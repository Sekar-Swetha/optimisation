\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C). Six categories of methods are investigated across six questions: (Q1)~adaptive step-size methods including the Polyak step, Adagrad, RMSprop, and Heavy Ball; (Q2)~momentum-based methods including Nesterov acceleration and Adam, together with mini-batch SGD; (Q3)~Newton's method with local quadratic approximations; (Q4)~derivative-free approaches including finite differences, Nesterov random search, Nelder--Mead simplex, and grid search; (Q5)~constrained optimisation via projected gradient descent and penalty methods; and (Q6)~the Frank--Wolfe conditional gradient algorithm for linear-constrained problems. Each method is evaluated with respect to convergence speed, trajectory behaviour, feasibility enforcement, stability, and computational trade-offs, with results interpreted in the context of first- and second-order optimisation theory.
\end{abstract}

\subsection{Benchmark A: Linear Regression Quadratic Loss}
\label{subsec:bench_A}

A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2),\quad y^{(i)} = {\theta^{\star}}^{\!\top} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^\top x^{(i)} - y^{(i)}\bigr)^2 = \frac{1}{2m}\norm{X\theta - y}^2.
\end{equation}
The Hessian is $H = \frac{1}{m}X^\top X$ (constant, positive definite), making $J$ a strictly convex quadratic. The gradient is $\grad J(\theta) = \frac{1}{m}X^\top(X\theta - y)$. The OLS minimiser is $\theta_{\text{OLS}} = (X^\top X)^{-1}X^\top y \approx (3.028,\, 4.017)^\top$. Due to observation noise, the minimum loss is $J^{\star} = J(\theta_{\text{OLS}}) \approx 0.4826$, not zero. Starting point: $\theta_0 = (0, 0)^\top$.

\begin{remark}
For a strongly convex quadratic with Hessian $H$, the condition number $\kappa(H) = \lambda_{\max}/\lambda_{\min}$ governs GD convergence: the per-iteration contraction is $(1 - 2\mu/(L + \mu))$ where $\mu = \lambda_{\min}$ and $L = \lambda_{\max}$. For Benchmark~A (with \texttt{np.random.seed(42)}, $m=1000$, $d=2$), the eigenvalues of $X^\top X/m$ are $\lambda_{\min} \approx 0.925$ and $\lambda_{\max} \approx 1.033$, giving $\kappa \approx 1.12$. This extremely low condition number means GD converges very rapidly on Benchmark~A, explaining why almost all first-order methods reach the noise floor $J^{\star} \approx 0.4826$ within 40--80 iterations. Note: the safe GD step is $1/L \approx 0.968$; the conservative choice $\alpha = 0.08 \approx 1/12L$ is used for comparability with other benchmarks.
\end{remark}

\subsection{Benchmark B: Toy Neural Network Loss}
\label{subsec:bench_B}

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\grad f = \bigl[2(x_1 - 1) + \cos(x_1),\; 10(x_2 - 2)\bigr]^\top$ and the Hessian is
\begin{equation}
H(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
At the minimum $(x_1^{\star} \approx 0.5824,\, x_2^{\star} = 2)$, the eigenvalues of $H$ are $\lambda_1 = 2 - \sin(0.5824) \approx 1.450$ and $\lambda_2 = 10$, giving a condition number $\kappa \approx 6.9$. This moderate anisotropy means gradient descent converges noticeably faster in the $x_2$ direction than in $x_1$. The minimum value is $f^{\star} \approx 0.7244$. Starting point: $x_0 = (-1, 4)^\top$.

\subsection{Benchmark C: Rosenbrock Function}
\label{subsec:bench_C}

\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum has eigenvalues $\lambda_1 \approx 0.399$ and $\lambda_2 \approx 1003$, giving a condition number $\kappa \approx 2508$. This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions. Starting point: $x_0 = (-1, 1)^\top$.

\begin{remark}[Non-Convexity Structure of Rosenbrock]
The Rosenbrock Hessian is $H(x) = \begin{pmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{pmatrix}$. The minimum eigenvalue is negative (Hessian indefinite, function non-convex) in the region where $x_2 > x_1^2 + c$ for a small constant $c > 0$. On the valley $x_2 = x_1^2$ itself, $H_{11} = 2 + 800x_1^2 \geq 2$ always, so the Hessian is \emph{positive definite everywhere on the valley}. The starting point $x_0 = (-1, 1)$ lies exactly on the valley boundary ($x_2 = x_1^2 = 1$). As gradient methods descend from $x_0$, they quickly enter the convex valley region and remain in it; the non-convexity is above the valley and does not trap gradient methods in this experiment. However, the extreme ill-conditioning ($\kappa \approx 2508$) along the valley --- not the non-convexity --- is the primary source of slow convergence.
\end{remark}

\subsection{Algorithm Formulations}
\label{subsec:q1_algs}

Four adaptive methods are compared against a constant step-size gradient descent baseline (120 iterations, all three benchmarks). All algorithms minimise $f \colon \R^d \to \R$ using only first-order (gradient) information; they differ in how they compute the effective per-iteration step size.

\begin{algorithm}[H]
\caption{Gradient Descent (Baseline)}
\label{alg:gd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, constant step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\, \grad f(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[GD Convergence]
If $f$ is $\mu$-strongly convex and $L$-smooth (i.e.\ $\norm{\grad f(x) - \grad f(y)} \leq L\norm{x-y}$), then GD with $\alpha \leq 1/L$ satisfies
\begin{equation}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k \!\bigl(f(x_0) - f^{\star}\bigr).
\end{equation}
The convergence factor $\rho = 1 - \mu/L = (\kappa - 1)/\kappa$ approaches 1 as $\kappa = L/\mu \to \infty$, explaining GD's slow progress on ill-conditioned problems such as Rosenbrock ($\kappa \approx 2508$).
\end{theorem}

\begin{theorem}[First-Order Oracle Lower Bound (Nesterov 1983)]
\label{thm:first_order_lb}
For any first-order algorithm that generates iterates $x_1, \ldots, x_k$ from gradients $\grad f(x_0), \ldots, \grad f(x_{k-1})$, there exists an $L$-smooth convex function $f$ such that
\begin{equation}
f(x_k) - f^{\star} \geq \frac{3L\|x_0 - x^{\star}\|^2}{32(k+1)^2}.
\end{equation}
This $\Omega(1/k^2)$ lower bound matches Nesterov's upper bound (Theorem~\ref{thm:nesterov_rate}), proving that the $O(1/k^2)$ rate is \emph{optimal} for first-order methods. No gradient-based algorithm can exceed this convergence rate on $L$-smooth convex objectives without additional structural assumptions. This is the reason why the Nesterov acceleration achieves an ``optimal'' rate: it cannot be improved upon within the first-order oracle model. The lower bound is achieved by the \emph{worst-case} function $f(x) = \frac{L}{4}\left[\frac{1}{2}\sum_{i=1}^{k}(x_i-x_{i+1})^2 + x_1^2 - x^{\star}_{k}x_k\right]$ (Chebyshev-like hard instance). For strongly convex functions, the analogous lower bound is $\Omega((1-1/\sqrt{\kappa})^k)$, matching Nesterov's linear rate.
\end{theorem}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \grad f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\norm{g_k}^2 + \varepsilon}$ \Comment{Adaptive, data-driven step}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step is derived by minimising the step that satisfies $f(x_{k+1}) \leq f^{\star}$ along the gradient direction. For a subgradient $g_k \in \partial f(x_k)$, the step $x_{k+1} = x_k - \alpha_k g_k$ minimises the upper bound on $\|x_{k+1} - x^{\star}\|^2$: expanding gives
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 = \|x_k - x^{\star}\|^2 - 2\alpha_k g_k^\top(x_k - x^{\star}) + \alpha_k^2\|g_k\|^2
\leq \|x_k - x^{\star}\|^2 - 2\alpha_k(f(x_k) - f^{\star}) + \alpha_k^2\|g_k\|^2,
\end{equation}
using the subgradient inequality $g_k^\top(x_k - x^{\star}) \geq f(x_k) - f^{\star}$. Minimising over $\alpha_k$ yields $\alpha_k^{\star} = (f(x_k) - f^{\star}) / \|g_k\|^2$. When $f^{\star}$ is known exactly and $f$ is convex, this yields $O(1/\sqrt{k})$ convergence for subgradient methods: $\min_{k=0}^{T} f(x_k) - f^{\star} \leq \|x_0 - x^{\star}\| \cdot (\max_k \|g_k\|) / \sqrt{T}$. \emph{Crucially, if $f^{\star}$ is misspecified}, the step size $\alpha_k = (f(x_k) - f^{\star}_{\text{wrong}})/(\norm{g_k}^2 + \varepsilon)$ diverges as $\norm{g_k} \to 0$ while the numerator remains nonzero, inducing instability.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \R^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \grad f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative sum of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad uses a \emph{monotonically decreasing} per-coordinate step size. Coordinates with historically large gradients receive smaller steps, automatically normalising for features with different scales. The effective step is $(\alpha_0 / (\sqrt{\sum_{j=0}^{k} g_{j,i}^2} + \varepsilon))$ per coordinate $i$, which decays as $O(1/\sqrt{k})$. This can cause \emph{premature stagnation} on problems where gradients remain large throughout, since the denominator grows without bound.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \grad f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's full history with an exponential moving average (EMA) of squared gradients. The forgetting factor $\beta$ controls the effective window: only the most recent $\sim 1/(1-\beta)$ gradients have significant influence. This prevents the step from decaying to zero on stationary problems, making RMSprop better suited to non-stationary objectives such as neural network training.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \grad f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method adds a \emph{momentum} term $\beta z_k$ to the gradient update, which accumulates gradient contributions from previous iterations. Physically, this corresponds to a ball rolling down a loss surface with friction $\beta$. For strongly convex quadratics, the optimal parameters $(\alpha, \beta)$ achieve the \emph{theoretically optimal} linear convergence rate with contraction factor $((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$, dramatically outperforming plain GD on ill-conditioned problems. However, Heavy Ball can diverge on non-quadratic functions and lacks Nesterov's theoretical guarantees.

\begin{figure}[H]

\begin{itemize}[nosep,leftmargin=*]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.024$, reaching $f < 1$ at iteration~5). Without $f^{\star}$, Nesterov (Q2) and Heavy~Ball are best on ill-conditioned problems.
\item \textbf{Heavy Ball on quadratics:} For a strongly convex quadratic with $\mu = \lambda_{\min}$ and $L = \lambda_{\max}$, the theoretically optimal Heavy~Ball parameters are $\alpha^{\star} = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2}$ and $\beta^{\star} = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$, giving contraction factor $\rho^{\star} = \beta^{\star} = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2 \ll (1-1/\kappa)^2$. For Benchmark~A ($\kappa \approx 1.12$), $\sqrt{\kappa} \approx 1.058$, so $\beta^{\star} = (0.058/2.058)^2 \approx 0.00079$, giving $\rho^{\star} \approx 0.00079$ vs.\ GD's $\rho \approx 1 - 1/1.12 \approx 0.107$ with optimal step. Since Benchmark~A is nearly spherical ($\kappa \approx 1$), both GD and Heavy~Ball converge very rapidly and the advantage of Heavy~Ball is minimal. For Benchmark~C ($\kappa \approx 2508$), $\sqrt{\kappa} \approx 50.1$, so $\beta^{\star} \approx (49.1/51.1)^2 \approx 0.923$, giving $\rho^{\star} \approx 0.923$ vs.\ GD's $\rho \approx 1 - 1/2508 \approx 0.9996$. The gap is dramatic for large $\kappa$: Heavy~Ball would require $\sim\!\!2508$ GD iterations to match the accuracy of $\sim\!\!13$ optimal Heavy~Ball iterations on Rosenbrock.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to the EMA forgetting factor. Heavy~Ball shows characteristic overshoot followed by damped oscillation.
\item \textbf{Stability:} GD is most stable (monotone decrease for correct $\alpha \leq 1/L$). Polyak is unstable when $f^{\star}$ is misspecified. Adagrad is uniformly stable but subject to step decay; RMSprop avoids this through forgetting.
\item \textbf{Anisotropy handling:} Adagrad and RMSprop adapt per-coordinate step sizes, reducing sensitivity to $\kappa$. Heavy~Ball accumulates momentum in the persistent descent direction along the valley floor.
\item \textbf{Practical recommendation:} Since exact $f^{\star}$ is rarely known in practice, RMSprop (stable, adaptive) and Nesterov (optimal first-order rate) are the most robust choices. Adagrad is preferred for sparse gradient problems (e.g.\ NLP embeddings).
\end{itemize}

\subsection{Algorithm Formulations}
\label{subsec:q2_algs}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \grad f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Optimal Rate]
\label{thm:nesterov_rate}
For $L$-smooth convex $f$, Nesterov's accelerated gradient method satisfies
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\end{equation}
This $O(1/k^2)$ rate is \emph{optimal} for first-order methods on smooth convex functions (Nesterov, 1983), compared to the $O(1/k)$ rate of plain gradient descent. The adaptive momentum schedule $\beta_k = (k-1)/(k+2)$ is precisely the sequence required to achieve this optimal rate. Combined with the lower bound in Theorem~\ref{thm:first_order_lb}, this confirms that no first-order algorithm can do better than $O(1/k^2)$ on this function class.
\end{theorem}

The key distinction from Heavy~Ball is the \emph{lookahead}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This ``look before you leap'' correction allows Nesterov to correct its trajectory before overshooting. The momentum is gradually increased by the $(k-1)/(k+2)$ schedule, starting near zero and approaching $\beta_{\max}$.

\begin{remark}[Nesterov vs.\ Heavy Ball: Theoretical Guarantees]
Both methods accelerate first-order convergence, but their theoretical guarantees differ:
\begin{itemize}[nosep,leftmargin=*]
\item \textbf{Nesterov} achieves the \emph{information-theoretic optimal} $O(1/k^2)$ rate for \emph{any} smooth convex $f$, and $O((1-1/\sqrt{\kappa})^k)$ linear rate for $\mu$-strongly convex $f$. These guarantees hold globally (not just near the optimum).
\item \textbf{Heavy Ball} achieves the optimal $O(((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^{2k})$ rate for \emph{strongly convex quadratics} with optimal parameters. However, for non-quadratic objectives, Heavy Ball can \emph{diverge} or fail to converge (Lessard et al., 2016 showed counterexamples). Its convergence for non-quadratic strongly convex objectives is not guaranteed with fixed parameters.
\end{itemize}
In practice, Nesterov is the safer choice for general smooth functions (it always converges with $\alpha \leq 1/L$), while Heavy Ball with optimal parameters outperforms Nesterov on quadratics --- as seen in Q1 (Benchmark~C: Heavy~Ball $f = 1.40$ vs.\ Nesterov $f = 0.48$, but using suboptimal parameters for robustness). The theoretical $\sqrt{\kappa}$ speedup of Nesterov over GD is $\sqrt{2508} \approx 50$-fold, confirmed empirically by their rate ratio in Figure~\ref{fig:conv_rate_theory}.
\end{remark}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \grad f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{1st moment (EMA of gradients)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{2nd moment (EMA of squared gradients)}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$;\quad $\hat{v}_t \gets v_t / (1 - \beta_2^t)$ \Comment{Bias-corrected estimates}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum (via the first-moment EMA $m_t \approx \mathbb{E}[g_t]$) with per-coordinate adaptive scaling (via $v_t \approx \mathbb{E}[g_t^2]$). The bias-correction terms $(1-\beta_1^t)$ and $(1-\beta_2^t)$ prevent the initial iterates from being biased towards zero because $m_0 = v_0 = 0$. The effective per-coordinate step is $\alpha\,|\hat{m}_t|/(\sqrt{\hat{v}_t}+\varepsilon)$, which approximates $\alpha\,\text{sign}(g_t)$ when the EMA windows are well-estimated. Adam inherits RMSprop's adaptive scaling and Heavy~Ball's momentum.

\begin{remark}[Why Adam Underperforms Nesterov on Deterministic Benchmarks]
Despite Adam's popularity in deep learning, it exhibits suboptimal performance compared to Nesterov on the deterministic benchmarks in this study. The key reasons are:
\begin{itemize}[nosep,leftmargin=*]
\item \textbf{Adaptive scaling hurts symmetry:} Adam's per-coordinate normalisation by $\sqrt{\hat{v}_t}$ effectively distorts the loss landscape, destroying the favourable symmetry that Nesterov exploits. On Rosenbrock, where the valley has a specific curved structure, this normalisation can misalign the step direction.
\item \textbf{Fixed step in sparse direction:} The bias-corrected $\hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$ normalises away the gradient magnitude, making Adam's effective step size $\approx \alpha$ in each dimension regardless of curvature. Nesterov preserves gradient magnitude, allowing it to take larger steps along the valley where curvature is shallow.
\item \textbf{Convergence guarantees:} Nesterov has proven $O(1/k^2)$ convergence for convex functions (and $O((1-1/\sqrt{\kappa})^k)$ for strongly convex). Adam's convergence guarantees are weaker: the original Adam paper (Kingma \& Ba, 2015) does not prove optimal rates, and Reddi et al.\ (2018) showed Adam can fail to converge even on simple convex problems without modifications (AMSGrad). Adam's advantage is primarily in \emph{stochastic} settings where gradient variance is high.
\end{itemize}
\end{remark}

\begin{remark}[Bias--Variance Trade-off in Mini-Batch SGD]
Let $\hat{g}_{\mathcal{B}} = \frac{1}{b}\sum_{i \in \mathcal{B}} \grad_\theta \ell(\theta; x^{(i)}, y^{(i)})$ be the mini-batch gradient. Then:
\begin{equation}
\mathbb{E}[\hat{g}_{\mathcal{B}}] = \grad J(\theta) \quad \text{(unbiased)}, \qquad
\mathrm{Var}[\hat{g}_{\mathcal{B}}] = \frac{\sigma_g^2}{b},
\end{equation}
where $\sigma_g^2 = \mathrm{Var}[\grad_\theta \ell(\theta; x, y)]$ is the per-sample gradient variance. The noise floor of SGD is approximately $\sigma_g^2 / b$ (for fixed $\alpha$), confirming that doubling the batch size halves the variance --- at the cost of halving the number of gradient steps per epoch.
\end{remark}

\subsection{Learning Rate Schedules for SGD}
\label{subsec:q2_schedules}

A key limitation of SGD with a fixed learning rate is the persistent noise floor: for any $\alpha > 0$, the iterate oscillates around the optimum with an error of order $O(\alpha \sigma_g^2 / b)$. A \emph{decaying} learning rate schedule $\{\alpha_t\}$ resolves this trade-off: large initial steps allow rapid progress while smaller later steps reduce oscillation near the minimum. Figure~\ref{fig:q2_lr_schedules} compares four schedules on Benchmark~A.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q2_lr_schedules.pdf}
\caption{Q2: Learning rate schedule comparison on Benchmark~A ($b=10$, 60~epochs). \emph{Left}: suboptimality $J(\theta_k) - J^{\star}$ on a log scale. The large fixed rate ($\alpha=0.05$) converges quickly but plateaus above the optimal noise floor due to persistent SGD oscillations ($\Delta J \approx 0.0005$ above $J^{\star}$). The small fixed rate ($\alpha=0.005$) converges accurately but slowly. Both decay schedules --- $1/\sqrt{t}$ and cosine annealing --- converge faster and more accurately than either fixed rate, demonstrating the advantage of adaptive scheduling. \emph{Right}: the effective learning rate over gradient steps, showing the characteristic warm-up (implicit in cosine) and decay patterns.}
\label{fig:q2_lr_schedules}
\end{figure}

\begin{remark}[Convergence Guarantees for Decaying SGD]
For $\mu$-strongly convex $f$ with $L$-smooth components and mini-batch gradients with variance $\sigma^2$, SGD with $\alpha_t = \alpha_0 / t$ achieves:
\begin{equation}
\mathbb{E}[J(\theta_T)] - J^{\star} \leq \frac{L\,\alpha_0^2\,\sigma^2}{2\mu\,(\alpha_0\mu - 1)\,T},
\end{equation}
(for $\alpha_0 > 1/\mu$), giving an $O(1/T)$ rate. In contrast, fixed-$\alpha$ SGD converges to a noise floor of $O(\alpha\sigma^2/b)$. \emph{Cosine annealing} $\alpha_t = (\alpha_0/2)(1 + \cos(\pi t / T_{\max}))$ is widely used in deep learning for its smooth warmup-then-decay profile: it allows exploration early in training (high $\alpha$) and refinement late in training (low $\alpha$). Unlike polynomial decay, cosine annealing automatically matches the decay length to the total budget $T_{\max}$.
\end{remark}

\begin{remark}[Linear Warmup for Learning Rate Schedules]
Cosine annealing and polynomial decay schedules share a common practical limitation: they may start with a large learning rate $\alpha_0$ that causes instability in early training, especially for deep networks where the initial gradient norms are large and vary drastically across layers. \emph{Linear warmup} addresses this by gradually increasing $\alpha$ from 0 to $\alpha_0$ over the first $T_{\text{warm}}$ steps:
\begin{equation}
\alpha_t = \frac{t}{T_{\text{warm}}}\,\alpha_0 \quad (t \leq T_{\text{warm}}), \qquad
\alpha_t = \text{cosine/poly}(\alpha_0, t - T_{\text{warm}}) \quad (t > T_{\text{warm}}).
\end{equation}
Warmup is standard practice in Transformer training (Vaswani et al., 2017) and reduces the probability of divergence during the first steps when Adam's bias-correction terms $1/(1-\beta_1^t)$ are far from 1. For the benchmarks in this study (small $d=2$, analytic objectives, well-scaled gradients), warmup provides no benefit and the standard decaying schedules are sufficient.
\end{remark}

\begin{remark}[Gradient Clipping for Stability]
In regions with large gradients (e.g.\ near the start $(-1, 4)$ for Benchmark~B), the gradient magnitude can be very large, causing instability without a conservative step size. \emph{Gradient clipping} modifies SGD updates as:
\begin{equation}
g_t^{\text{clip}} = \begin{cases} g_t & \text{if } \|g_t\| \leq \tau \\ \tau\, g_t/\|g_t\| & \text{otherwise,}\end{cases}
\end{equation}
where $\tau > 0$ is the clipping threshold. This is standard practice in recurrent neural network (RNN) training where exploding gradients are a known issue. The clipped update is always bounded: $\|g_t^{\text{clip}}\| \leq \tau$, guaranteeing that the step size is at most $\alpha \tau$ regardless of the gradient magnitude. For the deterministic benchmarks in this study, where the Hessian and gradient magnitudes are bounded, gradient clipping is unnecessary; its utility emerges when gradient magnitudes vary drastically across different parameter coordinates or training samples.
\end{remark}

\begin{remark}[Implicit Regularisation of SGD]
An important but often overlooked property of SGD is its role as an \emph{implicit regulariser}. For overparameterised models (where the number of parameters $d \gg m$ samples), there are infinitely many solutions with zero training loss. Among these, GD converges to the \emph{minimum-norm} solution $\theta^{\star} = \arg\min_{\theta: J(\theta)=0} \|\theta\|$, while SGD with small batches tends to find solutions with better \emph{generalisation} (lower test error). Intuitively, the gradient noise in SGD acts as a random perturbation that steers the iterate away from sharp minima (high Hessian trace) and towards flat minima, which are known to generalise better (Keskar et al., 2016). For Benchmark~A (a well-specified, underparameterised problem), this effect is absent: all methods converge to the unique OLS minimiser and the implicit regularisation of SGD provides no benefit.
\end{remark}

\begin{remark}[Variance Reduction: SVRG and SAGA]
A fundamental limitation of standard mini-batch SGD is that its convergence rate degrades with gradient variance: even for strongly convex $f$, the best achievable rate with fixed step size is a noise floor rather than linear convergence to $x^{\star}$. \emph{Variance reduction} methods eliminate this noise floor by periodically anchoring the stochastic gradient to a full gradient snapshot. Two leading methods are:
\begin{itemize}[nosep,leftmargin=*]
\item \textbf{SVRG} (Johnson \& Zhang, 2013): every $s$ steps, compute a full gradient $\tilde{\nabla} = \nabla f(\tilde{x})$ at a snapshot $\tilde{x}$. The stochastic update uses a control variate:
\begin{equation}
g_k = \nabla f_{i_k}(x_k) - \nabla f_{i_k}(\tilde{x}) + \tilde{\nabla},
\end{equation}
which is unbiased ($\mathbb{E}[g_k] = \nabla f(x_k)$) but has variance $O(\|x_k - \tilde{x}\|^2)$, shrinking to zero as $x_k \to x^{\star}$. SVRG achieves \emph{linear} convergence $O((1-\mu/(2L))^k)$ with $O(m)$ gradient evaluations per effective pass --- matching full GD rate.
\item \textbf{SAGA} (Defazio et al., 2014): stores the last gradient $\nabla f_i(\phi_i)$ for every data point $i$, and updates only the sampled component at each step. The update $g_k = \nabla f_{i_k}(x_k) - \nabla f_{i_k}(\phi_{i_k}) + \frac{1}{m}\sum_j \nabla f_j(\phi_j)$ similarly achieves linear convergence without periodic snapshot recomputation, at the cost of $O(md)$ storage.
\end{itemize}
Figure~\ref{fig:q2_svrg} illustrates this empirically and theoretically: SVRG achieves linear convergence matching full GD (per epoch), whereas fixed-step SGD plateaus at a noise floor. For Benchmark~A with $m = 1000$ samples, SVRG/SAGA would converge at the full GD linear rate $O((1-\kappa^{-1})^k) \approx O((1-1/1.12)^k)$ while performing only $O(b)$ gradient computations per step. Their practical disadvantage is implementation complexity and, for SAGA, the $O(md)$ memory requirement.
\end{remark}

\subsection{Local Approximations of $g(x) = x^4$}
\label{subsec:q3_approx}

At $x_0 = 0.25$: $g(x_0) = 0.0039$, $g'(x_0) = 4 \cdot 0.25^3 = 0.0625$, $g''(x_0) = 12 \cdot 0.25^2 = 0.75$.

\textbf{First-order (tangent line) approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625\,(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}\,g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\end{equation}

The Newton step minimises $\hat{g}_2$: setting $\frac{d\hat{g}_2}{dx} = 0$ gives $x_{\text{Newton}} = x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 = 0.25 - 0.0833 = 0.167$. The true minimum of $g(x) = x^4$ is at $x=0$; Newton's step moves in the correct direction but overshoots for non-quadratic functions.

\subsection{Newton's Method with Damping}
\label{subsec:q3_newton}

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \grad f(x_k)$
    \State $H_k \gets \Hess f(x_k) + \lambda I$ \Comment{Regularised Hessian (ensures positive definiteness)}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: solve linear system}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Newton Quadratic Convergence]
Suppose $f$ is twice continuously differentiable with $L$-Lipschitz Hessian and $\mu$-strongly convex. Then in a neighbourhood of $x^{\star}$, Newton's method (with $\alpha = 1$, $\lambda = 0$) satisfies
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{L}{2\mu}\,\|x_k - x^{\star}\|^2.
\end{equation}
This \emph{quadratic} (superlinear) convergence means the number of correct decimal digits approximately doubles per iteration. In contrast, GD satisfies $\|x_{k+1} - x^{\star}\| \leq (1 - \mu/L)\|x_k - x^{\star}\|$ (linear convergence). Newton's method thus requires $O(\log\log(1/\varepsilon))$ iterations vs.\ $O(\log(1/\varepsilon))$ for GD to achieve $\varepsilon$ accuracy.
\end{theorem}

Damping ($\lambda = 10^{-8}$ in all experiments) prevents numerical issues when the Hessian is nearly singular. The step-size factor $\alpha < 1$ (used for Benchmarks~B and~C) is a \emph{damped Newton step}, required when the function is not well-approximated by its local quadratic model far from the optimum.

\begin{remark}[Armijo Line Search for Newton's Method]
An \emph{Armijo backtracking line search} selects the step size $\alpha_k$ automatically, avoiding the need to hand-tune a damping factor. Starting from $\alpha = 1$, the step is halved until the \emph{sufficient decrease condition} is satisfied:
\begin{equation}
f(x_k - \alpha_k p_k) \leq f(x_k) - c\,\alpha_k\, \grad f(x_k)^\top p_k,
\end{equation}
where $c \in (0, 1)$ (typically $c = 10^{-4}$) and $p_k = H_k^{-1}\grad f(x_k)$ is the Newton direction. Since the Newton direction satisfies $\grad f(x_k)^\top p_k = \grad f(x_k)^\top H_k^{-1}\grad f(x_k) > 0$ (when $H_k \succ 0$), the Armijo condition is always achievable for small enough $\alpha_k$. Combined with a Wolfe condition on the gradient, this gives \emph{damped Newton with line search}, which retains the quadratic convergence rate of pure Newton's method while guaranteeing global convergence from any starting point. For Benchmark~C, line search would automatically find the aggressive-but-stable step size along the high-curvature Rosenbrock valley.
\end{remark}

\begin{remark}[Why Newton Converges in One Step for Quadratics]
For a strictly convex quadratic $J(\theta) = \frac{1}{2}\theta^\top H\theta - b^\top\theta + c$ with constant Hessian $H \succ 0$, the Newton step from any $\theta_0$ is:
\begin{equation}
\theta_1 = \theta_0 - H^{-1}\grad J(\theta_0) = \theta_0 - H^{-1}(H\theta_0 - b) = H^{-1}b = \theta^{\star}.
\end{equation}
This is the exact minimiser regardless of starting point --- Newton's method is a one-shot solver for quadratic objectives. For Benchmark~A, $J(\theta) = \frac{1}{2m}\|X\theta - y\|^2$ is quadratic with $H = X^\top X/m$ and $b = X^\top y/m$, so $\theta^{\star} = (X^\top X)^{-1}X^\top y$ (OLS solution). With $\alpha = 1$ and $\lambda = 10^{-8} \approx 0$, Newton reaches $\theta^{\star}$ in exactly one iteration.
\end{remark}

\begin{remark}[Conjugate Gradient as Optimal Krylov Method for Quadratics]
A class of methods that bridges first-order and second-order approaches is the \emph{Conjugate Gradient} (CG) method. For a strictly convex quadratic $\min_\theta \frac{1}{2}\theta^\top H\theta - b^\top\theta$, CG generates a sequence of \emph{$H$-conjugate} directions $\{p_0, p_1, \ldots\}$ satisfying $p_i^\top H p_j = 0$ for $i \neq j$, and performs an exact line search along each. The resulting algorithm:
\begin{equation}
\theta_{k+1} = \theta_k + \alpha_k p_k, \quad \alpha_k = -\frac{r_k^\top p_k}{p_k^\top H p_k}, \quad p_{k+1} = r_{k+1} + \beta_k p_k, \quad \beta_k = \frac{r_{k+1}^\top r_{k+1}}{r_k^\top r_k},
\end{equation}
where $r_k = H\theta_k - b$ is the residual. CG converges in at most $d = 2$ steps for our 2D problems, and in at most $r \leq d$ steps in general, where $r$ is the number of distinct eigenvalues of $H$. When $H$ has only $s \ll d$ distinct eigenvalues, CG converges in $s$ steps, making it exponentially faster than GD for structured matrices. For the non-quadratic benchmarks B and C, the \emph{nonlinear} CG generalisation (Fletcher--Reeves or Polak--Ribi\`ere) is applied iteratively with restarts, achieving super-linear convergence in practice. CG requires only $O(d)$ memory (no Hessian storage) and $O(md)$ per step, making it the method of choice for large-scale quadratic problems (e.g.\ solving normal equations in linear regression with $d \gg 10^5$).
\end{remark}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark~B)}
\label{subsec:q4_partA}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta\, e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector and $\delta > 0$ is the perturbation size. The approximation error is the sum of a \emph{truncation error} $O(\delta)$ (from the Taylor expansion remainder) and a \emph{round-off error} $O(\varepsilon_{\text{mach}}/\delta)$ (from floating-point cancellation). The optimal balance gives $\delta^{\star} \sim \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ in double precision. The experiments test $\delta = 0.05$ (a good practical choice, yielding $O(10^{-2})$ truncation error) and $\delta = 0.8$ (a poor choice where the gradient is averaged over a region much wider than the local curvature scale, producing inaccurate directions).

Each gradient estimate requires $d+1 = 3$ function evaluations (one at $x$, one per coordinate), compared to one evaluation for an exact gradient oracle. For large $d$, finite differences become expensive: $O(d)$ evaluations per gradient, motivating randomised methods.

\begin{remark}[Optimal Finite Difference Step Size]
The forward-difference error decomposes as:
\begin{equation}
\left|\frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| \leq \underbrace{\frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(\xi)\right|}_{\text{truncation error}} + \underbrace{\frac{2\varepsilon_{\text{mach}}\|f\|}{\delta}}_{\text{round-off error}},
\end{equation}
where $\xi$ is some point between $x$ and $x + \delta e_i$. Minimising the sum over $\delta$ gives the optimal perturbation:
\begin{equation}
\delta^{\star} = \sqrt{\frac{2\varepsilon_{\text{mach}}\|f\|}{|f''|}} \approx \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}.
\end{equation}
The assignment's ``good'' choice $\delta = 0.05$ gives truncation error $\sim 0.025|f''|$, which is acceptable for Benchmark~B's moderate curvature. The ``poor'' choice $\delta = 0.8$ gives truncation error $\sim 0.4|f''|$, which can exceed the gradient magnitude itself and produces misleading descent directions.
\end{remark}

Forward differences (blue) follow the $O(\delta)$ reference (dotted) for small $\delta$ before round-off dominates at $\delta < 10^{-8}$; the two gradient components have different errors because the second partial ($\partial_2 f = 10(x_2-2)$) is larger and more sensitive to perturbation. \begin{remark}[Central Differences: $O(\delta^2)$ Accuracy]
The \emph{central difference} approximation uses two symmetric evaluations:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta},
\end{equation}
with truncation error $O(\delta^2)$ (vs.\ $O(\delta)$ for forward differences). This requires $2d$ evaluations instead of $d+1$, but achieves much higher accuracy for the same $\delta$: with $\delta = 0.05$, central differences give truncation error $\sim 0.00125|f'''|$, roughly 20-fold better than forward differences ($\sim 0.025|f''|$). In practice, central differences are preferred for Newton's method implementations where high gradient accuracy is needed to compute accurate Newton directions.
\end{remark}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Stochastic directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates a directional derivative along a random unit vector $u_k$. Only \emph{two} function evaluations are required per iteration regardless of dimension $d$, making it the most scalable method in this comparison. The estimator satisfies $\mathbb{E}[\hat{g}_k] = \frac{1}{d}\grad f(x_k)$ for smooth $f$, so on average it takes a step proportional to the gradient, albeit with high variance (estimating one of $d$ directions randomly). This variance scales as $O(1/d)$ in the directional accuracy, explaining slow convergence for $d \gg 1$.

\begin{theorem}[Nesterov Random Search Convergence]
Let $f$ be $L$-smooth and convex. Nesterov's random search with step $\alpha$ and perturbation $\delta \to 0$ satisfies, in expectation over the random directions $\{u_k\}$:
\begin{equation}
\mathbb{E}[f(x_k)] - f^{\star} \leq \frac{d\,\|x_0 - x^{\star}\|^2}{2\alpha\,k}.
\end{equation}
This is an $O(d/k)$ rate --- slower than GD's $O(1/k)$ by a factor of $d$ --- because the random direction estimator is effectively equivalent to a gradient with variance inflated by a factor of $d$. For $d = 2$ (as in our experiments), the effective slowdown is only $2\times$ per iteration vs.\ exact GD; for $d = 10^4$ (typical high-dimensional problems), random search would require $10^4 \times$ more iterations to achieve the same accuracy.
\end{theorem}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\R^2$ and iteratively transforms it using four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): reflect the worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2.0$): if reflection improved, try moving further in the same direction.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection worsened, contract the worst vertex towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, shrink all vertices towards the best.
\end{enumerate}
No gradient information is required. The simplex adapts its shape to the local landscape: it elongates in productive descent directions and contracts when near a minimum.

\begin{remark}[Nelder--Mead Convergence Conditions]
Despite its widespread practical use, Nelder--Mead has no general convergence guarantee for $d \geq 2$. McKinnon (1998) constructed a counterexample where Nelder--Mead converges to a non-stationary point in $d = 2$. However, convergence can be established under specific conditions: if the simplex never becomes degenerate (all vertices remain non-coplanar) and the objective is strictly convex with bounded level sets, the algorithm generates a sequence with $f(x_k) \to f^{\star}$ (Lagarias et al., 1998). In practice, the algorithm is augmented with \emph{restarts} (periodically reinitialising the simplex near the current best point) to recover from degeneracy. Its empirical success on smooth, unimodal 2D problems --- as seen in this study --- contrasts with its theoretical limitations in higher dimensions and on multimodal functions.
\end{remark}

\begin{remark}[Regret Bound for Adagrad]
Adagrad's adaptive step sizes can be analysed through the lens of online convex optimisation. For a sequence of convex losses $f_1, \ldots, f_T$ with subgradients $g_t = \partial f_t(x_t)$, Adagrad achieves the regret bound:
\begin{equation}
\sum_{t=1}^{T} f_t(x_t) - \min_{x \in \mathcal{X}} \sum_{t=1}^{T} f_t(x) \leq 2\alpha_0 \sum_{i=1}^{d} \left(\sum_{t=1}^{T} g_{t,i}^2\right)^{1/2}.
\end{equation}
This bound adapts to the problem: if only a few coordinates $i$ have large cumulative gradients, those coordinates have small effective step sizes, improving over the coordinate-uniform bound. For sparse gradient problems (e.g.\ NLP embeddings where most coordinates are zero most of the time), Adagrad's regret can be $O(\sqrt{T})$ with a much smaller constant than non-adaptive methods.
\end{remark}

\begin{remark}[Coordinate Descent as an Alternative Derivative-Free Method]
An important class of gradient-free (or gradient-aware) methods not covered in the assignment is \emph{coordinate descent} (CD). At each step, CD minimises $f$ exactly along a single coordinate direction $e_i$ (cycling or randomised):
\begin{equation}
x_{k+1} = x_k - \frac{\partial f}{\partial x_i}(x_k)\, e_i \cdot \alpha_i^{\star}, \qquad \alpha_i^{\star} = \arg\min_\alpha\, f(x_k - \alpha e_i),
\end{equation}
requiring only a 1D line search per iteration rather than a full gradient. This is advantageous when partial derivatives are cheap or when $d$ is very large but the function has separable or sparse coupling structure. For \emph{separable} $f$ (i.e.\ $f(x) = \sum_i f_i(x_i)$), CD achieves exact convergence in one sweep. For general strongly convex $f$, randomised CD converges at rate $(1 - \mu/(Ld))^k$ --- slower than full GD by the dimension $d$ --- because each step updates only one of $d$ coordinates. This matches the $O(d/k)$ rate of Nesterov random search and highlights a general principle: \emph{per-coordinate methods exchange gradient computation for more iterations}. In contrast to Nelder--Mead, CD has theoretical convergence guarantees for smooth convex objectives.
\end{remark}

\subsection{Problem Setup}
\label{subsec:q5_setup}

Minimise Benchmark~B, $f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)$, subject to the inequality constraint $x_1 \geq 0.5$. The starting point $x_0 = (0.2, 4.0)^\top$ is \emph{infeasible} ($x_1 = 0.2 < 0.5$). The unconstrained minimum is at $x^{\star} \approx (0.582, 2.000)$, which satisfies $x_1^{\star} = 0.582 > 0.5$. Therefore, the constraint is \emph{inactive} at the optimum, and the constrained minimum coincides with the unconstrained minimum. The methods differ in how they handle the infeasible starting point and enforce feasibility during optimisation.

\begin{theorem}[Karush--Kuhn--Tucker Conditions]
\label{thm:kkt}
For the constrained problem $\min_{x \in \mathcal{X}} f(x)$ with $\mathcal{X} = \{x : g(x) \leq 0\}$ (here $g(x) = 0.5 - x_1$), if $x^{\star}$ is a local minimum and a constraint qualification holds (e.g.\ linear independence of active constraint gradients), then there exists a dual variable $\mu^{\star} \geq 0$ such that the KKT conditions hold:
\begin{align}
&\text{(Stationarity)}\quad \nabla f(x^{\star}) + \mu^{\star}\, \nabla g(x^{\star}) = 0, \label{eq:kkt_stat}\\
&\text{(Primal feasibility)}\quad g(x^{\star}) \leq 0, \label{eq:kkt_primal}\\
&\text{(Dual feasibility)}\quad \mu^{\star} \geq 0, \label{eq:kkt_dual}\\
&\text{(Complementary slackness)}\quad \mu^{\star}\, g(x^{\star}) = 0. \label{eq:kkt_cs}
\end{align}
For our problem, $g(x) = 0.5 - x_1$, so $\nabla g = (-1, 0)^\top$. Since the constraint is \emph{inactive} at $x^{\star}$ ($g(x^{\star}) = 0.5 - 0.582 < 0$), complementary slackness~\eqref{eq:kkt_cs} forces $\mu^{\star} = 0$. Stationarity~\eqref{eq:kkt_stat} then reduces to $\nabla f(x^{\star}) = 0$: the constrained minimum satisfies the same first-order condition as the unconstrained minimum. This is why all methods converge to the same point.
\end{theorem}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \grad f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection $\Pi_{\mathcal{X}}(z) = \max(0.5, z_1)$ is trivially $O(1)$. For general convex constraint sets, projection can be expensive (e.g.\ $O(d \log d)$ for simplex constraints), motivating penalty methods as an alternative.

\begin{theorem}[Projected Gradient Descent Convergence]
\label{thm:pgd_conv}
Let $f$ be $\mu$-strongly convex and $L$-smooth on $\mathcal{X}$, and let $\Pi_{\mathcal{X}}$ be the Euclidean projection onto the closed convex set $\mathcal{X}$. Then projected GD with step $\alpha \leq 1/L$ satisfies:
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 \leq (1 - \alpha\mu)^2\,\|x_k - x^{\star}\|^2,
\end{equation}
giving linear convergence with the same rate $(1-\mu/L)^k$ (for $\alpha = 1/L$) as unconstrained GD. The proof uses the \emph{non-expansiveness} of the projection: $\|\Pi_{\mathcal{X}}(u) - \Pi_{\mathcal{X}}(v)\| \leq \|u - v\|$ for all $u, v \in \R^d$, together with the descent lemma for $f$.
\end{theorem}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda\, \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $x_{k+1} \gets x_k - \alpha\, \grad F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty gradient is $\grad F_\lambda(x) = \grad f(x) + \lambda\, \partial_x[\max(0,\, 0.5 - x_1)]$, where $\partial_{x_1}[\max(0, 0.5 - x_1)] = -\mathbf{1}[x_1 < 0.5]$.

Settings: Projected GD ($\alpha = 0.08$); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$ to compensate for the stiffened objective); Unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\subsection{Comparative Discussion}
\label{subsec:q5_discuss}

\textbf{Projected GD} enforces feasibility at every iteration via the projection operator, making infeasibility impossible by construction. The projection cost is $O(1)$ for a half-space constraint ($\Pi_{\mathcal{X}} = \max(0.5, \cdot)$) but can be as expensive as $O(d \log d)$ or require a QP solve for complex constraint sets.

\textbf{Penalty methods} trade exact feasibility for implementation simplicity: only a modified gradient is needed, with no projection oracle required. The weight $\lambda$ controls the trade-off between constraint enforcement and objective optimisation. Since the constraint is inactive at the true minimum, all penalty methods converge to the same feasible point regardless of $\lambda$ (given sufficient iterations). The key differences are: (a) speed of leaving the infeasible region, and (b) the conditioning of the penalised objective (larger $\lambda$ increases the effective Lipschitz constant, requiring smaller step sizes to maintain stability).

Figure~\ref{fig:q5_penalty_bias} contrasts the inactive case (as in Q5) with the theoretically important active constraint case.

\textbf{When the constraint is active} at the constrained minimum (i.e.\ when the unconstrained minimum is outside $\mathcal{X}$), the penalty method with finite $\lambda$ converges to a \emph{biased} solution at distance $O(1/\lambda)$ from the true constrained minimum. To see this, note that the penalty minimiser $x^{\star}_\lambda$ satisfies $\grad f(x^{\star}_\lambda) + \lambda \partial_x P(x^{\star}_\lambda) = 0$, where $P(x) = \max(0, 0.5-x_1)$. At the true constrained minimum $x^{\star}_c$ (with active constraint), the KKT conditions give $\grad f(x^{\star}_c) + \mu^{\star} e_1 = 0$ for some dual variable $\mu^{\star} > 0$. The penalty method recovers this only in the limit $\lambda \to \infty$.

\begin{remark}[Newton's Method Applied to KKT System]
Newton's method can be applied directly to the KKT conditions (Theorem~\ref{thm:kkt}) to solve constrained problems. The KKT stationarity condition $\nabla f(x) + \mu \nabla g(x) = 0$ and complementary slackness $\mu g(x) = 0$ form a nonlinear system $F(x, \mu) = 0$, where:
\begin{equation}
F(x, \mu) = \begin{bmatrix} \nabla f(x) + \mu \nabla g(x) \\ \min(\mu, -g(x)) \end{bmatrix} = 0.
\end{equation}
Newton's method on $F$ gives the \emph{primal-dual interior point} update at each step:
\begin{equation}
\begin{bmatrix} \nabla^2 f + \mu \nabla^2 g & \nabla g \\ \nabla g^\top & 0 \end{bmatrix} \begin{bmatrix} \Delta x \\ \Delta \mu \end{bmatrix} = -\begin{bmatrix} \nabla f + \mu \nabla g \\ g \end{bmatrix}.
\end{equation}
This is a $(d+1) \times (d+1)$ linear system (here $d=2$, so $3 \times 3$) that yields the joint primal-dual Newton direction $(\Delta x, \Delta \mu)$. Applying Newton's method to the KKT system achieves quadratic convergence in both primal and dual variables simultaneously --- much faster than the first-order methods studied in Q5. For our problem with an inactive constraint ($\mu^{\star} = 0$), the KKT system reduces to $\nabla f(x^{\star}) = 0$, collapsing to unconstrained Newton's method and converging in $\sim$5 steps on Benchmark~B.
\end{remark}

\begin{remark}[Augmented Lagrangian for Active Constraints]
The \emph{augmented Lagrangian} (Multiplier Method) avoids the penalty bias by combining a penalty term with an explicit dual variable update:
\begin{equation}
\mathcal{L}_\lambda(x, \mu) = f(x) + \mu^\top c(x) + \frac{\lambda}{2}\|c(x)\|^2,
\end{equation}
where $c(x) = \max(0, 0.5 - x_1)$ is the constraint violation. The dual variable $\mu$ is updated as $\mu_{k+1} = \mu_k + \lambda\, c(x^{\star}_k)$ after each inner minimisation. This achieves \emph{exact} convergence to $x^{\star}_c$ for \emph{finite} $\lambda$, unlike the pure penalty method which requires $\lambda \to \infty$. For the inactive constraint case studied here (where the unconstrained minimum is already feasible), augmented Lagrangian provides no advantage over direct GD.
\end{remark}

\begin{remark}[Proximal Gradient Descent for Composite Objectives]
Both projected GD and the penalty method are special cases of the broader \emph{proximal gradient} framework. For a composite objective $f(x) + g(x)$ where $f$ is smooth and $g$ is convex but possibly non-smooth, the proximal gradient update is:
\begin{equation}
x_{k+1} = \mathrm{prox}_{\alpha g}\!\bigl(x_k - \alpha\,\nabla f(x_k)\bigr), \quad
\mathrm{prox}_{\alpha g}(z) = \arg\min_{x} \Bigl\{g(x) + \frac{1}{2\alpha}\|x - z\|^2\Bigr\}.
\end{equation}
\emph{Projected GD} is the special case where $g(x) = \iota_{\mathcal{X}}(x)$ (the indicator of the feasible set: $0$ if $x \in \mathcal{X}$, $+\infty$ otherwise), giving $\mathrm{prox}_{\alpha \iota_\mathcal{X}}(z) = \Pi_\mathcal{X}(z)$. \emph{ISTA} (Iterative Shrinkage-Thresholding Algorithm) is the case where $g(x) = \lambda\|x\|_1$ (Lasso regularisation), giving the soft-thresholding proximal operator $\mathrm{prox}_{\alpha\lambda\|\cdot\|_1}(z)_i = \mathrm{sign}(z_i)\max(|z_i| - \alpha\lambda, 0)$. Proximal gradient descent achieves the same $O((1-\mu/L)^k)$ convergence as projected GD, and FISTA (with Nesterov momentum applied to the proximal step) achieves $O(1/k^2)$. For the Q5 problem, projecting onto the half-space $\{x : x_1 \geq 0.5\}$ is trivially $O(1)$, but for constraints like the $\ell_1$ ball or nuclear norm ball (matrix problems), the proximal operator requires $O(d)$ or $O(d^3)$ computation respectively.
\end{remark}

\subsection{Linear Programme on Feasible Set $\mathcal{X}$}
\label{subsec:q6_lp}

Feasible set: $\mathcal{X} = \{x \in \R^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$ (a box / bounded polytope).

\textbf{Linear Programme:} Minimise $f(x) = a^\top x = x_1 + 2x_2$ over $\mathcal{X}$ (with $a = [1, 2]^\top$).

Since $f$ is linear and $\mathcal{X}$ is a bounded convex polytope, the \emph{fundamental theorem of linear programming} guarantees that the minimum is attained at a vertex of $\mathcal{X}$. With both components of $a$ positive, the minimum occurs at the lower bounds of both coordinates:
\begin{equation}
x^{\star} = (0.5,\; -5), \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}

\begin{remark}[LP Duality and Verification of Optimality]
The \emph{dual} of the box-constrained LP $\min_{l \leq x \leq u} a^\top x$ is obtained via LP duality. Writing the primal as $\min_x a^\top x$ subject to $x \geq l$, $-x \geq -u$, the Lagrangian is:
\begin{equation}
L(x, \lambda, \mu) = a^\top x - \lambda^\top(x - l) - \mu^\top(u - x) = (a - \lambda + \mu)^\top x + \lambda^\top l - \mu^\top u,
\end{equation}
with $\lambda, \mu \geq 0$. The dual objective is $\inf_x L = \lambda^\top l - \mu^\top u$ (if $a - \lambda + \mu = 0$, else $-\infty$). The dual is $\max_{\lambda, \mu \geq 0} \lambda^\top l - \mu^\top u$ subject to $\lambda - \mu = a$. For our LP, $a = (1, 2)$, $l = (0.5, -5)$, $u = (5, 10)$. Setting $\lambda_i = \max(a_i, 0)$ and $\mu_i = \max(-a_i, 0)$ gives $\lambda = (1, 2)$, $\mu = (0, 0)$, and dual value $\lambda^\top l - \mu^\top u = 1(0.5) + 2(-5) = -9.5 = f^{\star}$. The primal and dual values match, confirming \emph{strong duality} and verifying the optimality of $x^{\star} = (0.5, -5)$ via the dual certificate.
\end{remark}

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}
\label{subsec:q6_fw}

\begin{algorithm}[H]
\caption{Frank--Wolfe with Fixed Mixing Parameter}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\; \grad f(x_k)^\top x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Fixed convex combination step}
\EndFor
\end{algorithmic}
\end{algorithm}

The standard Frank--Wolfe algorithm uses the decaying step-size schedule $\gamma_k = 2/(k+2)$ (i.e.\ $\beta_k = k/(k+2)$), yielding $O(1/k)$ convergence for smooth convex objectives. The implementation here uses a \emph{fixed} mixing parameter $\beta$, equivalent to a constant step size $\gamma = 1 - \beta$ towards the LP vertex $z_k$. This variant converges geometrically in early iterations but may oscillate near the optimum as $z_k$ jumps between vertices.

\begin{theorem}[Frank--Wolfe Convergence, standard schedule]
For an $L$-smooth convex function $f$ with diameter $D = \sup_{x,y \in \mathcal{X}}\|x - y\|$, the standard Frank--Wolfe method with $\gamma_k = 2/(k+2)$ satisfies
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2}.
\end{equation}
This $O(1/k)$ rate is inferior to GD's $O((1-\mu/L)^k)$ but the per-iteration cost is lower: only a linear minimisation oracle (LMO) is required, with no projection or gradient step over $\mathcal{X}$.
\end{theorem}

\begin{remark}[Proof Sketch of FW $O(1/k)$ Rate]
The key idea is to bound $f(x_{k+1})$ using the $L$-smoothness of $f$ and the step $x_{k+1} = x_k + \gamma_k(z_k - x_k)$:
\begin{align}
f(x_{k+1}) &\leq f(x_k) + \gamma_k \grad f(x_k)^\top(z_k - x_k) + \frac{L\gamma_k^2}{2}\|z_k - x_k\|^2 \\
 &\leq f(x_k) - \gamma_k g_k + \frac{L\gamma_k^2 D^2}{2},
\end{align}
where $g_k = \grad f(x_k)^\top(x_k - z_k)$ is the FW gap and we used $\|z_k - x_k\| \leq D$. Since $z_k = \arg\min_{z \in \mathcal{X}} \grad f(x_k)^\top z$, we have $\grad f(x_k)^\top(z_k - x_k) \leq \grad f(x_k)^\top(x^{\star} - x_k) \leq -(f(x_k) - f^{\star})$ (by convexity). Substituting and using $h_k = f(x_k) - f^{\star}$:
\begin{equation}
h_{k+1} \leq (1 - \gamma_k)\,h_k + \frac{L\gamma_k^2 D^2}{2}.
\end{equation}
With $\gamma_k = 2/(k+2)$, this recursion yields $h_k \leq 2LD^2/(k+2)$ by induction.
\end{remark}

For box constraints, the LMO decomposes per-coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\grad f(x_k))_i > 0 \\ u_i & \text{if } (\grad f(x_k))_i \leq 0 \end{cases}
\label{eq:fw_lmo}
\end{equation}
requiring only $O(d)$ operations --- no linear programming solver needed.

\subsection{Interior vs.\ Boundary Behaviour}
\label{subsec:q6_discuss}

\textbf{Interior optimum:} Frank--Wolfe converges to the interior optimum, though at an $O(1/k)$ rate (slower than GD's linear rate). The characteristic ``zig-zag'' pattern arises because $z_k$ jumps between extreme vertices of $\mathcal{X}$, and the convex combination gradually averages these jumps. The standard FW schedule $\gamma_k = 2/(k+2)$ (i.e.\ $\beta_k = k/(k+2)$) recovers the $O(1/k)$ convergence guarantee; the fixed-$\beta$ schedule used here converges geometrically initially but can stall near the optimum when vertex oscillations dominate. Specifically, with fixed $\gamma = 1-\beta$, the error satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq (1-\gamma)^k(f(x_0)-f^{\star}) + \frac{\gamma^2 L D^2}{2(1-(1-\gamma)^k)/\gamma},
\end{equation}
showing that a permanent variance floor of order $O(\gamma^2)$ persists for fixed $\gamma$.

\textbf{Boundary optimum:} Frank--Wolfe is generally better suited to boundary optima on \emph{polytopes} than interior optima, because the LMO naturally selects vertices close to the constrained optimum. However, when the constrained optimum lies on an \emph{edge} (not a vertex), such as $(0.5, 0)$ in Case~III, the LMO cannot select this point and instead alternates between adjacent vertices, producing oscillation. Reaching edge-optimal points requires either more iterations, a modified step-size schedule, or pairwise Frank--Wolfe variants that can move away from vertices.

\textbf{Key advantage of Frank--Wolfe:} Unlike projected gradient descent, FW never requires a projection oracle. For many structured constraint sets (boxes, simplices, $\ell_1$ balls, spectral norm balls, flow polytopes), the LMO can be solved in near-linear time, while projection onto the same sets may require $O(d \log d)$ sorting (simplex) or $O(d^3)$ SVD (spectral norm ball). FW also produces \emph{sparse} iterates when started from a vertex and constrained to polytopes, since each step only moves towards one vertex --- a property exploited in sparse coding and matrix factorisation.

\begin{remark}[Frank--Wolfe and Mirror Descent]
Both Frank--Wolfe and \emph{mirror descent} generalise projected gradient descent to handle structured constraints, but through different mechanisms:
\begin{itemize}[nosep,leftmargin=*]
\item \textbf{Frank--Wolfe}: replaces the projection step with a linear minimisation (LMO). Each iterate is a convex combination of LMO solutions (vertices of $\mathcal{X}$), guaranteeing feasibility without projection.
\item \textbf{Mirror descent}: replaces the Euclidean proximal term with a Bregman divergence $D_\psi(x, y) = \psi(x) - \psi(y) - \nabla\psi(y)^\top(x-y)$, where $\psi$ is a mirror map adapted to the geometry of $\mathcal{X}$. For the simplex, the entropy mirror map $\psi(x) = \sum_i x_i \log x_i$ gives the \emph{exponentiated gradient} (Hedge) algorithm, which updates multiplicatively: $x_{k+1,i} \propto x_{k,i} \exp(-\alpha g_{k,i})$. This achieves $O(1/k)$ convergence with $\ell_\infty$-adapted step sizes, outperforming projected GD when the geometry is ``simplex-like''.
\end{itemize}
For the box constraint $\mathcal{X}$ in this study, both FW and mirror descent recover similar rates, but FW's sparsity property makes it preferred for high-dimensional matrix factorisation problems.
\end{remark}

\begin{remark}[Frank--Wolfe Gap as Stopping Criterion]
The \emph{Frank--Wolfe gap} $g_k = \grad f(x_k)^\top(x_k - z_k)$ satisfies $g_k \geq f(x_k) - f^{\star}$ for convex $f$. It provides a computable, certificate-based stopping criterion: when $g_k < \varepsilon$, the current iterate is within $\varepsilon$ of optimality. This is analogous to checking $\|\grad f(x_k)\| < \varepsilon$ for unconstrained problems.
\end{remark}

\begin{remark}[FW Rate Tightness and Comparison with Projected Methods]
The $O(1/k)$ rate of Frank--Wolfe is \emph{tight}: for a linear objective on the simplex, the standard schedule $\gamma_k = 2/(k+2)$ achieves exactly $h_k \approx 2h_0/(k+2)$. Figure~\ref{fig:q6_fw_rate} demonstrates this tightness empirically, and contrasts FW with Projected GD and FISTA (Nesterov-accelerated proximal GD) on a quadratic objective over the simplex. The figure shows that:
\begin{itemize}[nosep,leftmargin=*]
\item FW achieves $O(1/k)$ convergence (linear on log scale, slope $-1$)
\item Projected GD achieves linear convergence (faster on strongly convex)
\item FISTA (Nesterov + projection) achieves $O(1/k^2)$ convergence
\end{itemize}
This illustrates the fundamental trade-off: FW avoids projections (using only LMO), but the $O(1/k)$ rate is generally inferior to projected methods for strongly convex objectives.
\end{remark}

\begin{remark}[Away-Step Frank--Wolfe]
The standard Frank--Wolfe algorithm cannot move \emph{away} from previously selected vertices, causing slow progress when the optimum lies on an edge. The \emph{Away-Step FW} variant (Wolfe, 1970) addresses this by, at each iteration, choosing between two steps: (i) the standard FW step towards the LP vertex $z_k$, and (ii) an \emph{away step} that decreases the weight of the currently most-active vertex. Away-step FW achieves \emph{linear} convergence on strongly convex objectives over polytopes, compared to the $O(1/k)$ rate of standard FW, at the cost of tracking a \emph{support} --- the set of active vertices in the current convex combination.
\end{remark}

\begin{remark}[Interior Point Methods for Linear and Quadratic Programming]
For the LP in Question~6 with linear objective and box constraints, the optimal solution was found by vertex enumeration --- a simple $O(2^d)$ approach feasible for $d=2$. In general, \emph{interior point methods} (IPMs) provide a superior algorithmic approach:
\begin{itemize}[nosep,leftmargin=*]
\item \textbf{Barrier method}: instead of working on the boundary, replace the LP $\min_{x \in \mathcal{X}} c^\top x$ with a sequence of unconstrained problems $\min_x c^\top x - \frac{1}{t}\sum_i \log(s_i(x))$, where $s_i(x) \geq 0$ are the constraint slack variables. As $t \to \infty$, the barrier terms vanish and the solution converges to the LP optimum. Each unconstrained subproblem is solved with Newton's method in $O(\log\log 1/\varepsilon)$ steps.
\item \textbf{Convergence}: IPMs solve LPs to $\varepsilon$-accuracy in $O(\sqrt{m} \log(1/\varepsilon))$ Newton iterations, where $m$ is the number of constraints, compared to the simplex method's exponential worst-case (though polynomial average-case).
\item \textbf{For Q5 and Q6 in this study}: since both the constraint set and objective are simple (one inequality constraint / box constraints, $d=2$), projected GD and Frank--Wolfe are more appropriate choices. IPMs become advantageous for large-scale structured problems (e.g.\ traffic flow, support vector machines with $10^5$ constraints) where Newton's quadratic convergence amortises the $O(m^3)$ cost per outer iteration.
\end{itemize}
The ADMM (Alternating Direction Method of Multipliers) is another approach that combines the decomposability of dual methods with the convergence of augmented Lagrangian, widely used in distributed optimisation and signal processing.
\end{remark}

\subsection{Cross-Cutting Conclusions}

\begin{enumerate}[leftmargin=*]

\item \textbf{Problem structure dominates method choice.} On the well-conditioned quadratic (Benchmark~A), virtually all first-order methods perform equivalently. The differences emerge on ill-conditioned ($\kappa \approx 6.9$, Benchmark~B) and pathologically ill-conditioned ($\kappa \approx 2508$, Benchmark~C) problems. Newton's method is the most robust across all benchmarks because it directly accounts for curvature via the Hessian, removing condition-number dependence.

\item \textbf{Adaptive step sizes are powerful but sensitive.} The Polyak step achieves the best first-order result on Rosenbrock ($f = 0.024$) when $f^{\star}$ is exactly known, but diverges catastrophically when $f^{\star}$ is wrong. This sensitivity makes it impractical in most real settings. Adagrad and RMSprop are more robust, with RMSprop's exponential forgetting preventing step-size stagnation.

\item \textbf{Momentum acceleration is broadly beneficial.} Nesterov momentum achieves the theoretical optimal $O(1/k^2)$ convergence rate for convex functions and shows the strongest first-order performance on Rosenbrock ($f = 0.479$ vs.\ GD's $f = 3.388$). Adam, while combining momentum and adaptive scaling, performs below Nesterov on the deterministic benchmarks here; its advantage emerges in stochastic settings where gradient variance is high.

\item \textbf{Stochastic methods trade noise for scalability.} Mini-batch SGD on Benchmark~A shows that smaller batches converge faster per epoch (more gradient updates) but exhibit higher oscillation near the minimum. Observation noise raises the irreducible loss floor proportionally to $\sigma^2$, confirmed by the 36-fold increase in noise floor when $\sigma$ is increased from 1 to 6. Importantly, decaying learning rate schedules (polynomial $O(1/\sqrt{t})$ or cosine annealing) allow SGD to escape this noise floor and converge to the true optimum, as demonstrated in Section~\ref{subsec:q2_schedules}.

\item \textbf{Second-order information provides dramatic speedup.} Newton's method converges in 1 step for the quadratic benchmark and in $\sim5$ steps for Benchmark~B, compared to 50--80 steps for first-order methods. The cost is computing and inverting the $d \times d$ Hessian at each step, which scales as $O(d^3)$ --- infeasible for large-scale problems but highly effective in small dimensions.

\item \textbf{Derivative-free methods are necessary without a gradient oracle.} Nelder--Mead simplex navigates the Rosenbrock valley effectively without any gradient information, though without convergence guarantees. Finite-difference gradient approximation closely matches exact GD when $\delta$ is small ($\delta = 0.05$), but degrades significantly for $\delta = 0.8$. Nesterov random search is the most scalable (2 evaluations per step) but converges slowest due to high directional variance.

\item \textbf{Constraint handling requires careful design.} When the constraint is inactive at the optimum (as in Question~5), projected GD and penalty methods converge to the same point. Projected GD enforces feasibility at every step by construction; penalty methods are simpler to implement but require tuning $\lambda$ and may converge to a biased solution when constraints are active. Frank--Wolfe maintains feasibility without projection by working exclusively with convex combinations of feasible points, at the cost of slower $O(1/k)$ convergence.

\end{enumerate}

\begin{remark}[Memory Bottleneck of Second-Order Methods]
The memory column of Table~\ref{tab:complexity} reveals a key practical limitation of Newton's method: it requires $O(d^2)$ memory to store the Hessian and $O(d^3)$ arithmetic operations to invert it. For a neural network with $d = 10^6$ parameters, the Hessian would require $\sim 4\,\text{TB}$ of memory (float32) --- entirely infeasible. L-BFGS reduces this to $O(m_h d)$ with $m_h = 5$--$20$ history pairs, requiring only $\sim 80\,\text{MB}$ for the same network. This makes L-BFGS the practical quasi-Newton method of choice for moderately large problems. For very large-scale problems ($d \gg 10^6$), all gradient methods reduce to $O(d)$ memory, and the dominant memory cost is storing the model parameters and a small number of gradient vectors.
\end{remark}

\subsection{Practical Recommendations}

For practitioners facing optimisation problems in data analysis:
\begin{itemize}[nosep,leftmargin=*]
\item \textbf{Small-scale, smooth, unconstrained:} Newton's method (or quasi-Newton, e.g.\ L-BFGS) if second-order information is available.
\item \textbf{Large-scale or noisy (e.g.\ deep learning):} Adam or RMSprop with appropriate learning rate tuning.
\item \textbf{Ill-conditioned convex:} Nesterov acceleration for deterministic settings; Adam for stochastic.
\item \textbf{Gradient unavailable:} Finite differences (small $d$), Nelder--Mead (small $d$, non-smooth), or Nesterov random search (large $d$).
\item \textbf{Constrained with simple sets:} Projected gradient descent. For structured constraint sets: Frank--Wolfe.
\end{itemize}

\begin{remark}[Quasi-Newton Methods and L-BFGS]
A key limitation of exact Newton's method is the $O(d^3)$ cost of forming and inverting the Hessian. \emph{Quasi-Newton methods} approximate $H_k^{-1}$ from gradient differences, avoiding explicit Hessian computation. The \emph{Broyden--Fletcher--Goldfarb--Shanno} (BFGS) update maintains a positive definite approximation $B_k \approx H_k$ using rank-2 updates from successive gradient changes:
\begin{equation}
s_k = x_{k+1} - x_k, \quad y_k = \grad f(x_{k+1}) - \grad f(x_k), \quad
B_{k+1} = B_k + \frac{y_k y_k^\top}{y_k^\top s_k} - \frac{B_k s_k s_k^\top B_k}{s_k^\top B_k s_k}.
\end{equation}
\emph{Limited-Memory BFGS} (L-BFGS) stores only the last $m = 5$--$20$ curvature pairs $(s_k, y_k)$, reducing memory from $O(d^2)$ to $O(md)$. L-BFGS achieves superlinear convergence and is the method of choice for large-scale smooth unconstrained problems: for example, it is used as the backend optimiser in \texttt{scipy.optimize.minimize} with \texttt{method='L-BFGS-B'} and in many neural network training pipelines for small-to-medium networks. On the benchmarks in this study ($d=2$), L-BFGS would converge in $\sim$3--5 iterations on all three benchmarks.
\end{remark}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z          # evaluate gradient at extrapolated point
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)      # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)   # solve (H + lambda*I) p = g
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe with Fixed Mixing Parameter}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # Linear minimisation oracle for box constraints
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z       # fixed convex combination
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
