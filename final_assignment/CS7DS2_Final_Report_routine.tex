\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to
three benchmark problems of increasing difficulty: a convex quadratic linear regression loss
(Benchmark~A), a non-convex toy neural network loss with sinusoidal perturbation
(Benchmark~B), and the ill-conditioned Rosenbrock function (Benchmark~C). Six families of
methods are systematically investigated: (1)~adaptive step-size methods (Polyak, Adagrad,
RMSprop, Heavy Ball), which automate learning rate selection; (2)~momentum-based and
stochastic methods (Nesterov acceleration, Adam, mini-batch SGD), which exploit gradient
history and achieve super-linear convergence rates; (3)~Newton's method and local
second-order approximations, demonstrating quadratic convergence on near-quadratic functions;
(4)~derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead
simplex, grid search), applicable when analytic gradients are unavailable; (5)~constrained
optimisation via projected gradient descent and linear penalty methods; and (6)~the
Frank--Wolfe conditional gradient algorithm for linearly constrained problems. Each method is
evaluated in terms of convergence speed, trajectory behaviour, stability, and computational
trade-offs. Key findings include: the Polyak step size diverges catastrophically when the
optimal value is misspecified; Nesterov acceleration achieves the optimal $O(1/k^2)$ rate for
convex problems; Newton's method converges in one iteration on the quadratic benchmark; the
constraint in Question~5 is inactive at the unconstrained minimum, so all methods that
achieve feasibility ultimately reach the same optimum; and Frank--Wolfe exhibits a
characteristic $O(1/k)$ convergence rate with zig-zag behaviour near the optimum. All
experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated as:
\begin{equation}
  \theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, I_2), \quad
  y^{(i)} = {\theta^{\star}}^{\!\top} x^{(i)} + \varepsilon^{(i)}, \quad
  \varepsilon^{(i)} \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss is:
\begin{equation}
  J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
            = \frac{1}{2m}\|X\theta - y\|^2.
  \label{eq:loss_A}
\end{equation}
This is a \emph{strongly convex quadratic}: the Hessian $H_A = \frac{1}{m}X^T X$ is constant
and positive definite. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$.
Because $\varepsilon$ is present, the true minimiser $\theta^{\star}_{\text{OLS}} =
H_A^{-1}\frac{1}{m}X^T y$ differs slightly from $\theta^{\star}$, and the minimum loss is
$J^{\star} \approx 0.4826$ (not zero). The smoothness constant is $L = \lambda_{\max}(H_A)$
and the strong convexity constant is $\mu = \lambda_{\min}(H_A)$; with $m=1000$ i.i.d.\ draws
from $\mathcal{N}(0, I)$, the eigenvalues of $H_A$ concentrate near 1, giving condition
number $\kappa(H_A) = L/\mu \approx 1.2$. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
  \label{eq:loss_B}
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is
$H_B(x) = \mathrm{diag}(2 - \sin(x_1),\ 10)$. The $\sin(x_1)$ term introduces a
nonlinearity but does not destroy convexity: since $2 - \sin(x_1) \geq 1 > 0$ for all
$x_1 \in \mathbb{R}$, the Hessian is uniformly positive definite and $f_B$ is strictly
convex. The condition number ranges from $10/1 = 10$ (where $\sin(x_1) = -1$) to
$10/3 \approx 3.3$ (where $\sin(x_1) = 1$), creating moderate anisotropy that makes
per-coordinate adaptive methods advantageous. The global minimum lies at $x^{\star} =
(x_1^{\star}, 2)$ where $x_1^{\star}$ satisfies the transcendental equation
$2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$; numerically $x_1^{\star} \approx 0.582$,
giving $f^{\star} = (0.582-1)^2 + \sin(0.582) \approx 0.174 + 0.549 = 0.724$.
\begin{remark}
  The contour trajectory figures (e.g.\ Figure~\ref{fig:q1_contour_B}) mark the optimum
  at $(0.61, 2.0)$ for plotting purposes; the more precise value $x_1^{\star} \approx 0.582$
  is confirmed by the Newton method results in Table~\ref{tab:q3_final}.
\end{remark}
Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
  f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
  \label{eq:loss_C}
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is:
\begin{equation}
  H_C(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues approximately $0.40$ and $1002$, giving a condition number
$\kappa \approx 2504$. This extreme ill-conditioning, combined with a narrow curved valley
along the parabola $x_2 = x_1^2$, makes the Rosenbrock function one of the most challenging
standard test problems for gradient-based optimisation. The gradient is:
\begin{align}
  \partial_{x_1} f &= -2(1 - x_1) - 400x_1(x_2 - x_1^2), \\
  \partial_{x_2} f &= 200(x_2 - x_1^2).
\end{align}
Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}

For Questions~5 and~6, the feasible set is the box:
\begin{equation}
  \mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
Note that the unconstrained minimiser of Benchmark~B, $x_1^{\star} \approx 0.582 > 0.5$,
lies in the interior of this feasible set. This will have important consequences for the
constrained problem in Section~\ref{sec:q5}.

\section{Question 1: Adaptive Step-Size Methods}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

Gradient descent with fixed step size $\alpha$ satisfies:
\begin{equation}
  f(x_{k+1}) \leq f(x_k) - \alpha\bigl(1 - \tfrac{L\alpha}{2}\bigr)\|\nabla f(x_k)\|^2,
\end{equation}
for $L$-smooth $f$. Stability requires $\alpha < 2/L$. The convergence rate is linear:
$f(x_k) - f^{\star} \leq (1 - \mu/L)^k [f(x_0) - f^{\star}]$, where the ratio $(1 -
\mu/L) = (1 - 1/\kappa)$ gives the per-iteration reduction. Adaptive methods seek to
automatically set $\alpha$ to reduce dependence on $L$ or $\kappa$.

\begin{definition}[Polyak Step Size]
  For a function $f$ with known minimum $f^{\star}$, the Polyak step size is:
  \begin{equation}
    \alpha_k = \frac{f(x_k) - f^{\star}}{\|\nabla f(x_k)\|^2 + \varepsilon},
    \label{eq:polyak}
  \end{equation}
  where $\varepsilon > 0$ is a small regularisation term for numerical stability.
\end{definition}

\begin{theorem}[Polyak Convergence]
  If $f$ is convex with $\|\nabla f(x)\| \leq G$ and $f^{\star}$ is known exactly, the
  Polyak step size guarantees $\min_{k \leq K} f(x_k) - f^{\star} \leq \frac{G^2
  \|x_0 - x^{\star}\|^2}{2K}$.
\end{theorem}

Critically, the guarantee fails if $f^{\star}$ is misspecified: when the true minimum exceeds
the specified value (i.e.\ $f_{\text{true}}^{\star} > 0 = f_{\text{specified}}^{\star}$), as
$\|\nabla f(x_k)\| \to 0$ while $f(x_k) - f_{\text{specified}}^{\star} \to
f_{\text{true}}^{\star} > 0$, the step size $\alpha_k \to \infty$, inducing oscillation.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value estimate $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets (f(x_k) - f^{\star}) \;/\; (\|g_k\|^2 + \varepsilon)$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0 > 0$, $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Coordinate-wise accumulation of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad applies a \emph{larger} effective rate to coordinates with small historical gradients
and a \emph{smaller} rate to coordinates with large historical gradients, automatically
adapting to the problem geometry. However, because $G_k$ grows monotonically, the effective
learning rate $\alpha_0 / (\sqrt{G_k} + \varepsilon)$ decays to zero, which can cause
premature stagnation on non-stationary or very long-running problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop remedies Adagrad's monotonic decay by using an exponential moving average with
forgetting factor $\beta$. The stationary effective step is approximately
$\alpha_0 / \sqrt{\mathbb{E}[g_i^2]}$, adapting to the recent gradient magnitude in each
coordinate without being dominated by early large gradients.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball update can be rewritten as $x_{k+1} = x_k - \alpha\nabla f(x_k) + \beta(x_k
- x_{k-1})$, revealing that it adds a fraction $\beta$ of the previous step. For a quadratic
$f$ with $L$-smooth, $\mu$-strongly convex, the optimal parameters $\alpha = 4/({\sqrt{L} +
\sqrt{\mu}})^2$ and $\beta = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$ achieve
$O((1 - 2/(\sqrt{\kappa}+1))^k)$ convergence, matching the optimal linear rate.

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly (Benchmark~C), Polyak
converges fastest because its step size is calibrated to the optimality gap at every
iteration, effectively implementing an oracle step size. On Benchmarks~A and~B where
$f^{\star}$ is misspecified, Adagrad and Heavy Ball match or exceed GD performance.

\item \textbf{Stability:} The GD baseline is most stable but slowest, constrained to
$\alpha < 2/L$. Polyak is \emph{conditionally} stable: well-behaved when $f^{\star}$ is
correct, catastrophically unstable when misspecified. Adagrad is uniformly stable because its
monotonically decreasing step size guarantees $\alpha_k \to 0$ and the effective learning rate
is bounded: $\alpha_0 / (\sqrt{G_k} + \varepsilon) \leq \alpha_0 / \varepsilon$.

\item \textbf{Adagrad vs RMSprop:} On Benchmark~A, Adagrad exactly matches the GD floor
while RMSprop slightly overshoots ($0.5027$ vs $0.4826$). The key difference is that
RMSprop's exponential moving average $v_k = \beta v_{k-1} + (1-\beta)g_k^2$ forgets old
gradients with timescale $\tau = 1/(1-\beta) = 10$ steps (for $\beta = 0.9$), allowing it
to adapt to changing curvature. Adagrad's monotonic $G_k$ permanently down-weights
coordinates where large early gradients occurred, which is undesirable when the landscape
changes (but beneficial on sparse problems where informative gradients occur rarely).

\item \textbf{Heavy Ball on Rosenbrock:} Heavy Ball with $(\alpha = 0.0008, \beta = 0.86)$
achieves $f = 1.40$ vs GD's $3.51$ on Rosenbrock, the largest relative improvement in Q1
outside Polyak. The momentum accumulation in the consistent valley-floor direction
effectively increases the step in that direction by factor $\approx \alpha / (1-\beta)
= 0.0008 / 0.14 \approx 0.006$, a $5\times$ effective amplification over the bare step.

\item \textbf{Practical guideline:} For well-conditioned problems ($\kappa \lesssim 10$),
vanilla GD with a tuned step size suffices. For ill-conditioned problems, adaptive methods
(Adagrad, RMSprop, Heavy Ball) provide meaningful acceleration at the same per-iteration
cost. When $f^{\star}$ is known (e.g.\ classification losses with known data-generating
distribution), Polyak is optimal.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods}
\label{sec:q2}
% ============================================================

\subsection{Theoretical Background}

\begin{theorem}[Nesterov Optimal Rate]
  For a convex $L$-smooth function $f$ with minimiser $x^{\star}$, Nesterov's accelerated
  gradient method satisfies:
  \begin{equation}
    f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2},
  \end{equation}
  which is optimal among first-order methods in the sense that no algorithm can improve
  this $O(1/k^2)$ rate using only gradient information.
\end{theorem}

This contrasts with GD's $O(1/k)$ rate for convex functions. For strongly convex functions,
Nesterov achieves a linear rate $(1 - \sqrt{\mu/L})^k$, matching the optimal bound.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, maximum momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated
at $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This ``gradient correction''
is the source of the theoretical speedup: by evaluating the gradient at the anticipated next
position, the method effectively reduces the oscillation that limits Heavy Ball on anisotropic
problems. The schedule $\beta_k = (k-1)/(k+2) \nearrow 1$ gradually builds momentum from
zero to $\beta_{\max}$, avoiding the instability that arises when full momentum is applied
from the first iteration. On the Rosenbrock benchmark, this momentum accumulation along the
curved valley is the primary reason Nesterov outperforms all other first-order methods.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines two complementary ideas: (1)~\emph{momentum} via the first moment $m_t \approx
\mathbb{E}[g_t]$, which builds velocity in consistent gradient directions and dampens
oscillations; and (2)~\emph{per-coordinate adaptive scaling} via $v_t \approx
\mathbb{E}[g_t^2]$, which normalises the effective step size in each coordinate, making the
algorithm invariant to the gradient scale. The bias correction factors $(1-\beta_1^t)$ and
$(1-\beta_2^t)$ ensure that $\hat{m}_t$ and $\hat{v}_t$ are unbiased at step $t$, preventing
the warm-up period from being biased towards zero. In these experiments, non-standard $\beta_1$
values ($0.80$--$0.82$ rather than the default $0.9$) are used to reduce the effective memory
for momentum, allowing faster adaptation of the search direction.

Notably, Adam underperforms Nesterov on the Rosenbrock function despite combining both
momentum and adaptive scaling. This occurs because the Rosenbrock valley is highly
\emph{anisotropic and curved}: the direction of descent changes continuously as the iterate
traverses the parabola $x_2 = x_1^2$. Adam's per-coordinate scaling normalises the gradient
magnitude but does not align with the principal curvature directions of the Hessian (which
rotate along the valley). Nesterov's momentum, by contrast, builds a consistent velocity
\emph{along the valley floor direction}, which remains approximately constant over many steps.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A is run with learning rate $\alpha = 0.06$, batch sizes
$b \in \{5, 40\}$, for 50 epochs, with data shuffled uniformly at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{epoch $e = 1, 2, \ldots, E$}
    \State Randomly permute indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \Comment{Stochastic gradient}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $g_{\mathcal{B}}$ is an unbiased estimator of the full gradient:
$\mathbb{E}_{\mathcal{B}}[g_{\mathcal{B}}] = \nabla J(\theta)$. Its variance satisfies
$\mathrm{Var}[g_{\mathcal{B}}] = \frac{1}{b}\mathrm{Var}[\nabla \ell(\theta; x^{(i)},
y^{(i)})]$, decreasing as $1/b$. Thus larger batches give more accurate gradient estimates
at the cost of fewer updates per epoch.

\subsection{Variance Reduction: Beyond Vanilla SGD}

Vanilla mini-batch SGD maintains a gradient variance floor of $\sigma^2/b$ that prevents
convergence below the noise floor even for strongly convex objectives, yielding at best an
$O(1/\sqrt{k})$ rate (versus GD's linear rate). \textbf{Variance-reduced} methods break this
barrier by incorporating periodic full-gradient corrections.

The \textbf{SVRG} (Stochastic Variance Reduced Gradient) update is:
\begin{equation}
  \theta_{k+1} = \theta_k - \alpha\Bigl[\nabla \ell(\theta_k; \xi_k)
  - \nabla \ell(\tilde\theta; \xi_k) + \nabla J(\tilde\theta)\Bigr],
  \label{eq:svrg}
\end{equation}
where $\tilde\theta$ is a \emph{snapshot} recomputed every $p$ steps via a full gradient pass.
The control variate $[\nabla\ell(\theta_k; \xi_k) - \nabla\ell(\tilde\theta; \xi_k)]$ has
zero mean and variance $O(\|\theta_k - \tilde\theta\|^2)$ --- it shrinks to zero as the
iterates converge.  This restores the linear convergence rate $O(\rho^k)$ for strongly convex
$f$, matching full gradient descent but at an amortised cost of $O(m/(p+1))$ gradient
evaluations per ``fast'' step.

\begin{tabular}{lccl}
\toprule
\textbf{Method} & \textbf{Cost/iter} & \textbf{Rate} & \textbf{Note} \\
\midrule

  is small relative to $J^{\star}$.  Variance reduction pays off when $\kappa$ is large or
  when low noise floors are required, as in training deep neural networks with many epochs.
\end{remark}

\subsection{Local Approximations of $g(x) = x^4$}

At a point $x_0 = 0.25$, the function $g(x) = x^4$ has:
$g(x_0) = 0.25^4 = 3.906 \times 10^{-3}$,
$g'(x_0) = 4 \times 0.25^3 = 0.0625$, and
$g''(x_0) = 12 \times 0.25^2 = 0.75$.

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
  \hat{g}^{(1)}(x) = g(x_0) + g'(x_0)(x - x_0) = 3.906\times10^{-3} + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
  \hat{g}^{(2)}(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
  = 3.906\times10^{-3} + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton step minimises $\hat{g}^{(2)}$, setting $\frac{d\hat{g}^{(2)}}{dx} = 0$:
\begin{equation}
  x_{\text{Newton}} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\end{equation}
The true minimiser of $g(x) = x^4$ is $x^{\star} = 0$. Starting from $x_0 = 0.25$, each
Newton step moves to the minimum of the local quadratic model rather than taking a fixed
gradient step, achieving quadratic convergence near $x^{\star}$.

\subsection{Newton's Method}

\begin{definition}[Newton's Method]
  Given $\nabla f(x_k)$ and $\nabla^2 f(x_k)$, the Newton step is $p_k = [\nabla^2
  f(x_k)]^{-1}\nabla f(x_k)$, and the update is $x_{k+1} = x_k - \alpha_k p_k$ where
  $\alpha_k = 1$ for pure Newton and $\alpha_k < 1$ for damped Newton.
\end{definition}

\begin{theorem}[Quadratic Convergence of Newton's Method]
  If $f$ is twice continuously differentiable, $\nabla^2 f$ is Lipschitz continuous near
  $x^{\star}$, and $\nabla^2 f(x^{\star}) \succ 0$, then there exists $\delta > 0$ such
  that for $\|x_0 - x^{\star}\| \leq \delta$, Newton's method with $\alpha_k = 1$ satisfies:
  \begin{equation}
    \|x_{k+1} - x^{\star}\| \leq C \|x_k - x^{\star}\|^2,
  \end{equation}
  for a constant $C = \frac{M}{2\mu}$ where $M$ is the Lipschitz constant of $\nabla^2 f$
  and $\mu = \lambda_{\min}(\nabla^2 f(x^{\star}))$.
\end{theorem}

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda_{\text{damp}} > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda_{\text{damp}}\, I$ \Comment{Tikhonov regularisation ensures $H_k \succ 0$}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction via linear system solve}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

All experiments use Tikhonov damping $\lambda_{\text{damp}} = 10^{-8}$, which ensures the
linear system is non-singular without noticeably altering the Newton direction (since
$\lambda_{\text{damp}} \ll \lambda_{\min}(H)$ for all three benchmarks).

\subsection{Quasi-Newton Methods: Practical Compromise}

Full Newton's method costs $O(d^3)$ per iteration to solve the linear system $H_k p_k = g_k$
and requires forming the $d \times d$ Hessian.  For large $d$ this is prohibitive.
\textbf{Quasi-Newton} methods build a \emph{low-rank approximation} $B_k \approx H_k$ using
only gradient information, reducing cost to $O(d^2)$ (BFGS) or $O(md)$ (L-BFGS with $m$
stored vector pairs).

The \textbf{BFGS} rank-2 Hessian update after step $s_k = x_{k+1} - x_k$ and
gradient change $y_k = \nabla f(x_{k+1}) - \nabla f(x_k)$ is:
\begin{equation}
  B_{k+1} = B_k - \frac{B_k s_k s_k^T B_k}{s_k^T B_k s_k} + \frac{y_k y_k^T}{y_k^T s_k}.
  \label{eq:bfgs}
\end{equation}
$B_{k+1}$ satisfies the \emph{secant condition} $B_{k+1} s_k = y_k$, ensuring the
approximate Hessian matches the true curvature along the most recent step.  Under mild
conditions, BFGS achieves \emph{superlinear} convergence: $\|x_{k+1} - x^{\star}\| /
\|x_k - x^{\star}\| \to 0$, slower than Newton's quadratic rate but much faster than the
linear rate of GD.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation}
\label{sec:q4}

\subsection{Part A: Finite Difference Gradient Approximation (Benchmark B)}

\subsubsection{Finite Differences: Error Analysis}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
  \label{eq:fd}
\end{equation}
By Taylor expansion, the approximation error decomposes as:
\begin{equation}
  \text{Error} = \underbrace{\frac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x)
  + O(\delta^2)}_{\text{truncation error: } O(\delta)}
  + \underbrace{\frac{2\varepsilon_{\text{machine}}|f(x)|}{\delta}}_{\text{floating-point error: }
  O(\varepsilon_{\text{machine}}/\delta)}.
\end{equation}
The optimal $\delta$ balances these two terms, giving $\delta_{\text{opt}} \approx
\sqrt{2\varepsilon_{\text{machine}}} \approx \sqrt{2 \times 2.2 \times 10^{-16}} \approx
2 \times 10^{-8}$. Figure~\ref{fig:fd_error} confirms this analysis empirically: the error
decreases as $O(\delta)$ for large $\delta$ (truncation dominated) and increases as
$O(1/\delta)$ for tiny $\delta$ (floating-point dominated), with a minimum near
$\delta \approx 10^{-8}$. In practice, $\delta = 0.05$ keeps the truncation error
manageable while staying well above the catastrophic cancellation regime. The choice
$\delta = 0.8$ incurs truncation error of order $\approx 0.4 \times \partial^2 f /
\partial x_i^2$, which can be comparable to or larger than the true gradient, explaining the
degraded convergence seen in Figure~\ref{fig:q4_fval_B}.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/fd_error_vs_delta.pdf}
\caption{Q4A: Finite difference gradient approximation error vs step size $\delta$ for
Benchmark~B at $x = (0.5, 3.0)$ (log--log scale). Both theoretical regimes are clearly
visible: $O(\delta)$ truncation error for large $\delta$ and $O(\varepsilon_{\rm machine}/\delta)$
round-off error for small $\delta$. The optimal $\delta \approx 10^{-8}$ minimises total
error. The choices $\delta = 0.05$ (green dashed) and $\delta = 0.8$ (orange dashed) used

\label{fig:fd_error}
\end{figure}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Method)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, smoothing parameter $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This method requires only \emph{two function evaluations per iteration}, independent of
dimension $d$, making it suitable when gradients are unavailable or expensive. The estimate
$\hat{g}_k$ is an unbiased estimator of a \emph{smoothed gradient}: $\mathbb{E}[\hat{g}_k] =
\nabla f_{\delta}(x_k)$ where $f_{\delta}(x) = \mathbb{E}_{u \sim \mathcal{U}(S^{d-1})}
[f(x + \delta u)]$ is the $\delta$-smoothed version of $f$. As $\delta \to 0$,
$f_{\delta} \to f$ and convergence to the true minimum is recovered.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$
and iteratively reshapes it through four operations:
\begin{enumerate}[nosep]
  \item \textbf{Reflection:} move the worst vertex through the centroid of the remaining
  $n$ vertices by factor $\alpha_r = 1$.
  \item \textbf{Expansion:} if reflection improved, try further expansion by factor
  $\gamma = 2$.
  \item \textbf{Contraction:} if reflection worsened, contract towards the centroid by
  factor $\rho = 0.5$.
  \item \textbf{Shrink:} if contraction also fails, contract all vertices towards the
  best vertex by factor $\sigma = 0.5$.
\end{enumerate}
These operations require at most 2--3 function evaluations per iteration and adapt the
simplex geometry to the local landscape without any gradient information. The method lacks
convergence guarantees in general (it can cycle or stagnate), but works reliably in practice
for low-dimensional problems. Initial simplex: step $= 0.35$, 160 iterations.

\subsection{Problem Formulation}

The constrained problem is:
\begin{equation}
  \min_{x \in \mathbb{R}^2}\ f_B(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)
  \quad \text{subject to } x_1 \geq 0.5,
  \label{eq:constrained_B}
\end{equation}
starting from the infeasible point $x_0 = (0.2, 4.0)$.

\begin{remark}[KKT Conditions and Constraint Activity]
  The Karush--Kuhn--Tucker (KKT) conditions for Problem~\eqref{eq:constrained_B} are:
  (Stationarity) $\nabla f_B(x^{\star}) - \mu^{\star}\, e_1 = 0$,
  (Primal feasibility) $x_1^{\star} \geq 0.5$,
  (Dual feasibility) $\mu^{\star} \geq 0$, and
  (Complementary slackness) $\mu^{\star}(x_1^{\star} - 0.5) = 0$.
  Since the unconstrained minimiser satisfies $\nabla f_B(0.582, 2.0) = 0$ and
  $0.582 > 0.5$, we can set $\mu^{\star} = 0$ (inactive constraint) and all KKT conditions
  are satisfied. Thus the constrained and unconstrained optima coincide:
  $x_{\text{constrained}}^{\star} = x_{\text{unconstrained}}^{\star} = (0.582, 2.0)$.
\end{remark}

This observation has an important practical implication: \emph{all} methods that achieve
feasibility will ultimately converge to the same point $(0.582, 2.0)$ because the constraint
$x_1 \geq 0.5$ does not restrict the optimal solution. The differences between methods
manifest in (a)~how quickly they achieve feasibility from the infeasible starting point
$(0.2, 4.0)$, and (b)~how quickly they then converge to the optimum within the feasible set.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$ (possibly infeasible), step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{$\Pi_{\mathcal{X}}$: $x_1 \gets \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space constraint $\{x_1 \geq 0.5\}$, the projection $\Pi_{\mathcal{X}}$ is
simply the coordinate-wise clip $x_1 \gets \max(0.5, x_1)$, computable in $O(1)$ time.
This projection satisfies $\|\Pi_{\mathcal{X}}(y) - x^{\star}\| \leq \|y - x^{\star}\|$ for
all $y$ and all $x^{\star} \in \mathcal{X}$ (non-expansiveness), which is the key property
ensuring convergence.

\begin{algorithm}[H]
\caption{Linear Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
    \Comment{$\partial F_{\lambda}/\partial x_1 = \partial f/\partial x_1 - \lambda \mathbf{1}[x_1 < 0.5]$}
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method replaces the hard constraint by a soft penalty term. As $\lambda \to
\infty$, the penalised minimiser converges to the constrained minimiser, but large $\lambda$
makes the objective less smooth (the penalty gradient is discontinuous at $x_1 = 0.5$),
requiring smaller step sizes for stability.

Settings: Projected GD ($\alpha = 0.08$); Penalty with $\lambda = 0.15$, $1.8$ ($\alpha =
0.05$); Penalty with $\lambda = 4.5$ ($\alpha = 0.03$); Unconstrained GD ($\alpha = 0.07$).
All run for 100 iterations.

\textbf{Projected GD} enforces feasibility as a hard constraint at every iteration, requiring
only a cheap projection oracle (here a scalar $\max$ operation). Its convergence guarantee

\textbf{Penalty methods} avoid the projection oracle but trade exact feasibility for
simplicity. The penalty weight $\lambda$ controls the trade-off: larger $\lambda$ drives
the iterates closer to feasibility at the cost of a less smooth objective and the need for
smaller step sizes. For the sequence $\lambda^{(j)} \to \infty$, the penalty method is
guaranteed to converge to the constrained optimum (penalty method theory), but the
conditioning worsens as $\lambda$ grows.

\textbf{Key insight:} In this particular problem, since the constraint is inactive at the
optimum, unconstrained GD also reaches the correct answer despite passing through the
infeasible region. In general (when the optimum \emph{is} on the constraint boundary),
projected GD is necessary for correctness.

\subsection{Augmented Lagrangian: Bridging Penalty and Dual Methods}

The pure penalty method suffers a fundamental trade-off: large $\lambda$ enforces feasibility
but makes the penalised objective $F_\lambda$ ill-conditioned (condition number grows as
$O(\lambda)$). The \textbf{Augmented Lagrangian} method resolves this by augmenting with
\emph{both} a quadratic penalty \emph{and} a Lagrange multiplier estimate $\mu$:
\begin{equation}
  \mathcal{L}_\rho(x, \mu) = f(x) + \mu \max(0,\, 0.5 - x_1)
  + \frac{\rho}{2}\max(0,\, 0.5 - x_1)^2.
  \label{eq:auglag}
\end{equation}
The multiplier $\mu$ is updated after each inner optimisation as
$\mu^{(j+1)} = \max\!\bigl(0,\; \mu^{(j)} - \rho\,(x_1^{(j)} - 0.5)\bigr)$,
where the $\max$ enforces dual feasibility.  Unlike the pure penalty method, $\rho$ can be
kept moderate (and hence the conditioning bounded) while the multiplier $\mu^{(j)}$ converges
to the true KKT multiplier $\mu^{\star} = 0$ (since the constraint is inactive here).

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm}
\label{sec:q6}
% ============================================================

\subsection{Feasible Set and Linear Programme}

The feasible set is the box $\mathcal{X} = [0.5, 5] \times [-5, 10]$.

\textbf{Linear Programme:} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$.
Since $f$ is linear, by the fundamental theorem of linear programming, its minimum over a
bounded polytope is attained at a \emph{vertex} of the polytope. For a box, the vertices
are the $2^d = 4$ corners. The minimum of each coordinate contribution is obtained
independently: since $a_1 = 1 > 0$ and $a_2 = 2 > 0$, both coordinates are minimised at
their lower bounds:
\begin{equation}
  x^{\star} = (0.5,\, -5), \qquad f^{\star} = a_1 \cdot 0.5 + a_2 \cdot (-5) = 0.5 - 10 = -9.5.
\end{equation}

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q6_lp.pdf}
\caption{Q6-I: Feasible region $\mathcal{X}$ (blue box) and linear objective contours $f(x)
= x_1 + 2x_2$. The contour lines (parallel hyperplanes) decrease in value towards the
lower-left, and the minimum is at the corner $(0.5, -5)$. This illustrates the fundamental
theorem of LP: the minimum of a linear function over a convex polytope is attained at an
extreme point (vertex).}
\label{fig:q6_lp}
\end{figure}

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe Algorithm with Fixed Mixing Parameter}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$
    \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
    \Comment{Convex combination: $x_{k+1} \in \mathcal{X}$ if $x_k \in \mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Frank--Wolfe Convergence Rate]
  For $f$ convex and $L$-smooth with minimiser $x^{\star} \in \mathcal{X}$, and using the
  optimal step size $\gamma_k = 2/(k+2)$ (rather than fixed $\beta$), the Frank--Wolfe
  algorithm satisfies:
  \begin{equation}
    f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2},
  \end{equation}
  where $D = \max_{x,y \in \mathcal{X}}\|x - y\|$ is the diameter of $\mathcal{X}$. This
  $O(1/k)$ rate is strictly worse than projected gradient descent's $O((1 - \mu/L)^k)$
  linear rate, but Frank--Wolfe has the advantage of requiring only linear minimisation
  (the LMO) rather than projection.
\end{theorem}

For box constraints, the LMO decomposes per coordinate:
\begin{equation}
  (z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if }
  (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
costing $O(d)$ operations, compared to $O(d)$ for box projection. The fundamental
advantage of Frank--Wolfe is for problems where projection is expensive (e.g.\ nuclear norm
ball) but LMO is cheap (e.g.\ SVD).

The fixed-$\beta$ scheme used here differs from the standard $\gamma_k = 2/(k+2)$ step-size
schedule. With fixed $\beta$, the update $x_{k+1} = \beta x_k + (1-\beta) z_k$ is a
geometric average, and the step towards $z_k$ is always $(1-\beta)$. This introduces
permanent oscillation near the optimum rather than convergence, since the iterate can never
reach $z_k$ exactly unless $z_k = x^{\star}$.

\subsection{Interior Optimum: $f(x) = (x_1 - 1)^2 + (x_2 - 5)^2$}

The unconstrained minimum $(1, 5)$ lies strictly inside $\mathcal{X}$ (since $0.5 < 1 < 5$
and $-5 < 5 < 10$). Starting point: $x_0 = (1, 1)$. Two mixing parameters are tested:

\subsection{Interior vs Boundary Behaviour Analysis}

\textbf{Interior optimum:} Frank--Wolfe exhibits a characteristic ``zig-zag'' pattern in
$z_k$, which switches between extreme vertices of $\mathcal{X}$ as the gradient changes
direction near the optimum. The convex combination $x_{k+1} = \beta x_k + (1-\beta) z_k$
dampens these jumps, allowing $x_k$ to converge smoothly. The general convergence rate is
$O(1/k)$ for the standard $\gamma_k = 2/(k+2)$ schedule. The gap between the achieved
objective ($0.0046$) and the true minimum ($0.0$) reflects both the finite iteration count
and the residual oscillation inherent in the fixed $\beta$ schedule.

\textbf{Fixed $\beta$ vs decaying schedule:} Figure~\ref{fig:fw_schedule} demonstrates the
superiority of the theoretically optimal decaying schedule $\gamma_k = 2/(k+2)$ over fixed
$\beta$. With fixed $\beta$, the iterate can never fully converge because even when close
to the optimum, each step moves $(1-\beta)$ of the way toward a vertex $z_k$ that may be
far from $x^{\star}$. The decaying schedule reduces this displacement geometrically,
allowing convergence to arbitrarily small error.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/fw_schedule_comparison.pdf}
\caption{Q6: Frank--Wolfe with fixed mixing parameter vs optimal decaying schedule
$\gamma_k = 2/(k+2)$ on the interior optimum ($f(x) = (x_1-1)^2 + (x_2-5)^2$). The
decaying schedule (red) achieves a final value of $\approx 0.001$ after 200 iterations,
significantly outperforming both fixed-$\beta$ variants. The theoretical $O(1/k)$ bound
(dashed black) is loose but confirms the rate; the actual convergence is faster in early
iterations due to the favourable geometry of the quadratic objective within the feasible box.}
\label{fig:fw_schedule}
\end{figure}

\textbf{Boundary optimum:} Frank--Wolfe is well-suited to boundary optima because the LMO
solution $z_k$ frequently selects the vertex of $\mathcal{X}$ closest to the unconstrained
minimum, naturally driving the iterate towards the constraint boundary. The fixed $\beta =
0.93$ prevents full convergence: using the standard decaying schedule with $L = 2$ and
$D = \sqrt{(5-0.5)^2 + (10+5)^2} \approx 15.7$ gives the bound $f(x_k) - f^{\star} \leq
2LD^2/(k+2) \approx 985/(k+2)$, requiring $\approx 3940$ iterations to reach
$f - f^{\star} < 0.25$. The actual convergence in practice is faster because the quadratic
objective has small curvature relative to the box diameter, but this illustrates why
decaying step sizes are essential for convergence to the true optimum.

\textbf{Projection-free advantage:} The key practical advantage of Frank--Wolfe over
projected gradient descent is that it replaces projection ($\text{proj}_{\mathcal{X}}(y)
= \arg\min_{x \in \mathcal{X}} \|x - y\|^2$, a quadratic program) with linear minimisation
(a linear program). For structured constraint sets such as the nuclear norm ball
$\mathcal{X} = \{A : \|A\|_* \leq 1\}$, the LMO requires only one SVD while projection
requires solving a semidefinite program. Similarly, for the convex hull of a combinatorial
set (e.g.\ the set of doubly stochastic matrices), the LMO corresponds to solving an
assignment problem while projection is much harder.

\subsection{Frank--Wolfe Duality Gap as a Convergence Certificate}

A key feature of Frank--Wolfe absent from projected gradient methods is a built-in,
\emph{computable upper bound} on the suboptimality:

\begin{definition}[Frank--Wolfe Gap]
  At iterate $x_k \in \mathcal{X}$, the \emph{Frank--Wolfe gap} (or duality gap) is
  \begin{equation}
    g_k = \nabla f(x_k)^T (x_k - z_k),
    \label{eq:fw_gap}
  \end{equation}
  where $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ is the LMO solution.
\end{definition}

\begin{proposition}
  For convex $f$, the Frank--Wolfe gap satisfies
  $f(x_k) - f^{\star} \leq g_k$.
\end{proposition}

\begin{proof}[Proof sketch]
  By convexity, $f(x^{\star}) \geq f(x_k) + \nabla f(x_k)^T(x^{\star} - x_k)$.
  Since $z_k$ minimises $\nabla f(x_k)^T z$ over $\mathcal{X} \ni x^{\star}$, we have
  $\nabla f(x_k)^T z_k \leq \nabla f(x_k)^T x^{\star}$.  Combining:
  $f^{\star} \geq f(x_k) + \nabla f(x_k)^T(x^{\star} - x_k) \geq
   f(x_k) + \nabla f(x_k)^T(z_k - x_k) = f(x_k) - g_k$.
\end{proof}

The gap $g_k$ is computed \emph{for free} at every iteration (the LMO is already evaluated),
providing a rigorous stopping criterion without knowing $f^{\star}$.  When $g_k < \varepsilon$,
the iterate is guaranteed $\varepsilon$-optimal.  This contrasts with gradient-based methods
where $\|\nabla f(x_k)\| < \varepsilon$ is only a first-order stationarity condition, not a
primal suboptimality bound.  For the interior-optimum experiments, the gap decays
as $g_k \approx 2LD^2/(k+2)$, consistent with the $O(1/k)$ convergence theorem.

\subsection{Key Takeaways}

\begin{enumerate}[nosep]
\item \textbf{Step size adaptation is critical on ill-conditioned problems.} The Rosenbrock
function ($\kappa \approx 2504$ at the optimum) requires a GD step of $\alpha < 2/L \approx
0.002$, making vanilla GD impractically slow. Adaptive methods implicitly estimate local
curvature and take larger effective steps in low-curvature directions.

\item \textbf{Momentum achieves faster convergence via trajectory accumulation.} Nesterov's
$O(1/k^2)$ rate versus GD's $O(1/k)$ directly explains its superiority on convex problems.
Heavy Ball achieves similar acceleration but lacks theoretical guarantees for non-convex
functions. Adam combines both benefits at the cost of additional hyperparameters.

\item \textbf{Second-order information is transformative when available.} Newton's method
converges in one iteration on the quadratic benchmark and dramatically outperforms all
first-order methods on the others, at the cost of $O(d^3)$ per iteration for the linear
system solve. For large-scale problems, quasi-Newton methods (L-BFGS) provide an effective
compromise.

\item \textbf{Derivative-free methods are essential but inherently less efficient.}
Finite differences with appropriate $\delta$ closely approximate gradient descent but scale
as $O(d)$ in function evaluations per step. Nelder--Mead and random search are practical for
low-dimensional black-box problems but cannot scale to high dimensions.

\item \textbf{Constraint handling requires careful algorithm selection.} Projected gradient
descent provides hard feasibility guarantees but requires a projection oracle. Penalty methods
offer simplicity but require increasing $\lambda$ for convergence, degrading conditioning.
Frank--Wolfe avoids projection entirely via the linear minimisation oracle, at the cost of a
slower $O(1/k)$ convergence rate.

\item \textbf{The Polyak step size is a special case:} optimal when $f^{\star}$ is known
exactly, catastrophically unstable otherwise. It serves as a useful theoretical benchmark
rather than a practical algorithm.
\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)   # Cumulative squared gradient
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)   # Exponential moving average of g^2
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball (Polyak Momentum)}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g   # Velocity accumulation
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum (Accelerated Gradient)}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)   # Adaptive momentum
        lookahead = x + beta_k * z                    # Evaluate gradient ahead
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g          # First moment
        v_vec = beta2 * v_vec + (1 - beta2) * g**2       # Second moment
        m_hat = m_vec / (1 - beta1**t)                    # Bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))   # Ensure positive definiteness
        try:
            p = np.linalg.solve(H_reg, g)       # Solve H p = g
        except np.linalg.LinAlgError:
            p = g                                # Fallback to gradient step
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # Linear minimisation oracle (LMO) for box constraints
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # Convex combination update
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
