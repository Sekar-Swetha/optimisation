\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three
benchmark problems spanning the spectrum from well-conditioned quadratic losses ($\kappa \approx 1.12$)
to severely ill-conditioned nonlinear landscapes ($\kappa \approx 2{,}508$).
Six categories of methods are investigated across Questions 1--6: \textbf{(i)}~adaptive step-size
methods (Polyak, Adagrad, RMSprop, Heavy Ball); \textbf{(ii)}~momentum and stochastic methods
(Nesterov acceleration, Adam, mini-batch SGD); \textbf{(iii)}~Newton's method with second-order local
approximations; \textbf{(iv)}~derivative-free methods (finite differences, Nesterov random search,
Nelder--Mead, grid search); \textbf{(v)}~constrained optimisation via projected gradient descent and
penalty methods; and \textbf{(vi)}~the Frank--Wolfe algorithm for linear-constrained problems.

Key quantitative findings include: Newton's method converges in a single iteration on the quadratic
benchmark and to $f = 0.686$ on Rosenbrock in only 20 iterations (vs.\ GD's $f = 3.73$ in 80
iterations); Nesterov acceleration achieves $f = 0.479$ on Rosenbrock compared to GD's $f = 3.39$;
the Polyak step diverges on benchmarks where $f^{\star}$ is misspecified, reaching $f = 24.1$ on
Benchmark~B; and the Frank--Wolfe algorithm converges to an interior optimum with residual
$f = 0.0046$ in 180 iterations. Throughout, the condition number of the problem emerges as the
primary determinant of convergence speed, with Adagrad and Newton effectively pre-conditioning the
gradient direction.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

\noindent This report evaluates 16 optimisation algorithms across six broad categories,
applied to three carefully chosen benchmark problems.  The benchmarks span the range from
perfectly conditioned quadratic losses to highly non-convex, ill-conditioned landscapes,
enabling a rigorous assessment of how algorithmic properties translate to practical
performance.  The experimental findings are connected throughout to the theoretical
convergence guarantees studied in the CS7DS2 course, with particular attention to the role
of the condition number~$\kappa$, the availability of gradient and Hessian information,
and the presence of constraints.

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for full reproducibility.

\subsection{Notation}

Throughout this report the following notation is used.
$f : \mathbb{R}^n \to \mathbb{R}$ denotes the objective function; $\nabla f(x)$ its gradient;
$\nabla^2 f(x)$ (or $H_x$) its Hessian.
$x^{\star}$ denotes a minimiser; $f^{\star} = f(x^{\star})$ the optimal value.
$\|\cdot\|$ is the Euclidean norm. $\odot$ denotes elementwise multiplication.
$\lambda_{\min}$ and $\lambda_{\max}$ are the smallest and largest eigenvalues of a symmetric matrix.
The condition number $\kappa = \lambda_{\max}/\lambda_{\min}$.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1{,}000$ samples and $n = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \sim \mathcal{N}(0, I_2), \quad
y^{(i)} = (\theta^{\star})^T x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
= \frac{1}{2} \theta^T H \theta - b^T \theta + c,
\quad H = \frac{X^T X}{m},\quad b = \frac{X^T y}{m}.
\end{equation}
Since $J$ is a convex quadratic, its Hessian $H = X^T X / m$ is constant.
With $m \gg n$ the sample covariance concentrates near $I_2$, yielding
$\lambda_{\min}(H) \approx 0.925$, $\lambda_{\max}(H) \approx 1.033$, and
$\kappa \approx 1.12$ (nearly isotropic).
The OLS minimiser is $\theta_{\mathrm{OLS}} = H^{-1}b \approx (3.028,\, 4.017)^T$,
close to $\theta^{\star}$ up to estimation noise.
Because observation noise is irreducible, the minimum loss is $J^{\star} \approx 0.4826$, not zero.
Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient and Hessian are
\begin{equation}
\nabla f =
\begin{bmatrix} 2(x_1-1) + \cos(x_1) \\ 10(x_2-2) \end{bmatrix},
\qquad
H(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The unique minimiser satisfies $2(x_1^{\star}-1) + \cos(x_1^{\star}) = 0$ and $x_2^{\star} = 2$,
giving $x^{\star} \approx (0.582,\, 2.000)$ and $f^{\star} \approx 0.7244$.
At the optimum, $H(x^{\star}) = \mathrm{diag}(1.450,\, 10.000)$, so $\kappa \approx 6.90$.
The sinusoidal term introduces mild non-quadratic behaviour but the function remains
strongly convex throughout.  Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$.
The Hessian at the optimum is
\begin{equation}
H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $\lambda_1 \approx 0.40$ and $\lambda_2 \approx 1{,}001.6$,
giving $\kappa \approx 2{,}508$.
This extreme ill-conditioning, combined with the narrow curved valley $x_2 \approx x_1^2$,
makes the Rosenbrock function one of the most challenging standard test problems.
For gradient descent with step $\alpha = 1/\lambda_{\max}$, the linear convergence factor
is approximately $({\kappa - 1})/({\kappa + 1}) \approx 0.9992$, meaning the optimality
gap decays by only $0.08\%$ per iteration.

\textbf{Convexity note.}  Although $f$ is non-convex (the Hessian is indefinite along some
directions away from the minimum), the starting point $x_0 = (-1, 1)$ leads all methods
tested to the unique global minimum at $(1, 1)$; no local minima are encountered in these
experiments.  For multi-start or stochastic settings, derivative-free methods (Nelder--Mead,
grid search, random search) have the advantage of not being trapped by local curvature information.
Starting point: $x_0 = [-1, 1]^T$.

\subsection{Theoretical Background}

For a differentiable function $f$ with $L$-Lipschitz gradient
($\|\nabla f(x) - \nabla f(y)\| \leq L\|x - y\|$ for all $x, y$),
gradient descent with step $\alpha \leq 1/L$ satisfies the descent lemma
\begin{equation}
f(x_{k+1}) \leq f(x_k) - \frac{1}{2L}\|\nabla f(x_k)\|^2.
\end{equation}
For convex $f$ this yields $f(x_k) - f^{\star} \leq \frac{L\|x_0 - x^{\star}\|^2}{2k}$ ($O(1/k)$ rate).
When $f$ is also $\mu$-strongly convex, the convergence is linear:
$f(x_k) - f^{\star} \leq \left(1 - \mu/L\right)^k [f(x_0) - f^{\star}]$.
The adaptive methods below each address a different weakness of vanilla GD.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline
across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive, problem-specific step}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Theory.}
\begin{proposition}[Polyak step optimality~\cite{polyak1969}]
For convex $f$ with exact $f^{\star}$, the Polyak step satisfies
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - \frac{(f(x_k) - f^{\star})^2}{\|\nabla f(x_k)\|^2 + \varepsilon},
\end{equation}
guaranteeing $f(x_k) \to f^{\star}$ monotonically.
\end{proposition}
When $f^{\star}$ is misspecified as $f_{\rm spec}^{\star} < f^{\star}_{\rm true}$,
the step $\alpha_k = (f(x_k) - f_{\rm spec}^{\star}) / (\|\nabla f(x_k)\|^2 + \varepsilon)$
diverges as $x_k \to x^{\star}$: the numerator approaches $f^{\star}_{\rm true} - f_{\rm spec}^{\star} > 0$
while the denominator $\|\nabla f(x_k)\|^2 \to 0$, driving $\alpha_k \to \infty$.
This is precisely the divergence observed on Benchmarks~A and~B.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Theory.} Adagrad~\cite{duchi2011} automatically adapts the learning rate per
coordinate: coordinates with consistently large gradients receive smaller effective rates.
The effective rate for coordinate $i$ after $k$ steps is
$\alpha_0 / (\sqrt{\sum_{t=0}^k g_{t,i}^2} + \varepsilon)$, which monotonically decreases,
ensuring stability but risking premature convergence.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Theory.} RMSprop replaces Adagrad's cumulative sum with an exponential moving
average ($\beta = 0.9$), so older gradients are down-weighted.  This prevents the effective step
from decaying to zero, making RMSprop more suitable for non-stationary problems and deep learning.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Theory.} For a strongly convex quadratic with condition number $\kappa$, the
optimal Heavy Ball parameters are
$\alpha^* = \frac{4}{(\sqrt{L} + \sqrt{\mu})^2}$ and
$\beta^* = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$,
giving convergence factor $\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$ per iteration,
which is substantially smaller than the GD factor $(1 - 1/\kappa)$ for large $\kappa$.

\begin{itemize}[nosep]
\item \textbf{Condition number sensitivity:} On nearly isotropic Benchmark~A ($\kappa = 1.12$),
all methods converge rapidly; differences are negligible. The theoretical GD linear convergence
factor of $(1-1/\kappa)^k \approx (0.895)^k$ is already fast. The $\kappa$ sensitivity becomes
critical on Rosenbrock ($\kappa = 2{,}508$) where the GD factor is $(0.9992)^k$, essentially
stagnant. Methods that implicitly pre-condition (Polyak, Adagrad) gain a decisive advantage.

\item \textbf{$f^{\star}$ sensitivity:} Polyak is the only method requiring exact knowledge of
$f^{\star}$. When $f^{\star}$ is correct (Benchmark~C), Polyak is the best of all five methods
($f = 0.009$); when wrong (Benchmarks A and B), it diverges catastrophically
($f = 2.72$ and $f = 24.07$ respectively). In practice, $f^{\star}$ is almost never known,
limiting Polyak's applicability to settings where a tight lower bound is available.

\item \textbf{Stability:} Adagrad is the most uniformly stable because its effective step is
monotonically non-increasing. GD is stable when $\alpha \leq 1/L$ but sensitive to the choice of
$L$. RMSprop and Heavy Ball can oscillate near the minimum; Heavy Ball with $\beta$ near 1
exhibits damped oscillations typical of under-damped second-order systems.

\item \textbf{Practical recommendation:} For problems where curvature information is unavailable
and $f^{\star}$ is unknown, Adagrad or Heavy Ball with modest $\beta \in [0.85, 0.95]$ provide the
best risk-adjusted performance. RMSprop is preferred when the gradient statistics are non-stationary.
\end{itemize}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\; \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Gradient at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Theory.}
The critical distinction from Heavy Ball is the \emph{lookahead}: gradients are evaluated at
$\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$, allowing the algorithm to
\emph{correct} before overshooting.
The schedule $\beta_k = (k-1)/(k+2)$ is precisely the sequence that achieves
the $O(1/k^2)$ optimal rate for smooth convex functions~\cite{nesterov1983}:

\begin{theorem}[Nesterov, 1983~\cite{nesterov1983}]
Let $f$ be convex and $L$-smooth.  Nesterov's accelerated gradient method with
$\alpha = 1/L$ satisfies
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2},
\end{equation}
which is optimal (matching the oracle lower bound for smooth convex optimisation).
\end{theorem}
This $O(1/k^2)$ rate versus GD's $O(1/k)$ rate represents a genuine improvement,
not merely a constant factor.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (gradient EMA)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (squared gradient EMA)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction, first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias correction, second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Theory.}
Adam~\cite{kingma2015} combines RMSprop's per-coordinate scaling (via $v_t$, effectively normalising
gradients by their recent magnitude) with gradient momentum (via $m_t$, reducing variance and
providing acceleration). The bias correction terms $(1 - \beta_j^t)$ prevent the initial estimates
from being biased towards zero, which is important in early iterations when $\beta_j^t$ is not yet
close to zero. Adam's convergence on general non-convex problems is not guaranteed by classical
theory; it performs best when gradients are informative and the learning rate is tuned carefully.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$,
batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch.
The gradient is computed as $g = \frac{1}{b} X_{\mathcal{B}}^T (X_{\mathcal{B}}\theta - y_{\mathcal{B}})$,
which is an unbiased estimator of $\nabla J(\theta)$ with variance
$\mathrm{Var}[g] \propto \sigma^2 / b$ (for label noise $\sigma^2$).

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Unbiased, noisy gradient}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\centering
\includegraphics[width=0.85\textwidth]{figures/q2_sgd_batch.pdf}

achieves the theoretically optimal $O(1/k^2)$ rate for convex functions. On Rosenbrock, Nesterov

local minima in non-convex problems. For the convex Benchmark~A, the optimal batch size is the
full dataset (batch GD), but mini-batching provides a useful stochastic regularisor.
\end{itemize}

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{Taylor expansion} around $x_0 = 0.25$:
with $g'(x) = 4x^3$, $g''(x) = 12x^2$:
\begin{equation}
g(0.25) = 0.003906,\quad g'(0.25) = 0.0625,\quad g''(0.25) = 0.75.
\end{equation}

\textbf{First-order (linear) approximation:}
\begin{equation}
q_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.003906 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
q_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.003906 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton step minimises $q_2$, setting $q_2'(x) = 0$:
\begin{equation}
x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3} = \frac{2}{3}x_k.
\end{equation}
Starting from $x_0 = 0.25$, this gives $x_1 = 1/6$, $x_2 = 1/9$, \ldots, $x_k = (2/3)^k \cdot 0.25 \to 0$.
Convergence is \emph{linear} with ratio $2/3$, \emph{not} quadratic, because $g''(0) = 0$:
the Hessian vanishes at the optimum, so the standard quadratic-convergence guarantee does not apply.
This illustrates that Newton's quadratic convergence requires a non-degenerate (positive definite)
Hessian at the minimiser.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Theory.}
\begin{theorem}[Newton quadratic convergence~\cite{nocedal2006}]
Let $f \in C^2$ with $\nabla^2 f(x^{\star}) \succ 0$ and Hessian Lipschitz continuous with
constant $M$.  For $x_0$ sufficiently close to $x^{\star}$ and step $\alpha = 1$, Newton's method
satisfies
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2,
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^{\star}))$.
Once $\|x_0 - x^{\star}\| < 2\mu / M$, the iterates converge to $x^{\star}$ quadratically.
\end{theorem}
For a quadratic $f$ with $M = 0$, the Newton step is exact in one iteration regardless of $\kappa$.
Specifically, for $J(\theta) = \frac{1}{2}\theta^T H\theta - b^T\theta + c$,
the Newton update is $\theta_1 = \theta_0 - H^{-1}(H\theta_0 - b) = H^{-1}b = \theta_{\rm OLS}$.
With $\alpha = 1$ (pure Newton step) near $x^{\star}$, the number of correct decimal digits
roughly \emph{doubles} each iteration due to the quadratic convergence rate.
The Tikhonov regularisation $\lambda I$ (damping with $\lambda = 10^{-8}$) ensures
$H_k + \lambda I \succ 0$ globally, making the algorithm well-defined even far from the optimum.

Settings: GD runs 80 iterations; Newton runs 20 iterations with damping $\lambda = 10^{-8}$.

\subsection{Why Newton Works: Curvature-Aware Steps}

Newton's method is effective because the step direction $H^{-1}\nabla f(x)$ performs an implicit
pre-conditioning: it scales the gradient by the inverse of the local curvature in each direction.
For a quadratic with Hessian $H = \mathrm{diag}(\lambda_1, \ldots, \lambda_n)$, the Newton step
moves distance $|\partial f / \partial x_i| / \lambda_i$ in each coordinate, which is exactly
the distance to the minimum in that direction --- hence convergence in one step.
For non-quadratic functions, the Hessian provides the best available local approximation to this
ideal step, leading to quadratic convergence once the iterates are in the region where the
quadratic model is accurate.

\textbf{Limitations and extensions.} The main limitations of Newton's method are:
\begin{itemize}[nosep]
\item \textbf{Cost}: Computing and factorising the Hessian requires $O(n^2)$ and $O(n^3)$
operations respectively, prohibitive for large $n$ (e.g., deep learning with millions of parameters).
\item \textbf{Non-positive-definite Hessians}: Far from the optimum or at saddle points,
$\nabla^2 f$ may not be positive definite, making the Newton direction non-descent.
Tikhonov regularisation $(\nabla^2 f + \lambda I)^{-1}$ addresses this.
\item \textbf{Storage}: The full Hessian matrix requires $O(n^2)$ memory.
\end{itemize}
\textbf{Quasi-Newton methods} (e.g., BFGS) approximate $H^{-1}$ using rank-1 or rank-2
updates from gradient differences across iterates, achieving superlinear convergence without
explicit Hessian computation. L-BFGS limits memory to $O(n \cdot m)$ for a window of $m$
gradient pairs, enabling Newton-like performance at scale.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark~B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with
\begin{equation}
\hat{g}_i(x;\delta) = \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th unit vector.  The true error satisfies
\begin{equation}
\left|\hat{g}_i(x;\delta) - \frac{\partial f}{\partial x_i}(x)\right|
\leq \frac{\delta}{2} M_2 + \frac{\varepsilon_{\mathrm{mach}}}{\delta},
\end{equation}
where $M_2 = \max\|\nabla^2 f\|$ bounds the truncation term and
$\varepsilon_{\mathrm{mach}} \approx 10^{-16}$ accounts for floating-point cancellation.
The optimal step balancing these two errors is
$\delta^* = \sqrt{2\varepsilon_{\mathrm{mach}} / M_2} \approx 10^{-8}$.
For the assignment, $\delta = 0.05$ (near-optimal) and $\delta = 0.8$
(poor, dominated by truncation error) are compared.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Stochastic directional derivative}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent This zeroth-order method estimates a directional derivative along a random unit vector
$u_k$, requiring only \emph{two} function evaluations per iteration regardless of dimension
$n$.  The expected gradient estimate satisfies
$\mathbb{E}[\hat{g}_k] = \mathbb{E}_{u}\left[\frac{f(x + \delta u) - f(x)}{\delta} u\right]
\approx \frac{1}{n} \nabla f(x)$ (projection onto a random direction),
so the effective gradient signal is diluted by a factor of $1/n$ in $n$ dimensions.
This contrasts with finite differences which use $n+1$ function evaluations but recover
the full gradient.

\subsection{Part B: Derivative-Free Methods (Benchmark~C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm~\cite{nelder1965} maintains a simplex of $n+1 = 3$ vertices in
$\mathbb{R}^2$ and iteratively applies four operations to deform it:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): move worst vertex through centroid of remaining vertices.
\item \textbf{Expansion} ($\gamma = 2.0$): if reflection improved, try moving further in the same direction.
\item \textbf{Outside contraction} ($\rho = 0.5$): if reflection improved slightly, contract outside.
\item \textbf{Inside contraction} ($\rho = 0.5$): if reflection worsened, contract inside.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction failed, contract all vertices towards the best.
\end{enumerate}
Settings: initial simplex step $= 0.35$, iterations $= 160$.

\subsection{Centred Differences and Automatic Differentiation}

The forward difference formula achieves $O(\delta)$ accuracy.  The \emph{centred} difference
\begin{equation}
\hat{g}_i^{\rm cd}(x;\delta) = \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}
\end{equation}
achieves $O(\delta^2)$ accuracy at the cost of $2d$ function evaluations per gradient, because
the first-order error terms cancel by symmetry.  For the assignment's ``good'' $\delta = 0.05$,
the centred approximation error would be $O(0.05^2 \cdot M_3) \approx 0.0025 M_3$, potentially
100$\times$ more accurate than the forward difference.

\textbf{Automatic differentiation} (AD) provides exact gradients at $O(d)$ cost
(forward mode) or $O(1)$ gradient evaluations for a scalar loss (reverse mode, i.e., backpropagation).
AD eliminates both truncation and round-off errors entirely, making it the preferred method
when the function is implemented in a differentiable framework (PyTorch, JAX, TensorFlow).
The finite-difference approach tested here is most useful when the function is a black box
or implemented in non-differentiable code.

\subsection{Problem Formulation and KKT Analysis}

The constrained problem is:
\begin{equation}
\min_{x \in \mathbb{R}^2} f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)
\quad \text{subject to} \quad x_1 \geq 0.5,
\end{equation}
with infeasible starting point $x_0 = (0.2, 4.0)$.

\textbf{KKT conditions.}
Writing the inequality constraint as $g(x) = 0.5 - x_1 \leq 0$, the KKT conditions are:
\begin{align}
\nabla f(x^{\star}) + \mu^{\star} \nabla g(x^{\star}) &= 0, \label{eq:kkt_stat}\\
\mu^{\star} &\geq 0, \label{eq:kkt_dual}\\
\mu^{\star} g(x^{\star}) &= 0. \label{eq:kkt_comp}
\end{align}
Since $\nabla g = (-1, 0)^T$, equation \eqref{eq:kkt_stat} gives
$2(x_1^{\star}-1) + \cos(x_1^{\star}) = \mu^{\star}$ and
$10(x_2^{\star}-2) = 0 \Rightarrow x_2^{\star} = 2$.
The unconstrained minimiser of $f$ satisfies $x_1^{\star} \approx 0.582 > 0.5$,
so the constraint $x_1 \geq 0.5$ is \emph{inactive} at the unconstrained minimum.
By complementary slackness \eqref{eq:kkt_comp}: since $g(x^{\star}) = 0.5 - 0.582 < 0$,
we need $\mu^{\star} = 0$. Substituting back into \eqref{eq:kkt_stat}:
$2(0.582-1) + \cos(0.582) \approx -0.836 + 0.836 = 0$ \checkmark.

\begin{remark}
The constrained minimum coincides with the unconstrained minimum $x^{\star} = (0.582, 2.000)$
with $f^{\star} = 0.7244$, because the feasible region $\{x_1 \geq 0.5\}$ contains the
unconstrained minimiser. The challenge is the \emph{infeasible starting point}: all methods
must first navigate from $(0.2, 4.0)$ into the feasible half-space, then converge to the
interior optimum.
\end{remark}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $(x_1)_+ = \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Theory.}
\begin{theorem}[Projected GD convergence~\cite{bubeck2015}]
For a closed convex feasible set $\mathcal{X}$ and $L$-smooth convex objective $f$, projected GD
with step size $\alpha \leq 1/L$ satisfies
\begin{equation}
f(x_k) - f^{\star} \leq \frac{\|x_0 - x^{\star}\|^2}{2\alpha k},
\end{equation}
i.e., $O(1/k)$ convergence, identical to unconstrained GD.  The key property used is
non-expansiveness of the projection:
$\|\Pi_{\mathcal{X}}(x) - \Pi_{\mathcal{X}}(y)\| \leq \|x - y\|$ for all $x, y$.
\end{theorem}
The projection for $x_1 \geq 0.5$ is the halfspace projection $\max(0.5, x_1)$,
computable in $O(1)$.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$ \Comment{Gradient of $F_\lambda$}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Theory.}
The penalty method approximates the constrained problem by the unconstrained minimisation of
$F_\lambda$.  As $\lambda \to \infty$, the minimiser of $F_\lambda$ approaches the constrained
minimiser.  For finite $\lambda$, the residual violation at the penalty minimiser $x_\lambda^{\star}$
satisfies $0.5 - (x_\lambda^{\star})_1 \leq f(x_0)/\lambda$, so larger $\lambda$ implies
smaller violation but also a stiffer (larger Lipschitz constant) objective, requiring
smaller step sizes for stability.

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$
($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$, reflecting the
steeper penalised landscape), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\noindent\textbf{Summary of trade-offs.}
Projected GD enforces hard feasibility at every iteration at the cost of requiring a
projection oracle. For simple constraints (e.g., box or halfspace constraints), projection
is $O(n)$; for more complex sets (e.g., nuclear norm balls, Birkhoff polytopes), projection
can be expensive.
Penalty methods require only a gradient oracle for the modified objective and are easy to
implement for any differentiable constraint function, but provide only approximate feasibility
for finite $\lambda$.  Increasing $\lambda$ improves feasibility but requires smaller step sizes,
creating a computational trade-off.

\noindent\textbf{Augmented Lagrangian improvement.}
The \emph{augmented Lagrangian} method (Multiplier Method) overcomes the ill-conditioning
of pure penalty methods by combining the penalty with a Lagrange multiplier estimate:
\begin{equation}
L_\lambda(x, \mu) = f(x) + \mu \max(0,\, 0.5 - x_1) + \frac{\lambda}{2}[\max(0,\, 0.5 - x_1)]^2.
\end{equation}
The multiplier $\mu$ is updated as $\mu_{k+1} = \mu_k + \lambda \max(0, 0.5 - x_{1,k})$,
which allows convergence to the exact constrained optimum for any fixed finite $\lambda > 0$,
avoiding the ill-conditioning that grows with $\lambda$ in the pure penalty approach.
For the current problem, since the constraint is inactive ($\mu^{\star} = 0$), the augmented
Lagrangian reduces to the pure penalty at optimality.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$
(a box with diameter $\mathrm{diam}(\mathcal{X}) = \|(4.5, 15)\| \approx 15.66$).

\textbf{Linear Programme:} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$.
Since $f$ is linear, the minimum is attained at a vertex of the box.  As both components of $a$
are positive, minimising $a^T x$ independently per coordinate gives the lower bounds:
\begin{equation}
x^{\star} = (0.5,\; -5), \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}
This exemplifies the \emph{fundamental theorem of linear programming}: a linear objective over a
bounded polytope is minimised at a vertex (extreme point).

The contour lines are parallel with slope $-1/2$; decreasing $c$ moves them towards the lower-left.

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: stays in $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Linear minimisation oracle.}
For box constraints, the LP subproblem decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
This $O(d)$ operation replaces the projection oracle of projected GD.

\noindent\textbf{Convergence theory.}
\begin{theorem}[Frank \& Wolfe, 1956~\cite{frank1956}]
Let $f$ be convex and $L$-smooth over a compact convex set $\mathcal{X}$ with diameter $D$.
The Frank--Wolfe algorithm with step-size rule $\gamma_k = 2/(k+2)$ satisfies
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2}.
\end{equation}
\end{theorem}
The $O(1/k)$ rate matches (unconstrained) gradient descent, but with the critical advantage that
all iterates remain in $\mathcal{X}$ by construction (convex combination of feasible points),
requiring no projection.

\noindent\textbf{Frank--Wolfe gap.}
A practical convergence certificate is the \emph{Frank--Wolfe gap}:
\begin{equation}
g_{\mathrm{FW}}(x_k) = \nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - f^{\star},
\end{equation}
which provides an upper bound on suboptimality, computable without knowing $f^{\star}$.
The fixed-$\beta$ variant used here replaces $\gamma_k = 2/(k+2)$ with a constant mixing weight
$1 - \beta$, which does not enjoy the theoretical $O(1/k)$ guarantee but is more predictable in
practice.

\subsection{Interior vs Boundary Behaviour Analysis}

\textbf{Interior optimum.}
Frank--Wolfe converges but exhibits a characteristic zig-zag oscillation: the LP vertex $z_k$
alternates between opposing extreme points of $\mathcal{X}$ as the gradient rotates around the
interior optimum, and the convex combination gradually averages these jumps.
The theoretical $O(1/k)$ rate applies with the standard $\gamma_k = 2/(k+2)$ schedule;
the fixed-$\beta$ variant used here achieves similar empirical behaviour but without a formal
guarantee. For the interior case, the Frank--Wolfe gap at the final iterate is
$g_{\mathrm{FW}}(x_k) = \nabla f(x_k)^T(x_k - z_k)$, which can be evaluated without knowledge
of $f^{\star}$ and provides an upper bound on suboptimality.

\textbf{Boundary optimum.}
Frank--Wolfe is particularly well-suited to problems where $x^{\star}$ lies on the boundary:
the LMO consistently selects the vertex in the direction of steepest descent, naturally steering
the iterate towards the constraint boundary without explicit projection.
The residual gap $f(x_{140}) - f^{\star} = 0.364 - 0.25 = 0.114$ after 140 iterations.
Using the convergence bound with Lipschitz constant $L = 2$ (for $f(x) = x_1^2 + x_2^2$) and
$D \approx 15.66$, the theoretical bound gives
$f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2} = \frac{2 \cdot 2 \cdot 245}{142} \approx 6.9$,
which is a very loose bound due to the large diameter.
Using $\gamma_k = 2/(k+2)$ instead of fixed $\beta$ would tighten convergence in theory.

\textbf{Frank--Wolfe vs Projected GD.}
On box-constrained problems, both algorithms are computationally $O(d)$ per iteration
(projection = clipping, LMO = sign-of-gradient per coordinate).
Frank--Wolfe maintains feasibility by construction and requires no explicit projection, but
converges at $O(1/k)$ regardless of condition number.  Projected GD can accelerate to
$O(1/k^2)$ via Nesterov-style variants and adapts step sizes, but needs a projection oracle.
The LMO in Frank--Wolfe always returns a vertex of $\mathcal{X}$, which makes the method
attractive for problems where sparse/vertex solutions are desirable (e.g., trace-norm
constrained matrix factorisation, sparsity-constrained machine learning).

\noindent\textbf{Addressing the zig-zagging problem.}
The oscillation of $z_k$ between vertices of $\mathcal{X}$ (observed in Figure~\ref{fig:q6_evolution}
for $\beta = 0.90$) is a fundamental limitation of standard Frank--Wolfe.
The \emph{away-step} variant addresses this: at each iteration, the algorithm considers both
moving \emph{towards} a vertex (standard FW step) and moving \emph{away} from the current
atom with maximum weight in the convex decomposition of $x_k$.  Away steps prevent the
algorithm from repeatedly re-introducing the same extreme point, achieving a linear convergence
rate $O((1 - \mu/(LC^2))^k)$ on strongly convex objectives~\cite{bubeck2015}, compared to the
$O(1/k)$ rate of the standard FW algorithm.

\subsection{Key Findings}

\begin{enumerate}

\item \textbf{Condition number governs convergence.}
Benchmark A ($\kappa = 1.12$): all methods converge in $< 20$ iterations.
Benchmark B ($\kappa = 6.9$): methods diverge in performance; adaptive/momentum methods gain.
Benchmark C ($\kappa = 2{,}508$): dramatic divergence; GD is ineffective without pre-conditioning
or second-order information.

\item \textbf{Second-order methods dominate.}
Newton's method is the only algorithm tested that is immune to the condition number: it converges
in 1 step on any quadratic, regardless of $\kappa$.  For strongly non-quadratic problems
(Rosenbrock), even a damped Newton step achieves $5.4\times$ better results than 4$\times$ more
GD iterations.  The cost is computing and inverting the Hessian ($O(n^3)$ per step).

\item \textbf{Momentum is effective for ill-conditioned smooth problems.}
Nesterov acceleration achieves the theoretically optimal $O(1/k^2)$ rate and achieves
$f = 0.479$ on Rosenbrock vs GD's $f = 3.39$ --- a $7\times$ improvement.  Heavy Ball is
competitive but without the convergence guarantee.

\item \textbf{Polyak step size is optimal when $f^{\star}$ is known.}
On Benchmark C where $f^{\star} = 0$ is correct, Polyak achieves the best performance among
all first-order methods (f = 0.009). However, its sensitivity to $f^{\star}$ misspecification
makes it impractical when the true minimum is unknown.

\item \textbf{Adaptive methods reduce tuning burden.}
Adagrad and Adam/RMSprop are uniformly competitive across all benchmarks without requiring the
careful step-size selection that GD needs. For Benchmark A, all methods converge equally; for C,
adaptive methods outperform constant GD by implicitly addressing the anisotropy.

\item \textbf{Derivative-free methods trade accuracy for generality.}
Nelder--Mead achieves competitive results on Rosenbrock without any gradient information.
Grid search guarantees global optimality to grid resolution but is impractical beyond $d = 3$.
Nesterov random search provides a principled gradient-free baseline with dimension-independent
cost per step.

\item \textbf{Constrained optimisation: projection vs penalty.}
Projected GD enforces hard feasibility at every iteration; penalty methods are simpler to
implement but provide only approximate feasibility. Since the constrained minimum coincides with
the unconstrained minimum in this problem (constraint inactive at optimum), both methods converge
to the same point; projected GD reaches it faster due to immediate feasibility enforcement.

\item \textbf{Frank--Wolfe is projection-free and suited to boundary optima.}
The $O(1/k)$ rate and LMO-based iteration make Frank--Wolfe competitive with projected GD on
box-constrained problems. Its natural tendency to favour vertices of the feasible set makes it
ideal for problems requiring sparse solutions.

\end{enumerate}

\subsection{Method Selection Guidelines}

\begin{itemize}[nosep]
\item \textbf{Quadratic or near-quadratic problem, $n$ small}: use Newton's method.
\item \textbf{Smooth strongly convex, first-order access only}: use Nesterov or Adam.
\item \textbf{$f^{\star}$ known exactly}: Polyak step is optimal in the strongly convex case.
\item \textbf{No gradient available}: Nelder--Mead (low $d$), or random search (any $d$).
\item \textbf{Noisy gradients (stochastic setting)}: mini-batch SGD with decreasing step size.
\item \textbf{Linear constraint set}: Frank--Wolfe avoids projection overhead.
\item \textbf{Simple constraint set (halfspace, box)}: projected gradient descent.
\end{itemize}

\begin{lstlisting}[caption={Gradient Descent}]
def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex}]
def nelder_mead(loss_fn, x0, step, n_iters, alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1])
        f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1])
            f_e = loss_fn(x_e)
            if f_e < f_r: simplex[-1] = x_e; f_vals[-1] = f_e
            else: simplex[-1] = x_r; f_vals[-1] = f_r
        else:
            if f_r < f_vals[-1]:
                x_c = centroid + rho * (x_r - centroid)
                f_c = loss_fn(x_c)
                if f_c <= f_r: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
            else:
                x_c = centroid + rho * (simplex[-1] - centroid)
                f_c = loss_fn(x_c)
                if f_c < f_vals[-1]: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (Conditional Gradient)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
