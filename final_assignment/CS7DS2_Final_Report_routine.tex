\begin{abstract}
% ============================================================
This report presents a systematic comparative study of first-order, second-order, derivative-free, and constrained optimisation algorithms applied to three benchmark problems of increasing difficulty: a convex quadratic linear-regression loss (Benchmark A), a mildly non-quadratic toy neural-network loss with sinusoidal perturbation (Benchmark B), and the Rosenbrock function (Benchmark C). Six families of methods are investigated across six questions worth 30 marks each.

\textbf{Question 1} compares four adaptive step-size strategies against a constant-step gradient descent baseline. The Polyak step size achieves the fastest convergence ($f = 0.0094$) on Rosenbrock when $f^\star$ is known exactly, but diverges on Benchmarks A and B because the misspecified $f^\star = 0$ causes the effective step size to grow without bound. Adagrad and Heavy Ball both match the gradient descent noise floor on convex problems, while RMSprop's exponential forgetting of past gradients provides smoother convergence.

\textbf{Question 2} shows that Nesterov acceleration ($f_C = 0.479$) and Adam ($f_C = 1.578$) both substantially outperform gradient descent ($f_C = 3.388$) on Rosenbrock. Mini-batch SGD demonstrates that smaller batch size ($b = 5$) achieves faster per-epoch progress through more frequent updates, but exhibits higher gradient variance. Increasing observation noise by a factor of six raises the irreducible loss floor by $\approx 36\times$ and reduces the relative advantage of smaller batches.

\textbf{Question 3} confirms Newton's quadratic convergence rate: for the quadratic Benchmark A, a single Newton step reaches machine-precision accuracy, while gradient descent requires 80 iterations. On the ill-conditioned Rosenbrock, damped Newton ($\alpha = 0.22$) reaches $f = 0.686$ in only 20 steps versus $f = 3.73$ for gradient descent in 80 steps.

\textbf{Question 4} demonstrates that finite-difference gradient approximation with a well-chosen perturbation ($\delta = 0.05$) closely tracks exact gradient descent, while a large perturbation ($\delta = 0.8$) introduces significant truncation error. The Nelder--Mead simplex method navigates the Rosenbrock valley without any gradient information, and grid search provides global coverage at the cost of $O(N^d)$ function evaluations.

\textbf{Question 5} shows that projected gradient descent enforces the constraint $x_1 \geq 0.5$ exactly from the first iteration, whereas penalty methods enforce it only asymptotically, with higher $\lambda$ providing faster constraint satisfaction at the cost of requiring smaller step sizes.

\textbf{Question 6} demonstrates that the Frank--Wolfe algorithm naturally handles box-constrained problems without explicit projections. When the unconstrained optimum lies in the interior, convergence is smooth; when it lies on the boundary, the fixed-$\beta$ schedule results in residual oscillation.
\end{abstract}

\subsection{Benchmark A: Linear Regression Quadratic Loss}
% ------------------------------------------------------------
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
  \theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \sim \mathcal{N}(0, I_2), \quad
  y^{(i)} = {\theta^{\star}}^{T} x^{(i)} + \varepsilon^{(i)}, \quad
  \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The mean-squared-error loss is
\begin{equation}
  J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
            = \frac{1}{2m}\norm{X\theta - y}^2.
\end{equation}
This is a \emph{strongly convex quadratic} with constant Hessian $H = \frac{1}{m}X^TX$.  The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$.  Due to observation noise, the irreducible noise floor is $J^\star \approx 0.4826$ (not zero); the optimal parameter is the OLS solution $\hat{\theta}_{\mathrm{OLS}} = (X^TX)^{-1}X^Ty \approx [3, 4]^T$.  Starting point: $\theta_0 = [0,\,0]^T$.

The smoothness constant is $L = \lambda_{\max}(H)$ and the strong convexity parameter is $\mu = \lambda_{\min}(H)$, giving condition number $\kappa = \lambda_{\max}/\lambda_{\min}$.  For a random design matrix, $\kappa$ is typically moderate (around 2--5 here), making Benchmark A relatively easy.

\subsection{Benchmark B: Toy Neural Network Loss}
% ------------------------------------------------------------
\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = \bigl[2(x_1-1)+\cos x_1,\; 10(x_2-2)\bigr]^T$ and the Hessian is
$H(x) = \mathrm{diag}(2-\sin x_1,\; 10)$.
The $\sin$ term makes the function non-quadratic along $x_1$, creating a mild non-convexity near the unconstrained minimum.  Setting the gradient to zero yields $x_2^\star = 2$ and $2(x_1^\star - 1) + \cos x_1^\star = 0$, which gives $x_1^\star \approx 0.5823$ by Newton bisection.  Thus $f^\star = (0.5823-1)^2 + \sin(0.5823) \approx 0.1745 + 0.5499 = 0.7244$.

At the minimum the Hessian eigenvalues are $\lambda_1 = 2-\sin(0.5823) \approx 1.45$ and $\lambda_2 = 10$, giving $\kappa \approx 6.9$.  Starting point: $x_0 = [-1,\,4]^T$.

\begin{remark}
  The code marks the visual optimum in contour plots at $(0.61, 2.0)$, which is an approximation; the true minimiser is at $(x_1^\star, 2) \approx (0.5823, 2.0)$ as confirmed by Newton's method (Table~\ref{tab:q3_final}).
\end{remark}

\subsection{Algorithm Formulations and Theoretical Background}

Four adaptive methods are compared against a constant step-size gradient descent (GD) baseline for 120 iterations on all three benchmarks.

\begin{algorithm}[H]
\caption{Constant Step-Size Gradient Descent (Baseline)}
\label{alg:gd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\,\nabla f(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[GD Convergence for $L$-smooth Convex Functions]
For a convex, $L$-smooth function with step size $\alpha \leq 1/L$, gradient descent satisfies
$f(x_k) - f^\star \leq \frac{\norm{x_0 - x^\star}^2}{2\alpha k}$, i.e., an $O(1/k)$ rate.
For $\mu$-strongly convex functions, the rate improves to linear: $f(x_k) - f^\star \leq (1 - \mu\alpha)^k(f(x_0) - f^\star)$.
\end{theorem}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, target value $f^\star$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^\star}{\norm{g_k}^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\,g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Polyak Step-Size Derivation]
The Polyak step minimises the worst-case distance to the optimum.  For a subgradient $g_k$ and step $\alpha_k$, we have
$\norm{x_{k+1}-x^\star}^2 = \norm{x_k - \alpha_k g_k - x^\star}^2 = \norm{x_k-x^\star}^2 - 2\alpha_k\inner{g_k}{x_k-x^\star} + \alpha_k^2\norm{g_k}^2$.
Using the subgradient inequality $\inner{g_k}{x_k-x^\star} \geq f(x_k)-f^\star$ and minimising over $\alpha_k$ yields $\alpha_k^\star = (f(x_k)-f^\star)/\norm{g_k}^2$.
\textbf{Critical requirement:} the method requires exact knowledge of $f^\star$.  When $f^\star$ is misspecified as zero but $f^\star_{\mathrm{true}} > 0$, the numerator $f(x_k)-0$ does not vanish as $\|\nabla f(x_k)\| \to 0$, causing $\alpha_k \to \infty$ and divergence.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Elementwise squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Adagrad Properties]
Adagrad adapts the learning rate per coordinate: directions with persistently large gradients receive small updates, and infrequently activated coordinates receive larger ones.  The denominator $\sqrt{\sum_{j=0}^k g_{k,i}^2}$ grows without bound, so the effective step size $\alpha_0/(\sqrt{G_{k+1,i}}+\varepsilon) \to 0$ monotonically.  This guarantees stability but can lead to premature convergence for non-sparse problems.
\end{remark}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\,v_k + (1-\beta)\,g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[RMSprop vs Adagrad]
RMSprop addresses Adagrad's vanishing step size by replacing the full cumulative sum $G_k$ with an exponential moving average $v_k$.  With decay $\beta = 0.9$, only the last $1/(1-\beta) = 10$ effective steps of gradient history are retained, allowing the effective step to remain non-zero throughout training.  This makes RMSprop better suited to non-stationary objectives (e.g., neural network training) at the expense of losing Adagrad's convergence guarantees.
\end{remark}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\,z_k + \alpha\,\nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Heavy Ball on Quadratics and Beyond]
For a $\mu$-strongly convex, $L$-smooth quadratic with $\kappa = L/\mu$, Heavy Ball with optimal parameters $\alpha^\star = 4/(\sqrt{L}+\sqrt{\mu})^2$ and $\beta^\star = \bigl((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)\bigr)^2$ achieves a convergence rate of $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$ per iteration --- the \emph{optimal} first-order rate for quadratics --- compared to $(1-1/\kappa)$ for gradient descent.  The speed-up factor grows as $\kappa \to \infty$: for $\kappa = 2508$ (Rosenbrock at optimum), the Heavy Ball rate $\approx 0.969$ is dramatically better than GD's $\approx 0.9996$ per iteration.

On \emph{non-quadratic} functions such as Rosenbrock, Heavy Ball has no such guarantee: the curvature changes along the trajectory, so the optimal parameters for one region may cause overshoot or oscillation in another.  However, the Rosenbrock valley is approximately quadratic locally along its axis, so momentum continues to be beneficial: our results show Heavy Ball achieves $f = 1.40$ vs GD's $f = 3.51$ on Rosenbrock (a $2.5\times$ improvement), confirming that momentum provides practical acceleration even without theoretical guarantees in the non-quadratic setting.
\end{remark}

\subsection{Comparative Analysis}
% ------------------------------------------------------------

\begin{itemize}[nosep,leftmargin=*]
\item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest due to its theoretically optimal step size (Bench.\ C: $f=0.009$ at 120 iters). Otherwise, Adagrad and Heavy Ball match or exceed GD performance on convex problems, while Polyak fails catastrophically on A and B.
\item \textbf{Smoothness:} RMSprop produces the smoothest curves due to exponential smoothing with $\beta = 0.9$ that averages over $\approx 10$ recent gradient-squared values. Heavy Ball exhibits characteristic momentum overshoot (visible as a slight initial overshoot on Benchmark B before the trajectory settles) followed by exponential damping. Adagrad's monotonically decreasing step gives smooth but perpetually decelerating convergence --- once $G_k$ is large, the effective step can never increase again, even if the loss landscape becomes less curved.
\item \textbf{Stability:} GD with a well-chosen constant step is uniformly stable. Polyak is conditionally stable (requires exact $f^\star$). Both Adagrad and RMSprop are unconditionally stable for all $\alpha_0 > 0$ and $\varepsilon > 0$ due to the bounded effective step $\leq \alpha_0/\varepsilon$.
\item \textbf{Per-coordinate adaptation vs isotropic methods:} Adagrad and RMSprop implicitly perform \emph{diagonal preconditioning}: effectively approximating the preconditioned step $D^{-1}\nabla f$ where $D = \mathrm{diag}(\sqrt{G_k}+\varepsilon)$ adapts to coordinate-wise gradient magnitudes. This is particularly valuable for Benchmark B where $\kappa \approx 7$ --- the $x_2$ direction has $\approx 6.9\times$ more curvature than $x_1$, so Adagrad allocates a proportionally smaller step to $x_2$. In contrast, GD and Polyak use the same step in all directions.
\item \textbf{RMSprop vs Adagrad:} The key difference is the ``memory horizon''. Adagrad's $G_k = \sum_{j=0}^k g_j^2$ remembers \emph{all} past gradients equally. For a stationary problem, this causes over-accumulation leading to vanishing steps. RMSprop's $v_k = 0.9\,v_{k-1} + 0.1\,g_k^2$ has an effective horizon of $\approx 10$ steps, maintaining responsiveness to recent gradient changes. This makes RMSprop better for non-stationary objectives (e.g., neural network training with mini-batches) but provides no theoretical convergence guarantee, unlike Adagrad's guaranteed $O(1/\sqrt{T})$ regret bound in convex online learning.
\item \textbf{Practical implication:} Since exact $f^\star$ is rarely available, Polyak has limited applicability in practice. RMSprop and Adam (Section~\ref{sec:q2}) are preferred for neural network training. For sparse or heterogeneous-gradient problems, Adagrad provides automatic learning rate adaptation without hyperparameter tuning.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}

\subsection{Algorithm Formulations and Theoretical Background}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\; \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\,z_k$ \Comment{Look-ahead point}
    \State $z_{k+1} \gets \beta_k\,z_k - \alpha\,\nabla f\!\left(\tilde{x}_k\right)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Optimal Rate]
For a convex, $L$-smooth function with $\alpha = 1/L$ and the schedule $\beta_k = (k-1)/(k+2)$, Nesterov's accelerated gradient method satisfies
$f(x_k) - f^\star \leq \frac{2L\norm{x_0 - x^\star}^2}{(k+1)^2},$
an $O(1/k^2)$ rate, compared to the $O(1/k)$ rate of gradient descent.  This is \emph{optimal} among first-order methods for smooth convex minimisation.
\end{theorem}

\begin{remark}[Lookahead Mechanism]
The key distinction from Heavy Ball is that the gradient is evaluated at the \emph{look-ahead} point $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$.  This allows the method to ``look ahead'' in the direction of momentum and correct its trajectory before overshooting, giving the characteristic smooth convergence trajectory.  The increasing $\beta_k = (k-1)/(k+2)$ schedule gradually builds momentum as trust in the direction grows.
\end{remark}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\quad v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\,m_{t-1} + (1-\beta_1)\,g_t$ \Comment{1st moment (momentum)}
    \State $v_t \gets \beta_2\,v_{t-1} + (1-\beta_2)\,g_t \odot g_t$ \Comment{2nd moment (adaptive scaling)}
    \State $\hat{m}_t \gets m_t\,/\,(1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\,/\,(1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \dfrac{\alpha\,\hat{m}_t}{\sqrt{\hat{v}_t} + \varepsilon}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Adam Design Rationale and Limitations]
Adam combines two ideas: (i) momentum via the first-moment estimate $m_t$, which accumulates gradient history to smooth updates; and (ii) per-coordinate adaptive scaling via $v_t$ (as in RMSprop), which normalises the effective step in each coordinate by recent gradient magnitudes.  The bias correction factors $(1-\beta_1^t)^{-1}$ and $(1-\beta_2^t)^{-1}$ compensate for the zero-initialised moments being biased toward zero in the first few iterations.  With $\beta_1 = 0.82$ and $\beta_2 = 0.999$, the first moment retains $1/(1-0.82) \approx 5.6$ effective gradient steps of history, and the second moment retains $1/(1-0.999) = 1000$ steps.

\textbf{Why Nesterov outperforms Adam on Rosenbrock:} The Rosenbrock valley is a one-dimensional manifold $x_2 \approx x_1^2$ embedded in $\mathbb{R}^2$.  Progress along this valley requires \emph{correlated} steps: the gradient in the $x_2$ direction must consistently combine with the $x_1$ direction to follow the curved valley.  Nesterov's momentum $z_k$ accumulates this consistent directional signal, producing an effective ``valley navigator'' after several iterations.  Adam's per-coordinate scaling \emph{destroys this correlation} by independently normalising the $x_1$ and $x_2$ updates, treating each coordinate as if it were independent.  The coordinate with the largest recent gradient variance (typically $x_1$ near the valley walls) gets a smaller effective step, while the valley-floor direction (involving both coordinates) receives no special treatment.  This is the fundamental reason why Nesterov ($f = 0.479$) substantially outperforms Adam ($f = 1.579$) on Rosenbrock despite Adam's superior empirical performance on deep learning tasks.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}
% ------------------------------------------------------------

Mini-batch SGD on Benchmark A uses $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data index set $\{1,\ldots,m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b}X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\,g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[SGD Variance Analysis]
With batch size $b$, each mini-batch gradient $g_b$ satisfies $\mathbb{E}[g_b] = \nabla J(\theta)$ and $\mathrm{Var}(g_b) = \sigma^2/b$, where $\sigma^2$ measures per-sample gradient variance.  With $b = 5$ and $m = 1000$ samples, each epoch performs $1000/5 = 200$ gradient updates, while $b = 40$ gives only $1000/40 = 25$ updates.  The smaller batch processes the same data 8$\times$ more frequently, explaining the faster per-epoch loss decrease at the cost of higher oscillation near the minimum.
\end{remark}

\begin{remark}[Noise Decomposition and Learning Rate Implications]
The total gradient noise has two components: (i) \emph{observation noise} $\propto \sigma^2$ from the data labels, which is irreducible regardless of batch size; and (ii) \emph{sampling noise} $\propto \sigma^2/b$ from using a subset of data.  Under high $\sigma = 6$, component (i) dominates and the relative advantage of $b = 5$ (which reduces sampling noise) is small compared to the irreducible floor.

Formally, with fixed learning rate $\alpha$, SGD converges to a region around the optimum with radius $r \approx \alpha\sigma^2/b$ (from a standard noise-floor analysis).  With $\sigma = 6$ and $b = 5$: $r \propto \alpha \cdot 36/5 = 7.2\alpha$.  With $b = 40$: $r \propto \alpha \cdot 36/40 = 0.9\alpha$.  The ratio of noise floors is $7.2/0.9 = 8$, meaning under high noise, the large-batch method converges to a tighter region around $\theta^\star$.  This is the opposite effect from the low-noise case, confirming that larger batches are preferable under high observation noise, while smaller batches are preferable when observation noise is low and sampling variance dominates.

In practice, this motivates \emph{learning rate decay} schedules (e.g., $\alpha_k = \alpha_0/(1 + \lambda k)$) which allow SGD to converge to the exact optimum rather than oscillating in a noise floor: as $\alpha_k \to 0$, the noise floor $r \to 0$.
\end{remark}

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
  g(x) \approx g(x_0) + g'(x_0)(x - x_0)
            = 0.25^4 + 4\cdot 0.25^3\,(x - 0.25)
            = 0.0039 + 0.0625\,(x - 0.25).
\end{equation}
This is a tangent line that correctly predicts the function's slope but not its curvature.

\textbf{Second-order approximation:}
\begin{align}
  g(x) &\approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 \nonumber \\
       &= 0.0039 + 0.0625\,(x-0.25) + \tfrac{1}{2}\cdot 12\cdot 0.25^2\,(x-0.25)^2 \nonumber \\
       &= 0.0039 + 0.0625\,(x-0.25) + 0.375\,(x-0.25)^2.
\end{align}
Minimising this quadratic approximation yields the Newton update:
\begin{equation}
  x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)}.
\end{equation}
In multiple dimensions, this generalises to $x_{k+1} = x_k - [\nabla^2 f(x_k)]^{-1}\nabla f(x_k)$, the damped version of which (with step $\alpha$) is implemented here.

\centering
\includegraphics[width=0.88\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$. The first-order approximation (blue dashed tangent line) correctly represents the slope at $x_0$ but predicts no curvature, extrapolating linearly and therefore locating the minimum incorrectly (at $-\infty$). The second-order approximation (red dash-dot parabola) captures the local curvature and provides a well-defined minimum estimate at $x_1^* = x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 = 0.1667$. This motivates Newton's method: the true minimum of $x^4$ is at 0, and Newton iterates would converge to it by repeatedly refitting the local quadratic.}

\subsection{Newton's Method}
% ------------------------------------------------------------

\begin{algorithm}[H]
\caption{Newton's Method with Damping and Hessian Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \leq 1$, damping $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov regularisation ensures $H_k \succ 0$}
    \State Solve $H_k\,p_k = g_k$ for Newton direction $p_k$
    \State $x_{k+1} \gets x_k - \alpha\,p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Quadratic Convergence of Newton's Method]
Let $f$ be twice continuously differentiable with Lipschitz-continuous Hessian.  Near a non-degenerate local minimum $x^\star$, Newton's method (with $\alpha = 1$) satisfies:
$\norm{x_{k+1} - x^\star} \leq C\,\norm{x_k - x^\star}^2$
for some constant $C > 0$.  This is \emph{quadratic convergence}: the number of correct decimal places doubles at each iteration.  For quadratic functions, the convergence is exact in one step ($\alpha = 1$), since the second-order model is exact.
\end{theorem}

\begin{remark}[Necessity of Damping and Regularisation]
For non-quadratic functions such as Rosenbrock, two modifications are needed: (i) a step size $\alpha < 1$ (damping) prevents Newton steps from overshooting when the local quadratic model is inaccurate far from the minimum; and (ii) a regularisation $\lambda I$ added to the Hessian ensures positive definiteness when the Hessian is indefinite (e.g., near saddle points or in regions of negative curvature).  Here $\lambda = 10^{-8}$ is small enough to have negligible effect near the minimum but prevents numerical errors.
\end{remark}

Settings: GD uses $\alpha_{\mathrm{GD}}$ for 80 iterations; Newton uses $\alpha_{\mathrm{N}}$ for 20 iterations; damping $\lambda = 10^{-8}$.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}
% ============================================================

\subsection{Part A: Finite Differences and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of the partial derivative is:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector.  Replacing the exact gradient with this approximation and running gradient descent yields a \emph{pseudo-gradient descent} method.

\begin{remark}[Bias-Variance Trade-off in Finite Differences]
The approximation error decomposes into two parts:
\begin{align}
  \frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)
  &= \underbrace{\frac{\delta}{2}f''_{ii}(x) + O(\delta^2)}_{\text{truncation bias, } O(\delta)}
   + \underbrace{O\!\left(\frac{\varepsilon_{\mathrm{mach}}}{\delta}\right)}_{\text{rounding error}}.
\end{align}
The total error is minimised at $\delta_{\mathrm{opt}} \approx \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$ for double precision.  The two values tested here are:
\begin{itemize}[nosep]
  \item $\delta = 0.05$: truncation bias $\approx 0.025\,f''_{ii}$ --- small, leading to accurate gradient approximation.
  \item $\delta = 0.8$: truncation bias $\approx 0.4\,f''_{ii}$ --- large relative to the gradient, severely distorting the descent direction.
\end{itemize}
\end{remark}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise $u_k \gets u_k / \norm{u_k}$
    \State Estimate directional derivative: $\hat{g}_k \gets \dfrac{f(x_k + \delta\,u_k) - f(x_k)}{\delta}\,u_k$
    \State $x_{k+1} \gets x_k - \alpha\,\hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Random Search Properties]
Nesterov random search requires only \emph{two} function evaluations per iteration regardless of dimension $d$, making it dimension-independent in per-step cost. The expected gradient estimator satisfies $\mathbb{E}[{\hat{g}_k}] = \frac{d}{d+\delta^2}\nabla f_\delta(x_k)$, where $f_\delta(x) = \mathbb{E}_{u}[f(x+\delta u)]$ is a smoothed version of $f$. For small $\delta$, $f_\delta \approx f$.  However, the estimator has high variance because it projects onto a single random direction rather than the full gradient.  For dimension $d = 2$ (Benchmark B), this is less severe than in high dimensions.
\end{remark}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations:

\begin{enumerate}[nosep,label=(\roman*)]
  \item \textbf{Reflection} ($\alpha_r = 1$): move worst vertex through the centroid of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{\text{worst}})$.
  \item \textbf{Expansion} ($\gamma = 2$): if reflection improved, try a further step: $x_e = \bar{x} + \gamma(\bar{x} - x_{\text{worst}})$.
  \item \textbf{Contraction} ($\rho = 0.5$): if reflection worsened, contract towards the centroid.
  \item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all vertices towards the best.
\end{enumerate}

Settings: initial simplex step $= 0.35$ (starting from $x_0 = (-1, 1)$, simplex vertices at $(-1,1)$, $(-0.65, 1)$, $(-1, 1.35)$), iterations $= 160$.

\begin{remark}[Why Nelder--Mead Handles Rosenbrock]
The Nelder--Mead simplex \emph{deforms} to fit the local landscape geometry through its four operations.  On the Rosenbrock banana valley:
\begin{itemize}[nosep]
  \item In the initial descent to the valley floor, \textbf{reflection} and \textbf{expansion} allow large steps in the steep $x_2$ direction.
  \item Once on the valley floor, the simplex elongates along the valley axis (the $x_2 \approx x_1^2$ curve) through successive expansions, effectively aligning the simplex with the slow-varying direction.
  \item Near the optimum $(1,1)$, \textbf{contraction} and \textbf{shrink} reduce the simplex size for fine-scale convergence.
\end{itemize}
This geometric adaptation allows Nelder--Mead to traverse the curved valley without the coordinate-coupling information that gradient methods need to handle the off-diagonal Hessian term $H_{12} = -400x_1$.  The function-value comparisons among simplex vertices serve as a \emph{shape learning} mechanism, implicitly discovering the valley's orientation.
\end{remark}

\section{Question 5: Constrained Optimisation \hfill [30 marks]}

\subsection{Problem Setup}

Minimise $f(x_1, x_2) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)$ subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$.

The unconstrained minimum $x^\star \approx (0.5823, 2.0)$ lies within the feasible region ($x_1^\star \approx 0.5823 > 0.5$), so the \textbf{constrained minimum coincides with the unconstrained minimum}.  The constraint is \emph{inactive} at the optimal solution.  This means a sufficiently powerful constrained method will converge to the same point $(0.5823, 2.0)$ as the unconstrained method, but must do so while maintaining feasibility throughout (or converging to it).

\begin{remark}[KKT Conditions]
The Karush--Kuhn--Tucker (KKT) first-order necessary conditions for the constrained problem
$\min_x f(x) \ \text{subject to}\ x_1 \geq 0.5$
are: $\nabla f(x^\star) = \mu e_1$ for some $\mu \leq 0$ (complementary slackness: $\mu(x_1^\star - 0.5) = 0$).
Since the constraint is inactive ($x_1^\star = 0.5823 > 0.5$), complementary slackness gives $\mu = 0$, reducing to $\nabla f(x^\star) = 0$ --- the unconstrained stationarity condition.  This explains why all methods that converge to $x_1^\star \approx 0.5823$ achieve the same minimum value.
\end{remark}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\,\nabla f(x_k)$ \Comment{Unconstrained step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Projection: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Convergence of Projected GD]
For a convex objective and convex feasible set $\mathcal{X}$, projected gradient descent retains the same $O(1/k)$ convergence rate as unconstrained GD.  The projection $\Pi_\mathcal{X}$ onto a closed convex set is non-expansive: $\|\Pi_\mathcal{X}(a) - \Pi_\mathcal{X}(b)\| \leq \|a - b\|$, which preserves the descent property.  For our simple half-space constraint, the projection is $\Pi_\mathcal{X}(x) = (\max(0.5, x_1), x_2)$, computed in $O(1)$ time.
\end{remark}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\State Define penalised objective: $F_\lambda(x) := f(x) + \lambda\max(0,\,-x_1+0.5)$
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute $\nabla F_\lambda(x_k)$: gradient of $f$ plus $-\lambda e_1$ when $x_1 < 0.5$
    \State $x_{k+1} \gets x_k - \alpha\,\nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Penalty Method Analysis]
The penalty term $\lambda\max(0,-x_1+0.5)$ is a linear penalty (L1-type) that pushes the iterate towards the feasible region.  As $\lambda \to \infty$, the penalised minimum converges to the constrained minimum.  For finite $\lambda$, there is a residual constraint violation proportional to $1/\lambda$: the gradient of the penalty at a point $x_1 < 0.5$ is $-\lambda$, balancing the objective gradient $\partial f/\partial x_1$ at a point where $x_1 = 0.5 - \|\nabla_{x_1}f\|/\lambda$, i.e., the iterate stops slightly inside the infeasible region.  Higher $\lambda$ requires smaller step sizes (reduced to $\alpha = 0.03$ for $\lambda = 4.5$) to maintain stability of the gradient iterations.
\end{remark}

Settings: Projected GD ($\alpha = 0.08$); Penalty $\lambda = 0.15, 1.8$ ($\alpha = 0.05$); Penalty $\lambda = 4.5$ ($\alpha = 0.03$); Unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\subsection{Comparative Discussion}
% ------------------------------------------------------------

\begin{itemize}[nosep,leftmargin=*]
\item \textbf{Projected GD} enforces feasibility as a \emph{hard constraint} at every iteration. The projection oracle is trivial for half-space constraints but can be expensive for complex feasible sets (e.g., semidefinite constraints require $O(d^3)$ eigendecomposition per step).
\item \textbf{Penalty methods} trade exact feasibility for implementation simplicity --- only a modified gradient is required. The fundamental limitation is the $\lambda$-accuracy trade-off: for $\lambda$ large enough to enforce the constraint precisely, the step size must decrease, potentially slowing convergence. A proper augmented Lagrangian method (which adds both a linear multiplier term and a quadratic penalty) overcomes this by estimating the optimal multiplier and achieves exact feasibility with finite $\lambda$.
\item \textbf{Practical guidance:} When the constraint set has a cheap projection, projected GD is preferred. When the constraint is complex or the dimension is high, penalty methods (or interior-point methods) may be more tractable.
\end{itemize}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} Minimise $f(x) = a^Tx = x_1 + 2x_2$ over $\mathcal{X}$.

\begin{proposition}[LP Optimum at Vertex]
The optimum of a linear objective over a bounded convex polytope is always attained at a \emph{vertex} (extreme point) of the polytope.  For a box constraint with positive coefficients $a_i > 0$, the minimum of $\sum_i a_i x_i$ is achieved at the lower bound of each coordinate.  Here both $a_1 = 1 > 0$ and $a_2 = 2 > 0$, so
\begin{equation}
  x^\star = (0.5,\,-5), \qquad f^\star = 0.5 + 2(-5) = -9.5.
\end{equation}
\end{proposition}

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \displaystyle\argmin_{x \in \mathcal{X}}\,\nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\,x_k + (1-\beta)\,z_k$ \Comment{Convex combination (stays in $\mathcal{X}$)}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Frank--Wolfe: Projection-Free Property]
The Frank--Wolfe method replaces the projection step of projected GD with a linear minimisation oracle (LMO): find $z_k = \argmin_{x\in\mathcal{X}} \nabla f(x_k)^T x$.  For a box constraint, this decomposes per coordinate:
\begin{equation}
  (z_k)_i = \begin{cases} l_i & \text{if } [\nabla f(x_k)]_i > 0 \\ u_i & \text{if } [\nabla f(x_k)]_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations. Crucially, the iterate $x_{k+1} = \beta x_k + (1-\beta)z_k$ is a convex combination of two feasible points ($x_k, z_k \in \mathcal{X}$) and thus remains feasible without explicit projection.  This is the key advantage when the LMO is cheap but projection is expensive (e.g., for trace-norm or nuclear-norm constraints in matrix optimisation).
\end{remark}

\begin{remark}[Frank--Wolfe Duality Gap]
The \emph{Frank--Wolfe gap} $D_k = \inner{\nabla f(x_k)}{x_k - z_k}$ is a certificate of suboptimality:
$f(x_k) - f(x^\star) \leq D_k$.
At the true optimum, $D_k = 0$ (since $z_k = x_k$ when $\nabla f(x_k)^T(x-x_k) \geq 0$ for all $x \in \mathcal{X}$). This gap can be computed at zero additional cost and serves as a stopping criterion.  The Frank--Wolfe convergence rate satisfies $f(x_k) - f^\star \leq O(1/k)$ for smooth convex functions, identical to gradient descent but without projection.
\end{remark}

\subsection{Interior vs Boundary Behaviour: Comparison}
% ------------------------------------------------------------

\begin{itemize}[nosep,leftmargin=*]
\item \textbf{Interior optimum (fixed $\beta$):} The iterate $x_{k+1} = \beta x_k + (1-\beta)z_k$ can be written as a geometric series:
  $x_k = \beta^k x_0 + (1-\beta)\sum_{j=0}^{k-1}\beta^{k-1-j}z_j$.
  The iterate is a weighted average of all past LP vertices, with weights decaying geometrically.  As $k \to \infty$ with fixed $\beta$, the sum converges if and only if $z_j \to z^\star$ (a fixed vertex).  The ``zig-zag'' behaviour occurs when $z_j$ alternates between two vertices at opposite corners of $\mathcal{X}$: consecutive LP solutions at $(0.5, 10)$ and $(0.5, -5)$ cancel, limiting convergence.  Using the standard decaying schedule $1 - \beta_k = 2/(k+2)$ would guarantee $O(1/k)$ convergence to the true optimum.
\item \textbf{Boundary optimum (fixed $\beta$):} The constrained minimum at $(0.5, 0)$ lies at a boundary vertex of $\mathcal{X}$.  The LP vertex $z_k$ consistently selects $(0.5, -5)$ for most iterations (gradient $\nabla f = 2x$ points away from the lower-left corner once $x_2 < 0$). The iterate converges to the corner but the fixed $\beta = 0.93$ prevents reaching exact $(0.5, 0)$: at convergence, $x^\infty = (1-\beta)\sum_{j=0}^\infty \beta^j z_j \approx (0.5, -5)$ if $z_j$ is always $(0.5, -5)$, but $x_2^{(k)}$ overshoots and oscillates.  The observed final value $f(x) = 0.364 > f^\star = 0.25$ quantifies this residual.
\item \textbf{Projection-free advantage:} Both cases demonstrate Frank--Wolfe's key property: iterates remain feasible throughout without any projection, since $x_{k+1}$ is a convex combination of $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$.  For the box constraint here, projection is trivial, but for constraints like the nuclear-norm ball $\{X : \|X\|_* \leq 1\}$ in matrix optimisation, the LMO requires only a rank-1 SVD ($O(d^2)$) while projection requires a full SVD and thresholding ($O(d^3)$), making Frank--Wolfe the method of choice.
\item \textbf{Convergence rate summary:} Frank--Wolfe with optimal decaying step $\gamma_k = 2/(k+2)$ satisfies $f(x_k) - f^\star \leq 4LC^2/(k+2)$ where $C = \max_{x,y\in\mathcal{X}}\|x-y\|$ is the diameter.  The fixed-$\beta$ variant used here does not have this guarantee, as confirmed by the residual gaps in Table~\ref{tab:q6_final}.
\end{itemize}

\section{Conclusion}
\label{sec:conclusion}

\subsection*{Key Results by Algorithm Family}

\begin{description}[style=nextline]
\item[Adaptive Step-Size Methods (Q1)]
The Polyak step size is theoretically optimal when $f^\star$ is known exactly, achieving the best convergence on Rosenbrock ($f = 0.009$ at 120 iterations). However, its complete failure when $f^\star$ is misspecified (as on Benchmarks A and B where the noise floor is nonzero) limits its practical utility. Adagrad provides uniformly stable convergence with per-coordinate scaling, making it robust to anisotropic curvature. RMSprop's exponential forgetting prevents Adagrad's step-size collapse but at the cost of some convergence guarantees. Heavy Ball achieves the best momentum-based performance in Q1 with a $2.5\times$ improvement over GD on Rosenbrock.

\item[Momentum and Stochastic Methods (Q2)]
Nesterov acceleration achieves the most significant improvement on Rosenbrock ($7.1\times$ better than GD), confirming its $O(1/k^2)$ theoretical advantage. Adam performs well across all benchmarks, combining momentum and per-coordinate adaptation, but slightly lags Nesterov on strongly anisotropic problems. Mini-batch SGD's convergence is dominated by the gradient variance from the noise model under high-noise conditions, reducing the relative benefit of smaller batch sizes.

\item[Newton's Method (Q3)]
For quadratic functions, Newton's method achieves machine-precision accuracy in a single step ($\alpha = 1$), dramatically outperforming first-order methods. On the non-quadratic Rosenbrock, damped Newton ($\alpha = 0.22$) still achieves a $5.4\times$ improvement over GD using only 25\% as many iterations, at the cost of computing and inverting the Hessian ($O(d^3)$ per step). The quadratic convergence rate makes Newton the method of choice whenever Hessian computation is affordable.

\item[Derivative-Free Methods (Q4)]
Finite-difference gradient approximation with a small perturbation ($\delta = 0.05$) closely tracks exact GD, offering a practical gradient-free alternative when analytic derivatives are unavailable. A large $\delta = 0.8$ severely corrupts the gradient estimate. Nelder--Mead navigates the Rosenbrock valley effectively without any gradient information, demonstrating the power of geometric simplex methods for low-dimensional problems. Grid search provides global coverage guarantees but scales exponentially with dimension, limiting it to $d \leq 3$.

\item[Constrained Optimisation (Q5)]
Projected gradient descent enforces feasibility exactly from iteration 1, making it the most reliable method for this problem. Penalty methods offer a simpler implementation but require $\lambda \to \infty$ for exact feasibility; the $\lambda = 4.5$ case achieves near-zero violation within 10 iterations while $\lambda = 0.15$ never fully enforces the constraint. The practical take-away is that projected GD should be preferred when a cheap projection exists.

\item[Frank--Wolfe / Linear Programming (Q6)]
The LP fundamental theorem is confirmed: the minimum of a linear objective over a polytope is attained at a vertex. Frank--Wolfe's projection-free property makes it valuable for constraint sets where the LMO is cheaper than the projection. Interior optima are approached via zig-zagging convex combinations; boundary optima are handled more naturally, as the LMO consistently selects the boundary-nearest vertex.
\end{description}

\subsection*{Limitations and Scope}

Several limitations apply to the current experimental study:
\begin{itemize}[nosep,leftmargin=*]
  \item \textbf{Small dimension}: all benchmarks are $d = 2$. The relative performance of methods, particularly derivative-free approaches and Newton's method, would change significantly for $d \gg 1$.
  \item \textbf{Hyperparameter sensitivity}: all methods were run with problem-specific hyperparameters tuned for the assignment. Real-world performance requires learning-rate search, and the rankings may not generalise.
  \item \textbf{Fixed iteration budget}: methods are compared at fixed iteration counts (120/150/80/20). If compared by total function evaluations, the rankings would differ (e.g., Newton uses the Hessian at $O(d^3)$ cost per step).
  \item \textbf{Non-convex globalisation}: Newton's method and gradient-based methods can converge to local minima in general non-convex problems. Rosenbrock is convex and has a unique global minimum, so this is not an issue here, but the Toy Neural Network's sinusoidal term could create local minima for larger amplitude.
  \item \textbf{No stochastic noise in Q3--Q6}: Questions 3--6 use the full (exact) objective gradient, making the comparison cleaner but potentially overly optimistic compared to the stochastic setting of Q2.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad and RMSprop}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2                            # accumulate squared gradients
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)

def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2    # exponential moving average
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball and Nesterov Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g            # momentum accumulation
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z          # evaluate gradient at look-ahead point
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g      # 1st moment
        v_vec = beta2 * v_vec + (1 - beta2) * g**2   # 2nd moment
        m_hat = m_vec / (1 - beta1**t)               # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))         # regularise for PD
        try:
            p = np.linalg.solve(H_reg, g)            # Newton direction
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm with Box LMO}]
def fw_lp_box(grad, bounds):
    # Linear minimisation oracle: arg min_{x in X} grad^T x for box X
    z = np.zeros(len(grad))
    for i in range(len(grad)):
        z[i] = bounds[i][0] if grad[i] > 0 else bounds[i][1]
    return z

def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = fw_lp_box(g, bounds)
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z               # convex combination
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
