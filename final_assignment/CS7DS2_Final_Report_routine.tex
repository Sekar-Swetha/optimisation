\begin{abstract}
This report presents a rigorous comparative study of twelve optimisation algorithms applied to four benchmark problems spanning strongly convex quadratics, non-convex landscapes with mild anisotropy, and the highly ill-conditioned Rosenbrock function ($\kappa \approx 2508$ at the optimum). Six methodological categories are investigated: (1)~adaptive step-size methods --- Polyak, Adagrad, RMSprop, and Heavy Ball --- demonstrating that Polyak's optimality depends critically on knowledge of the true minimum value; (2)~accelerated and stochastic methods --- Nesterov momentum achieves a provably optimal $O(1/k^2)$ rate on convex functions versus GD's $O(1/k)$, while mini-batch SGD trades gradient variance for computational throughput; (3)~Newton's method, which converges in a single step on strictly quadratic objectives and exhibits near-quadratic convergence elsewhere, at the cost of $O(d^3)$ per iteration; (4)~derivative-free methods, where finite-difference accuracy depends critically on the step size $\delta$ through a $\sqrt{\varepsilon_{\rm mach}}$-optimal trade-off, and Nelder--Mead adapts to local geometry without any gradient information; (5)~constrained optimisation via projected gradient descent --- which enforces feasibility as a hard constraint from iteration one --- and quadratic penalty methods, whose conditioning worsens as the penalty weight $\lambda$ increases; and (6)~the Frank--Wolfe algorithm, which avoids projections entirely by replacing them with computationally cheap linear minimisation oracle calls, converging at $O(1/k)$ on smooth convex objectives. Empirical results consistently show that curvature information (Newton, quasi-Newton, adaptive methods) is the decisive factor for ill-conditioned problems, while first-order methods remain competitive when the Hessian condition number is moderate.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. We use four benchmarks of increasing difficulty, characterised by their spectral properties.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strongly convex quadratic with constant Hessian $H = \frac{1}{m}X^T X$. The spectral properties computed from the actual dataset are: eigenvalues $\lambda_1 = 0.9247$, $\lambda_2 = 1.0325$, giving Lipschitz constant $L = \lambda_{\max} \approx 1.033$, strong convexity modulus $\mu = \lambda_{\min} \approx 0.925$, and condition number $\kappa = L/\mu \approx 1.12$. This near-isotropic Hessian makes Benchmark A the easiest problem. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the true minimum (OLS solution) is $\hat{\theta} \approx [3.028, 4.017]^T$ with $J^{\star} \approx 0.4826$, not zero. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
\label{eq:bench_B}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$. Setting $\nabla f = 0$ yields $x_2^{\star} = 2$ and $2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$, solved numerically as $x_1^{\star} \approx 0.5824$. The Hessian is $H = \mathrm{diag}(2 - \sin(x_1),\ 10)$, with eigenvalues in the range $[1, 3] \times [10, 10]$, giving a curvature ratio of approximately $3$--$10$, creating \emph{moderate anisotropy}. The minimum is at $(x_1^{\star}, x_2^{\star}) \approx (0.5824, 2.0)$ with $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
\label{eq:bench_C}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. At the optimum, the Hessian is
\begin{equation}
H = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $\lambda_1 \approx 0.40$ and $\lambda_2 \approx 1001.6$, giving condition number $\kappa \approx 2508$. This severe ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions. Starting point: $x_0 = [-1, 1]^T$.

For constrained problems (Questions~5--6): $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\subsection{Algorithm Formulations}

Four adaptive step-size methods are compared against a constant-step gradient descent (GD) baseline across all three benchmarks for 120 iterations. The key distinction from plain GD is that each method adjusts the learning rate using information accumulated during the optimisation trajectory.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step is derived by minimising $\|x_{k+1} - x^{\star}\|^2$ over $\alpha_k$. Expanding:
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 = \|x_k - x^{\star}\|^2 - 2\alpha_k g_k^T(x_k - x^{\star}) + \alpha_k^2 \|g_k\|^2.
\end{equation}
Minimising over $\alpha_k$ gives $\alpha_k^* = g_k^T(x_k - x^{\star})/\|g_k\|^2$. By convexity, $g_k^T(x_k - x^{\star}) \geq f(x_k) - f^{\star}$, so the Polyak step $\alpha_k = (f(x_k) - f^{\star})/\|g_k\|^2$ is a conservative approximation. Substituting back yields $\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - (f(x_k) - f^{\star})^2/\|g_k\|^2$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative squared gradient}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's per-coordinate denominator $\sqrt{G_{k+1,i}}$ accumulates the sum of squared gradients for coordinate $i$. This automatically reduces the effective step for frequently-updated (large gradient) coordinates while maintaining larger steps for infrequently-updated ones. The effective step is monotonically non-increasing, which guarantees stability but may cause premature stalling.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's cumulative sum with an exponential moving average (EMA) of squared gradients (decay $\beta = 0.9$). The EMA gives more weight to recent gradients, allowing the effective step size to recover when gradients become small --- addressing Adagrad's monotonic decay issue. RMSprop can be interpreted as a diagonal preconditioning by the running second moment, approximating the curvature per coordinate.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method accumulates a momentum term $z_k$, which acts as an exponential weighted average of past gradients. The update rule is equivalent to $x_{k+1} = x_k - \alpha\nabla f(x_k) + \beta(x_k - x_{k-1})$, which damps oscillations and accelerates progress in consistent gradient directions. For strongly convex quadratics with optimal parameters $\alpha^* = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2}$ and $\beta^* = \left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^2$, Heavy Ball achieves linear convergence with rate $(\sqrt{\kappa} - 1)/(\sqrt{\kappa}+1)$, matching Chebyshev acceleration.

\subsection{Comparative Analysis and Theoretical Context}

\begin{theorem}[GD Convergence, Strongly Convex Case]
\label{thm:gd}
If $f$ is $\mu$-strongly convex and $L$-smooth, gradient descent with $\alpha = 1/L$ satisfies:
\begin{equation}
\label{eq:gd_rate}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k \bigl(f(x_0) - f^{\star}\bigr).
\end{equation}
For Benchmark~A, $\kappa = L/\mu \approx 1.12$, so GD with $\alpha = 1/L$ converges at rate $(1 - 1/1.12)^k = (0.107)^k$ --- extremely fast, reaching machine precision in $\approx 13$ iterations. The GD baseline uses a substantially smaller $\alpha = 0.08 \ll 1/L \approx 0.969$ (roughly $12\times$ below the optimal step). At this sub-optimal step, the convergence factor is $\max(|1-\alpha\mu|,|1-\alpha L|) = \max(0.926, 0.917) = 0.926$, giving a rate of $(0.926)^k$ --- far slower than the theoretical optimum of $(0.107)^k$ at $\alpha = 1/L$. To reach machine precision at $\alpha=0.08$ would require $\approx 360$ iterations, versus $\approx 13$ at the optimal step. The conservative choice ensures stability across all three benchmarks, where $L$ varies greatly.
\end{theorem}

\begin{itemize}[nosep]
\item \textbf{Polyak:} Theoretically optimal step for convex functions when $f^{\star}$ is exact. Catastrophically sensitive to misspecification of $f^{\star}$ --- a fundamental practical limitation. On Benchmark~C (where $f^{\star} = 0$ is exact), it achieves $f \approx 0.024$ in 120 iterations, outperforming all other first-order methods.
\item \textbf{Adagrad:} Accumulates curvature information via $G_k$, automatically scaling per coordinate. The formal guarantee is $O(\sqrt{T})$ cumulative regret in the online learning setting (Duchi et al.~2011): for any sequence of convex losses $\ell_1, \ldots, \ell_T$, $\sum_{t=1}^T \ell_t(x_t) - \min_x \sum_t \ell_t(x) \leq O(\|x^{\star}\|\sqrt{T \sum_j \|g_{\cdot j}\|})$. The coordinate-wise scaling via $G_k^{(j)} = \sum_{t \leq k} g_t^{(j)2}$ is optimal for \emph{sparse gradient} settings where some coordinates rarely receive non-zero gradient. The monotone step decay is a fundamental limitation for dense gradient problems.
\item \textbf{RMSprop:} Addresses Adagrad's stalling via exponential forgetting ($\beta = 0.9$). No formal convergence guarantee exists for general non-convex functions, but it is the most widely used first-order method in deep learning practice. Its effective preconditioner $\hat{G}_k^{(j)} = \beta G_{k-1}^{(j)} + (1-\beta)(g_k^{(j)})^2$ approximates the diagonal of the \emph{Fisher information matrix} in the neighbourhood of the current iterate, connecting adaptive methods to natural gradient descent.
\item \textbf{Heavy Ball:} For strongly convex quadratics with Lipschitz constant $L$ and strong convexity $\mu$, the optimal parameters $\alpha^* = 4/(\sqrt{L}+\sqrt{\mu})^2$ and $\beta^* = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$ achieve convergence rate $\rho^* = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$ --- the Chebyshev-optimal rate. For non-quadratic problems (Rosenbrock), global convergence is not guaranteed. The connection to Polyak's heavy ball and Nesterov momentum can be understood via the \emph{triple momentum method} (Van Scoy et al.~2018), which achieves the optimal convergence rate for strongly convex functions by interpolating between the two.
\end{itemize}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Look-ahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{look-ahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This ``corrects'' the trajectory before overshooting, enabling the following optimal convergence guarantee.

\begin{theorem}[Nesterov Acceleration Rate]
\label{thm:nesterov}
For a convex, $L$-smooth function $f$, the Nesterov method with $\alpha = 1/L$ and $\beta_k = (k-1)/(k+2)$ satisfies:
\begin{equation}
\label{eq:nesterov_rate}
f(x_k) - f^{\star} \leq \frac{2L\,\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\end{equation}
This $O(1/k^2)$ rate is optimal among first-order methods for smooth convex optimisation (Nesterov 1983).
\end{theorem}

By contrast, GD achieves only $O(1/k)$ (Theorem~\ref{thm:gd}). For Benchmark~A with $L \approx 1.033$, $\|x_0 - x^{\star}\|^2 \approx \|[3.028, 4.017]\|^2 \approx 25.3$, the bound of Theorem~\ref{thm:nesterov} gives convergence to within $10^{-4}$ in $k \gtrsim \sqrt{2 \times 1.033 \times 25.3 / 10^{-4}} \approx 723$ iterations. However, strong convexity (present in Benchmark~A) allows the practical convergence to be much faster. For $\mu$-strongly convex, $L$-smooth objectives, Nesterov with a modified momentum $\beta_k = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$ (constant) achieves the linear rate:
\begin{equation}
f(x_k) - f^{\star} \leq \left(1 - \sqrt{\frac{\mu}{L}}\right)^k (f(x_0) - f^{\star}),
\end{equation}
which is the minimax-optimal first-order rate for strongly convex problems, matching the Heavy Ball Chebyshev rate (Theorem~\ref{thm:gd}). For Benchmark~A, $\sqrt{\mu/L} = \sqrt{1/1.12} \approx 0.945$, giving rate $(1-0.945)^k = (0.055)^k$ vs GD's $(0.107)^k$. The speedup is by a factor of $\sqrt{\kappa} \approx 1.06$ in iteration count --- small for this near-isotropic problem, but growing to $\sqrt{\kappa} \approx 50\times$ for Benchmark~C ($\kappa \approx 2508$).

\textbf{Interpretation of the look-ahead:} The Nesterov update evaluates the gradient at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This can be understood as: rather than computing the gradient at the \emph{current position}, compute it at the \emph{expected next position} (if we continued moving with current momentum). This anticipatory step prevents overshooting and is the mechanism by which Nesterov achieves the $O(1/k^2)$ rate that plain gradient descent cannot.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines RMSprop's per-coordinate adaptive scaling (via $v_t$) with a momentum term (via $m_t$). The effective update is approximately:
\begin{equation}
x_{t+1} \approx x_t - \frac{\alpha}{\sqrt{\hat{v}_t} + \varepsilon} \hat{m}_t,
\end{equation}
where $\hat{m}_t$ is an exponential moving average of gradients (momentum) and $1/\sqrt{\hat{v}_t}$ is an exponential moving average of inverse gradient RMS (adaptive scaling). The bias corrections $(1-\beta_1^t)^{-1}$ and $(1-\beta_2^t)^{-1}$ are critical in early iterations when $m_t$ and $v_t$ are initialised to zero: without correction, the effective step size is $\alpha(1-\beta_1^t)/\sqrt{1-\beta_2^t}$, which for $\beta_2 = 0.999$ can significantly deviate from the intended $\alpha$. The bias corrections normalise both moving averages to ensure the effective step remains close to $\alpha$ regardless of $t$. With $\beta_2 = 0.999$, the second moment adapts slowly, providing stable curvature estimates, while $\beta_1 = 0.8$--$0.9$ gives moderate momentum. With standard parameters $\beta_1 = 0.9$, $\beta_2 = 0.999$, $\varepsilon = 10^{-8}$, Adam is widely used in deep learning due to its robustness to hyperparameter choice and its ability to handle sparse gradients effectively.

\begin{remark}[Adam convergence caveats]
Adam does \emph{not} guarantee convergence for general non-convex functions with constant $\alpha$. Reddi et al.~(2018) showed that Adam can diverge on simple convex problems due to the interaction between the decaying effective step size and the EMA second moment. The \emph{AMSGrad} fix uses $\hat{v}_t = \max(\hat{v}_{t-1}, v_t/( 1-\beta_2^t))$, ensuring the effective step is non-increasing. For strongly convex objectives with sufficiently decaying $\alpha_t$, Adam achieves $O(1/\sqrt{T})$ regret (matching SGD), but the practical advantage lies in faster \emph{initial} progress and robustness to learning rate tuning, not in asymptotic rate improvement over SGD.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs with data shuffled every epoch. The gradient estimate for mini-batch $\mathcal{B}$ is:
\begin{equation}
\hat{g}_{\mathcal{B}}(\theta) = \frac{1}{|\mathcal{B}|} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}}),
\end{equation}
which is an unbiased estimator of $\nabla J(\theta)$ with variance $O(1/b)$.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{Gradient Noise as a Function of Batch Size}

The SGD gradient estimator $\hat{g}_{\mathcal{B}} = \frac{1}{b}\sum_{i \in \mathcal{B}} \nabla \ell_i(\theta)$ is unbiased ($\mathbb{E}[\hat{g}_{\mathcal{B}}] = \nabla J(\theta)$) with variance scaling as $O(1/b)$. More precisely, for a loss that decomposes as $J = \frac{1}{m}\sum_{i=1}^{m}\ell_i$:
\begin{equation}
\mathrm{Var}[\hat{g}_{\mathcal{B}}] = \frac{1}{b}\sigma^2_{\rm sample}, \qquad \sigma^2_{\rm sample} = \frac{1}{m}\sum_{i=1}^{m}\|\nabla \ell_i(\theta) - \nabla J(\theta)\|^2.
\end{equation}
At the optimum $\theta^{\star}$, $\nabla J(\theta^{\star}) = 0$, so $\sigma^2_{\rm sample} = \frac{1}{m}\sum_{i=1}^{m}\|\nabla \ell_i(\theta^{\star})\|^2$ measures the inherent data heterogeneity. SGD near the optimum effectively performs a random walk with step variance $\alpha^2 \sigma^2_{\rm sample}/b$, which determines the oscillation floor.

\textbf{Variance reduction:} The fundamental bottleneck of SGD is this irreducible variance floor. \emph{Variance reduction} methods (SVRG, SAG, SAGA) eliminate this floor by periodically computing the full gradient and using it to correct subsequent stochastic estimates. SVRG (Johnson \& Zhang, 2013) uses gradient updates of the form $\hat{g}_k = \nabla \ell_{i_k}(\theta_k) - \nabla \ell_{i_k}(\tilde{\theta}) + \nabla J(\tilde{\theta})$, where $\tilde{\theta}$ is a periodically-updated reference point. This gives $\mathbb{E}[\hat{g}_k] = \nabla J(\theta_k)$ with variance that vanishes at the optimum, enabling \emph{linear convergence} $O(\rho^k)$ with $\rho < 1$ for strongly convex objectives --- matching full GD convergence but at $O(1)$ cost per step instead of $O(m)$. For our small-scale experiments, batch sizes up to $m = 1000$ are feasible; the variance reduction benefit would become critical for large-scale problems with $m \gg 10^4$.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton step minimises $g_2(x)$: setting $g_2'(x) = 0$ gives
\begin{equation}
x^{\rm N} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{4(0.25)^3}{12(0.25)^2} = 0.25 - \frac{1}{3} \approx -0.083.
\end{equation}
Since $g(x) = x^4$ has its minimum at $x = 0$, and $g''(x) = 12x^2 > 0$ only for $x \neq 0$, Newton's method from $x_0 = 0.25$ will converge to $x^{\star} = 0$ with quadratic convergence rate once sufficiently close to the optimum.

\subsection{Newton's Method with Damping}

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The regularisation $\lambda I$ with $\lambda = 10^{-8}$ serves two purposes: (i) it ensures the Hessian is positive definite even when the true Hessian has near-zero eigenvalues, and (ii) it provides numerical stability when solving $H_k p_k = g_k$ via Cholesky or LU factorisation. The computational cost per Newton step is $O(d^3)$ (Hessian construction and factorisation), compared to $O(d)$ for GD --- a critical distinction in high dimensions.

\begin{theorem}[Quadratic Convergence of Newton's Method]
\label{thm:newton}
Let $f$ be twice continuously differentiable with Lipschitz Hessian. If $x_0$ is sufficiently close to a non-degenerate local minimum $x^{\star}$ (where $\nabla^2 f(x^{\star}) \succ 0$), then Newton's method with $\alpha = 1$ satisfies:
\begin{equation}
\label{eq:newton_rate}
\|x_{k+1} - x^{\star}\| \leq C\, \|x_k - x^{\star}\|^2,
\end{equation}
for some constant $C > 0$ depending on $\nabla^2 f(x^{\star})$ and the Hessian Lipschitz constant. This quadratic convergence means the number of correct digits approximately doubles each iteration.
\end{theorem}

For quadratic $f$ (Benchmark~A), the convergence is exact in a single step with $\alpha = 1$. The quadratic convergence constant $C$ depends on the ratio $\|\nabla^3 f\| / \lambda_{\min}(\nabla^2 f(x^{\star}))$: functions with large third-order derivatives relative to the minimum curvature converge more slowly.

\begin{remark}[Failure modes of Newton's method]
Newton's method can fail if: (1) the Hessian is not positive definite (e.g., at a saddle point), causing $H_k^{-1} g_k$ to point in an ascent direction; (2) the initial point is outside the basin of attraction where the Hessian Lipschitz condition ensures quadratic convergence; or (3) the Hessian becomes ill-conditioned, making the linear solve $H_k p = g_k$ numerically unstable. The Tikhonov regularisation $H_k \leftarrow H_k + \lambda I$ with $\lambda = 10^{-8}$ addresses case (3) but adds a small bias to the Newton direction.
\end{remark}

\noindent\textbf{Damped Newton on Rosenbrock ($\alpha = 0.22$):} A full Newton step ($\alpha = 1$) is unsuitable for Rosenbrock because the Hessian changes rapidly along the trajectory --- the Hessian Lipschitz condition is violated globally, so the quadratic model is only accurate in a small neighbourhood. \noindent\textbf{Connection to quasi-Newton methods:} Full Newton requires forming and factorising $\nabla^2 f$ at cost $O(d^2)$ (formation) + $O(d^3)$ (solve) per iteration. For large $d$, BFGS maintains a low-rank approximation $B_k \approx \nabla^2 f(x_k)$ using the secant condition $B_{k+1}(x_{k+1}-x_k) = \nabla f(x_{k+1}) - \nabla f(x_k)$, updated with the rank-2 BFGS formula. The L-BFGS variant stores only the last $m$ secant pairs (typically $m = 10$--$20$), giving $O(md)$ memory and $O(md)$ per-iteration cost while retaining superlinear convergence. For our $d=2$ experiments, Newton's exact Hessian is always used; BFGS becomes essential for $d \geq 10^3$.

\subsection{Part A: Finite Difference Gradient Approximation}

\subsubsection{Theory of Finite Differences}

The forward finite-difference (FD) approximation of the $i$-th partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
A Taylor expansion of $f(x + \delta e_i)$ around $x$ yields:
\begin{equation}
\frac{f(x + \delta e_i) - f(x)}{\delta} = \frac{\partial f}{\partial x_i}(x) + \underbrace{\frac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x)}_{\text{truncation error}} + O(\delta^2).
\end{equation}
In floating-point arithmetic, each function evaluation incurs a rounding error $\sim \varepsilon_{\rm mach} |f(x)|$, so the total error bound is approximately:
\begin{equation}
\label{eq:fd_error}
\left\|\hat{g} - \nabla f\right\| \lesssim \underbrace{\frac{M_2 \delta}{2}}_{\rm truncation} + \underbrace{\frac{2\varepsilon_{\rm mach}\,|f(x)|}{\delta}}_{\rm rounding},
\end{equation}
where $M_2 = \max_i \left|\frac{\partial^2 f}{\partial x_i^2}\right|$ is a second-derivative bound. The optimal trade-off minimises the sum of truncation and rounding errors, yielding:
\begin{equation}
\delta^{\star} = \sqrt{\frac{2\varepsilon_{\rm mach}\,|f(x)|}{M_2}} \approx \sqrt{\varepsilon_{\rm mach}} \approx 10^{-8} \quad (\text{for } \varepsilon_{\rm mach} \approx 10^{-16}).
\end{equation}
This makes \emph{central differences} ($O(\delta^2)$ truncation error rather than $O(\delta)$) preferable when extra function evaluations are affordable: $\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x+\delta e_i) - f(x-\delta e_i)}{2\delta}$, which shifts the optimal $\delta_{\rm CD}^* = (3\varepsilon_{\rm mach}|f|/M_3)^{1/3} \approx 10^{-5.3}$ and achieves approximately $10 \times$ lower minimum error than forward differences. The cost is $2d$ evaluations per gradient instead of $d+1$, which is negligible in the small-$d$ setting of our experiments.

The ``U-shaped'' curve reflects equation~\eqref{eq:fd_error}: truncation error dominates for large $\delta$ (slope $+1$ on log-log scale), while rounding error dominates for small $\delta$ (slope $-1$). The minimum error occurs at $\delta \approx \sqrt{\varepsilon_{\rm mach}} \approx 10^{-8}$. \subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates a directional derivative along a uniformly random direction $u_k \in \mathcal{S}^{d-1}$. The estimator's expectation follows from the identity $\mathbb{E}[uu^T] = \frac{1}{d}I_d$ for uniform $u \in \mathcal{S}^{d-1}$: by Taylor expansion, $f(x+\delta u) = f(x) + \delta u^T \nabla f(x) + O(\delta^2)$, so $\hat{g}_k = \frac{f(x+\delta u)-f(x)}{\delta} u \approx (u^T\nabla f)u$ and $\mathbb{E}[\hat{g}_k] = \mathbb{E}[(u^T\nabla f)u] = \mathbb{E}[uu^T]\nabla f = \frac{1}{d}\nabla f(x_k) + O(\delta^2)$. Thus $\hat{g}_k$ is a biased estimate scaled by $1/d$. Consequently, the method requires $O(d)$ more iterations than full-gradient methods to achieve the same accuracy. More precisely, if $f$ is $\mu$-strongly convex and $L$-smooth:
\begin{equation}
\mathbb{E}[f(x_k) - f^{\star}] \leq \left(1 - \frac{\mu\alpha}{d}\right)^k (f(x_0) - f^{\star}) + \frac{L\delta^2\alpha d}{2},
\end{equation}
showing an effective convergence rate of $O(1 - \mu\alpha/d)$ compared to GD's $O(1 - \mu\alpha)$. The method uses only \emph{two} function evaluations per iteration, making it appropriate when gradient computation is expensive or unavailable (black-box oracle setting).

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations: \textbf{reflection} ($\alpha_r = 1$): move worst vertex through the centroid of the rest; \textbf{expansion} ($\gamma = 2$): if reflection improves, try further; \textbf{contraction} ($\rho = 0.5$): if reflection worsens, contract towards centroid; \textbf{shrink} ($\sigma = 0.5$): contract all vertices towards best if contraction fails.

No gradient or Hessian is required. The simplex adapts geometrically to the local landscape --- elongating along valleys and contracting near optima. Settings: initial simplex step $= 0.35$, iterations $= 160$.

Grid search is the only method that provides a \emph{global} optimality guarantee (to within grid resolution), but its $O(N^d)$ cost makes it computationally intractable beyond $d \approx 3$ for any reasonable $N$. For $d = 10$, the same resolution would require $55^{10} \approx 2.5 \times 10^{17}$ evaluations, completely infeasible.

Nelder--Mead is well-suited to low-dimensional problems ($d \leq 10$--$20$) where gradient information is expensive or unavailable. Its adaptive simplex avoids the curse of dimensionality of grid search while being more structured than random search. The key limitation is that convergence guarantees are weak --- Nelder--Mead can stagnate on non-smooth or high-dimensional functions.

Nesterov random search is the preferred choice in high dimensions with only function evaluation access, as it costs only 2 function evaluations per iteration independent of $d$, and can leverage the problem's smoothness through its derivative estimate. However, it requires $O(d)$ more iterations than exact gradient descent to achieve the same accuracy, making it inefficient when gradient computation is feasible.

\textbf{Beyond the methods studied:} For black-box optimisation in higher dimensions, two further families are important: (1)~\emph{Evolution Strategies} such as CMA-ES (Covariance Matrix Adaptation) maintain a multivariate Gaussian $\mathcal{N}(\mu_k, C_k)$ over candidate solutions, adapting the full covariance to capture correlations in the landscape. CMA-ES is scale-invariant and typically the state-of-the-art for $d \leq 100$ with expensive black-box objectives ($\leq 10^4$ evaluations). It achieves superlinear convergence on strongly convex smooth objectives. (2)~\emph{Bayesian Optimisation} (BO) fits a Gaussian Process (GP) surrogate to the function, then uses an acquisition function (e.g., Expected Improvement) to select evaluation points. BO excels for $d \leq 20$ with very expensive objectives ($\leq 10^3$ evaluations). Unlike the methods in this report, both CMA-ES and BO can escape local optima by maintaining population or probabilistic diversity.

\subsection{Problem Setup}

Minimise $f(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)$ (Benchmark~B) subject to the half-space constraint $x_1 \geq 0.5$. The unconstrained minimum of Benchmark~B lies at $x_1^{\star} \approx 0.5824 > 0.5$, so the constraint is \emph{inactive} at the solution: $x_{\rm constrained}^{\star} = x_{\rm unconstrained}^{\star} \approx (0.5824, 2.0)$ with $f^{\star} \approx 0.7244$. The starting point $x_0 = (0.2, 4.0)$ is \emph{infeasible}: $x_0^{(1)} = 0.2 < 0.5$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{$\Pi_{\mathcal{X}}(x) = [\max(0.5, x_1),\ x_2]^T$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space $\{x : x_1 \geq 0.5\}$, the projection is simply $\Pi_{\mathcal{X}}(x) = (\max(0.5, x_1),\ x_2)^T$, requiring $O(1)$ operations. The convergence theory for PGD mirrors Theorem~\ref{thm:gd}:

\begin{theorem}[PGD Convergence]
\label{thm:pgd}
Let $f$ be $\mu$-strongly convex and $L$-smooth. Then projected gradient descent with $\alpha = 1/L$ satisfies:
\begin{equation}
\label{eq:pgd_rate}
\|x_{k+1} - x^{\star}\|^2 \leq \left(1 - \frac{\mu}{L}\right)^k \|x_0 - x^{\star}\|^2,
\end{equation}
where $x^{\star} = \Pi_{\mathcal{X}}(x^{\star}_{\rm unconstrained})$ is the constrained optimum.
\end{theorem}

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda\, \max(0,\, 0.5 - x_1)$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty gradient is $\nabla F_\lambda(x) = \nabla f(x) - \lambda \mathbf{1}[x_1 < 0.5]\, e_1$, where $e_1 = (1,0)^T$. Note that this is a linear (not quadratic) penalty; a quadratic penalty would use $\lambda \max(0, 0.5 - x_1)^2$ and have a smooth gradient. The linear penalty has a kink at the constraint boundary $x_1 = 0.5$, which can cause the algorithm to oscillate when iterates cross the boundary.

\textbf{Effect of $\lambda$ on Lipschitz constant:} For the penalty objective $F_\lambda$, the gradient $\nabla F_\lambda(x)$ has a discontinuity at $x_1 = 0.5$ of magnitude $\lambda$ in the $e_1$ direction. This means the effective Lipschitz constant of $F_\lambda$ near the constraint is at least $\lambda$, requiring step size $\alpha \leq 2/(\lambda + L_f)$ for stability, where $L_f \approx 10$ is the Lipschitz constant of $\nabla f$. For $\lambda = 4.5$, this gives $\alpha \leq 2/14.5 \approx 0.138$; the used $\alpha = 0.03$ is conservative and stable.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q5_penalty_conditioning.pdf}
\caption{Q5 supplementary: Effect of penalty weight $\lambda$ on problem conditioning. \textbf{Left:} Effective Lipschitz constant $L_\text{eff} = L_f + \lambda$ (left $y$-axis, blue) and maximum safe step size $\alpha_\text{max} = 2/L_\text{eff}$ (right $y$-axis, green) as functions of $\lambda$. As $\lambda$ increases to enforce the constraint more strictly, the problem becomes harder to optimise: the safe step size shrinks from $\approx 0.18$ (unconstrained) to $\approx 0.14$ at $\lambda = 4.5$ and $\approx 0.023$ at $\lambda = 100$. \subsection{Hyperparameter Settings}

PGD uses $\alpha = 0.08$. Penalty method: $\lambda \in \{0.15, 1.8, 4.5\}$, with $\alpha = 0.05$ for $\lambda \leq 1.8$ and $\alpha = 0.03$ for $\lambda = 4.5$ (smaller step needed for stability as $\lambda$ increases the effective Lipschitz constant). Unconstrained GD uses $\alpha = 0.07$. All methods run for 100 iterations. The Benchmark~B Lipschitz constant on the compact domain can be estimated: $L = \max_{x} \lambda_{\max}(\nabla^2 f(x)) \leq 10$ (the largest eigenvalue of the diagonal Hessian).

\begin{remark}[Constraint Activity]
Since the unconstrained minimum of Benchmark~B lies at $x_1^{\star} \approx 0.5824 > 0.5$, the constraint $x_1 \geq 0.5$ is \emph{inactive} at the constrained solution. This is why the unconstrained GD (starting from $x_0 = (0.2, 4.0)$, which is infeasible) converges to the same point $(0.582, 2.0)$ as the PGD, even though it passes through the infeasible region. Had the constraint been $x_1 \geq 0.7$ (greater than the unconstrained optimum), the constrained and unconstrained solutions would differ, making the constraint \emph{active} and genuinely binding.
\end{remark}

\textbf{Key distinction:}

\textbf{Projected GD} enforces feasibility as a \emph{hard constraint} at every iteration via the projection operator $\Pi_{\mathcal{X}}$. The method guarantees: (i) all iterates are feasible, and (ii) convergence to the constrained optimum. The cost is a projection oracle --- for complex constraint sets this can be expensive, but for box/half-space constraints it is $O(d)$.

\textbf{Penalty methods} solve an unconstrained surrogate $F_\lambda(x) = f(x) + \lambda p(x)$ where $p(x) \geq 0$ measures constraint violation. As $\lambda \to \infty$, the penalised solution converges to the constrained optimum, but the penalised problem becomes increasingly ill-conditioned: the effective Lipschitz constant of $F_\lambda$ grows as $\lambda$, requiring smaller step sizes (and more iterations) to maintain stability. This is the fundamental trade-off of pure penalty methods.

\begin{remark}[Augmented Lagrangian Methods]
The ill-conditioning of penalty methods as $\lambda \to \infty$ motivates \emph{augmented Lagrangian} (AL) methods, which add a Lagrange multiplier $\nu$ alongside the quadratic penalty:
\begin{equation}
\mathcal{L}_\lambda(x, \nu) = f(x) - \nu \cdot (x_1 - 0.5) + \tfrac{\lambda}{2}\max(0,\, 0.5 - x_1)^2.
\end{equation}
The dual variable $\nu$ is updated by a gradient step on the dual problem after each primal minimisation, allowing exact constraint satisfaction to be achieved with finite $\lambda$. ADMM (Alternating Direction Method of Multipliers) is a widely used variant that handles separable constraints via alternating minimisation of $\mathcal{L}_\lambda$ over the primal variables.
\end{remark}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box/polytope with 4 vertices).

\textbf{Linear Programme:} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$.

Since $f$ is linear, its minimum over the compact convex set $\mathcal{X}$ is attained at a \emph{vertex}. Since both components of $a$ are positive ($a_1 = 1 > 0$, $a_2 = 2 > 0$), we minimise $f$ by selecting the minimum value of each coordinate independently:
\begin{equation}
x^{\star} = \arg\min_{x \in \mathcal{X}} (x_1 + 2x_2) = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}

This illustrates the \textbf{fundamental theorem of linear programming}: the optimum of a linear objective over a bounded polytope $\mathcal{X} = \{x : Ax \leq b\}$ is always attained at a vertex (extreme point) of $\mathcal{X}$.

\begin{proposition}[LP Optimum at Vertex]
Let $\mathcal{X} \subset \mathbb{R}^d$ be a bounded polytope and $f(x) = c^T x$ a linear objective. Any optimal solution $x^{\star}$ is an extreme point (vertex) of $\mathcal{X}$.
\end{proposition}
\begin{proof}[Proof sketch]
Suppose $x^{\star}$ is optimal but not a vertex. Then $x^{\star} = \lambda x_1 + (1-\lambda) x_2$ for some $x_1, x_2 \in \mathcal{X}$, $\lambda \in (0,1)$. By linearity, $f(x^{\star}) = \lambda f(x_1) + (1-\lambda) f(x_2)$. If $f(x^{\star}) \leq f(x)$ for all $x \in \mathcal{X}$, then $f(x_1) = f(x_2) = f(x^{\star})$ (otherwise one would be strictly better). Continuing this argument, any optimal point in the relative interior of a face can be replaced by an optimal vertex.
\end{proof}

For our box $\mathcal{X}$, the four vertices are $(0.5, -5)$, $(5, -5)$, $(0.5, 10)$, $(5, 10)$ with objective values $-9.5$, $-5$, $20.5$, and $25$ respectively. The minimum $-9.5$ is at $(0.5, -5)$.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial $x_0 \in \mathcal{X}$, fixed mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination, stays in $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Key advantage:} Unlike projected gradient descent, Frank--Wolfe never requires a projection onto $\mathcal{X}$. Instead, at each step it solves an \emph{easier} subproblem --- minimising a linear function over $\mathcal{X}$. For a box $\mathcal{X}$, the LMO decomposes per coordinate:
\begin{equation}
\label{eq:fw_lmo}
(z_k)_i = \begin{cases} l_i & \text{if } \bigl(\nabla f(x_k)\bigr)_i > 0 \\ u_i & \text{if } \bigl(\nabla f(x_k)\bigr)_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations. The update $x_{k+1} = \beta x_k + (1-\beta) z_k$ is a convex combination, so if $x_k, z_k \in \mathcal{X}$ then $x_{k+1} \in \mathcal{X}$ by convexity --- feasibility is maintained without any projection.

\begin{theorem}[Frank--Wolfe Convergence Rate]
\label{thm:fw}
Let $f$ be convex and $L$-smooth on a convex set $\mathcal{X}$ with diameter $D = \max_{x,y \in \mathcal{X}} \|x - y\|$. With step sizes $\gamma_k = 2/(k+2)$, the Frank--Wolfe iterates satisfy:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2} = O\!\left(\frac{1}{k}\right).
\end{equation}
For our box $\mathcal{X}$, $D = \sqrt{(5-0.5)^2 + (10-(-5))^2} = \sqrt{20.25 + 225} \approx 15.67$.
\end{theorem}

\begin{proof}[Proof sketch]
By $L$-smoothness with $x_{k+1} = x_k + \gamma_k(z_k - x_k)$:
$f(x_{k+1}) \leq f(x_k) + \gamma_k \nabla f(x_k)^T(z_k-x_k) + \frac{L\gamma_k^2}{2}\|z_k-x_k\|^2 \leq f(x_k) + \gamma_k \nabla f(x_k)^T(z_k-x_k) + \frac{L\gamma_k^2 D^2}{2}$.
Since $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$, we have $\nabla f(x_k)^T z_k \leq \nabla f(x_k)^T x^{\star}$. By convexity, $\nabla f(x_k)^T(x^{\star}-x_k) \leq f^{\star}-f(x_k)$, so $\nabla f(x_k)^T(z_k-x_k) \leq f^{\star}-f(x_k)$. Setting $h_k = f(x_k) - f^{\star}$:
\[h_{k+1} \leq (1-\gamma_k)h_k + \tfrac{L\gamma_k^2 D^2}{2}.\]
Choosing $\gamma_k = 2/(k+2)$ and applying induction yields $h_k \leq 2LD^2/(k+2)$.
\end{proof}

\noindent The implementation uses a \emph{fixed} $\beta$ (equivalent to $\gamma_k = 1 - \beta$ constant), which does not achieve the $O(1/k)$ rate but is simpler to tune.

\textbf{FW gap:} A natural convergence certificate is the Frank--Wolfe duality gap:
\begin{equation}
g(x_k) = \nabla f(x_k)^T (x_k - z_k) \geq f(x_k) - f^{\star} \geq 0,
\end{equation}
which upper bounds the suboptimality at each iterate and can be monitored without knowing $f^{\star}$.

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits characteristic ``zig-zag'' oscillation because $z_k$ jumps between extreme points of $\mathcal{X}$ (whose location is determined by the sign of $\nabla f(x_k)$). The convex combination $\beta x_k + (1-\beta)z_k$ gradually averages these jumps towards the interior minimum. With fixed step $\gamma = 1-\beta$, the recurrence $h_{k+1} \leq \beta h_k + \tfrac{L\gamma^2 D^2}{2}$ converges to an irreducible floor $h_\infty = \tfrac{L\gamma D^2}{2} = \tfrac{L(1-\beta)D^2}{2}$, so smaller $\gamma$ (larger $\beta$) gives a lower floor but slower geometric descent to it. The optimal $O(1/k)$ rate (achieved without this floor) requires the decaying schedule $\gamma_k = 2/(k+2)$ (as shown empirically in Fig.~\ref{fig:q6_fw_analysis}).

\textbf{Boundary optimum:} Frank--Wolfe is particularly natural for boundary optima. The LP subproblem repeatedly selects the vertex $(0.5, -5)$ or $(0.5, 10)$ (depending on the sign of $(\nabla f)_2$), and the iterates are progressively pulled toward $x_1 = 0.5$. The fixed $\gamma = 0.07$ schedule does not converge to the exact constrained optimum $(0.5, 0)$ in 140 iterations; a decaying schedule $\gamma_k = 2/(k+2)$ would give $O(1/k)$ convergence by Theorem~\ref{thm:fw}. The FW duality gap $g(x_k)$ (Fig.~\ref{fig:q6_fw_analysis}) provides a convergence certificate computable at each iteration without knowledge of $f^{\star}$.

\noindent\textbf{Frank--Wolfe vs Projected Gradient:} For our box constraint, projection is $O(d)$ (component-wise clipping) and the LMO is also $O(d)$ (component-wise argmin), so both methods have the same per-iteration cost. Frank--Wolfe's advantage appears for constraints where projection is expensive: for example, on the nuclear norm ball $\{X : \|X\|_* \leq r\}$, projection requires a full SVD ($O(d^3)$), while the LMO (largest singular vector) costs only $O(d^2)$. In such settings, Frank--Wolfe is the preferred method for large-scale structured problems.

\textbf{Improving beyond $O(1/k)$:} Standard FW achieves $O(1/k)$ on smooth convex problems (Theorem~\ref{thm:fw}), which cannot be improved in general. However, for strongly convex $f$ on polytopes, the \emph{pairwise FW} (Lacoste-Julien \& Jaggi, 2015) and \emph{away-step FW} variants achieve \emph{linear} convergence $O((1-\delta)^k)$ for some $\delta > 0$ depending on the geometric properties of $\mathcal{X}$ (the pyramidal width). Away steps select a vertex $s_k = \arg\max_{s \in \text{active}}\, \nabla f(x_k)^T s$ and move \emph{away} from it, allowing the algorithm to drop vertices from the active set and escape the zig-zag pattern that causes the $O(1/k)$ bound. For our box constraint experiments, the fixed $\beta$ schedule is equivalent to a standard FW step without away-steps, explaining the slow boundary convergence observed.

\subsection{Theoretical Insights}

\paragraph{Condition number is the primary difficulty.}
All algorithms studied show dramatically different behaviour on Benchmark~A ($\kappa \approx 1.12$) versus Benchmark~C ($\kappa \approx 2508$). By Theorem~\ref{thm:gd}, GD with optimal step converges at rate $(1-\mu/L) = (1-1/\kappa)$ per iteration, equal to $0.107$ for Benchmark~A (fast) but $0.9996$ for Benchmark~C (essentially no progress per step). This motivates preconditioning (Newton, quasi-Newton) or curvature-adaptive methods (Adam, Adagrad) as the first line of defence against ill-conditioning.

\paragraph{Adaptive vs accelerated methods.}
Adaptive methods (Adagrad, RMSprop, Adam) address anisotropy by independently scaling each coordinate using second-moment estimates. This is effective when the Hessian is approximately diagonal (as in Benchmarks~A and~B) but less effective when off-diagonal coupling is strong (as in Rosenbrock, where $H_{12} = -400x_1$ can be large). Nesterov acceleration, by contrast, achieves the optimal $O(1/k^2)$ rate for convex problems by exploiting momentum regardless of Hessian structure.

\paragraph{The price of derivative-free methods.}
As shown in Table~\ref{tab:q4_comparison}, moving from full gradient to zeroth-order information incurs a cost of $O(d)$ additional function evaluations per effective gradient step (for FD gradient) or $O(d)$ more iterations (for random search). On non-convex problems (Rosenbrock, Benchmark~C), Nelder--Mead's geometry-adaptive simplex often outperforms random search by adaptively elongating along the valley.

\paragraph{Constraint handling philosophy.}
Projected GD and Frank--Wolfe represent two fundamentally different philosophies for constrained optimisation. PGD uses the projection oracle to enforce feasibility \emph{explicitly} at each step; Frank--Wolfe avoids projections entirely by replacing them with a linear minimisation subproblem. For problems where projection is expensive but the constraint set is a polytope (e.g., a simplex), Frank--Wolfe's $O(d)$ LMO cost per iteration may be substantially cheaper than the $O(d^2)$--$O(d^3)$ cost of computing a general projection.

\paragraph{Lower bounds and optimality gaps.}
A key question is whether the observed rates are optimal. The classical Nesterov lower bound for smooth convex optimisation states that any first-order method requires at least $\Omega(1/\sqrt{\varepsilon})$ gradient evaluations to achieve $f(x_k) - f^{\star} \leq \varepsilon$. This corresponds to $O(1/k^2)$ in iteration count, matching Theorem~\ref{thm:nesterov} --- so Nesterov's method is \emph{minimax optimal} among first-order methods on smooth convex problems. For strongly convex $f$ (modulus $\mu > 0$), the optimal rate improves to $O((1-\sqrt{\mu/L})^k)$, again matched by Nesterov with optimal momentum. Newton's method achieves quadratic convergence (Theorem~\ref{thm:newton}), which is beyond the reach of any method that accesses only zeroth- or first-order oracle information, at the cost of higher per-iteration computation. These complexity-theoretic results formally justify the empirical ordering we observe: Newton $>$ Nesterov $>$ GD on convex problems with $\kappa > 1$.

\subsection{Practical Recommendations}

\begin{enumerate}[nosep]
\item \textbf{Start with Nesterov or Adam} for smooth, unconstrained problems with moderate dimension. Adam is more robust to hyperparameter choice; Nesterov has stronger theoretical guarantees for convex objectives.
\item \textbf{Use Newton or quasi-Newton} (BFGS/L-BFGS) when the Hessian condition number is large and $d \leq 10^3$. The cost of curvature information is amortised over dramatically fewer iterations.
\item \textbf{Polyak step requires exact $f^{\star}$.} In practice, use the Polyak-step heuristic only when the true minimum is known (e.g., feasibility problems), or use an estimate updated online.
\item \textbf{For constrained problems:} use PGD for simple constraint sets; Frank--Wolfe for polytopes where projection is expensive; augmented Lagrangian / ADMM for complex constraints where both projection and LMO are expensive.
\item \textbf{Derivative-free:} use Nelder--Mead for $d \leq 10$ (low-dimensional black-box); Nesterov random search for $d > 10$ with only function evaluation access.
\item \textbf{SGD convergence:} to avoid the oscillation floor of constant-$\alpha$ SGD, use a decaying schedule $\alpha_k = \alpha_0 / (1 + \gamma k)$ (Robbins--Monro conditions: $\sum \alpha_k = \infty$, $\sum \alpha_k^2 < \infty$), or switch to variance reduction (SVRG) for strongly convex problems to achieve linear convergence.
\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
