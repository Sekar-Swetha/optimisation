\begin{abstract}
This report presents a comprehensive comparative study of six categories of optimisation algorithms applied to three canonical benchmark problems: a quadratic linear regression loss ($\kappa \approx 1.1$), a nonlinear toy neural network loss with sinusoidal perturbation ($\kappa \approx 5.6$), and the ill-conditioned Rosenbrock function ($\kappa \approx 2500$). The methods investigated span the full convergence rate hierarchy: (1) adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball) achieving $O(1/k)$ or linear rates; (2) momentum-based accelerated methods (Nesterov $O(1/k^2)$, Adam) achieving oracle-optimal acceleration on smooth convex problems; (3) Newton's method achieving quadratic convergence and its interpretation as Sequential Quadratic Programming; (4) derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead simplex, grid search) for gradient-unavailable settings; (5) constrained optimisation via projected gradient descent (exact feasibility at every step, admits FISTA acceleration) and quadratic penalty methods (soft enforcement, $O(1/\lambda)$ gap); and (6) the Frank--Wolfe conditional gradient algorithm achieving $O(1/k)$ convergence projection-free. Key findings: Polyak diverges when $f^\star$ is misspecified as the step becomes $O(1/\|\nabla f\|^2)$ while $f(x_k) - f^\star \to f^\star_{\rm true} - \tilde{f}^\star > 0$; Heavy Ball's effective step size is $\alpha/(1-\beta) = 10\alpha$ for $\beta=0.9$, explaining its accelerated convergence; Adagrad and RMSprop do not outperform constant step-size GD on ill-conditioned Rosenbrock despite coordinate-wise scaling; Nesterov's theoretical $O(1/k^2)$ advantage over Heavy Ball is primarily realised with large fixed $\beta$ rather than matched schedules; and FISTA combines Nesterov momentum with projection to achieve the optimal rate for constrained smooth optimisation.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) to ensure reproducibility. The three benchmark functions span a spectrum of difficulty: a well-conditioned convex quadratic, a moderately anisotropic nonlinear function, and a severely ill-conditioned non-convex valley.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters $\theta \in \mathbb{R}^2$ is generated as follows:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss is:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strictly convex quadratic with \emph{constant} Hessian $H = \frac{1}{m}X^T X \in \mathbb{R}^{2 \times 2}$, gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$, and a unique global minimiser $\theta^{\star}_{\text{OLS}} = (X^TX)^{-1}X^Ty$. Due to the irreducible observation noise, the minimum loss is $J^{\star} \approx 0.4826 \neq 0$. Starting point: $\theta_0 = [0, 0]^T$.

\textbf{Condition number:} The Hessian $H = \frac{1}{m}X^TX$ has eigenvalues $\lambda_{\min}$ and $\lambda_{\max}$. Since $X_{\text{data}} \sim \mathcal{N}(0, I_{m \times 2})$ by construction and $m = 1000 \gg d = 2$, the law of large numbers gives $H \approx \mathbb{E}[x^{(i)} (x^{(i)})^T] = I_2$, so $\kappa(H) = \lambda_{\max}/\lambda_{\min} \approx 1.0$. In practice, with finite samples and the specific random seed, the empirical eigenvalues deviate slightly from 1, giving $\kappa \approx 1.1$. This near-perfect conditioning explains why all gradient-based methods converge rapidly on Benchmark A with any reasonable step size.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
\label{eq:bench_B}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient and Hessian are:
\begin{align}
\nabla f(x) &= \begin{bmatrix} 2(x_1 - 1) + \cos(x_1) \\ 10(x_2 - 2) \end{bmatrix}, \quad
H(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{align}
The eigenvalue ratio of the Hessian varies between $\approx 1.8{:}10$ and $\approx 3{:}10$ depending on the iterate $x_1$, creating moderate anisotropy with condition number $\kappa \approx 5.6$. The global minimum is at approximately $x^\star \approx (0.582, 2.0)$ (solving $2(x_1-1) + \cos(x_1) = 0$) with $f^\star \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
\label{eq:rosenbrock}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The Hessian at the global minimum $(1,1)$ is:
\begin{equation}
\nabla^2 f(1,1) = \begin{bmatrix} 2 + 1200\cdot1 - 400\cdot1 & -400\cdot1 \\ -400\cdot1 & 200 \end{bmatrix} = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}.
\end{equation}
The eigenvalues are $\frac{802+200}{2} \pm \sqrt{\left(\frac{802-200}{2}\right)^2 + 400^2} = 501 \pm \sqrt{250601} \approx 501 \pm 500.6$, giving $\lambda_{\max} \approx 1001.6$ and $\lambda_{\min} \approx 0.4$, so $\kappa \approx 2500$. This extreme ill-conditioning, combined with the non-convex banana-shaped valley $\{x : x_2 = x_1^2\}$, makes Rosenbrock the canonical challenge. GD requires $O(\kappa \log(1/\varepsilon)) \approx O(2500 \log(1/\varepsilon))$ iterations for $\varepsilon$ accuracy. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems (Questions 5--6): $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$, a bounded box polytope with vertices at the four corners.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Motivation}

Standard gradient descent with a fixed step size $\alpha$ requires $\alpha \leq 2/L$ for convergence on $L$-smooth functions, and its convergence rate on a $\mu$-strongly convex function is:
\begin{equation}
f(x_k) - f^\star \leq \left(1 - \frac{\mu}{L}\right)^k \bigl(f(x_0) - f^\star\bigr) = \left(\frac{\kappa - 1}{\kappa}\right)^k \bigl(f(x_0) - f^\star\bigr),
\end{equation}
where $\kappa = L/\mu$ is the condition number. For ill-conditioned problems ($\kappa \gg 1$), convergence is extremely slow. Adaptive methods address this by adjusting the effective step size based on observed gradient information, removing the need to tune $\alpha$ with knowledge of $L$ and $\mu$.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive step: larger when far from minimum}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is theoretically motivated by choosing $\alpha_k$ to directly reduce the distance to the optimum. For a convex function with known $f^\star$:
\begin{align}
\|x_{k+1} - x^\star\|^2 &= \|x_k - \alpha_k g_k - x^\star\|^2 = \|x_k - x^\star\|^2 - 2\alpha_k g_k^T(x_k - x^\star) + \alpha_k^2\|g_k\|^2 \notag\\
&\leq \|x_k - x^\star\|^2 - 2\alpha_k(f(x_k) - f^\star) + \alpha_k^2\|g_k\|^2,
\end{align}
where we used convexity $g_k^T(x_k - x^\star) \geq f(x_k) - f^\star$. Minimising this over $\alpha_k$ gives exactly $\alpha_k = (f(x_k)-f^\star)/\|g_k\|^2$, and the minimum value is $\|x_k - x^\star\|^2 - (f(x_k)-f^\star)^2/\|g_k\|^2$. This yields global convergence without any knowledge of the Lipschitz constant $L$.

\begin{remark}[First-order Taylor motivation for Polyak step]
An alternative motivation for the Polyak step: apply a first-order Taylor expansion to the objective after a gradient step $x^+ = x - \alpha g$:
\[
f(x - \alpha g) \approx f(x) - \alpha \|g\|^2.
\]
Setting this equal to $f^\star$ and solving for $\alpha$ gives $\alpha = (f(x) - f^\star)/\|g\|^2$ --- exactly the Polyak step. This derivation shows that Polyak chooses $\alpha$ to make the linearised objective hit $f^\star$ in a single step, a more intuitive ``progress guarantee'' framing than the distance-to-optimum derivation.
\end{remark}

When $f^\star$ is \emph{misspecified} as $\tilde{f}^\star < f^\star_{\text{true}}$, the step size $\alpha_k = (f(x_k) - \tilde{f}^\star)/\|g_k\|^2$ grows as $\|g_k\| \to 0$ while the numerator $f(x_k) - \tilde{f}^\star \to f^\star_{\text{true}} - \tilde{f}^\star > 0$ remains bounded away from zero, causing $\alpha_k \to \infty$ and instability.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative squared gradient sum (per-coordinate)}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad scales each coordinate by the inverse of its historical gradient magnitude. Coordinates with large, frequent gradients receive smaller steps; rarely-updated coordinates receive larger steps. This is particularly beneficial for sparse gradients (e.g., word embeddings) but its monotonically non-increasing effective step size can cause premature stopping on dense problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average (EMA) of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop (Tieleman \& Hinton, 2012) addresses Adagrad's decaying step problem by replacing the cumulative sum with an exponential moving average controlled by the forgetting factor $\beta$. With $\beta = 0.9$, each squared gradient contributes $0.9^k$ weight after $k$ steps, so effectively only the most recent $1/(1-\beta) = 10$ gradients influence the scaling.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Momentum accumulates gradient history}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

Heavy Ball adds an inertia term: the update direction is a weighted combination of the current gradient and the previous update direction. For quadratic objectives, the optimal parameters $\alpha^\star, \beta^\star$ can be chosen analytically to achieve the optimal convergence rate $\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^k$, which is dramatically faster than gradient descent's $\left(\frac{\kappa-1}{\kappa}\right)^k$ for large $\kappa$.

\begin{remark}[Heavy Ball effective step size]
Unrolling the momentum recurrence $z_{k+1} = \beta z_k + \alpha \nabla f(x_k)$ gives $z_{k+1} = \sum_{j=0}^{k} \beta^{k-j}\,\alpha\,\nabla f(x_j)$. Since $\sum_{j=0}^{\infty} \beta^j = 1/(1-\beta)$, the \emph{effective step size} relative to any single gradient is $\alpha/(1-\beta)$. For $\beta = 0.9$, this equals $10\alpha$, which explains why the Heavy Ball step size $\alpha$ must be set approximately $10\times$ smaller than the GD step size to maintain stability. Equivalently, momentum amplifies the step in persistent gradient directions by a factor of $1/(1-\beta)$, providing acceleration exactly where it is most beneficial.
\end{remark}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest on Rosenbrock ($f = 0.009$), achieving more than a $370\times$ reduction over GD. On Benchmarks A and B (where $f^\star$ is misspecified), Adagrad and Heavy Ball match or exceed the GD baseline without any oracle knowledge.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to EMA gradient averaging. Heavy Ball shows characteristic underdamped oscillations (overshoot-recover cycles) before settling.
\item \textbf{Stability:} GD baseline is most stable but slowest. Polyak is \emph{critically unstable} when $f^{\star}$ is misspecified --- even a small error in $f^\star$ causes divergence. Adagrad is uniformly stable due to its monotonically non-increasing step size, though this monotone decay can cause premature stagnation on non-stationary loss surfaces.
\item \textbf{Practical implication:} In practice, exact $f^{\star}$ is rarely available (it is equivalent to knowing the solution), so Polyak is mainly of theoretical interest. RMSprop and Adagrad require only the objective's gradient and a base rate $\alpha_0$, making them more practical. Heavy Ball requires step size $\alpha$ and momentum $\beta$, which can be chosen near-optimally for quadratic objectives using spectral analysis.
\item \textbf{Library availability:} Despite its theoretical elegance, Polyak step size is \emph{not} implemented in common machine learning frameworks (PyTorch, TensorFlow, JAX). Adagrad and RMSprop are both standard optimisers in these libraries, reflecting their practical robustness across ML tasks. This gap reflects that Polyak's requirement for $f^\star$ makes it impractical for the vast majority of deep learning applications where the global minimum is unknown.
\item \textbf{Adaptive methods on ill-conditioned problems:} A notable observation from these experiments (consistent with lecture note findings) is that Adagrad and RMSprop, even with tuned base rates $\alpha_0$, do \emph{not} converge faster than a well-tuned constant step size on the ill-conditioned Rosenbrock function. Table~\ref{tab:q1_final} shows RMSprop achieving only $f = 3.12$ vs.\ GD's $f = 3.51$ --- a marginal improvement. The per-coordinate scaling provides no benefit for the Rosenbrock valley because the slow direction (along the valley) and the fast direction (perpendicular to it) involve both coordinates in a coupled way, making coordinate-wise rescaling ineffective.
\end{itemize}

\section{Question 2: Momentum-Based and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Theoretical Motivation}

Gradient descent achieves an $O(1/k)$ convergence rate for convex smooth objectives. A fundamental question in optimisation theory is: can first-order methods do better? Nesterov (1983) proved that $O(1/k^2)$ is achievable for convex smooth objectives, and his accelerated method is optimal in the oracle complexity sense. This motivates the Nesterov momentum formulation below. Adam (Kingma \& Ba, 2015) combines momentum with per-coordinate adaptive scaling, inheriting the practical benefits of both RMSprop and momentum methods.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Update momentum at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This ``look before you leap'' strategy allows the algorithm to correct its trajectory before overshooting. The schedule $\beta_k = (k-1)/(k+2) \to 1$ gradually increases momentum, achieving the optimal $O(1/k^2)$ convergence rate for convex smooth objectives:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2},
\end{equation}
where $L$ is the gradient's Lipschitz constant.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in [0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (EMA of gradients)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (EMA of sq.\ gradients)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction: compensates zero initialisation}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam's bias correction is essential: without it, early iterates $m_1 = (1-\beta_1)g_1 \approx 0$ would be severely biased towards zero, causing overconfident small steps in the first few iterations. The bias-corrected estimate $\hat{m}_t = m_t/(1-\beta_1^t)$ debiases by the factor $(1-\beta_1^t)$ which grows from $\approx 0$ to $\approx 1$ as $t \to \infty$.

\subsection{Mini-Batch Stochastic Gradient Descent (Benchmark A)}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\pi$ \Comment{New permutation each epoch}
    \For{each mini-batch $\mathcal{B}_j = \{\pi(jb+1), \ldots, \pi((j+1)b)\}$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}_j}^T(X_{\mathcal{B}_j}\theta - y_{\mathcal{B}_j})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{SGD Gradient Variance Analysis}

The theoretical reason smaller batches converge faster per epoch but oscillate more is the \emph{bias-variance tradeoff} of the stochastic gradient estimate. For a mini-batch $\mathcal{B}$ of size $b$ drawn without replacement from $m$ samples:
\begin{equation}
\text{Var}[\hat{g}_b] = \frac{1}{b}\left(1 - \frac{b}{m}\right)\sigma_g^2 \approx \frac{\sigma_g^2}{b} \quad \text{for } b \ll m,
\end{equation}
where $\sigma_g^2$ is the population gradient variance. This $1/b$ scaling means batch size $b=5$ produces $40/5 = 8\times$ higher gradient variance than $b=40$.

\begin{figure}[H]
\centering
\includegraphics[width=0.75\textwidth]{figures/q2_sgd_variance.pdf}
\caption{Q2: Empirical gradient variance vs batch size at $\theta = \theta^\star$ on Benchmark A (300 trials per batch size). The variance follows the theoretical $\sim 1/b$ relationship (red dashed line) over the full range $b \in [1, 1000]$. At the Q2 batch sizes ($b=5$: high variance, $b=40$: $8\times$ lower variance), the trade-off between update frequency (favours small $b$) and gradient accuracy (favours large $b$) determines the observed convergence behaviour. Full-batch GD ($b=1000$) uses the exact gradient but only one update per epoch, explaining why it is slowest per epoch despite having the smallest noise.}

\begin{remark}[Diminishing step sizes for exact SGD convergence]
With a constant step size $\alpha$, SGD does not converge to $\theta^\star$ but to a neighbourhood of size $O(\alpha \sigma_g^2/\mu)$ around it (for $\mu$-strongly convex $f$). The irreducible noise floor can be eliminated using a \emph{diminishing step sequence} $\alpha_k \to 0$ satisfying the Robbins-Monro conditions: $\sum_k \alpha_k = \infty$ (step stays large enough to escape any basin) and $\sum_k \alpha_k^2 < \infty$ (step decreases fast enough to suppress noise). A common choice is $\alpha_k = \alpha_0 / (1 + k/k_0)$. However, with diminishing steps the convergence rate degrades to $O(1/k)$ on strongly convex problems (vs.\ linear rate with constant step), illustrating the fundamental tension in stochastic optimisation between bias (large $\alpha$ noise) and variance (slow $O(1/k)$ rate with diminishing $\alpha$).
\end{remark}

\subsection{Nesterov vs.\ Adam: Theoretical Comparison}

\begin{itemize}[nosep]
\item \textbf{Convergence rate:} Nesterov achieves the information-theoretically optimal $O(1/k^2)$ rate for smooth convex objectives (proved by Nesterov, 1983). Adam does not have provably optimal rates in the deterministic setting; Reddi et al.\ (2018) showed Adam can fail to converge on simple convex problems without additional hyperparameter assumptions.
\item \textbf{Memory:} Both require $O(d)$ extra memory. Adam uses $2d$ extra scalars ($m_t, v_t$) vs.\ $d$ scalars ($z_k$) for Nesterov. For $d = 10^8$ (large language models), this $2\times$ overhead matters.
\item \textbf{Adaptivity:} Adam adapts step sizes per coordinate, which is beneficial when gradient magnitudes differ vastly across coordinates (e.g., embedding tables). Nesterov uses a single global step size, which can be suboptimal in highly anisotropic settings but maintains the theoretically proven $O(1/k^2)$ rate.
\item \textbf{Rosenbrock performance:} Nesterov outperforms Adam ($f = 0.479$ vs.\ $f = 1.579$) because the banana valley has a persistent gradient direction --- momentum accumulation is exactly the right inductive bias. Adam's gradient-magnitude scaling inadvertently suppresses steps in the dominant valley direction when that direction accumulates large $v_t$ values, limiting its effectiveness on this structured problem.
\item \textbf{Nesterov vs.\ Heavy Ball with matched schedules:} When both methods use the same increasing momentum schedule $\beta_k = \min((k-1)/(k+2), \beta_{\max})$ (rather than a fixed $\beta = 0.9$ for Heavy Ball), Nesterov and Heavy Ball perform very similarly in practice (see Figure~\ref{fig:q2_nesterov_hb_schedule}). The theoretical $O(1/k^2)$ advantage of Nesterov over Heavy Ball is primarily realised when $\beta$ is large and fixed. With an adaptive schedule that mimics Nesterov's prescribed $\beta_k$, the lookahead correction in Nesterov's update becomes a minor second-order effect.
\item \textbf{Adam step-size tolerance:} Adam allows significantly larger base step sizes without diverging compared to constant-step GD or Nesterov. For Rosenbrock, constant-step GD diverges for $\alpha > 1$, while Adam remains stable at $\alpha = 0.1$--$0.5$ due to the $1/\sqrt{v_t}$ normalisation bounding effective per-coordinate steps. This tolerance for larger $\alpha$ compensates for the absence of a proven $O(1/k^2)$ rate by allowing more aggressive descent per step.
\end{itemize}

\subsection{First- and Second-Order Local Approximations of $g(x) = x^4$}

Consider $g(x) = x^4$ with $g'(x) = 4x^3$, $g''(x) = 12x^2$. The Taylor approximations around $x_0 = 0.25$ are:

\textbf{First-order approximation} (tangent line):
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.0039 + 0.0625\,(x - 0.25).
\end{equation}
Minimising $\hat{g}_1$ is a linear programme with no finite solution (the linear function $\to -\infty$), so the first-order model alone cannot localise the minimum.

\textbf{Second-order approximation} (quadratic):
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\end{equation}
Setting $\hat{g}_2'(x) = 0$ gives the \textbf{Newton update}:
\begin{equation}
x^+ = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{4 \cdot 0.25^3}{12 \cdot 0.25^2} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.083 = 0.167.
\end{equation}
More generally: $x_{k+1} = x_k - [g''(x_k)]^{-1}\, g'(x_k)$. The second-order model captures local curvature, enabling a much larger and better-directed step than gradient descent.

The second-order approximation (parabola) captures local curvature and fits well in a neighbourhood of $x_0$. The Newton update steps to the parabola's vertex, providing a far better estimate of the function minimum than a simple gradient step. \subsection{Newton's Method with Hessian Damping}

For the multivariate case, Newton's method with step-size $\alpha$ and Tikhonov regularisation $\lambda$ is:

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0,1]$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised Hessian: ensures positive definiteness}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: solve $H_k p_k = g_k$}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{$\alpha=1$ gives pure Newton; $\alpha<1$ gives damped Newton}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{proposition}[Newton converges in one step on quadratics]
For a purely quadratic objective $f(x) = \frac{1}{2}x^T H x + b^T x + c$ with constant Hessian $H \succ 0$, Newton's method with $\alpha = 1$ converges exactly to $x^\star = -H^{-1}b$ in a \emph{single} iteration from any starting point $x_0$:
\[
x_1 = x_0 - H^{-1}\nabla f(x_0) = x_0 - H^{-1}(Hx_0 + b) = x_0 - x_0 - H^{-1}b = x^\star.
\]
This is because the second-order model is \emph{exact} for quadratics, so the Newton step lands exactly at the minimiser.
\end{proposition}

\begin{theorem}[Local Quadratic Convergence of Newton's Method]
If $f$ is twice continuously differentiable, $x^\star$ is a non-degenerate local minimiser (i.e., $\nabla^2 f(x^\star) \succ 0$), and $\nabla^2 f$ is Lipschitz continuous with constant $M$ near $x^\star$, then pure Newton's method ($\alpha = 1$, $\lambda = 0$) satisfies:
\begin{equation}
\|x_{k+1} - x^\star\| \leq \frac{M}{2\mu}\, \|x_k - x^\star\|^2,
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^\star)) > 0$. This quadratic convergence means the number of correct digits \emph{doubles} at each iteration.
\end{theorem}

\subsection{Newton as Sequential Quadratic Programming and Practical Limitations}

As noted in the course lectures, Newton's method can be interpreted as a \textbf{Sequential Quadratic Programme (SQP)}: at each iteration, fit a local quadratic model to $f$ at the current point $x_k$, find the exact minimiser of this quadratic model, then repeat. This ``model-trust-repeat'' structure is precisely why Newton achieves quadratic convergence --- each step solves the local quadratic problem exactly, whereas GD only takes a small step along the gradient.

\textbf{Connection to Adagrad (from lectures):} Newton's method can be written as:
\begin{equation}
x_{t+1} = x_t - \alpha \bigl[b_1 \tfrac{\partial f}{\partial x_1}(x_t),\ \ldots,\ b_n \tfrac{\partial f}{\partial x_n}(x_t)\bigr]
\end{equation}
where $b_i$ are the rows of $(H_k)^{-1}$. When $b_1 = \cdots = b_n = 1$, this is standard GD. Adagrad uses $b_i = 1/\sqrt{\sum_t (\partial f/\partial x_i)^2}$ --- a diagonal approximation to $H^{-1}$ based on gradient history. Newton uses the \emph{full} inverse Hessian, capturing cross-coordinate curvature information that Adagrad misses.

\textbf{Computational cost for large $d$:}
\begin{enumerate}[nosep]
\item Storing $\nabla^2 f$: $O(d^2)$ memory. For $d = 10^6$: a $10^6 \times 10^6$ matrix with $10^{12}$ entries ($\approx 8000$ GB at 64-bit precision) --- infeasible.
\item Computing $\nabla^2 f$: $O(d^2)$ gradients.
\item Inverting $\nabla^2 f$: $O(d^3)$ in general (but only $O(d)$ for the diagonal Hessians of Benchmarks B and A).
\end{enumerate}
For $d=2$ as in this assignment, Newton is clearly superior (20 vs 80 iterations, minimal overhead). For large $d$, \textbf{quasi-Newton methods} such as BFGS or its memory-efficient variant L-BFGS build an approximation to $H^{-1}$ using gradient-difference vectors from recent iterates, requiring only $O(d)$ memory and computation per step. L-BFGS is the default solver in \texttt{sklearn.linear\_model.LogisticRegression} and is available in PyTorch (\texttt{torch.optim.LBFGS}).

\begin{remark}[BFGS superlinear convergence]
BFGS builds an approximate inverse Hessian $B_k \approx H_k^{-1}$ using the secant condition: given $s_k = x_{k+1} - x_k$ and $y_k = \nabla f(x_{k+1}) - \nabla f(x_k)$, the update
\begin{equation}
B_{k+1} = B_k + \frac{(s_k - B_k y_k) s_k^T + s_k (s_k - B_k y_k)^T}{s_k^T y_k} - \frac{(s_k - B_k y_k)^T y_k}{(s_k^T y_k)^2} s_k s_k^T
\end{equation}
ensures $B_{k+1} y_k = s_k$ (the approximation satisfies the secant equation for the latest step). BFGS achieves \emph{superlinear convergence}: the error ratio $\|x_{k+1} - x^\star\| / \|x_k - x^\star\| \to 0$ as $k \to \infty$, faster than any linear rate but without the strict $O(\|x_k - x^\star\|^2)$ quadratic bound of Newton. Intuitively, the Hessian approximation $B_k^{-1} \to H(x^\star)^{-1}$ as iterates approach $x^\star$, so BFGS behaves more and more like Newton in the limit. L-BFGS stores only the last $m$ pairs $(s_k, y_k)$ (typically $m = 5$--$20$), reducing memory from $O(d^2)$ to $O(md)$ at the cost of a less accurate Hessian approximation.
\end{remark}

\subsection{Part A: Finite Difference Approximation and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta} = \frac{\partial f}{\partial x_i}(x) + \underbrace{\frac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x)}_{\text{truncation error: }O(\delta)} + O(\delta^2).
\end{equation}
There is also \emph{numerical cancellation error}: if $f(x)$ is evaluated with floating-point error $\varepsilon_\text{mach} \approx 10^{-16}$, then the subtraction $f(x+\delta e_i) - f(x)$ has absolute error $\approx \varepsilon_\text{mach}/\delta$, growing unboundedly as $\delta \to 0$. Balancing truncation ($O(\delta)$) and cancellation ($O(\varepsilon_\text{mach}/\delta)$) errors gives the optimal step size:
\begin{equation}
\delta^\star = \sqrt{\varepsilon_\text{mach}} \approx 10^{-8}, \quad \text{with minimum total error} \approx 2\sqrt{\varepsilon_\text{mach}} \approx 10^{-8}.
\end{equation}
However, using $\delta^\star$ is not always practical: near a stationary point where $f(x+\delta e_i) \approx f(x)$, catastrophic cancellation dominates even at $\delta = \delta^\star$.

\begin{remark}[Central differences vs. forward differences]
The \emph{central} finite difference $\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x+\delta e_i) - f(x-\delta e_i)}{2\delta}$ achieves $O(\delta^2)$ truncation error (the $O(\delta)$ term cancels by symmetry), at the cost of $2d$ evaluations per gradient instead of $d+1$. For $d=2$, central differences use 4 evaluations per gradient vs.\ 3 for forward; for large $d$, this doubles the cost but substantially improves accuracy. The optimal $\delta^\star_{\rm central} = \varepsilon_{\rm mach}^{1/3} \approx 10^{-5}$ (balancing $O(\delta^2)$ vs.\ $O(\varepsilon_{\rm mach}/\delta)$), giving minimum error $\approx 3\varepsilon_{\rm mach}^{2/3} \approx 10^{-10}$, approximately three orders of magnitude better than forward differences.
\end{remark}

\item \textbf{Poor setting} ($\delta = 0.8$): truncation error $O(0.8)$ is of the same order as the gradient itself; the FD gradient points in a meaningfully wrong direction, degrading descent quality.
\end{itemize}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Random unit vector}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates a \emph{directional derivative} along a random direction, requiring only \textbf{two} function evaluations per iteration regardless of dimension $d$. The gradient estimate satisfies $\mathbb{E}[\hat{g}_k] = \frac{1}{d}\nabla f(x_k)$ (up to $O(\delta^2)$ error), so it is an unbiased gradient estimate modulo a $1/d$ scaling factor which effectively reduces the step size by $1/d$. For $d=2$, this halving means convergence is approximately twice as slow as exact GD with the same $\alpha$.

\begin{proposition}[Nesterov random search gradient estimate]
Let $u \in \mathbb{R}^d$ be uniform on the unit sphere $\mathbb{S}^{d-1}$. Then
\[
\mathbb{E}\!\left[\frac{f(x + \delta u) - f(x)}{\delta}\, u\right] = \frac{1}{d}\,\nabla f(x) + O(\delta).
\]
\end{proposition}
\begin{proof}
By Taylor expansion, $(f(x+\delta u)-f(x))/\delta = \nabla f(x)^T u + O(\delta)$. Therefore:
\[
\mathbb{E}\!\left[\frac{f(x + \delta u) - f(x)}{\delta}\,u\right] = \mathbb{E}\!\left[(\nabla f(x)^T u)\,u\right] + O(\delta) = \mathbb{E}[u u^T]\,\nabla f(x) + O(\delta).
\]
For a uniform unit vector, $\mathbb{E}[u_i u_j] = 0$ for $i \neq j$ (by odd symmetry), and $\mathbb{E}[u_i^2] = \mathbb{E}[u_j^2]$ (by isotropy). Since $\sum_i u_i^2 = 1$, taking expectations gives $d\,\mathbb{E}[u_i^2] = 1$, so $\mathbb{E}[u_i^2] = 1/d$. Thus $\mathbb{E}[u u^T] = \frac{1}{d} I$, and the result follows.
\end{proof}

\subsection{Part B: Derivative-Free Methods on Benchmark C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\rho_r = 1$): reflect the worst vertex through the centroid of the best $n$ vertices.
\item \textbf{Expansion} ($\gamma = 2$): if reflection improved, try an extended step.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection worsened, contract towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, shrink all vertices towards the best.
\end{enumerate}
These geometric operations gradually steer the simplex into lower-objective regions without any gradient information.

\begin{remark}[Nelder--Mead has no convergence guarantees]
Despite its widespread practical use, the Nelder--Mead algorithm has \emph{no general convergence theory for $d \geq 2$}. McKinnon (1998) constructed a family of strictly convex functions on $\mathbb{R}^2$ for which the algorithm converges to a non-optimal point: the simplex degenerates (two or more vertices collapse towards the same point) and subsequent reflection and expansion steps fail to escape. In practice, degenerate simplices stagnate because reflections pass through the centroid of nearly-collinear vertices, generating steps of negligible length. In higher dimensions ($d \geq 5$), empirical evidence suggests Nelder--Mead often fails to make progress once the simplex contracts below the scale of the objective's curvature. In contrast, Nesterov random search and finite-difference gradient descent have explicit $O(1/k)$ convergence rates (with appropriate step sizes), making them theoretically preferable whenever gradients can be estimated. Nelder--Mead's practical advantage lies in robustness to noisy, discontinuous, or simulation-based objectives where gradient estimation is impossible or unreliable.
\end{remark}

Settings: initial simplex step $= 0.35$, iterations $= 160$.

\subsection{Problem Formulation and KKT Conditions}

We minimise Benchmark B subject to the inequality constraint $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$:
\begin{equation}
\min_{x \in \mathbb{R}^2} f(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1) \quad \text{subject to} \quad x_1 \geq 0.5.
\end{equation}

The unconstrained minimum $(0.582, 2.0)$ lies in the feasible set $\{x: x_1 \geq 0.5\}$, so the constrained minimum coincides with it. The KKT conditions at the constrained optimum $(x_1^\star, x_2^\star) = (0.582, 2.0)$ are:
\begin{align}
\nabla f(x^\star) + \mu^\star \begin{bmatrix} -1 \\ 0 \end{bmatrix} &= 0, \quad \mu^\star \geq 0, \quad \mu^\star(0.5 - x_1^\star) = 0.
\end{align}
Since $x_1^\star = 0.582 > 0.5$, the constraint is \emph{inactive} at the optimum, so $\mu^\star = 0$ (complementary slackness), and the constrained and unconstrained optima coincide.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = \begin{bmatrix}\max(0.5,\, z_{k+1,1}) \\ z_{k+1,2}\end{bmatrix}$ \Comment{Project onto $\mathcal{X} = \{x: x_1 \geq 0.5\}$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection onto the half-space $\{x: x_1 \geq 0.5\}$ is the $\max$ operation, computationally trivial. The theory of projected gradient descent guarantees convergence to the constrained minimiser when $f$ is convex and $\alpha \leq 1/L$.

\begin{remark}[General half-plane projection]
For a general half-plane $\mathcal{X} = \{x : a^T x \leq b\}$ with $a \in \mathbb{R}^d$, the projection is:
\begin{equation}
\Pi_{\mathcal{X}}(z) = \begin{cases} z & \text{if } a^T z \leq b \\ z - \dfrac{a^T z - b}{\|a\|^2}\, a & \text{if } a^T z > b \end{cases}
\end{equation}
This is an $O(d)$ operation: compute $a^T z$, check feasibility, and subtract the outward normal scaled by the violated amount. For our problem, $a = -e_1 = (-1, 0)^T$ and $b = -0.5$, giving $\Pi_\mathcal{X}(z)_1 = \max(0.5, z_1)$ and $\Pi_\mathcal{X}(z)_2 = z_2$ --- consistent with the algorithm above.
\end{remark}

\begin{remark}[FISTA: Nesterov acceleration for constrained optimisation]
Projected gradient descent converges at $O(1/k)$ for convex $f$. Combining Nesterov's momentum with the projection operator yields the \textbf{Fast Iterative Shrinkage-Thresholding Algorithm} (FISTA):
\begin{align}
y_t &= x_t + \beta_t(x_t - x_{t-1}) \quad (\text{Nesterov lookahead}) \\
x_{t+1} &= \Pi_{\mathcal{X}}\!\left(y_t - \alpha\, \nabla f(y_t)\right) \quad (\text{gradient step then project})
\end{align}
with $\beta_t = (t-1)/(t+2)$ as in the unconstrained Nesterov scheme. FISTA achieves the optimal $O(1/k^2)$ rate for constrained convex optimisation, matching the unconstrained Nesterov bound. Notably, the literature on combining Polyak step size, RMSprop, Heavy Ball, and Adam \emph{with} projected gradient descent is sparse --- these adaptive methods are rarely studied in the constrained setting, and their convergence guarantees under projection are not fully established.
\end{remark}

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $\nabla F_\lambda(x) \gets \nabla f(x) - \lambda \mathbf{1}[x_1 < 0.5]\, e_1$ \Comment{Subgradient}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Key insights:}
\begin{enumerate}[nosep]
\item \textbf{Projected GD} enforces feasibility immediately and exactly at every iteration --- the projection $\Pi_\mathcal{X}$ is computationally free ($O(d)$) for simple constraints (box, half-space, simplex) but can be expensive or computationally challenging for complex constraints (semidefinite cone: $O(d^3)$ SVD; flow polytope: $O(d^2)$ network flow).

\item \textbf{Penalty methods} trade exact feasibility for implementation simplicity: only a modified gradient is needed, with no projection oracle. The weight $\lambda$ controls the feasibility-optimality trade-off. As $\lambda \to \infty$, the penalty method solution converges to the constrained optimum (for exact penalty $\ell_1$) or the gap shrinks as $O(1/\lambda)$ (for quadratic penalty). Large $\lambda$ creates ill-conditioned loss landscapes near the constraint boundary, requiring smaller $\alpha$ for stability, which in turn slows convergence. This Goldilocks problem with $\lambda$ motivates \emph{augmented Lagrangian methods} (ALM) that adaptively increase $\lambda$.

\begin{remark}[Augmented Lagrangian: exact feasibility without ill-conditioning]
The \emph{Augmented Lagrangian Method} (ALM, also called the \emph{method of multipliers}) adds a dual variable $\mu$ to avoid the ill-conditioning of pure penalty. For equality constraints $g(x)=0$, ALM alternates:
\begin{align}
x_{k+1} &= \arg\min_{x}\; f(x) + \mu_k^\top g(x) + \tfrac{\lambda}{2}\|g(x)\|^2 \\
\mu_{k+1} &= \mu_k + \lambda\, g(x_{k+1}).
\end{align}
The dual update $\mu_{k+1}$ shifts the penalty centre: the inner minimisation targets $g(x) = -\mu_k/\lambda$ rather than $g(x)=0$, so the quadratic well is not artificially tightened around the constraint. This allows $\lambda$ to remain \emph{fixed} (no ill-conditioning) while the dual variable $\mu_k$ drives the violation $g(x_k) \to 0$ at a linear rate. In practice, ALM achieves exact constraint satisfaction in finite $\lambda$, whereas pure penalty requires $\lambda \to \infty$. For inequality constraints $g(x) \leq 0$, the inequality form uses $\mu_{k+1} = \max(0, \mu_k + \lambda g(x_{k+1}))$ (projecting the dual onto the non-negative orthant).
\end{remark}

\item \textbf{Feasibility speed:} Projected GD achieves zero violation at iteration 1 (hard constraint). Penalty methods achieve $O(1/\lambda)$ violation asymptotically. For time-critical safety constraints, projected GD is preferable; for soft constraints or when projection is expensive, penalty or augmented Lagrangian methods are better.

\item For this specific problem, the unconstrained optimum $(0.582, 2.0)$ is strictly feasible ($x_1 = 0.582 > 0.5$), so the constraint is inactive (KKT multiplier $\mu^\star = 0$), and all methods ultimately converge to the same point. The difference would be more pronounced if the constraint were active, e.g., $x_1 \geq 1.0$, where the constrained optimum is at the boundary $(1.0, 2.0)$ and weak penalty methods would converge to a point strictly inside the infeasible region.

\item \textbf{Change-of-variables alternative:} For simple non-negativity constraints $x_1 \geq 0$, an alternative to projection or penalty methods is a change of variables: let $x_1 = e^{z_1}$, so $z_1 \in \mathbb{R}$ automatically satisfies $x_1 > 0$ for all $z_1$. Similarly, for box constraints $0 \leq x_1 \leq 1$, the sigmoid $x_1 = \sigma(z_1) = 1/(1+e^{-z_1})$ is always in $(0,1)$. However, this approach has a drawback: even if the original problem is strongly convex (and hence easy), the reparametrised problem $h(z) = f(e^z)$ is typically \emph{not} strongly convex in $z$, making the transformed problem harder to minimise. Projected gradient descent, which directly operates in the original constrained space, avoids this distortion.
\end{enumerate}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box polytope with 4 vertices).

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is obtained by independent coordinate minimisation. Since all components of $a$ are positive, the minimum is at the lower bound for each coordinate:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}
This illustrates the \textbf{fundamental theorem of linear programming}: the optimum of a linear objective over a bounded polytope is always attained at a \emph{vertex} of the feasible set. Vertices are where constraints are simultaneously active; the box $\mathcal{X}$ has exactly 4 vertices: $(0.5, -5)$, $(5, -5)$, $(0.5, 10)$, $(5, 10)$.

\begin{remark}[Why Frank--Wolfe uses an LP at every step]
The Frank--Wolfe algorithm replaces the quadratic descent direction $x - \alpha\nabla f(x)$ with an LP direction: at each step, it finds $z_k = \arg\min_{z\in\mathcal{X}} \nabla f(x_k)^T z$, i.e., the vertex that minimises the \emph{linearisation} of $f$ around $x_k$. By the fundamental theorem of LP, this vertex is always an extreme point of $\mathcal{X}$. The update $x_{k+1} = (1-\gamma_k)x_k + \gamma_k z_k$ is a convex combination of two feasible points, hence feasible --- no projection is needed. This LP structure is the reason Frank--Wolfe is ``projection-free'': the feasibility of iterates follows from LP optimality at each step, not from an explicit projection onto $\mathcal{X}$.
\end{remark}

\subsection{Frank--Wolfe Algorithm}

Frank--Wolfe (or conditional gradient) replaces each gradient step with a \emph{linear minimisation} over $\mathcal{X}$:

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: maintains $x_{k+1} \in \mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Key property:} Since $x_k, z_k \in \mathcal{X}$ and $\mathcal{X}$ is convex, every iterate $x_{k+1} = \beta x_k + (1-\beta)z_k \in \mathcal{X}$ --- the method is \emph{projection-free} and automatically maintains feasibility.

For box constraints, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
This requires $O(d)$ operations per iteration and always selects a \emph{vertex} of $\mathcal{X}$ as the linear minimiser.

\textbf{Convergence theory:} For a smooth convex objective with $L$-Lipschitz gradient, Frank--Wolfe with $\gamma_k = 2/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\,\mathrm{diam}(\mathcal{X})^2}{k+2},
\end{equation}
where $\mathrm{diam}(\mathcal{X}) = \sqrt{(5-0.5)^2 + (10+5)^2} \approx 15.7$ for our box. This $O(1/k)$ rate is slower than projected gradient descent's $O(1/k^2)$ with Nesterov acceleration.

The \textbf{Frank--Wolfe gap} $g(x_k) = \max_{y \in \mathcal{X}} \nabla f(x_k)^T(x_k - y)$ serves as a certificate of optimality: if $g(x_k) = 0$, then $x_k$ is a constrained minimum (since $\nabla f(x_k)^T(y - x_k) \geq 0$ for all $y \in \mathcal{X}$ --- the KKT condition for a minimum). The gap also provides a computable upper bound $f(x_k) - f^\star \leq g(x_k)$, making it useful as a stopping criterion.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q6_fw_gap.pdf}
\caption{Q6: Frank--Wolfe gap $g(x_k)$ vs.\ objective $f(x_k)$ for the interior optimum ($\beta = 0.985$, 140 iterations). \textbf{Left}: semi-log scale. \textbf{Right}: log-log scale with $O(1/k)$ reference. The gap satisfies the required bound $f(x_k) - f^\star \leq g(x_k)$ at every iteration (gap above objective). For boundary optima where the optimum \emph{is} a vertex, the gap tightens significantly.}

Note: the fixed-$\beta$ scheme $x_{k+1} = \beta x_k + (1-\beta)z_k$ used in this assignment is non-standard and does not achieve the $O(1/k)$ convergence rate guarantee. The decaying schedule $\gamma_k = 2/(k+2)$ (equivalently $\beta_k = k/(k+2)$) is the theoretically principled choice.

\subsection{Interior vs.\ Boundary Behaviour: Comparative Discussion}

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits characteristic ``zig-zag'' oscillations. Since $z_k$ always selects a \emph{vertex} of $\mathcal{X}$, and the interior optimum is far from any vertex, the convex combinations $x_{k+1} = \beta x_k + (1-\beta)z_k$ converge slowly as the averaging process brings the iterate towards the interior. The general $O(1/k)$ rate is evident. The standard Frank--Wolfe schedule $\gamma_k = 2/(k+2)$ (instead of fixed $\beta$) would achieve faster convergence in theory.

\textbf{Boundary optimum:} Frank--Wolfe is naturally well-suited because the LP vertex $z_k$ frequently selects the vertex \emph{nearest} to the constrained optimum ($(0.5, -5)$ in this case), and convex combinations with this vertex pull the iterate towards the constraint boundary. The iterate naturally approaches the boundary without any projection, which is the key advantage over projected gradient descent when the projection is expensive. However, the fixed $\beta = 0.93$ creates residual oscillations --- a decaying step sequence would reduce this.

\textbf{Frank--Wolfe advantages:}
\begin{itemize}[nosep]
\item Projection-free: only requires solving a linear programme over $\mathcal{X}$, which is $O(d)$ for boxes.
\item Natural for structured constraint sets where LP is easy (e.g., nuclear norm, flow polytopes) but projection is hard.
\item Iterates are always convex combinations of extreme points, which is important when sparsity of the solution is desired.
\end{itemize}

\textbf{Frank--Wolfe disadvantages:}
\begin{itemize}[nosep]
\item $O(1/k)$ convergence rate is slower than projected gradient descent's rate when the objective is smooth and strongly convex.
\item The LP vertex oscillation near non-vertex optima creates a ``zig-zagging'' that limits final accuracy.
\item The fixed $\beta$ formulation here is non-standard; the classical line-search or $\gamma_k = 2/(k+2)$ schedule would be more principled.
\end{itemize}

\subsection{Key Takeaways}

\begin{enumerate}
\item \textbf{No universally best method:} The Polyak step size is optimal when $f^\star$ is known (Benchmark C, $f = 0.009$) but catastrophically unstable when $f^\star$ is misspecified. Newton's method is fastest on quadratics (single iteration) but expensive per step. The choice of method must account for the problem's convexity, condition number, differentiability, and dimensionality.

\item \textbf{Condition number drives difficulty:} Benchmark A ($\kappa \approx 1.1$) is trivial for all methods; Benchmark B ($\kappa \approx 5.6$) is moderately challenging; Benchmark C ($\kappa \approx 2500$) separates methods dramatically. On ill-conditioned problems, preconditioning (Newton's method) or curvature-aware momentum (Nesterov) are essential.

\item \textbf{Adaptive methods are not always adaptive enough:} Adagrad and RMSprop adapt per-coordinate step sizes but cannot account for off-diagonal Hessian structure (cross-coordinate coupling). Newton's method, by inverting the full Hessian, handles all curvature information.

\item \textbf{Stochastic noise creates an irreducible floor:} Mini-batch SGD converges to the noise floor $J^\star = \sigma^2/2$, which is determined by the data noise $\sigma$, not the batch size or learning rate. Reducing $\sigma$ (better data) is the only way to lower this floor.

\item \textbf{Constraints change the geometry:} Projected gradient descent enforces feasibility exactly at every iteration, making it preferable when feasibility is a hard requirement. Penalty methods offer simplicity at the cost of approximate feasibility and instability at large $\lambda$.

\item \textbf{Frank--Wolfe is projection-free but slower:} Its $O(1/k)$ rate is inferior to projected gradient descent's accelerated variants, but it is uniquely suited to constraint sets where the LP oracle is efficient (flow polytopes, nuclear norm balls, etc.) while projection is expensive.

\item \textbf{Derivative-free methods scale differently:} For large $d$, Nesterov random search (2 evaluations per step) is the only scalable derivative-free method. Grid search is strictly limited to $d \leq 2$; Nelder--Mead to $d \leq 5$.
\end{enumerate}

\subsection{Connections to Course Theory}

The algorithms studied span the full hierarchy of optimisation theory:
\begin{itemize}[nosep]
\item \textbf{$O(1/k)$ rate}: vanilla GD (convex), Frank--Wolfe.
\item \textbf{$O(1/k^2)$ rate}: Nesterov acceleration (optimal for first-order methods on smooth convex problems).
\item \textbf{Linear rate} $(1-\mu/L)^k$: GD on strongly convex problems; Heavy Ball with optimal parameters.
\item \textbf{Quadratic rate}: Newton's method near the minimiser.
\end{itemize}
These rates form a hierarchy: quadratic $\gg$ linear $\gg$ $O(1/k^2)$ $\gg$ $O(1/k)$. However, each faster rate comes with a cost: Newton requires computing and inverting the Hessian ($O(d^3)$ per iteration); Nesterov requires convexity and knowledge of $L$; linear rates require strong convexity with known $\mu$. The art of optimisation lies in matching algorithm properties to problem structure.

Further connections to course material:
\begin{itemize}[nosep]
\item \textbf{Oracle complexity lower bound:} Nesterov's $O(1/k^2)$ rate is tight --- no first-order method can achieve a better rate for all smooth convex functions (information-theoretic lower bound). This means that FISTA, Nesterov momentum, and accelerated variants are all \emph{optimal} in the oracle complexity sense.
\item \textbf{Heavy Ball vs Nesterov on quadratics:} For purely quadratic objectives, both Heavy Ball (with optimal $\alpha^\star, \beta^\star$) and Nesterov achieve the same optimal rate $\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^k$. On non-quadratic objectives, Nesterov's convergence guarantee holds but Heavy Ball can diverge, explaining why Nesterov is preferred in general.
\item \textbf{SGD rates:} For SGD with batch size $b$ and constant step $\alpha$, the convergence rate is $O(1/k)$ on strongly convex problems but with an irreducible noise floor $O(\sigma^2/b)$, balanced only by diminishing step sizes (which kill the rate). This tension motivates variance reduction methods (SVRG, SARAH) not covered in this assignment.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)

def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)

def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):

\begin{lstlisting}[caption={Nesterov Momentum / Acceleration}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Hessian Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

def mini_batch_sgd(X, y, theta0, alpha, batch_size, n_epochs, seed=42):
    rng = np.random.RandomState(seed)
    theta = theta0.copy().astype(float)
    n = len(y)
    loss_fn_local = lambda th: 0.5 * np.mean((X @ th - y)**2)
    epoch_losses = [loss_fn_local(theta)]
    for _ in range(n_epochs):
        idx = rng.permutation(n)
        for i in range(0, n, batch_size):
            batch_idx = idx[i:i+batch_size]
            Xb, yb = X[batch_idx], y[batch_idx]
            g = Xb.T @ (Xb @ theta - yb) / len(yb)
            theta = theta - alpha * g
        epoch_losses.append(loss_fn_local(theta))
    return np.array(epoch_losses)
\end{lstlisting}

def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
