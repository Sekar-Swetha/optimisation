\begin{abstract}
% ============================================================
This report presents a comprehensive comparative study of six categories of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C). The methods investigated are: (1)~adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy~Ball); (2)~momentum-based methods (Nesterov acceleration, Adam) and mini-batch stochastic gradient descent; (3)~Newton's method with second-order local approximations; (4)~derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead simplex, grid search); (5)~constrained optimisation via projected gradient descent and exterior penalty methods; and (6)~the Frank--Wolfe conditional gradient algorithm for linearly constrained problems. Key findings include: the Polyak step size achieves the fastest convergence on Benchmark~C (where $f^{\star}=0$ is known exactly) but diverges when $f^{\star}$ is misspecified; Nesterov acceleration outperforms Adam on the ill-conditioned Rosenbrock function ($f = 0.479$ vs $f = 1.578$ after 150 iterations); Newton's method solves the quadratic regression loss in a single step; finite-difference gradient approximation with $\delta = 0.05$ closely tracks exact gradient descent; and projected gradient descent enforces hard constraint satisfaction from iteration~1, while penalty methods enforce feasibility progressively with $\lambda$-dependent speed.
\end{abstract}

\subsection{Benchmark Problems}
\label{sec:benchmarks}

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for exact reproducibility.

\subsubsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and two parameters is generated as:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a \emph{strongly convex quadratic} with constant Hessian $H = \frac{1}{m}X^T X$ and gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the minimum loss is $J^{\star} \approx 0.4826$ (not zero), attained at $\theta_{\mathrm{OLS}} = (X^TX)^{-1}X^Ty$. The condition number $\kappa(H) \approx 1.15$ makes this the easiest benchmark. Starting point: $\theta_0 = [0, 0]^T$.

\subsubsection{Benchmark B: Toy Neural Network Loss}

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is
\begin{equation}
H(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The diagonal structure means curvature along $x_1$ varies between $1$ and $3$ depending on position (as $\sin x_1 \in [-1,1]$), while curvature along $x_2$ is fixed at 10. The effective condition number is approximately $\kappa \approx 10/1 = 10$, creating moderate anisotropy. The sinusoidal term introduces mild non-convexity in $x_1$. The minimum lies near $(0.582, 2.0)$ with $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsubsection{Benchmark C: Rosenbrock Function}

\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian is:
\begin{equation}
H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum, $H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}$ with eigenvalues $\approx 0.5$ and $\approx 1001$, giving $\kappa \approx 2002$. This extreme ill-conditioning, combined with the narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock the most challenging standard test function. Small gradient steps must be taken to maintain stability ($\alpha \lesssim 2/L$ where $L$ is the Lipschitz constant of $\nabla f$, which is very large near the cusp). Starting point: $x_0 = [-1, 1]^T$.

\subsubsection{Benchmark D: Constrained Feasible Set}

For constrained optimisation (Questions 5 and 6):
\begin{equation}
\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a simple box (hyperrectangle), whose Euclidean projection decomposes per coordinate:
$[\Pi_{\mathcal{X}}(x)]_i = \max(l_i, \min(u_i, x_i))$.

\subsection{Theoretical Background}

Standard gradient descent uses a fixed step size $\alpha$, requiring $\alpha \leq 2/L$ for convergence where $L$ is the smoothness parameter. Adaptive methods adjust $\alpha_k$ automatically, either using knowledge of the function value gap (Polyak) or accumulating gradient statistics to scale each coordinate independently (Adagrad, RMSprop, Heavy Ball).

\begin{theorem}[Gradient Descent Convergence]
For an $L$-smooth convex function with gradient descent and step $\alpha = 1/L$:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{L\|x_0 - x^{\star}\|^2}{2k}.
\end{equation}
For $\mu$-strongly convex functions: $f(x_k) - f^{\star} \leq (1 - \mu/L)^k (f(x_0) - f^{\star})$, i.e., linear convergence with rate $(1-1/\kappa)$.
\end{theorem}

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Optimal step towards level set $\{f \leq f^{\star}\}$}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Polyak Derivation]
The Polyak step size minimises the squared distance to the optimal level set $\{x : f(x) \leq f^{\star}\}$ within the gradient direction. For a Lipschitz-continuous objective, it achieves $O(1/k)$ convergence without knowing the Lipschitz constant, provided $f^{\star}$ is known exactly. The update can be derived as:
\begin{equation}
\alpha_k^{\mathrm{Polyak}} = \arg\min_{\alpha \geq 0} \left\| (x_k - \alpha g_k) - x^{\star} \right\|^2 \quad \text{(linearised)}.
\end{equation}
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$ \Comment{Coordinate-wise adaptive scaling}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Adagrad Properties]
Adagrad implicitly performs online learning on a diagonal pre-conditioned problem. Coordinates with large historical gradients receive smaller effective step sizes, automatically balancing learning across directions of different curvature. The monotonically decreasing effective rate $\alpha_k^{(i)} = \alpha_0/\sqrt{\sum_{j=0}^k g_j^{(i)2}}$ is both a strength (stability) and weakness (premature stagnation in non-stationary settings).
\end{remark}

\begin{remark}[Adagrad Stagnation Rate]
Under the assumption that gradient magnitudes stabilise at $|g_\infty^{(i)}|$ for large $k$, the cumulative second-moment $G_k^{(i)} \approx k\,|g_\infty^{(i)}|^2$, giving:
\begin{equation}
\alpha_k^{(i)} \approx \frac{\alpha_0}{|g_\infty^{(i)}| \sqrt{k}} \qquad (k \text{ large}).
\end{equation}
The total displacement $\sum_{k=0}^{K} \alpha_k^{(i)} \approx \frac{2\alpha_0}{|g_\infty^{(i)}|}\sqrt{K} \to \infty$, so Adagrad is not ``stuck'' in the sense that convergence remains possible as $K \to \infty$. However, the $O(\sqrt{k})$ cumulative learning budget means progress decelerates severely: halving the remaining gap requires \emph{quadrupling} the number of iterations. This matches Adagrad's optimal $O(\sqrt{T})$ regret bound in the online convex optimisation setting. In our Benchmark~C experiments, the stagnating step size (Figure~\ref{fig:q1_stepsize_C}) causes Adagrad to ``freeze'' well above the Rosenbrock minimum, whereas RMSprop's exponential forgetting ($\beta = 0.9$, window $\approx 10$ iterations) allows the step size to partially recover when gradient magnitudes decrease.
\end{remark}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[RMSprop vs Adagrad]
RMSprop fixes Adagrad's vanishing step-size problem by replacing cumulative summation with an exponential moving average. The effective memory window has length $\approx 1/(1-\beta)$ iterations. With $\beta = 0.9$, only the last $\sim$10 gradient magnitudes significantly influence the current scale, allowing the step size to recover if the gradient magnitude decreases.
\end{remark}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Velocity accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Heavy Ball Interpretation]
The Heavy Ball method (Polyak, 1964) can be interpreted as a discretisation of a damped second-order ODE: $\ddot{x}(t) + (1-\beta)\dot{x}(t) = -\alpha\nabla f(x(t))$. The momentum term $\beta z_k$ carries forward information about previous gradients, effectively reducing the zig-zag oscillation of gradient descent on ill-conditioned problems. For a quadratic $f$ with Hessian eigenvalues in $[\mu, L]$, the optimal Heavy Ball parameters achieve the convergence rate $(\kappa^{1/2}-1)/(\kappa^{1/2}+1)$ per step (vs $(\kappa-1)/(\kappa+1)$ for GD), a significant improvement when $\kappa \gg 1$.
\end{remark}

\subsubsection{Step Size Rationale: Lipschitz Constant Analysis}

The GD step sizes are chosen to satisfy the stability condition $\alpha \leq 2/L$, where $L$ is the Lipschitz constant of $\nabla f$:
\begin{itemize}[nosep]
\item \textbf{Benchmark A:} $L = \lambda_{\max}(H) = \lambda_{\max}(X^TX/m) \approx 1.03$ (estimated from the data), so $2/L \approx 1.94$. The chosen $\alpha = 0.08 \ll 2/L$, providing a generous stability margin, since the data matrix is near-isotropic ($\kappa \approx 1.15$ implies $\lambda_{\min} \approx 0.90$, $\lambda_{\max} \approx 1.03$). A larger step would work but does not significantly accelerate convergence on this well-conditioned problem.
\item \textbf{Benchmark B:} $L = \max_{x} \lambda_{\max}(H(x)) = \max_{x}(2 - \sin(x_1), 10) = 10$ along the trajectory. The stability bound gives $\alpha \leq 0.2$; the chosen $\alpha = 0.06$ is conservative to ensure stability given the non-quadratic structure.
\item \textbf{Benchmark C:} Near the starting point $(-1, 1)$, $H_{11} = 2 + 1200 \times 1 - 400 \times 1 = 802$, giving $L_{\max} \approx 1000$. The stability bound $2/L \approx 0.002$, and $\alpha = 0.0012$ is chosen slightly below this, giving GD a narrow safety margin against the extreme curvature.
\end{itemize}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$, $120$ iterations). Among universally applicable methods, Heavy Ball ($f = 1.40$) outperforms GD, Adagrad, and RMSprop on Rosenbrock due to momentum's ability to accelerate along the consistent valley-floor direction.
\item \textbf{Stability and robustness:} GD baseline is most stable but slowest. Polyak is \emph{unstable} when $f^{\star}$ is misspecified --- a fundamental limitation that prevents its practical deployment. Adagrad is uniformly stable due to its monotonically decreasing step size, though this causes stagnation on non-stationary or high-dimensional problems. RMSprop offers a practical compromise with its tunable memory window.
\item \textbf{Anisotropy handling:} On Benchmark B (moderate anisotropy $\kappa \approx 10$), all adaptive methods improve over GD. On Benchmark C ($\kappa \approx 2000$), the advantage of Heavy Ball is clear: momentum reduces the \emph{effective} condition number experienced by the optimiser, since the accumulated velocity aligns with the dominant curvature direction.
\item \textbf{Practical implication:} In practice, exact $f^{\star}$ is rarely known. RMSprop and Adam (an extension that adds bias correction and momentum to RMSprop) are the preferred choices for deep learning. Adagrad remains useful for sparse gradient problems (e.g., NLP feature learning). The Polyak step is most applicable in specialised settings where tight lower bounds on $f^{\star}$ are available.
\end{itemize}

\subsubsection{Why Adagrad and RMSprop Did Not Win on Benchmark C}

On the Rosenbrock function, Adagrad ($f = 2.20$) and RMSprop ($f = 3.12$) underperform Heavy Ball ($f = 1.40$). This is explained by two factors:
\begin{enumerate}[nosep]
\item \textbf{Coordinate-independence assumption:} Both Adagrad and RMSprop scale each coordinate independently, ignoring off-diagonal Hessian terms. On Rosenbrock, the off-diagonal Hessian term $H_{12} = -400x_1$ is large and couples $x_1$ and $x_2$, making per-coordinate scaling an incomplete preconditioner.
\item \textbf{Step size magnitude:} The conservative base rates ($\alpha_0 = 0.45$ for Adagrad, $\alpha_0 = 0.0035$ for RMSprop on Benchmark C) are chosen to prevent instability in the early phase, but this limits the speed of valley traversal in the later phase. Heavy Ball with fixed step and momentum builds consistent velocity more effectively.
\end{enumerate}

\subsection{Theoretical Background}

Momentum methods accelerate gradient descent by accumulating a velocity vector that smooths the gradient signal. The key theoretical result is Nesterov's $O(1/k^2)$ convergence rate for convex functions, compared to GD's $O(1/k)$ --- a provably optimal rate among first-order methods on this function class.

\begin{theorem}[Nesterov Acceleration]
For an $L$-smooth convex function with Nesterov's accelerated gradient method using step $\alpha = 1/L$ and the canonical momentum schedule:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\end{equation}
This matches the minimax lower bound for first-order methods on convex functions.
\end{theorem}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update using lookahead gradient}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This seemingly minor change allows Nesterov's method to ``correct'' its trajectory before overshooting, enabling the optimal convergence rate. Intuitively, evaluating the gradient at the predicted future position provides a ``corrected'' gradient that partially accounts for the momentum-induced displacement. The adaptive schedule $\beta_k = (k-1)/(k+2) \nearrow 1$ increases momentum gradually, which is crucial for the $O(1/k^2)$ guarantee --- using fixed momentum from the start would not achieve the optimal rate.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction (eliminates warm-up bias)}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum (via $m_t$, analogous to Nesterov/Heavy Ball) with per-coordinate adaptive scaling (via $v_t$, analogous to RMSprop). The bias correction terms $1/(1-\beta_j^t)$ are essential during early iterations when the exponential averages are initialised to zero: without correction, $m_1 = (1-\beta_1)g_1$ would be much smaller than the true gradient signal, slowing early-iteration convergence.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at each epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Dataset $(X, y)$ with $m$ samples, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute indices $\{1,\ldots,m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$ from permuted indices}
        \State $\hat{g} \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The stochastic gradient $\hat{g}$ is an unbiased estimate of the full gradient $\nabla J(\theta)$, with variance inversely proportional to batch size: $\mathrm{Var}(\hat{g}) \propto \sigma^2/b$. The variance-speed trade-off is central to SGD behaviour.

More precisely, the stochastic gradient for a mini-batch $\mathcal{B}$ is:
\begin{equation}
\hat{g}(\theta; \mathcal{B}) = \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}}) \approx \nabla J(\theta) + \underbrace{\xi}_{\text{noise}}, \quad \mathbb{E}[\xi] = 0, \quad \mathbb{E}[\|\xi\|^2] = O(\sigma^2/b).
\end{equation}

For a $\mu$-strongly convex loss with $L$-smooth gradient, mini-batch SGD with constant step $\alpha \in (0, 1/L]$ satisfies at epoch $e$ (each epoch = $m/b$ updates):
\begin{equation}
\mathbb{E}\bigl[J(\theta_{e \cdot m/b}) - J^{\star}\bigr] \leq \underbrace{\left(1 - \alpha\mu\right)^{e \cdot m/b}\!\!\bigl(J(\theta_0) - J^{\star}\bigr)}_{\text{transient (geometrically decaying)}} + \underbrace{\frac{\alpha\sigma^2}{2\mu b}}_{\text{noise floor (irreducible)}}.
\label{eq:sgd_bound}
\end{equation}
The two terms in \eqref{eq:sgd_bound} govern a fundamental trade-off:
\begin{itemize}[nosep]
\item \textbf{Small batch ($b = 5$):} Each epoch performs $m/b = 1000/5 = 200$ updates. The transient decays as $(1-\alpha\mu)^{200}$ per epoch (fast). But the noise floor $\propto 1/b$ is $8\times$ larger than for $b = 40$.
\item \textbf{Large batch ($b = 40$):} Each epoch performs $m/b = 25$ updates. The transient decays $(1-\alpha\mu)^{25}$ per epoch (slow, $8\times$ fewer updates). The noise floor is $8\times$ smaller.
\end{itemize}
This explains the experimental observation: with $b = 5$, the loss drops rapidly in early epochs (200 gradient steps per epoch vs 25), but the two batches converge to the same ultimate value because both noise floors equal $J^{\star}_{\mathrm{true}} = J(\theta_{\mathrm{OLS}})$ (the irreducible observation-noise floor, independent of $b$). The observation noise $\sigma^2$ dominates the mini-batch sampling noise $\sigma^2/b$ in this experiment, so further increasing $b$ would not reduce the final loss.

The \emph{irreducible noise floor} is thus:
\begin{equation}
\liminf_{k \to \infty} J(\theta_k) \geq J^{\star} + \frac{\alpha \sigma^2}{2\mu b} \xrightarrow[\text{Bench.\ A}]{} J(\theta_{\mathrm{OLS}}) \approx 0.483.
\end{equation}
To reduce the floor by half: either halve $\alpha$ (but slower transient) or double $b$ (but more computation per update). In the high-noise ($\sigma = 6$) case, the floor rises to $\approx 17.4 \propto \sigma^2$, consistent with Figure~\ref{fig:q2_sgd_noisy}.

\subsection{Analysis: Nesterov vs Adam}

The experimental results reveal a fundamental difference between the two methods on ill-conditioned problems.

\begin{itemize}[nosep]
\item \textbf{Nesterov on Rosenbrock:} Momentum accumulates in a consistent direction along the valley floor. The Rosenbrock valley $\{(x_1,x_2): x_2 \approx x_1^2\}$ has a locally stable descent direction tangent to the parabola, and Nesterov's velocity vector $z_k$ progressively aligns with this direction after a few iterations. The lookahead step $\tilde{x}_k = x_k + \beta_k z_k$ then evaluates the gradient at a point already partially advanced along the valley, providing a corrected gradient that pre-empts overshooting. This is not possible for Heavy Ball, which evaluates at the current (un-corrected) position.
\item \textbf{Adam's diagonal failure on Rosenbrock:} Adam's adaptive scaling mimics the diagonal preconditioner $D_t^{-1} = \text{diag}(\hat{v}_t)^{-1/2}$, effectively approximating the \emph{diagonal entries} of $H^{-1}$. However, the ideal preconditioner for Rosenbrock is the full $H^{-1}(x)$, which has a large off-diagonal entry $H_{12} = -400x_1$ (coupling $x_1$ and $x_2$). The key insight is:
\begin{equation}
H^{-1}(x) \approx \frac{1}{\det H}\begin{pmatrix}200 & 400x_1 \\ 400x_1 & 802\end{pmatrix},
\end{equation}
which is far from diagonal near $x_1 = \pm 1$. Adam's diagonal approximation $D^{-1} = \text{diag}(1/\sqrt{E[g_1^2]}, 1/\sqrt{E[g_2^2]})$ discards this off-diagonal coupling, so its ``scaled gradient'' $D^{-1}g$ is not aligned with the true Newton direction $H^{-1}g$. This misalignment causes Adam to oscillate across the valley rather than progress along it, explaining the inferior $f = 1.579$ vs Nesterov's $f = 0.479$.
\item \textbf{Adam's superiority on Benchmark B:} Benchmark B has a nearly diagonal Hessian: $H(x) = \text{diag}(2-\sin(x_1), 10)$ with small off-diagonal terms. Here, Adam's diagonal approximation is accurate, and per-coordinate scaling correctly re-scales the gradient to handle the 5:1 curvature ratio. In this setting, Adam's effective update $\approx H_{\text{diag}}^{-1}g$ closely approximates the Newton direction, explaining why it matches Nesterov on Benchmark B.
\item The theoretical $O(1/k^2)$ acceleration of Nesterov applies to convex functions; Rosenbrock is non-convex, and the observed acceleration reflects the near-quadratic structure within the valley rather than global convexity. Adam has no convergence rate guarantee on non-convex functions that is competitive with Nesterov for this valley geometry.
\end{itemize}

\begin{remark}[Adam's Convergence Subtlety]
Adam is not guaranteed to converge to a stationary point on non-convex problems with the standard parameter settings. Reddi et al.\ (2018) showed that Adam can diverge even on simple convex functions when the step size is fixed; the AMSGrad variant (which uses a non-decreasing $\hat{v}_t$ denominator) restores convergence guarantees. In practice, Adam's empirical success in deep learning arises from the favourable loss landscape structure (saddle points rather than local minima), not from formal convergence theory.
\end{remark}

\subsubsection{Heavy Ball vs Nesterov: Theoretical Comparison}

On a quadratic $f(x) = \frac{1}{2}x^T H x - b^T x$ with eigenvalues $\mu \leq \lambda_i \leq L$ (condition number $\kappa = L/\mu$), both methods can be analysed exactly:

\begin{itemize}[nosep]
\item \textbf{GD} with optimal $\alpha = 2/(L+\mu)$: contraction rate per step $\rho_{\mathrm{GD}} = (\kappa-1)/(\kappa+1)$.
\item \textbf{Heavy Ball} with optimal $\alpha = 4/((\sqrt{L}+\sqrt{\mu})^2)$, $\beta = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$: contraction rate $\rho_{\mathrm{HB}} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$, achieved per step.
\item \textbf{Nesterov} with $\alpha = 1/L$: $f(x_k) - f^{\star} \leq O(1/k^2)$ (not linear convergence, but better constant).
\end{itemize}

For $\kappa \approx 10$ (Benchmark B), $\rho_{\mathrm{GD}} = 9/11 \approx 0.818$ and $\rho_{\mathrm{HB}} = (\sqrt{10}-1)/(\sqrt{10}+1) \approx 0.52$, so Heavy Ball needs roughly $\log(0.52)/\log(0.818) \approx 3.3 \times$ fewer iterations than GD to reach the same accuracy. Nesterov's adaptive momentum schedule achieves an even better rate in the convex setting. These theoretical predictions are consistent with the observed experimental advantage of Nesterov and Heavy Ball over GD across all benchmarks.

\subsubsection{Adam's $\beta_1$ Hyperparameter}

The $\beta_1$ values chosen here (0.80--0.82 vs standard 0.9) are intentionally lower to reduce momentum accumulation. On the smooth benchmark functions, higher momentum can cause overshooting and instability near the minimum. The lower $\beta_1$ trades some momentum benefit for more stable fine-convergence behaviour. The standard $\beta_1 = 0.9$ is optimal for deep learning's noisy gradient environments where momentum serves as a variance-reduction mechanism; in our deterministic setting, lower momentum avoids overshooting the sharp minimum.

\subsection{Local Approximations of $g(x) = x^4$}

For a smooth function $g: \mathbb{R} \to \mathbb{R}$, the Taylor expansion around $x_0$ gives:
\begin{align}
g(x) &\approx g(x_0) + g'(x_0)(x - x_0) && \text{(first-order / linear)} \\
g(x) &\approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 && \text{(second-order / quadratic)}.
\end{align}

For $g(x) = x^4$ at $x_0 = 0.25$: $g(0.25) = 0.25^4 = 0.00391$, $g'(0.25) = 4 \times 0.25^3 = 0.0625$, $g''(0.25) = 12 \times 0.25^2 = 0.75$.

\textbf{First-order approximation:}
\begin{equation}
g(x) \approx 0.00391 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
g(x) \approx 0.00391 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton update minimises the second-order approximation:
\begin{equation}
x^+ = \arg\min_x \left[g(x_0) + g'(x_0)(x-x_0) + \tfrac{1}{2}g''(x_0)(x-x_0)^2\right] = x_0 - \frac{g'(x_0)}{g''(x_0)}.
\end{equation}

For $g(x) = x^4$: $x^+ = x_0 - \frac{4x_0^3}{12x_0^2} = x_0 - \frac{x_0}{3} = \frac{2}{3}x_0$. Starting from $x_0 = 0.25$: $x^+ = 0.167$, which is closer to the true minimum $x^{\star} = 0$ than the gradient step $x_0 - \alpha g'(x_0) = 0.25 - 0.0625\alpha$.

\subsection{Newton's Method with Damped Step}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0,1]$, Hessian damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise Hessian: ensures positive definiteness}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{$\alpha = 1$: full Newton; $\alpha < 1$: damped Newton}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Convergence Rates of Newton's Method]
Under standard regularity conditions (Lipschitz-continuous Hessian), the undamped Newton method ($\alpha = 1$) achieves \emph{quadratic} local convergence:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq C \|x_k - x^{\star}\|^2,
\end{equation}
where $C$ depends on the Hessian Lipschitz constant and the smallest eigenvalue of $\nabla^2 f(x^{\star})$. This means that once $\|x_k - x^{\star}\|$ is small (within the ``quadratic convergence regime''), the number of correct decimal digits roughly doubles per iteration. For exact quadratic functions, Newton achieves the minimum in exactly one step.
\end{remark}

Settings: GD runs for 80 iterations; Newton runs for 20 iterations with Hessian damping $\lambda = 10^{-8}$ and per-benchmark step sizes.

\subsubsection{Newton Decrement and Stopping Criterion}

A natural, scale-invariant stopping criterion for Newton's method is the \emph{Newton decrement}:
\begin{equation}
\lambda(x)^2 := \nabla f(x)^T H^{-1}(x) \nabla f(x) = p(x)^T H(x) p(x),
\end{equation}
where $p(x) = H^{-1}\nabla f(x)$ is the Newton direction. The decrement satisfies:
\begin{equation}
f(x) - f(x - p(x)) \approx \tfrac{1}{2}\lambda(x)^2, \quad \text{and} \quad f(x) - f^{\star} \leq \tfrac{1}{2}\lambda(x)^2.
\end{equation}
Stopping when $\lambda(x)^2/2 \leq \varepsilon_{\mathrm{tol}}$ guarantees the iterate is within $\varepsilon_{\mathrm{tol}}$ of optimal. This criterion was not explicitly used in the code (a fixed number of iterations was used instead), but the update magnitude plot (Figure~\ref{fig:q3_update}) serves as a proxy: small update norms indicate small Newton decrements.

\subsection{Computational Cost Comparison}

Newton's method requires computing and factorising the $d \times d$ Hessian at each iteration, costing $O(d^3)$ per step (plus $O(d^2)$ for the solve). For small $d = 2$, this is negligible. However, for high-dimensional problems ($d \sim 10^6$ in deep learning), the Hessian is too expensive to store ($O(d^2)$ memory) or compute, motivating quasi-Newton methods (L-BFGS) that approximate $H^{-1}$ using gradient differences.

\subsection{Newton's Method: Global vs.\ Local Convergence}

Newton's method exhibits a fundamental two-phase convergence behaviour not shared by first-order methods.

\textbf{Phase~1 --- Global phase (damped Newton).} Far from $x^{\star}$, the local quadratic model $m_k(d) = f(x_k) + \nabla f(x_k)^T d + \frac{1}{2}d^T H(x_k)\, d$ may not accurately represent $f$ over the full Newton step $d_k^N = H^{-1}(x_k)\nabla f(x_k)$. A concrete example on Benchmark~C illustrates the danger. At $x_0 = (-1, 1)$ the Rosenbrock function evaluates to $f(x_0) = 100(1 - 1)^2 + (1+1)^2 = 4$, and:
\begin{align}
\nabla f(x_0) &= \bigl(-400(-1)(1-1) - 2(1-(-1)),\; 200(1-1)\bigr)^T = (-4,\;0)^T, \\
H(x_0) &= \begin{pmatrix} 1200(-1)^2 - 400(1) + 2 & -400(-1) \\ -400(-1) & 200 \end{pmatrix} = \begin{pmatrix} 802 & 400 \\ 400 & 200 \end{pmatrix}.
\end{align}
Solving $H(x_0)\,d = \nabla f(x_0)$ gives $d_0^N = (-2,\, 4)^T$, placing the full Newton iterate at $x_1^N = x_0 - d_0^N = (1,\,-3)$, where
\[
f(1,-3) = 100(-3 - 1)^2 + (1-1)^2 = 1600,
\]
an \emph{increase} of $400\times$ over $f(x_0) = 4$. This catastrophic failure demonstrates that the quadratic model is valid only within a small neighbourhood of $x_k$; the full step of magnitude $\|d_0^N\| = \sqrt{4+16} \approx 4.5$ far exceeds this neighbourhood.

The damping factor $\alpha = 0.22$ prevents this failure by confining the step to a region of model accuracy. A rigorous alternative is \emph{Armijo backtracking}: choose the largest $\alpha_k \in \{1, \rho, \rho^2, \ldots\}$ satisfying
\begin{equation}
f(x_k - \alpha_k d_k^N) \leq f(x_k) - c\,\alpha_k\,\nabla f(x_k)^T d_k^N, \quad c \in (0, \tfrac{1}{2}),\; \rho \in (0,1),
\label{eq:armijo}
\end{equation}
which guarantees $f(x_{k+1}) < f(x_k)$ for all $k$ and global convergence to a stationary point at a linear rate during the global phase.

\textbf{Phase~2 --- Local phase (pure Newton, quadratic convergence).} Once $x_k$ enters the basin of attraction of $x^{\star}$ --- the region where the Hessian mapping $\nabla^2 f$ is Lipschitz-continuous with constant $M$ over the full Newton step --- the pure Newton step ($\alpha_k = 1$) achieves:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\,\lambda_{\min}(H(x^{\star}))}\,\|x_k - x^{\star}\|^2,
\label{eq:newton_quadratic_cv}
\end{equation}
where $\lambda_{\min}(H(x^{\star}))$ is the smallest eigenvalue of the Hessian at the optimum. The quadratic bound means the number of correct significant digits \emph{doubles} at each iteration: if $\|x_k - x^{\star}\| \sim 10^{-3}$, then $\|x_{k+1} - x^{\star}\| \lesssim 10^{-6}$. In our Benchmark~C experiments (20 iterations, $\alpha = 0.22$ throughout), the iterates do not reach this local phase --- the starting point $x_0 = (-1,1)$ is far from $x^{\star} = (1,1)$, and heavy damping keeps the method in the global phase for all 20 steps, explaining the approximately \emph{linear} (rather than quadratic) observed convergence in Figure~\ref{fig:q3_fval_C}.

\begin{remark}[Newton Decrement as Stopping Criterion]
The Newton decrement $\lambda(x)^2 = \nabla f(x)^T H^{-1}(x)\nabla f(x)$ provides a scale-invariant proximity-to-optimum certificate:
\begin{equation}
f(x) - f^{\star} \lesssim \frac{\lambda(x)^2}{2} \qquad (\text{tight as } x \to x^{\star}).
\end{equation}
Stopping when $\lambda(x)^2/2 < \varepsilon_{\mathrm{tol}}$ certifies $\varepsilon_{\mathrm{tol}}$-optimality without knowledge of $f^{\star}$ --- a major advantage over the gradient norm $\|\nabla f(x_k)\|$, which is not scale-invariant. For a poorly scaled function (e.g., $f(x) = 10^6 g(x)$), the gradient can be large even near the minimum, but $\lambda(x)^2$ correctly identifies proximity by accounting for the Hessian scaling.
\end{remark}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

When the analytical gradient is unavailable (e.g., simulation-based objectives, black-box functions), the partial derivatives can be approximated by finite differences:

\textbf{Forward difference:}
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \qquad \text{error} = O(\delta) + O\!\left(\frac{\varepsilon_{\mathrm{mach}}}{\delta}\right).
\label{eq:fd}
\end{equation}

The total error in \eqref{eq:fd} is minimised by balancing truncation error $O(\delta)$ against floating-point cancellation error $O(\varepsilon_{\mathrm{mach}}/\delta)$, giving optimal $\delta^{\star} = \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$. The assignment tests $\delta = 0.05$ (reasonably close to optimal for this smooth function) and $\delta = 0.8$ (far too large, causing substantial truncation error). Each full gradient evaluation requires $d$ forward-difference function evaluations (plus one baseline), for $d+1 = 3$ total per iteration.

\textbf{Central difference} (not used here but more accurate):
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}, \qquad \text{error} = O(\delta^2).
\end{equation}

\subsubsection{Nesterov Random Search (Zeroth-Order Method)}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$ (uniform on $S^{d-1}$)
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method requires only \emph{two} function evaluations per iteration regardless of dimension $d$, making it dimension-independent in per-step cost. The key property is that $\mathbb{E}[\hat{g}_k] = \mathbb{E}_{u}[\nabla f(x_k)^T u\, u] = \nabla f(x_k)/d$ (the gradient scaled by $1/d$), so the method is an unbiased gradient estimator up to a scaling factor. The effective step size is $\alpha/d$, requiring larger $\alpha$ to compensate.

The convergence rate is $O(d/k)$ for smooth convex functions (compared to $O(1/k)$ for first-order methods), reflecting the $d$-dimensional gradient estimation from a single directional probe. This is the fundamental cost of zeroth-order optimisation.

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively improves it using four geometric operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection:} Move the worst vertex $x_h$ to $x_r = (1+\alpha_r)\bar{x} - \alpha_r x_h$ (here $\alpha_r = 1$), where $\bar{x}$ is the centroid of the remaining vertices.
\item \textbf{Expansion:} If reflection improves on the best vertex, extend further: $x_e = (1+\gamma)\bar{x} - \gamma x_h$ ($\gamma = 2$).
\item \textbf{Contraction:} If reflection is worse than the second-worst, contract: $x_c = \bar{x} + \rho(x_h - \bar{x})$ ($\rho = 0.5$).
\item \textbf{Shrink:} If contraction fails, pull all vertices towards the best by factor $\sigma = 0.5$.
\end{enumerate}

No gradient information is required. The simplex adapts its shape to the local landscape, elongating along descent directions and contracting near the minimum. Initial simplex: $x_0 = (-1, 1)$, $x_0 + 0.35\,e_1 = (-0.65, 1)$, $x_0 + 0.35\,e_2 = (-1, 1.35)$. Total iterations: 160.

\begin{remark}[Nelder--Mead Convergence Theory]
Nelder--Mead lacks a complete convergence theory for $d \geq 2$. McKinnon (1998) constructed a pathological $\mathbb{R}^2$ example where Nelder--Mead converges to a non-stationary point (a non-critical point of $f$). Kelley (1999) identified a \emph{sufficient decrease} condition --- the simplex must contract geometrically as it approaches the minimum --- that guarantees convergence but may not hold in general. The per-iteration cost is at most $d + 2 = 4$ function evaluations (for reflection + contraction + shrink), but the number of iterations to reach $\varepsilon$-accuracy has no polynomial bound in $d$ in the worst case. In practice, Nelder--Mead is empirically reliable for $d \leq 5$--$10$ and smooth unimodal functions, but can stagnate on multi-modal or noisy objectives.

Each iteration replaces at most one vertex ($O(d)$ work), so the computational cost per step is $O(d \cdot \text{FE})$ where $\text{FE} \leq d+2$ is the number of function evaluations. For our $d = 2$ problem, this is at most 4 evaluations per iteration and 640 total for 160 iterations. Compare with grid search's $55^2 = 3025$ evaluations: Nelder--Mead requires $5\times$ fewer function evaluations while achieving far lower final values (because it \emph{adapts} its search to the landscape rather than covering it uniformly).
\end{remark}

\subsubsection{Zeroth-Order Oracle Complexity}

A fundamental result in optimisation theory compares the query complexity (number of function evaluations) required by first-order vs zeroth-order methods to achieve $\varepsilon$-optimality on $L$-smooth convex functions:
\begin{align}
\text{First-order (gradient access):} &\quad K_{\mathrm{FO}} = O\!\left(\sqrt{L/\varepsilon}\right) \text{ (Nesterov acceleration)} \\
\text{Zeroth-order (function-value access):} &\quad K_{\mathrm{ZO}} = O\!\left(d \cdot \sqrt{L/\varepsilon}\right) \text{ (random search)}
\end{align}
The $d$-fold gap reflects the cost of estimating a $d$-dimensional gradient from scalar function evaluations. For our $d = 2$ problems, the gap is only $2\times$, making zeroth-order methods quite competitive. However, for high-dimensional problems ($d \sim 10^6$ in deep learning), zeroth-order methods become impractical and first-order methods are essential. This explains why automatic differentiation (which provides exact gradients at $O(d)$ cost) is transformative for modern machine learning.

\subsubsection{Finite Difference Error Analysis}

The approximation error of the forward finite difference formula is precisely:
\begin{equation}
\left|\frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| \leq \underbrace{\frac{\delta}{2}\max_{t \in [0,1]}\left|\frac{\partial^2 f}{\partial x_i^2}(x + t\delta e_i)\right|}_{\text{truncation error}} + \underbrace{\frac{2\epsilon_{\mathrm{mach}}\|f\|_\infty}{\delta}}_{\text{cancellation error}},
\end{equation}
where $\epsilon_{\mathrm{mach}} \approx 10^{-16}$. For Benchmark B near $x_0 = (-1,4)$: $\partial^2 f/\partial x_1^2 = 2 - \sin(x_1) \approx 2.84$ and $\|f(x_0)\| \approx 45$. At $\delta = 0.05$: truncation error $\approx 0.05 \times 2.84 / 2 \approx 0.071$; at $\delta = 0.8$: truncation error $\approx 0.8 \times 2.84 / 2 \approx 1.14$. Since the true gradient magnitude $\|\nabla f(x_0)\| \approx 24$, the large-$\delta$ error of $1.14$ introduces a relative gradient error of $\sim$5\%, sufficient to significantly bias the descent direction and explain the observed poor convergence.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Setup}

Minimise Benchmark B subject to the constraint $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$:
\begin{equation}
\min_{x \in \mathcal{C}} f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1), \quad \mathcal{C} = \{x : x_1 \geq 0.5\}.
\end{equation}

Since the unconstrained minimum $x^{\star}_{\mathrm{unc}} = (0.582, 2.0)$ satisfies $x_1^{\star} = 0.582 > 0.5$, the constraint is \emph{inactive} at the solution: the constrained and unconstrained minima coincide. This means Projected GD will eventually reach the same minimum as unconstrained GD, but must traverse the feasible region.

\subsection{Methods}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1}) = \max(0.5,\, z_{k+1,1}),\, z_{k+1,2})$ \Comment{Hard feasibility enforcement}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Exterior Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $\nabla F_{\lambda}(x) \gets \nabla f(x) - \lambda \mathbf{1}[x_1 < 0.5]\, e_1$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Theoretical Distinction]
Projected GD solves the original constrained problem exactly (maintains hard feasibility at every step). The penalty method solves an unconstrained surrogate and converges to the constrained solution only as $\lambda \to \infty$. For finite $\lambda$, there remains a residual feasibility error of order $O(1/\lambda)$ unless the constraint is inactive at the solution (as in our case, where $x_1^{\star} = 0.582 > 0.5$, so both methods converge to the same point).
\end{remark}

Settings: Projected GD ($\alpha = 0.08$, 100 iterations); Penalty GD ($\lambda \in \{0.15, 1.8, 4.5\}$, step sizes $\alpha = 0.05$ for $\lambda \leq 1.8$, $\alpha = 0.03$ for $\lambda = 4.5$, 100 iterations); Unconstrained GD ($\alpha = 0.07$, 100 iterations).

\subsection{Convergence Guarantees}

\begin{theorem}[Projected Gradient Descent Convergence]
\label{thm:pgd}
Let $f : \mathbb{R}^d \to \mathbb{R}$ be $L$-smooth and convex, and let $\mathcal{C}$ be a closed convex set with constrained minimiser $x^{\star} = \arg\min_{x \in \mathcal{C}} f(x)$. Projected GD with step size $\alpha \in (0, 1/L]$ satisfies:
\begin{equation}
f(x_k) - f(x^{\star}) \leq \frac{\|x_0 - x^{\star}\|^2}{2\alpha k}.
\label{eq:pgd_rate}
\end{equation}
If additionally $f$ is $\mu$-strongly convex on $\mathcal{C}$ (with $\kappa = L/\mu$), convergence is linear:
\begin{equation}
\|x_k - x^{\star}\|^2 \leq \left(1 - \frac{\mu}{L}\right)^k \|x_0 - x^{\star}\|^2.
\label{eq:pgd_linear}
\end{equation}
\end{theorem}

\textit{Proof sketch.} From $L$-smoothness: $f(x_{k+1}) \leq f(x_k) + \nabla f(x_k)^T(x_{k+1}-x_k) + \frac{L}{2}\|x_{k+1}-x_k\|^2$. The projection optimality condition $(\nabla f(x_k) + \frac{1}{\alpha}(x_{k+1}-x_k))^T(z - x_{k+1}) \geq 0$ for all $z \in \mathcal{C}$ and non-expansiveness of $\Pi_{\mathcal{C}}$ yield $\nabla f(x_k)^T(x_{k+1}-x^{\star}) \leq \frac{1}{2\alpha}(\|x_k - x^{\star}\|^2 - \|x_{k+1}-x^{\star}\|^2 - \|x_{k+1}-x_k\|^2)$. Summing over $k=0,\ldots,K-1$ and applying convexity ($f(x_k) - f^{\star} \leq \nabla f(x_k)^T(x_k-x^{\star})$) gives \eqref{eq:pgd_rate}. Strong convexity adds $-\frac{\mu}{2}\|x_k-x^{\star}\|^2$ to the descent lemma, giving the geometric bound \eqref{eq:pgd_linear}. $\square$

\textbf{Application to our experiment:} Benchmark~B has smoothness constant $L = \max(2 + \sin(0.582), 10) \approx 10$ (the $5(x_2-2)^2$ term dominates) and strong convexity $\mu \approx 2$ (from the $(x_1-1)^2$ and $5(x_2-2)^2$ terms far from the constraint). The starting distance $\|x_0 - x^{\star}\| = \|(0.2, 4.0) - (0.582, 2.0)\| \approx 2.05$. With $\alpha = 0.08$ (slightly below $1/L = 0.1$), \eqref{eq:pgd_rate} guarantees $f(x_{100}) - f^{\star} \leq \frac{2.05^2}{2 \times 0.08 \times 100} \approx 0.263$. The observed gap is $< 10^{-4}$ (essentially converged), far better than the bound predicts --- the strong convexity of Benchmark~B activates the linear rate \eqref{eq:pgd_linear}, which converges exponentially in $k$ rather than as $O(1/k)$.

\subsection{KKT Conditions and Constraint Activity}

The Karush--Kuhn--Tucker (KKT) necessary conditions for the constrained problem $\min_{x_1 \geq 0.5} f(x_1, x_2)$ are:
\begin{align}
\nabla f(x^{\star}) - \mu^{\star} e_1 &= 0 \quad (\text{stationarity, where } \mu^{\star} \geq 0) \\
x_1^{\star} - 0.5 &\geq 0 \quad (\text{primal feasibility}) \\
\mu^{\star}(x_1^{\star} - 0.5) &= 0 \quad (\text{complementary slackness}).
\end{align}
Since the unconstrained minimum of Benchmark B is at $(0.582, 2.0)$ with $x_1^{\star} = 0.582 > 0.5$, the constraint $x_1 \geq 0.5$ is \emph{inactive}. By complementary slackness, this requires $\mu^{\star} = 0$. Substituting into the stationarity condition: $\nabla f(0.582, 2.0) = [2(0.582-1) + \cos(0.582), 0]^T \approx [0, 0]^T$, which is indeed approximately zero (the gradient vanishes at the unconstrained minimum). Therefore the constrained and unconstrained optima coincide at $(0.582, 2.0)$ with $\mu^{\star} = 0$ (inactive constraint, zero Lagrange multiplier).

This analysis explains why all methods (Projected GD and penalty with $\lambda \geq 1.8$) converge to the \emph{same} point as unconstrained GD: the constraint does not change the optimum, it only constrains the feasible path taken to reach it.

\begin{remark}[Augmented Lagrangian]
A natural improvement over the pure exterior penalty is the \emph{augmented Lagrangian} (Method of Multipliers), which adds a Lagrange multiplier update to progressively tighten constraint enforcement without requiring $\lambda \to \infty$:
\begin{equation}
\mathcal{L}_\lambda(x, \mu) = f(x) + \mu\max(0, 0.5 - x_1) + \frac{\lambda}{2}\max(0, 0.5 - x_1)^2.
\end{equation}
Updating $\mu_{k+1} = \mu_k + \lambda \max(0, 0.5 - x_{1,k})$ allows the multiplier to carry constraint violation information across iterations, achieving exact feasibility at moderate $\lambda$ without the conditioning issues of the pure penalty method.
\end{remark}

\textbf{Projected GD} is provably correct: every iterate is feasible by construction, and convergence theory guarantees $f(x_k) \to f^{\star}_{\mathcal{C}}$ for appropriate $\alpha$. The projection cost is $O(1)$ for this simple constraint ($\max$ operation), but can be $O(d^3)$ for general polytopes (via linear programming).

\textbf{Penalty methods} offer simplicity (only a gradient modification is needed), but sacrifice exact feasibility for finite $\lambda$. A key theoretical trade-off is the growth of the Lipschitz constant of $\nabla F_\lambda$ with $\lambda$.

The penalised gradient is $\nabla F_\lambda(x) = \nabla f(x) - \lambda\,\mathbf{1}[x_1 < 0.5]\,e_1$. The penalty component $P(x_1) = \lambda\max(0, 0.5 - x_1)$ has gradient $\partial_1 P = -\lambda\mathbf{1}[x_1<0.5]$, which is piecewise constant with a jump of magnitude $\lambda$ at $x_1 = 0.5$. When iterates straddle the constraint boundary (one step inside, next outside the feasible region), GD sees an \emph{effective} gradient change of magnitude $\lambda$ in one step, equivalent to a local Lipschitz constant of approximately $L_\lambda \approx L_f + \lambda/\alpha$ (per step). Stable iteration requires:
\begin{equation}
\alpha \leq \frac{2}{L_f + \lambda}, \quad \text{or more conservatively } \alpha \leq \frac{2}{L_f + c\lambda} \text{ for some constant } c > 1.
\end{equation}
For $L_f \approx 10$: with $\lambda = 1.8$, $\alpha_{\max} \approx 2/11.8 \approx 0.169$; with $\lambda = 4.5$, $\alpha_{\max} \approx 2/14.5 \approx 0.138$. The chosen steps ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$) are substantially more conservative, reflecting the gradient discontinuity at $x_1 = 0.5$ which creates large local variation. As $\lambda \to \infty$, $\alpha \to 0$ and the iteration becomes infeasible in finite steps, which is why the augmented Lagrangian method (Remark above) is preferred: it achieves exact feasibility without requiring $\lambda \to \infty$.

\subsection{Feasible Set and Linear Programme}

The feasible set for Questions 6 is the box:
\begin{equation}
\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}

\textbf{Linear Programme:} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$.

\begin{proposition}[LP Solution]
Since both components $a_1 = 1 > 0$ and $a_2 = 2 > 0$ are positive, the LP objective is minimised by choosing the smallest feasible value for each coordinate independently:
\begin{equation}
x^{\star} = \begin{pmatrix} 0.5 \\ -5 \end{pmatrix}, \qquad f^{\star} = 0.5 + 2 \times (-5) = -9.5.
\end{equation}
This illustrates the \emph{Fundamental Theorem of Linear Programming}: the optimum of a linear objective over a bounded polytope is always attained at a vertex. The box $\mathcal{X}$ has $2^d = 4$ vertices; the LP selects the vertex in the direction opposite to $a$.
\end{proposition}

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe Algorithm}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LP subproblem)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (stays in $\mathcal{X}$)}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Frank--Wolfe Properties]
The Frank--Wolfe algorithm (Frank \& Wolfe, 1956; also called the Conditional Gradient method) replaces the projection step of projected gradient descent with an LP solve over $\mathcal{X}$. Key properties:
\begin{enumerate}[nosep]
\item \textbf{Projection-free:} Iterates are maintained as convex combinations of feasible points, so $x_k \in \mathcal{X}$ for all $k$ without an explicit projection oracle.
\item \textbf{Sparse iterates:} Each step moves towards a vertex of $\mathcal{X}$, making the iterates ``sparse'' in the vertex representation --- valuable in matrix optimisation over spectrahedra.
\item \textbf{Convergence rate:} $O(1/k)$ for smooth convex functions with fixed mixing parameter $1-\beta = 2/(k+2)$.
\item \textbf{Oracle cost:} For box constraints, the LP subproblem decomposes per coordinate and costs $O(d)$.
\end{enumerate}
\end{remark}

For the box constraint, the LP subproblem has the closed-form solution:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations per iteration. The choice between Frank--Wolfe and projected GD depends critically on the relative cost of the LP oracle vs.\ the projection for the specific constraint set:

\centering

For our box constraint ($d = 2$), both methods have $O(d)$ oracle cost and Frank--Wolfe offers no computational advantage. Its value is pedagogical: it demonstrates projection-free feasibility maintenance and provides the FW gap certificate.

\subsection{Frank--Wolfe Gap: A Convergence Certificate}

A key advantage of Frank--Wolfe is the availability of a \emph{primal gap certificate} at each iteration. The \emph{Frank--Wolfe gap} is:
\begin{equation}
g_{\mathrm{FW}}(x_k) := \nabla f(x_k)^T (x_k - z_k) = \max_{z \in \mathcal{X}} \nabla f(x_k)^T (x_k - z).
\end{equation}
This quantity is always non-negative (since $z_k$ minimises $\nabla f^T z$ over $\mathcal{X}$) and serves as an upper bound on the suboptimality gap:
\begin{equation}
f(x_k) - f^{\star} \leq g_{\mathrm{FW}}(x_k).
\end{equation}
For $L$-smooth convex $f$, the standard Frank--Wolfe convergence guarantee is:
\begin{equation}
g_{\mathrm{FW}}(x_k) \leq \frac{2L D^2}{k+2},
\end{equation}
where $D = \max_{x, y \in \mathcal{X}} \|x - y\|$ is the diameter of the feasible set. For our box with $D = \sqrt{(5-0.5)^2 + 15^2} \approx 15.7$, this gives a slow $O(1/k)$ bound. The fixed-$\beta$ scheme used in the code (instead of the decreasing $\gamma_k = 2/(k+2)$ schedule) does not exactly match the standard analysis, but empirically performs comparably.

\subsection{Interior vs Boundary Convergence Analysis}

\subsubsection{Mathematical Analysis of the Zig-Zag Phenomenon}

The zig-zag failure of Frank--Wolfe for interior optima has a precise mathematical explanation rooted in the discontinuity of the LP oracle near zero-gradient points.

\textbf{Oracle map for box constraints.} For $\mathcal{X} = [l_1,u_1]\times[l_2,u_2]$, the LP solution is:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } \partial_i f(x_k) > 0, \\ u_i & \text{if } \partial_i f(x_k) \leq 0. \end{cases}
\end{equation}
This sign-dependent map is \emph{discontinuous} at $\nabla f = 0$: any infinitesimal perturbation in the gradient direction at the optimum can flip $z_k$ from one vertex to the diametrically opposite vertex.

\textbf{Oscillation mechanics for Case~II.} Consider the $x_1$ coordinate with $f(x) = (x_1-1)^2 + (x_2-5)^2$ and $x^{\star}_1 = 1 \in (0.5, 5)$ strictly interior. At iterate $x_k$ near $x^{\star}_1 = 1$:
\[
\partial_1 f(x_k) = 2(x_{1,k} - 1) \begin{cases} > 0 & \Rightarrow z_{1,k} = 0.5 \\ < 0 & \Rightarrow z_{1,k} = 5. \end{cases}
\]
The update $x_{1,k+1} = \beta x_{1,k} + (1-\beta)z_{1,k}$ creates a fixed-point analysis. Let $\delta_k = x_{1,k} - 1$ (signed deviation). If $\delta_k > 0$ (right of optimum):
\[
\delta_{k+1} = \beta(1+\delta_k) + (1-\beta)\cdot 0.5 - 1 = \beta\,\delta_k - \frac{1-\beta}{2}.
\]
The deviation crosses zero (triggers vertex flip) when $\delta_k < \delta^{\star} := \frac{(1-\beta)/2}{\beta}$. For $\beta = 0.90$: $\delta^{\star} = 0.05/0.90 \approx 0.056$. Once $\delta_k < \delta^{\star}$, the iterate crosses $x_1^{\star}=1$ and now $\delta_{k+1} < 0$, so the oracle switches to $z_{1,k+1} = 5$:
\[
\delta_{k+1} = \beta\delta_k + (1-\beta)(5-1) = \beta\,\delta_k + 4(1-\beta).
\]
Starting from $\delta_k \approx -\delta^{\star} \approx -0.056$: $\delta_{k+1} \approx 0.90\times(-0.056) + 0.4 \approx 0.35$.

Thus after each crossing event, the iterate \emph{overshoots} by $\approx 4(1-\beta)$ in the opposite direction. For $\beta = 0.90$ this gives an overshoot of $0.4$, meaning $x_{1,k}$ oscillates between roughly $x_1 \approx 0.944$ and $x_1 \approx 1.35$ in steady state. The residual squared deviation $(x_1-1)^2 \approx (0.2)^2 = 0.04$ contributes significantly to the final objective $f \approx 0.0089$.

For $\beta = 0.985$: $\delta^{\star} \approx \frac{0.015/2}{0.985} \approx 0.0076$ and overshoot $\approx 4(1-0.985) = 0.06$. The oscillation amplitude is $\approx 0.03$, yielding $(x_1-1)^2 \approx 10^{-3}$, consistent with the observed $f \approx 0.0046$. This quantitative analysis confirms that larger $\beta$ (smaller step towards vertices) dramatically reduces zig-zag amplitude.

\textbf{Remedies.} The \emph{Away-Step Frank--Wolfe} variant (Lacoste-Julien \& Jaggi, 2015) adds a component that moves away from the vertex contributing most to the current iterate, breaking the oscillation cycle and achieving \emph{linear} convergence on strongly convex functions over polytopes:
\[
x_{k+1} = x_k + \gamma_k (s_k - x_k) - \eta_k (v_k - x_k),
\]
where $s_k$ is the standard FW step (towards LP solution) and $v_k$ is the ``away'' step (away from the active vertex). This modification avoids the zig-zag by maintaining a ``drop condition'' that permanently reduces the weight on vertices contributing to oscillation.

\textbf{Boundary optimum:} Frank--Wolfe excels for boundary optima. When the unconstrained minimum lies outside $\mathcal{X}$ (Case~III, $x^{\star}_{\mathrm{unc}} = (0,0) \notin \mathcal{X}$), the gradient $\partial_1 f(x_k) = 2x_{1,k} > 0$ for all feasible $x_{1,k} \geq 0.5$, so the LP oracle \emph{consistently} selects $z_{1,k} = 0.5$ without oscillation. The iterate converges monotonically toward the boundary $x_1 = 0.5$ at rate $|x_{1,k} - 0.5| = \beta^k|x_{1,0} - 0.5|$ (geometric with rate $\beta = 0.93$). This eliminates the zig-zag entirely, explaining the smoother convergence curve in Figure~\ref{fig:q6_evolution} (bottom-left).

The key practical advantage of Frank--Wolfe over projected gradient descent is the \emph{oracle}: for complex constraint sets where projection is expensive (e.g., nuclear norm ball, spectrahedra in semidefinite programming), solving the LP subproblem is often far cheaper than computing the projection. The zig-zag issue is a known limitation specifically for interior optima and can be addressed by the away-step variant or by using a decreasing $1-\beta_k = 2/(k+2)$ schedule.

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This study has systematically evaluated six categories of optimisation algorithms on three benchmark functions of increasing difficulty. The following conclusions emerge:

\subsection{Method-Level Findings}

\begin{itemize}[nosep]
\item \textbf{Adaptive Step-Size Methods (Q1):} The Polyak step is theoretically motivated and empirically fastest when $f^{\star}$ is known exactly, but fails catastrophically when $f^{\star}$ is misspecified --- a fundamental limitation in practice. Adagrad provides uniform stability across all benchmarks at the cost of eventual stagnation. RMSprop and Heavy Ball offer the best practical balance of speed and stability, with Heavy Ball particularly effective on ill-conditioned problems due to momentum.

\item \textbf{Momentum Methods (Q2):} Nesterov's accelerated gradient achieves the theoretical $O(1/k^2)$ convergence rate on convex functions, yielding a $7\times$ improvement over GD on the Rosenbrock valley. Adam combines momentum with per-coordinate adaptive scaling, making it the most robust general-purpose method across diverse problem geometries. Mini-batch SGD demonstrates that smaller batches provide faster convergence per epoch at the cost of higher variance, with the benefit disappearing under high observation noise.

\item \textbf{Newton's Method (Q3):} Second-order methods achieve qualitatively superior convergence --- one step on quadratics, quadratic local convergence near minima. The computational cost ($O(d^3)$ per step) limits applicability to small $d$, but damped Newton is clearly superior to all first-order methods on the Rosenbrock function even with only 20 iterations. The local approximation analysis confirms that curvature information (second-order Taylor expansion) provides dramatically better gradient descent guidance than linear information alone.

\item \textbf{Derivative-Free Methods (Q4):} Finite-difference gradient approximation with appropriate $\delta$ nearly matches exact gradient descent, confirming that smooth functions allow high-quality numerical differentiation. The $\delta = 0.8$ case illustrates the severe consequences of poor step selection. Nelder--Mead is a practical, gradient-free option for low-dimensional problems. Grid search provides coverage guarantees but is computationally impractical beyond $d = 2$.

\item \textbf{Constrained Optimisation (Q5):} Projected gradient descent is the theoretically sound approach, maintaining hard feasibility at every iteration with identical per-iteration cost to unconstrained GD (for cheap projections). Penalty methods offer implementation simplicity at the cost of exact feasibility and increased conditioning as $\lambda$ grows.

\item \textbf{Frank--Wolfe (Q6):} The conditional gradient algorithm provides projection-free constrained optimisation, naturally producing sparse iterates as convex combinations of vertices. It converges more slowly than projected GD on strongly convex interiors but handles boundary optima naturally and is particularly advantageous for complex constraint sets where projection is computationally expensive.
\end{itemize}

\subsection{Theoretical Connections}

The methods studied here form a coherent hierarchy in terms of information access and convergence rates:

\begin{table}[H]
\centering
\caption{Convergence rate hierarchy for smooth convex optimisation (condition number $\kappa$).}
\label{tab:hierarchy}
\begin{tabular}{llll}
\toprule
\textbf{Category} & \textbf{Information} & \textbf{Convex rate} & \textbf{Str.\ convex rate} \\
\midrule
Grid search & $f$ values (dense) & $O(1/N^{1/d})$ & same \\
Random search & $f$ values (sparse) & $O(d/k)$ & $O(d/k)$ \\
Frank--Wolfe & $\nabla f$ + LP oracle & $O(1/k)$ & $O(1/k)$ \\
Gradient descent & $\nabla f$ & $O(1/k)$ & $O(\rho^k)$, $\rho=1-1/\kappa$ \\
Heavy Ball / Nesterov & $\nabla f$ + momentum & $O(1/k^2)$ & $O(\rho^k)$, $\rho=1-1/\sqrt{\kappa}$ \\

The experimental evidence from Benchmarks A, B, and C confirms these theoretical predictions: methods higher in the information hierarchy universally outperform those below on the challenging Rosenbrock function (the only benchmark where the differences are large enough to discriminate between methods). Benchmark A's near-isotropic quadratic structure means all first-order methods perform similarly (the $\kappa \approx 1.15$ makes the condition-number-dependent terms negligible).

\subsection{Practical Recommendations}

\begin{enumerate}[nosep]
\item \textbf{Default choice:} Adam combines momentum and adaptive scaling, offering robust performance across problem types without extensive hyperparameter tuning. It is the standard choice in deep learning.
\item \textbf{Well-conditioned convex:} Nesterov accelerated gradient achieves the optimal first-order rate and outperforms Adam on simple geometries.
\item \textbf{Small-scale with known Hessian:} Newton's method (or L-BFGS for quasi-Newton) is strongly preferred for its quadratic local convergence --- just 20 Newton iterations outperform 80 GD iterations on the Rosenbrock function.
\item \textbf{No gradient available:} Nelder--Mead for $d \leq 5$; Nesterov random search or Bayesian optimisation for higher dimensions.
\item \textbf{Constrained problems:} Projected GD when projection is cheap (box, simplex, ball); Frank--Wolfe when the LP subproblem is cheaper than projection (spectrahedra, nuclear norm ball, flow polytopes).
\item \textbf{Stochastic setting:} Mini-batch SGD with small batch sizes converges faster per epoch. Under high observation noise, the irreducible noise floor dominates and batch size becomes less important --- focus on algorithmic momentum (Adam) instead.
\item \textbf{Known $f^{\star}$:} The Polyak step size is theoretically optimal and achieves the best performance on Benchmark C. In practice, a conservative lower bound on $f^{\star}$ can be used safely without risking divergence.
\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
