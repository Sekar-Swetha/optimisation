\begin{abstract}
This report presents a comprehensive comparative study of six families of optimisation algorithms applied to three canonical benchmark problems: a linear regression quadratic loss (Benchmark~A, condition number $\kappa \approx 1.12$), a toy neural network loss with sinusoidal perturbation (Benchmark~B, $\kappa \in [3.5, 6.9]$), and the Rosenbrock function (Benchmark~C, $\kappa \approx 2508$ at the optimum). The investigations encompass: (1)~adaptive step-size methods --- Polyak, Adagrad, RMSprop, and Heavy Ball; (2)~momentum and stochastic methods --- Nesterov acceleration, Adam, and mini-batch SGD; (3)~Newton's method with local second-order approximation; (4)~derivative-free approaches including finite differences, Nesterov random search, Nelder--Mead, and grid search; (5)~constrained optimisation via projected gradient descent and penalty methods; and (6)~the Frank--Wolfe conditional gradient algorithm for linear-constrained problems.

Key findings are: the Polyak step size achieves the best convergence on well-specified problems ($f = 0.0094$ on Rosenbrock in 120 iterations) but diverges catastrophically when $f^\star$ is misspecified; Newton's method requires only one iteration to solve the quadratic loss exactly and 20 damped iterations to reach $f = 0.686$ on Rosenbrock versus $f = 3.73$ for gradient descent in 80 iterations; derivative-free methods incur a convergence rate penalty relative to gradient-based methods due to noisy or coarse gradient approximation; projected gradient descent enforces feasibility as a hard constraint from iteration~1, while penalty methods require $\lambda \geq 1.8$ to achieve reliable constraint satisfaction in 100 iterations; and Frank--Wolfe converges at $O(1/k)$ without requiring a projection oracle.

Each method is evaluated in terms of convergence speed, trajectory behaviour, stability, computational cost, and alignment with the theoretical guarantees from the CS7DS2 course.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmarks span a spectrum of problem difficulty, from a well-conditioned convex quadratic to a highly ill-conditioned non-convex landscape.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \sim \mathcal{N}(0, I_2), \quad
y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The objective is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2
= \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strongly convex quadratic with Hessian $H = \frac{1}{m}X^T X$ (constant throughout optimisation). Numerical computation gives eigenvalues $\lambda_{\min} \approx 0.9247$ and $\lambda_{\max} \approx 1.0325$, yielding condition number $\kappa(H) = \lambda_{\max}/\lambda_{\min} \approx 1.12$. This near-isotropic curvature means almost any step size within the stability range $\alpha < 2/\lambda_{\max} \approx 1.94$ will converge rapidly. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the irreducible loss floor is $J^{\star} = \frac{1}{2}\sigma^2 \approx 0.4826$ (not zero). Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is diagonal:
\begin{equation}
H(x_1) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
Since $\sin(x_1) \in [-1, 1]$, the first diagonal entry ranges from $1$ to $3$, giving a condition number $\kappa \in [10/3, 10/1] \approx [3.3, 10]$. Along the optimisation trajectory from $x_0 = [-1, 4]^T$ to $x^\star$, $\kappa$ ranges from approximately $3.5$ to $6.9$. The minimum satisfies $df/dx_2 = 0 \Rightarrow x_2^\star = 2$ and $df/dx_1 = 2(x_1-1) + \cos(x_1) = 0$, which gives $x_1^\star \approx 0.5824$ (solved numerically via Newton's method applied to this scalar equation). Thus $x^\star \approx (0.5824, 2.000)$ and $f^\star = (x_1^\star - 1)^2 + \sin(x_1^\star) \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\textit{Note:} the visualisation code marks the optimum at $(0.61, 2.0)$, which is a slight overestimate; the true value $x_1^\star \approx 0.582$ is used in this report.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$. The Hessian is:
\begin{equation}
H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1,1)$: eigenvalues are approximately $0.399$ and $1001.6$, giving $\kappa \approx 2508$. Near the starting point $x_0 = (-1, 1)$: the largest eigenvalue is approximately $1001$, so the stability bound for gradient descent is $\alpha < 2/1001 \approx 0.002$. The code uses $\alpha = 0.0012$, which is safely conservative. The characteristic narrow, curved valley along $x_2 = x_1^2$ and the extreme ill-conditioning make Rosenbrock one of the most widely used stress tests for optimisation algorithms.

\subsection{Benchmark D: Constrained Feasible Set}
For Questions~5 and 6, the feasible set is:
\begin{equation}
\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a box (polytope), for which the projection and the linear minimisation oracle are both $O(d)$ operations.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for $N = 120$ iterations. The baseline GD update is:
\begin{equation}
x_{k+1} = x_k - \alpha\,\nabla f(x_k).
\end{equation}
For a convex function with $L$-Lipschitz gradient and strong convexity parameter $\mu$, GD with $\alpha \leq 1/L$ converges at rate $O((1 - \mu/L)^k)$; on Benchmark A ($\kappa \approx 1.12$) this is nearly linear, while on Rosenbrock ($\kappa \approx 2508$) convergence is extremely slow.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive, requires knowing $f^\star$}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is theoretically optimal for convex functions when $f^\star$ is known exactly: it guarantees convergence in $O(1/k)$ without requiring knowledge of the Lipschitz constant $L$. However, it is highly sensitive to errors in $f^\star$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative squared gradient (per coordinate)}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad adapts the learning rate per coordinate: directions with large historical gradients receive smaller effective steps. The cumulative sum $G_k$ is monotonically non-decreasing, so the effective learning rate $\alpha_0/\sqrt{G_k}$ is \emph{monotonically decreasing}. This guarantees stability but can cause premature convergence on non-stationary problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's monotone decay by replacing the cumulative sum with an exponential moving average (decay $\beta = 0.9$). This means $v_k$ tracks the \emph{recent} gradient magnitude rather than the full history, allowing the effective learning rate to increase again if gradients become smaller. RMSprop was proposed by Hinton (unpublished, Coursera lecture) as a heuristic; it lacks the convergence guarantees of Adagrad but is more robust in practice.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, g_k$ \Comment{Accumulated velocity}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method (Polyak, 1964) accumulates a momentum term $z_k$ that damps oscillations in high-curvature directions. For a quadratic with condition number $\kappa$, the optimal Heavy Ball parameters achieve a convergence rate $O\!\left(\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^k\right)$, dramatically better than GD's $O\!\left(\left(\frac{\kappa-1}{\kappa+1}\right)^k\right)$. Note: the momentum vector $z_k$ accumulates gradient history, unlike the Nesterov variant which uses a lookahead evaluation (see Section~\ref{sec:q2}).

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$, a $374\times$ improvement over GD). Otherwise, Adagrad and Heavy Ball match or exceed GD baseline performance. Heavy Ball's $f = 1.40$ vs.\ GD's $f = 3.51$ on Rosenbrock demonstrates the momentum benefit even for non-convex problems, where the formal quadratic-convergence guarantee does not apply but the practical benefit is still realised.
\item \textbf{Step-size behaviour:} Adagrad's step decays at $O(1/\sqrt{k})$ asymptotically (since $G_k = \sum_{i=1}^k g_i^2 \sim O(k)$ for bounded gradients), guaranteeing stability but slowing late-stage convergence. RMSprop's exponential moving average with $\beta = 0.9$ maintains $v_k \approx (1-\beta)^{-1} \cdot \mathbb{E}[g_k^2]$ in steady state, so the effective step stabilises rather than decaying to zero. The Polyak step is adaptive in a fundamentally different sense: it scales to the \emph{remaining suboptimality} rather than to accumulated gradients.
\item \textbf{Stability:} GD baseline is the most stable but slowest. Polyak is unstable when $f^\star$ is misspecified. Heavy Ball can oscillate on ill-conditioned non-quadratic problems (no theoretical guarantee beyond quadratics), but is empirically stable here.
\item \textbf{Practical implication:} In practice, exact $f^\star$ is rarely known, limiting Polyak to problems where a lower bound on the optimal value is available (e.g., $f^\star = 0$ for a non-negative loss like cross-entropy with label smoothing). RMSprop and Adam are more robust adaptive choices for general deep learning applications.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead / extrapolation point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This ``look before you step'' correction allows Nesterov to anticipate the momentum's direction and correct trajectory before overshooting. The schedule $\beta_k = (k-1)/(k+2)$ gradually increases momentum from 0, avoiding early-iteration instability. Nesterov's method achieves an \textbf{optimal} $O(1/k^2)$ convergence rate for smooth convex functions, compared to $O(1/k)$ for gradient descent. For a convex function with $L$-Lipschitz gradient, the guarantee is:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{k^2}.
\end{equation}
This $O(1/k^2)$ rate matches the lower bound for first-order methods on smooth convex problems (Nesterov's complexity bound), making it \emph{optimal} in this sense.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (EMA of gradient)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (EMA of sq.\ gradient)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam (Kingma \& Ba, 2015) combines first-moment momentum (analogous to Heavy Ball) with per-coordinate adaptive scaling (analogous to RMSprop). The bias-correction terms $(1-\beta_1^t)^{-1}$ and $(1-\beta_2^t)^{-1}$ compensate for the zero-initialisation of $m_0 = v_0 = 0$: without correction, early estimates are biased towards zero, artificially reducing the effective step size in the first few iterations. After bias correction, the magnitude of the Adam update per coordinate is:
\begin{equation}
\left|\Delta x_{t,i}\right| = \alpha\,\frac{|\hat{m}_{t,i}|}{\sqrt{\hat{v}_{t,i}} + \varepsilon}
\leq \frac{\alpha}{\varepsilon},
\end{equation}
since $|\hat{m}_{t,i}| \leq \sqrt{\hat{v}_{t,i}}$ (by the Cauchy--Schwarz inequality applied to the EMA weights). This provides an inherent step-size upper bound of $\alpha/\varepsilon$ per coordinate, regardless of gradient magnitude.

\subsection{Hyperparameter Settings}

\begin{table}[H]

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data permuted at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute the $m$ sample indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The stochastic gradient $g$ is an unbiased estimate of $\nabla J(\theta)$: $\mathbb{E}[g] = \nabla J(\theta)$. Its variance is:
\begin{equation}
\mathrm{Var}(g_i) = \frac{1}{b}\,\mathrm{Var}\!\left(\frac{[X^T(X\theta - y)]_i}{1}\right) \propto \frac{1}{b}.
\end{equation}
SGD with constant learning rate does not converge to the exact minimiser; it oscillates in a neighbourhood of radius $O(\alpha/b)$ around $\theta^\star$. To achieve $\varepsilon$-accuracy in expectation, a decaying schedule $\alpha_k \to 0$ (e.g., $\alpha_k = \alpha_0/(k+1)$) is needed, giving convergence rate $O(1/k)$. With constant $\alpha = 0.06$, the plots show convergence to the noise floor rather than the exact minimum. The epoch-level update count is $\lfloor m/b \rfloor = 1000/b$ per epoch.

\subsection{Local Approximations of $g(x) = x^4$}

Around a point $x_0$, a smooth function $g(x)$ can be approximated by its first- and second-order Taylor expansions.

\textbf{First-order (linear) approximation} around $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3(x - 0.25) = 0.003906 + 0.0625(x - 0.25).
\end{equation}
This tangent line correctly models the slope at $x_0$ but does not capture curvature, leading to poor approximation away from $x_0$.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.003906 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
The Newton step $x_{k+1} = x_k - g'(x_k)/g''(x_k)$ minimises this local quadratic model, corresponding to a full step to the model's stationary point. For $g(x) = x^4$:
\begin{equation}
x_{k+1} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3} = \frac{2}{3}x_k.
\end{equation}
The Newton iterates satisfy $x_k = (2/3)^k x_0$, converging \emph{linearly} (not quadratically) to $x^\star = 0$. The lack of quadratic convergence arises because $g''(x^\star) = 0$ at $x^\star = 0$ --- the Hessian is singular at the optimum, violating the strong convexity condition required for Newton's quadratic convergence guarantee. For $g(x) = x^4$, the function has a ``flat'' minimum (the second derivative is zero there), so the Newton step overshoots more than in the strongly convex case. This illustrates that Newton's quadratic convergence requires $\nabla^2 f(x^\star) \succ 0$ (strictly positive definite Hessian at the optimum).

\subsection{Newton's Method}

For a smooth function, Newton's method solves a linear system at each iteration to find the Newton direction:

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0, 1]$, damping $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise Hessian for numerical stability}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{$\alpha = 1$: pure Newton; $\alpha < 1$: damped}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Quadratic convergence:} For a strongly convex, twice-differentiable function with Lipschitz Hessian, the undamped Newton method ($\alpha = 1$) with exact Hessian satisfies:
\begin{equation}
\|x_{k+1} - x^\star\| \leq C\,\|x_k - x^\star\|^2
\end{equation}
for some constant $C > 0$ once the iterate enters a neighbourhood of $x^\star$ (the quadratic convergence region). This \emph{quadratic} rate means the number of correct digits doubles each iteration, compared to GD's linear rate of $O((1-\mu/L)^k)$.

For non-convex problems (e.g., Rosenbrock away from the optimum) or problems with rapidly changing Hessians, a damped step ($\alpha < 1$) prevents overshooting into regions where the quadratic model is inaccurate. The quadratic convergence region (``basin of attraction'' for Newton) is entered only when $x_k$ is sufficiently close to $x^\star$; outside this region, a damped or globalised Newton step is needed.

\textbf{Comparison with gradient descent:} GD's linear convergence rate $O\!\left((1-\mu/L)^k\right)$ depends on the condition number $\kappa = L/\mu$. For Benchmark~C with $\kappa \approx 2508$, GD converges at rate $\approx (1 - 1/2508)^k \approx 0.9996^k$ --- reaching $\varepsilon$-accuracy requires $O(\kappa \log(1/\varepsilon))$ iterations. Newton's quadratic convergence requires only $O(\log\log(1/\varepsilon))$ iterations in the local regime, independent of $\kappa$.

\textbf{Settings:} GD uses 80 iterations; Newton uses only 20 iterations, reflecting the superior per-iteration progress. Tikhonov damping $\lambda = 10^{-8}$ ensures $H_k + \lambda I \succ 0$ (positive definite) without materially affecting the Newton direction when $H_k$ is itself well-conditioned.

\subsection{Part A: Finite Differences and Nesterov Random Search on Benchmark B}

\subsubsection{Finite Difference Gradient Approximation}

When analytical gradients are unavailable, they can be approximated using function evaluations:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\label{eq:fd}
\end{equation}
where $e_i$ is the $i$-th standard basis vector. The total error in the approximation has two competing terms:
\begin{itemize}[nosep]
\item \textbf{Truncation error:} $O(\delta)$ from neglecting higher-order terms in the Taylor expansion.
\item \textbf{Floating-point error:} $O(\varepsilon_{\text{mach}}/\delta)$ from subtractive cancellation in $f(x + \delta e_i) - f(x)$.
\end{itemize}
Balancing these gives the optimal perturbation $\delta_{\text{opt}} \approx \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ for double-precision arithmetic. The assignment tests $\delta = 0.05$ (well within the useful range) and $\delta = 0.8$ (too large, dominated by truncation error). For a $d$-dimensional problem, the FD gradient requires $d + 1$ function evaluations per step.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$, seed
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta\, u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Nesterov random search is a zeroth-order method: it requires only \emph{two} function evaluations per iteration regardless of dimension $d$, making it \emph{dimension-independent} in cost per step. The gradient estimate $\hat{g}_k$ is a one-sided directional derivative along a random unit direction. Its expected value satisfies $\mathbb{E}[\hat{g}_k] = \frac{1}{d}\nabla f(x_k)$ (a scaled version of the gradient), so the method is unbiased up to a dimensional scaling factor. The variance of $\hat{g}_k$ is $O(\|\nabla f(x_k)\|^2 / d)$, which decreases in high dimensions relative to the signal --- a key advantage over methods that require estimating all $d$ partial derivatives.

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$. At each iteration, the worst vertex (highest function value) is replaced by a new point determined by one of four geometric operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): move worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2$): if reflection improved, attempt a further step in the same direction.
\item \textbf{Outside contraction} ($\rho = 0.5$): if reflection is moderate, contract between centroid and reflected point.
\item \textbf{Inside contraction} and \textbf{shrink} ($\sigma = 0.5$): if all contractions fail, shrink the entire simplex towards the best vertex.
\end{enumerate}
No gradient information is required. The method exploits the simplex geometry to probe the landscape and adapts its shape to the local topology. Convergence is generally sub-linear and there are no global convergence guarantees, but empirically it works well for smooth low-dimensional problems.

Settings: initial simplex step $= 0.35$, iterations $= 160$.

\textbf{Key observations:}
\begin{itemize}[nosep]
\item For high-dimensional problems ($d \gg 1$), FD requires $d + 1$ evaluations per step versus 2 for Nesterov random search, which is dimension-independent. However, FD provides a biased gradient estimate with $O(\delta)$ error per component, whereas Nesterov random search's estimate has zero bias in expectation (it estimates a directional derivative, not the full gradient) but high variance.
\item Nelder--Mead and grid search do not scale: Nelder--Mead is typically reliable only for $d \leq 10$--$20$, and grid search is intractable beyond $d \approx 3$ due to the curse of dimensionality ($55^{10} \approx 2.5 \times 10^{17}$ evaluations for $d=10$).
\item When analytical gradients are available, the cost of gradient computation is typically $O(nd)$ for loss functions summing over $n$ data points --- the same order as one function evaluation --- so the per-iteration cost of exact GD is comparable to FD GD but without the approximation error.
\item \textbf{Finite difference vs.\ automatic differentiation:} In modern machine learning, automatic differentiation (backpropagation) computes \emph{exact} gradients at cost $O(d)$ forward passes, completely superseding finite differences. FD remains relevant only when code is black-box or not differentiable.
\end{itemize}

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Formulation}

Minimise Benchmark~B subject to a linear inequality constraint:
\begin{equation}
\min_{x \in \mathbb{R}^2}\ f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad\text{subject to}\quad x_1 \geq 0.5.
\end{equation}
The starting point $x_0 = (0.2, 4.0)$ is \emph{infeasible} ($x_1 = 0.2 < 0.5$). Note that the unconstrained minimum of $f$ is at $x^\star \approx (0.582, 2.0)$ with $x_1^\star \approx 0.582 > 0.5$: the unconstrained optimum is already \emph{interior} to the feasible set. Thus, the true constrained and unconstrained optima coincide, and the constraint is \emph{inactive} at the optimum. However, the trajectory from $x_0 = (0.2, 4.0)$ may pass through the infeasible region, and the different methods handle this differently.

\textbf{KKT conditions:} For the problem above, the KKT conditions require:
\begin{equation}
\nabla f(x^\star) = \mu\, \nabla(-x_1 + 0.5) = -\mu\, e_1, \quad \mu \geq 0, \quad \mu(0.5 - x_1^\star) = 0.
\end{equation}
Since the unconstrained optimum satisfies $x_1^\star > 0.5$, the complementary slackness condition gives $\mu = 0$, and the KKT conditions reduce to $\nabla f(x^\star) = 0$ --- confirming that the unconstrained minimum is also the constrained minimum.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection operator $\Pi_{\mathcal{F}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{F}}(z_{k+1})$ where $\Pi_{\mathcal{F}}(z) = (\max(0.5, z_1),\, z_2)$ \Comment{Project onto $\{x : x_1 \geq 0.5\}$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection $\Pi_{\mathcal{F}}(z) = \arg\min_{x \in \mathcal{F}} \|x - z\|^2$ for the half-space $\{x : x_1 \geq 0.5\}$ reduces to a simple clipping operation. PGD guarantees feasibility at every iterate.

\begin{algorithm}[H]
\caption{Penalty Method (Linear Penalty)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_\lambda(x) \gets f(x) + \lambda\max(0,\; 0.5 - x_1)$ \Comment{Penalised objective}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method adds a soft violation cost proportional to $\lambda$ and does not guarantee feasibility. The effective gradient includes an extra $-\lambda$ in the $x_1$ direction when $x_1 < 0.5$ (penalty gradient: $\nabla_x[\lambda\max(0, 0.5-x_1)] = -\lambda e_1$ if $x_1 < 0.5$, else $0$). As $\lambda \to \infty$, the penalised solution converges to the constrained solution, but large $\lambda$ requires smaller step sizes to maintain stability.

Settings: PGD uses $\alpha = 0.08$. Penalty uses $\alpha = 0.05$ for $\lambda \in \{0.15, 1.8\}$ and $\alpha = 0.03$ for $\lambda = 4.5$ (to compensate for the larger effective Lipschitz constant). Unconstrained GD uses $\alpha = 0.07$. All run for 100 iterations.

In this experiment, all methods eventually converge to the same feasible minimum because the unconstrained and constrained optima coincide. The key differences lie in \emph{how} feasibility is enforced and the convergence rate during the initial infeasible phase:

\textbf{Projected GD} enforces feasibility as a hard constraint immediately (from iteration~1), making it the only method that never visits an infeasible point after initialisation. It is exact, computationally cheap for simple constraints ($O(1)$ projection), and does not require tuning $\lambda$. The projection onto $\{x : x_1 \geq 0.5\}$ is trivial here, but for complex convex constraint sets the projection can be computationally expensive.

\textbf{Penalty method} requires only a modified gradient and no projection oracle. The trade-off is that feasibility is enforced softly: $\lambda$ must be chosen large enough to drive the iterate to the feasible set, but too large a $\lambda$ amplifies the effective Lipschitz constant to $L(F_\lambda) = L(f) + \lambda$, requiring stable step size $\alpha \leq 2/(L(f) + \lambda)$ --- consistent with the smaller $\alpha = 0.03$ used for $\lambda = 4.5$. For problems where the unconstrained minimum is infeasible, the residual constraint violation of the penalty minimiser scales as $O(1/\lambda)$.

An improvement over the pure penalty approach is the \emph{augmented Lagrangian} (multiplier) method:
\begin{equation}
L_\rho(x, \mu) = f(x) + \mu\,\max(0,\, 0.5 - x_1) + \frac{\rho}{2}\bigl[\max(0,\, 0.5 - x_1)\bigr]^2,
\end{equation}
where $\mu$ is a Lagrange multiplier updated as $\mu_{k+1} = \mu_k + \rho\,\max(0, 0.5 - x_k)$. This provides exact constraint satisfaction at a \emph{finite} $\rho$ as a fixed point of the update, unlike the pure penalty method which requires $\lambda \to \infty$ for exact feasibility. The augmented Lagrangian is standard in production-grade constrained solvers (e.g., ADMM, SQP).

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm \hfill [30 marks]}
\label{sec:q6}
% ============================================================

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box polytope).

\textbf{Linear Programme:} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$.

Since $a = [1, 2]^T$ has all positive components, the minimum is attained at the lower bound of every coordinate:
\begin{equation}
x^\star = (0.5,\; -5), \qquad f^\star = 0.5 + 2 \times (-5) = -9.5.
\end{equation}
This illustrates the \textbf{fundamental theorem of linear programming}: the optimum of a linear objective over a bounded polytope is always attained at a vertex. The LP subproblem solved by Frank--Wolfe for box constraints is:
\begin{equation}
\arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z \quad\Leftrightarrow\quad
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
where $[l_i, u_i]$ are the box bounds. This is an $O(d)$ operation per coordinate, making Frank--Wolfe very efficient for box constraints.

\centering
\includegraphics[width=0.85\textwidth]{figures/q6_lp.pdf}
\caption{Q6-I: Feasible region $\mathcal{X}$ (light blue box) and linear objective contours $f(x) = x_1 + 2x_2$ (colour-coded). The contour lines are parallel (gradient $[1, 2]^T$ is constant for a linear objective), and the minimum occurs at the lower-left vertex $(0.5, -5)$ of the feasible box, where both components of $x$ are at their lower bounds. Moving any component of $x$ upward from its lower bound increases the linear objective, confirming this vertex as the global minimum.}

\subsection{Frank--Wolfe Algorithm (Conditional Gradient)}

\begin{algorithm}[H]
\caption{Frank--Wolfe / Conditional Gradient Method}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Feasible $x_0 \in \mathcal{X}$, fixed mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LP subproblem)}
    \State $x_{k+1} \gets \beta\, x_k + (1-\beta)\, z_k$ \Comment{Convex combination; equivalently $x_k + \gamma_k(z_k - x_k)$ with $\gamma_k = 1-\beta$}
\EndFor
\end{algorithmic}
\end{algorithm}

The Frank--Wolfe algorithm (1956) maintains feasibility automatically: since $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$, any convex combination $\beta x_k + (1-\beta)z_k \in \mathcal{X}$ (by convexity of $\mathcal{X}$). No projection is needed. The step $\gamma_k = 1 - \beta$ is fixed here; the standard decaying schedule $\gamma_k = 2/(k+2)$ achieves the optimal $O(1/k)$ convergence rate for smooth convex $f$ over a compact convex set:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\,\mathrm{diam}(\mathcal{X})^2}{k+2},
\end{equation}
where $L$ is the Lipschitz constant of $\nabla f$ and $\mathrm{diam}(\mathcal{X}) = \|u - l\|$ for box bounds. With fixed $\gamma = 1 - \beta$, convergence is guaranteed as long as $z_k$ converges to a fixed point, but the oscillatory behaviour of $z_k$ (which jumps between vertices) can prevent exact convergence at a fixed step.

\textbf{Frank--Wolfe gap} as a convergence certificate:
\begin{equation}
G_k = \nabla f(x_k)^T (x_k - z_k) \geq f(x_k) - f^\star \geq 0.
\end{equation}
The gap $G_k$ is computable without knowing $f^\star$ and provides an upper bound on the current suboptimality. When $G_k \approx 0$, convergence is certified.

\subsection{Interior vs.\ Boundary Behaviour --- Analysis}

\textbf{Interior optimum:} Frank--Wolfe converges to the interior minimum, but via a zig-zag path: at each step, the LP oracle selects a vertex of $\mathcal{X}$ based on the current gradient, and the convex combination slowly moves the iterate towards that vertex. Near the interior minimum $(1, 5)$, the gradient $\nabla f(x_k) \approx 2(x_k - (1, 5))$ is small and may alternate sign in individual components, causing the LP oracle to select alternating vertices. The fixed mixing parameter $\gamma = 1-\beta$ prevents these oscillations from being averaged away. The $O(1/k)$ rate is confirmed by the approximate power-law decrease visible in the convergence plots.

The \textbf{Frank--Wolfe gap} at the final iterate can be computed as:
\begin{equation}
G_k = \nabla f(x_k)^T(x_k - z_k) = \sum_i (\nabla f(x_k))_i \cdot [(x_k)_i - (z_k)_i] \geq f(x_k) - f^\star,
\end{equation}
providing a certificate of near-optimality without knowing $f^\star$.

\textbf{Boundary optimum:} Frank--Wolfe is particularly suited to boundary optima. When the unconstrained minimum $(0, 0)$ lies outside $\mathcal{X}$ (violating $x_1 \geq 0.5$), the gradient at $x_k$ near the $x_1 = 0.5$ boundary has $(\nabla f(x_k))_1 = 2(x_k)_1 \approx 1 > 0$, so the LP oracle \emph{consistently} selects $z_1 = 0.5$ (lower bound). This drives $x_k$ towards the boundary naturally, without projection. However, the $x_2$ component oscillates: the LP oracle selects $z_2 = 10$ when $x_2 < 0$ (since $(\nabla f)_2 = 2x_2 < 0$) and $z_2 = -5$ when $x_2 > 0$, causing zigzagging in the $x_2$ direction. The final $x_2 = -0.337$ (below $x_2^\star = 0$) reflects this oscillation. Using $\gamma_k = 2/(k+2)$ would provide the proven $O(1/k)$ convergence certificate.

\textbf{Comparison with Projected GD:} Frank--Wolfe avoids the projection operation at the cost of a slower $O(1/k)$ convergence rate compared to Projected GD's linear rate $O\!\left((1-\mu/L)^k\right)$. Projected GD with $\mu = 1$ (strong convexity of $f(x) = x_1^2 + x_2^2$) and $L = 2$ would converge at rate $0.5^k$, reaching $\varepsilon = 0.01$ in $\log_{0.5}(0.01) \approx 7$ iterations --- far faster than Frank--Wolfe's $O(1/k)$ rate. For problems with expensive projections but cheap LP subproblems (e.g., nuclear norm balls, flow polytopes, spectrahedra in semidefinite programming), Frank--Wolfe is preferred.

\item \textbf{Benchmark A} ($\kappa \approx 1.12$): The problem is so well-conditioned that virtually all methods (except Polyak) converge to the noise floor in 120--150 iterations. Differentiation between methods is minimal; any step size $\alpha \in (0, 1.94)$ is stable and converges rapidly.

\item \textbf{Benchmark C} (Rosenbrock, $\kappa \approx 2508$): The strongly ill-conditioned, non-convex landscape exposes the full range of method capabilities (see Figure~\ref{fig:comprehensive_C}). Best result around 120 iterations: Polyak ($f = 0.009$, requires known $f^\star = 0$). Best first-order method without $f^\star$: Nesterov ($f = 0.479$, 150 iterations, $O(1/k^2)$ rate). Best with second-order information: Newton ($f = 0.686$, only 20 iterations; $f = 0.017$ in 40 iterations, demonstrating dramatic per-iteration improvement). Nelder--Mead (160 iterations, no gradient) navigates the valley through simplex adaptation, showing derivative-free methods can be competitive in low dimensions.
\end{enumerate}

% ============================================================
\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has investigated six families of optimisation algorithms across three canonical benchmarks. The key findings are summarised below, ordered by the problem insight each method illuminates.

\textbf{Adaptive step-size methods (Q1):} The Polyak step is theoretically optimal ($O(1/k)$ convergence) when the optimal value $f^\star$ is known exactly (Rosenbrock: $f = 0.0094$, a $374\times$ improvement over GD in the same 120 iterations), but diverges catastrophically when $f^\star$ is misspecified (Benchmark A: $f = 2.72$ vs.\ $f^\star = 0.4826$). The failure mode is analytically clear: $\alpha_k = (f(x_k) - f^\star_{\text{wrong}})/\|\nabla f(x_k)\|^2 \to \infty$ as $\|\nabla f\| \to 0$ with $f(x_k) \not\to f^\star_{\text{wrong}}$. Adagrad's monotone decaying step size provides stability at the cost of slowing late convergence ($O(1/\sqrt{k})$ decay); RMSprop's exponential forgetting factor ($\beta = 0.9$) prevents this premature decay but lacks Adagrad's rigorous convergence guarantee for sparse gradient settings. Heavy Ball benefits from momentum acceleration proportional to $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$, which for Rosenbrock's $\kappa \approx 2508$ gives a rate $\approx 0.9996^k$ for both GD \emph{and} Heavy Ball near the optimum (Heavy Ball's guarantee is for quadratics only).

\textbf{Momentum and SGD (Q2):} Nesterov's $O(1/k^2)$ optimal rate (vs.\ GD's $O(1/k)$) provides a $7\times$ improvement over GD on Rosenbrock in 150 iterations. The key mechanism is the lookahead: by evaluating $\nabla f(\tilde{x}_k)$ at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$, Nesterov ``corrects'' the momentum direction before overshooting --- this is fundamentally different from Heavy Ball which evaluates the gradient at $x_k$ and cannot self-correct. Adam's per-coordinate scaling handles anisotropy but is less effective on the Rosenbrock valley ($f = 1.579$ vs.\ Nesterov's $f = 0.479$) because the valley's curvature is highly nonlinear: per-coordinate scaling cannot capture the off-diagonal structure of the Rosenbrock Hessian. Mini-batch SGD shows the fundamental batch-size trade-off: smaller batches $b = 5$ give $200$ gradient steps per epoch (faster progress) but gradient variance $\propto 1/b$; larger batches $b = 40$ give $25$ steps per epoch with lower variance. When observation noise dominates sampling noise ($\sigma = 6$), both batch sizes converge to the same elevated floor $J^\star \approx 18$.

\textbf{Newton's method (Q3):} Newton achieves \emph{one-step} convergence on the quadratic loss (Benchmark A) --- this is exact for quadratics since Newton's direction is $H^{-1}\nabla J$ and $H^{-1}\nabla J(\theta_0) = \theta_0 - \theta^{\text{OLS}}$ gives the exact solution in a single step. For near-quadratic problems (Benchmark B: $\sim$5 iterations), quadratic local convergence begins immediately. On Rosenbrock (Benchmark C), Newton achieves a $5.4\times$ objective improvement over GD using only 20 vs.\ 80 iterations; the heavy damping $\alpha = 0.22$ is necessary because the Rosenbrock Hessian changes rapidly along the curved valley (20 steps with $\alpha = 0.22$ have not yet entered the quadratic convergence basin). Newton's $O(d^3)$ per-iteration cost (Hessian inversion) limits scalability to large-scale problems, motivating quasi-Newton methods (L-BFGS) that approximate $H^{-1}$ from gradient differences.

\textbf{Derivative-free methods (Q4):} Finite differences with $\delta = 0.05$ successfully approximate the gradient with negligible accuracy loss, while $\delta = 0.8$ introduces significant truncation error. Nesterov random search requires only 2 function evaluations per iteration (dimension-independent) but converges slowly due to one-dimensional gradient projection. Nelder--Mead adapts its simplex geometry to navigate the Rosenbrock valley without any derivative information. Grid search is the only method with a global optimality guarantee (to within grid resolution) but is ruled out by the curse of dimensionality for $d > 3$.

\textbf{Constrained optimisation (Q5):} Projected GD enforces the constraint as a hard condition from iteration~1 via a simple $\max(0.5, \cdot)$ clip, achieving exact feasibility at zero additional cost for this simple constraint set. Penalty methods approximate the constrained problem and require $\lambda \geq 1.8$ for reliable convergence in 100 iterations; the augmented Lagrangian method (not implemented) would eliminate the need for $\lambda \to \infty$ by jointly updating the Lagrange multiplier.

\textbf{Frank--Wolfe (Q6):} The conditional gradient method exploits the LP oracle to maintain feasibility automatically ($\mathcal{X}$-iterates are guaranteed by convexity of $\mathcal{X}$) without projection. It achieves $O(1/k)$ convergence with a computable convergence certificate (the Frank--Wolfe gap $G_k$). The fixed $\beta$ schedule used here causes oscillation near the optimum; switching to the decreasing $\gamma_k = 2/(k+2)$ schedule would achieve the proven $O(1/k)$ rate bound.

\textbf{Overall recommendation:} For well-conditioned smooth problems (Benchmark A, B), Adam or Nesterov momentum are practical first choices offering marginal improvements over GD with minimal hyperparameter sensitivity. For very ill-conditioned problems (Benchmark C, $\kappa \approx 2508$), damped Newton provides $O(\log\log(1/\varepsilon))$ quadratic convergence once in the local regime, dramatically outperforming all first-order methods. Derivative-free optimisation is appropriate when gradients are unavailable (black-box, non-smooth, or legacy code), with Nelder--Mead being the preferred low-dimensional solver. When constraints are present and the projection is cheap, projected GD provides the strongest guarantee (hard feasibility from iteration~1). Frank--Wolfe trades the $O(1/k)$ rate for projection-free operation, making it the method of choice when the LP oracle is cheaper than projection (e.g., nuclear norm ball, flow polytopes, spectrahedra).

\begin{lstlisting}[caption={Gradient Descent}]
def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g   # velocity accumulates gradient
        x = x - z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum (lookahead)}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)  # increasing schedule
        lookahead = x + beta_k * z                  # extrapolation point
        g = grad_fn(lookahead)                       # gradient at lookahead
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser (with bias correction)}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # bias-corrected
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))       # Tikhonov regularisation
        try:
            p = np.linalg.solve(H_reg, g)           # solve H*p = g
        except np.linalg.LinAlgError:
            p = g                                    # fallback to gradient
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (Fixed Step)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LP oracle for box constraints: O(d) per coordinate
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # gamma = 1-beta towards z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
