\begin{abstract}
This report presents a systematic comparative study of twelve optimisation algorithms across
three canonical benchmark functions of increasing difficulty: a linear regression quadratic loss
($\kappa \approx 1.12$, Benchmark A), a toy neural network loss with sinusoidal perturbation
($\kappa \leq 10$, Benchmark B), and the Rosenbrock function ($\kappa \approx 2508$,
Benchmark C).  Six algorithm families are investigated: (Q1) adaptive step-size methods
(Polyak, Adagrad, RMSprop, Heavy Ball); (Q2) momentum and stochastic methods (Nesterov
acceleration, Adam, mini-batch SGD); (Q3) Newton's method and second-order local
approximation; (Q4) derivative-free approaches (finite-difference gradients, Nesterov random
search, Nelder--Mead, grid search); (Q5) constrained optimisation (projected gradient descent,
penalty methods); and (Q6) the Frank--Wolfe conditional-gradient algorithm.

Key findings include: (i) the Polyak step achieves the best first-order convergence on
Rosenbrock when $f^{\star}$ is known, but diverges catastrophically under misspecification
--- confirmed both theoretically and numerically; (ii) Nesterov's $O(1/k^2)$ acceleration
yields a $7\times$ improvement over GD on Rosenbrock despite the function's non-convexity;
(iii) Newton's method converges in \emph{one step} on the quadratic benchmark and requires
only 20 damped steps to outperform 80 GD steps on Rosenbrock, confirming its
condition-number-independent quadratic convergence; (iv) finite differences with $\delta = 0.05$
approximate the exact gradient to within $1.2\%$ error, while $\delta = 0.8$ introduces $20\%$
error and substantially degrades convergence; (v) projected gradient descent enforces
feasibility from the first iteration as a hard constraint, while penalty methods require
$\lambda \geq 1.8$ for adequate feasibility enforcement; and (vi) Frank--Wolfe with fixed
$\beta = 0.985$ converges more smoothly than $\beta = 0.90$ for an interior optimum, and
naturally handles boundary optima without explicit projection.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments are conducted in Python 3 using a fixed random seed
(\texttt{np.random.seed(42)}) for reproducibility.  The three benchmark problems are chosen
to represent a spectrum of optimisation difficulty: a well-conditioned convex quadratic (A), a
moderately nonlinear convex function (B), and a highly nonlinear, ill-conditioned non-convex
function (C).

\subsection{Mathematical Preliminaries}
\label{sec:prelims}

We recall the standard regularity conditions used throughout:

\begin{definition}[$L$-smooth gradient]
A differentiable function $f:\mathbb{R}^d \to \mathbb{R}$ has an $L$-Lipschitz gradient
(is $L$-smooth) if for all $x, y$:
\[
\|\nabla f(x) - \nabla f(y)\| \leq L\,\|x - y\|,
\]
equivalently $f(y) \leq f(x) + \nabla f(x)^T(y-x) + \frac{L}{2}\|y-x\|^2$.
\end{definition}

\begin{definition}[$\mu$-strongly convex]
A differentiable function $f$ is $\mu$-strongly convex if for all $x, y$:
\[
f(y) \geq f(x) + \nabla f(x)^T(y-x) + \frac{\mu}{2}\|y-x\|^2.
\]
\end{definition}

For a twice-differentiable function, $L$ equals the maximum eigenvalue of $\nabla^2 f$ and
$\mu$ equals the minimum eigenvalue (when $\nabla^2 f \succ 0$). The condition number
$\kappa = L/\mu$ governs the convergence of gradient-based methods: GD requires
$O(\kappa \log(1/\varepsilon))$ iterations while Newton's method is $\kappa$-independent.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \stackrel{\mathrm{iid}}{\sim} \mathcal{N}(0, I_2),\quad
y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \stackrel{\mathrm{iid}}{\sim} \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss is:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2
= \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
The gradient and Hessian are constant:
\begin{equation}
\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y) = H\theta - b, \qquad
H = \frac{1}{m}X^T X, \quad b = \frac{1}{m}X^T y.
\end{equation}
For this dataset (seed 42), the Hessian has eigenvalues $\lambda_1 \approx 0.925$ and
$\lambda_2 \approx 1.033$, yielding condition number $\kappa \approx 1.12$ --- an extremely
well-conditioned problem.  The smoothness constant is $L = \lambda_{\max} \approx 1.033$ and
the strong convexity constant is $\mu = \lambda_{\min} \approx 0.925$.  The OLS minimiser is
$\theta^{\star}_{\mathrm{OLS}} = H^{-1}b \approx (3.028,\,4.017)^T$ (close to the true
$\theta^{\star} = (3, 4)^T$, differing due to observation noise), with minimum loss
$J^{\star} = J(\theta^{\star}_{\mathrm{OLS}}) \approx 0.4826$.  Starting point:
$\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient and Hessian are:
\begin{equation}
\nabla f = \begin{bmatrix} 2(x_1 - 1) + \cos(x_1) \\ 10(x_2 - 2) \end{bmatrix},
\qquad
\nabla^2 f = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The Hessian is diagonal; the $(1,1)$ entry $2 - \sin(x_1)$ ranges in $[1, 3]$, so the
condition number ranges between $10/3 \approx 3.3$ and $10/1 = 10$.  The global minimum
satisfies $x_2^{\star} = 2$ exactly, and $x_1^{\star}$ is the solution of:
\[
2(x_1 - 1) + \cos(x_1) = 0 \implies x_1^{\star} \approx 0.5824,
\]
giving $f^{\star} \approx 0.7244$.  Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The Rosenbrock function is not convex. Its unique global minimum is at $(1, 1)$ with
$f^{\star} = 0$.  The Hessian is:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 - 400x_2 + 1200x_1^2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}

\textbf{Hessian at the starting point} $x_0 = (-1, 1)$:
\[
\nabla^2 f(-1,1) = \begin{bmatrix} 802 & 400 \\ 400 & 200 \end{bmatrix}, \quad
\lambda_{\min} \approx 0.40,\ \lambda_{\max} \approx 1001.6,\ \kappa \approx 2504.
\]

\textbf{Hessian at the optimum} $(1, 1)$:
\begin{equation}
\nabla^2 f(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}, \quad
\lambda_{\min} \approx 0.399,\ \lambda_{\max} \approx 1001.6,\ \kappa \approx 2508.
\end{equation}

Notably, the condition number is nearly \emph{identical} at the starting point and the optimum,
despite the off-diagonal sign changing.  This is because the Hessian structure is dominated by
the $100 \cdot (x_2 - x_1^2)$ coupling term.  Along the valley floor where $x_2 \approx x_1^2$,
the Hessian eigenvalues remain approximately $\{0.4, 1001.6\}$ throughout the trajectory.
This explains why gradient-based methods struggle all the way from start to finish on Rosenbrock:
the ill-conditioning does not worsen near the optimum, but it does not improve either.

This severe ill-conditioning, combined with the narrow curved valley along $x_2 \approx x_1^2$,
makes Rosenbrock one of the most challenging standard test functions.  Starting point:
$x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For Questions 5 and 6: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\
-5 \leq x_2 \leq 10\}$.  This is a compact, convex box polytope.

\section{Question 1: Adaptive Step-Size Methods}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

For a quadratic function with Hessian $H$ (eigenvalues $\mu \leq L$), standard GD with
fixed step $\alpha \leq 1/L$ satisfies:
\begin{equation}
\label{eq:gd_rate}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k \bigl[f(x_0) - f^{\star}\bigr]
= \left(\frac{\kappa - 1}{\kappa}\right)^k \bigl[f(x_0) - f^{\star}\bigr].
\end{equation}
This linear convergence rate degrades as $\kappa \to \infty$.  Adaptive step-size methods
attempt to overcome this limitation by automatically adjusting the effective learning rate.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline
across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, regularisation $\varepsilon > 0$,
         iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is motivated by the inequality: for any $L$-smooth function with
minimum $f^{\star}$,
\[
f(x_k) - f^{\star} \geq \frac{1}{2L}\|\nabla f(x_k)\|^2,
\]
so $\alpha_k = (f(x_k) - f^{\star})/\|\nabla f(x_k)\|^2$ is the maximal step that minimises
an upper bound on the next iterate.  When $f^{\star}$ is known exactly and $f$ is convex,
the Polyak step satisfies the contraction:
\[
\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - \frac{(f(x_k) - f^{\star})^2}{\|\nabla f(x_k)\|^2},
\]
guaranteeing strict decrease of the distance to $x^{\star}$ at every step.  For strongly
convex $f$ with constant $\mu$, this implies $f(x_k) - f^{\star} = O(e^{-k\mu/L})$ --- the
same linear rate as GD with optimal fixed step, but without requiring knowledge of $L$.

When $f^{\star}$ is unknown, one uses an estimate $\tilde{f}^{\star}$; misspecification
causes $\alpha_k$ to grow unboundedly (see Section~\ref{sec:q1_convergence}).

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, regularisation $\varepsilon > 0$,
         iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$
    \Comment{Per-coordinate squared gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's per-coordinate update normalises gradients by the square root of cumulative squared
gradients.  This allows larger steps in directions with historically small gradients (rarely
updated coordinates) and smaller steps where gradients are large.  Its theoretical guarantee
in the online convex optimisation setting is the regret bound:
\[
\sum_{k=1}^{T} f_k(x_k) - \sum_{k=1}^{T} f_k(x^{\star})
\leq \frac{\|x_0 - x^{\star}\|^2}{2\alpha_0}\sum_{i=1}^{d}\sqrt{\sum_{k=1}^{T} g_{k,i}^2},
\]
which is $O(\sqrt{T}\,G_{1:T})$ where $G_{1:T}$ is the total gradient magnitude.  This adapts
to the geometry of the loss: if many coordinates have small gradients (as in sparse problems),
Adagrad achieves much better bounds than standard SGD.  The monotonically decreasing effective
step $\alpha_0 / \sqrt{G_k}$ is a feature (guaranteed convergence) but also a drawback: the
step size can become negligibly small on long runs, causing premature stagnation.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$,
         regularisation $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$
    \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's diminishing-step problem by replacing the cumulative sum with an
exponential moving average controlled by $\beta = 0.9$ (standard setting).  The effective
denominator $\sqrt{v_k}$ tracks recent gradient magnitudes, preventing over-decay and
maintaining non-negligible steps throughout training.

\paragraph{Effective window size.}
The EMA $v_k = \beta v_{k-1} + (1-\beta)g_k^2$ weights the gradient at step $j$ by
$(1-\beta)\beta^{k-j}$.  The effective number of past gradients contributing to $v_k$ is
approximately $1/(1-\beta) = 10$ for $\beta = 0.9$.  This means RMSprop forgets gradients
older than $\approx 10$ steps, using only recent history to estimate the local gradient
scale.  For Benchmark A ($m = 1000$, many iterations), this ``short memory'' means RMSprop
cannot exploit the consistent long-run gradient structure that Adagrad accumulates.
For problems with non-stationary gradients (e.g., mini-batch SGD on changing data), the
short memory is beneficial --- this is why RMSprop is preferred in online learning settings.
The parameter $\beta$ therefore controls a \emph{stationarity-adaptability trade-off}: larger
$\beta$ (longer memory) is better for stationary problems, smaller $\beta$ for non-stationary.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum coefficient $\beta \in [0,1)$,
         iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball update can be written as:
$x_{k+1} = x_k - \alpha \nabla f(x_k) + \beta(x_k - x_{k-1})$,
adding a fraction of the previous displacement.  For a quadratic $f$ with eigenvalues
$[\mu, L]$, the optimal parameters are $\alpha^{\star} = 4/(\sqrt{L}+\sqrt{\mu})^2$ and
$\beta^{\star} = ((\sqrt{L}-\sqrt{\mu})/(\sqrt{L}+\sqrt{\mu}))^2$, achieving the
$O((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^k$ convergence rate --- the same asymptotic rate as
Nesterov acceleration for strongly convex quadratics.

\paragraph{Heavy Ball vs Nesterov: a key distinction.}
Both methods use a momentum term $z_k$, but differ in \emph{where} the gradient is evaluated:
\begin{itemize}[noitemsep]
\item \textbf{Heavy Ball:} gradient at $x_k$ (current point)
\item \textbf{Nesterov:} gradient at $\tilde{x}_k = x_k + \beta_k z_k$ (lookahead point)
\end{itemize}
The lookahead allows Nesterov to evaluate the gradient at the point it will arrive at
\emph{before} applying the update, enabling it to correct overshooting.  Heavy Ball lacks this
correction: its gradient can point in the wrong direction once momentum carries the iterate
past the minimum, causing oscillations.  Importantly, Heavy Ball only provably achieves the
optimal $O(1/\sqrt{\kappa})$ rate for \emph{quadratic} functions; for general non-quadratic
functions it can diverge even on smooth convex objectives.  Nesterov's $O(1/k^2)$ guarantee
holds for all smooth convex functions.

\paragraph{Why Heavy Ball beats GD on Rosenbrock.}
Despite lacking Nesterov's guarantees on general functions, Heavy Ball ($f = 1.40$) significantly
outperforms GD ($f = 3.51$) on Rosenbrock in 120 iterations.  The mechanism is
\emph{momentum accumulation along the valley direction}.  The Rosenbrock valley lies roughly
along the curve $x_2 \approx x_1^2$, which curves gradually as $x_1$ increases from $-1$
to $1$.  The gradient perpendicular to the valley (the ``steep'' direction) is large and
oscillates in sign as the iterate crosses the valley --- momentum in this direction
averages out rapidly (the contributions cancel).  The gradient \emph{along} the valley (the
``shallow'' direction) is small but consistently points towards the minimum, so momentum
builds up in this direction and amplifies progress along the floor.

More precisely, the accumulated velocity $z_k = \alpha\sum_{j=0}^{k}\beta^{k-j}\nabla f(x_j)$
weights recent gradients by $\beta^{k-j}$.  For the along-valley component of $\nabla f$,
the signs are consistent across iterations, so $z_k$ accumulates constructively.  For the
perpendicular component, the signs alternate (overshoot-undershoot oscillation), so $z_k$
remains small.  The net effect is a velocity vector that is biased towards the valley
direction, achieving faster progress than GD's step-by-step moves.

This is precisely the advantage that Heavy Ball's ``physical'' interpretation captures: the
iterate behaves like a particle with inertia rolling down the curved valley, building speed
along it.  GD behaves as an overdamped system with no inertia, taking tiny steps in all
directions equally.

\paragraph{Heavy Ball stability region.}
For a quadratic $f$ with eigenvalues $[\mu, L]$, the Heavy Ball method is stable (converges)
if and only if:
\[
\alpha < \frac{4}{(\sqrt{L} + \sqrt{\mu})^2}(1 + \beta), \quad \beta < 1.
\]
For Benchmark A ($\mu = 0.925$, $L = 1.033$): the stability limit is approximately
$\alpha < 4(1+\beta) / (\sqrt{1.033} + \sqrt{0.925})^2 \approx 4(1+0.88) / (1.016 + 0.962)^2
\approx 7.52 / 3.91 \approx 1.92$.  The chosen $\alpha = 0.045$ is well within the stable
region.  For Benchmark C ($\mu = 0.399$, $L = 1001.6$):
$(\sqrt{L} + \sqrt{\mu})^2 \approx (31.65 + 0.632)^2 \approx 1042$, so the stability limit
is $\alpha < 4(1+0.86)/1042 \approx 0.0071$.  The chosen $\alpha = 0.0008$ is conservative
(roughly $9\times$ below the limit), ensuring stability on the highly nonlinear Rosenbrock
trajectory where the effective condition number fluctuates.

\paragraph{Key finding: Polyak sensitivity.}
The Polyak step size is theoretically optimal when $f^{\star}$ is known exactly, but
catastrophically fails when $f^{\star}$ is misspecified.  Let $\Delta = f^{\star}_{\mathrm{true}} - f^{\star}_{\mathrm{spec}}$ (the misspecification error, positive when we underestimate the minimum).  Then:
\[
\alpha_k = \frac{f(x_k) - f^{\star}_{\mathrm{spec}}}{\|\nabla f(x_k)\|^2 + \varepsilon}
= \frac{[f(x_k) - f^{\star}_{\mathrm{true}}] + \Delta}{\|\nabla f(x_k)\|^2 + \varepsilon}.
\]
For Benchmarks A and B, $f^{\star}_{\mathrm{spec}} = 0 < f^{\star}_{\mathrm{true}} \approx 0.48$,
so $\Delta = 0.48 > 0$.  As $x_k \to x^{\star}$, the excess $f(x_k) - f^{\star}_{\mathrm{true}} \to 0$
while $\|\nabla f\|^2 \to 0$.  The numerator approaches $\Delta > 0$ (bounded below), while
the denominator approaches $\varepsilon$ (small regularisation), giving
$\alpha_k \to \Delta/\varepsilon \approx 0.48/10^{-4} = 4800 \gg 1$.
Such a large step size moves the iterate far past the minimum, causing the divergent
oscillations observed in Figures~\ref{fig:q1_fval_A} and~\ref{fig:q1_fval_B}.

\textbf{Practical implication:} The Polyak step is usable when a known lower bound on
$f^{\star}$ is available (e.g., $f^{\star} = 0$ for a valid probability or norm).
Any positive overestimate $f^{\star}_{\mathrm{spec}} > f^{\star}_{\mathrm{true}}$ similarly
causes divergence.  In practice, one often uses a \emph{running minimum} estimate
$\tilde{f}_k = \min_{j \leq k} f(x_j)$ as a conservative approximation, though this still
lacks guarantees.

\paragraph{Adagrad vs RMSprop.}
On Benchmarks A and B (low condition number), both methods perform similarly.  On Benchmark C
(high condition number), Adagrad's accumulated history enforces progressively smaller steps,
which --- counterintuitively --- helps on this problem because the gradient magnitudes along
the valley decrease over time.  RMSprop's exponential forgetting loses this history, causing
its effective step to occasionally increase and destabilise.

\paragraph{Per-coordinate Adagrad analysis on Rosenbrock.}
The value of Adagrad on Rosenbrock can be understood through its per-coordinate step-size
adaptation.  Let $G_{k,i} = \sum_{j=1}^{k} g_{j,i}^2$ denote the accumulated squared gradient
for coordinate $i$.  The Rosenbrock gradient at iteration $j$ has two characteristic regimes:
\begin{enumerate}[noitemsep]
\item \textbf{Descent into the valley} (early iterations, $j$ small): the $x_2$ component of
  $\nabla f$ is large (driving $x_2 \to x_1^2$), so $G_{k,2}$ grows rapidly, shrinking the
  effective step in $x_2$.  This prevents overshooting perpendicular to the valley.
\item \textbf{Progress along the valley floor}: the $x_1$ gradient is consistently non-zero
  (since $f = (1-x_1)^2 + 100(x_2 - x_1^2)^2$ and $x_1 \to 1$ slowly), so $G_{k,1}$ also
  grows, but the $x_1$ step stabilises at a moderate value.
\end{enumerate}
The resulting effective step sizes $\alpha_0/\sqrt{G_{k,i}}$ are thus \emph{automatically
differentiated} by the historical gradient magnitudes: larger past gradients in $x_2$ produce
smaller steps there, effectively reducing the anisotropy.  This is a form of \emph{diagonal
preconditioning} analogous to using $H_{\mathrm{diag}}^{-1/2}$ as a preconditioner.

In contrast, RMSprop's exponential forgetting $G_{k,i} \leftarrow \rho G_{k-1,i} + (1-\rho)g_{k,i}^2$
means that when the algorithm is near the valley floor and gradients become small, old large
gradients are discarded.  This causes the effective step $\alpha_0/\sqrt{G_{k,i}}$ to \emph{increase}
again, risking instability.  The base rate $\alpha_0 = 0.0035$ for RMSprop is therefore chosen
conservatively, limiting overall progress.

\subsection{Backtracking Line Search: An Alternative Adaptive Step}

All adaptive methods above still require a fixed base learning rate.  An alternative is the
\emph{Armijo backtracking line search}, which at each iteration selects the step size by
checking the sufficient decrease (Armijo) condition:
\[
f(x_k - \alpha g_k) \leq f(x_k) - c\,\alpha\,\|g_k\|^2, \quad c \in (0, 1),
\]
starting from $\alpha_0$ and reducing $\alpha \leftarrow \rho\alpha$ ($\rho \in (0,1)$)
until the condition is satisfied.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q1_backtracking.pdf}
\caption{Q1 Supplement: Backtracking line search vs fixed-step GD and Nesterov on
Benchmark~C.  Backtracking GD (starting $\alpha_0 = 1$, $\rho = 0.5$, $c = 0.3$) adapts its
step size automatically to the local curvature, achieving substantially faster convergence
than fixed-step GD ($\alpha = 0.0012$) without requiring prior knowledge of $L$.  Nesterov
with a well-tuned fixed step is faster still, illustrating that momentum provides additional
benefit beyond just adaptive step sizing.}
\label{fig:q1_backtrack}
\end{figure}

Backtracking line search costs 1--2 extra function evaluations per iteration (amortised),
but can use a large initial $\alpha_0 = 1$ (or the Newton default) and automatically reduces
it, removing the need to specify $L$ or $\kappa$.  This makes it a practical default for
first-order methods in medium-scale problems.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest
  (Rosenbrock: $f = 0.0094$, Heavy Ball second at $f = 1.40$, GD worst at $f = 3.51$).
  Otherwise Adagrad and Heavy Ball match or exceed GD.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to its
  exponential averaging, damping per-coordinate oscillations.  Heavy Ball exhibits
  characteristic overshoot followed by damped oscillation, especially on anisotropic surfaces.
\item \textbf{Stability:} GD baseline is the most stable but slowest.  Polyak is
  \emph{catastrophically unstable} when $f^{\star}$ is misspecified.  Adagrad is uniformly
  stable due to its monotonically non-increasing step size.
\item \textbf{Practical recommendation:} In practice, exact $f^{\star}$ is rarely available,
  limiting Polyak's applicability to settings where a good lower bound is known (e.g.,
  non-negative loss functions with a natural zero floor).  RMSprop and Adagrad are more robust
  choices for general problems; Heavy Ball is effective when the landscape is approximately
  quadratic and step/momentum parameters can be tuned.
\end{itemize}

\subsection{Theoretical Background}

\begin{theorem}[Nesterov optimal rate, 1983]
For a convex $L$-smooth function $f$ and Nesterov's accelerated gradient method with
step $\alpha \leq 1/L$ and the momentum schedule $\beta_k = (k-1)/(k+2)$:
\[
f(x_k) - f^{\star} \leq \frac{2L\,\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\]
\end{theorem}
This $O(1/k^2)$ rate is \emph{optimal} for first-order methods on smooth convex functions:
Nesterov (1983) proved a matching \emph{lower bound} showing no first-order method can achieve
a better worst-case rate for this problem class.  The gap between GD ($O(1/k)$) and Nesterov
($O(1/k^2)$) grows with iteration count: after $k = 100$ steps, Nesterov's bound is
$10^4/k^2 \approx 1$ while GD's is $L\|x_0\|^2/k \approx L\|x_0\|^2/100$, so Nesterov
converges $100\times$ faster in the convex case.  For strongly convex functions, the optimal
first-order method achieves $O(e^{-k/\sqrt{\kappa}})$ (which Nesterov's method achieves with
fixed momentum $\beta = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$), vs GD's
$O(e^{-k/\kappa})$.  For Rosenbrock with $\kappa \approx 2508$, the ratio of rates is
$e^{-(k/\sqrt{2508}) - (-k/2508)} \approx e^{k(1/2508 - 1/50.08)} \approx e^{-0.020k}$:
after $k = 100$ steps, Nesterov is $e^{2} \approx 7.4\times$ better, close to the observed
$7\times$ factor in the experiments.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$
    \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at
$\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$.  By evaluating at the extrapolated
point, Nesterov can ``anticipate'' its next position and correct overshooting before it occurs.
The adaptive schedule $\beta_k = (k-1)/(k+2)$ gradually increases momentum, preventing the
instability that arises from immediately using high momentum:
\begin{itemize}[noitemsep]
\item At $k=1$: $\beta_1 = 0$ (pure gradient step, no momentum)
\item At $k=5$: $\beta_5 = 4/7 \approx 0.57$
\item At $k=50$: $\beta_{50} = 49/52 \approx 0.94$ (capped at $\beta_{\max}$)
\end{itemize}
In the implementation, $\beta_{\max} \in \{0.90, 0.92\}$ prevents momentum from approaching 1,
which would cause oscillation.

\centering
\includegraphics[width=\textwidth]{figures/q2_nesterov_schedule.pdf}
\caption{Q2: Nesterov momentum schedule.  Left: $\beta_k = (k-1)/(k+2)$ increases from 0 to
$\beta_{\max}$ over iterations; the cap prevents excessive momentum.  Right: rate bounds for
Nesterov ($2/(k+1)^2$) vs GD ($(1-1/\kappa)^k$) for $\kappa = 10$ (Benchmark B).  Nesterov's
$O(1/k^2)$ rate overtakes GD's geometric rate around iteration 20, illustrating when the
acceleration advantage becomes significant.}
\label{fig:q2_nesterov_sched}
\end{figure}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, first-moment decay $\beta_1$,
         second-moment decay $\beta_2$, regularisation $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$
    \Comment{First moment (gradient EMA)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$
    \Comment{Second moment (squared gradient EMA)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$
    \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam synthesises RMSprop (per-coordinate adaptive scaling via $v_t$) and momentum (via $m_t$).
The bias-correction terms $(1-\beta_1^t)$ and $(1-\beta_2^t)$ are essential in early iterations:
without them, $m_1 = (1-\beta_1)g_1 \approx 0.1 g_1$ (initialised far below the true gradient
magnitude), causing inefficiently small steps.  With bias correction,
$\hat{m}_1 = m_1 / (1-\beta_1) = g_1$ --- exactly recovering the first gradient.
As $t \to \infty$, the corrections approach 1 and the bias becomes negligible.

\paragraph{Adam's effective step magnitude.}
In coordinate $i$, the effective step at iteration $t$ is:
\[
\Delta x_{t,i} = \alpha \frac{\hat{m}_{t,i}}{\sqrt{\hat{v}_{t,i}} + \varepsilon}.
\]
When the gradient is approximately constant ($g_{t,i} \approx c$ for all $t$), the EMA estimates
converge: $\hat{m}_{t,i} \to c$, $\hat{v}_{t,i} \to c^2$, so $\Delta x_{t,i} \to \alpha$,
independent of $|c|$.  This \emph{sign-following} behaviour gives Adam near-constant effective
step in each coordinate regardless of gradient magnitude, which is both its strength (robustness
to ill-scaling) and its weakness (it may take steps of similar magnitude even near the minimum
where the step should shrink).  In practice, as $x_t \to x^{\star}$, $g_t \to 0$ and
$\hat{v}_{t,i} \to 0$, so the effective step $\to \alpha / \varepsilon \cdot \hat{m}_{t,i}$
--- bounded only by $\varepsilon$.  This is why careful tuning of $\alpha$ and $\varepsilon$
is critical for Adam near the optimum.

\begin{remark}[Adam non-convergence]
Despite its empirical success, Adam is not guaranteed to converge for general non-convex
objectives.  Reddi et al.\ (2018) showed that Adam can diverge even on convex online problems
when $\beta_2$ is close to 1, because the exponential moving average of $g^2$ can forget
gradients that pointed in the right direction.  The fix (AMSGrad) maintains a running maximum
of $v_t$, but at the cost of less adaptivity.  In practice, Adam with $\beta_1 = 0.9$,
$\beta_2 = 0.999$ works well for most deep learning problems but may require careful tuning
on ill-conditioned optimisation landscapes like Rosenbrock.
\end{remark}

\begin{remark}[Rosenbrock is non-convex]
Rosenbrock is \emph{not} a convex function, so the $O(1/k^2)$ theorem does not directly apply.
The function has a unique global minimum at $(1,1)$ (it is \emph{pseudo-convex}: $\nabla f(x) =
0 \Rightarrow x = (1,1)$), but $\nabla^2 f$ is not positive semidefinite everywhere --- for
example, at $x_0 = (-1, 1)$ one can verify $\nabla^2 f$ has a positive determinant but the
off-diagonal entries $H_{12} = 400$ can cause curvature to be ``ill-directed'' along some paths.

Nevertheless, Nesterov acceleration empirically outperforms GD on non-convex problems in
practice, particularly when the trajectory remains in a locally convex region for many
iterations.  Near the valley floor of Rosenbrock the function is locally well-approximated by
a convex quadratic (as confirmed by the near-constant Hessian eigenvalues along the trajectory),
and Nesterov's momentum effectively exploits this structure.  The $7\times$ improvement of
Nesterov over GD on Rosenbrock is therefore attributable to this local convexity, not the
global convergence theorem.
\end{remark}

\paragraph{Why Nesterov beats Adam on Rosenbrock.}
Adam's per-coordinate scaling $1/\sqrt{\hat{v}_{t,i}}$ is effective when the ill-conditioning
is axis-aligned --- i.e., when different coordinates have very different gradient magnitudes.
However, Rosenbrock's ill-conditioning arises from a \emph{coupled} (off-diagonal) Hessian:
$H_{12} = -400x_1$ at the starting point, so the eigenvalues are not aligned with the
coordinate axes.  Adam's diagonal approximation cannot correct this off-diagonal curvature.
Nesterov's momentum, by contrast, accumulates velocity along the valley floor direction
(which is consistent across many iterations), effectively exploiting the local curvature
without needing to approximate the full Hessian.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD partitions the $m = 1000$ training samples into batches of size $b$ and
performs one parameter update per batch.  The mini-batch gradient is an unbiased estimate of
the full gradient:
\[
\hat{g}(\theta; \mathcal{B}) = \frac{1}{b}\sum_{i \in \mathcal{B}}
\nabla_\theta \ell(\theta; x^{(i)}, y^{(i)}), \quad
\mathbb{E}_{\mathcal{B}}[\hat{g}] = \nabla J(\theta).
\]
The variance of the mini-batch gradient decreases with $b$:
$\mathrm{Var}[\hat{g}] = \sigma^2 / b$, where $\sigma^2$ is the per-sample gradient variance.

\paragraph{Random shuffling.}  The data is randomly permuted at every epoch (Algorithm~\ref{alg:sgd}).
This is important for two reasons: (1) it prevents \emph{cyclic bias} where the same gradients
appear in the same order across epochs, which can cause the iterates to cycle; (2) it ensures
the mini-batch gradient estimates are independent across batches within an epoch.  Without
shuffling, mini-batches that are contiguous in the dataset may be correlated, violating the
unbiasedness assumption.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\centering
\includegraphics[width=0.85\textwidth]{figures/q2_sgd_batch.pdf}
\caption{Q2: Mini-batch SGD with batch sizes $b = 5$ and $b = 40$ on Benchmark~A
($\alpha = 0.06$, 50 epochs).  With $b = 5$: $1000/5 = 200$ updates per epoch, rapid

\paragraph{Optimal batch size.}
For the linear regression objective on Benchmark A, the optimal batch size (in terms of
total gradient evaluations to reach accuracy $\varepsilon$) can be derived analytically.
Let $T_b$ be the number of epochs to reach $\varepsilon$-accuracy with batch size $b$.
Total gradient evaluations $= (m/b) \cdot T_b$.  Using the constant-step SGD convergence
bound, $T_b \propto 1/(\alpha\mu - \alpha^2 L^2/b)$ for appropriate noise scaling.  The
optimal $b$ balances the number of updates per epoch ($m/b$) against the step-size limit.
In the \emph{linear scaling regime} (Goyal et al., 2017), one can increase batch size by
$k$ and learning rate by $k$ while maintaining the same convergence speed per epoch up to
a threshold $b^{\star}$ beyond which the gradient noise is negligible compared to the
signal.  For Benchmark A with $m = 1000$, $b^{\star} \approx m \cdot \mu^2 / (L \sigma^2)
\approx 1000 \cdot 0.856 / (1.033 \cdot \sigma^2)$, which for $\sigma = 1$ gives
$b^{\star} \approx 800$.  Beyond this, increasing $b$ adds no benefit --- consistent with
the sweep showing $b = 200$ and $b = 1000$ performing similarly per epoch.

\subsection{SGD under Increased Observation Noise}

The experiment is repeated with output noise scaled by a factor of 6 ($\sigma = 6$).
Since $J^{\star} = \frac{1}{2}\mathbb{E}[(y - \theta^{\star T}x)^2] = \frac{\sigma^2}{2}$,
scaling $\sigma \mapsto 6\sigma$ raises the irreducible loss floor by factor $36$.

\centering

\caption{Q2: Effect of observation noise on SGD convergence.  Left: $\sigma = 1$, minimum

\label{fig:q2_sgd_noisy}
\end{figure}

\paragraph{Discussion.}
The key insight is that SGD has two sources of gradient noise: (1) \emph{mini-batch sampling
noise} $\propto 1/b$, which can be reduced by increasing batch size; and (2) \emph{irreducible
observation noise} from $\sigma^2$, which cannot be eliminated by changing $b$.  When the
observation noise is small ($\sigma = 1$), small batches provide a meaningful speed advantage.
When observation noise dominates ($\sigma = 6$), there is little to be gained from small
batches, and the smoother convergence of larger batches becomes preferable.

For the linear model, the mini-batch gradient $\hat{g}(\theta; \mathcal{B})$ has variance:
\[
\mathrm{Var}[\hat{g}] = \frac{\sigma^2}{b}\,\frac{1}{m}\sum_{i=1}^{m}\|x^{(i)}\|^2,
\]
so increasing $b$ linearly decreases gradient variance.  However, the irreducible floor
$J^{\star} = \sigma^2/2$ cannot be reduced regardless of $b$.

\begin{theorem}[SGD convergence rate]
Let $f = \frac{1}{m}\sum_{i=1}^m f_i$ be $\mu$-strongly convex and $L$-smooth, and let
$\hat{g}_k$ be an unbiased gradient estimate with bounded variance
$\mathbb{E}[\|\hat{g}_k - \nabla f(x_k)\|^2] \leq \sigma^2$.  Then:
\begin{itemize}[noitemsep]
\item \textbf{Constant step $\alpha$}: $\mathbb{E}[f(\bar{x}_T)] - f^{\star}
  \leq \frac{L\|x_0 - x^{\star}\|^2}{2\mu^2 T} + \frac{\alpha L \sigma^2}{2\mu^2}$,
  converging to a \emph{noise floor} $O(\alpha \sigma^2)$.
\item \textbf{Decaying step $\alpha_k = c/(k + k_0)$}: $\mathbb{E}[f(\bar{x}_T)] - f^{\star}
  = O(1/T)$ for appropriate $c, k_0$ (same asymptotic rate as full GD for strongly convex $f$,
  but with a larger constant).
\end{itemize}
\end{theorem}
The noise floor with constant step explains why SGD plateaus in the experiments: even with
$\alpha = 0.06$ and $b = 5$ on Benchmark A, the term $\alpha L \sigma^2 / (2\mu^2)
\approx 0.06 \cdot 1.033 \cdot (\sigma^2/5) / (2 \cdot 0.925^2)$ sets a lower bound on
the achievable loss.  Reducing $\alpha$ decreases this floor but also slows the initial
convergence, confirming the bias-variance trade-off observed empirically.

\paragraph{Variance reduction methods.}
A fundamental limitation of mini-batch SGD is that even with decreasing step sizes
$\alpha_k \to 0$, the convergence is $O(1/\sqrt{T})$ for convex functions (vs $O(1/T)$ for
full gradient), due to gradient noise.  \emph{Variance reduction} methods (SVRG, SAGA, SAG)
eliminate this gap by maintaining gradient estimates that become progressively more accurate:
\begin{itemize}[noitemsep]
\item \textbf{SVRG} (Johnson \& Zhang, 2013): At each \emph{outer iteration} (epoch), compute
  the full gradient snapshot $\tilde{g} = \frac{1}{m}\sum_{i=1}^{m}\nabla f_i(\tilde{x})$ at
  a reference point $\tilde{x}$.  Each \emph{inner step} then uses the variance-reduced estimator:
  \[
  v_k = \nabla f_{i_k}(x_k) - \nabla f_{i_k}(\tilde{x}) + \tilde{g},
  \]
  where $i_k$ is a random sample index.  By construction $\mathbb{E}[v_k] = \nabla f(x_k)$
  (unbiased), and crucially $\mathrm{Var}[v_k] = O(\|x_k - \tilde{x}\|^2) \to 0$ as
  $x_k \to x^{\star}$.  This shrinking variance allows a \emph{constant step size} $\alpha$,
  achieving linear convergence $O(e^{-\mu k/L})$ for $\mu$-strongly convex $f$ --- the same
  as full gradient descent, at $O(m)$ cost per epoch.

\item \textbf{SAGA} (Defazio et al., 2014): stores a table $\{g_i^{(k)}\}_{i=1}^m$ of the most
  recent per-sample gradient at each data point.  After sampling $i_k$, the update is
  $v_k = \nabla f_{i_k}(x_k) - g_{i_k}^{(k)} + \frac{1}{m}\sum_i g_i^{(k)}$, and the table
  entry $g_{i_k}^{(k+1)} \leftarrow \nabla f_{i_k}(x_k)$ is refreshed.  SAGA achieves the
  same linear rate as SVRG but updates only one gradient per step (vs a full pass for SVRG's
  snapshot), at the cost of $O(md)$ memory for the gradient table.

\item \textbf{SAG} (Schmidt et al., 2013): a simpler precursor that stores only the last
  per-sample gradient; slightly biased but similarly achieves linear convergence.
\end{itemize}
For the Benchmark A linear regression with $m = 1000$ and $d = 2$, both SVRG and SAGA would
achieve the same $O(e^{-\mu k/L})$ convergence as full GD but using only a random subset of
samples per step.  The variance reduction advantage grows with problem size: for
$m = 10^6$ samples, SVRG can achieve the same accuracy as GD in $O(\sqrt{m})$ fewer
total gradient evaluations.

\subsection{Local Approximations of $g(x) = x^4$}

\paragraph{Connection to Q1 and Q2.}
Questions 1 and 2 studied first-order methods (gradient information only), whose convergence
rate $O(e^{-\mu k/L})$ depends critically on $\kappa = L/\mu$.  Newton's method in this
question is motivated by using \emph{second-order information} (Hessian) to remove this
$\kappa$ dependence entirely.  This is not merely a quantitative improvement: on Benchmark C
($\kappa \approx 2508$), GD requires $O(\kappa) \approx 2500$ iterations for linear
convergence while Newton is condition-number-independent.

Newton's method is motivated by the second-order Taylor approximation.  Given a function
$g : \mathbb{R} \to \mathbb{R}$, expanded around $x_0 = 0.25$:

\textbf{First-order (linear) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0)
= (0.25)^4 + 4(0.25)^3(x - 0.25)
= 0.0039 + 0.0625(x - 0.25).
\label{eq:first_order}
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\label{eq:second_order}
\end{equation}

Setting the derivative of \eqref{eq:second_order} to zero:
\[
g'(x_0) + g''(x_0)(x - x_0) = 0 \implies x_{\mathrm{Newton}} = x_0 - \frac{g'(x_0)}{g''(x_0)}
= 0.25 - \frac{0.0625}{0.75} \approx 0.167.
\]
Comparing: the exact minimum of $g(x) = x^4$ is $x^{\star} = 0$.  The Newton step brings us
to 0.167, which is substantially closer than the gradient step of size $\alpha \cdot 0.0625$
for any moderate $\alpha$.  This illustrates the curvature-awareness of Newton's method.

\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$.  The first-order

\subsection{Newton's Method}

In higher dimensions, Newton's method solves the linear system $H_k\, p_k = \nabla f(x_k)$
at each step, where $H_k = \nabla^2 f(x_k) + \lambda I$ is the (regularised) Hessian.

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, Hessian damping $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$
    \Comment{Tikhonov regularisation for positive definiteness}
    \State $p_k \gets H_k^{-1}\, g_k$
    \Comment{Newton direction via linear solve}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton direction $p_k = H_k^{-1}g_k$ is the gradient in the metric defined by $H_k^{-1}$
(the \emph{natural gradient}).  It is invariant to affine transformations of the domain ---
Newton's method is \emph{affine-invariant}, meaning its iterates are identical under any
invertible linear change of variables.  Gradient descent does not share this property, which
explains why GD struggles on ill-conditioned problems while Newton does not.

\paragraph{Newton decrement and stopping criterion.}
The \emph{Newton decrement} at $x_k$ is defined as:
\[
\lambda(x_k) = \left(\nabla f(x_k)^T H_k^{-1} \nabla f(x_k)\right)^{1/2} = \|p_k\|_{H_k},
\]
which satisfies $f(x_k) - \inf_{x} q_k(x) = \tfrac{1}{2}\lambda(x_k)^2$ where $q_k$ is the
local quadratic model.  Thus $\tfrac{1}{2}\lambda(x_k)^2$ is a tight lower bound on how
much $f$ can decrease in one Newton step, making it a rigorous stopping criterion:
$\lambda(x_k)^2 / 2 < \varepsilon$ implies $f(x_k) - f^{\star} \leq \varepsilon$ (for
strictly convex $f$).  A common practical threshold is $\lambda(x_k) \leq 10^{-3}$
(corresponding to a bound of $5 \times 10^{-7}$ on the optimality gap).

\paragraph{Newton decrement evolution in the benchmarks.}
For Benchmark~A (quadratic), the decrement drops from $\lambda(x_0) \approx 2.3$ to
$\lambda(x_1) \approx 10^{-14}$ in \emph{one step}, directly reflecting one-step exactness.
For Benchmark~B, $\lambda$ drops from $\approx 3.1$ at $x_0$ to $\approx 0.04$ after three
steps and below $10^{-6}$ by step 10.  For Benchmark~C (Rosenbrock), the damped Newton step
($\alpha = 0.22$) gives a much slower decrement trajectory: $\lambda(x_0) \approx 47$ (heavily
ill-conditioned start), decreasing to $\approx 1.8$ at step 10 and $\approx 0.4$ at step 20.
This is above the convergence threshold, confirming that 20 iterations is insufficient for
Newton to enter the quadratic phase on Rosenbrock from the starting point.  In practice,
Rosenbrock would require $\approx 60$--80 Newton steps with a proper line search to satisfy
$\lambda \leq 10^{-3}$, reflecting the extreme non-convexity and the long narrow valley.

\begin{theorem}[Local quadratic convergence]
Let $f$ be twice continuously differentiable near $x^{\star}$ with $\nabla^2 f(x^{\star})$
positive definite, and $\nabla^2 f$ Lipschitz continuous.  For the full Newton step
($\alpha = 1$) and $x_0$ sufficiently close to $x^{\star}$:
\[
\|x_{k+1} - x^{\star}\| \leq C\,\|x_k - x^{\star}\|^2.
\]
\end{theorem}
This \emph{quadratic convergence} means the number of correct digits doubles at each iteration
--- dramatically faster than GD's linear convergence.

The Tikhonov regularisation $H_k + \lambda I$ with $\lambda = 10^{-8}$ serves three purposes:
(1) ensures positive definiteness ($\lambda_{\min}(H_k + \lambda I) \geq 10^{-8} > 0$);
(2) allows stable Cholesky factorisation via \texttt{np.linalg.solve};
(3) adds robustness near saddle points where $H_k$ may have near-zero or negative eigenvalues.
The choice $\lambda = 10^{-8}$ is negligible for Benchmarks A and B (whose Hessian eigenvalues
are $O(1)$--$O(10)$), and only slightly distorts the Newton direction for Benchmark C.

\paragraph{Choice of damping $\alpha$ for Rosenbrock.}
The full Newton step ($\alpha = 1.0$) works on Benchmarks A and B because the functions
are close to quadratic --- the quadratic model is accurate over a large neighbourhood.
For Rosenbrock, the full Newton step overshoots badly at the starting point $x_0 = (-1, 1)$:
the Newton direction points correctly towards $(1,1)$, but the step length under $\alpha = 1$
is $\|\alpha p_0\| = \|H_0^{-1} g_0\| \approx 2.3$, which would carry the iterate past the
minimum and onto the opposite side of the valley.

The damping $\alpha = 0.22$ is chosen to ensure that the initial step remains within the
region where the Hessian $H$ is a reasonable approximation of the true curvature.  This
corresponds to the ``cautious'' Newton step in globalisation strategies.  A Backtracking line
search (Armijo with $c = 0.3$, $\rho = 0.5$) would automatically select an appropriate
$\alpha_k$ at each step, but for simplicity a fixed $\alpha = 0.22$ is used in the
implementation.  The cost is that early iterations are more conservative than necessary (the
optimal $\alpha$ increases as the iterate approaches $x^{\star}$), but 20 iterations with
$\alpha = 0.22$ still dramatically outperforms 80 GD iterations.

\subsection{Convergence Comparison}

\paragraph{Exact one-step convergence for quadratics.}
For the strictly quadratic Benchmark A loss $J(\theta) = \frac{1}{2}\theta^T H\theta - b^T\theta + c$:
\begin{itemize}[noitemsep]
\item Gradient: $\nabla J(\theta) = H\theta - b$
\item Hessian: $\nabla^2 J(\theta) = H$ (constant)
\item Newton step from $\theta_0 = [0, 0]^T$:
  $\theta_1 = \theta_0 - H^{-1}\nabla J(\theta_0) = \mathbf{0} - H^{-1}(H\cdot\mathbf{0} - b)
  = H^{-1}b = \theta^{\star}_{\mathrm{OLS}}$
\end{itemize}
This algebra shows that \emph{any} quadratic is solved exactly in one Newton step (with full step
$\alpha = 1$), regardless of the starting point and regardless of the condition number.  For
Benchmark A, $H = \frac{1}{1000}X^T X$ and $b = \frac{1}{1000}X^T y$, so
$\theta^{\star}_{\mathrm{OLS}} = H^{-1}b = (X^T X)^{-1} X^T y \approx (3.028, 4.017)^T$.

\paragraph{Why Newton outperforms GD on ill-conditioned problems.}
Gradient descent is sensitive to the condition number $\kappa = L/\mu$ because it applies the
same step in all directions.  Newton's method rescales the gradient by the inverse Hessian,
effectively normalising the curvature: a step of $H^{-1}\nabla f$ has the same magnitude in
every eigendirection of $H$, regardless of the corresponding eigenvalue.  This
\emph{affine invariance} explains why Newton's convergence rate (in terms of iterations) is
independent of $\kappa$, making it far superior on Rosenbrock where $\kappa \approx 2508$.

\paragraph{Cost comparison and quasi-Newton methods.}
Each Newton iteration requires forming and factoring the $d \times d$ Hessian at cost
$O(d^3)$, in addition to one gradient evaluation at cost $O(d)$.  For large $d$ (e.g.,
$d = 10^6$ in neural networks), $O(d^3)$ is completely impractical.  This motivates
\emph{quasi-Newton} methods that approximate the inverse Hessian $H_k^{-1}$ from gradient
differences:
\begin{itemize}[noitemsep]
\item \textbf{BFGS} (Broyden--Fletcher--Goldfarb--Shanno): maintains a dense $d \times d$
  approximation $B_k^{-1} \approx H_k^{-1}$, updated via rank-2 corrections at $O(d^2)$
  per step.  Memory cost $O(d^2)$.
\item \textbf{L-BFGS} (Limited-memory BFGS): stores only the last $m$ gradient differences
  and curvature pairs ($m = 5$--$20$), giving an \emph{implicit} $H_k^{-1}$ approximation
  at cost $O(md)$ per step.  This is the standard method for large-scale smooth optimisation.
\end{itemize}
Both methods achieve \emph{superlinear} convergence (faster than linear, slower than quadratic)
in theory, and typically mimic Newton's convergence in practice at $O(d^2)$ or $O(md)$ cost.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation}
\label{sec:q4}
% ============================================================

\paragraph{Position in the algorithm hierarchy.}
Questions 1--3 assumed gradient (and Hessian) access.  This question investigates what happens
when gradient information is unavailable or expensive.  The key trade-off is between
\emph{information quality} and \emph{information cost}: exact gradients provide $O(1/\varepsilon)$
iterations but require $O(d)$ gradient calls; zeroth-order methods use only function values
(cost $O(1)$ per call) but require $O(d/\varepsilon^2)$ iterations --- a factor-$d$ penalty
for being derivative-free.  Section~\ref{sec:q4} makes this trade-off concrete on the
benchmarks from Q1--Q3.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\label{eq:fd}
\end{equation}
where $e_i$ is the $i$-th coordinate vector.  The total error has two components:
\begin{itemize}
\item \textbf{Truncation error:} $O(\delta)$ from Taylor expansion (the higher-order terms
  omitted in \eqref{eq:fd}).
\item \textbf{Round-off error:} $O(\varepsilon_{\mathrm{machine}} / \delta)$ from floating
  point arithmetic.
\end{itemize}
These balance at $\delta^{\star} \sim \sqrt{\varepsilon_{\mathrm{machine}}} \approx 10^{-8}$,
giving a minimum total error $\approx \sqrt{\varepsilon_{\mathrm{machine}}} \approx 10^{-8}$.
In practice, the choice $\delta \approx 0.01$--$0.1$ (truncation-error dominated) is common.

A centred difference $[f(x + \delta e_i) - f(x - \delta e_i)]/(2\delta)$ achieves $O(\delta^2)$
truncation error at the cost of $2d$ function evaluations instead of $d+1$.  The optimal
$\delta$ for centred differences is $\delta^{\star} \sim \varepsilon_{\mathrm{machine}}^{1/3}
\approx 10^{-5.3}$, roughly $10^{-5}$, giving accuracy $\approx 10^{-10}$ --- far better than
forward differences.  In practice, $\delta \in [10^{-5}, 10^{-4}]$ is a good default for
centred differences.  When noise is present in $f$ (as in stochastic settings), the
round-off floor is replaced by a noise floor $\sigma_f$, and the optimal $\delta$ is
$\delta^{\star} \sim \sigma_f^{1/2}$ (forward) or $\sigma_f^{1/3}$ (centred) --- larger
$\delta$ reduces sensitivity to noise.

\textbf{Why the implementation uses $\delta = 0.05$:}  Benchmark B's $\sin(x_1)$ component
makes the function slightly nonlinear, so very small $\delta$ picks up the quadratic term
rather than the linear term.  At $x_0 = (-1, 4)$, the second derivative $|f''| \leq 3$,
so forward difference error $\approx \delta/2 \cdot |f''| \approx 0.075$ at $\delta = 0.05$.
This is acceptable for the step sizes used.  By contrast, $\delta = 0.8$ gives truncation
error $\approx 0.8/2 \cdot 3 = 1.2$, dwarfing the true gradient value.

Figure~\ref{fig:q4_fd_error} quantifies the gradient approximation error as a function of
$\delta$ at the starting point $x_0 = (-1, 4)$ for Benchmark~B.

\begin{figure}[H]
\centering

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \Comment{Stochastic directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Nesterov random search estimates a directional derivative along a random unit vector $u_k$.
The expected update direction can be derived by integrating over the unit sphere:
\[
\mathbb{E}_{u}[\hat{g}_k] = \mathbb{E}_u\!\left[(\nabla f(x_k)^T u + O(\delta))\,u\right]
= \frac{1}{d}\nabla f(x_k) + O(\delta),
\]
using the fact that $\mathbb{E}[uu^T] = \frac{1}{d}I_d$ for $u$ uniform on the unit sphere.
This means the method is an unbiased estimator of $\frac{1}{d}\nabla f(x_k)$ in the limit
$\delta \to 0$.  For $d = 2$ (our experiments), this $1/d = 1/2$ reduction is modest; for
$d = 100$, the effective signal per step is $100\times$ smaller.

Only \emph{two} function evaluations are needed per iteration regardless of $d$,
making per-step cost $O(1)$ (vs $O(d)$ for full gradient or finite-difference methods).
The overall complexity to achieve $\varepsilon$-accuracy is $O(d/\varepsilon^2)$ steps for
smooth convex functions, vs $O(1/\varepsilon)$ for exact gradient methods --- the
factor-$d$ penalty is the price of being derivative-free.

\subsection{Part B: Derivative-Free Methods on Benchmark C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices (for $n = 2$
dimensions) and iteratively transforms it through four operations:
\begin{enumerate}[noitemsep]
\item \textbf{Reflection} ($\alpha_r = 1$): move the worst vertex $x_w$ through the centroid
  $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_w)$.
\item \textbf{Expansion} ($\gamma = 2$): if reflection improved, try farther:
  $x_e = \bar{x} + \gamma(\bar{x} - x_w)$.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection did not improve, contract towards
  centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all vertices towards
  the best vertex.
\end{enumerate}
Settings: initial simplex step $= 0.35$, iterations $= 160$.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q4_nelder_mead_C.pdf}
\caption{Q4B: Nelder--Mead on Rosenbrock.  Left: the centroid trajectory navigates the curved
valley without any gradient information, adapting the simplex geometry to the local landscape.
Right: the best objective value decreases steadily over 160 iterations, with occasional
plateaus during shrink operations where all vertices collapse towards the current best point.
Final value $f \approx 10^{-3}$--$10^{-2}$, substantially closer to the optimum than GD
achieved in a comparable number of function evaluations.}
\label{fig:q4_nm}
\end{figure}

\paragraph{How Nelder--Mead adapts to Rosenbrock.}
The key strength of Nelder--Mead on Rosenbrock is its ability to \emph{orient} the simplex
along the valley.  As the algorithm progresses, the simplex elongates in the direction of
easiest descent (the valley floor) and contracts perpendicular to it.  This implicit
adaptation to the local geometry --- without any gradient information --- makes Nelder--Mead
remarkably effective on smooth, low-dimensional problems.

\paragraph{Lack of convergence guarantee.}
Nelder--Mead has no general convergence guarantee.  McKinnon (1998) constructed explicit smooth
functions in $d = 2$ for which the method converges to non-stationary points.  The McKinnon
counterexample uses a function $f$ for which the reflection and expansion operations always
produce worse points, causing the simplex to shrink indefinitely towards a non-minimum point.
The key issue is that the standard operations (reflection, expansion, contraction) are designed
for locally convex functions; on non-convex surfaces the simplex can become trapped.

In practice, Nelder--Mead often works well on smooth, low-dimensional functions like Rosenbrock
because: (1) the Rosenbrock function has a unique global minimum and is effectively unimodal
in any bounded region; (2) the valley structure allows the simplex to orient correctly after a
few shrink operations.  Extending to $d \geq 10$, convergence becomes unreliable: the number of
shrink operations grows (each costs $d+1$ evaluations), and the simplex cannot efficiently
explore the high-dimensional space.  For $d > 10$, trust-region derivative-free methods
(e.g., COBYLA, BOBYQA) are preferred due to their convergence guarantees.

\paragraph{Function evaluation budget.}
With 160 iterations, the total function evaluations are at most $4 \times 160 = 640$ (one
reflection/expansion + check per iteration), plus the initial $d + 1 = 3$ evaluations.
In contrast, FD-GD requires $(d+1) \times T = 3 \times 120 = 360$ evaluations for 120
iterations.  Nelder--Mead uses somewhat more evaluations per unit of progress but is gradient-free.

\paragraph{Discussion.}
The methods represent a trade-off between gradient information and scalability.  When gradient
access is available and cheap (as on Benchmark B), exact GD with proper step size is
unambiguously best.  Finite differences are a practical substitute when the gradient is
unavailable but function evaluations are cheap, provided $\delta$ is chosen carefully.
Zeroth-order methods (Nesterov random search, Nelder--Mead) sacrifice per-step efficiency for
gradient-free applicability; they are valuable for black-box, simulation-based, or
non-differentiable objectives.  Grid search is only viable in very low dimensions ($d \leq 3$).

\paragraph{Practical guidance.}
The key question when choosing a derivative approximation method is: \emph{what information
is available about the objective?}
\begin{itemize}[noitemsep]
\item \textbf{Exact gradient available}: use GD, Nesterov, or Adam directly.
\item \textbf{Function only, smooth objective}: use centred finite differences
  ($\delta \approx \sqrt[3]{\varepsilon_{\mathrm{machine}}}$ for $O(\delta^2)$ accuracy).
\item \textbf{Function only, non-smooth or noisy}: use Nelder--Mead (low $d$) or
  derivative-free trust-region methods.
\item \textbf{Function only, large $d$}: use Nesterov random search or ES-based methods.
\item \textbf{Global optimum needed, small $d$}: use grid/random search with Bayesian
  refinement.
\end{itemize}

\section{Question 5: Constrained Optimisation}
\label{sec:q5}
% ============================================================

\subsection{Problem Formulation}

The constrained problem is:
\begin{equation}
\min_{x \in \mathbb{R}^2}\; f_B(x) \quad \text{subject to}\quad x_1 \geq 0.5,
\label{eq:q5_problem}
\end{equation}
starting from the infeasible point $x_0 = (0.2, 4.0)$.

The feasible set $\mathcal{C} = \{x : x_1 \geq 0.5\}$ is a half-plane (a convex set).
The starting point $x_0 = (0.2, 4.0)$ is \emph{infeasible} ($0.2 < 0.5$), meaning
initialisation violates the constraint.  This tests whether the algorithms can recover
feasibility while also minimising the objective.

The unconstrained minimum $x^{\star}_{\mathrm{unc}} \approx (0.582, 2.0)$ lies \emph{inside}
the feasible set ($0.582 > 0.5$), so the constraint is \emph{inactive} at the solution: the
constrained minimiser equals the unconstrained minimiser.  This means both projected GD and
penalty methods should converge to the same point --- the practical interest lies in
\emph{how} they handle the infeasible starting point.

\paragraph{KKT conditions.}
The constrained problem~\eqref{eq:q5_problem} satisfies the KKT conditions at
$x^{\star} = (0.582, 2.0)$:
\begin{align}
\nabla f_B(x^{\star}) + \mu \nabla c(x^{\star}) &= 0, \quad c(x) = 0.5 - x_1 \leq 0, \\
\mu &\geq 0, \quad \mu c(x^{\star}) = 0.
\end{align}
Since $c(x^{\star}) = 0.5 - 0.582 < 0$ (strict feasibility), the complementary slackness
condition forces $\mu = 0$, and the KKT condition reduces to $\nabla f_B(x^{\star}) = 0$ ---
the standard unconstrained stationarity.

\subsection{Algorithms}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$
    \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1}) = (\max(0.5, z_{k+1,1}),\, z_{k+1,2})$
    \Comment{Project onto $\mathcal{C}$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection $\Pi_{\mathcal{C}}(z) = (\max(0.5, z_1), z_2)$ is closed-form for this
half-plane constraint.  Each step maintains feasibility as a \emph{hard constraint}: the
iterate is always in $\mathcal{C}$ regardless of step size.  For smooth convex $f$ and convex
$\mathcal{C}$, projected GD with $\alpha \leq 1/L$ satisfies:
\[
f(x_k) - f(x^{\star}) \leq \frac{\|x_0 - x^{\star}\|^2}{2\alpha k},
\]
the standard $O(1/k)$ rate.

\paragraph{Projected GD convergence for this problem.}
For the specific Q5 problem, $f = f_B$ is $\mu \approx 1.45$-strongly convex
(the minimum Hessian eigenvalue of $f_B$ at the solution), and $L \approx 10$ (the maximum).
With $\alpha = 0.08 \leq 1/L = 0.1$, the strongly-convex projected GD rate is:
\[
f(x_k) - f^{\star} \leq \left(1 - 2\mu\alpha + \alpha^2 L^2/(\mu \text{ term})\right)^k
 [f(x_0) - f^{\star}] \approx (1-\mu\alpha)^k \cdot [f(x_0) - f^{\star}].
\]
Using $\mu \approx 1.45$ and $\alpha = 0.08$: contraction $\rho = 1 - 0.116 = 0.884$.
Initial gap $f(x_0) - f^{\star} \approx f_B(0.2, 4.0) - 0.7244 \approx 9.32 - 0.72 = 8.60$.
After 100 iterations: $(0.884)^{100} \cdot 8.60 \approx 4.3 \times 10^{-6}$, consistent with
the observed convergence to near machine precision.  Note: the infeasibility of $x_0$ is
handled immediately by the projection at step 1, which maps $(0.2, 4.0) \to (0.5, 4.0)$,
reducing the initial gap to $f(0.5, 4.0) - 0.7244 \approx 9.27 - 0.72 = 8.55$.

\begin{algorithm}[H]
\caption{Penalty Method (Gradient Descent on Penalised Objective)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\State Define $F_\lambda(x) = f(x) + \lambda\,\max(0, -x_1 + 0.5)$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
    \Comment{GD on penalised objective}
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method replaces the constrained problem with an unconstrained problem by adding a
term $\lambda \max(0, -x_1 + 0.5)$ that penalises violations.  As $\lambda \to \infty$, the
solution of the penalised problem approaches the constrained solution.  However, very large
$\lambda$ increases the effective Lipschitz constant of $F_\lambda$, forcing a smaller step
size to maintain stability.

Settings: Projected GD ($\alpha = 0.08$, 100 iterations); Penalty with $\lambda \in \{0.15,
1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$, $\alpha = 0.03$ for $\lambda = 4.5$);
unconstrained GD ($\alpha = 0.07$, 100 iterations).

\subsection{Comparative Discussion}

\textbf{Projected GD} enforces feasibility as a hard constraint at every step, with guaranteed
convergence for convex $f$ and convex $\mathcal{C}$.  The cost is requiring a projection oracle
--- trivial for half-planes and boxes ($O(d)$ per step), but expensive for general convex sets
(solving a QP, $O(d^3)$).

\textbf{Penalty methods} require only gradient computations of the augmented objective, with
no projection.  The weight $\lambda$ controls the trade-off between constraint satisfaction
and step-size stability.  A key limitation is that the penalty method can never achieve
\emph{exact} feasibility with finite $\lambda$ (since the gradient of $\max(0, c(x))$ is zero
in the feasible region); it can only approach feasibility asymptotically.

\paragraph{Augmented Lagrangian connection.}
The \emph{augmented Lagrangian} (method of multipliers) overcomes this limitation by iterating:
\begin{align}
x_{k+1} &= \arg\min_x\, f(x) + \mu_k c^+(x) + \tfrac{\lambda}{2}\|c^+(x)\|^2, \\
\mu_{k+1} &= \mu_k + \lambda\, c^+(x_{k+1}),
\end{align}
where $c^+(x) = \max(0, c(x))$.  The dual variable $\mu_k$ is updated to keep track of the
accumulated constraint violation.  Unlike the pure penalty method, the augmented Lagrangian
achieves exact feasibility even for fixed finite $\lambda$, and the conditioning of the
augmented objective does not deteriorate with $\lambda$.

In this problem, since the unconstrained minimum is inside the feasible set, the constraint
is inactive and both methods converge to the same point.  The projected GD is more
direct and achieves zero violation from the first iteration, making it the preferred method
for this problem.

\paragraph{What if the constraint were active?}
If the objective were shifted so that $x^{\star}_{\mathrm{unc}}$ lay outside $\mathcal{C}$
(e.g., if $x^{\star}_{\mathrm{unc},1} < 0.5$), then the constrained optimum would lie on
the boundary $x_1 = 0.5$.  In this case:
\begin{itemize}[noitemsep]
\item \textbf{Projected GD} would ``slide along'' the boundary $x_1 = 0.5$ until the
  projected gradient has no component pointing further into the feasible set.  The KKT
  multiplier $\mu > 0$ would be nonzero, indicating the constraint is binding.
\item \textbf{Penalty method} with $\lambda \to \infty$ would approach the constrained
  optimum, but very large $\lambda$ creates a nearly-singular objective that requires very
  small step sizes.  The augmented Lagrangian would handle this more gracefully.
\end{itemize}
This distinction motivates the study of both approaches even when they give the same answer
on this specific problem.

\paragraph{Penalty parameter selection.}
A key practical question is: how large should $\lambda$ be?  The sensitivity of the penalty
solution can be quantified analytically.  At the penalty minimiser $x_\lambda$, the
optimality condition is:
\begin{equation}
\nabla f(x_\lambda) = \lambda \, \nabla c^+(x_\lambda),
\label{eq:penalty_kkt}
\end{equation}
where $c^+(x) = \max(0, 0.5 - x_1)$ and $\nabla c^+ = -e_1$ when $x_1 < 0.5$.
Comparing with the KKT condition $\nabla f(x^{\star}) + \mu^{\star} \nabla c(x^{\star}) = 0$
(where $\mu^{\star} = 0$ since the constraint is inactive), we see that for the penalty
method the implicit dual variable is $\mu_\lambda = \lambda \, c^+(x_\lambda)$.  As
$\lambda \to \infty$, $\mu_\lambda \to \mu^{\star} = 0$ and $c^+(x_\lambda) \to 0$ at rate
$O(1/\lambda)$.

For the current experiment (inactive constraint), the true multiplier $\mu^{\star} = 0$, so
any $\lambda > 0$ eventually drives $x_\lambda \to x^{\star}_{\mathrm{unc}}$ as the penalty
term becomes negligible at the unconstrained minimum.  The convergence residual satisfies
$c^+(x_\lambda) = O(1/\lambda)$, consistent with the sweep in Figure~\ref{fig:q5_lambda_sweep}.

\textbf{Choosing $\lambda$ in practice.}  A common heuristic is to start with a small
$\lambda_0$ and double it every few iterations (``penalty continuation''), which avoids both
the instability of large $\lambda$ and the infeasibility of small $\lambda$.  The augmented
Lagrangian method sidesteps this entirely by adapting the dual variable explicitly, achieving
exact feasibility at fixed $\lambda$.

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm}
\label{sec:q6}
% ============================================================

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\
-5 \leq x_2 \leq 10\}$.  This is a closed convex box (polytope with four vertices).

\textbf{Linear programme:} Minimise $f(x) = a^T x = x_1 + 2x_2$ ($a = [1, 2]^T$) over
$\mathcal{X}$.  Since $a$ has strictly positive components, the minimum is attained at the
coordinate-wise lower bounds:
\begin{equation}
x^{\star} = (0.5,\ -5), \qquad f^{\star} = 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}

\begin{proposition}
The minimum of a linear function $f(x) = a^T x$ over a bounded polytope is attained at a
vertex.
\end{proposition}
\begin{proof}
A linear function has no interior critical points ($\nabla f = a \neq 0$).  By the extreme
value theorem the minimum exists on the compact set $\mathcal{X}$, and since any boundary
point of a polytope that is not a vertex can be written as a convex combination of two other
boundary points, a non-vertex boundary minimiser could be improved by moving to one of those
points.
\end{proof}

\paragraph{LP duality.}
The LP $\min_{x \in \mathcal{X}} a^T x$ over the box $l_i \leq x_i \leq u_i$ can be
written in standard form as $\min_x a^T x$ subject to $Ax \leq b$ (with $A = [I; -I]$,
$b = [u; -l]$).  The \emph{dual problem} is $\max_{\mu \geq 0} -b^T\mu$ subject to
$A^T\mu = -a$, giving dual variables $\mu = [\mu^+; \mu^-]$ with $\mu^+ - \mu^- = -a$.
By \emph{strong duality} (Slater's condition holds since $l_i < u_i$), the optimal dual
variables $\mu^{\star}$ satisfy $\mu_i^{\star+} \geq 0$ and $\mu_i^{\star-} \geq 0$, with
$\mu_i^{\star+}(x_i^{\star} - u_i) = 0$ and $\mu_i^{\star-}(l_i - x_i^{\star}) = 0$ (complementary
slackness).  For coordinate 1: $a_1 = 1 > 0$, so $x_1^{\star} = l_1 = 0.5$ (lower bound active,
$\mu_1^{\star-} = 1$).  For coordinate 2: $a_2 = 2 > 0$, so $x_2^{\star} = l_2 = -5$
(lower bound active, $\mu_2^{\star-} = 2$).  The optimal dual value equals the primal:
$f^{\star} = -b^T\mu^{\star} = -(5 \cdot 0 + (-0.5)(-1) + 10 \cdot 0 + (-(-5))(-2)) = -9.5$.
This confirms strong duality and the vertex optimum.

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe / Conditional Gradient}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial feasible $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$,
         iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\; \nabla f(x_k)^T z$
    \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
    \Comment{Convex combination: maintains feasibility}
\EndFor
\end{algorithmic}
\end{algorithm}

For box constraints, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if }
(\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ work per iteration, \emph{no projection}.  The key property is that
since both $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$, and $\mathcal{X}$ is convex, the
convex combination $x_{k+1} = \beta x_k + (1-\beta) z_k \in \mathcal{X}$ automatically.

\begin{theorem}[Frank--Wolfe convergence, O(1/k)]
For a convex $L$-smooth $f$ over a compact convex set $\mathcal{X}$ with diameter $D$, and
the step-size schedule $\gamma_k = 2/(k+2)$:
\[
f(x_k) - f^{\star} \leq \frac{2 L D^2}{k + 2}.
\]
\end{theorem}

\begin{proof}[Proof sketch]
Let $h_k = f(x_k) - f^{\star}$.  At each step, using $L$-smoothness:
\[
f(x_{k+1}) \leq f(x_k) + \gamma_k \nabla f(x_k)^T(z_k - x_k) + \frac{L\gamma_k^2}{2}D^2,
\]
where $D^2 \geq \|z_k - x_k\|^2$.  Since $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$
and $x^{\star} \in \mathcal{X}$:
$\nabla f(x_k)^T(z_k - x_k) \leq \nabla f(x_k)^T(x^{\star} - x_k) \leq f(x^{\star}) - f(x_k)
= -h_k$ (by convexity of $f$).
Substituting: $h_{k+1} \leq (1-\gamma_k)h_k + \frac{L\gamma_k^2 D^2}{2}$.
With $\gamma_k = 2/(k+2)$, induction gives $h_k \leq 2LD^2/(k+2)$.
\end{proof}

This $O(1/k)$ rate matches projected GD in iterations.  The advantage of FW lies in problems
where the LMO is much cheaper than the projection oracle (e.g., nuclear norm constraints,
simplex constraints).  For box constraints (this experiment), both oracles are $O(d)$, so
neither has a clear per-iteration advantage --- the practical difference is implementation
simplicity (FW needs no projection code).

Note: in the implementation, a fixed $\beta \in (0,1)$ is used as the mixing parameter,
corresponding to $(1-\beta)$ fraction moved towards $z_k$ at each step.  The standard
decreasing schedule $\gamma_k = 2/(k+2)$ (corresponding to $(1-\beta_k)$ in this notation)
guarantees the $O(1/k)$ rate; fixed $\beta$ does not.  Fixed $\beta$ close to 1 (small steps)
leads to smoother convergence but slower initial progress and may not converge to exact optimum.

\paragraph{Frank--Wolfe gap and step size schedules.}
A useful primal convergence certificate is the \emph{Frank--Wolfe gap} at iteration $k$:
\[
g(x_k) = \nabla f(x_k)^T(x_k - z_k) = \max_{z \in \mathcal{X}} \nabla f(x_k)^T(x_k - z),
\]
which satisfies $g(x_k) \geq f(x_k) - f^{\star}$ for convex $f$.  Thus when $g(x_k)$ is
small, the current iterate is provably near-optimal.  Computing $g(x_k)$ requires only one
additional LMO call (which returns $z_k$ anyway), making it a practical convergence criterion.

For the step size, three choices are common:
\begin{itemize}[noitemsep]
\item \textbf{Fixed} $\gamma = 1-\beta$: simple but does not guarantee the $O(1/k)$ rate.
\item \textbf{Decaying} $\gamma_k = 2/(k+2)$: guarantees $O(1/k)$ for smooth convex $f$.
\item \textbf{Exact line search}: $\gamma_k^{\star} = \arg\min_{\gamma \in [0,1]} f(x_k + \gamma(z_k - x_k))$, achieves the best per-step decrease.
\end{itemize}

\subsection{Interior vs Boundary Behaviour}

\paragraph{Interior optimum.}
When the optimum lies strictly inside $\mathcal{X}$, the gradient near the optimum is small
and the LP vertex $z_k$ jumps erratically between opposite vertices of $\mathcal{X}$ (a
vertex-chasing phenomenon).  The iterates average out these jumps through the convex
combination.  The $O(1/k)$ Frank--Wolfe convergence is slower than projected GD's $O(1/k)$
by a factor, but both achieve the same asymptotic rate.

\paragraph{Boundary optimum.}
When the optimum lies on the boundary, Frank--Wolfe is naturally suited: the LP oracle
$z_k$ will repeatedly select the vertex $(0.5, \text{argmin})$ once the gradient points
towards the infeasible region ($\nabla f_1 > 0$ drives $z_{k,1} \to 0.5$).  The iterate
converges to the constrained optimum without explicit projection.

However, the fixed $\beta = 0.93$ schedule ($\gamma = 0.07$ per step) does not decrease
sufficiently fast, leaving residual error ($f = 0.364$ vs $f^{\star} = 0.25$) after
140 iterations.  The FW convergence theorem gives a bound:
\[
f(x_k) - f^{\star} \leq \frac{2 L D^2}{k + 2}
\]
where $L$ is the smoothness constant (here $L = 2$ for $f = x_1^2 + x_2^2$) and
$D$ is the diameter of $\mathcal{X}$ ($D = \sqrt{(5-0.5)^2 + 15^2} \approx 15.7$).
This gives: $f(x_{140}) - f^{\star} \leq \frac{2 \cdot 2 \cdot 15.7^2}{142} \approx 6.9$,
a loose bound.  The empirical residual $f = 0.364 - 0.25 = 0.114$ is much smaller,
confirming that the fixed-$\beta$ schedule is practically reasonable but that a decaying
schedule $\gamma_k = 2/(k+2)$ would push residuals below $0.01$ in comparable iterations.

\paragraph{Comparison with projected gradient descent.}
Both methods maintain feasibility and achieve $O(1/k)$ convergence for smooth convex $f$.
The advantage of Frank--Wolfe is its \emph{projection-free} nature: it requires only a linear
minimisation oracle (solvable in $O(d)$ for boxes), while projection can require $O(d^3)$ for
complex convex sets.  Frank--Wolfe also produces \emph{sparse iterates} --- since $x_{k+1}$
is a convex combination of a small number of vertices, the iterate has at most $k+1$ nonzero
``vertex weights'' in its history.  This is a key practical advantage in:
\begin{itemize}[noitemsep]
\item \textbf{Low-rank matrix optimisation}: the nuclear norm ball's LMO is a rank-1 SVD,
  while projection requires a full truncated SVD.
\item \textbf{Sparse optimisation} (Lasso): the $\ell_1$ ball's LMO is $O(d)$ (argmax of
  absolute values), while projection is $O(d \log d)$ (sorting for water-filling).
\item \textbf{Large-scale ML}: when the constraint set is a probability simplex (e.g.,
  boosting, attention mechanisms), the LMO is $O(d)$ (argmin), while projection
  requires $O(d \log d)$.
\end{itemize}

\paragraph{Quantifying the zig-zag phenomenon.}
The zig-zag is an intrinsic consequence of Frank--Wolfe's LP oracle.  Near the interior
optimum $x^{\star}$, the gradient $\nabla f(x_k)$ is small and its direction rotates as
$x_k$ oscillates.  Each LMO call $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$
maps a gradient sign pattern to a vertex.  For the box $\mathcal{X}$ with vertices
$v \in \{0.5, 5\} \times \{-5, 10\}$, any small rotation of $\nabla f$ may switch which
vertex is selected.  This causes $z_k$ to jump between, say, $(0.5, -5)$ and $(5, 10)$
--- a distance of $\|v_1 - v_2\| \approx 16.8$ --- even when $x_k$ is within $0.1$ of $x^{\star}$.
Each such jump contributes a step of magnitude $(1-\beta) \cdot 16.8$ in the direction of the
new vertex, creating the oscillatory convergence visible in Figure~\ref{fig:q6_fw_gap}.

This is the fundamental reason Frank--Wolfe cannot achieve a linear convergence rate on
strongly convex problems over polytopes: the LP oracle introduces an $O(D)$ oscillation
that does not shrink below $O(1/k)$ regardless of $\beta$.  Methods that alleviate this
include \emph{away-step Frank--Wolfe} (which can also move \emph{away} from a vertex, removing
it from the active set) and \emph{pairwise Frank--Wolfe}, both of which achieve linear rates
on polytopes.

\paragraph{Limitation: $O(1/k)$ is tight.}
A key weakness of Frank--Wolfe is that it cannot achieve a linear (geometric) convergence
rate even for strongly convex $f$ over a polytope, due to the zig-zag phenomenon near the
optimum.  Lower bound results (Jaggi, 2013) confirm that $O(1/k)$ is tight for vanilla
Frank--Wolfe.  Projected gradient descent and quasi-Newton methods achieve linear rates
$O(e^{-\mu k / L})$ for strongly convex problems.  This makes Frank--Wolfe primarily
attractive when projection is expensive, not when fast convergence to high accuracy is required.

\section{Summary and Conclusions}
\label{sec:conclusions}
% ============================================================

This report has systematically investigated six families of optimisation algorithms across
three benchmark functions of increasing difficulty.  The key conclusions are as follows.

\subsection{Per-Question Summary}

\paragraph{Q1: Adaptive Step-Size Methods.}
The Polyak step size achieves the best convergence when $f^{\star}$ is known exactly
(Rosenbrock: $f = 0.0094$ in 120 iterations) but diverges catastrophically when $f^{\star}$ is
misspecified.  Adagrad and Heavy Ball are robust alternatives that consistently match or
outperform constant-step GD.  RMSprop's exponential forgetting provides smoother convergence
but can cause it to lose the accumulated gradient history that helps Adagrad on
ill-conditioned problems.

\paragraph{Q2: Momentum and Stochastic Methods.}
Nesterov acceleration's theoretical $O(1/k^2)$ advantage over GD's $O(1/k)$ translates to
substantial empirical gains on the ill-conditioned Rosenbrock function ($f = 0.479$ vs
$f = 3.388$ for GD).  Adam, which combines momentum with adaptive scaling, performs well on
all benchmarks but falls behind Nesterov on Rosenbrock.  Mini-batch SGD reveals the
bias-variance trade-off inherent in batch size selection: small batches provide more updates
per epoch but higher gradient variance, and the optimal batch size depends on the signal-to-noise
ratio of the data.

\paragraph{Q3: Newton's Method.}
Newton's method demonstrates its quadratic convergence: one step suffices for the quadratic
benchmark, and only 20 heavily-damped steps are needed to dramatically outperform 80 steps of
GD on Rosenbrock ($f = 0.686$ vs $f = 3.732$).  The affine invariance of Newton's method
explains its condition-number independence, in stark contrast to GD which degrades as
$O(\kappa^{-k})$.

\paragraph{Q4: Derivative-Free Methods.}
Finite differences with appropriate step size $\delta$ closely approximate exact GD gradients.
Poor $\delta$ choice introduces significant truncation error.  Zeroth-order methods (Nesterov
random search, Nelder--Mead) converge more slowly but are applicable to black-box functions.
Grid search is only feasible in very low dimensions due to the curse of dimensionality.

\paragraph{Q5: Constrained Optimisation.}
Projected gradient descent enforces feasibility as a hard constraint from the first iteration,
making it the gold standard when a projection oracle is available.  Penalty methods offer a
simpler implementation but require careful tuning of the penalty weight $\lambda$: too small
leaves constraints violated, too large degrades step-size stability.

\paragraph{Q6: Frank--Wolfe.}
Frank--Wolfe's projection-free nature and $O(1/k)$ convergence make it attractive for
problems with computationally expensive projections.  Its characteristic zig-zag behaviour
near interior optima is a consequence of the LP oracle selecting extreme vertices; smaller
step sizes mitigate this.  A decaying step-size schedule $\gamma_k = 2/(k+2)$ is recommended
for better practical convergence guarantees.

\subsection{Key Lessons Learned}

Three meta-level insights emerge from the six experiments:

\begin{enumerate}
\item \textbf{Condition number is the dominant performance driver for first-order methods.}
  On Benchmark A ($\kappa \approx 1.12$), all methods converge in fewer than 10 effective
  iterations; the choice of method is nearly irrelevant.  On Rosenbrock ($\kappa \approx 2508$),
  the best first-order method (Nesterov, $f = 0.479$) is $7\times$ better than the worst (GD,
  $f = 3.388$) in 150 iterations, and Newton's 20-iteration result ($f = 0.686$) requires
  understanding that even second-order methods need damping on highly non-convex functions.
  The theoretical prediction that acceleration reduces effective iterations by $\sqrt{\kappa}$
  is confirmed by the data.

\item \textbf{Robustness and optimality are in tension.}
  The Polyak step is theoretically optimal (achieves $(1-\mu/L)^k$ without tuning $L$) but
  catastrophically fragile: a $0.48$-unit misspecification of $f^{\star}$ on Benchmarks A/B
  causes divergence to $f \approx 24$, while a correct specification on Rosenbrock gives the
  best result ($f = 0.0094$).  Newton is the most powerful method in terms of convergence
  rate but requires Hessian computation and careful damping.  The most practically robust
  methods are those that require minimal structural knowledge: Adam and Nesterov perform well
  across all three benchmarks with minimal tuning.

\item \textbf{The ``derivative ladder'' provides increasing accuracy at increasing cost.}
  The experiments reveal a clear hierarchy: exact gradients $\to$ FD gradients $\to$ random
  directional derivatives $\to$ simplex/grid search.  Each step down the ladder loses gradient
  fidelity (increasing error from $O(\delta)$ to $O(d/\sqrt{k})$ to no guarantees), but
  broadens applicability (from smooth differentiable to black-box to non-smooth functions).
  Similarly, the second-order information in Newton's Hessian costs $O(d^3)$ per step but
  eliminates the $\kappa$ dependence.  Choosing the right level of information to exploit is
  the core practical decision in algorithm selection.
\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (Conditional Gradient)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
