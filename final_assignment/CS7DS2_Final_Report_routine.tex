\begin{abstract}
This report presents a comprehensive comparative study of six families of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A, $\kappa \approx 1.12$), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C, $\kappa \approx 2500$). The families investigated are: (1)~adaptive step-size methods --- Polyak, Adagrad, RMSprop, and Heavy Ball --- demonstrating the critical sensitivity of Polyak's step to $f^\star$ accuracy, formalised in Proposition~1; (2)~momentum and stochastic methods --- Nesterov acceleration and Adam --- with Nesterov achieving $f=0.479$ on Rosenbrock (seven-fold improvement over GD), consistent with the information-theoretically optimal $O(1/k^2)$ rate (Theorem~1); (3)~Newton's method with local quadratic approximations, converging to Benchmark~A's minimum in a \emph{single iteration} due to quadratic convergence (Theorem~2), and reducing Rosenbrock from $f=3.73$ to $f\approx 0$ in 150 damped steps; (4)~derivative-free methods --- finite differences, Nesterov random search, Nelder--Mead, and grid search --- revealing the $O(\delta)$-truncation vs.\ $O(\varepsilon_{\mathrm{mach}}/\delta)$-rounding trade-off with optimal $\delta^\star = \sqrt{\varepsilon_{\mathrm{mach}}}$; (5)~constrained optimisation via projected gradient descent and quadratic penalty methods, with KKT analysis showing that the constraint is inactive at the optimum ($\mu^\star = 0$ by complementary slackness); and (6)~linear programming and the Frank--Wolfe conditional gradient algorithm, whose duality gap $G_k = \nabla f(x_k)^T(x_k - z_k)$ certifies global optimality and tracks the $O(1/k)$ rate, confirmed by LP strong duality. The report also provides a method selection guide across 14 algorithms and benchmark-spanning analysis of all eight methods on Rosenbrock, showing a four-order-of-magnitude performance spread. All experiments use \texttt{np.random.seed(42)} for reproducibility.
\end{abstract}

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1{,}000$ samples and $d = 2$ parameters is generated as follows:
\begin{equation}
  \theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \sim \mathcal{N}(0, I_2), \quad
  y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad
  \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss is
\begin{equation}
  J(\theta) = \frac{1}{2m}\sum_{i=1}^{m}\bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
            = \frac{1}{2m}\|X\theta - y\|^2.
  \label{eq:loss_A}
\end{equation}
This is a convex quadratic with constant Hessian $H_A = \frac{1}{m}X^TX$ and gradient
$\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. The smoothness constant is
$L_A = \lambda_{\max}(H_A) \approx 1.033$ and the strong-convexity constant is
$\mu_A = \lambda_{\min}(H_A) \approx 0.925$; the condition number
$\kappa_A = L_A/\mu_A \approx 1.12$ is extremely low (near-spherical curvature),
making this benchmark nearly trivial for any gradient-based method. The GD convergence
factor per iteration is $\rho \approx 1 - \mu_A/L_A \approx 0.10$, meaning the error
is reduced by $\approx 90\%$ every iteration. Due to observation noise, the irreducible
loss at the ordinary-least-squares solution $\hat\theta_{\mathrm{OLS}} = (X^TX)^{-1}X^Ty$ is
$J^\star \approx 0.4826 \neq 0$. Starting point: $\theta_0 = [0,\,0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}

\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
  \label{eq:loss_B}
\end{equation}
The gradient and Hessian are
\begin{align}
  \nabla f(x) &= \bigl[2(x_1 - 1) + \cos(x_1),\;\; 10(x_2 - 2)\bigr]^T, \\
  \nabla^2 f(x) &= \mathrm{diag}\bigl(2 - \sin(x_1),\;\; 10\bigr).
\end{align}
The minimum satisfies $\partial f/\partial x_2 = 0 \Rightarrow x_2^\star = 2$ and
$2(x_1^\star - 1) + \cos(x_1^\star) = 0$, which yields $x_1^\star \approx 0.5828$ by
Newton's method on the scalar equation. Hence
$f^\star = f(0.5828, 2) \approx 0.7244$. The Hessian eigenvalue ratio at the optimum
is $\approx 10/(2 - \sin(0.5828)) \approx 7$, creating moderate anisotropy.
Starting point: $x_0 = [-1,\,4]^T$.

\begin{remark}
Contour plot annotations in the generated figures mark the approximate optimum at
$(0.61,\,2.0)$ as a visual aid; the true computed minimum is $(0.5828,\,2.0)$ per the
first-order condition above.
\end{remark}

\subsection{Benchmark C: Rosenbrock Function}

\begin{equation}
  f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
  \label{eq:loss_C}
\end{equation}
The global minimum is $f^\star = 0$ at $(x_1^\star, x_2^\star) = (1,\,1)$. The Hessian at the
optimum is $\nabla^2 f(1,1) = \bigl[\begin{smallmatrix}802 & -400\\ -400 & 200\end{smallmatrix}\bigr]$,
with eigenvalues $\approx 0.4$ and $\approx 1002$ (computed via characteristic polynomial
$\lambda^2 - 1002\lambda + 400 = 0$), yielding condition number $\kappa_C \approx 2500$. This extreme ill-conditioning, combined with the narrow
curved valley along $x_2 = x_1^2$, makes Rosenbrock the canonical hard test
function for first-order methods. Starting point: $x_0 = [-1,\,1]^T$.

\subsection{Theoretical Background}

Standard gradient descent with constant step size $\alpha$ satisfies
\begin{equation}
  f(x_k) - f^\star \;\leq\; \left(1 - \frac{\mu}{L}\right)^k [f(x_0) - f^\star]
  \label{eq:gd_rate}
\end{equation}
for $\mu$-strongly convex, $L$-smooth functions, giving linear convergence but requiring
$\alpha \leq 1/L$. Adaptive methods seek to overcome the conservative $1/L$ bound by
exploiting gradient information to set per-iteration or per-coordinate step sizes.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size GD baseline across all three
benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, approximate optimal value $f^{\star}$, tolerance $\varepsilon > 0$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{proposition}[Polyak Convergence]\label{prop:polyak}
For a convex function with known $f^\star$, the Polyak step satisfies
$\|x_{k+1} - x^\star\|^2 \leq \|x_k - x^\star\|^2 - \frac{(f(x_k)-f^\star)^2}{\|g_k\|^2+\varepsilon}$,
guaranteeing monotone reduction in distance to the optimum.
When $f^\star$ is misspecified as $\tilde f^\star < f^\star_{\mathrm{true}}$, the denominator
$\|g_k\|^2 \to 0$ while the numerator $f(x_k) - \tilde f^\star \to f^\star_{\mathrm{true}} - \tilde f^\star > 0$,
causing $\alpha_k \to \infty$ and divergence.
\end{proposition}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, $\varepsilon > 0$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative squared gradients, per-coordinate}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's effective step size $\alpha_0/(\sqrt{G_k}+\varepsilon)$ decays monotonically, providing
large steps early (large gradient, fast learning) and small steps later (accumulated curvature,
stability). For online convex optimisation, Adagrad achieves regret $O(\sqrt{T})$, which is
optimal for the problem class~\cite{duchi2011}.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, $\varepsilon > 0$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop fixes Adagrad's monotonic decay by replacing the cumulative sum with an exponential
moving average. The decay parameter $\beta = 0.9$ means past gradients are discounted geometrically,
so the effective step size adapts to recent curvature rather than all history.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in [0,1)$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method maintains a velocity $z_k$ that accumulates past gradients with geometric
decay $\beta$. For a quadratic with condition number $\kappa$, the optimal parameters
$\alpha^\star = \bigl(\frac{2}{\sqrt{L}+\sqrt{\mu}}\bigr)^2$,
$\beta^\star = \bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\bigr)^2$ give a \emph{per-iteration}
squared-error decay factor of $\bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\bigr)^2$,
substantially faster than GD's $\bigl(\frac{\kappa-1}{\kappa+1}\bigr)^2$.
For the benchmarks in this study:
\begin{itemize}[nosep]
  \item \textbf{Benchmark~A} ($\kappa_A \approx 1.12$):
        HB factor $\approx\!\left(\frac{\sqrt{1.12}-1}{\sqrt{1.12}+1}\right)^{\!2}\!\approx 0.0008$
        vs.\ GD factor $\approx 0.003$ --- both so small that convergence is near-instant;
        the 4$\times$ speedup is irrelevant when $\kappa \approx 1$.
  \item \textbf{Benchmark~C} ($\kappa_C \approx 2500$):
        HB factor $\approx\!\left(\frac{50-1}{50+1}\right)^{\!2}\!\approx 0.923$ vs.\ GD factor
        $\approx 0.999$ per iteration. Heavy Ball requires $\approx 13$ iterations per decade
        of error reduction, while GD requires $\approx 625$; a \textbf{48$\times$ speedup}.
        This improvement is only realised for quadratic objectives with exactly optimal $\alpha^\star,\beta^\star$;
        on the non-quadratic Rosenbrock, the empirical advantage is smaller (HB $f=1.40$ vs.\ GD $f=3.51$ at 120 iterations) because the curvature varies along the trajectory.
\end{itemize}

\paragraph{Key finding: Polyak step sensitivity.}
The Polyak step is the most performant method \emph{when $f^\star$ is known exactly}
(Rosenbrock: $f = 0.009$, a 370-fold improvement over GD). However, it is
the \emph{least robust}: even a small underestimate of $f^\star$ triggers divergence
because the step size grows without bound as the gradient shrinks. In practice,
$f^\star$ is almost never known exactly, so Polyak's applicability is limited to
scenarios where a lower bound on $f^\star$ can be established (e.g., via duality).
Adagrad and Heavy Ball provide the best general-purpose performance across all three
benchmarks.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
  \item \textbf{Convergence speed:} When $f^\star$ is exactly known, Polyak converges
        fastest (Rosenbrock: $f = 0.009$, factor 370$\times$ over GD). Otherwise,
        Heavy Ball and Adagrad match or exceed GD baseline. Momentum parameter
        sensitivity (Figure~\ref{fig:q1_hb_beta}) shows that $\beta = 0.88$ is
        near-optimal for Benchmark~A.

  \item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to
        exponential averaging of squared gradients, which filters out high-frequency
        oscillations in the effective step size. Heavy Ball exhibits characteristic
        overshoot-damp behaviour from momentum.

  \item \textbf{Stability:} GD baseline is most stable but slowest. Polyak is
        \emph{unstable} under $f^\star$ misspecification. Adagrad is uniformly stable
        due to its monotonically decreasing step size, though this can cause premature
        stagnation on non-stationary problems (a known limitation).

  \item \textbf{Computational cost:} All four methods require exactly one gradient
        evaluation per iteration, so cost is $O(m \cdot d)$ per step for Benchmark~A,
        identical to GD.

  \item \textbf{Practical recommendation:} In problems where $f^\star$ cannot be
        reliably estimated, Heavy Ball with tuned momentum and RMSprop with $\beta = 0.9$
        are the most robust adaptive choices, consistent with their widespread use in deep
        learning practice. Note that per-coordinate step adaptation (Adagrad, RMSprop)
        provides the most benefit when the loss Hessian is nearly diagonal; for problems
        with strong off-diagonal coupling (like Rosenbrock, see Figure~\ref{fig:q1_adagrad_percoord}),
        the advantage over plain GD is reduced and Newton-type preconditioning is preferred.
\end{itemize}

\subsection{Theoretical Background}

The theoretical motivation for momentum is the following acceleration result due to Nesterov:
any first-order method using only gradient evaluations on convex smooth functions cannot achieve
a convergence rate better than $O(1/k^2)$ in the function value. Nesterov's accelerated gradient
method \emph{achieves} this optimal rate, making it information-theoretically optimal among
gradient-based methods for convex problems.

\begin{theorem}[Nesterov Acceleration]\label{thm:nesterov}
For $L$-smooth convex $f$ and step size $\alpha \leq 1/L$, the Nesterov accelerated
gradient method satisfies
$f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2}$,
compared to GD's $O(L\|x_0-x^\star\|^2/k)$ rate.
\end{theorem}

\begin{remark}[Information-Theoretic Lower Bound]
The $O(1/k^2)$ rate of Theorem~\ref{thm:nesterov} is \emph{optimal}: Nesterov~\cite{nesterov1983}
proved that no first-order method can achieve $f(x_k) - f^\star = o\!\bigl(1/(k+1)^2\bigr)$
in the worst case over the class of $L$-smooth convex functions, regardless of the
choice of query points (the ``black-box'' lower bound). Nesterov's algorithm therefore
saturates this lower bound, making it information-theoretically optimal for first-order
convex optimisation. By contrast, GD's $O(1/k)$ rate is sub-optimal by a factor of $k$.
\end{remark}

Adam~\cite{kingma2014}, while empirically powerful, lacks the same clean convergence guarantees;
it may not converge in general non-convex settings without additional assumptions on gradient variance.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\; \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead / extrapolation point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The fundamental distinction from Heavy Ball is that the gradient is evaluated at the
\emph{extrapolated point} $\tilde x_k = x_k + \beta_k z_k$ rather than at $x_k$. This
``lookahead'' step allows the algorithm to anticipate overshoot and correct its trajectory
before crossing the optimum. The adaptive schedule $\beta_k = (k-1)/(k+2)$ gradually
increases momentum from 0 to $\beta_{\max}$, maintaining stability in early iterations.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, $\varepsilon > 0$
\State $m_0 \gets \mathbf{0},\quad v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (RMSprop)}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$ \Comment{Bias correction for cold start}
    \State $\hat{v}_t \gets v_t / (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam synthesises Heavy Ball momentum (via $m_t \approx$ exponential average of past gradients)
with RMSprop's per-coordinate scaling (via $v_t \approx$ exponential average of squared
gradients). The bias correction terms $(1-\beta_j^t)^{-1}$ prevent the moment estimates
from being biased towards zero in the first few iterations when $m_0 = v_0 = 0$.
The effective step size per coordinate is $\alpha\hat m_{t,i}/(\sqrt{\hat v_{t,i}}+\varepsilon)$,
which normalises each component by its recent root-mean-square gradient magnitude.

\begin{remark}[Adam Convergence]
Unlike Nesterov, which achieves a provable $O(1/k^2)$ rate on smooth convex functions,
Adam lacks strong theoretical convergence guarantees in the general non-convex setting.
Specifically, Adam has been shown to not converge to stationary points on simple non-convex
problems without strong additional assumptions (Reddi et al.~\cite{reddi2018} ---
``On the Convergence of Adam and Beyond''). The AMSGrad variant addresses this by using a running maximum of
$v_t$ rather than the exponential average. Nevertheless, Adam's practical empirical
performance on deep learning benchmarks typically exceeds that of Nesterov, due to its
robustness to hyperparameter choices and handling of sparse gradients.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD decomposes the full-batch gradient into unbiased estimates computed on
randomly sampled subsets $\mathcal{B} \subset \{1,\ldots,m\}$ of size $b$:
\begin{equation}
  g_{\mathcal{B}}(\theta) = \frac{1}{b}\sum_{i \in \mathcal{B}} x^{(i)}\bigl(\theta^T x^{(i)} - y^{(i)}\bigr)
  = \frac{1}{b}X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}}).
\end{equation}
Since $\mathbb{E}[g_{\mathcal{B}}] = \nabla J(\theta)$, this is an unbiased estimator. The variance
of the stochastic gradient scales as $\mathrm{Var}[g_{\mathcal{B}}] \propto \sigma^2/b$, so
smaller batches introduce more noise but provide more updates per epoch
(each epoch with batch size $b$ produces $m/b$ updates).

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\pi$
    \For{$i = 0, b, 2b, \ldots$}
        \State $\mathcal{B} \gets \{\pi_i, \ldots, \pi_{i+b-1}\}$
        \State $\theta \gets \theta - \alpha \cdot \frac{1}{b}X_\mathcal{B}^T(X_\mathcal{B}\theta - y_\mathcal{B})$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

Experiments on Benchmark~A use $\alpha = 0.06$, $b \in \{5, 40\}$, 50 epochs, data
shuffled per epoch. With $m = 1{,}000$, batch $b=5$ gives 200 updates/epoch and
batch $b=40$ gives 25 updates/epoch.

\subsection{Effect of Observation Noise on SGD}

The noisy experiment scales the observation noise by factor 6 ($\sigma_{\text{noisy}} = 6$
vs.\ $\sigma = 1$), generating $y_{\text{noisy}}^{(i)} = \theta^{\star T}x^{(i)} + 6\varepsilon^{(i)}$.
The irreducible loss floor scales quadratically with noise:
\begin{equation}
  J^\star_{\text{noisy}}
  = \frac{1}{2m}\|6\varepsilon\|^2
  = 36 \cdot \frac{1}{2m}\|\varepsilon\|^2
  = 36 \cdot J^\star_{\text{orig}}
  \approx 36 \times 0.4826 \approx 17.4
  \;\approx\; \frac{\sigma_{\text{noisy}}^2}{2} = \frac{36}{2} = 18.
\end{equation}
The last equality uses $J^\star \approx \sigma^2/2$ by the law of large numbers; the $\approx 3\%$
discrepancy is due to the finite sample ($m=1{,}000$).

\subsection{Variance Reduction and the SGD Noise Floor}

A fundamental limitation of mini-batch SGD with a fixed learning rate $\alpha$ is that it
cannot converge to $f^\star$ exactly for strongly convex objectives: the final iterate satisfies
\begin{equation}
  \mathbb{E}[f(\theta_k)] - f^\star \;\geq\; \frac{\alpha\sigma^2}{2b}
  \qquad (\text{irreducible SGD noise floor})
\end{equation}
where $\sigma^2 = \mathrm{Var}[g_i(\theta^\star)]$ is the per-sample gradient variance and $b$
is the batch size. This is seen in Figure~\ref{fig:q2_sgd_batch}: even with $b=40$, the
objective plateaus above $J^\star$ unless $\alpha \to 0$.

\textbf{Variance reduction techniques} break this barrier. The SVRG algorithm
(Stochastic Variance Reduced Gradient, Johnson \& Zhang 2013) computes a full-batch gradient
$\tilde g = \frac{1}{m}X^T(X\theta - y)$ every $m$ inner iterations and uses it as a control variate:
$g_{\mathrm{SVRG}} = g_{\mathcal{B}}(\theta) - g_{\mathcal{B}}(\tilde\theta) + \tilde g$.
This estimator satisfies $\mathrm{Var}[g_{\mathrm{SVRG}}] \to 0$ as iterates converge to $\theta^\star$
(since both terms approach $\nabla J(\theta^\star)$ when $\theta = \tilde\theta = \theta^\star$).
The resulting convergence rate is \emph{linear} (geometric), matching full-batch GD but with
$O(m/b)$ fewer gradient evaluations per outer iteration. For Benchmark~A, SVRG would achieve
$J(\theta_k) - J^\star \leq \rho^k$ for some $\rho < 1$, in contrast to mini-batch SGD's
$O(1/k) + \alpha\sigma^2/(2b)$ floor visible in the experiments.

\textbf{Practical implication:} For finite-sum objectives ($J = \frac{1}{m}\sum_i f_i$), variance
reduction methods (SVRG, SAG, SAGA) achieve the best of both worlds: $O(m)$-evaluation inner
loops with convergence rates comparable to full-batch gradient methods, effectively eliminating
the noise floor that limits plain SGD.

\subsection{Local Approximations of $g(x) = x^4$}

At $x_0 = 0.25$: $g(0.25) = 0.25^4 = 3.906 \times 10^{-3}$,
$g'(0.25) = 4 \cdot 0.25^3 = 0.0625$,
$g''(0.25) = 12 \cdot 0.25^2 = 0.75$.

\textbf{First-order (Taylor) approximation:}
\begin{equation}
  \hat g_1(x) = g(x_0) + g'(x_0)(x - x_0)
              = 3.906 \times 10^{-3} + 0.0625(x - 0.25).
\end{equation}
The minimum of $\hat g_1$ is at $-\infty$ (linear function), so the first-order
model cannot locate the minimiser.

\textbf{Second-order approximation:}
\begin{equation}
  \hat g_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
              = 3.906 \times 10^{-3} + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
The minimiser of $\hat g_2$ (i.e., the Newton step) is:
\begin{equation}
  x_1 = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\end{equation}
For $g(x) = x^4$ with true minimum at $x^\star = 0$, the error reduces from $|x_0 - x^\star| = 0.25$
to $|x_1 - x^\star| = 0.167$ --- a significant one-step improvement despite the quartic being
far from quadratic globally.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Hessian Regularisation (Damped Newton)}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0,1]$, damping $\lambda > 0$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{Damped Newton step}
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton direction $p_k = H_k^{-1} g_k$ minimises the local quadratic model
$m_k(d) = f(x_k) + g_k^T d + \frac{1}{2}d^T H_k d$. Regularisation ($\lambda = 10^{-8}$
throughout) ensures $H_k$ is positive definite even when the Hessian has near-zero or
negative eigenvalues, as can occur for non-convex functions or at saddle points.

\paragraph{Computational cost comparison.}
Each Newton step requires: (i)~a gradient evaluation ($O(md)$ for dataset size $m$,
dimension $d$); (ii)~a Hessian computation ($O(md^2)$ for empirical losses, $O(d^2)$ for
analytic Hessians); and (iii)~a linear system solve ($O(d^3)$ via Cholesky factorisation).
For the benchmarks here ($m=1{,}000$, $d=2$), Newton is trivial: $H_A$ is $2\times 2$ and
computed once; each solve costs $O(4)$ operations. In contrast, a neural network with
$d=10^6$ parameters would require $O(d^2) = 10^{12}$ entries to store $H$, consuming $\sim 8$~TB
of memory --- entirely infeasible. This is why practical second-order methods use \emph{approximations}
(L-BFGS: $O(md)$ storage, quasi-Newton updates) rather than the exact Newton step.

\begin{theorem}[Quadratic Convergence of Newton's Method]\label{thm:newton}
For twice-continuously differentiable $f$ with positive-definite Hessian in a neighbourhood
of $x^\star$, pure Newton ($\alpha = 1$) satisfies
$\|x_{k+1} - x^\star\| \leq \frac{M}{2\mu}\|x_k - x^\star\|^2$
when $\|x_k - x^\star\|$ is sufficiently small, where $M = \sup_x \|\nabla^3 f(x)\|$
and $\mu = \lambda_{\min}(\nabla^2 f(x^\star))$. This is \emph{quadratic convergence} ---
the number of correct digits doubles each iteration.
\end{theorem}

\subsection{Quasi-Newton Methods and Adam as Diagonal Newton}

\textbf{Practical limitation of Newton's method.}
Theorem~\ref{thm:newton} guarantees quadratic convergence but requires forming and inverting
the $d \times d$ Hessian at each step, costing $O(d^3)$ per iteration.
For $d > 10^4$, this is infeasible (e.g., $d = 10^6$ for a neural network requires
$8 \times 10^{12}$ bytes for $H$ alone). \emph{Quasi-Newton} methods approximate the inverse
Hessian $H^{-1}$ without computing it explicitly:

\begin{itemize}[nosep]
  \item \textbf{L-BFGS} (Limited-memory BFGS): stores $m \ll d$ past curvature pairs
        $(s_k, y_k) = (x_k - x_{k-1}, \nabla f_k - \nabla f_{k-1})$ and uses the
        secant condition $H_k s_k = y_k$ to build a rank-$2m$ update to $H_0 = I$.
        Storage: $O(md)$; convergence: superlinear; preferred for $d \leq 10^7$.
  \item \textbf{Adam as diagonal quasi-Newton}: the Adam update can be written as
        $x_{t+1} = x_t - D_t^{-1}\hat m_t$ where $D_t = \mathrm{diag}(\sqrt{\hat v_t} + \varepsilon)$.
        The matrix $D_t$ approximates $\mathrm{diag}(|\nabla^2 f(x_t)|)^{1/2}$ via the
        gradient second-moment estimator $\hat v_{t,i} \approx [\nabla^2 f]_{ii}$
        (squared gradient $\approx$ diagonal Hessian entry for well-separated eigenvalues).
        Thus Adam is a \emph{diagonal quasi-Newton method}, which explains both its
        effectiveness for axis-aligned curvature (Benchmark~B, $7$:$1$ ratio) and its
        inferiority to exact Newton on problems with strong off-diagonal coupling (Rosenbrock,
        Figure~\ref{fig:q3_hessian_geometry}).
\end{itemize}

The performance hierarchy on Rosenbrock (Figure~\ref{fig:all_methods_rosenbrock}) exactly
mirrors this theoretical ordering: exact Newton $\gg$ first-order methods with diagonal
scaling (Adam, Adagrad) $\geq$ first-order methods without scaling (GD), as predicted
by the completeness of each method's curvature approximation.

\subsection{Part~A: Finite Differences and Nesterov Random Search on Benchmark~B}

\subsubsection{Forward Finite Difference Gradient Approximation}

When $\nabla f(x)$ is unavailable, each partial derivative is approximated via:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
  \label{eq:fd}
\end{equation}
where $e_i$ is the $i$-th standard basis vector. The total approximation error has two components:
\begin{equation}
  \text{Error} = \underbrace{\tfrac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x)}_{\text{truncation: } O(\delta)}
               + \underbrace{\frac{2\varepsilon_{\mathrm{mach}}}{\delta}\|f(x)\|}_{\text{rounding: } O(\varepsilon_{\mathrm{mach}}/\delta)}.
\end{equation}
The optimal step $\delta^\star = \sqrt{2\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$ balances these
components. In practice, $\delta \approx 10^{-5}$ to $10^{-3}$ is used to remain robust against
numerical noise. Each full gradient evaluation requires $d + 1 = 3$ function calls for $d = 2$.

\subsubsection{Nesterov Random Search (Zeroth-Order Method)}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Gradient Estimation)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation scale $\delta$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Uniform on $S^{d-1}$}
    \State $\hat g_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Stochastic directional derivative}
    \State $x_{k+1} \gets x_k - \alpha\, \hat g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Note $\mathbb{E}[\hat g_k] = \nabla f_\delta(x_k)$ where $f_\delta(x) = \mathbb{E}[f(x + \delta u)]$
is a smoothed version of $f$. Only \textbf{2} function evaluations are needed per iteration,
regardless of dimension $d$ --- a critical advantage in high dimensions where finite differences
cost $O(d)$ evaluations per step. The price is a much noisier gradient estimate (scalar
projection onto a random direction).

\subsection{Part~B: Derivative-Free Methods on Benchmark~C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices
$\{v_0, v_1, v_2\} \subset \mathbb{R}^2$ sorted by $f(v_0) \leq f(v_1) \leq f(v_2)$,
and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r = 1$): $x_r = \bar x + \alpha_r(\bar x - v_2)$
        where $\bar x = \frac{1}{2}(v_0 + v_1)$ is the centroid of the best two.
  \item \textbf{Expansion} ($\gamma = 2$): if $f(x_r) < f(v_0)$, try
        $x_e = \bar x + \gamma(\bar x - v_2)$ and accept whichever of $x_r$, $x_e$ is better.
  \item \textbf{Contraction} ($\rho = 0.5$): if reflection did not improve, contract
        towards $\bar x$.
  \item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, shrink all vertices towards
        the best.
\end{enumerate}
Initial simplex step: $0.35$; iterations: 160.

\subsection{Problem Formulation}

Minimise $f_B(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)$ subject to the inequality
constraint $g(x) = 0.5 - x_1 \leq 0$ (equivalently, $x_1 \geq 0.5$), starting from the
\emph{infeasible} initial point $x_0 = (0.2,\, 4.0)$.

The unconstrained minimiser of $f_B$ is $x^\star_{\mathrm{unc}} \approx (0.5828, 2.0)$.
Since $x^\star_{\mathrm{unc},1} \approx 0.5828 > 0.5$, the unconstrained minimiser is
\emph{feasible}, so the constrained minimum coincides with the unconstrained minimum:
$x^\star_{\mathrm{con}} = x^\star_{\mathrm{unc}} \approx (0.5828, 2.0)$ with
$f^\star \approx 0.7244$. The constraint is \emph{inactive} at the solution.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial $x_0$, step size $\alpha > 0$, projection $\Pi_\mathcal{X}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_\mathcal{X}(z_{k+1}) = \bigl(\max(0.5,\; z_{k+1,1}),\; z_{k+1,2}\bigr)$ \Comment{Project onto $x_1 \geq 0.5$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial $x_0$, step size $\alpha > 0$, penalty weight $\lambda > 0$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_\lambda(x) \gets f(x) + \lambda\max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State Compute $\nabla F_\lambda(x)$: penalty sub-gradient $-\lambda$ if $x_1 < 0.5$, else 0
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Note that the penalty function $\max(0, 0.5-x_1)$ is not differentiable at $x_1 = 0.5$;
a sub-gradient is used at this point. The penalised objective $F_\lambda$ is equivalent to
the $\ell_1$ penalty formulation for the constraint $g(x) \leq 0$.

Settings: projected GD ($\alpha = 0.08$), penalty with $\lambda \in \{0.15, 1.8, 4.5\}$
($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained
GD ($\alpha = 0.07$). All methods run for 100 iterations.

\subsection{Karush--Kuhn--Tucker Optimality Conditions}

The constrained problem is: minimise $f_B(x)$ subject to $g(x) = 0.5 - x_1 \leq 0$.
For this inequality-constrained convex programme, the KKT conditions are necessary and
sufficient for a global minimum:
\begin{align}
  \text{Stationarity:}      &\quad \nabla f_B(x^\star) - \mu^\star e_1 = 0, \label{eq:kkt_stat}\\
  \text{Primal feasibility:}&\quad x_1^\star \geq 0.5, \label{eq:kkt_pf}\\
  \text{Dual feasibility:}  &\quad \mu^\star \geq 0, \label{eq:kkt_df}\\
  \text{Complementary slackness:}&\quad \mu^\star(0.5 - x_1^\star) = 0, \label{eq:kkt_cs}
\end{align}
where $\mu^\star \geq 0$ is the Lagrange multiplier for the constraint $g(x) \leq 0$,
$e_1 = (1,0)^T$, and the penalty gradient of $\max(0, 0.5 - x_1)$ in $x_1$ is $-1$
when $x_1 < 0.5$ (i.e., the outward normal to the constraint $g \leq 0$).
Since the unconstrained minimum satisfies $x_1^\star = 0.5828 > 0.5$ (strictly interior
to the feasible set), complementary slackness~\eqref{eq:kkt_cs} forces $\mu^\star = 0$:
the constraint is \emph{inactive} and its shadow price is zero. With $\mu^\star = 0$,
stationarity~\eqref{eq:kkt_stat} reduces to $\nabla f_B(x^\star) = 0$ --- exactly the
unconstrained first-order condition. Therefore $x^\star_{\mathrm{con}} = x^\star_{\mathrm{unc}}$
formally by KKT, consistent with the numerical results in Table~\ref{tab:q5_final}.
In the general case where the unconstrained minimum is infeasible ($x_1^\star < 0.5$),
the active constraint forces $x_1^\star = 0.5$, complementary slackness gives $\mu^\star > 0$,
and stationarity becomes $\partial f_B / \partial x_1 = \mu^\star > 0$, meaning the
gradient perpendicular to the boundary points inward (as expected at a constrained boundary minimum).

\subsection{Comparative Discussion}

\textbf{Projected gradient descent} enforces feasibility as a hard constraint at every
iteration. The projection onto $\{x : x_1 \geq 0.5\}$ is a simple $\max$ operation
($O(1)$ per step). This elegance makes PGD the preferred method when the constraint set
admits a cheap projection oracle. Convergence is guaranteed for convex $f$ on convex
sets: $\|x_k - x^\star\|^2 \leq \|x_0 - x^\star\|^2 / (1 + 2\alpha k \mu)$ for
$\mu$-strongly convex $f$.

\textbf{Penalty methods} soften the constraint into an objective term, requiring no
projection but introducing a bias: the penalised minimiser $x^\star_\lambda$ satisfies
$x_{\lambda,1} \geq 0.5 - C/\lambda$ for some constant $C$, so feasibility is achieved
only as $\lambda \to \infty$. Very large $\lambda$ makes $F_\lambda$ poorly conditioned
(large gradient near the boundary), necessitating smaller step sizes. The sequence
$\lambda_k \to \infty$ (progressive penalty) recovers exact feasibility in the limit.

In this problem the unconstrained minimum is already feasible (Table~\ref{tab:q5_final}),
so both methods converge to the same point $x^\star \approx (0.5828, 2.0)$ with
$f^\star \approx 0.7244$. This is an important special case: when the unconstrained minimum
lies inside $\mathcal{X}$, the constraint is \emph{inactive} at the optimum, and
$x^\star_{\mathrm{con}} = x^\star_{\mathrm{unc}}$.

In the more general case where $x^\star_{\mathrm{unc}}$ is infeasible, projected GD
converges to the constrained minimum $x^\star_{\mathrm{con}} \in \partial\mathcal{X}$,
while the penalty method converges to an $O(1/\lambda)$ neighbourhood of it for any finite
$\lambda$ (systematic bias). In that case, projected GD is strictly superior for exact
constraint satisfaction.

\subsection{Feasible Set and Linear Programme}

Feasible box: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$.

\textbf{Linear programme:} minimise $f(x) = a^Tx$ with $a = [1, 2]^T$ over $\mathcal{X}$.
Since $f$ is linear, its minimum over the compact convex set $\mathcal{X}$ is attained at
a \emph{vertex} (extreme point) of $\mathcal{X}$ by the fundamental theorem of linear programming.
The four vertices of the box are $\{(0.5,-5), (5,-5), (0.5,10), (5,10)\}$; since both
components of $a$ are positive, the minimum is at the vertex with smallest coordinates:
\begin{equation}
  x^\star = (0.5,\; -5), \qquad f^\star = a^Tx^\star = 0.5 + 2(-5) = -9.5.
  \label{eq:lp_solution}
\end{equation}

\paragraph{LP duality and strong duality.}
The dual of the LP $\min_{l \leq x \leq u} a^Tx$ is derived via the Lagrangian:
$\mathcal{L}(x, \mu, \nu) = a^Tx - \mu^T(x-l) - \nu^T(u-x)$ where $\mu, \nu \geq 0$.
The dual function is $d(\mu,\nu) = \mu^Tl - \nu^Tu$ whenever $a - \mu + \nu = 0$
(else $-\infty$). The dual problem is:
\begin{equation}
  \max_{\mu, \nu \geq 0}\; \mu^Tl - \nu^Tu \quad \text{s.t.}\quad \mu - \nu = a,
\end{equation}
and by LP strong duality, $d^\star = f^\star = -9.5$ (no duality gap).
At the optimal vertex $x^\star = (0.5, -5)$: both constraints $x_i \geq l_i$ are active,
so by complementary slackness $\nu_i = 0$ and $\mu_i = a_i > 0$ ($i=1,2$).
The dual certificate is $\mu^\star = (1, 2) = a$, confirming $f^\star = \mu^{\star T}l - 0 = 1\cdot 0.5 + 2\cdot(-5) = -9.5$.
In Frank--Wolfe, the LP vertex $z_k$ is precisely the dual certificate at each iteration:
the duality gap $G_k = \nabla f(x_k)^T(x_k - z_k)$ measures how far the current iterate
is from satisfying the LP dual conditions.

The level sets are parallel lines with slope $-1/2$, and the minimum lies where the

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe with Fixed Mixing Parameter}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\; \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination; step size $\gamma = 1 - \beta$}
\EndFor
\end{algorithmic}
\end{algorithm}

\paragraph{Key properties.}
(i)~\textbf{Projection-free:} every iterate is a convex combination of feasible points, hence
automatically feasible.
(ii)~\textbf{Sparse iterates:} $x_k$ is a convex combination of at most $k+1$ vertices,
giving interpretable sparse solutions in machine learning applications.
(iii)~\textbf{LMO for box:} the subproblem $\min_{z \in \mathcal{X}} g^Tz$ decomposes
per coordinate:
\begin{equation}
  (z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
  \label{eq:fw_lmo}
\end{equation}
requiring $O(d)$ operations per step.

\paragraph{Standard vs.\ fixed step.}
The standard Frank--Wolfe algorithm uses the decaying schedule $\gamma_k = 2/(k+2)$,
achieving the optimal $O(1/k)$ convergence rate for smooth convex functions:
$f(x_k) - f^\star \leq 2LC^2/(k+2)$ where $C = \text{diam}(\mathcal{X})$.
The implementation here uses a \emph{fixed} $\gamma = 1 - \beta$, which corresponds to
a constant step towards the LP vertex. Fixed step does not give $O(1/k)$ convergence
in general, but can converge faster in early iterations when the optimal step size
is approximately constant.

\subsection{Interior vs.\ Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits characteristic ``zig-zag'' behaviour because
the LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ as the gradient direction
changes sign near the optimum. The convex combination progressively averages these jumps,
driving $x_k$ toward the interior minimum. For the standard decaying step, convergence
satisfies $f(x_k) - f^\star \leq 2LC^2/(k+2) = O(1/k)$; for fixed $\gamma$, the algorithm
can oscillate persistently for interior optima (Figure~\ref{fig:q6_fw_step_comparison},
left), with $\beta = 0.985$ (smaller step) outperforming $\beta = 0.90$ at 180 and 300
iterations.

\textbf{Boundary optimum:} Frank--Wolfe is particularly well suited to boundary optima
because the LP vertex $z_k$ consistently selects the box corner nearest to the
unconstrained minimum, naturally driving iterates toward the constraint boundary.
Fixed $\beta = 0.93$ achieves $f = 0.364$ after 140 iterations but plateaus due to
$x_2$-component oscillation; the decaying schedule converges to $f = 0.251 \approx f^\star$
in 300 iterations (Figure~\ref{fig:q6_fw_step_comparison}, right).

\textbf{Frank--Wolfe vs.\ Projected GD:} PGD achieves exact convergence to the constrained
minimum with the correct (linear) convergence rate for strongly convex $f$. FW~\cite{frankwolfe1956}
converges at the sublinear $O(1/k)$ rate. However, FW is preferred when the projection is expensive
(e.g., onto a nuclear-norm ball requires a full SVD, $O(n^3)$), whereas the LMO
(Equation~\eqref{eq:fw_lmo}) is always $O(d)$ for box constraints. In high-dimensional
machine learning problems this scalability advantage often outweighs the slower convergence rate.

\paragraph{Frank--Wolfe duality gap.}
The \emph{Frank--Wolfe gap} at iteration $k$ is
\begin{equation}
  G_k = \nabla f(x_k)^T (x_k - z_k) \geq 0,
  \label{eq:fw_gap}
\end{equation}
where $z_k$ is the current LP vertex. It satisfies $G_k \geq f(x_k) - f^\star$ by convexity of
$f$ (since $z_k$ minimises $\nabla f(x_k)^T z$ over $\mathcal{X}$, the linearisation
lower-bounds $f^\star$). Thus $G_k = 0$ certifies \emph{global optimality} without knowing
$f^\star$, making it the natural convergence certificate for Frank--Wolfe. The standard
algorithm's $O(1/k)$ rate can equivalently be written as $G_k \leq 2LC^2/(k+2)$. In
practice, monitoring $G_k$ rather than $f(x_k) - f^\star$ allows convergence to be
verified even when the optimal value is unknown.

This study compared 12 distinct optimisation algorithms across three benchmark functions
spanning different difficulty levels (quadratic, mildly non-convex, strongly ill-conditioned).
The following overarching conclusions emerge:

\begin{enumerate}[nosep]
  \item \textbf{Algorithm--problem matching is critical.} Newton's method with exact Hessian
        is unmatched for (near-)quadratic problems, converging in 1 step on Benchmark~A.
        However, on strongly non-convex landscapes (Rosenbrock), it requires damped steps
        and still does not reach the minimum in 20 iterations. Nesterov acceleration
        provides the best first-order convergence on all benchmarks.

  \item \textbf{Adaptive step sizes help but are fragile.} Polyak achieves the best
        Rosenbrock convergence when $f^\star$ is known, but diverges catastrophically when
        $f^\star$ is misspecified. Adagrad and Heavy Ball are more robust alternatives.

  \item \textbf{The Rosenbrock function ($\kappa \approx 2500$) exposes every algorithm's
        limitations.} Even Newton's method with damping only reaches $f = 0.686$ in 20 steps
        from $x_0 = (-1,1)$; Nesterov with 150 steps achieves $f = 0.479$; GD barely moves
        with $f = 3.39$ in 150 steps. Ill-conditioning is the dominant challenge for
        first-order methods.

  \item \textbf{Derivative-free methods trade gradient efficiency for applicability.}
        Finite differences (small $\delta$) closely track exact GD at cost $d+1$ evaluations/step.
        Nesterov random search requires only 2 evaluations/step but converges much more slowly.
        Nelder--Mead adapts its simplex without any gradient information and is reliable for
        small-dimensional problems.

  \item \textbf{Constraint handling strategy depends on the projection cost.} Projected GD
        is simpler, guarantees hard feasibility, and converges at the same rate as unconstrained
        GD. Penalty methods are more flexible but introduce bias controlled by $\lambda$.
        Frank--Wolfe avoids projection entirely at the cost of a sublinear $O(1/k)$ rate.

  \item \textbf{Mini-batch SGD batch size governs variance--speed trade-off.} Smaller batches
        provide more updates per epoch but higher gradient variance. Under high observation
        noise, the batch-size advantage diminishes as the irreducible loss floor dominates.
\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z   # extrapolation point
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)  # Newton direction
        except np.linalg.LinAlgError:
            p = g                           # fallback to gradient
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LMO: minimise g^T z over box
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # fixed-step convex combination
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
