\begin{abstract}
This report presents a comprehensive comparative study of six categories of optimisation algorithms applied to three benchmark problems of increasing difficulty: a well-conditioned linear regression quadratic loss (Benchmark~A, $\kappa \approx 1.12$), a mildly anisotropic toy neural network loss with sinusoidal perturbation (Benchmark~B, $\kappa \in [3.5,\,6.9]$), and the ill-conditioned Rosenbrock function (Benchmark~C, $\kappa \approx 2508$).

\textbf{Question~1} investigates adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball). The Polyak step achieves the best convergence on Benchmark~C ($f = 0.024$) where $f^{\star} = 0$ is exact, but oscillates on A and B due to $f^{\star}$ misspecification. Heavy Ball achieves $f = 1.40$ on Benchmark~C, outperforming vanilla GD ($f = 3.51$).

\textbf{Question~2} examines momentum-based and stochastic methods. Nesterov acceleration achieves the best convergence on Benchmark~C ($f = 0.479$), demonstrating its $O(1/k^2)$ advantage over GD's $O(1/k)$ rate. Mini-batch SGD with smaller batches converges faster per epoch but with higher variance; increasing observation noise by $6\times$ raises the irreducible loss floor by $\approx 36\times$.

\textbf{Question~3} demonstrates Newton's method's superiority: it solves the quadratic Benchmark~A exactly in one iteration, and reaches $f = 0.686$ on Benchmark~C in only 20 steps compared to GD's $f = 3.732$ in 80 steps, exhibiting the characteristic quadratic convergence rate.

\textbf{Question~4} analyses derivative-free approaches. Finite differences with a well-chosen perturbation ($\delta = 0.05$) closely track exact GD, while $\delta = 0.8$ degrades accuracy. Nelder--Mead navigates the Rosenbrock valley without gradient information. Grid search ($55 \times 55$) is accurate to the grid resolution but scales exponentially with dimension.

\textbf{Question~5} compares projected gradient descent (hard constraint enforcement) with penalty methods ($\lambda \in \{0.15, 1.8, 4.5\}$) for the constraint $x_1 \geq 0.5$. Since the unconstrained minimum of Benchmark~B lies at $x_1^{\star} \approx 0.582 > 0.5$, the constraint is inactive at the optimum; projected GD achieves exact feasibility from iteration~1 and converges to the same point as unconstrained GD.

\textbf{Question~6} analyses the Frank--Wolfe algorithm on a box-constrained feasible set. The linear programme's minimum is identified at a vertex $(0.5, -5)$ with $f^{\star} = -9.5$. Frank--Wolfe with $\beta = 0.985$ achieves $f = 0.0046$ for the interior case, better than $\beta = 0.90$ ($f = 0.0089$), because the conservative step size reduces oscillation near the optimum.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmark functions span a range of geometric difficulty, quantified by the condition number $\kappa = L/\mu$ where $L$ is the gradient Lipschitz constant and $\mu$ is the strong convexity parameter (when applicable).

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \in \mathbb{R}^2, \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2}\theta^T H\theta - b^T\theta + c,
\end{equation}
where $H = \frac{1}{m}X^T X$ (constant Hessian), $b = \frac{1}{m}X^T y$. With seed 42, the eigenvalues of $H$ are approximately $\lambda_{\min} \approx 0.925$ and $\lambda_{\max} \approx 1.033$, giving condition number $\kappa \approx 1.12$ --- an exceptionally well-conditioned problem. The gradient is $\nabla J(\theta) = H\theta - b = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the minimum loss is $J^{\star} \approx 0.4826$ (the irreducible noise variance), not zero. The OLS solution $\theta^{\star}_{\text{OLS}} \approx (3.028, 4.017)^T$ deviates slightly from $\theta^{\star} = (3, 4)^T$ due to finite-sample noise. Starting point: $\theta_0 = (0, 0)^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = \bigl[2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)\bigr]^T$ and the Hessian is $H(x) = \operatorname{diag}\!\bigl(2 - \sin(x_1),\ 10\bigr)$. The first eigenvalue $2 - \sin(x_1)$ varies with $x_1$: at the starting point $x_1 = -1$, it equals $2 - \sin(-1) \approx 2.84$; at the optimum $x_1 \approx 0.582$, it is approximately $1.45$. Consequently, the condition number ranges from $\kappa \approx 3.5$ (near the starting point) to $\kappa \approx 6.9$ (at the optimum). The global minimum satisfies $\frac{\partial f}{\partial x_2} = 10(x_2 - 2) = 0$ (giving $x_2^{\star} = 2.0$) and $\frac{\partial f}{\partial x_1} = 2(x_1-1) + \cos(x_1) = 0$, solved numerically to give $x_1^{\star} \approx 0.582$. The minimum function value is $f^{\star} \approx 0.7244$. Starting point: $x_0 = (-1, 4)^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is
\begin{equation}
H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $\lambda_{\min} \approx 0.40$ and $\lambda_{\max} \approx 1001.6$, giving condition number $\kappa \approx 2508$. This extreme ill-conditioning, combined with the narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock the most challenging standard test function. The gradient varies rapidly along the valley, requiring very small steps to maintain stability. Starting point: $x_0 = (-1, 1)^T$.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theoretical basis:} For $L$-smooth functions, the Polyak step $\alpha_k = (f(x_k)-f^\star)/\|g_k\|^2$ achieves the optimal per-step decrease when $f^\star$ is known exactly. For strongly convex functions with parameter $\mu$, it yields linear convergence: $\|x_{k+1} - x^\star\| \leq (1 - \mu/L)^k \|x_0 - x^\star\|$. The key sensitivity is in the numerator: if $f^\star$ is underestimated (here, $f^\star = 0$ while the true minimum may be positive), then as $x_k \to x^\star_{\text{true}}$, the numerator remains $\approx f^\star_{\text{true}} > 0$ while $\|g_k\|^2 \to 0$, causing $\alpha_k \to \infty$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theoretical basis:} Adagrad's effective learning rate for coordinate $i$ is $\alpha_k^{(i)} = \alpha_0 / (\sqrt{\sum_{j=0}^{k} g_j^{(i)2}} + \varepsilon)$, which decreases monotonically. This provides large steps along coordinates with small historical gradients (rare but important features) and small steps along frequently-updated coordinates. In the online convex setting, Adagrad achieves regret $O(\sqrt{T})$. The drawback is that the rate decays to zero for all coordinates, even those still needing updates.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theoretical basis:} RMSprop addresses Adagrad's vanishing learning rate by replacing the cumulative sum of squared gradients with an exponential moving average (EMA) with decay $\beta = 0.9$. The EMA ``forgets'' old gradients, maintaining a non-zero effective step size throughout training. This is particularly beneficial for non-stationary loss landscapes.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theoretical basis:} The Heavy Ball method adds a momentum term $\beta z_k$ to the gradient update, allowing the algorithm to build up speed along consistent gradient directions and dampen oscillations in transverse directions. For strongly convex quadratics with $L$-smooth gradients, the optimal choice $\alpha^{\star} = (2/(\sqrt{L}+\sqrt{\mu}))^2$ and $\beta^{\star} = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$ achieves convergence rate $((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^k$, compared to $((1-1/\kappa))^k$ for GD.

For Benchmark~C ($\kappa \approx 2508$): GD rate $\approx 1 - 1/2508 \approx 0.9996$; Heavy Ball optimal rate $\approx ((\sqrt{2508}-1)/(\sqrt{2508}+1))^2 \approx ((\sqrt{2508}-1)/(\sqrt{2508}+1))^2 \approx ((50.1-1)/(50.1+1))^2 \approx (0.961)^2 \approx 0.924$. Thus Heavy Ball requires approximately $\log(1/\varepsilon)/\log(1/0.924) \approx 13\log(1/\varepsilon)$ iterations, while GD requires $2508\log(1/\varepsilon)$ --- a $\sim 190\times$ speedup. The observed Heavy Ball advantage ($f = 1.40$ vs GD's $3.51$) is consistent with this theoretical prediction, though the non-quadratic nature of Benchmark~C means the optimal $\alpha^{\star}$ and $\beta^{\star}$ for the quadratic setting are not exactly applicable.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed (when $f^{\star}$ known):} Polyak is fastest on Benchmark~C ($f = 0.024$ vs GD's $3.514$, a $147\times$ improvement). On Benchmarks~A and~B where $f^{\star} = 0$ is incorrect, Polyak oscillates above the true minimum (ending at $f = 4.77$ and $0.98$ respectively). Heavy Ball is the second-best method on ill-conditioned Benchmark~C ($f = 1.40$, a $2.5\times$ improvement over GD), demonstrating the value of momentum in high-$\kappa$ settings.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to EMA of squared gradients. Heavy Ball exhibits characteristic overshoot followed by damped oscillation on anisotropic landscapes. Adagrad's monotonically decreasing step produces a nearly monotone loss curve.
\item \textbf{Stability:} GD baseline is most robust across all settings. Adagrad is uniformly stable due to its monotonically decreasing step size, though this causes premature stagnation on ill-conditioned problems. Polyak is \emph{unreliable} when $f^{\star}$ is misspecified, exhibiting large oscillations. Heavy Ball can also exhibit instability if $\beta$ and $\alpha$ are not jointly tuned.
\item \textbf{Comparison with Q2 methods:} Heavy Ball (Q1) and Nesterov (Q2) both use momentum but differ in how the gradient is evaluated: Heavy Ball evaluates at $x_k$, while Nesterov evaluates at the lookahead point $x_k + \beta_k z_k$. This single change gives Nesterov its theoretical $O(1/k^2)$ guarantee while Heavy Ball has no such guarantee on non-quadratic functions, explaining why Nesterov ($f = 0.479$) significantly outperforms Heavy Ball ($f = 1.40$) on the nonlinear Benchmark~C.
\item \textbf{Practical recommendation:} For problems where $f^{\star}$ is unknown, Adagrad and Heavy Ball offer the best trade-off of stability and performance. RMSprop is preferred for non-stationary objectives. Polyak is best reserved for settings with oracle access to $f^{\star}$ (e.g., some convex optimisation frameworks, or problems with known optimal value from duality arguments).
\end{itemize}

\section[Q2: Momentum and Stochastic Methods]{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$, allowing Nesterov's method to anticipate and correct its trajectory before overshooting. This is equivalent to the update $x_{k+1} = \tilde{x}_k - \alpha \nabla f(\tilde{x}_k)$.

\begin{theorem}[Nesterov Acceleration, 1983]
For an $L$-smooth convex function $f$, Nesterov's method with the adaptive schedule $\beta_k = (k-1)/(k+2)$ achieves:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} = O\!\left(\frac{1}{k^2}\right),
\end{equation}
compared to gradient descent's $O(1/k)$ rate. This improvement from linear to quadratic convergence speed in $1/k$ is optimal for first-order methods on smooth convex functions (Nesterov's lower bound).
\end{theorem}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scaling)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum (via $m_t$, exponential average of gradients) with per-coordinate adaptive scaling (via $v_t$, exponential average of squared gradients), inheriting advantages of both RMSprop and momentum methods. The bias-correction terms $(1-\beta_1^t)$ and $(1-\beta_2^t)$ prevent the early iterates from being biased towards zero, which is critical when $\beta_1, \beta_2 \approx 1$ and $t$ is small.

\textbf{Noise analysis:} The SGD gradient on a mini-batch $\mathcal{B}$ is $\hat{g}_\mathcal{B} = \frac{1}{b}X_\mathcal{B}^T(X_\mathcal{B}\theta - y_\mathcal{B})$. The variance of this estimator is:
\begin{equation}
\mathbb{V}[\hat{g}_\mathcal{B}] = \frac{\sigma^2}{b} \cdot \frac{1}{m} X^T X = \frac{\sigma^2}{b} H.
\end{equation}
With $\sigma = 1, b = 5$: $\mathbb{V} = 0.2H$; with $\sigma = 6, b = 5$: $\mathbb{V} = 7.2H$ (36$\times$ larger). This explains why the convergence stagnates earlier under high noise: the stochastic gradient noise $O(\sigma^2/b)$ dominates the signal $O(\|\nabla J(\theta)\|^2)$ once $\|\nabla J(\theta)\|$ becomes small. The SGD convergence rate near the optimum is governed by the noise-to-signal ratio $\sigma^2/(b\|\nabla J\|^2)$, not just the learning rate or batch size.

Furthermore, with $\sigma = 6$ and fixed $\alpha = 0.06$, the noise-amplified gradients cause the iterate to oscillate around the minimum with radius $O(\sigma\alpha/\sqrt{b})$. Reducing the learning rate $\alpha$ or increasing the batch size $b$ would reduce this oscillation, at the cost of slower initial convergence.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.0039 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton update minimises the second-order surrogate: setting $\frac{d}{dx}[g'(x_0)(x-x_0) + \frac{1}{2}g''(x_0)(x-x_0)^2] = 0$ yields $x_{\text{Newton}} = x_0 - g'(x_0)/g''(x_0)$.

For $g(x) = x^4$: $g'(0.25) = 4(0.25)^3 = 0.0625$, $g''(0.25) = 12(0.25)^2 = 0.75$, so
\begin{equation}
x_{\text{Newton}} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\end{equation}
This is a better estimate of the true minimum $x^{\star} = 0$ than a gradient step would provide, illustrating why second-order information accelerates convergence.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised Hessian (ensures PD)}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The damping term $\lambda = 10^{-8}$ regularises the Hessian to ensure positive-definiteness, guarding against near-singularity. The step size $\alpha < 1$ (``damped Newton'') is used on Benchmarks~B and~C to prevent overshooting in regions where the quadratic surrogate is a poor global approximation.

\textbf{Connection to the linear regression case:} For Benchmark~A, Newton's method reduces to the \emph{Gauss-Newton method}: since $J(\theta) = \frac{1}{2m}\|X\theta - y\|^2$, the Hessian is $H = \frac{1}{m}X^T X$ (constant), and the Newton update is:
\begin{equation}
\theta_{\text{new}} = \theta - H^{-1}\nabla J(\theta) = \theta - \left(\frac{X^T X}{m}\right)^{-1}\frac{X^T(X\theta - y)}{m} = (X^T X)^{-1} X^T y = \theta^{\star}_{\text{OLS}}.
\end{equation}
The last equality shows that a \emph{single} Newton step from \emph{any} starting point gives the exact OLS solution $\theta^{\star}_{\text{OLS}}$. This is a fundamental property of quadratic objectives: Newton's method is globally convergent in one step.

\textbf{Convergence theory:} Newton's method has two phases:
\begin{itemize}[nosep]
  \item \emph{Damped phase:} When far from the optimum, each step makes progress at a linear rate.
  \item \emph{Quadratic convergence phase:} Near the optimum $x^{\star}$, for $f$ with $M$-Lipschitz Hessian and strong convexity parameter $\mu$:
  \begin{equation}
  \|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2.
  \end{equation}
  This means the number of correct decimal digits \emph{doubles} at each iteration.
\end{itemize}
The iteration complexity to reach $\varepsilon$-accuracy is $O(\log\log(1/\varepsilon))$ for Newton versus $O(\kappa \log(1/\varepsilon))$ for GD. The $\kappa$-fold advantage of Newton is most pronounced on ill-conditioned problems (large $\kappa$).

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
The approximation error decomposes into two competing terms:
\begin{equation}
\left|\frac{\partial f}{\partial x_i}(x) - \frac{f(x + \delta e_i) - f(x)}{\delta}\right| \leq \underbrace{\frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(x)\right|}_{\text{truncation error}} + \underbrace{\frac{2\varepsilon_{\text{mach}}}{\delta}}_{\text{rounding error}}.
\end{equation}
Minimising over $\delta$ gives optimal $\delta^{\star} \sim \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ for 64-bit arithmetic. In practice, $\delta = 0.05$ is a safe choice for smooth functions with bounded second derivatives (truncation-dominated regime), while $\delta = 0.8$ introduces a truncation error of order $0.8 \times \frac{1}{2}|f''| \approx 0.4 |f''|$, which significantly corrupts the gradient direction on curved functions.

Cost: $d + 1 = 3$ function evaluations per iteration (for $d = 2$), compared to 1 for exact gradient computation. The ratio grows linearly with $d$, making FD costly in high dimensions.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates a directional derivative along a random unit vector $u_k$. Only \textbf{two} function evaluations are required per iteration regardless of dimension, giving it a dimension-independent cost per step. However, it sacrifices gradient accuracy: the expected value $\mathbb{E}[\hat{g}_k] = \frac{1}{d}\nabla f(x_k)$ is a scaled gradient, meaning the effective step size is $\alpha/d$ per coordinate on average. For $d = 2$, the overhead is only $2\times$, but for large $d$ this method becomes prohibitively slow per dimension.

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it using four operations:
\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r = 1.0$): move the worst vertex through the centroid of the remaining vertices.
  \item \textbf{Expansion} ($\gamma = 2.0$): if reflection improved substantially, try a further step.
  \item \textbf{Contraction} ($\rho = 0.5$): if reflection worsened, contract towards the centroid.
  \item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all vertices towards the best vertex.
\end{enumerate}
No gradient information is required. The simplex geometry adapts to the local landscape, effectively scaling to the local curvature.

Settings: initial simplex step $= 0.35$, iterations $= 160$.

\textbf{Curse of dimensionality:} Grid search with $N$ points per dimension requires $N^d$ total evaluations. For $N=55$, $d=2$: 3025 evaluations; for $d=10$: $55^{10} \approx 2.5 \times 10^{17}$ evaluations --- infeasible. \textbf{Oracle complexity comparison:} For a given optimality $\varepsilon$ target:
\begin{itemize}[nosep]
  \item Exact GD: $O(\kappa \log(1/\varepsilon))$ gradient evaluations. Each gradient evaluation costs $O(d)$ derivative evaluations.
  \item FD GD ($\delta$ optimal): Same iteration count as GD, but each iteration requires $d+1$ function evaluations ($d\times$ more expensive than exact GD in function evaluation terms).

\end{itemize}

\subsection{Problem Setup and KKT Analysis}

The constrained problem is:
\begin{equation}
\min_{x \in \mathbb{R}^2} f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad \text{subject to} \quad x_1 \geq 0.5,
\end{equation}
starting from the infeasible initial point $x_0 = (0.2, 4.0)$.

\textbf{KKT Analysis:} The KKT conditions for this inequality-constrained problem are:
\begin{align}
\nabla f(x^{\star}) - \mu^{\star}\, e_1 &= \mathbf{0}, \quad \mu^{\star} \geq 0, \quad \mu^{\star}(x_1^{\star} - 0.5) = 0,
\end{align}
where $\mu^{\star}$ is the Lagrange multiplier and $e_1 = (1,0)^T$. Two cases arise:
\begin{enumerate}[nosep]
  \item \textbf{Interior optimum} ($x_1^{\star} > 0.5$, constraint inactive): $\mu^{\star} = 0$ and $\nabla f(x^{\star}) = \mathbf{0}$. The constrained optimum coincides with the unconstrained optimum.
  \item \textbf{Boundary optimum} ($x_1^{\star} = 0.5$, constraint active): $\mu^{\star} = \frac{\partial f}{\partial x_1}(x^{\star}) > 0$, reflecting the cost of enforcing the constraint.
\end{enumerate}

\textbf{Key observation:} The unconstrained minimum of Benchmark~B is at $x^{\star} \approx (0.582, 2.0)$ with $x_1^{\star} \approx 0.582 > 0.5$. Since $x^{\star}$ lies in the \emph{interior} of the feasible set $\{x_1 \geq 0.5\}$, the constraint is \emph{inactive} at the optimum and Case~1 applies. The constrained minimum is identical to the unconstrained minimum, and no feasibility cost is incurred. All methods that enforce the constraint should converge to the same point $(0.582, 2.0)$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, -x_1 + 0.5)$ \Comment{Penalised objective}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Penalty method theory:} As $\lambda \to \infty$, the penalty solution converges to the constrained optimum: $x^{\star}(\lambda) \to x^{\star}$ and $\mu^{\star}(\lambda) \to \mu^{\star}$. However, large $\lambda$ stiffens the Hessian of $F_\lambda$ near the boundary (adding eigenvalue $\lambda$ in the $x_1$ direction), requiring smaller step sizes for stability.

\textbf{Projected GD} enforces feasibility immediately at every iteration as a hard constraint. The cost is requiring a projection oracle: for $x_1 \geq 0.5$, projection is trivial ($\max$ operation, $O(1)$), but for complex convex sets (e.g., semidefinite constraints, nuclear norm balls), projection can require $O(d^3)$ operations. For our problem, Projected GD achieves the best of both worlds: zero violation from iteration 1 AND convergence to the true optimum (since the constraint is inactive there).

\textbf{Penalty methods} only require a modified gradient, making implementation simpler. Large $\lambda$ asymptotically recovers the constrained solution, but stiffens the Hessian: the penalised Hessian near the boundary has eigenvalue $\lambda$ in the $x_1$ direction (from the penalty term), increasing the effective condition number to $\kappa_\lambda = \max(L, \lambda)/\mu \approx \lambda/\mu$ for large $\lambda$. This requires step sizes $\alpha \leq 2/(\lambda + L)$, explaining why $\lambda = 4.5$ requires the reduced step $\alpha = 0.03$ (vs $\alpha = 0.05$ for $\lambda \leq 1.8$).

\textbf{Comparison of feasibility speed:} From the violation plot (Figure~\ref{fig:q5_violation}), the feasibility speed ordering is: Projected GD ($\leq 1$ iteration) $\gg$ Penalty $\lambda=4.5$ ($\approx 10$ iters) $>$ Penalty $\lambda=1.8$ ($\approx 30$ iters) $>$ Penalty $\lambda=0.15$ (never fully feasible). This ordering is consistent with the theory: the penalty reduces violation at rate $O(1/\lambda)$ per iteration, so higher $\lambda$ enforces feasibility faster (at the cost of a stiffer optimisation).

\textbf{The augmented Lagrangian method} (Hestenes-Powell multiplier method, not implemented here) avoids the stiffness issue by maintaining a Lagrange multiplier estimate $\mu_k$ that is updated alongside the iterates:
\begin{equation}
F_{\text{AL}}(x;\mu,\lambda) = f(x) + \mu \max(0, -x_1+0.5) + \frac{\lambda}{2}\max(0,-x_1+0.5)^2.
\end{equation}
With multiplier updates $\mu_{k+1} \leftarrow \mu_k + \lambda\max(0,-x_1^{(k)}+0.5)$, the method achieves exact feasibility at a \emph{fixed} (non-diverging) $\lambda$, avoiding the ill-conditioning of the pure penalty approach.

\section[Q6: Frank--Wolfe Algorithm]{Question 6: Linear Programmes and Frank--Wolfe Algorithm \hfill [30 marks]}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = (1, 2)^T$, the minimum over $\mathcal{X}$ is obtained by minimising each coordinate independently. Since both components of $a$ are positive, the minimum is at the lower bounds:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
\textbf{Fundamental theorem of linear programming:} The optimum of a linear objective over a bounded polytope (here, a rectangle) is always attained at a \emph{vertex}. This is because the objective $a^T x$ is linear and thus has no interior critical points; gradient descent applied to a linear objective always moves towards a boundary, and the linear constraints concentrate the minimum at an extreme point.

The gradient of $f$ is $a = (1, 2)^T$ pointing upper-right; to minimise, we move in direction $-a$ (lower-left) until hitting the boundary. The minimum occurs at the lower-left vertex $(0.5, -5)$ with $f^{\star} = -9.5$.}

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (feasible)}
\EndFor
\end{algorithmic}
\end{algorithm}

The step $x_{k+1} = \beta x_k + (1-\beta)z_k$ is a convex combination of two feasible points ($x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$), hence $x_{k+1} \in \mathcal{X}$ automatically --- \textbf{no projection is ever required}. This is the central advantage over projected gradient descent for problems where projection is expensive.

For box constraints, the LP subproblem decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations per iteration --- much cheaper than projection onto general convex sets.

\textbf{Convergence theory:} For $L$-smooth convex $f$ with $\beta_k = 1 - \gamma_k$ and $\gamma_k = 2/(k+2)$ (decreasing step schedule):
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LC^2}{k+1} = O\!\left(\frac{1}{k}\right),
\end{equation}
where $C = \max_{x,y \in \mathcal{X}} \|x - y\|$ is the diameter of $\mathcal{X}$. The \emph{Frank--Wolfe gap} $g_k = \nabla f(x_k)^T(x_k - z_k) \geq 0$ provides a computable upper bound on the suboptimality and serves as a stopping criterion. With a fixed $\beta$ (as used here, $\beta = 0.90$ or $0.985$), the effective step size $\gamma = 1-\beta$ is constant, leading to less smooth convergence but potentially faster initial progress.

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits a characteristic ``zig-zag'' pattern in the $z_k$ sequence: the LP vertex jumps between extreme points of $\mathcal{X}$ at each iteration, and the convex combination gradually averages these jumps towards the true interior minimum. The $O(1/k)$ convergence rate is optimal for this algorithm class (unlike GD's superior rate on smooth problems), reflecting the fundamental trade-off: FW avoids expensive projections but loses the direct curvature-following ability of gradient steps. The smaller step size $\gamma = 0.015$ ($\beta = 0.985$) converges more smoothly than $\gamma = 0.10$ ($\beta = 0.90$) because it gives the averaging effect more iterations to stabilise.

\textbf{Boundary optimum:} Frank--Wolfe is naturally suited to boundary optima. When the unconstrained minimum lies outside $\mathcal{X}$, the gradient $\nabla f(x_k)$ consistently points towards the infeasible region, causing the LP oracle to select nearby vertices repeatedly. After 140 iterations with $\beta = 0.93$, $x_1$ has converged to $0.5$ (the active constraint boundary) but $x_2 = -0.337$ has not yet reached $0$, reflecting the $O(1/k)$ rate. To estimate the residual: the FW rate gives $f(x_{140}) - f^\star \leq 2LC^2/(140+1)$. Here $L = 2$ (Hessian eigenvalue of $f = x_1^2 + x_2^2$), $C = \text{diam}(\mathcal{X}) = \sqrt{(5-0.5)^2 + (10+5)^2} \approx 15.7$, giving the bound $2\times 2 \times 246 / 141 \approx 7.0$. The observed $f = 0.364$ is much better, indicating the $O(1/k)$ bound is loose for this specific problem. The residual $x_2 = -0.337$ (vs $x_2^\star = 0$) reflects that $z_k$ alternates between $(0.5, -5)$ (when $\nabla f \cdot e_2 > 0$, i.e., $x_2 > 0$) and other vertices, averaging towards but not quite reaching $x_2 = 0$.

Using the adaptive schedule $\gamma_k = 2/(k+2)$ (equivalently $\beta_k = k/(k+2)$) instead of a fixed $\beta$ would improve convergence by providing larger steps early (fast initial progress) and smaller steps later (reduced oscillation), matching the theoretical $O(1/k)$ bound more closely.

\textbf{Frank--Wolfe vs Projected Gradient Descent:} The two approaches for constrained optimisation have complementary strengths:
\begin{itemize}[nosep]
  \item \textbf{Frank--Wolfe}: $O(1/k)$ rate regardless of constraint complexity; each step requires solving an LP (cheap for box constraints); iterates are always feasible; does \emph{not} accelerate to $O(1/k^2)$ in general.
  \item \textbf{Projected GD}: $O(1/k)$ rate with standard GD; can be accelerated to $O(1/k^2)$ with Nesterov momentum (FISTA); requires a projection oracle; projection is cheap for boxes but expensive for general sets.
\end{itemize}
For box constraints (as in this assignment), both approaches are computationally cheap per iteration. Frank--Wolfe has a practical advantage when the LP oracle is very efficient (e.g., for simplex constraints, the LMO is just an $\arg\max$), while Projected GD is preferred when acceleration is needed and projection is cheap.

\textbf{Key cross-cutting insights:}

\begin{enumerate}
\item \textbf{Condition number is the primary discriminator.} On well-conditioned problems (Benchmark~A, $\kappa \approx 1.12$), virtually all methods converge quickly and differences are negligible. On ill-conditioned Rosenbrock ($\kappa \approx 2508$), differences span orders of magnitude: GD achieves $f = 3.51$ in 120 iters while Polyak achieves $f = 0.024$ --- a $146\times$ difference --- from the same starting point. The theoretical GD iteration count to reach $\varepsilon = 0.01$ on Benchmark~C is $\approx \kappa \log(1/\varepsilon) \approx 11,500$ iterations; in contrast, Newton achieves this in $O(\log\log(1/\varepsilon)) \approx 4$ iterations.

\item \textbf{Second-order information is transformative.} Newton's method eliminates the $\kappa$ factor from the iteration count, reducing the number of iterations needed for Benchmark~A from $\sim$20 (GD) to 1 (Newton). The cost is $O(d^3)$ per iteration for computing and factoring the Hessian, which is feasible for $d = 2$ but becomes prohibitive for large $d$.

\item \textbf{Momentum helps most on ill-conditioned problems.} Nesterov's $O(1/k^2)$ acceleration provides the clearest benefit on Benchmark~C, where the high condition number makes $O(1/k)$ convergence extremely slow. The Heavy Ball method also shows significant improvement on Rosenbrock by accumulating momentum along the consistent valley-floor direction.

\item \textbf{Adaptive methods are robust but not optimal.} Adagrad and Adam converge reliably across all settings but do not match the theoretical optimality of Nesterov acceleration for smooth convex problems. Their strength is practical: they work well without careful tuning on non-convex, non-stationary problems typical in deep learning.

\item \textbf{Constraint handling trade-offs.} Projected GD offers the cleanest guarantee (hard feasibility at every step) at the cost of requiring a projection oracle. Penalty methods are simpler to implement but require $\lambda$ tuning; as $\lambda$ increases, the problem becomes more ill-conditioned. Frank--Wolfe avoids both projections and ill-conditioning from large $\lambda$, but its $O(1/k)$ rate cannot be accelerated to $O(1/k^2)$ in general (unlike projected GD with Nesterov momentum).
\end{enumerate}

\textbf{Practical recommendations:}
\begin{itemize}[nosep]
\item For small-scale, smooth, well-specified problems: use Newton's method.
\item For large-scale smooth convex problems: use Nesterov's method.
\item For deep learning and non-convex problems: use Adam.
\item When $f^\star$ is known: Polyak step with proper $f^\star$ is highly effective.
\item For black-box/derivative-free settings: use FD gradient approximation with $\delta \approx \sqrt{\varepsilon_{\text{mach}}}$ or Nelder--Mead for low-dim problems.
\item For constrained problems with simple sets: projected GD; for structured polytope constraints: Frank--Wolfe.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex}]
def nelder_mead(loss_fn, x0, step, n_iters, alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1])
        f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1])
            f_e = loss_fn(x_e)
            if f_e < f_r: simplex[-1] = x_e; f_vals[-1] = f_e
            else: simplex[-1] = x_r; f_vals[-1] = f_r
        else:
            if f_r < f_vals[-1]:
                x_c = centroid + rho * (x_r - centroid)
                f_c = loss_fn(x_c)
                if f_c <= f_r: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
            else:
                x_c = centroid + rho * (simplex[-1] - centroid)
                f_c = loss_fn(x_c)
                if f_c < f_vals[-1]: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
