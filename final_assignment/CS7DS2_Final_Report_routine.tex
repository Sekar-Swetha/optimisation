\begin{abstract}
This report presents a comprehensive comparative study of first- and second-order optimisation algorithms applied to three benchmark problems spanning a range of curvature conditions: a linear regression quadratic loss (condition number $\kappa \approx 1$), a toy neural network loss with sinusoidal perturbation ($\kappa \approx 7$), and the Rosenbrock function ($\kappa \approx 2500$).  Six classes of method are investigated and theoretically grounded: (1)~adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), (2)~momentum-based methods (Nesterov acceleration, Adam) and mini-batch stochastic gradient descent, (3)~Newton's method exploiting second-order curvature for superlinear convergence, (4)~derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead, grid search), (5)~constrained optimisation via projected gradient descent and penalty methods, analysed through KKT conditions, and (6)~the Frank--Wolfe conditional-gradient algorithm for linearly-constrained problems with convergence characterised by the Frank--Wolfe gap.  Key findings include: the Polyak step size diverges when $f^{\star}$ is misspecified but achieves the best convergence on Rosenbrock when $f^{\star} = 0$ is exact; Nesterov momentum achieves the theoretically optimal $O(1/k^2)$ rate and outperforms Adam on the ill-conditioned Rosenbrock valley; Newton's method solves the quadratic benchmark in a single iteration and reaches the Rosenbrock vicinity in 20 steps versus 80 for gradient descent; finite-difference accuracy depends critically on step size $\delta$, with the optimal $\delta \approx \sqrt{\varepsilon_{\mathrm{mach}}}$; KKT analysis confirms the constrained and unconstrained Benchmark B minima coincide; and Frank--Wolfe converges at $O(1/k)$ with a characteristic zig-zag pattern driven by the linear minimisation oracle.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.  The three benchmark functions are deliberately chosen to cover the spectrum from well-conditioned quadratics to highly ill-conditioned non-convex landscapes, allowing a systematic evaluation of how each algorithm's performance degrades (or improves) with problem difficulty.

\begin{definition}[Smoothness and Strong Convexity]
A function $f:\mathbb{R}^n \to \mathbb{R}$ is \emph{$L$-smooth} if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x - y\|$ for all $x, y$, and \emph{$\mu$-strongly convex} if $f(y) \geq f(x) + \nabla f(x)^T(y-x) + \frac{\mu}{2}\|y-x\|^2$.  The condition number is $\kappa = L/\mu \geq 1$.
\end{definition}

\subsection{Benchmark A: Linear Regression Quadratic Loss}
\label{sec:bench_a}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \overset{\mathrm{i.i.d.}}{\sim} \mathcal{N}(0, I_2), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \overset{\mathrm{i.i.d.}}{\sim} \mathcal{N}(0, 1).
\end{equation}
The loss is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2}\theta^T H \theta - b^T\theta + c,
\end{equation}
where $H = \frac{1}{m}X^T X$, $b = \frac{1}{m}X^T y$, and $c = \frac{1}{2m}\|y\|^2$.  The gradient and Hessian are:
\begin{equation}
\nabla J(\theta) = H\theta - b = \frac{1}{m}X^T(X\theta - y), \qquad \nabla^2 J(\theta) = H = \frac{1}{m}X^T X.
\end{equation}
Since $X \sim \mathcal{N}(0, I_{m \times 2})$, by the law of large numbers $H \approx I_2$ for large $m$, giving $L \approx \mu \approx 1$ and $\kappa \approx 1$.  The noise floor prevents exact minimisation: $J^{\star} = \frac{1}{2m}\|y - X\hat\theta\|^2 \approx 0.4826$.  Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\label{sec:bench_b}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient and Hessian are:
\begin{equation}
\nabla f = \begin{bmatrix} 2(x_1 - 1) + \cos(x_1) \\ 10(x_2 - 2) \end{bmatrix}, \qquad
\nabla^2 f = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
Since $\sin(x_1) \in [-1, 1]$, the Hessian diagonal entries satisfy $2-\sin(x_1) \in [1, 3]$ and the constant 10.  Thus $L \approx 10$ and $\mu \approx 1$, giving $\kappa \approx 10$.  The minimum is the solution to $2(x_1 - 1) + \cos(x_1) = 0$, which yields $x_1^{\star} \approx 0.582$ (verified by the converged Newton and Adam iterates), giving $f^{\star} \approx 0.7244$.  Starting point: $x_0 = [-1, 4]^T$.

\begin{remark}
The contour plots mark the approximate minimum at $(0.61, 2.0)$; the numerically precise minimum is $(0.582, 2.0)$ as confirmed by all converging algorithms.
\end{remark}

\subsection{Benchmark C: Rosenbrock Function}
\label{sec:bench_c}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$.  The Hessian is:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1,1)$: $H^{\star} = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}$.  The eigenvalues are $\lambda_{1,2} = 501 \pm \sqrt{501^2 - (802 \cdot 200 - 400^2)} \approx 1001.6$ and $0.399$, giving $\kappa \approx 2512$.  This extreme ill-conditioning, combined with the narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions.  Starting point: $x_0 = [-1, 1]^T$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

For an $L$-smooth, $\mu$-strongly convex function, constant step-size gradient descent with $\alpha = 1/L$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k \bigl(f(x_0) - f^{\star}\bigr) = \left(1 - \frac{1}{\kappa}\right)^k \bigl(f(x_0) - f^{\star}\bigr).
\label{eq:gd_rate}
\end{equation}
For $\kappa \approx 2512$ (Rosenbrock), achieving $\varepsilon$-accuracy requires $k = O(\kappa \log(1/\varepsilon))$ iterations, explaining why constant step-size GD barely progresses on Benchmark C.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak Step Size Optimality]
For a convex function with known minimum $f^{\star}$ and $\varepsilon = 0$, the Polyak step size minimises the distance to an optimal solution:
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - \frac{(f(x_k) - f^{\star})^2}{\|g_k\|^2}.
\end{equation}
If $f^{\star}$ is known exactly, this guarantees $f(x_k) \to f^{\star}$ without requiring knowledge of $L$ or $\mu$.
\end{theorem}

\begin{remark}
The Polyak formula $\alpha_k = (f(x_k) - f^{\star}) / (\|g_k\|^2 + \varepsilon)$ is ill-posed when $f^{\star}$ is overestimated (set to 0 while the true minimum is positive).  Near convergence, $f(x_k) \to f^{\star}_{\mathrm{true}} > 0$ while $\|g_k\| \to 0$, so $\alpha_k \to \infty$, inducing divergence.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The effective per-coordinate step $(\alpha_0 / \sqrt{G_k^{(i)} + \varepsilon})$ decreases monotonically, effectively normalising each coordinate by the root-mean-square of past gradients.  This makes Adagrad especially effective for sparse or ill-scaled problems: coordinates with historically small gradients receive larger effective steps.  However, the monotone decay means Adagrad's step eventually vanishes, which can cause premature stagnation on non-sparse problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's vanishing step by replacing the cumulative sum with an exponential moving average with decay $\beta$.  The effective step size stabilises around $\alpha_0 / \sqrt{(1-\beta^k)/(1-\beta) \cdot \overline{g^2}}$ rather than decaying monotonically.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{proposition}[Heavy Ball Optimal Rate for Quadratics]
For the quadratic $f(x) = \frac{1}{2}x^T H x - b^T x$ with $H \succ 0$ having eigenvalues in $[\mu, L]$, the Heavy Ball method with optimal parameters $\alpha^* = 4/(\sqrt{L} + \sqrt{\mu})^2$ and $\beta^* = ((\sqrt{\kappa} - 1)/(\sqrt{\kappa} + 1))^2$ achieves:
\begin{equation}
\|x_k - x^{\star}\| \leq C \left(\frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1}\right)^k \|x_0 - x^{\star}\|,
\end{equation}
which improves on GD's rate $(({\kappa} - 1)/({\kappa} + 1))^k$ by working with $\sqrt{\kappa}$ instead of $\kappa$.
\end{proposition}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Condition number dependence:} On Benchmark A ($\kappa \approx 1$) all methods converge rapidly and similarly; on Benchmark C ($\kappa \approx 2512$) differences between methods are stark, with only Polyak (correct $f^{\star}$) and Heavy Ball making significant progress.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging, but this comes at the cost of requiring careful tuning of $\beta$ and $\alpha_0$.
\item \textbf{Stability:} GD baseline is most stable but slowest.  Polyak is \emph{unstable} when $f^{\star}$ is misspecified.  Adagrad is uniformly stable due to its monotonically decreasing step size, though this causes premature convergence on non-sparse, stationary problems.
\item \textbf{Per-coordinate adaptation:} Both Adagrad and RMSprop benefit problems where different coordinates have vastly different curvatures (Benchmark B's $\approx 10:1$ ratio).  However, for Benchmark C the curvature varies \emph{along} the trajectory rather than \emph{between} coordinates, limiting coordinate-wise adaptation.
\item \textbf{Practical recommendation:} RMSprop is preferable to Adagrad in non-stationary settings (deep learning) due to its non-vanishing effective rate.  Polyak is a powerful oracle-based method but requires exact knowledge of $f^{\star}$, limiting its practical applicability.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Optimal Rate]
For a convex $L$-smooth function with $\alpha = 1/L$ and $\beta_k = (k-1)/(k+2)$:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\end{equation}
This $O(1/k^2)$ rate is \emph{optimal} for first-order methods on the class of convex $L$-smooth functions, improving on GD's $O(1/k)$.  For $\mu$-strongly convex functions, the rate improves to $O((1 - 1/\sqrt{\kappa})^k)$.
\end{theorem}

The key distinction from Heavy Ball (Algorithm~\ref{alg:heavyball}) is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$.  This lookahead acts as a correction step, allowing Nesterov to anticipate and counteract overshoot before it occurs.  The momentum schedule $\beta_k = (k-1)/(k+2) \to 1$ builds gradually, avoiding instability in early iterations.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (variance) estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction for $m_t$}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias correction for $v_t$}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines two ideas: (i) momentum via the first moment $m_t$ (an exponential moving average of gradients, similar to Heavy Ball), and (ii) per-coordinate adaptive scaling via $v_t$ (the second moment, similar to RMSprop).  The bias-correction terms $(1 - \beta_1^t)$ and $(1 - \beta_2^t)$ are essential at early iterations where $m_0 = v_0 = 0$ would otherwise under-estimate the moments.  The effective step for coordinate $i$ is approximately $\alpha \cdot \mathrm{sign}(g_t^{(i)}) \cdot (\text{SNR}_t^{(i)})$ for large $t$, where $\mathrm{SNR}$ is the signal-to-noise ratio of past gradients.

\subsection{Mini-Batch Stochastic Gradient Descent}

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}_j$ of size $b$}
        \State $g_j \gets \frac{1}{b} X_{\mathcal{B}_j}^T(X_{\mathcal{B}_j}\theta - y_{\mathcal{B}_j})$
        \State $\theta \gets \theta - \alpha\, g_j$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $g_j$ is an unbiased estimator of the full gradient: $\mathbb{E}[g_j] = \nabla J(\theta)$.  Its variance is:
\begin{equation}
\mathrm{Var}[g_j] = \frac{1}{b} \cdot \mathrm{Var}[\text{per-sample gradient}] = \frac{\sigma_g^2}{b},
\end{equation}
so larger batches reduce gradient noise by factor $\sqrt{b}$, at the cost of $b$ times more computation per update.  With $m = 1000$ samples and learning rate $\alpha = 0.06$, batches of $b = 5$ give $200$ updates per epoch while $b = 40$ gives $25$ updates per epoch.

\subsection{SGD with Increased Noise}

The label noise is scaled to $\sigma = 6$ (versus $\sigma = 1$ baseline): $y_{\mathrm{noisy}} = X\theta^{\star} + 6\varepsilon$.

\begin{proposition}[Noise Floor Scaling]
For the linear regression loss with noise $\varepsilon \sim \mathcal{N}(0, \sigma^2 I)$, the irreducible loss floor is:
\begin{equation}
J^{\star} = \frac{\sigma^2}{2m}\mathbb{E}\left[\|y - X\hat\theta\|^2\right] \approx \frac{\sigma^2}{2} \cdot \frac{m - d}{m} \approx \frac{\sigma^2}{2},
\end{equation}
so doubling $\sigma$ raises the noise floor by $4\times$, and the $\sigma = 6$ case gives $J^{\star} \approx 18$.
\end{proposition}

\subsection{Local Approximations of $g(x) = x^4$}

At $x_0 = 0.25$: $g(0.25) = 0.00391$, $g'(0.25) = 4(0.25)^3 = 0.0625$, $g''(0.25) = 12(0.25)^2 = 0.75$.

\textbf{First-order (Taylor) approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.00391 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.00391 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton update minimises $\hat{g}_2$, yielding:
\begin{equation}
x_1 = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\end{equation}
After one Newton step, $g(0.1667) \approx 7.7 \times 10^{-4}$, a $5\times$ improvement over the initial value despite $x^{\star} = 0$ being the true minimum.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised (positive-definite) Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solved via $H_k p_k = g_k$)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Quadratic Convergence of Newton's Method]
Let $f$ be twice continuously differentiable with $\nabla^2 f$ Lipschitz continuous with constant $M$.  If $\alpha = 1$ (full Newton step) and $x_0$ is sufficiently close to $x^{\star}$, then:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2,
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^{\star}))$.  This \emph{quadratic} convergence means the number of correct digits roughly \emph{doubles} each iteration, in contrast to GD's linear convergence.
\end{theorem}

The Tikhonov regularisation $H_k + \lambda I$ with $\lambda = 10^{-8}$ ensures positive definiteness, preventing numerical singularity without significantly altering the Newton direction when $H_k$ is well-conditioned.  The damped step $\alpha < 1$ extends Newton's convergence region beyond the local quadratic basin of attraction.

\textbf{Cost per iteration:} Newton requires computing the Hessian ($O(n^2)$ storage, $O(n^2)$ computation for diagonal $H$) and solving a linear system ($O(n^3)$ in general, $O(n)$ for diagonal $H$).  For the 2D problems here, this overhead is negligible.  For large $n$, quasi-Newton methods (BFGS, L-BFGS) approximate $H^{-1}$ to reduce cost.

Settings: GD uses 80 iterations; Newton uses only 20 iterations with $\lambda = 10^{-8}$.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
By Taylor expansion: $f(x + \delta e_i) = f(x) + \delta \frac{\partial f}{\partial x_i} + \frac{\delta^2}{2}\frac{\partial^2 f}{\partial x_i^2} + O(\delta^3)$.  Therefore:
\begin{equation}
\text{error} = \underbrace{\frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}\right|}_{\text{truncation error, } O(\delta)} + \underbrace{\frac{2\varepsilon_{\mathrm{mach}}}{\delta}}_{\text{rounding error, } O(\varepsilon_{\mathrm{mach}}/\delta)}.
\end{equation}
Minimising total error over $\delta$ gives the optimal step size $\delta^* = \sqrt{2\varepsilon_{\mathrm{mach}} / |f''|} \approx \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$ for double precision.  The assignment tests $\delta = 0.05$ (close to optimal for smooth $f$) and $\delta = 0.8$ (truncation error $O(0.8) = O(1)$, far from optimal).

This is a zeroth-order method requiring $d + 1$ function evaluations per gradient estimate ($d = 2$ here), but the cost scales linearly with dimension, making it less attractive than analytical gradients in high dimensions.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_n)$, normalise: $u_k \gets u_k / \|u_k\|$ (uniform on sphere)
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Stochastic directional derivative}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The estimator $\hat{g}_k$ satisfies $\mathbb{E}[\hat{g}_k] = \nabla f_\delta(x_k)$ where $f_\delta(x) = \mathbb{E}_{u}[f(x + \delta u)]$ is the Gaussian-smoothed function, which converges to $\nabla f(x)$ as $\delta \to 0$.  Crucially, only \emph{two} function evaluations are required per iteration regardless of dimension $d$, making it dimension-independent in cost per step but requiring $O(d)$ times more iterations than gradient methods for the same accuracy.

\subsection{Part B: Derivative-Free Methods on Benchmark C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it via four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): move the worst vertex through the centroid of the best $n$ vertices.
\item \textbf{Expansion} ($\gamma = 2$): if reflection improved on the best, try a further expansion step.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection worsened, contract towards centroid (inside or outside).
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all vertices towards the best vertex.
\end{enumerate}

Settings: initial simplex step $= 0.35$, 160 iterations.  No gradient information is used; the simplex geometry adapts to local landscape topology.

\subsection{Problem Setup and KKT Analysis}

Minimise Benchmark B subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$:
\begin{equation}
\min_{x \in \mathbb{R}^2}\ f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad \text{subject to} \quad g(x) = 0.5 - x_1 \leq 0.
\end{equation}

\begin{theorem}[KKT Necessary Conditions]
If $x^{\star}$ is a local minimiser and a constraint qualification holds (e.g.\ LICQ), then there exists a multiplier $\mu^{\star} \geq 0$ such that:
\begin{align}
\nabla f(x^{\star}) + \mu^{\star} \nabla g(x^{\star}) &= 0 \quad \text{(stationarity)}, \label{eq:kkt_stat} \\
g(x^{\star}) &\leq 0 \quad \text{(primal feasibility)}, \\
\mu^{\star} &\geq 0 \quad \text{(dual feasibility)}, \\
\mu^{\star} g(x^{\star}) &= 0 \quad \text{(complementary slackness)}.
\end{align}
\end{theorem}

\paragraph{Applying KKT to this problem.}
$\nabla g = (-1, 0)^T$.  Stationarity (Eq.~\ref{eq:kkt_stat}) gives:
\begin{equation}
\begin{bmatrix} 2(x_1^{\star} - 1) + \cos(x_1^{\star}) \\ 10(x_2^{\star} - 2) \end{bmatrix} + \mu^{\star} \begin{bmatrix} -1 \\ 0 \end{bmatrix} = \mathbf{0}.
\end{equation}
From the second component: $x_2^{\star} = 2$.  From the first: $\mu^{\star} = 2(x_1^{\star} - 1) + \cos(x_1^{\star})$.

\textbf{Case 1: Constraint inactive ($x_1^{\star} > 0.5$, so $\mu^{\star} = 0$ by complementary slackness).}
Then $2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$, giving $x_1^{\star} \approx 0.582 > 0.5$ \checkmark.

\textbf{Case 2: Constraint active ($x_1^{\star} = 0.5$, so $\mu^{\star}$ unrestricted).}
Then $\mu^{\star} = 2(0.5 - 1) + \cos(0.5) = -1 + 0.8776 = -0.1224 < 0$, violating dual feasibility \texttimes.

\begin{remark}
The KKT analysis confirms that the constrained optimum is identical to the unconstrained optimum at $(0.582, 2.0)$: the constraint $x_1 \geq 0.5$ is \emph{inactive} at the solution.  The interest of this problem lies entirely in the \emph{path} from the infeasible starting point $(0.2, 4.0)$ to the feasible minimum, not in the solution value itself.
\end{remark}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$, $x_2$ unchanged}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method (Linear Penalty)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\State Define penalised objective: $F_{\lambda}(x) = f(x) + \lambda \max(0,\, 0.5 - x_1)$
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute $\nabla F_{\lambda}(x_k)$: equals $\nabla f(x_k) - \lambda e_1$ if $x_1 < 0.5$, else $\nabla f(x_k)$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\paragraph{Penalty parameter selection.}  As $\lambda \to \infty$, the penalty method's solution converges to the constrained solution.  However, large $\lambda$ increases the effective Lipschitz constant of $F_\lambda$ from $L_f$ to $L_f + \lambda$, requiring $\alpha < 2/(L_f + \lambda)$ for convergence.  This explains why $\alpha = 0.03$ is used for $\lambda = 4.5$ versus $\alpha = 0.05$ for $\lambda \leq 1.8$.

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$, unconstrained GD ($\alpha = 0.07$).  All run for 100 iterations.

\textbf{Projected GD} is the gold standard when the projection oracle is cheap.  For the half-space constraint $x_1 \geq 0.5$, projection is $O(1)$ (a single $\max$ operation). Increasing $\lambda$ improves feasibility but requires reducing the step size to maintain stability, introducing a $O(1/\lambda)$ convergence slowdown.  The \emph{augmented Lagrangian} method addresses this by combining the penalty with a Lagrange multiplier update, achieving exact feasibility in the limit while avoiding the conditioning deterioration of pure penalty methods.

\subsection{Feasible Set and Linear Programme}

The feasible set is the box $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ with $\mathrm{diam}(\mathcal{X}) = \|(4.5, 15)^T\| \approx 15.66$.

\textbf{Linear Programme:} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$ with $a = [1, 2]^T$.

Since $f$ is linear, it is minimised at a vertex of the polytope $\mathcal{X}$.  For the box, each coordinate is minimised independently:
\begin{equation}
x_i^{\star} = \begin{cases} l_i & \text{if } a_i > 0 \\ u_i & \text{if } a_i < 0 \end{cases}, \quad \text{so } x^{\star} = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This is the \emph{fundamental theorem of linear programming}: the optimum of a linear objective over a bounded polyhedral feasible set is attained at a vertex (extreme point).  Interior-point methods reach this vertex from the interior via the central path.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient) with Fixed Mixing Parameter}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0, 1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: $x_{k+1} \in \mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the box constraint, the LMO decomposes per coordinate in $O(d)$:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}.
\end{equation}

\begin{remark}
The implementation uses a \emph{fixed} mixing parameter $\beta$, i.e.\ step size $\gamma = 1 - \beta$.  The standard Frank--Wolfe uses a diminishing schedule $\gamma_k = 2/(k+2)$ (or exact line search), achieving $O(1/k)$ convergence with this schedule.  A fixed $\gamma$ can also converge if $\gamma$ is small enough, but with a different rate profile.  The key invariant is that $x_k \in \mathcal{X}$ for all $k$, since convex combinations of feasible points are feasible.
\end{remark}

\begin{definition}[Frank--Wolfe Gap]
The \emph{Frank--Wolfe gap} at $x_k$ is:
\begin{equation}
\mathcal{G}(x_k) = \max_{z \in \mathcal{X}}\, \nabla f(x_k)^T (x_k - z) = \nabla f(x_k)^T (x_k - z_k).
\end{equation}
Since $\mathcal{G}(x_k) \geq 0$ for all $k$ and $\mathcal{G}(x^{\star}) = 0$, the gap provides an upper bound on the suboptimality gap: $f(x_k) - f^{\star} \leq \mathcal{G}(x_k)$.
\end{definition}

\begin{theorem}[Frank--Wolfe Convergence Rate]
For an $L$-smooth convex function with standard step size $\gamma_k = 2/(k+2)$:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2},
\end{equation}
where $D = \mathrm{diam}(\mathcal{X})$.  For $\mathcal{X} = [0.5, 5] \times [-5, 10]$ and Benchmark~B ($L = 10$), this gives $f(x_k) - f^{\star} \leq 2 \cdot 10 \cdot 15.66^2 / (k+2) \approx 4909 / (k+2)$.  The $O(1/k)$ rate is \emph{not improvable} in general for Frank--Wolfe, in contrast to the $O(1/k^2)$ of projected gradient methods.  However, Frank--Wolfe avoids the projection oracle entirely.
\end{theorem}

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits the ``zig-zag'' or ``staircase'' phenomenon, characteristic of its $O(1/k)$ rate.  Each iterate is a convex combination of the current point and an LP vertex, and these vertices lie on the boundary of $\mathcal{X}$, far from the interior optimum.  The algorithm asymptotically averages many vertex directions to produce a trajectory converging towards the interior.  The rate $O(1/k)$ means halving the optimality gap requires doubling the iteration count.

\textbf{Boundary optimum:} Frank--Wolfe is more naturally suited to boundary optima, because the LP vertex selected by the oracle frequently lies near the constrained optimum.  When $\nabla f(x_k)^T$ points towards a corner of $\mathcal{X}$, $z_k$ selects that corner, driving $x_{k+1}$ towards the constrained minimiser.  No projection oracle is needed: feasibility is maintained by construction through the convex combination.  This is the main advantage of Frank--Wolfe over projected gradient methods for problems where projection is expensive (e.g., nuclear norm balls, spectrahedra, flow polytopes).

\textbf{Comparison:} The boundary case's gap $f - f^{\star} = 0.114$ after 140 iterations reflects the $O(1/k)$ rate; the $O(1/k^2)$ projected gradient method would achieve a gap of $O(2 \cdot 10 \cdot 15.66^2 / 140^2) \approx 0.25$ from the same starting point, but without the vertex-tracking advantage at the boundary.

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has systematically evaluated six classes of optimisation methods across three benchmark problems that span condition numbers from $\kappa \approx 1$ (linear regression) through $\kappa \approx 10$ (toy neural network) to $\kappa \approx 2512$ (Rosenbrock).  The key conclusions are organised by method class below.

\subsection*{Adaptive Step-Size Methods (Q1)}

The Polyak step size provides theoretically optimal convergence when $f^{\star}$ is known exactly (Benchmark C: $f = 0.009$ at 120 iterations, the best Q1 result), but diverges catastrophically when $f^{\star}$ is overestimated (Benchmarks A and B with $f^{\star} = 0$).  This binary behaviour limits Polyak's practical applicability.  Adagrad is uniformly stable due to its monotone decaying step, making it a robust choice for sparse problems.  RMSprop improves on Adagrad through exponential forgetting, maintaining non-vanishing effective rates for non-stationary problems.  Heavy Ball provides momentum acceleration on ill-conditioned problems ($f = 1.40$ vs GD's $3.51$ on Rosenbrock) through accumulation in consistent descent directions.

\subsection*{Momentum and Stochastic Methods (Q2)}

Nesterov momentum achieves the theoretically optimal $O(1/k^2)$ convergence rate for smooth convex functions through the lookahead correction, substantially outperforming both GD and Adam on the ill-conditioned Rosenbrock ($f = 0.479$ vs $f = 1.578$ for Adam, $f = 3.388$ for GD at 150 iterations).  Adam's per-coordinate adaptive scaling is most effective when curvature varies between coordinates (anisotropy), but is less effective for the off-diagonal curvature structure of Rosenbrock.  Mini-batch SGD demonstrates a fundamental variance--accuracy trade-off: smaller batches provide more gradient updates per epoch but with higher variance, while larger batches are smoother but slower per epoch.

\subsection*{Newton's Method (Q3)}

Newton's method demonstrates the power of second-order information: it solves the quadratic benchmark in exactly one iteration, and achieves $5\times$ better objective value than GD on Rosenbrock (using $25\%$ as many iterations).  The quadratic convergence rate (error squares each iteration) is empirically visible in the update magnitude plots.  Hessian regularisation ($\lambda = 10^{-8}$) and step damping ($\alpha = 0.22$ for Rosenbrock) are necessary to handle regions where the Hessian varies rapidly or is nearly singular.

\subsection*{Derivative-Free Methods (Q4)}

Finite-difference gradient approximation with small $\delta$ (0.05) closely tracks exact gradient descent, confirming that the $O(\delta)$ truncation error is negligible when $\delta \ll 1/f''$.  Large $\delta$ (0.8) causes substantial gradient bias.  The optimal $\delta \approx \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$ balances truncation and rounding errors.  Nesterov random search demonstrates dimension-independent cost (2 function evaluations per iteration) at the price of very slow convergence due to high gradient estimation variance.  Nelder--Mead adapts its simplex geometry to the landscape without any gradient information, navigating the Rosenbrock valley effectively.  Grid search is viable only for low-dimensional ($d \leq 3$) problems due to exponential scaling.

\subsection*{Constrained Optimisation (Q5)}

KKT analysis confirms the constraint $x_1 \geq 0.5$ is inactive at the optimum $(0.582, 2.0)$ (dual variable $\mu^{\star} = 0$), so the constrained solution equals the unconstrained solution.  Projected gradient descent enforces feasibility exactly from iteration 1 via the cheap projection $x_1 \leftarrow \max(0.5, x_1)$, making it the preferred approach when the projection is computationally inexpensive.  Penalty methods approach the constrained solution asymptotically as $\lambda \to \infty$, but large $\lambda$ degrades the Lipschitz constant of the penalised objective, requiring smaller step sizes.

\subsection*{Frank--Wolfe (Q6)}

The Frank--Wolfe algorithm avoids projection entirely by solving a linear minimisation oracle at each step, maintaining feasibility through convex combinations.  Its $O(1/k)$ convergence rate is lower than projected gradient's $O(1/k^2)$, but FW is advantageous when the projection is expensive (e.g., semidefinite or nuclear norm constraints).  The algorithm exhibits a characteristic zig-zag pattern for interior optima, arising from the binary nature of the LP vertex selection.  For boundary optima, the LP oracle naturally steers iterates towards the constrained minimiser.  The Frank--Wolfe gap $\mathcal{G}(x_k) = \nabla f(x_k)^T(x_k - z_k)$ provides a computable, certificate of suboptimality at every iteration.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float); G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x); G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float); v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float); z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x); z = beta * z + alpha * g; x = x - z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float); z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g; x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Hessian Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                      for i in range(len(g))])   # LMO for box
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z            # fixed mixing
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
