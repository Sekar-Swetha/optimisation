\begin{abstract}
This report presents a comprehensive comparative study of six families of optimisation algorithms applied to three standard benchmark problems of increasing difficulty: a linear regression quadratic loss (Benchmark A), a toy neural network loss with a sinusoidal perturbation (Benchmark B), and the Rosenbrock function (Benchmark C). The six families investigated are: (i)~adaptive step-size methods (Polyak, Adagrad, RMSprop, Polyak/Heavy-Ball momentum); (ii)~accelerated and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD); (iii)~second-order Newton's method with local quadratic approximations; (iv)~derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead simplex, grid search); (v)~constrained optimisation via projected gradient descent and penalty methods; and (vi)~the Frank--Wolfe (conditional gradient) algorithm for linearly constrained problems. Each method is analysed for convergence speed, trajectory behaviour, stability, and computational trade-offs. Theoretical convergence rates are stated, proved informally where space permits, and empirically verified against log-log plots. The experiments confirm that the structure of the objective function---convexity, curvature, condition number, constraint geometry---is the primary determinant of which algorithm family is most suitable.
\end{abstract}

\section{Notation and Mathematical Preliminaries}
\label{sec:notation}
% ============================================================

Let $f : \mathbb{R}^d \to \mathbb{R}$ be the objective. Throughout, $x^{\star}$ denotes a (global) minimiser with $f^{\star} = f(x^{\star})$; $\nabla f(x) \in \mathbb{R}^d$ is the gradient; $\nabla^2 f(x) \in \mathbb{R}^{d \times d}$ is the Hessian. The symbol $\odot$ denotes element-wise (Hadamard) multiplication. $\|\cdot\|$ is the Euclidean norm. The projection onto a closed convex set $\mathcal{X}$ is $\Pi_{\mathcal{X}}(x) = \arg\min_{z \in \mathcal{X}} \|x - z\|$. The iteration index is $k$; $\rho^k$ denotes linear convergence at rate $\rho \in (0,1)$.

\begin{definition}[Smoothness]
$f$ is $L$-\emph{smooth} if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x-y\|$ for all $x,y$. Equivalently, $f(y) \leq f(x) + \nabla f(x)^T(y-x) + \tfrac{L}{2}\|y-x\|^2$ (descent lemma).
\end{definition}

\begin{definition}[Strong Convexity]
$f$ is $\mu$-\emph{strongly convex} if $f(y) \geq f(x) + \nabla f(x)^T(y-x) + \frac{\mu}{2}\|y-x\|^2$ for all $x,y$. The \emph{condition number} is $\kappa = L/\mu \geq 1$.
\end{definition}

For a $\mu$-strongly convex, $L$-smooth function, gradient descent with step $\alpha = 1/L$ satisfies:
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 \leq \left(1 - \frac{\mu}{L}\right)^k \|x_0 - x^{\star}\|^2 = \rho_{\text{GD}}^k\,\|x_0 - x^{\star}\|^2, \quad \rho_{\text{GD}} = 1 - \frac{1}{\kappa}.
\label{eq:gd_rate}
\end{equation}
This linear (geometric) rate forms the baseline against which accelerated methods are measured.

\paragraph{Derivation of GD's $O(1/k)$ rate for convex $f$.} The \emph{descent lemma} (from $L$-smoothness) states:
\begin{equation}
f(x_{k+1}) \leq f(x_k) + \nabla f(x_k)^T(x_{k+1}-x_k) + \frac{L}{2}\|x_{k+1}-x_k\|^2.
\label{eq:descent_lemma}
\end{equation}
With $x_{k+1} = x_k - \frac{1}{L}\nabla f(x_k)$, the right side simplifies to $f(x_k) - \frac{1}{2L}\|\nabla f(x_k)\|^2$, giving the per-step \emph{sufficient decrease}:
\begin{equation}
f(x_{k+1}) - f(x^{\star}) \leq f(x_k) - f(x^{\star}) - \frac{1}{2L}\|\nabla f(x_k)\|^2.
\label{eq:suff_decrease}
\end{equation}
Convexity then gives $f(x_k) - f(x^{\star}) \leq \nabla f(x_k)^T(x_k - x^{\star}) \leq \|\nabla f(x_k)\|\cdot\|x_k - x^{\star}\| \leq \|\nabla f(x_k)\| R$ where $R = \|x_0 - x^{\star}\|$. Combined with \eqref{eq:suff_decrease} and summing over $k$:
\begin{equation}
\min_{j \leq k} (f(x_j) - f^{\star}) \leq \frac{LR^2}{2k} = O\!\left(\frac{1}{k}\right).
\label{eq:gd_sublinear}
\end{equation}
This $O(1/k)$ rate is achieved for any convex $L$-smooth function; strong convexity improves this to the linear rate \eqref{eq:gd_rate}.

All experiments use \texttt{np.random.seed(42)} for reproducibility.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \overset{\text{iid}}{\sim} \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss and its gradient are:
\begin{equation}
J(\theta) = \frac{1}{2m}\|X\theta - y\|^2, \qquad \nabla J(\theta) = \frac{1}{m}X^T(X\theta - y) = H(\theta - \theta_{\text{OLS}}),
\end{equation}
where $H = X^TX/m$ is the (constant) Hessian and $\theta_{\text{OLS}} = H^{-1}X^Ty/m$ is the ordinary least-squares solution. This is a strongly convex quadratic; $L = \lambda_{\max}(H)$, $\mu = \lambda_{\min}(H)$. For the generated data, $\theta_{\text{OLS}} \approx (3.028, 4.017)$ and the minimum loss is $J^{\star} \approx 0.4826$ (the irreducible noise floor $\approx \frac{1}{2}\mathbb{E}[\varepsilon^2] = 0.5$). Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\label{eq:bench_B}
\end{equation}
Gradient: $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$. Hessian: $H(x) = \text{diag}(2 - \sin(x_1),\ 10)$. The function is \emph{not} quadratic due to the sinusoidal term; however, $|\sin(x_1)| \leq 1$ so the perturbation is bounded and the function remains strongly convex in a large neighbourhood ($\mu \geq 1$ since $H_{11} = 2 - \sin(x_1) \geq 1$, $H_{22} = 10$). The condition number $\kappa \in [10/3, 10]$ (moderate anisotropy). The minimiser satisfies $2(x_1-1) + \cos(x_1) = 0$, giving $x_1^{\star} \approx 0.5824$ and $x_2^{\star} = 2.0$, with $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\label{eq:rosenbrock}
\end{equation}
Global minimum at $(1,1)$ with $f^{\star} = 0$. The Hessian is:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At $x^{\star} = (1,1)$: eigenvalues $\approx 0.5$ and $\approx 801$, giving $\kappa \approx 1600$. The function is not strongly convex globally (it is convex but not strictly convex far from the minimum), and its curved narrow valley $\{x_2 = x_1^2\}$ makes it one of the most challenging standard test functions for first-order methods. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$, a box (bounded polytope with four vertices).

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}

\subsection{Motivation: Limitations of Fixed Step-Size GD}

Standard gradient descent $x_{k+1} = x_k - \alpha \nabla f(x_k)$ requires a global step bound $\alpha \leq 1/L$. For problems with large condition number $\kappa$, this forces an extremely small step that can barely move in the direction of low curvature (as seen on Rosenbrock, where $\alpha = 0.0012$). Adaptive methods address this by automatically adjusting the step based on local gradient information.

\subsubsection{Step-Size Stability Analysis}

The stability condition for GD is $\alpha < 2/L$ (for quadratics; stricter for nonlinear functions). Figure~\ref{fig:stepsize_sensitivity} shows the phase transition from fast convergence to divergence as $\alpha$ increases past the stability threshold.

\begin{figure}[H]
\centering
\includegraphics[width=0.95\textwidth]{figures/step_size_sensitivity.pdf}
\caption{Q1: Step-size sensitivity for GD. Left (Benchmark B): $\alpha \leq 0.12$ converges, $\alpha = 0.14$ diverges. The Lipschitz constant at the starting point $x_0 = (-1,4)$ is $L(x_0) = \|H(x_0)\| = \max(|2 - \sin(-1)|, 10) = 10$; the stability threshold is $\alpha^{\star} = 2/L = 0.2$. However, the nonlinear sinusoidal term reduces the practical threshold to $\approx 0.13$. Right (Rosenbrock): even $\alpha = 0.002$ causes oscillations; $\alpha = 0.003$ diverges. The Rosenbrock Hessian at $x_0 = (-1,1)$ has $\lambda_{\max} \approx 1802$, giving $\alpha^{\star} \approx 0.0011$. Within the stable regime, smaller $\alpha$ converges more reliably but more slowly; the optimal $\alpha = 1/L$ balances these.}
\label{fig:stepsize_sensitivity}
\end{figure}

\noindent\textit{Armijo line search:} A practical alternative to fixed $\alpha$ is the Armijo (sufficient decrease) condition: accept $\alpha_k = \rho^j \alpha_{\text{init}}$ for the smallest $j \geq 0$ such that
\begin{equation}
f(x_k - \alpha_k \nabla f(x_k)) \leq f(x_k) - c_1 \alpha_k \|\nabla f(x_k)\|^2, \quad c_1 \in (0,1),
\label{eq:armijo}
\end{equation}
with typical $c_1 = 10^{-4}$, $\rho = 0.5$. Combined with the Wolfe curvature condition $|\nabla f(x_{k+1})^T d_k| \leq c_2 |\nabla f(x_k)^T d_k|$ ($c_2 \in (c_1, 1)$), Wolfe line search guarantees $\alpha_k$ is not too large (sufficient decrease) or too small (avoids early stopping). BFGS with Wolfe line search is the standard second-order algorithm for general smooth unconstrained problems.

\noindent\textit{Preconditioning interpretation of adaptive methods:} Every adaptive gradient method can be written as a preconditioned gradient step:
\begin{equation}
x_{k+1} = x_k - P_k^{-1} \nabla f(x_k),
\end{equation}
where $P_k$ is a diagonal preconditioner. For Adagrad, $P_k^{(i,i)} = \sqrt{G_k^{(i)}} / \alpha_0$; for RMSprop, $P_k^{(i,i)} = \sqrt{v_k^{(i)}} / \alpha_0$; for Adam, $P_k^{(i,i)} = \sqrt{\hat{v}_k^{(i)}} / \alpha$. These diagonal preconditioners approximate the curvature in each coordinate direction, effectively reducing the coordinate-wise condition number. Full-matrix preconditioning would require $P_k = [\nabla^2 f(x_k)]^{1/2}$ (the Newton step), connecting adaptive methods to quasi-Newton methods through the degree of curvature estimation.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require $x_0$, known optimum $f^{\star}$, regularisation $\varepsilon > 0$, $N$ iterations
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Theoretically optimal single step if $f^{\star}$ is exact}
    \State $x_{k+1} \gets x_k - \alpha_k g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textit{Derivation:} Setting $x_{k+1} = x_k - \alpha g_k$ and bounding $\|x_{k+1} - x^{\star}\|^2$:
\begin{align}
\|x_{k+1} - x^{\star}\|^2 &= \|x_k - x^{\star}\|^2 - 2\alpha_k g_k^T(x_k - x^{\star}) + \alpha_k^2 \|g_k\|^2 \nonumber \\
&\leq \|x_k - x^{\star}\|^2 - 2\alpha_k(f(x_k) - f^{\star}) + \alpha_k^2 \|g_k\|^2. \label{eq:polyak_bound}
\end{align}
Minimising the right side over $\alpha_k$ gives the Polyak step $\alpha_k = (f(x_k) - f^{\star})/\|g_k\|^2$, and plugging back:
$\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - (f(x_k) - f^{\star})^2/\|g_k\|^2$, which telescopes to $\min_{j \leq k}(f(x_j) - f^{\star}) = O(1/\sqrt{k})$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require $x_0$, base rate $\alpha_0$, $\varepsilon > 0$, $N$ iterations; $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative sum of squared gradients, per coordinate}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$ \Comment{Coordinate-wise adaptive rate}
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad implicitly performs gradient descent in a scaled metric: large historical gradients in coordinate $i$ yield a smaller effective step $\alpha_0/\sqrt{G_k^{(i)}}$. Since $G_k^{(i)} = \sum_{j=0}^{k} (g_j^{(i)})^2$ is monotonically non-decreasing, the effective step can only decrease over time --- a known limitation for non-stationary objectives.

\noindent\textit{Regret-bound interpretation:} Adagrad was originally designed for \emph{online learning}, where a sequence of arbitrary loss functions $\ell_t$ is presented and the goal is to minimise regret $R_T = \sum_{t=1}^T \ell_t(x_t) - \min_x \sum_{t=1}^T \ell_t(x)$. Adagrad achieves:
\begin{equation}
R_T \leq \frac{\|x_1 - x^{\star}\|^2}{2\alpha_0} \sum_i \sqrt{G_T^{(i)}} + \frac{\alpha_0}{2} \sum_i G_T^{(i)} / \sqrt{G_T^{(i)}} = O\!\left(\sqrt{\sum_i G_T^{(i)}}\right),
\end{equation}
which adapts to the geometry: if gradients are small in coordinate $i$ (e.g., sparse features), the effective regret contribution from that coordinate is small. This \emph{preconditioning} effect is precisely why Adagrad excels on sparse-gradient problems (e.g., NLP embeddings) but underperforms when all coordinates receive large gradients throughout (e.g., dense regression).

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require $x_0$, rate $\alpha_0$, decay $\beta \in (0,1)$, $\varepsilon$, $N$ iterations; $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta v_k + (1 - \beta) g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's cumulative sum with an exponential moving average (EMA), with effective memory length $1/(1-\beta) = 10$ iterations for $\beta = 0.9$. This prevents the step from decaying to zero on stationary problems and adapts rapidly to changes in gradient scale.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require $x_0$, step $\alpha$, momentum $\beta \in [0,1)$, $N$ iterations; $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta z_k + \alpha \nabla f(x_k)$ \Comment{Velocity update: accumulate past gradients}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

For a $\mu$-strongly convex, $L$-smooth function, the optimal Heavy Ball parameters are:
\begin{equation}
\alpha^{\star} = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2}, \qquad \beta^{\star} = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2,
\end{equation}
achieving the rate $\rho_{\text{HB}}^k$ with $\rho_{\text{HB}} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$, which is much better than $\rho_{\text{GD}} = (\kappa-1)/(\kappa+1)$ for large $\kappa$. However, this optimal rate holds only for quadratic functions; for general nonlinear objectives, Heavy Ball may not converge.

\subsection{Theoretical Background: Optimal Convergence Rates and Lower Bounds}

A fundamental result in optimisation theory (Nesterov, 1983) is that no first-order method --- one that uses only $\nabla f$ evaluations --- can achieve a convergence rate better than $O(1/k^2)$ for smooth convex functions in the black-box oracle model. Standard GD achieves only $O(1/k)$; Nesterov's accelerated method achieves the optimal $O(1/k^2)$. This is a genuine information-theoretic improvement, not merely better tuning.

\noindent\textit{Lower bound argument (Nesterov 1983):} Consider the worst-case family of functions $f_{d,k}(x) = \frac{L}{8}\left(\|x\|^2 - 2x_1 + x_1^2 + \sum_{i=1}^{\min(k,d)-1}(x_{i+1}-x_i)^2\right)$ over $\mathbb{R}^d$. Any first-order method that after $k$ steps has access only to the gradient information $\{\nabla f(x_j) : j \leq k\}$ satisfies:
\begin{equation}
\min_{j \leq k}(f(x_j) - f^{\star}) \geq \frac{3L\|x_0 - x^{\star}\|^2}{32(k+1)^2}.
\end{equation}
This lower bound matches Nesterov's upper bound to within a constant factor, proving the $O(1/k^2)$ rate is \emph{tight}. Heavy Ball and other heuristic momentum methods cannot beat this bound in the worst case; Nesterov's specific lookahead mechanism is essential for achieving it.

\noindent\textit{Gradient flow ODE interpretation:} In the continuous-time limit ($\alpha \to 0$), GD becomes the \emph{gradient flow}:
\begin{equation}
\dot{x}(t) = -\nabla f(x(t)), \qquad x(0) = x_0,
\label{eq:gradient_flow}
\end{equation}
a first-order ODE with $\frac{d}{dt}f(x(t)) = -\|\nabla f(x(t))\|^2 \leq 0$, so $f$ decreases monotonically. The Euler discretisation of \eqref{eq:gradient_flow} with step $\alpha$ recovers discrete GD. Nesterov's method corresponds to a second-order ODE with vanishing friction:
\begin{equation}
\ddot{x}(t) + \frac{3}{t}\dot{x}(t) + \nabla f(x(t)) = 0,
\label{eq:nesterov_ode}
\end{equation}
where the $3/t$ damping coefficient gives exactly the $O(1/t^2)$ rate (Su, Boyd \& Candès 2016). Heavy Ball corresponds to \eqref{eq:nesterov_ode} with a \emph{constant} friction coefficient $\gamma$ instead of $3/t$, which is why it achieves linear convergence on quadratics but not the $O(1/k^2)$ rate on general convex functions.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require $x_0$, step $\alpha$, max momentum $\beta_{\max}$, $N$ iterations; $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\, \beta_{\max}\right)$ \Comment{Momentum increases from 0 towards $\beta_{\max}$}
    \State $\tilde{x}_k \gets x_k + \beta_k z_k$ \Comment{Lookahead / extrapolation point}
    \State $z_{k+1} \gets \beta_k z_k - \alpha \nabla f(\tilde{x}_k)$ \Comment{Velocity at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Convergence]
For $L$-smooth convex $f$, Nesterov's method with step $\alpha = 1/L$ and momentum $\beta_k = (k-1)/(k+2)$ achieves:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} = O\!\left(\frac{1}{k^2}\right).
\end{equation}
This rate is optimal: no first-order method can guarantee convergence faster than $O(1/k^2)$ for smooth convex objectives in the worst case.
\end{theorem}

The key mechanism is the \emph{lookahead}: evaluating $\nabla f$ at $\tilde{x}_k = x_k + \beta_k z_k$ (the extrapolated point) rather than $x_k$ (as in Heavy Ball) allows the method to anticipate and correct overshooting. This seemingly small change transforms a heuristic acceleration into a provably optimal algorithm.

\noindent\textit{Lyapunov proof sketch:} Define the energy sequence $E_k = (k+1)^2(f(x_k) - f^{\star}) + 2L\|z_k\|^2$ where $z_k$ is the velocity. One can show $E_{k+1} \leq E_k$ using the descent lemma and the convexity inequality at the extrapolated point $\tilde{x}_k$. Telescoping gives $f(x_k) - f^{\star} \leq E_0 / (k+1)^2 = 2L\|x_0 - x^{\star}\|^2 / (k+1)^2$, proving the theorem. The key step uses \emph{both} $L$-smoothness (for the quadratic upper bound) and convexity of $f$ at $\tilde{x}_k$ --- establishing a tight coupling between the iterate and velocity updates.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require $x_0$, rate $\alpha$, $\beta_1, \beta_2 \in (0,1)$, $\varepsilon$, $N$ iterations; $m_0 \gets \mathbf{0}$, $v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1 m_{t-1} + (1-\beta_1) g_t$ \Comment{EMA of gradient (first moment)}
    \State $v_t \gets \beta_2 v_{t-1} + (1-\beta_2) g_t \odot g_t$ \Comment{EMA of $g^2$ (second moment)}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$, \quad $\hat{v}_t \gets v_t / (1 - \beta_2^t)$ \Comment{Bias correction}
    \State $x_{t+1} \gets x_t - \alpha \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam's bias corrections are essential in early iterations: without them, $m_1 = (1-\beta_1)g_1$ severely underestimates the true gradient mean due to initialisation at zero. With corrections, $\hat{m}_1 = g_1$ regardless of $\beta_1$. The effective per-coordinate update is approximately $\alpha \cdot \text{sign}(\hat{m}_t^{(i)})$ when the signal-to-noise ratio is high, bounding the per-step learning rate in $[-\alpha/\varepsilon, \alpha/\varepsilon]$ from above.

\subsubsection{Learning Rate Schedules}

For non-convex optimisation (deep networks, Rosenbrock), a single constant learning rate often fails: too large causes divergence, too small is slow. \textbf{Learning rate schedules} gradually reduce $\alpha_t$ to allow large initial steps followed by fine-tuning:
\begin{align}
\text{Polynomial decay:} & \quad \alpha_t = \alpha_0 / t^p, \quad p \in \{0.5, 1\} \label{eq:poly_decay} \\
\text{Cosine annealing:} & \quad \alpha_t = \frac{\alpha_0}{2}\left(1 + \cos\!\left(\frac{\pi t}{T}\right)\right) \label{eq:cosine} \\
\text{Warmup + cosine:} & \quad \alpha_t = \begin{cases} \alpha_{\max} t / T_{\text{warm}} & t \leq T_{\text{warm}} \\ \frac{\alpha_{\max}}{2}\left(1 + \cos\!\tfrac{\pi(t-T_{\text{warm}})}{T-T_{\text{warm}}}\right) & t > T_{\text{warm}} \end{cases} \label{eq:warmup}
\end{align}
Polynomial decay \eqref{eq:poly_decay} with $p = 0.5$ matches the diminishing step schedule required for SGD to achieve $O(1/\sqrt{k})$ convergence to the true minimiser (eliminating the noise floor). Cosine annealing \eqref{eq:cosine} is popular in deep learning; the smooth periodic shape avoids the abrupt step-size changes of piecewise-constant schedules. Warmup \eqref{eq:warmup} prevents early instability when gradients are high-variance at iteration 1.

\subsection{Conjugate Gradient: The Optimal First-Order Method for Quadratics}

For quadratic objectives $f(x) = \frac{1}{2}x^T H x - b^T x$, the \textbf{Conjugate Gradient} (CG) method achieves \emph{exact minimisation in at most $d$ steps} (where $d$ is the dimension), making it strictly superior to GD and Nesterov for quadratics when $d$ is small relative to the allowed iterations.

\begin{algorithm}[H]
\caption{Conjugate Gradient (Hestenes--Stiefel, for quadratic $Hx = b$)}
\label{alg:cg}
\begin{algorithmic}[1]
\Require $x_0$, $r_0 \gets b - Hx_0$, $p_0 \gets r_0$
\For{$k = 0, 1, \ldots, d-1$}
    \State $\alpha_k \gets r_k^T r_k / (p_k^T H p_k)$ \Comment{Exact line search along conjugate direction $p_k$}
    \State $x_{k+1} \gets x_k + \alpha_k p_k$
    \State $r_{k+1} \gets r_k - \alpha_k H p_k$ \Comment{Update residual}
    \State $\beta_k \gets r_{k+1}^T r_{k+1} / (r_k^T r_k)$ \Comment{Gram-Schmidt coefficient}
    \State $p_{k+1} \gets r_{k+1} + \beta_k p_k$ \Comment{Conjugate direction: $p_i^T H p_j = 0$ for $i \neq j$}
\EndFor
\end{algorithmic}
\end{algorithm}

The key property: the search directions $\{p_0, \ldots, p_{d-1}\}$ are \emph{$H$-conjugate} ($p_i^T H p_j = 0$), so each iteration minimises over a larger subspace. After $k$ steps, $x_k$ is the exact minimiser of $f$ over the Krylov subspace $\mathcal{K}_k(H, r_0) = \text{span}\{r_0, Hr_0, \ldots, H^{k-1}r_0\}$. In particular, if $H$ has only $s \leq d$ distinct eigenvalues, CG terminates in $\leq s$ steps --- much faster than $d$ when the spectrum is clustered.

The convergence bound for CG after $k$ steps is:
\begin{equation}
\frac{f(x_k) - f^{\star}}{f(x_0) - f^{\star}} \leq \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^{2k} = \rho_{\text{CG}}^k,
\end{equation}
matching the optimal Heavy Ball rate $\rho_{\text{HB}}^k$ but provably without the nonlinear oscillation issues.

\begin{figure}[H]
\centering
\includegraphics[width=0.95\textwidth]{figures/conjugate_gradient.pdf}
\caption{Q2: Conjugate Gradient vs GD vs Nesterov on an ill-conditioned quadratic ($d=30$, $\kappa = 100$, eigenvalues log-spaced from 1 to 100). Left: over 200 iterations, GD (blue) barely converges ($f(x_{200}) - f^{\star} \approx 0.016$); Nesterov (cyan) achieves $\approx 10^{-5}$ with $O(1/k^2)$ rate; CG (red) terminates exactly at machine precision by iteration $d = 30$, reaching $\approx 10^{-10}$ in 30 steps vs Nesterov's 200. For general non-quadratic $f$, CG must be restarted (Polak--Ribière or Fletcher--Reeves formulas) and loses the exact termination guarantee, but often retains superior convergence to GD. Nesterov (without CG's exact line search) is more practical for non-quadratic objectives.

\subsection{Mini-Batch Stochastic Gradient Descent}

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require $(X, y)$, $\theta_0$, rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, \ldots, E$}
    \State Randomly permute $\{1, \ldots, m\}$ into batches $\mathcal{B}_1, \mathcal{B}_2, \ldots$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Unbiased gradient estimate: $\mathbb{E}[g] = \nabla J(\theta)$}
        \State $\theta \gets \theta - \alpha g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The variance of the mini-batch gradient estimator is $\text{Var}(g) = \frac{\sigma^2_{\text{per-sample}}}{b}$, where $\sigma^2_{\text{per-sample}}$ is the variance of a single-sample gradient. Larger batches reduce variance but provide fewer updates per epoch. The trade-off is: more frequent updates (small $b$) converge faster per epoch but introduce noise; fewer updates (large $b$) are noisier per unit computation but reduce gradient variance per update.

\subsection{Variance-Reduced Stochastic Methods}

Pure SGD with a fixed step $\alpha > 0$ converges only to a noise floor of order $O(\alpha \sigma^2_{\text{per-sample}}/b)$ rather than to the exact minimiser $\theta^{\star}$. Eliminating this floor requires either (a) diminishing steps $\alpha_k \to 0$ (giving sublinear $O(1/k)$ convergence, losing SGD's per-epoch speed advantage) or (b) \textbf{variance reduction}: modifying the stochastic estimator so its variance vanishes near the optimum.

\begin{algorithm}[H]
\caption{SVRG (Stochastic Variance Reduced Gradient)}
\label{alg:svrg}
\begin{algorithmic}[1]
\Require $\tilde{\theta}_0$, step $\alpha$, inner period $m$, epochs $S$
\For{$s = 0, 1, \ldots, S-1$}
    \State $\tilde{g}^{(s)} \gets \frac{1}{n}\sum_{i=1}^n \nabla f_i(\tilde{\theta}_s)$ \Comment{Full gradient at reference point: $O(n)$ cost}
    \State $\theta_0 \gets \tilde{\theta}_s$
    \For{$t = 0, 1, \ldots, m-1$}
        \State Sample $i \sim \text{Uniform}\{1, \ldots, n\}$
        \State $v_t \gets \nabla f_i(\theta_t) - \nabla f_i(\tilde{\theta}_s) + \tilde{g}^{(s)}$ \Comment{Variance-reduced estimator}
        \State $\theta_{t+1} \gets \theta_t - \alpha v_t$
    \EndFor
    \State $\tilde{\theta}_{s+1} \gets \theta_m$ (or uniform-random draw from $\{\theta_0, \ldots, \theta_{m-1}\}$)
\EndFor
\end{algorithmic}
\end{algorithm}

The key property of $v_t$ is that $\mathbb{E}_i[v_t] = \nabla J(\theta_t)$ (unbiased gradient), and crucially:
\begin{equation}
\mathbb{E}_i\|v_t\|^2 \leq 4L^2\|\theta_t - \tilde{\theta}_s\|^2 + 4L^2\|\tilde{\theta}_s - \theta^{\star}\|^2 \xrightarrow{\theta_t, \tilde{\theta}_s \to \theta^{\star}} 0.
\end{equation}
The variance \emph{vanishes at the optimum}, unlike SGD where $\text{Var}(g_t) = \sigma^2_{\text{per-sample}}$ regardless of proximity to $\theta^{\star}$. Each inner iteration requires only \textbf{2} gradient evaluations ($\nabla f_i$ at $\theta_t$ and $\tilde{\theta}_s$); the full gradient is amortised over $m$ inner steps, costing $O(n/m)$ full passes per inner step.

\begin{theorem}[SVRG Linear Convergence]
For $\mu$-strongly convex, $L$-smooth $J = \frac{1}{n}\sum f_i$, with $m \geq 16L/\mu$ and step $\alpha = 1/(10L)$:
\begin{equation}
\mathbb{E}[J(\tilde{\theta}_s) - J^{\star}] \leq \rho^s (J(\tilde{\theta}_0) - J^{\star}), \quad \rho = \frac{1}{2} < 1.
\end{equation}
This is \textbf{linear (geometric) convergence} per epoch --- identical in rate to full gradient descent --- while each epoch costs only $O(n + m) = O(n)$ gradient evaluations (vs $O(n)$ for full GD). The per-epoch cost is the same but SVRG's linear convergence is maintained even with stochastic updates, unlike SGD which only achieves $O(1/k)$.
\end{theorem}

Related methods: \textbf{SAG/SAGA} (Stochastic Average Gradient) also achieve linear convergence by maintaining a table of $n$ past gradients, updating one per iteration; suitable when $n$ fits in memory. \textbf{SARAH} and \textbf{SPIDER} achieve optimal oracle complexity of $O(\sqrt{n}/\epsilon)$ for non-convex finite-sum problems.

\begin{equation}
\hat{g}_1(x) = 0.0039 + 0.0625(x - 0.25). \label{eq:fo}

\begin{equation}

Setting $\hat{g}_2'(x) = 0$ gives the Newton step: $x_1 = x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 = 0.1\overline{6}$.

\subsection{Newton's Method with Damping}

\begin{algorithm}[H]
\caption{Newton's Method with Hessian Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require $x_0$, step $\alpha \in (0,1]$, damping $\lambda \geq 0$, $N$ iterations
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute $g_k = \nabla f(x_k)$, $H_k = \nabla^2 f(x_k) + \lambda I$
    \State Solve $H_k p_k = g_k$ for the Newton direction $p_k$ \Comment{$O(d^3)$: linear system solve}
    \State $x_{k+1} \gets x_k - \alpha p_k$ \Comment{$\alpha = 1$: full Newton; $\alpha < 1$: damped}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Quadratic Convergence of Newton]
Let $f \in C^3$ with $\nabla^2 f$ being $M$-Lipschitz and $\nabla^2 f(x^{\star}) \succeq \mu I$. With full Newton step ($\alpha = 1$) and $x_0$ close to $x^{\star}$:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2.
\end{equation}
This means the number of accurate digits approximately \emph{doubles} at each iteration.
\end{theorem}

\noindent\textit{One-step property for quadratics:} If $f(\theta) = \frac{1}{2}(\theta - \theta^{\star})^T H (\theta - \theta^{\star}) + c$ (quadratic), then $\nabla^2 f = H$ is constant and the full Newton step gives $\theta_1 = \theta_0 - H^{-1} H (\theta_0 - \theta^{\star}) = \theta^{\star}$ exactly. For Benchmark A, this means \emph{convergence in one step} regardless of condition number.

The \textbf{Newton decrement} $\lambda(x) = \sqrt{\nabla f(x)^T [\nabla^2 f(x)]^{-1} \nabla f(x)}$ is a natural convergence criterion for Newton's method that is scale-invariant (independent of parameterisation). It satisfies $f(x) - f^{\star} \leq \lambda(x)^2 / (2\mu)$ and the stopping criterion $\lambda(x_k)^2/2 \leq \varepsilon$ guarantees $f(x_k) - f^{\star} \leq \varepsilon$ for strongly convex $f$.

\noindent\textit{Trust-region methods} improve Newton by replacing the line search with a constraint $\|p_k\| \leq \Delta_k$: $p_k = \arg\min_{\|p\| \leq \Delta_k} g_k^T p + \frac{1}{2}p^T H_k p$. The trust radius $\Delta_k$ is grown or shrunk based on how well the quadratic model predicts the actual function decrease. Trust-region methods are more robust than line-search Newton when the Hessian has negative eigenvalues (non-convex regions) or when the model is inaccurate far from the optimum.

Settings: GD runs 80 iterations; Newton runs only 20 iterations.

\subsection{Quasi-Newton Methods: BFGS}

Newton's method requires (i) computing $\nabla^2 f(x_k) \in \mathbb{R}^{d \times d}$ explicitly (not always available in code) and (ii) solving a $d \times d$ linear system at cost $O(d^3)$. \textbf{Quasi-Newton methods} approximate the inverse Hessian from gradient differences alone, requiring only $O(d^2)$ per iteration.

The \textbf{BFGS} (Broyden--Fletcher--Goldfarb--Shanno) update maintains $B_k \approx [\nabla^2 f(x_k)]^{-1}$:
\begin{equation}
B_{k+1} = \underbrace{\left(I - \rho_k s_k y_k^T\right)}_{\text{rank-1 correction}} B_k \underbrace{\left(I - \rho_k y_k s_k^T\right)}_{\text{rank-1 correction}} + \rho_k s_k s_k^T, \qquad \rho_k = \frac{1}{y_k^T s_k},
\label{eq:bfgs}
\end{equation}
where $s_k = x_{k+1} - x_k$ and $y_k = \nabla f(x_{k+1}) - \nabla f(x_k)$ are the step and gradient-change vectors. The update enforces the \emph{secant condition} $B_{k+1} y_k = s_k$, which approximates $[\nabla^2 f]^{-1} y_k = s_k$ (the Newton relationship). Combined with a Wolfe-condition line search, BFGS achieves \textbf{super-linear} convergence: $\|x_{k+1} - x^{\star}\| = o(\|x_k - x^{\star}\|)$ near $x^{\star}$.

\textbf{L-BFGS} (limited-memory BFGS) avoids storing the full $d \times d$ matrix $B_k$ by retaining only the last $\ell = 5$--20 $(s_k, y_k)$ pairs. The product $B_k g$ is computed via the two-loop recursion in $O(\ell d)$ time, making L-BFGS the standard algorithm for large-scale smooth unconstrained optimisation (neural network training, logistic regression with $d \gg 1000$).

GD requires $>$40 iterations for the same accuracy. \begin{remark}
BFGS cannot exploit the exact Hessian structure even when it is available. For the linear regression Benchmark A, Newton terminates in 1 step; BFGS requires $\approx 5$ iterations because it builds up $B_k \approx H^{-1}$ incrementally. The payoff for BFGS is gradient-only access at $O(d^2)$ cost, while Newton needs $O(d^3)$ and explicit second derivatives.
\end{remark}

\noindent\textit{Wolfe conditions for line search:} The BFGS implementation requires not just sufficient decrease (Armijo) but also the \emph{curvature condition}:
\begin{equation}
\nabla f(x_k + \alpha_k p_k)^T p_k \geq c_2 \nabla f(x_k)^T p_k, \quad 0 < c_1 < c_2 < 1.
\label{eq:wolfe}
\end{equation}
Condition \eqref{eq:wolfe} ensures $\alpha_k$ is not too small (gradient at next point is less negative in direction $p_k$ than at $x_k$), which guarantees $y_k^T s_k > 0$ and thus positive definiteness of $B_{k+1}$ in the BFGS update. Without this curvature condition, the BFGS update may produce an indefinite $B_{k+1}$ that is no longer a valid Hessian-inverse approximation. In the experiments, simple Armijo backtracking is used with the convexity of $f$ near the optimum guaranteeing $y_k^T s_k > 0$ empirically.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference (FD) approximation:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \qquad i = 1, \ldots, d,
\label{eq:fd}
\end{equation}
has truncation error $O(\delta)$ (from higher-order Taylor terms) and round-off error $O(\varepsilon_{\text{machine}}/\delta)$. The total error is minimised at $\delta^{\star} = O(\sqrt{\varepsilon_{\text{machine}}}) \approx 10^{-8}$. The experiments use $\delta = 0.05$ (moderate, practical) and $\delta = 0.8$ (too large: truncation error $O(0.8 \cdot M_2/2)$ dominates, where $M_2 = \max|\partial^2 f/\partial x_i^2| \approx 10$, giving error $\approx 4$). Cost: $d+1 = 3$ function evaluations per gradient.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Gradient Estimation)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require $x_0$, step $\alpha$, perturbation $\delta$, $N$ iterations
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \text{Uniform}(\mathbb{S}^{d-1})$ (uniform on unit sphere)
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta} u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The estimator satisfies $\mathbb{E}_u[\hat{g}_k] = \nabla f_{\delta}(x_k)$ where $f_\delta$ is the $\delta$-Gaussian smoothing of $f$, and $\|\nabla f - \nabla f_\delta\| = O(L\delta^2)$. Only \textbf{2} function evaluations are needed per iteration regardless of dimension $d$, versus $d+1$ for FD. However, the estimator variance is $O(L^2 d/\delta^2)$, so more iterations are needed to compensate, making it most competitive in very high-dimensional settings.

\begin{theorem}[Zeroth-Order Oracle Complexity]
\label{thm:zo_complexity}
Let $f$ be $L$-smooth and $\mu$-strongly convex. To achieve $\mathbb{E}[f(x_k) - f^{\star}] \leq \varepsilon$:
\begin{itemize}
\item \textbf{First-order GD} (exact $\nabla f$): $O\!\left(\kappa \log \frac{1}{\varepsilon}\right)$ gradient evaluations, where $\kappa = L/\mu$.
\item \textbf{Finite Difference} ($d+1$ FE per gradient): $O\!\left((d+1)\kappa \log \frac{1}{\varepsilon}\right)$ function evaluations --- exactly $d+1$ times more expensive.
\item \textbf{Nesterov Random Search} (2 FE per step, $K$ steps needed): $O\!\left(d \kappa \log \frac{1}{\varepsilon}\right)$ function evaluations. Despite using only 2 FE per step, the high-variance estimator requires $O(d\kappa)$ steps to converge, giving the same $d$-fold overhead as FD.
\end{itemize}
For smooth (but not strongly convex) $f$, achieving $\min_{j \leq k}\|\nabla f(x_j)\|^2 \leq \varepsilon$ requires $O(d L / \varepsilon)$ function evaluations for zeroth-order methods vs $O(L/\varepsilon)$ for first-order --- the fundamental $d$-fold dimension penalty of derivative-free optimisation.
\end{theorem}

The practical conclusion from Theorem~\ref{thm:zo_complexity} is that zeroth-order methods are competitive only when (a) gradient computation is genuinely unavailable (black-box simulators, hardware-in-the-loop systems), or (b) $d$ is very large and the gradient computation is so expensive that even 2 cheap function evaluations per step are preferable. For small $d$ (as in all experiments here, $d = 2$), finite differences almost always dominate random search because $(d+1)/2 = 1.5$ --- only marginally more expensive per step while delivering far lower-variance estimates.

\subsection{Part B: Derivative-Free Methods on Rosenbrock (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

Nelder--Mead maintains a simplex of $n+1 = 3$ vertices and iteratively transforms it through four operations (with standard parameters $\alpha_r = 1, \gamma = 2, \rho = 0.5, \sigma = 0.5$): \textbf{Reflection} (move worst vertex through centroid of remaining vertices), \textbf{Expansion} (extend further if reflection succeeded), \textbf{Contraction} (contract towards centroid if reflection failed), \textbf{Shrink} (contract all vertices towards best if contraction failed). Each iteration uses $\leq 3$ function evaluations (in 2D). Settings: initial simplex step $= 0.35$, 160 iterations.

\begin{remark}[Nelder--Mead Convergence]
Nelder--Mead has \emph{no provable convergence guarantee} for $d \geq 2$. McKinnon (1998) constructed a 2D example where Nelder--Mead converges to a non-stationary point. Convergence is proven only for $d = 1$ (as a golden-section-like method). Despite this, empirical performance is often strong for smooth, low-dimensional problems ($d \leq 10$), making it the default in \texttt{scipy.optimize.minimize} with \texttt{method='Nelder-Mead'}. For $d > 20$, modern alternatives are preferable: \textbf{CMA-ES} (Covariance Matrix Adaptation Evolution Strategy) adapts the search distribution covariance matrix, achieving $O(d^2)$ function evaluations to $\varepsilon$ accuracy with reliable convergence, at cost $O(d^2)$ per iteration; \textbf{Bayesian Optimisation} (with a Gaussian Process surrogate) is optimal for expensive black-box functions ($>O(100)$ evaluations), achieving $O(\sqrt{k \log k})$ cumulative regret.
\end{remark}

\subsection{Problem Setup and Optimality Conditions}

Minimise $f(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)$ (Benchmark B) subject to $x_1 \geq 0.5$. Starting from $x_0 = (0.2, 4.0)$ (infeasible). The feasible set is the closed half-space $\mathcal{X} = \{x : h(x) \leq 0\}$ with $h(x) = 0.5 - x_1$.

The KKT optimality conditions for $\min_{x \in \mathcal{X}} f(x)$ with $h(x) = 0.5 - x_1 \leq 0$ are:
\begin{equation}
\nabla f(x^{\star}) + \lambda^{\star} \nabla h(x^{\star}) = 0, \quad \lambda^{\star}(0.5 - x_1^{\star}) = 0, \quad \lambda^{\star} \geq 0, \quad x_1^{\star} \geq 0.5.
\end{equation}
Since the unconstrained minimum $x^{\star}_{\text{unc}} = (0.582, 2.0)$ satisfies $x_1^{\star} = 0.582 > 0.5$ (constraint inactive), the KKT solution has $\lambda^{\star} = 0$ and $x^{\star} = x^{\star}_{\text{unc}}$.

\begin{remark}[Slater's condition]
Since $x^{\star}_{\text{unc}}$ is strictly feasible ($x_1 = 0.582 > 0.5$), Slater's condition holds, so strong duality holds and the KKT conditions are both necessary and sufficient for global optimality.
\end{remark}

\subsection{Algorithms}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require $x_0$, step $\alpha$, $N$ iterations
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = (\max(0.5, z_{k+1,1}),\; z_{k+1,2})$ \Comment{Project onto $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space $\{x : x_1 \geq 0.5\}$, projection is $O(1)$: simply clip $x_1$ to 0.5 if it falls below. For general polytopes $\{Ax \leq b\}$, projection requires solving a QP, which costs $O(d^2 m)$ with $m$ constraints.

\begin{algorithm}[H]
\caption{$L_1$ Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require $x_0$, step $\alpha$, penalty $\lambda > 0$, $N$ iterations
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_\lambda(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Constraint violation penalty}
    \State $x_{k+1} \gets x_k - \alpha \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The gradient of $F_\lambda$ is $\nabla F_\lambda(x) = \nabla f(x) + \lambda \mathbf{1}[x_1 < 0.5] (-e_1)$, a subdifferential of $\max(0, 0.5-x_1)$. The penalty term makes $F_\lambda$ non-smooth at the constraint boundary, requiring a smaller step $\alpha$ for stability (as seen in the $\lambda = 4.5$, $\alpha = 0.03$ choice vs $\lambda = 1.8$, $\alpha = 0.05$).

Settings: Projected GD ($\alpha = 0.08$), Penalty $\lambda \in \{0.15, 1.8, 4.5\}$ (step $\alpha = 0.05$ for $\lambda \leq 1.8$, $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All: 100 iterations.

\subsection{Results}

\begin{figure}[H]

\subsection{Proximal Gradient Method}

For problems of the form $\min_{x} F(x) = f(x) + g(x)$ where $f$ is smooth (gradient available) and $g$ is non-smooth but has a tractable \emph{proximal operator}, the \textbf{proximal gradient method} (ISTA) replaces the gradient step on $F$ with a proximal step:
\begin{equation}
x_{k+1} = \text{prox}_{\alpha g}(x_k - \alpha \nabla f(x_k)), \qquad \text{prox}_{\lambda g}(v) = \arg\min_x \left\{g(x) + \frac{1}{2\lambda}\|x - v\|^2\right\}.
\label{eq:prox_gd}
\end{equation}
Two key examples:
\begin{itemize}
\item \textbf{Projected GD:} $g = \mathbb{I}_{\mathcal{X}}$ (indicator function, $0$ if $x \in \mathcal{X}$, $+\infty$ otherwise). Then $\text{prox}_{\alpha g}(v) = \Pi_{\mathcal{X}}(v)$ --- the projection. So projected GD \emph{is} proximal gradient with an indicator function non-smooth term.
\item \textbf{LASSO (L1-regularised regression):} $g = \lambda\|x\|_1$. Then $\text{prox}_{\alpha g}(v)_i = \text{sign}(v_i)\max(|v_i| - \alpha\lambda, 0)$ (soft thresholding), which promotes sparsity. The resulting algorithm is \textbf{ISTA} (Iterative Shrinkage-Thresholding Algorithm).
\end{itemize}

\textbf{FISTA} (Fast ISTA, Beck \& Teboulle 2009) applies Nesterov acceleration to the proximal gradient step:
\begin{align}
y_{k+1} &= x_k + \frac{t_k - 1}{t_{k+1}}(x_k - x_{k-1}), \quad t_{k+1} = \frac{1 + \sqrt{1 + 4t_k^2}}{2} \nonumber \\
x_{k+1} &= \text{prox}_{\alpha g}(y_{k+1} - \alpha \nabla f(y_{k+1})),
\end{align}
achieving the optimal $O(1/k^2)$ convergence rate for the composite objective $F = f + g$ under $L$-smoothness of $f$ (same rate as Nesterov for smooth objectives).

\subsection{Comparative Analysis}

\textbf{Projected GD} enforces feasibility at every step as a hard constraint. The projection $\Pi_{\mathcal{X}}$ is trivially cheap for the half-space but may be expensive for complex constraint sets. Convergence rate matches unconstrained GD: $O(\rho^k)$ for strongly convex $f$ with $\alpha \leq 1/L$.

\textbf{Penalty methods} approximate the constrained problem by adding a feasibility violation cost to the objective. Larger $\lambda$ enforces feasibility more aggressively but increases the Lipschitz constant of $\nabla F_\lambda$ (the penalty term creates a kink at $x_1 = 0.5$), requiring a smaller step size. In the limit $\lambda \to \infty$, penalty methods recover exact feasibility at the cost of numerical instability.

\textbf{Augmented Lagrangian / ADMM} (not implemented here) would combine both approaches: $F_{\lambda, \mu}(x) = f(x) + \lambda h(x)_+ + \frac{\mu}{2}\|h(x)_+\|^2$, updating $\lambda$ alongside $x$, achieving exact feasibility without the $\lambda \to \infty$ instability.

\subsection{Augmented Lagrangian Method (Theoretical Analysis)}

The pure penalty method's instability arises because exact feasibility requires $\lambda \to \infty$, which inflates the Lipschitz constant of $\nabla F_\lambda$ as $O(\lambda)$ and forces step size $\alpha = O(1/\lambda) \to 0$. The \textbf{Augmented Lagrangian Method} (ALM) circumvents this by dualising the constraint, allowing exact feasibility at \emph{finite} penalty $\mu$:

\begin{align}
\mathcal{L}_\mu(x, \lambda) &= f(x) + \lambda h(x) + \frac{\mu}{2} h(x)^2, \label{eq:alm}\\
x^{(k+1)} &= \arg\min_x \mathcal{L}_\mu(x, \lambda^{(k)}), \label{eq:alm_x}\\
\lambda^{(k+1)} &= \lambda^{(k)} + \mu\, h(x^{(k+1)}). \label{eq:alm_dual}
\end{align}

The dual update \eqref{eq:alm_dual} increments the multiplier by the constraint violation scaled by $\mu$. As $\lambda^{(k)} \to \lambda^{\star}$ (the KKT multiplier), the inner minimisation \eqref{eq:alm_x} drives $h(x^{(k+1)}) \to 0$ even for fixed $\mu$. The Lipschitz constant of $\nabla_x \mathcal{L}_\mu$ is $O(L + \mu)$, which is bounded as long as $\mu$ is fixed --- no instability as with pure penalty.

For inequality constraints $h(x) \leq 0$ (as in our Q5 setting $h = 0.5 - x_1 \leq 0$), the update becomes:
\begin{equation}
\lambda^{(k+1)} = \max\!\left(0,\; \lambda^{(k)} + \mu\, h(x^{(k+1)})\right),
\end{equation}
where the projection onto $\lambda \geq 0$ enforces dual feasibility (non-negativity of the inequality multiplier).

\begin{proposition}[ALM Convergence]
For $\mu$-strongly convex $f$ and a constraint set satisfying Slater's condition, ALM with fixed $\mu > 0$ and dual updates \eqref{eq:alm_dual} converges linearly to the primal-dual optimum $(x^{\star}, \lambda^{\star})$:
\begin{equation}
\|x^{(k)} - x^{\star}\| + |\lambda^{(k)} - \lambda^{\star}| = O(\rho^k), \quad \rho = 1 - \frac{\mu}{\mu + L} < 1.
\end{equation}
No constraint on $\mu$ is required for convergence (unlike pure penalty's $\lambda \to \infty$).
\end{proposition}

In our Q5 experiment, the unconstrained minimiser $(0.582, 2.0) \in \mathcal{X}$ means $\lambda^{\star} = 0$ (inactive constraint, KKT complementarity $\lambda^{\star} h(x^{\star}) = 0$). ALM would converge with $\lambda^{(k)} \to 0$ and $x^{(k)} \to (0.582, 2.0)$, recovering the unconstrained solution --- confirming that projected GD and penalty methods correctly identify the unconstrained minimiser as the constrained optimum.

\subsection{ADMM (Alternating Direction Method of Multipliers)}

For \emph{separable} objectives $\min_{x, z} f(x) + g(z)$ subject to $Ax + Bz = c$, ADMM splits the ALM update into alternating $x$ and $z$ minimisations:
\begin{align}
x^{(k+1)} &= \arg\min_x f(x) + \frac{\rho}{2}\|Ax + Bz^{(k)} - c + u^{(k)}\|^2, \label{eq:admm_x}\\
z^{(k+1)} &= \arg\min_z g(z) + \frac{\rho}{2}\|Ax^{(k+1)} + Bz - c + u^{(k)}\|^2, \label{eq:admm_z}\\
u^{(k+1)} &= u^{(k)} + Ax^{(k+1)} + Bz^{(k+1)} - c. \label{eq:admm_u}
\end{align}
Each sub-problem \eqref{eq:admm_x}--\eqref{eq:admm_z} can be solved independently and in parallel, enabling \textbf{distributed optimisation}. ADMM is the standard algorithm for:
\begin{itemize}
\item \textbf{Distributed machine learning}: split data across machines, each solves a local subproblem; consensus enforced via the constraint $x_i = z$ (global variable).
\item \textbf{L1-regularised problems}: $f(x) = \frac{1}{2}\|Ax - b\|^2$, $g(z) = \lambda\|z\|_1$ with $x = z$; the $z$ step is soft-thresholding (same as proximal gradient), but $x$ and $z$ updates alternate.
\item \textbf{Model predictive control} and \textbf{signal processing}: sparse recovery under linear constraints.
\end{itemize}
ADMM converges at $O(1/k)$ for general convex objectives and linearly for strongly convex $f$ and $g$, with penalty parameter $\rho$ controlling the convergence speed (analogous to $\mu$ in ALM).

\subsection{Linear Programme over Box Constraints}

Feasible set: $\mathcal{X} = \{x : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (box = polytope with 4 vertices).

\textbf{LP:} Minimise $f(x) = x_1 + 2x_2$ over $\mathcal{X}$. For a linear objective $c^Tx$ over a box $\times_{i=1}^d [l_i, u_i]$, the minimum is:
\begin{equation}
x_i^{\star} = \begin{cases} l_i & \text{if } c_i > 0 \\ u_i & \text{if } c_i < 0 \end{cases}, \qquad f^{\star} = \sum_i c_i x_i^{\star}.
\end{equation}
With $c = (1, 2)$ (both positive): $x^{\star} = (0.5, -5)$, $f^{\star} = 0.5 + 2(-5) = -9.5$.

\begin{theorem}[Fundamental Theorem of Linear Programming]
The minimum of a linear functional $c^Tx$ over a non-empty, compact convex polytope $\mathcal{X}$ is attained at a vertex (extreme point) of $\mathcal{X}$.
\end{theorem}

\subsection{Frank--Wolfe Algorithm (Conditional Gradient)}

\begin{algorithm}[H]
\caption{Frank--Wolfe / Conditional Gradient}
\label{alg:fw}
\begin{algorithmic}[1]
\Require $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, $N$ iterations
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ \Comment{Linear Minimisation Oracle (LMO)}
    \State $x_{k+1} \gets \beta x_k + (1-\beta) z_k$ \Comment{Convex combination: $\gamma = 1-\beta$}
\EndFor
\end{algorithmic}
\end{algorithm}

For box constraints, the LMO decomposes per coordinate in $O(d)$ operations:
$(z_k)_i = l_i$ if $\partial_i f(x_k) > 0$, else $(z_k)_i = u_i$. Since $x_k, z_k \in \mathcal{X}$ and $\mathcal{X}$ is convex, $x_{k+1} = \beta x_k + (1-\beta) z_k \in \mathcal{X}$ automatically --- no projection needed.

\begin{proposition}[Frank--Wolfe Convergence, Standard Schedule]
For $L$-smooth convex $f$ with $\mathcal{X}$ of diameter $C$ and step schedule $\gamma_k = 2/(k+2)$:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LC^2}{k+2} = O\!\left(\frac{1}{k}\right).
\end{equation}
The Frank--Wolfe gap $g_k = \nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - f^{\star}$ provides a computable upper bound (duality gap).
\end{proposition}

\noindent The fixed-$\beta$ implementation used in the experiments corresponds to constant step $\gamma = 1-\beta$, which does not decay. This prevents exact convergence to $x^{\star}$ (the method cycles around the optimum) but may converge faster in early iterations.

\subsection{Iterate Sparsity and Structural Properties}

A key structural advantage of Frank--Wolfe is that the iterates are \emph{convex combinations of at most $k+1$ vertices} of $\mathcal{X}$:
\begin{equation}
x_k = \sum_{j=0}^{k} w_j z_j, \quad w_j \geq 0, \quad \sum_j w_j = 1, \quad z_j \in \text{vertices}(\mathcal{X}).
\end{equation}
This follows by induction: $x_{k+1} = (1-\gamma_{k+1}) x_k + \gamma_{k+1} z_k$ is a convex combination of $x_k$ (which uses $k$ vertices by hypothesis) and one new vertex $z_k$. Consequence: for a polytope with many vertices (e.g., an $n \times n$ permutation matrix polytope with $n!$ vertices), Frank--Wolfe naturally produces \textbf{sparse} iterates using at most $k+1$ active vertices --- critical in matrix factorisation, compressed sensing, and sparsity-constrained optimisation.

\subsection{LMO Cost Analysis for Specific Constraint Sets}

The computational advantage of Frank--Wolfe depends entirely on the LMO cost:
\begin{center}
\begin{tabular}{lcc}
\toprule
\textbf{Constraint Set $\mathcal{X}$} & \textbf{LMO Cost} & \textbf{Projection Cost} \\
\midrule
Box $\times_{i=1}^d [l_i, u_i]$ & $O(d)$ & $O(d)$ (same!) \\
Simplex $\{\sum x_i = 1, x \geq 0\}$ & $O(d)$ (min coord) & $O(d \log d)$ (sort + threshold) \\
$\ell_1$ ball $\{\|x\|_1 \leq r\}$ & $O(d)$ & $O(d \log d)$ \\
Spectrahedron $\{X \succeq 0, \text{tr}(X) = 1\}$ & $O(d^3)$ (top eigvec) & $O(d^3)$ (eigendecomp) \\
Nuclear norm ball $\{\|X\|_* \leq r\}$ & $O(d^2 r_{\text{sparse}})$ (sparse SVD) & $O(d^3)$ (full SVD) \\
\bottomrule
\end{tabular}
\end{center}
For the nuclear norm ball (matrices with bounded trace norm), the LMO requires only the top singular vector pair ($O(d^2)$ vs $O(d^3)$ for full SVD projection), making Frank--Wolfe the \emph{only} practical algorithm for low-rank matrix completion problems at scale. The $O(1/k)$ convergence rate is an acceptable trade-off when the projection itself is $O(d^3)$.

In the Q6 experiment on the box $\mathcal{X}$, both FW and projected GD have $O(d) = O(2)$ costs per iteration, so the theoretical projection-avoidance advantage vanishes; this is why projected GD's linear rate is the practical winner for this particular constraint set.

\subsection{Key Lessons Learned}

\begin{enumerate}
\item \textbf{Second-order information is transformative for well-conditioned problems.} Newton's method reduces a quadratic problem to one step and achieves quadratic convergence near the optimum of any smooth function, requiring $O(\log\log 1/\varepsilon)$ iterations vs GD's $O(\kappa \log 1/\varepsilon)$. The $O(d^3)$ cost per iteration limits it to small-$d$ problems; BFGS ($O(d^2)$) and L-BFGS ($O(\ell d)$) are the practical alternatives for large scale.

\item \textbf{Nesterov acceleration is provably optimal for first-order methods.} The $O(1/k^2)$ rate (or $O(\sqrt{\kappa}\log 1/\varepsilon)$ for strongly convex) is not merely an empirical improvement --- it matches the information-theoretic lower bound in the black-box oracle model. The key mechanism is gradient evaluation at the extrapolated point (lookahead), not just momentum accumulation (as in Heavy Ball, which is heuristic for nonlinear objectives).

\item \textbf{Adaptive step sizes improve robustness at the cost of requiring tuning.} Polyak step is theoretically optimal when $f^{\star}$ is known exactly but catastrophically unstable otherwise. Adagrad decays the step monotonically, which is unsuitable for non-stationary objectives. RMSprop and Adam use exponential averaging to prevent step decay, making Adam the most popular choice in deep learning despite lacking strong convergence guarantees in the nonconvex deterministic setting.

\item \textbf{Stochastic methods need variance reduction for linear convergence.} Pure SGD with constant step achieves only a noise floor $O(\alpha\sigma^2)$, not the true minimum. SVRG and SAGA restore linear convergence at the same per-epoch cost as full GD, by using variance-reduced gradient estimators whose variance vanishes at the optimum. For $n$ samples, SVRG achieves linear convergence at $O(n)$ gradient evaluations per epoch vs $O(n)$ for full GD --- the same cost but without the noise floor.

\item \textbf{Derivative-free methods pay a dimension factor.} By Theorem~\ref{thm:zo_complexity}, both finite differences and random search require $O(d)$ times more function evaluations than first-order GD. Grid search is the only globally optimal derivative-free method, at the prohibitive cost of $N^d$ evaluations. Nelder--Mead has no global convergence guarantee for $d \geq 2$ but is often practical.

\item \textbf{Constraint handling has two paradigms, with ALM bridging them.} Projected GD enforces feasibility at every step (hard constraint), while penalty methods relax constraints into the objective (soft). Augmented Lagrangian combines both: a penalty term for local geometry and a dual update for exact global feasibility at finite $\mu$. Frank--Wolfe avoids projection entirely via the LMO, achieving $O(1/k)$ for polytopic constraints but at the cost of slower convergence vs projected GD's linear rate.

\item \textbf{Problem structure determines algorithm choice.} The Rosenbrock function ($\kappa \approx 1600$, nonquadratic) is hardest for all first-order methods. Strongly convex quadratics are easiest (Newton in one step). The practitioner's decision tree: (i) second-order available and $d$ small? $\to$ Newton or BFGS. (ii) $d$ large, gradients available? $\to$ L-BFGS or Nesterov. (iii) Finite-sum structure with $n$ samples? $\to$ SVRG or Adam. (iv) No gradients? $\to$ FD for small $d$, random search for large $d$. (v) Constraints? $\to$ projected GD (hard), ALM (dual), or FW (polytope).
\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x, G = x0.copy().astype(float), np.zeros_like(x0)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x -= eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)

def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x, v = x0.copy().astype(float), np.zeros_like(x0)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2  # EMA
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x -= eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball and Nesterov Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x, z = x0.copy().astype(float), np.zeros_like(x0)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        z = beta * z + alpha * grad_fn(x)   # momentum accumulation
        x -= z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x, z = x0.copy().astype(float), np.zeros_like(x0)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)  # increasing schedule
        lookahead = x + beta_k * z           # extrapolation
        g = grad_fn(lookahead)               # gradient at lookahead
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser and Newton's Method}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)    # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x -= alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))   # Tikhonov regularisation
        try:
            p = np.linalg.solve(H_reg, g)       # Newton direction
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x -= alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={SVRG and BFGS (Illustrative Implementations)}]
def svrg(grad_fi_fn, grad_full_fn, loss_fn, theta0, alpha, m, n_epochs, n_data):
    """Stochastic Variance Reduced Gradient."""
    theta = theta0.copy()
    f_hist = [loss_fn(theta)]
    for epoch in range(n_epochs):
        theta_ref = theta.copy()
        g_ref = grad_full_fn(theta_ref)   # O(n): full gradient
        for t in range(m):               # inner loop
            i = np.random.randint(0, n_data)
            g_i      = grad_fi_fn(theta, i)      # gradient at current
            g_i_ref  = grad_fi_fn(theta_ref, i)  # gradient at reference
            v = g_i - g_i_ref + g_ref    # variance-reduced estimator
            theta -= alpha * v
        f_hist.append(loss_fn(theta))
    return theta, np.array(f_hist)

def bfgs(grad_fn, loss_fn, x0, max_iter=100, tol=1e-8):
    """BFGS with backtracking Armijo line search."""
    d = len(x0); x = x0.copy(); B = np.eye(d)
    f_hist = [loss_fn(x)]
    for _ in range(max_iter):
        g = grad_fn(x)
        if np.linalg.norm(g) < tol: break
        p = -B @ g                   # quasi-Newton direction
        alpha, f0, slope = 1.0, loss_fn(x), g @ p
        while loss_fn(x + alpha*p) > f0 + 0.01*alpha*slope:
            alpha *= 0.5             # Armijo backtracking
        s = alpha * p; x_new = x + s
        y = grad_fn(x_new) - g; sy = s @ y
        if sy > 1e-10:               # BFGS rank-2 update
            rho = 1.0 / sy; I = np.eye(d)
            B = (I - rho*np.outer(s,y)) @ B @ (I - rho*np.outer(y,s)) \
                + rho * np.outer(s, s)
        x = x_new; f_hist.append(loss_fn(x))
    return x, np.array(f_hist)
\end{lstlisting}

def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LMO: per-coordinate minimisation over box
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # convex combination
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
