\begin{abstract}
This report presents a comprehensive comparative study of six categories of optimisation algorithms applied to three benchmark problems of increasing difficulty: a nearly-spherical quadratic regression loss ($\kappa \approx 1.12$), a moderately anisotropic toy neural-network loss ($\kappa \approx 6.9$), and the highly ill-conditioned Rosenbrock function ($\kappa \approx 2508$). Six method families are investigated: (1)~adaptive step-size methods — Polyak, Adagrad, RMSprop, and Heavy Ball; (2)~momentum-based methods — Nesterov acceleration and Adam; (3)~Newton's method with local quadratic approximations; (4)~derivative-free approaches — finite differences, Nesterov random search, Nelder--Mead, and grid search; (5)~constrained optimisation via projected gradient descent and penalty methods; and (6)~the Frank--Wolfe algorithm for linear-constrained problems. Key findings: the Polyak step size oscillates chaotically when $f^\star$ is misspecified (reaching values $\approx 432$ on Benchmark B) but achieves the fastest convergence ($f < 0.1$ in 38 iterations on Rosenbrock) when $f^\star = 0$ is exact; Nesterov momentum achieves the theoretical optimal $O(1/k^2)$ rate and is the best gradient-only method on Rosenbrock ($f = 0.479$, $7\times$ better than GD); Newton's method solves the quadratic exactly in one step and reaches $f = 0.686$ on Rosenbrock in just 20 damped iterations vs $f = 3.732$ for GD in 80 iterations; Frank--Wolfe achieves $O(1/k)$ convergence at $O(d)$ LP cost per step without projection (90$\times$ more accurate than fixed-$\beta$ on the boundary problem at 250 iterations); and Nelder--Mead --- a derivative-free simplex method --- converges to the Rosenbrock global minimum $(1,1)$ with $f \approx 10^{-7}$ in just 120 iterations, outperforming gradient descent which cannot reach $f < 1.0$ in 600 iterations. Theoretical contributions include the information-theoretic lower bound (Nesterov 1983) proving $O(1/k^2)$ is optimal for first-order methods, a formal Armijo backtracking line search algorithm showing how Newton's method recovers quadratic convergence with global guarantees, and the finite-difference Hessian cost analysis ($O(d^2)$ evaluations) motivating quasi-Newton methods. All numerical results are verified with \texttt{np.random.seed(42)}.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. We adopt the notation: $L$ denotes the Lipschitz constant of the gradient (i.e., $\|\nabla^2 f(x)\|_2 \leq L$ for all $x$), $\mu$ denotes the strong convexity constant, and $\kappa = L/\mu$ denotes the condition number.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
\label{subsec:bench_A}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \sim \mathcal{N}(0, I_2), \quad
y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2 = \frac{1}{2m}\|X\theta - y\|^2,
\end{equation}
with gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$ and constant Hessian $H = \frac{1}{m}X^T X \succ 0$.

\medskip
\noindent\textbf{Theoretical properties.} Since $X^T X / m \to I$ as $m\to\infty$ (law of large numbers with i.i.d.\ $\mathcal{N}(0,I)$ inputs), this benchmark is nearly spherical. With seed 42 and $m=1000$, we compute $L \approx 1.0325$, $\mu \approx 0.9247$, and $\kappa = L/\mu \approx 1.12$. This extremely low condition number means \emph{all} first-order methods converge rapidly. Due to observation noise, the minimum loss is $J^\star = J(\hat\theta_{\text{OLS}}) \approx 0.4826$ (not zero), with the OLS solution $\hat\theta = (X^TX)^{-1}X^Ty \approx (3.028, 4.017)^T$. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\label{subsec:bench_B}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\label{eq:bench_B}
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is:
\begin{equation}
H(x_1) = \operatorname{diag}\!\bigl(2 - \sin(x_1),\ 10\bigr).
\end{equation}
\textbf{Theoretical properties.} Since $\sin(x_1) \in [-1, 1]$, the Hessian eigenvalues satisfy $\lambda_1(x_1) = 2 - \sin(x_1) \in [1, 3]$ and $\lambda_2 = 10$ (constant). Consequently, $L = 10$ and $\mu = 1$ globally, giving $\kappa = 10$. At the optimum $x_1^\star \approx 0.582$ (where $2(x_1-1) + \cos(x_1) = 0$), $\lambda_1 \approx 1.450$, giving a local condition number $\kappa_{\text{local}} \approx 6.9$. The global minimum is at $(x_1^\star, 2.0) \approx (0.582, 2.0)$ with $f^\star \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\label{subsec:bench_C}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\label{eq:rosenbrock}
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$. The Hessian is:
\begin{equation}
H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
\textbf{Theoretical properties.} At the optimum $(1,1)$: $H = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}$, with eigenvalues $\lambda_{\min} \approx 0.40$ and $\lambda_{\max} \approx 1001.6$, giving $\kappa \approx 2508$. This extreme ill-conditioning arises from the narrow curved valley along $x_2 = x_1^2$. Furthermore, the Rosenbrock function is \emph{not} globally convex: the Hessian $H(x)$ has $\lambda_{\min}(H) < 0$ whenever $x_2 < x_1^2 - 1/600$ (approximately), meaning Newton's method can produce ascent directions in non-convex regions. Numerically, $\approx 43\%$ of the grid $[-1.5,1.5]^2 \times [-1.5,2.5]^2$ is in the non-convex region. The maximum condition number in this domain exceeds $10^6$ (far above the valley, where $\lambda_{\min}$ approaches 0). Starting point: $x_0 = [-1, 1]^T$.

\section{Question 1: Adaptive Step-Size Methods}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}
\label{subsec:q1_algorithms}

Four adaptive methods are compared against constant step-size gradient descent across all three benchmarks for 120 iterations. We denote the gradient at step $k$ by $g_k = \nabla f(x_k)$.

\begin{algorithm}[H]
\caption{Gradient Descent (baseline)}
\label{alg:gd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, target value $f^\star$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $\alpha_k \gets \dfrac{f(x_k) - f^\star}{\|g_k\|^2 + \varepsilon}$
    \Comment{Adaptive step targeting $f^\star$}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $G_{k+1} \gets G_k + g_k \odot g_k$
    \Comment{Cumulative squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
    \Comment{Per-coordinate adaptive step}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$
    \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, g_k$
    \Comment{Accumulated gradient (velocity)}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{Theoretical Convergence Rates and Second-Order Connections}
\label{subsec:q1_theory}

For reference, we state convergence rates for the classical setting of $L$-smooth, $\mu$-strongly convex objectives (Benchmarks A and B satisfy this near the optimum):

\begin{theorem}[GD linear convergence]
\label{thm:gd_rate}
Let $f$ be $\mu$-strongly convex and $L$-smooth. With step size $\alpha = 1/L$:
\begin{equation}
f(x_k) - f^\star \leq \left(1 - \frac{\mu}{L}\right)^k \bigl(f(x_0) - f^\star\bigr).
\end{equation}
\end{theorem}

For Benchmark A ($\kappa \approx 1.12$, $\rho = 1 - \mu/L \approx 0.055$), GD converges in roughly $\log(1/\varepsilon)/\log(1/\rho) \approx 17\log(1/\varepsilon)$ iterations — explaining the rapid convergence seen in Figure~\ref{fig:q1_fval_A}.

\begin{theorem}[Polyak step with exact $f^\star$]
\label{thm:polyak_rate}
Let $f$ be $\mu$-strongly convex and $L$-smooth, and let $f^\star$ be the exact minimum value. Then the Polyak update satisfies $f(x_{k+1}) - f^\star \leq \bigl(1 - \mu/L\bigr)(f(x_k) - f^\star)$.
\end{theorem}

\noindent When $f^\star$ is misspecified (e.g., $f^\star = 0$ when the true minimum is $0.4826$ or $0.7244$), Theorem~\ref{thm:polyak_rate} no longer applies. As $x_k$ approaches the true minimum, $\|g_k\|^2 \to 0$ while $f(x_k) - 0$ remains bounded away from zero, causing $\alpha_k \to \infty$ and inducing oscillation.

\begin{proposition}[Adagrad monotone decay and diagonal preconditioning]
\label{prop:adagrad}
The effective per-coordinate step size $\alpha_0 / (\sqrt{G_k^{(i)}} + \varepsilon)$ is monotonically non-increasing in $k$ for each coordinate $i$. As a consequence, Adagrad is guaranteed not to diverge, but may suffer from premature convergence on non-stationary or ill-conditioned problems.
\end{proposition}

\noindent\textbf{Diagonal preconditioning interpretation.} The Adagrad update can be written as:
\begin{equation}
x_{k+1} = x_k - \underbrace{\operatorname{diag}\!\left(\frac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon}\right)}_{\approx \, \alpha_0 (G_{k+1})^{-1/2}} g_k.
\end{equation}
This is equivalent to preconditioned gradient descent with a diagonal preconditioner $(G_{k+1})^{-1/2}$. Coordinates with historically large gradients (high curvature) receive smaller effective step sizes; coordinates with small gradients (flat directions) receive larger steps. This implicitly adapts to the Hessian's diagonal structure, which is particularly beneficial for Benchmark B where the Hessian is diagonal ($H = \operatorname{diag}(2-\sin(x_1), 10)$) — Adagrad effectively learns separate step sizes for $x_1$ and $x_2$ without explicitly computing the Hessian.

\noindent \textbf{Step size vs theory.} For Benchmark A, the theoretical optimal GD step size is $\alpha^* = 2/(L+\mu) \approx 1.02$, which would give the maximum rate per iteration. The chosen $\alpha = 0.08$ is far more conservative, explaining why GD on Benchmark A still converges quickly ($\kappa \approx 1.12$ is so small that even a very small step is effective). For Benchmark C (Rosenbrock), the stability requirement $\alpha \leq 1/L$ gives $\alpha \leq 1/1001.6 \approx 0.001$, consistent with the chosen $\alpha = 0.0012$. The need for such tiny steps on Benchmark C ($\alpha$ is $\sim 850\times$ smaller than Benchmark A) directly reflects the $850\times$ larger Lipschitz constant.

\noindent \textbf{Optimal Heavy Ball parameters.} For $\mu$-strongly convex, $L$-smooth quadratics, the optimal Heavy Ball parameters are:
\begin{equation}
\alpha^{\text{HB}}_{\text{opt}} = \frac{4}{(\sqrt{L} + \sqrt{\mu})^2}, \qquad \beta^{\text{HB}}_{\text{opt}} = \left(\frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1}\right)^2,
\end{equation}
achieving the optimal rate $\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^k$ per iteration. For Benchmark A ($\kappa \approx 1.12$): $\alpha^* \approx 1.02$, $\beta^* \approx 0.001$ (very small momentum). For Benchmark B ($\kappa = 10$): $\alpha^* \approx 0.23$, $\beta^* \approx 0.27$. The used values ($\alpha = 0.035$--$0.045$, $\beta = 0.88$--$0.90$) use much larger momentum than optimal — a common heuristic choice that works well empirically on non-quadratic problems where the optimal parameters for the local quadratic approximation vary along the trajectory.

\noindent \textbf{Heavy Ball failure on non-quadratic functions.} Despite achieving the optimal rate on quadratics, Heavy Ball is not guaranteed to converge for general non-convex smooth functions. The key reason: the optimal parameters $(\alpha^*, \beta^*)$ above depend on the global constants $L$ and $\mu$, which describe the \emph{quadratic approximation} at the current iterate. For non-quadratic functions like Rosenbrock, the effective $L$ and $\mu$ vary dramatically along the trajectory — at the starting point $(-1,1)$, the local condition number is $\kappa_{\text{local}} \approx 46$, while at the valley floor the effective $\kappa$ can be $>10^3$. The fixed parameters $\beta = 0.86$, $\alpha = 0.0008$ on Benchmark C represent a compromise chosen for average-case behaviour along the trajectory, but are suboptimal at any specific location. This explains why Heavy Ball ($f = 1.40$ at $k=120$) is much slower than Polyak ($f = 0.024$), which adapts its step at every iteration.

\subsubsection{Polyak Step-Size Pathology}

Figure~\ref{fig:q1_polyak_B} provides a detailed view of Polyak's failure mode on Benchmark B. The mechanism is:
\begin{equation}
\alpha_k = \frac{f(x_k) - 0}{\|g_k\|^2 + \varepsilon}.
\end{equation}
When $x_k$ approaches the true minimum (where $f(x_k) \to 0.7244$ and $\|g_k\| \to 0$), the step size satisfies $\alpha_k \approx 0.7244/\varepsilon = 7244$, which is catastrophically large. The iterate is then thrown far from the minimum, the gradient magnitude recovers, and the cycle repeats — producing chaotic oscillation rather than convergence.

\begin{figure}[H]
\centering
\includegraphics[width=0.88\textwidth]{figures/q1_polyak_divergence_B.pdf}

\subsection{Comparative Analysis}
\label{subsec:q1_analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} On the well-conditioned Benchmark A ($\kappa \approx 1.12$), all correctly-specified methods converge rapidly ($\rho \approx 0.055$ per step). Differences between methods are negligible here. On Benchmark C (ill-conditioned, $\kappa \approx 2508$), momentum and adaptive methods provide significant advantages.

\item \textbf{Sensitivity to $f^\star$:} Polyak's performance is binary: optimal when $f^\star$ is exact (best convergence on Benchmark C), pathological when misspecified. This limits its applicability to settings where a good lower bound is available (e.g., non-negative losses with known zero lower bound).

\item \textbf{Coordinate adaptivity:} Adagrad and RMSprop scale each coordinate independently by its historical gradient magnitude, effectively providing a diagonal preconditioner. This is advantageous when coordinates have very different curvatures (Benchmark B: $\lambda_1 \approx 1.45$ vs $\lambda_2 = 10$).

\item \textbf{Stability:} GD baseline is most stable but slowest. Adagrad is uniformly stable (monotonically decreasing step) but can stagnate prematurely. Heavy Ball can exhibit momentum-induced oscillation near the minimum but recovers quickly on smooth objectives.

\item \textbf{Adagrad vs RMSprop:} Both maintain per-coordinate step-size adaptation, but differ in memory: Adagrad accumulates \emph{all} historical squared gradients ($G_k = \sum_{j=0}^{k} g_j \odot g_j$, growing monotonically), while RMSprop uses an exponential moving average ($v_{k+1} = \beta v_k + (1-\beta)g_k^2$, bounded). Adagrad thus eventually shrinks all step sizes to near-zero on long runs; RMSprop maintains a finite effective rate throughout by ``forgetting'' old squared gradients.

\item \textbf{Practical recommendation:} For problems without known $f^\star$, RMSprop and Heavy Ball are robust choices with tunable hyperparameters. Adagrad is reliable for sparse gradients. Polyak is unsuitable unless $f^\star$ is accurately known.
\end{itemize}

\subsection{Algorithm Formulations}
\label{subsec:q2_algorithms}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead / extrapolated point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update using lookahead gradient}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball (Algorithm~\ref{alg:heavyball}) is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. Evaluating the gradient at the extrapolated point allows the algorithm to correct its trajectory \emph{before} overshooting the minimum.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (mean) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (variance) estimate}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$,\quad $\hat{v}_t \gets v_t / (1 - \beta_2^t)$ \Comment{Bias correction}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines Nesterov-like momentum (via the first moment $m_t$, effectively an EMA of gradients) with per-coordinate adaptive scaling (via the second moment $v_t$, an EMA of squared gradients), inheriting advantages of both RMSprop and momentum methods. The bias correction in step 4 prevents early iterates from being biased towards zero (since $m_0 = v_0 = \mathbf{0}$).

\paragraph{Bias correction derivation.} Without bias correction, at $t=1$:
\begin{equation}
m_1 = (1-\beta_1)g_1 = (1-\beta_1)\nabla f(x_1).
\end{equation}
Its expectation is $\mathbb{E}[m_1] = (1-\beta_1)\nabla f(x_1) \neq \nabla f(x_1)$, underestimating the gradient by factor $(1-\beta_1)$. The bias-corrected estimate $\hat{m}_t = m_t/(1-\beta_1^t)$ satisfies $\mathbb{E}[\hat{m}_1] = (1-\beta_1)/(1-\beta_1) \cdot \nabla f(x_1) = \nabla f(x_1)$. For large $t$, $\beta_1^t \to 0$, so $\hat{m}_t \approx m_t$ and the correction is negligible. Similarly for $\hat{v}_t$. The bias correction is most important in the first $\sim 10$ iterations when $\beta_1^t$ is still significant.

\paragraph{Connection between Adam and RMSprop.} Setting $\beta_1 = 0$ (no first-moment memory) in Adam reduces it to a bias-corrected version of RMSprop: $\hat{m}_t = g_t$ (no smoothing) and $x_{t+1} = x_t - \alpha g_t/(\sqrt{\hat{v}_t} + \varepsilon)$. Conversely, RMSprop is the ``no momentum'' limit of Adam. The standard Adam step can also be interpreted as a Newton step in a coordinate system where each coordinate $i$ is normalised by $\sqrt{\hat{v}_t^{(i)}}$, giving an effective preconditioner $\operatorname{diag}(\hat{v}_t)^{-1/2}$ — a rough approximation to the diagonal of the inverse Hessian.

\begin{remark}[Nesterov vs Heavy Ball]
\label{rem:nesterov_hb}
Both Nesterov (Algorithm~\ref{alg:nesterov}) and Heavy Ball (Algorithm~\ref{alg:heavyball}) use momentum, but differ crucially in \emph{where the gradient is evaluated}:
\begin{align}
\text{Heavy Ball:} \quad & x_{k+1} = x_k - \alpha\,\nabla f(x_k) + \beta(x_k - x_{k-1}), \\
\text{Nesterov:} \quad & x_{k+1} = x_k - \alpha\,\nabla f(x_k + \beta_k(x_k - x_{k-1})) + \beta_k(x_k - x_{k-1}).
\end{align}
Nesterov evaluates the gradient at the \emph{lookahead} point $\tilde{x}_k = x_k + \beta_k(x_k - x_{k-1})$, effectively using a ``corrected'' gradient that anticipates where momentum will take the iterate. For convex functions, this lookahead is the key ingredient that achieves the optimal $O(1/k^2)$ rate (Theorem~\ref{thm:nesterov}). Heavy Ball does not have this guarantee for general convex functions, despite being effective in practice on strongly convex quadratics.
\end{remark}

\subsection{Theoretical Convergence Rates}
\label{subsec:q2_theory}

\begin{theorem}[Nesterov optimal rate for convex problems]
\label{thm:nesterov}
For $L$-smooth convex $f$ with the schedule $\beta_k = (k-1)/(k+2)$ and step size $\alpha \leq 1/L$:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2}.
\end{equation}
This $O(1/k^2)$ rate is \textbf{optimal} for first-order methods on convex problems (Nesterov 1983); gradient descent achieves only $O(1/k)$ for convex functions, and $(1-\mu/L)^k$ for strongly convex.
\end{theorem}

\begin{theorem}[Nesterov linear rate for strongly convex problems]
\label{thm:nesterov_sc}
Let $f$ be $\mu$-strongly convex and $L$-smooth. Nesterov's accelerated method with appropriate schedule achieves:
\begin{equation}
f(x_k) - f^\star \leq L\|x_0 - x^\star\|^2 \left(1 - \frac{1}{\sqrt{\kappa}}\right)^k, \qquad \kappa = \frac{L}{\mu}.
\end{equation}
\end{theorem}

\noindent This linear rate with constant $(1 - 1/\sqrt{\kappa})$ is dramatically better than GD's linear rate $(1 - 1/\kappa)$ for large $\kappa$. For Benchmark C ($\kappa \approx 2508$): GD rate $\approx (1 - 1/2508)^k \approx e^{-k/2508}$ (requires $\sim 17300$ iterations to reach $\varepsilon = 10^{-3}$ error); Nesterov rate $\approx (1 - 1/\sqrt{2508})^k \approx e^{-k/50}$ (requires only $\sim 345$ iterations). The ratio $\kappa/\sqrt{\kappa} = \sqrt{\kappa} \approx 50$ is the speedup factor — Nesterov is $\sqrt{\kappa}$-faster than GD in the strongly convex setting. This rate also explains the empirical 600-iteration behavior on Rosenbrock (Figure~\ref{fig:long_horizon}): Nesterov reaches lower objective values than GD in the same number of iterations, consistent with the $\sqrt{\kappa}$-fold speedup prediction.

\begin{theorem}[Lower bound for first-order methods on smooth convex functions, Nesterov 1983]
\label{thm:lower_bound}
There exists an $L$-smooth convex function $f$ such that for any first-order method (i.e., any method whose iterates $x_k$ lie in $\text{span}\{x_0, \nabla f(x_0), \ldots, \nabla f(x_{k-1})\}$), the error satisfies:
\begin{equation}
f(x_k) - f^\star \geq \frac{3L\|x_0 - x^\star\|^2}{32(k+1)^2}.
\end{equation}
\end{theorem}

\noindent Comparing Theorems~\ref{thm:nesterov} and \ref{thm:lower_bound}: Nesterov achieves $2L\|x_0-x^\star\|^2/(k+1)^2$ while the lower bound is $3L\|x_0-x^\star\|^2/(32(k+1)^2)$. The ratio is $64/3 \approx 21$, meaning Nesterov is within a constant factor of the information-theoretic optimum. No first-order method can achieve better than $O(1/k^2)$ convergence on general smooth convex problems, regardless of memory or complexity. This hardness result provides the fundamental motivation for second-order and quasi-Newton methods: to break the $O(1/k^2)$ barrier, one must use curvature information beyond gradients.

In contrast, \textbf{Adam lacks a general convergence guarantee} for non-convex problems. Reddi et al.\ (2018) constructed explicit counterexamples where Adam fails to converge (the AMSGrad variant was proposed as a fix). The key issue is that the exponential averaging in $v_t$ can allow the effective step size $\alpha/\sqrt{\hat{v}_t}$ to \emph{increase} at certain iterations, violating the descent lemma. Despite this, Adam performs well empirically on the benchmarks because the objectives are smooth and the adaptive scaling helps navigate anisotropic curvature.

\begin{remark}[Adam on stochastic vs deterministic problems]
Our benchmarks are evaluated with full-batch gradients (deterministic). In the deterministic setting, Adam's adaptive step sizes provide mainly a preconditioning benefit (the second moment $v_t$ approximates the coordinate-wise gradient magnitude, which correlates with the diagonal of the Hessian). In the stochastic setting, the first moment $m_t$ additionally provides variance reduction through gradient averaging ($\mathbb{E}[\hat{m}_t] \approx \nabla f(x)$). This dual role makes Adam particularly effective for large-scale stochastic problems, though its non-convergence counterexamples arise specifically in the online/stochastic setting (Reddi et al.\ 2018). For our smooth deterministic benchmarks, the convergence is empirically reliable.
\end{remark}

\begin{remark}[Schedule capping and effective Nesterov rate]
\label{rem:nesterov_schedule}
The schedule $\beta_k = \min((k-1)/(k+2), \beta_{\max})$ caps the momentum at $\beta_{\max}$. With $\beta_{\max} = 0.90$, the cap activates at $k \approx 28$ (when $(k-1)/(k+2) \geq 0.90$). For $k < 28$, the uncapped schedule applies and Nesterov's $O(1/k^2)$ rate (Theorem~\ref{thm:nesterov}) holds. For $k \geq 28$, fixed momentum $\beta = 0.90$ applies, which behaves similarly to Heavy Ball — the asymptotic convergence becomes linear rather than $O(1/k^2)$. This is why the cap should be chosen large (close to 1) to delay the transition, or the uncapped schedule used throughout.
\end{remark}

\begin{remark}[Adam hyperparameter choices]
\label{rem:adam_params}
The tuned values $\beta_1 \in \{0.82, 0.80\}$ (versus the typical $\beta_1 = 0.9$) represent a lower first-moment decay, meaning past gradients are down-weighted faster. This is advantageous on Rosenbrock where the gradient direction changes rapidly along the valley — fast forgetting allows Adam to adapt to local curvature changes rather than accumulating momentum in the wrong direction. The standard $\beta_2 = 0.999$ provides stable second-moment estimates across all three benchmarks.
\end{remark}

\begin{enumerate}[nosep]
\item \textbf{The asymptotic regime has not been reached.} The $50\times$ speedup is the \emph{asymptotic} ratio — valid once both methods are near the optimum and the linear convergence rates dominate. At $k=150$, both GD and Nesterov are in the \emph{transient phase}: they have descended to the valley floor but not yet converged to $(1,1)$. Rosenbrock is non-strongly-convex globally (only locally near $(1,1)$), so the $\sqrt{\kappa}$ rate applies only once $\|x_k - x^\star\| < r$ (Theorem~\ref{thm:newton}'s local convergence radius analogy). The long-run comparison (Figure~\ref{fig:long_horizon}) at 600 iterations confirms that the speedup grows with iteration count.
\item \textbf{Nesterov's capped schedule reduces to heavy-ball for $k \geq 28$.} With $\beta_{\max} = 0.90$ and the cap activating at $k \approx 28$ (Remark~\ref{rem:nesterov_schedule}), Nesterov uses fixed momentum $\beta = 0.90$ for iterations 29--150 rather than the optimal $\beta_k = (k-1)/(k+2)$ schedule of Theorem~\ref{thm:nesterov}. Fixed $\beta = 0.90$ is suboptimal — the uncapped schedule would give $\beta_{150} \approx 0.987$, which corresponds to a much more aggressive (and optimal) momentum coefficient. The cap thus limits the practical speedup below the theoretical $50\times$.
\end{enumerate}

\subsection{Mini-Batch Stochastic Gradient Descent}
\label{subsec:q2_sgd}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data randomly permuted at each epoch:

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\sigma \in S_m$
    \For{each mini-batch $\mathcal{B}_j = \{\sigma(jb+1), \ldots, \sigma((j+1)b)\}$}
        \State $\hat{g} \gets \frac{1}{b} X_{\mathcal{B}_j}^T(X_{\mathcal{B}_j}\theta - y_{\mathcal{B}_j})$ \Comment{Mini-batch gradient estimate}
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $\hat{g}$ is an unbiased estimator of the full gradient: $\mathbb{E}[\hat{g}] = \nabla J(\theta)$. The variance satisfies:
\begin{equation}
\operatorname{Var}[\hat{g}] = \frac{1}{b}\operatorname{Var}[g_{\text{single}}] = \frac{\sigma^2_{\text{data}}}{b},
\end{equation}
where $\sigma^2_{\text{data}}$ is the per-sample gradient variance. This gives the fundamental \textbf{bias-variance trade-off} of mini-batch SGD:
\begin{itemize}[nosep]
\item \textbf{Small batch} ($b = 5$): high variance $(\sigma^2_{\text{data}}/5)$, but $m/b = 200$ gradient steps per epoch $\Rightarrow$ faster learning in early epochs, noisier convergence near the minimum.
\item \textbf{Large batch} ($b = 40$): low variance $(\sigma^2_{\text{data}}/40)$, but only $m/b = 25$ steps per epoch $\Rightarrow$ slower initial convergence, but more stable convergence near the minimum.
\end{itemize}
In terms of total gradient computations, both reach the same total work per epoch ($m = 1000$ evaluations). However, small batches make $8\times$ more parameter updates per epoch, exploiting local information more frequently.

\subsubsection{SGD Convergence Theory}

\begin{theorem}[SGD convergence for $L$-smooth objectives, non-convex]
\label{thm:sgd_nonconvex}
Let $f$ be $L$-smooth (not necessarily convex), and suppose the stochastic gradient satisfies $\mathbb{E}[\hat{g}_k] = \nabla f(x_k)$ and $\mathbb{E}[\|\hat{g}_k - \nabla f(x_k)\|^2] \leq \sigma^2/b$. With constant step size $\alpha_k = \alpha = \tfrac{1}{\sqrt{KL}}$ for $K$ total steps:
\begin{equation}
\min_{k=0,\ldots,K-1} \mathbb{E}\!\left[\|\nabla f(x_k)\|^2\right]
\;\leq\; \frac{2\sqrt{L}(f(x_0) - f^\star)}{\sqrt{K}} + \frac{\sigma^2}{\sqrt{KL}\, b} \cdot L
\;=\; O\!\left(\frac{1}{\sqrt{K}} + \frac{\sigma^2/b}{\sqrt{K}}\right).
\end{equation}
\end{theorem}

\noindent This $O(1/\sqrt{K})$ rate for stationarity compares with GD's $O(1/K)$ guarantee for finding approximate stationary points; the gap arises from gradient noise. The key observation is that \emph{mini-batch size $b$ enters the bound additively with $\sigma^2$}: increasing $b$ reduces $\sigma^2/b$ and tightens the bound. In the large-batch limit ($b = m$, full batch), the $\sigma^2/b$ term vanishes and we recover GD's behaviour.

\begin{theorem}[SGD convergence for strongly convex objectives]
\label{thm:sgd_sc}
For $\mu$-strongly convex, $L$-smooth $f$ with unbiased stochastic gradients of variance $\sigma^2/b$, and diminishing step sizes $\alpha_k = \frac{2}{\mu(k+1)}$:
\begin{equation}
\mathbb{E}[f(x_K)] - f^\star \;\leq\; \frac{2L\|x_0 - x^\star\|^2}{K+1} + \frac{2\sigma^2}{\mu b K}.
\end{equation}
The first term is the optimisation error (decays as $O(1/K)$) and the second is the statistical error (decays as $O(1/(bK))$, reduced by mini-batching). The total $O(1/K)$ rate is slower than the linear rate of full-gradient GD but is achieved at cost $O(K \cdot b/m)$ data passes (vs $O(K)$ passes for full GD).
\end{theorem}

With $b = 5$: the statistical error $2\sigma^2/(\mu b K) \approx 2\sigma^2/(\mu \cdot 5K)$ dominates for moderate $K$. For the noise floor $J^\star \approx 0.4826$, the excess statistical error is $\approx \sigma^2_{\text{data}}/(bm)$ per epoch, explaining why both batch sizes converge to the same floor but at different rates. The practical optimum batch size balances parallelism (hardware efficiency) against the statistical efficiency gain from larger batches.

\noindent\textbf{Connection to Adam in the stochastic setting.} Adam's adaptive step size acts as a per-coordinate step-size schedule, with the second moment tracking recent gradient variance. In the stochastic setting, the second moment $v_t \approx \mathbb{E}[g^2]$ (for large $t$) effectively normalises by the coordinate-wise gradient variance, making Adam particularly robust to varying noise levels across coordinates.

\subsection{Local Taylor Approximations of $g(x) = x^4$}
\label{subsec:q3_taylor}

Around the expansion point $x_0 = 0.25$, with $g'(x) = 4x^3$ and $g''(x) = 12x^2$:

\textbf{First-order (linear) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625(x - 0.25).
\end{equation}
This tangent line captures only the local slope. A gradient step $x_{\text{new}} = x_0 - g'(x_0)/\alpha$ minimises a scaled linear model but ignores curvature.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
The Newton update minimises this quadratic model exactly: setting the derivative to zero gives
\begin{equation}
x^* = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667,
\end{equation}
which is much closer to the true minimiser $x^* = 0$ than a gradient step with any fixed step size would achieve.

\subsection{Newton's Method}
\label{subsec:q3_newton}

\begin{algorithm}[H]
\caption{Newton's Method with Damping and Hessian Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Levenberg--Marquardt regularisation}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{Damped Newton step}
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton direction $p_k = H_k^{-1}g_k$ changes coordinates to the natural geometry of the objective: in the rescaled space $\tilde{x} = H_k^{1/2}x$, the step is a unit gradient step. Regularisation $\lambda > 0$ (here $\lambda = 10^{-8}$) ensures $H_k$ is positive definite even when $\nabla^2 f$ is singular or near-singular.

\begin{theorem}[Newton quadratic convergence]
\label{thm:newton}
Let $f$ be twice continuously differentiable with $L_H$-Lipschitz continuous Hessian. There exist constants $r, c > 0$ such that if $\|x_0 - x^\star\| \leq r$ (local convergence region), then the full Newton step ($\alpha = 1$, $\lambda = 0$) satisfies:
\begin{equation}
\|x_{k+1} - x^\star\| \leq c\, \|x_k - x^\star\|^2.
\end{equation}
This \textbf{quadratic convergence} means the number of correct digits approximately doubles at each iteration, compared to the linear rate of gradient descent.
\end{theorem}

\begin{proposition}[Newton is exact in one step for quadratics]
\label{prop:newton_quadratic}
Let $f(\theta) = \tfrac{1}{2}\theta^T H \theta - b^T\theta + c$ be a quadratic with $H \succ 0$ (constant Hessian). Then the Newton step from any starting point $\theta_0$ satisfies:
\begin{equation}
\theta_1 = \theta_0 - H^{-1}\nabla f(\theta_0) = \theta_0 - H^{-1}(H\theta_0 - b) = H^{-1}b = \theta^\star.
\end{equation}
The Newton step reaches the exact minimiser $\theta^\star = H^{-1}b$ in \textbf{one iteration}, regardless of $\theta_0$ or the condition number $\kappa$.
\end{proposition}

This is precisely what occurs on Benchmark A: $J(\theta) = \tfrac{1}{2m}\|X\theta - y\|^2 = \tfrac{1}{2}\theta^T H_A \theta - (X^Ty/m)^T\theta + \text{const}$ with $H_A = X^TX/m$ and $b = X^Ty/m$. Newton's single step solves the normal equations $H_A\theta = b$ exactly, yielding $\hat\theta_{\text{OLS}} = (X^TX)^{-1}X^Ty$.

For Benchmark C (Rosenbrock), Newton achieves \emph{local} quadratic convergence but requires damping ($\alpha < 1$) to remain in the convergence region, since the Hessian varies rapidly along the curved valley ($L_H \propto \max\{1200, 400\}$, very large).

\subsection{Backtracking Line Search (Armijo Condition)}
\label{subsec:q3_armijo}

Both Remark~\ref{rem:damped_newton} (Section~\ref{subsec:q3_params}) and the note after Table~\ref{tab:q3_final} mention that a fixed damping parameter $\alpha$ is used here for simplicity. In practice, Newton's method employs \emph{backtracking line search} to adaptively select $\alpha_k$ at each iteration, guaranteeing global convergence while recovering quadratic convergence near the optimum.

\begin{algorithm}[H]
\caption{Backtracking Line Search (Armijo Condition)}
\label{alg:armijo}
\begin{algorithmic}[1]
\Require Current point $x_k$, descent direction $p_k$ (e.g., Newton direction $-H_k^{-1}g_k$), parameters $c \in (0, 0.5)$, $\tau \in (0, 1)$ (typical: $c = 10^{-4}$, $\tau = 0.5$)
\State $\alpha \gets 1$ \Comment{Start with full Newton step}
\While{$f(x_k + \alpha\, p_k) > f(x_k) + c\,\alpha\, \nabla f(x_k)^T p_k$}
    \State $\alpha \gets \tau \cdot \alpha$ \Comment{Reduce step size by factor $\tau$}
\EndWhile
\State \Return $\alpha_k \gets \alpha$
\end{algorithmic}
\end{algorithm}

The \textbf{Armijo condition} $f(x_k + \alpha p_k) \leq f(x_k) + c\alpha \nabla f(x_k)^T p_k$ requires that the actual decrease be at least a fraction $c$ of the decrease predicted by the linear model. Since $p_k = -H_k^{-1}g_k$ is a descent direction (when $H_k \succ 0$), the directional derivative $\nabla f(x_k)^T p_k = -g_k^T H_k^{-1} g_k < 0$, so the right-hand side is strictly negative and the condition is always satisfiable for small $\alpha$.

\noindent\textbf{Convergence guarantee.} The key properties are:
\begin{itemize}[nosep]
\item \textbf{Global convergence}: Backtracking guarantees that each iteration produces sufficient decrease, preventing divergence even far from $x^\star$.
\item \textbf{Local quadratic recovery}: When $x_k$ is close to $x^\star$, the unit step $\alpha = 1$ satisfies the Armijo condition (since the quadratic model is accurate locally), and the algorithm recovers full Newton quadratic convergence (Theorem~\ref{thm:newton}).
\item \textbf{Cost}: Each backtracking step requires one additional function evaluation $f(x_k + \alpha p_k)$. In practice, for Newton's method near the optimum, $\alpha = 1$ is accepted immediately (zero extra evaluations). Far from the optimum, typically 2--5 reductions are needed.
\end{itemize}

For our benchmarks, a backtracking Newton with $c = 10^{-4}$, $\tau = 0.5$ would: (i) accept $\alpha = 1$ immediately on Benchmark A (exact quadratic, model always accurate); (ii) require 1--2 reductions on Benchmark B (mild non-quadratic); (iii) require 3--5 reductions on Benchmark C near $x_0 = (-1,1)$ but accept $\alpha = 1$ once near $(1,1)$. The fixed $\alpha = 0.22$ used here approximately matches the worst-case step accepted by backtracking on Rosenbrock, at the cost of always using this conservative value even near the optimum.

\subsection{Quasi-Newton Methods: Beyond Exact Hessians}
\label{subsec:q3_quasinewton}

While Newton's method achieves quadratic convergence (Theorem~\ref{thm:newton}), its $O(d^3)$ per-iteration cost (Hessian solve) is prohibitive for large $d$. \textbf{Quasi-Newton methods} approximate the inverse Hessian $H_k^{-1}$ from first-order gradient information alone, achieving \emph{superlinear} convergence at $O(d^2)$ per step.

\subsubsection{BFGS Update Rule}

The Broyden--Fletcher--Goldfarb--Shanno (BFGS) method maintains a positive definite approximation $B_k \approx \nabla^2 f(x_k)$ (or equivalently $W_k = B_k^{-1} \approx H_k^{-1}$), updated at each step using the secant equation:
\begin{equation}
B_{k+1}\, s_k = y_k, \qquad \text{where} \quad s_k = x_{k+1} - x_k,\quad y_k = \nabla f(x_{k+1}) - \nabla f(x_k).
\end{equation}
The secant equation requires that $B_{k+1}$ maps the step direction $s_k$ to the gradient change $y_k$, mimicking the true Hessian's action. The BFGS update to the inverse Hessian approximation $W_k \approx H_k^{-1}$ is:
\begin{equation}
\label{eq:bfgs_update}
W_{k+1} = \left(I - \rho_k s_k y_k^T\right) W_k \left(I - \rho_k y_k s_k^T\right) + \rho_k s_k s_k^T, \qquad \rho_k = \frac{1}{y_k^T s_k}.
\end{equation}
The curvature condition $y_k^T s_k > 0$ ensures $\rho_k > 0$ and $W_{k+1} \succ 0$. This condition is guaranteed by the \emph{Wolfe conditions} (stronger than Armijo, Algorithm~\ref{alg:armijo}):
\begin{equation}
\text{Sufficient decrease: } f(x_k + \alpha p_k) \leq f(x_k) + c_1 \alpha \nabla f(x_k)^T p_k, \quad
\text{Curvature: } \nabla f(x_k + \alpha p_k)^T p_k \geq c_2 \nabla f(x_k)^T p_k,
\end{equation}
with $0 < c_1 < c_2 < 1$ (typically $c_1 = 10^{-4}$, $c_2 = 0.9$). The curvature condition prevents the step size from being too small and guarantees $y_k^T s_k = [\nabla f(x_{k+1}) - \nabla f(x_k)]^T s_k \geq (c_2 - 1)\nabla f(x_k)^T p_k > 0$ (since $\nabla f(x_k)^T p_k < 0$ for a descent direction). Without the Wolfe curvature condition (pure Armijo backtracking), the BFGS update can become ill-conditioned. The BFGS direction is $p_k = -W_k \nabla f(x_k)$, computed at $O(d^2)$ cost.

\begin{theorem}[BFGS superlinear convergence]
\label{thm:bfgs}
Let $f$ be twice continuously differentiable with Lipschitz continuous Hessian at $x^\star$, and let $\nabla^2 f(x^\star) \succ 0$. Then BFGS with exact line search, started sufficiently near $x^\star$, converges \emph{superlinearly}:
\begin{equation}
\lim_{k\to\infty} \frac{\|x_{k+1} - x^\star\|}{\|x_k - x^\star\|} = 0.
\end{equation}
That is, the ratio $\|x_{k+1}-x^\star\|/\|x_k-x^\star\| \to 0$, which is faster than linear convergence but slower than Newton's quadratic convergence ($\|x_{k+1}-x^\star\| \sim c\|x_k-x^\star\|^2$).
\end{theorem}

\subsubsection{Limited-Memory BFGS (L-BFGS)}

Standard BFGS stores the full $d \times d$ matrix $W_k$, requiring $O(d^2)$ memory. For $d \gg 1$, this is impractical. \textbf{L-BFGS} stores only the $m_L$ most recent curvature pairs $\{(s_j, y_j)\}_{j=k-m_L}^{k-1}$ (typically $m_L = 5$--$20$), using them to implicitly represent $W_k$:

\begin{algorithm}[H]
\caption{L-BFGS Two-Loop Recursion (compute $W_k g_k$ without forming $W_k$)}
\label{alg:lbfgs}
\begin{algorithmic}[1]
\Require Gradient $q \gets g_k = \nabla f(x_k)$, curvature pairs $\{(s_j, y_j)\}_{j=k-m_L}^{k-1}$, initial $W^0 = \gamma_k I$
\For{$j = k-1, k-2, \ldots, k-m_L$} \Comment{Backward pass}
    \State $\rho_j \gets 1/(y_j^T s_j)$,\quad $\alpha_j \gets \rho_j s_j^T q$
    \State $q \gets q - \alpha_j y_j$
\EndFor
\State $r \gets W^0 q$ \Comment{Scale by initial Hessian approximation}
\For{$j = k-m_L, \ldots, k-1$} \Comment{Forward pass}
    \State $\beta_j \gets \rho_j y_j^T r$
    \State $r \gets r + s_j(\alpha_j - \beta_j)$
\EndFor
\State \Return $p_k \gets -r$ \Comment{L-BFGS direction, approximating $-W_k g_k$}
\end{algorithmic}
\end{algorithm}

The two-loop recursion computes $W_k g_k$ in $O(m_L d)$ operations without forming $W_k$ explicitly, giving L-BFGS $O(m_L d)$ memory and $O(m_L d)$ per-step cost — vastly superior to Newton's $O(d^2)$ memory and $O(d^3)$ solve.

\begin{remark}[Damped vs full Newton convergence]
\label{rem:damped_newton}
Theorem~\ref{thm:newton} (quadratic convergence) applies to the \emph{undamped} Newton step ($\alpha = 1$). This is empirically confirmed: the code's $\alpha = 0.85$ on Benchmark B yields $\rho \approx 0.15$ per iteration (Figure~\ref{fig:q3_newton_phases}), while $\alpha=1.0$ achieves the quadratic constant $c \approx 0.25$ (each iteration roughly doubles the number of accurate digits). \end{remark}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}
\label{subsec:q4_partA}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of partial derivatives:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \qquad i = 1, \ldots, d.
\end{equation}
This requires $d+1$ function evaluations per iteration. The approximation error decomposes as:
\begin{equation}
\left|\frac{\partial f}{\partial x_i}(x) - \frac{f(x+\delta e_i) - f(x)}{\delta}\right| = \underbrace{\frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(\xi)\right|}_{\text{truncation error }O(\delta)} + \underbrace{\frac{2\varepsilon_{\text{mach}}}{\delta}\|f\|}_{\text{round-off error }O(\varepsilon_{\text{mach}}/\delta)},
\end{equation}
where $\xi$ is an intermediate point. Balancing these gives the optimal step $\delta_{\text{opt}} \sim \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ for double-precision arithmetic. The two test values $\delta = 0.05$ (good) and $\delta = 0.8$ (poor) both fall in the truncation-dominated regime.

\begin{remark}[Central vs forward differences]
The \emph{central difference} approximation, $\frac{f(x+\delta e_i) - f(x-\delta e_i)}{2\delta}$, has truncation error $O(\delta^2)$ (vs $O(\delta)$ for forward differences) but requires $2d$ function evaluations per step (vs $d+1$ for forward). For the test values $\delta = 0.05$ and $\delta = 0.8$, central differences would achieve errors of $O(\delta^2) \approx 0.0025$ and $0.64$, respectively — significantly better than forward differences for $\delta = 0.05$ but not $\delta = 0.8$. The forward difference is used here for simplicity and to minimise function evaluations.
\end{remark}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \text{Uniform}(\mathbb{S}^{d-1})$ \Comment{Random unit direction}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method uses only \emph{two} function evaluations per iteration regardless of dimension $d$, making it dimension-independent in \emph{cost per step}. The estimator is biased but consistent as $\delta \to 0$: $\mathbb{E}[u_k u_k^T] = I/d$, so $\mathbb{E}[\hat{g}_k] \approx \frac{1}{d}\nabla f(x_k)$ (scaled gradient). The effective step size is therefore $\alpha/d$.

\begin{theorem}[NRS convergence rate, Nesterov 2011]
\label{thm:nrs}
For $L$-smooth convex $f$ on $\mathbb{R}^d$, the NRS method with $\delta \to 0$ achieves:
\begin{equation}
\mathbb{E}[f(x_k)] - f^\star \leq \frac{Cd^2 L \|x_0 - x^\star\|^2}{k},
\end{equation}
for a universal constant $C$. Compared to first-order GD's $O(L\|x_0-x^\star\|^2/k)$ rate, NRS suffers a $d^2$ penalty factor. In $d=2$ dimensions, this is a $4\times$ slowdown; in $d=1000$, it is $10^6\times$.
\end{theorem}

This dimension dependence explains why NRS (220 iterations to approach the Benchmark B minimum) is much slower than GD (120 iterations) despite using the same total function evaluations for $d=2$: $220 \times 2 = 440$ vs $120 \times (2+1) = 360$.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): move the worst vertex $x_{\text{worst}}$ through the centroid $c = \frac{1}{n}\sum_{i=1}^{n} x_i$ (excluding worst): $x_r = c + \alpha_r(c - x_{\text{worst}})$.
\item \textbf{Expansion} ($\gamma = 2.0$): if reflection improved, try $x_e = c + \gamma(c - x_{\text{worst}})$.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection did not improve, contract towards centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, shrink all vertices towards the best.
\end{enumerate}
Settings: initial simplex step $= 0.35$, 160 iterations.

The simplex initially takes large exploratory steps, then contracts as it localises the minimum. Right: the best objective value decreases steadily, with the characteristic step-wise pattern of simplex methods (improvement only occurs when a vertex is updated).}

\noindent\textbf{Nelder--Mead convergence properties.} Unlike gradient methods, Nelder--Mead has limited theoretical convergence guarantees. In general, it is not guaranteed to converge to a local minimiser — there exist examples (McKinnon 1998) where Nelder--Mead converges to a non-stationary point. However, in practice it is effective for low-dimensional problems ($d \leq 10$) with smooth objectives. On Rosenbrock specifically, the narrow curved valley means the simplex must become very elongated to probe the valley efficiently. The standard parameters ($\alpha_r=1, \gamma=2, \rho=0.5, \sigma=0.5$) are generally robust, and the method's lack of gradient requirement makes it invaluable for noisy or black-box objectives where finite differences would be unreliable.

\noindent The key trade-offs are: (1) methods requiring gradient information achieve higher accuracy per iteration but are inapplicable to black-box functions; (2) Nesterov random search scales to high dimensions (FE per iteration $= 2$ regardless of $d$) but requires many iterations; (3) grid search is the only method that guarantees global optimum discovery, but at exponential computational cost.

\noindent\textbf{Grid search complexity bound.} For a function defined on $\mathcal{X} = [l_1, u_1] \times \cdots \times [l_d, u_d]$ with volume $V = \prod_i (u_i - l_i)$ and assumed to be $L$-Lipschitz, a uniform grid with spacing $\Delta x$ in each dimension guarantees:
\begin{equation}
f(x_{\text{best}}) - f^\star \leq L\sqrt{d}\,\Delta x, \qquad N_{\text{eval}} = \left\lceil \frac{V}{\Delta x^d} \right\rceil.
\end{equation}
To achieve error $\varepsilon$ with Lipschitz constant $L$, grid spacing $\Delta x = \varepsilon/(L\sqrt{d})$ is needed, requiring $N_{\text{eval}} = O\bigl((L\sqrt{d}/\varepsilon)^d\bigr)$ evaluations. This \emph{curse of dimensionality} ($\exp(d)$ evaluations) makes grid search impractical for $d \geq 5$. For our $d=2$ benchmark with $55 \times 55 = 3025$ evaluations and $\Delta x \approx 0.073$, the Rosenbrock's Lipschitz constant near $(1,1)$ is $\|[\nabla f]\|_{\max} \approx 200$, giving an error bound $\varepsilon \leq 200 \times 0.073\sqrt{2} \approx 20.7$ — consistent with the grid finding a point with $f \leq O(1)$ but not the true minimum.

\noindent\textbf{Connecting Q3 and Q4: Finite-difference Newton's method.} Newton's method (Section~\ref{sec:q3}) requires the Hessian $H_k = \nabla^2 f(x_k)$, which for black-box functions can be approximated by finite differences. The symmetric finite-difference Hessian approximation uses $d(d+1)/2 + d + 1$ function evaluations:
\begin{equation}
\label{eq:fd_hessian}
\frac{\partial^2 f}{\partial x_i \partial x_j}(x) \approx \frac{f(x + \delta e_i + \delta e_j) - f(x + \delta e_i) - f(x + \delta e_j) + f(x)}{\delta^2} \quad (i \neq j),
\end{equation}
with diagonal entries $\partial^2 f/\partial x_i^2 \approx (f(x + \delta e_i) - 2f(x) + f(x - \delta e_i))/\delta^2$. For $d=2$, this requires only 6 function evaluations for the full $2\times 2$ Hessian. The truncation error is $O(\delta)$ for off-diagonal entries and $O(\delta^2)$ for diagonal entries (central differences), with optimal $\delta_{\text{opt}} \approx \varepsilon_{\text{mach}}^{1/3} \approx 10^{-5}$ balancing truncation and round-off. For $d \gg 1$, however, the cost grows as $O(d^2)$ function evaluations per Newton step, making FD-Newton impractical (e.g., $d = 100$: $5151$ evaluations just for the Hessian). This motivates Nesterov random search (only 2 FE per step) and quasi-Newton methods (which avoid explicit Hessian formation entirely).

\noindent\textbf{Beyond grid search: Bayesian optimisation.} For black-box problems where the function is expensive to evaluate and $d \leq 20$, \emph{Bayesian optimisation} (BO) provides a principled alternative. BO maintains a probabilistic surrogate model (typically a Gaussian process) $p(f | \{(x_i, f(x_i))\})$ and selects the next evaluation point by maximising an \emph{acquisition function} (e.g., expected improvement) that balances exploration and exploitation. BO achieves $O(k^{-1/d})$ convergence rate on smooth objectives with $k$ evaluations — exponentially better than grid search's guarantee, though without the hard global optimum guarantee. For our $d=2$ problems, BO would find the global optimum in $\sim 50$ evaluations (vs $55^2 = 3025$ for grid search).

\section{Question 5: Constrained Optimisation}
\label{sec:q5}
% ============================================================

\subsection{Problem Setup}
\label{subsec:q5_setup}

Minimise Benchmark B (Equation~\ref{eq:bench_B}) subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)^T$:
\begin{equation}
\min_{x \in \mathcal{C}} f(x_1, x_2) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1), \quad \mathcal{C} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}.
\end{equation}
Since the unconstrained optimum $(0.582, 2.0)$ satisfies $x_1^\star \approx 0.582 > 0.5$, it lies \emph{inside} the feasible set $\mathcal{C}$. Therefore the constrained and unconstrained optima coincide, and the constraint is \textbf{inactive} at optimality. This has important implications: both Projected GD and the penalty method should converge to the same point $(0.582, 2.0)$ with $f^\star \approx 0.7244$.

\subsection{KKT Conditions}
\label{subsec:q5_kkt}

The constrained problem can be written in standard form as $\min_x f(x)$ subject to $g(x) \leq 0$ where $g(x) = 0.5 - x_1$ (i.e., $x_1 \geq 0.5$ is equivalent to $0.5 - x_1 \leq 0$). The Lagrangian is:
\begin{equation}
\mathcal{L}(x, \mu) = f(x) + \mu\,(0.5 - x_1), \qquad \mu \geq 0.
\end{equation}

\begin{theorem}[KKT Necessary Conditions]
If $x^\star$ is a local minimiser and the LICQ constraint qualification holds, then there exists $\mu^\star \geq 0$ such that:
\begin{align}
\nabla_x \mathcal{L}(x^\star, \mu^\star) &= \nabla f(x^\star) - \mu^\star e_1 = 0, \label{eq:kkt_stat} \\
0.5 - x_1^\star &\leq 0, \label{eq:kkt_primal} \\
\mu^\star &\geq 0, \label{eq:kkt_dual} \\
\mu^\star\,(0.5 - x_1^\star) &= 0. \label{eq:kkt_compl}
\end{align}
\end{theorem}

\noindent Since $x_1^\star \approx 0.582 > 0.5$, complementary slackness~\eqref{eq:kkt_compl} gives $\mu^\star = 0$, and stationarity~\eqref{eq:kkt_stat} reduces to $\nabla f(x^\star) = 0$ (unconstrained stationarity). The constraint is inactive, confirming that the constrained and unconstrained optima coincide. This provides a theoretical basis for both algorithms converging to $f^\star \approx 0.7244$.

\subsection{Algorithm Descriptions}
\label{subsec:q5_algorithms}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection $\Pi_{\mathcal{C}}(z) = (\max(0.5, z_1), z_2)^T$ is the nearest point in $\mathcal{C}$ to $z$. For the half-space $\mathcal{C} = \{x : x_1 \geq 0.5\}$, this is an $O(1)$ operation. The key property is that Projected GD satisfies $x_k \in \mathcal{C}$ for all $k \geq 1$ — feasibility is a hard constraint.

\begin{theorem}[Projected GD convergence]
\label{thm:pgd}
Let $f$ be $\mu$-strongly convex and $L$-smooth on a closed convex set $\mathcal{C}$. With step size $\alpha \leq 1/L$ and starting from $x_0 \in \mathcal{C}$:
\begin{equation}
\|x_k - x^\star\|^2 \leq (1 - 2\alpha\mu)^k \|x_0 - x^\star\|^2.
\end{equation}
This achieves the same linear rate as unconstrained GD when the constraint is active, and when the constraint is inactive (as in our case, $x_1^\star \approx 0.582 > 0.5$), projected GD converges at the same rate as unconstrained GD after the first step.
\end{theorem}

For our problem with $L = 10$, $\mu = 1$ (Benchmark B, Section~\ref{subsec:bench_B}), and $\alpha = 0.08 \leq 1/10 = 0.1$: convergence rate $\rho = 1 - 2\times 0.08 \times 1 = 0.84$. Since the constraint is inactive ($x_1^\star > 0.5$), projected GD behaves like unconstrained GD after the first projection step clips $x_1$ to $0.5$ at step 1.

\begin{algorithm}[H]
\caption{Penalty Method (Quadratic Penalty)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $\nabla F_\lambda(x) \gets \nabla f(x) - \lambda\,\mathbf{1}[x_1 < 0.5]\, e_1$ \Comment{Subgradient}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method converts the constrained problem to an unconstrained one by adding a term that penalises constraint violation. As $\lambda \to \infty$, the solution of the penalised problem converges to the constrained optimum (exact penalty theorem for convex constraints).

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\noindent\textbf{Projected GD} enforces feasibility \emph{at every iteration} as a hard guarantee. The cost is a projection oracle, which for $\mathcal{C} = \{x : x_1 \geq 0.5\}$ is trivial (a single $\max$ operation) but can be expensive for complex constraint sets (e.g., $O(n\log n)$ for simplex projection).

\noindent\textbf{Penalty method} converts the problem to an unconstrained one, eliminating the need for a projection oracle. However: (i)~feasibility is only approximate (unless $\lambda \to \infty$); (ii)~large $\lambda$ creates a poorly-conditioned objective, requiring smaller step sizes (here, $\alpha$ decreases from $0.05$ to $0.03$ as $\lambda$ increases from $1.8$ to $4.5$); (iii)~the optimal $\lambda$ must be tuned. \noindent\textbf{Connection to KKT:} As $\lambda \to \infty$, the penalty method's solution approaches the KKT point identified in Section~\ref{subsec:q5_kkt}: $(0.582, 2.0)$ with $\mu^\star = 0$ (inactive constraint). \noindent\textbf{Ill-conditioning from large penalty.} The penalised objective $F_\lambda(x) = f(x) + \lambda\max(0, 0.5-x_1)$ has a discontinuous gradient at $x_1 = 0.5$. Near the constraint boundary, the effective Lipschitz constant of $F_\lambda$ is $L + \lambda$, requiring $\alpha \leq 1/(L + \lambda)$ for gradient descent stability. This explains why we use $\alpha = 0.03$ for $\lambda = 4.5$ (vs $\alpha = 0.05$ for $\lambda \leq 1.8$): the step size must decrease roughly as $1/\lambda$ as the penalty grows.

\subsection{Augmented Lagrangian Method (Multiplier Method)}
\label{subsec:q5_alm}

The pure penalty method requires $\lambda \to \infty$ to achieve feasibility exactly, causing progressive ill-conditioning. The \emph{Augmented Lagrangian Method} (ALM, also called the Multiplier Method or ADMM in distributed settings) overcomes this by combining the penalty with an explicit dual variable:

\begin{equation}
\mathcal{L}_\rho(x, \mu) = f(x) + \mu\,g(x) + \tfrac{\rho}{2}\bigl[g(x)\bigr]^2, \qquad g(x) = \max(0,\; 0.5 - x_1) \geq 0.
\end{equation}

\begin{algorithm}[H]
\caption{Augmented Lagrangian Method (Method of Multipliers)}
\label{alg:alm}
\begin{algorithmic}[1]
\Require Initial $x_0$, dual variable $\mu_0 = 0$, penalty $\rho > 0$, inner iterations $N_{\text{in}}$
\For{$k = 0, 1, 2, \ldots$}
    \State \textbf{Primal update:} $x_{k+1} \gets \arg\min_x \mathcal{L}_\rho(x, \mu_k)$ \Comment{(Inner loop: gradient steps on $\mathcal{L}_\rho(\cdot, \mu_k)$)}
    \State \textbf{Dual update:} $\mu_{k+1} \gets \mu_k + \rho\, g(x_{k+1})$ \Comment{(Dual ascent step)}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[ALM convergence and exact penalty]
\label{thm:alm}
For the constrained problem $\min_x f(x)$ subject to $g(x) \leq 0$ with $f$ convex and $g$ convex, ALM with fixed penalty $\rho > 0$ converges to the constrained optimum $x^\star$ with $\mu_k \to \mu^\star$ (the optimal dual variable). Crucially, for any $\rho > 0$, the augmented Lagrangian $\mathcal{L}_\rho(\cdot, \mu^\star)$ is an \emph{exact penalty}: its unconstrained minimum is the constrained solution $x^\star$, eliminating the need for $\rho \to \infty$.
\end{theorem}

\noindent\textbf{Comparison of penalty approaches for our Q5 problem.}

For our problem with inactive constraint ($x_1^\star \approx 0.582 > 0.5$, so $\mu^\star = 0$), all three approaches (penalty, ALM, projected GD) converge to the same unconstrained minimum. The differences become important when constraints are active:

\noindent The ALM dual update $\mu_{k+1} = \mu_k + \rho g(x_{k+1})$ acts as an integral controller: it accumulates the constraint violation over iterations, automatically increasing the effective penalty pressure until the constraint is satisfied. At convergence, $g(x^\star) = 0$, so $\mu_k$ stabilises at $\mu^\star$. This is analogous to proportional-integral (PI) control in control theory.

\noindent For our Q5 problem, $\mu^\star = 0$ (inactive constraint), so the dual update adds no net pressure — ALM reduces to the penalty method in this special case. To observe ALM's advantage over pure penalty, one would need to modify the constraint to force $x_1^\star = 0.5$ (e.g., constrain $x_1 \geq 0.7$, making the constraint active).

\subsection{Feasible Set and Linear Programme}
\label{subsec:q6_lp}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (the box from Section~\ref{sec:benchmarks}).

\medskip
\noindent\textbf{Linear Programme:} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$. Since $f$ is linear, the minimum is attained at a vertex of $\mathcal{X}$:

\begin{theorem}[Fundamental Theorem of Linear Programming]
The minimum (and maximum) of a linear objective over a non-empty bounded polytope is attained at a vertex of the polytope.
\end{theorem}

Since both components of $a = [1, 2]^T$ are positive, the objective $f = x_1 + 2x_2$ is minimised by taking both $x_1$ and $x_2$ as small as possible:
\begin{equation}
x^\star = (0.5,\, -5), \qquad f^\star = 1 \times 0.5 + 2 \times (-5) = -9.5.
\end{equation}

\subsection{Frank--Wolfe Algorithm}
\label{subsec:q6_fw}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$
    \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
    \Comment{Convex combination $\Rightarrow$ $x_{k+1} \in \mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

The LMO with the fixed mixing parameter $\beta$ corresponds to a step of size $(1-\beta)$ towards the extreme point of $\mathcal{X}$ in the steepest descent direction. Since $x_k, z_k \in \mathcal{X}$ and $\mathcal{X}$ is convex, $x_{k+1} \in \mathcal{X}$ automatically — no projection is needed.

For the box $\mathcal{X}$, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}, \qquad i = 1, \ldots, d.
\end{equation}
This requires only $O(d)$ operations per iteration — no matrix factorisation or projection onto a complex constraint set.

\subsubsection{Frank--Wolfe Gap and Convergence}

\begin{theorem}[Frank--Wolfe $O(1/k)$ convergence]
\label{thm:fw}
For $L$-smooth convex $f$ over a convex compact $\mathcal{X}$ with diameter $D = \max_{x,y\in\mathcal{X}}\|x-y\|$, using the standard step size $\gamma_k = 2/(k+2)$:
\begin{equation}
f(x_k) - f^\star \leq \frac{2LD^2}{k+2}.
\end{equation}
\end{theorem}

The \textbf{Frank--Wolfe gap} $\Delta_k = \nabla f(x_k)^T(x_k - z_k) \geq 0$ provides a computable upper bound on $f(x_k) - f^\star$ (since $f(x_k) - f^\star \leq \nabla f(x_k)^T(x_k - x^\star) \leq \Delta_k$ for convex $f$). The gap goes to zero as convergence is achieved.

In our implementation, a fixed $\beta$ (rather than the diminishing schedule $\gamma_k = 2/(k+2)$) is used, corresponding to fixed step size $1-\beta$ towards the LP vertex. The standard update $x_{k+1} = (1-\gamma_k)x_k + \gamma_k z_k$ with $\gamma_k = 2/(k+2)$ satisfies Theorem~\ref{thm:fw}. The fixed-$\beta$ variant can be interpreted as fixed step size $\gamma = 1-\beta$, which does not decay and thus lacks the $O(1/k)$ convergence guarantee. However, for well-tuned $\beta$, fixed step size often achieves better empirical performance in the first few hundred iterations. The relation between our $\beta$ and the standard step: $\gamma = 1 - \beta = 0.10$ (for $\beta=0.90$) and $\gamma = 0.015$ (for $\beta=0.985$); the standard diminishing schedule reaches $\gamma_k = 0.10$ at $k = 18$ and $\gamma_k = 0.015$ at $k = 131$.

\subsection{Interior vs Boundary Behaviour}
\label{subsec:q6_analysis}

\textbf{Interior optimum:} The LP oracle $z_k$ always selects a vertex of $\mathcal{X}$, pulling the iterate towards an extreme point. When the optimum is interior, iterates must converge to the optimum as a convex combination of many vertex oracle outputs, producing a characteristic ``zig-zag'' pattern. The theoretical $O(1/k)$ rate of Theorem~\ref{thm:fw} applies with the standard diminishing schedule; with fixed $\beta$, the rate depends on how far $z_k$ oscillates.

\textbf{Boundary optimum:} Frank--Wolfe is particularly well-suited to boundary optima because the LP oracle $z_k$ naturally selects extreme points adjacent to the constrained optimum. Iterates approach the boundary without explicit projection — a major advantage over projected GD when projection is expensive. The fixed $\beta$ schedule leads to residual oscillation near the boundary; a diminishing schedule $\gamma_k = 2/(k+2)$ (as in Theorem~\ref{thm:fw}) would provide the $O(1/k)$ convergence guarantee.

\textbf{Advantages of Frank--Wolfe over Projected GD:}
\begin{itemize}[nosep]
\item No projection oracle needed — only an LP oracle over $\mathcal{X}$ (cost $O(d)$ for box constraints)
\item Iterates are automatically feasible (convex combination of feasible points)
\item Applicable when $\mathcal{X}$ has structure that makes LP easy but projection hard (e.g., nuclear norm balls, spectrahedra, flow polytopes where LP is polynomial but projection is expensive)
\item Sparse iterates: after $k$ iterations, $x_k$ is a convex combination of at most $k+1$ vertices of $\mathcal{X}$, promoting sparsity in the solution
\item The FW gap $\Delta_k$ provides a computable optimality certificate (Figure~\ref{fig:q6_fw_gap})
\end{itemize}

\textbf{Limitations of Frank--Wolfe:}
\begin{itemize}[nosep]
\item Convergence rate $O(1/k)$ is slower than projected GD's $O(1/k^2)$ (with optimal step sizes) or linear rate for strongly convex objectives
\item The fixed $\beta$ schedule (as used here) does not satisfy the theoretical guarantee of Theorem~\ref{thm:fw}; the standard $\gamma_k = 2/(k+2)$ schedule is needed
\item Poor performance for well-conditioned problems or when projection is cheap — projected GD is preferred in such cases
\end{itemize}

\subsection{Away-Steps Frank--Wolfe}
\label{subsec:q6_awaysteps}

Standard Frank--Wolfe has a well-known weakness for interior optima: iterates can only move \emph{towards} extreme points (vertices) of $\mathcal{X}$, never \emph{away} from vertices that have accumulated too much weight. The \textbf{Away-Steps FW} variant (Wolfe 1970, Lacoste-Julien \& Jaggi 2015) adds an ``away step'' that moves away from the vertex contributing most to the current iterate:

\begin{equation}
\text{Away direction:} \quad d_k^{\text{away}} = x_k - v_k, \quad \text{where} \quad v_k = \arg\max_{v \in S_k} \nabla f(x_k)^T v,
\end{equation}
with $S_k$ the \emph{active set} of vertices with positive weights in the current iterate's convex combination representation. The iteration uses either the standard FW step (towards $z_k$) or the away step (away from $v_k$), whichever decreases the objective more.

\begin{theorem}[Away-Steps FW linear convergence for strongly convex objectives]
For $\mu$-strongly convex, $L$-smooth $f$ over a polytope $\mathcal{X}$, Away-Steps FW with exact line search achieves:
\begin{equation}
f(x_k) - f^\star \leq \left(1 - \frac{\mu}{C_{\mathcal{X}}^2 L}\right)^k (f(x_0) - f^\star),
\end{equation}
where $C_{\mathcal{X}}$ is the \emph{pyramidal width} of $\mathcal{X}$ (a geometric constant depending on the polytope). This is \emph{linear} convergence — dramatically faster than standard FW's $O(1/k)$.
\end{theorem}

\noindent For the box $\mathcal{X}$ in our problem, $C_{\mathcal{X}} = \Theta(D/\sqrt{n})$ (where $n$ is the number of vertices), and the linear rate is comparable to projected GD's. The trade-off is computational: Away-Steps FW requires maintaining and updating the active set $S_k$ (at $O(|S_k| d)$ cost), though $|S_k|$ is bounded by $\min(k, n)$ where $n$ is the number of vertices of $\mathcal{X}$. For our $2d$ box with 4 vertices, this overhead is negligible.

For our $d=2$, $n=1000$ problems, Newton's $O(d^3) = O(8)$ solve cost is negligible. In high dimensions ($d \gg 1$), the cubic cost of solving $H^{-1}g$ becomes the bottleneck, motivating quasi-Newton methods (L-BFGS) that approximate $H^{-1}$ with $O(d)$ storage and $O(d^2)$ per step. In particular:
\begin{itemize}[nosep]
\item \textbf{At $d = 1000$}: Newton costs $O(10^9)$ per step vs $O(10^3)$ for GD — a million-fold overhead.
\item \textbf{When derivatives are unavailable}: Finite difference Newton ($d(d+1)/2 + d$ function evaluations for Hessian) is extremely expensive, motivating derivative-free methods (Q4).
\item \textbf{When Hessian is indefinite} (non-convex regions, saddle points): Newton's direction may ascend rather than descend; Levenberg--Marquardt regularisation ($\lambda I$) ensures positive definiteness but at the cost of taking a gradient step rather than Newton step when $\lambda$ is large.
\end{itemize}

\subsection{Key Findings}

\begin{enumerate}
\item \textbf{Condition number is the primary difficulty driver.} Benchmark A ($\kappa \approx 1.12$) is trivial for all methods. Benchmark B ($\kappa \approx 10$) reveals meaningful differences. Benchmark C ($\kappa \approx 2508$, non-convex) is the true challenge where algorithmic properties matter most.

\item \textbf{Polyak step size is uniquely sensitive to $f^\star$ specification.} When $f^\star$ is exact (Benchmark C), Polyak achieves the best convergence among adaptive step-size methods. When misspecified, it oscillates chaotically with step sizes reaching $\mathcal{O}(f^\star_{\text{true}} / \varepsilon) \sim 10^3$--$10^4$.

\item \textbf{Nesterov acceleration is the best first-order method on Benchmark C.} The $O(1/k^2)$ rate (Theorem~\ref{thm:nesterov}) for convex functions — with the stronger $(1-1/\sqrt{\kappa})^k$ linear rate for strongly convex (Theorem~\ref{thm:nesterov_sc}) — and the lookahead gradient evaluation combine to make Nesterov the clear winner on the ill-conditioned Rosenbrock, reaching $f = 0.479$ vs GD's $f = 3.388$ (empirical $7\times$ speedup, with theoretical asymptotic speedup of $\sqrt{\kappa} \approx 50\times$ once in the linear convergence regime).

\item \textbf{Newton's method provides dramatic speedup on smooth objectives.} On the quadratic Benchmark A, Newton converges in one iteration. On Benchmark C with damping $\alpha = 0.22$, Newton (20 iterations) is $5.4\times$ better than GD (80 iterations). The computational cost per Newton step ($O(d^3)$ for Hessian solve vs $O(d)$ for GD) is justified for small-$d$ problems with high curvature.

\item \textbf{Derivative-free methods are essential for black-box functions.} Nesterov random search achieves rate $O(d^2 L \|x_0-x^\star\|^2/k)$ for smooth convex objectives (Theorem~\ref{thm:nrs}), using only 2 function evaluations per step but suffering a $d^2$ factor vs first-order GD. Nelder--Mead adapts to local landscape geometry without gradients. Grid search guarantees global optimum discovery at exponential cost $O(N^d)$.

\item \textbf{Mini-batch SGD offers a data-efficient convergence-noise tradeoff.} For $\mu$-strongly convex objectives with per-sample gradient variance $\sigma^2$, mini-batch SGD achieves $O(1/K) + O(\sigma^2/(bK))$ error after $K$ steps with batch size $b$ (Theorem~\ref{thm:sgd_sc}). The $\sigma^2/b$ noise term shows that doubling the batch halves the statistical error, but $m/b$ steps per epoch means the total function evaluation budget per epoch is identical regardless of $b$. This makes batch size a tradeoff between step quality and update frequency.

\item \textbf{Projected GD enforces hard feasibility; penalty methods and ALM offer soft alternatives.} When the constraint is inactive at optimality (as in Benchmark B with $x_1^\star \approx 0.582 > 0.5$), all three approaches (projected GD, penalty, ALM) converge to the same point. The Augmented Lagrangian Method (Section~\ref{subsec:q5_alm}) improves over pure penalty by maintaining a dual variable that accumulates constraint violation, enabling exact convergence at moderate penalty weight $\rho$ without the ill-conditioning of $\lambda \to \infty$.

\item \textbf{Frank--Wolfe's fixed-$\beta$ implementation is suboptimal vs the theoretically guaranteed schedule.} The standard $\gamma_k = 2/(k+2)$ schedule achieves $O(1/k)$ convergence (Theorem~\ref{thm:fw}); the fixed-$\beta$ code achieves the same asymptotic accuracy only for very small $1-\beta$ (slow step size). On the boundary case, the standard schedule is $\mathbf{90\times}$ more accurate than fixed $\beta=0.93$ at 250 iterations (Figure~\ref{fig:q6_schedule}). Away-Steps FW (Section~\ref{subsec:q6_awaysteps}) further improves this to linear convergence for strongly convex objectives.
\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
