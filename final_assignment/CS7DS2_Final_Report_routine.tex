\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A, $J^{\star} \approx 0.4826$), a toy neural network loss with sinusoidal perturbation (Benchmark~B, $f^{\star} \approx 0.7244$ at $(0.582, 2.0)$), and the Rosenbrock function (Benchmark~C, $f^{\star}=0$, condition number $\kappa \approx 2508$). Six categories of methods are systematically investigated: (1) adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), where Polyak diverges on benchmarks with misspecified $f^{\star}$ but converges best when $f^{\star}$ is exact; (2) momentum-based methods (Nesterov, Adam), where Nesterov achieves $f = 0.479$ on Rosenbrock vs.\ GD's $f = 3.388$, exploiting its $O(1/k^2)$ convergence rate; (3) Newton's method with Hessian-based local quadratic approximation, achieving one-step convergence on the quadratic benchmark and $f = 0.686$ on Rosenbrock in only 20 iterations; (4) derivative-free methods (finite differences, Nesterov random search, Nelder--Mead, grid search) revealing the gradient fidelity requirements and dimension scalability trade-offs; (5) constrained optimisation via projected gradient descent and quadratic penalty methods, demonstrating the hard vs.\ soft feasibility enforcement trade-off; and (6) the Frank--Wolfe algorithm for linear-constrained problems, exploiting the structure of the feasible set without a projection oracle. Each method is evaluated in terms of convergence speed, trajectory behaviour, stability, and computational trade-offs, and the results are connected to theoretical convergence guarantees from the course.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmark functions exercise fundamentally different aspects of the optimisation landscape: quadratic structure (A), near-quadratic with sinusoidal perturbation (B), and severe non-convex ill-conditioning (C).

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated as follows. Feature vectors $x^{(i)} \in \mathbb{R}^2$ are drawn i.i.d.\ from $\mathcal{N}(0, I)$, and labels are $y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}$ with ground-truth parameters $\theta^{\star} = [3, 4]^T$ and noise $\varepsilon^{(i)} \sim \mathcal{N}(0, 1)$. The loss is mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
The Hessian is $H = \frac{1}{m}X^T X$ (constant, positive definite) with eigenvalues $\lambda_{\min} \approx 0.925$ and $\lambda_{\max} \approx 1.033$, giving a near-isotropic condition number $\kappa \approx 1.12$. The Lipschitz constant of the gradient is $L = \lambda_{\max} \approx 1.033$, so GD is stable for any $\alpha < 2/L \approx 1.94$ --- the choice $\alpha = 0.08$ is extremely conservative, reflecting that larger steps would not help on this already near-isotropic problem. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the minimum loss is the irreducible noise floor $J^{\star} \approx 0.4826$ attained at the OLS estimator $\hat{\theta}_{\mathrm{OLS}} \approx [3.028, 4.017]^T \neq \theta^{\star}$. The difference arises because the OLS estimator minimises the empirical loss with the specific noise realization, not the population loss. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
\label{eq:bench_B}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The sinusoidal term makes the Hessian position-dependent. The $(1,1)$ diagonal entry $2 - \sin(x_1)$ ranges from $1$ to $3$ as $x_1$ varies; the $(2,2)$ entry is always exactly $10$. The global Lipschitz constant of the gradient is therefore $L = 10$ (the largest possible Hessian eigenvalue), giving a stability constraint $\alpha \leq 2/L = 0.2$. At the starting point $x_0 = [-1, 4]^T$, the Hessian diagonal is $[2 + \sin(1), 10] \approx [2.84, 10]$ (local condition number $\kappa \approx 3.5$). At the minimum $x^{\star} \approx (0.582, 2.0)$, it is $[2 - \sin(0.582), 10] \approx [1.45, 10]$ (local condition number $\kappa \approx 6.9$). The minimum satisfies $\nabla f(x^{\star}) = 0$: the $x_2$ equation gives $x_2^{\star} = 2$ trivially, and the $x_1$ equation $2(x_1 - 1) + \cos(x_1) = 0$ is solved numerically (Newton's method) as $x_1^{\star} \approx 0.5824$ with $f^{\star} = (0.5824-1)^2 + \sin(0.5824) \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
\label{eq:rosenbrock}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian is:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1, 1)$, this evaluates to $H^{\star} = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}$ with $\det(H^{\star}) = 802 \times 200 - 400^2 = 400$ and $\mathrm{tr}(H^{\star}) = 1002$. The eigenvalues satisfy $\lambda_{1,2} = (1002 \pm \sqrt{1002^2 - 4 \times 400})/2 = (1002 \pm \sqrt{1{,}002{,}404})/2 \approx \{0.399,\, 1001.6\}$, giving a condition number $\kappa = \lambda_{\max}/\lambda_{\min} \approx 2508$. Remarkably, the Hessian at the \emph{starting point} $(-1, 1)$ has identical eigenvalues (both $\det$ and $\mathrm{tr}$ are the same, since $|-400(-1)| = 400 = |{-}400(1)|$), so the extreme ill-conditioning $\kappa \approx 2508$ persists throughout the trajectory. The Lipschitz constant varies along the trajectory; near the start and optimum it is dominated by $\lambda_{\max} \approx 1002$, requiring step sizes $\alpha \leq 2/1002 \approx 0.002$ for GD stability. Combined with the narrow curved valley $x_2 \approx x_1^2$, this makes Rosenbrock the most challenging benchmark. Starting point: $x_0 = [-1, 1]^T$.

\noindent\textbf{Benchmark~A} (quadratic) is globally strongly convex. All first-order convergence guarantees (GD, Nesterov, heavy ball, Polyak) apply globally. \textbf{Benchmark~B} has a position-dependent Hessian ranging from $\mathrm{diag}(1, 10)$ to $\mathrm{diag}(3, 10)$; it is locally strongly convex near the optimum and locally convex in a neighbourhood. \textbf{Benchmark~C} (Rosenbrock) is globally non-convex: the Hessian is indefinite at many points, and the function has a curved valley that is only locally convex along the valley floor. The PL condition holds near the global minimum $(1,1)$, which is why gradient methods eventually converge there, but global convergence guarantees require additional care.

The \emph{convexity hierarchy} is: strongly convex $\Rightarrow$ convex $\Rightarrow$ PL (near local min) $\Rightarrow$ non-convex. Methods with global guarantees require at least convexity; methods that converge to \emph{stationary points} (not necessarily global optima) apply to non-convex objectives. All gradient-based methods in this assignment converge to the global minimum on Benchmark~C empirically because the Rosenbrock has a single global minimum and the starting point $(-1, 1)$ lies in its basin of attraction.

\subsection{Algorithm Formulations}
\label{subsec:q1_algo}

Four adaptive methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, estimated optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive step uses gap to optimum}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak Convergence \cite{polyak1969minimization}]
If $f$ is convex, $L$-smooth, $\mu$-strongly convex, and $f^{\star}$ is known exactly, the Polyak step satisfies:
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 \leq \left(1 - \frac{\mu}{L}\right)\|x_k - x^{\star}\|^2,
\end{equation}
giving linear convergence with rate $1 - \mu/L = 1 - 1/\kappa$. This matches the optimal GD rate with the ideal constant step size $\alpha = 2/(\mu+L)$, but the Polyak step achieves this \emph{automatically} without knowing $L$ or $\mu$. Importantly, the convergence also holds under the weaker \emph{Polyak--\L{}ojasiewicz} (PL) condition:
\begin{equation}
\label{eq:PL}
\frac{1}{2}\|\nabla f(x)\|^2 \geq \mu(f(x) - f^{\star}), \quad \forall x,
\end{equation}
which holds for some non-convex functions including the Rosenbrock function near the optimum. The PL condition is strictly weaker than strong convexity: it does not require the function to be convex, nor the sublevel sets to be bounded. Under PL, gradient descent with $\alpha = 1/L$ satisfies $f(x_k) - f^{\star} \leq (1 - \mu/L)^k(f(x_0) - f^{\star})$, identical to the strongly convex case. This explains why gradient descent achieves linear convergence on Rosenbrock \emph{eventually} (once the iterates enter the region where PL holds), despite Rosenbrock being globally non-convex.
\end{theorem}

\noindent When $f^{\star}$ is \emph{misspecified} as $\tilde{f}^{\star} < f^{\star}_{\text{true}}$, the step $\alpha_k = (f(x_k) - \tilde{f}^{\star})/\|\nabla f(x_k)\|^2$ diverges: as $x_k \to x^{\star}$ and $\|\nabla f(x_k)\| \to 0$, we have $f(x_k) - \tilde{f}^{\star} \to f^{\star}_{\text{true}} - \tilde{f}^{\star} > 0$, so $\alpha_k \to +\infty$, destabilising the iteration. This is the mechanism behind the divergence observed on Benchmarks~A and~B.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Monotone per-coordinate accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The effective per-coordinate step $\alpha_0/(\sqrt{G_k^{(i)}} + \varepsilon)$ is large when coordinate $i$ has historically received small gradients, providing automatic adaptation to the gradient scale. Because $G_k^{(i)} = \sum_{j=0}^{k-1}(g_j^{(i)})^2$ is monotonically non-decreasing, the step size is \emph{guaranteed to shrink}, which provides stability but can cause premature convergence on non-stationary objectives.

\noindent Adagrad was originally designed for the \emph{online convex optimisation} setting, where the objective $f_t(x)$ changes at each time step and the goal is to minimise the cumulative regret $R_T = \sum_{t=1}^T f_t(x_t) - \sum_{t=1}^T f_t(x^{\star})$. The key Adagrad regret bound \cite{duchi2011adagrad} is:
\begin{equation}
R_T \leq 2\alpha_0 \sum_{i=1}^d \left\|g_{1:T}^{(i)}\right\|_2 = 2\alpha_0 \sum_{i=1}^d \sqrt{\sum_{t=1}^T (g_t^{(i)})^2},
\end{equation}
where $g_{1:T}^{(i)} = (g_1^{(i)}, \ldots, g_T^{(i)})$. This bound adapts to the \emph{observed gradient structure}: for sparse gradients (where many coordinates rarely receive non-zero gradients), $\|g_{1:T}^{(i)}\|_2$ is small for sparse coordinates, and Adagrad's per-coordinate adaptation achieves much better regret than any fixed-step-size method. For dense, isotropic gradients (as in Benchmark~A), $\|g_{1:T}^{(i)}\|_2 \approx \sqrt{T}\bar{\sigma}$ for all $i$, and the bound becomes $O(\sqrt{T})$ --- matching the minimax optimal rate for online convex optimisation. This is the theoretical basis for Adagrad's popularity in natural language processing, where word features are naturally sparse.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's cumulative sum with an exponential moving average parameterised by $\beta$. This ``forgets'' old gradients with timescale $1/(1-\beta)$ iterations, preventing the step from decaying to zero. In stationary regions where the gradient converges to a constant $g^{(i)}_{\infty}$, the EMA converges:
\begin{equation}
v_k^{(i)} \to (1-\beta)\sum_{j=0}^{\infty}\beta^j (g_\infty^{(i)})^2 = (g_\infty^{(i)})^2,
\end{equation}
and the effective per-coordinate step stabilises at $\alpha_0 / (|g_\infty^{(i)}| + \varepsilon) \approx \alpha_0 / |g_\infty^{(i)}|$. This is the \emph{gradient-normalised step}: coordinates with large gradients receive small effective steps (avoiding oscillation in steep directions) and coordinates with small gradients receive large effective steps (making progress in flat directions). The resulting convergence is \emph{independent of the gradient scale}, a property that makes RMSprop robust across widely different problem scales.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

For a strongly convex quadratic $f$ with $\mu$-strong convexity and $L$-smoothness, the optimal parameters of Heavy Ball are $\alpha^{\star} = 4/(\sqrt{L}+\sqrt{\mu})^2$ and $\beta^{\star} = \left((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)\right)^2$, yielding convergence rate $\rho = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$ per iteration --- a dramatic improvement over GD's rate of $(\kappa-1)/(\kappa+1)$.

\begin{remark}[Heavy Ball on non-quadratic objectives]
The Heavy Ball convergence rate result applies \emph{only} to quadratic objectives. On general smooth strongly convex functions, Heavy Ball does not have a known global convergence guarantee and can oscillate. The mechanism is the following: the accumulated velocity $z_k = \sum_{j \leq k} \beta^{k-j} \alpha \nabla f(x_j)$ is a linear combination of past gradients, but gradient directions at past iterates may conflict with the current descent direction when the Hessian changes rapidly. On Benchmark~C (Rosenbrock), the strongly varying Hessian along the valley causes the accumulated momentum to occasionally point away from the valley floor, producing oscillations visible in Figure~\ref{fig:q1_fval_C} as non-monotone convergence near the valley. Nesterov's method corrects for this by using the gradient at the \emph{look-ahead} point, which is more predictive of the gradient at the next iterate, achieving the $O(1/k^2)$ rate on all smooth convex functions.
\end{remark}

\noindent The Benchmark~C step sizes are approximately two orders of magnitude smaller than those for~A and~B, reflecting the large Lipschitz constant $L \approx 1001.6$ near the minimum (requiring $\alpha \leq 2/L \approx 0.002$ for GD stability). \begin{remark}[Heavy Ball optimal parameters for quadratics]
For Benchmark~A with $\mu \approx 0.925$ and $L \approx 1.033$, the theoretically optimal Heavy Ball parameters are $\alpha^{\star} = 4/(\sqrt{L}+\sqrt{\mu})^2 \approx 0.978$ and $\beta^{\star} = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2 \approx 0.007$, with convergence rate $\rho \approx 0.007$ per iteration. For Benchmark~C (Rosenbrock) with $\mu \approx 0.399$ and $L \approx 1002$ (local values at the optimum), the optimal parameters would be $\alpha^{\star} = 4/(\sqrt{1002}+\sqrt{0.399})^2 \approx 4/(31.65 + 0.632)^2 \approx 3.8 \times 10^{-3}$ and $\beta^{\star} = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2 \approx 0.980$. The chosen $\alpha = 0.0008$, $\beta = 0.86$ are more conservative to ensure global stability across the entire trajectory from $(-1,1)$ to $(1,1)$, trading theoretical optimality for robustness.
\end{remark}

\subsection{Comparative Analysis}
\label{subsec:q1_compare}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly (Rosenbrock), Polyak converges fastest due to its oracle-guided step size. Otherwise, Adagrad and Heavy Ball match or exceed GD on all benchmarks.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging. Heavy Ball shows characteristic overshoot followed by damped oscillation --- a direct consequence of the inertial term $\beta z_k$.
\item \textbf{Stability:} GD with $\alpha < 2/L$ is always stable. Polyak is \emph{unstable} when $f^{\star}$ is misspecified. Adagrad is uniformly stable due to its monotonically decreasing step, though this causes premature convergence on non-stationary problems. On a long run (500 iterations, Figure~\ref{fig:q1_ada_rms}), Adagrad's effective step decays to near-machine-precision, effectively freezing the iterate --- a known limitation of Adagrad for non-quadratic problems. RMSprop is meta-stable: the exponential average prevents unbounded step growth but can cause oscillation near the minimum.
\item \textbf{Practical implication:} Exact $f^{\star}$ is rarely available in practice, limiting Polyak's applicability. For deep learning, Adam (combining RMSprop's adaptive scaling with momentum) is the practical default, addressing both the frozen-step issue of Adagrad and the misspecification issue of Polyak.
\end{itemize}

\paragraph{Quantitative validation of GD rate on Benchmark~A.}
For Benchmark~A with $\alpha = 0.08$, $L \approx 1.033$, $\mu \approx 0.925$, and $\kappa \approx 1.12$, the theoretical linear convergence rate is:
\begin{equation}
\rho_{\text{GD}} = 1 - \alpha \mu = 1 - 0.08 \times 0.925 = 0.926.
\end{equation}
After $k = 120$ iterations, the predicted gap is $(J(\theta_0) - J^{\star}) \times 0.926^{120} \approx 12.63 \times 0.926^{120}$. Computing: $0.926^{120} \approx (0.926)^{120} \approx e^{120 \ln 0.926} \approx e^{-9.04} \approx 1.2 \times 10^{-4}$, giving predicted gap $\approx 1.5 \times 10^{-3}$. The initial distance to the noise floor is $J(\theta_0) - J^{\star} = J([0,0]) - 0.4826 \approx 13.107 - 0.4826 = 12.63$, so the predicted final value is $J^{\star} + 1.5 \times 10^{-3} \approx 0.484$. This closely matches the observed final value (Table~\ref{tab:q1_final}: $J \approx 0.4826$) to within the simulation precision. This confirms that the linear convergence theory is quantitatively accurate on this benchmark.

\subsection{Algorithm Formulations}
\label{subsec:q2_algo}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient (NAG)}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, maximum momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Gradient at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball (Algorithm~\ref{alg:heavyball}) is the \emph{lookahead}: the gradient is evaluated at the predicted next position $x_k + \beta_k z_k$ rather than at $x_k$. This corrects the trajectory \emph{before} overshooting, yielding an improved convergence guarantee:

\begin{theorem}[Nesterov Optimal Rate \cite{nesterov1983method}]
For $L$-smooth convex $f$ and the step schedule $\alpha = 1/L$, $\beta_k = (k-1)/(k+2)$, Nesterov's method satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} = O(1/k^2).
\end{equation}
This rate is \emph{optimal} for first-order methods on smooth convex problems \cite{nesterov1983method}, compared to GD's $O(1/k)$ rate.
\end{theorem}

\begin{remark}[Information-theoretic lower bound for first-order methods]
Nesterov's \emph{lower bound} theorem states that for \emph{any} first-order algorithm (method that accesses $f$ only through function values and gradients) applied to an $L$-smooth convex problem, there exists a function $f$ such that the best achievable convergence rate satisfies:
\begin{equation}
f(x_k) - f^{\star} \geq \frac{3L\|x_0 - x^{\star}\|^2}{32(k+1)^2}.
\end{equation}
The constructing example is a chain-structured quadratic: $f(x) = \frac{L}{8}(x_1^2 + \sum_{i=1}^{k-1}(x_i - x_{i+1})^2 - x_1)$ in $\mathbb{R}^k$, whose gradient structure ensures that any first-order method can only ``discover'' one new coordinate per iteration. The Nesterov rate $O(2L\|x_0-x^{\star}\|^2/(k+1)^2)$ and this lower bound $O(3L\|x_0-x^{\star}\|^2/(32(k+1)^2))$ match to within a constant factor, proving that Nesterov's method is \emph{optimal to within a constant} among all first-order methods. This result is historically significant: it resolved the open question of whether a first-order method can do better than $O(1/k)$, showing that $O(1/k^2)$ is simultaneously achievable and tight. The lower bound holds regardless of the dimension $d$, showing that the limitation is fundamental and not an artifact of a specific algorithm.
\end{remark}

\begin{algorithm}[H]
\caption{Adam Optimiser \cite{kingma2014adam}}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (mean) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (variance) estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum (tracking gradient direction via $m_t$) with per-coordinate adaptive scaling (via $v_t$, the second moment of gradients). The bias corrections $1 - \beta_1^t$ and $1 - \beta_2^t$ compensate for the zero-initialisation of $m_0, v_0$ in early iterations, preventing underestimated step magnitudes at $t=1$. Adam inherits advantages of both RMSprop and Heavy Ball, making it the de facto standard in deep learning optimisation.

\begin{remark}[Adam's convergence limitations]
Adam does not converge in general for non-convex objectives: Reddi et al.\ (2018) demonstrated a simple convex example where Adam's adaptive second-moment estimate grows without bound due to its exponential moving average, preventing convergence. Their proposed fix, \emph{AMSGrad}, replaces $v_t$ with $\hat{v}_t = \max(\hat{v}_{t-1}, v_t)$ (a monotone maximum), restoring the guarantee that step sizes decay. In practice, Adam converges reliably on deep learning tasks because the objectives are approximately strongly convex in the neighbourhood of typical attractors, but this theoretical gap is relevant for convex optimisation guarantees. For the deterministic benchmarks studied here, Adam converges reliably because the full gradient is available ($\sigma^2 = 0$), but its empirical performance on Rosenbrock (worse than Nesterov) reflects the structural mismatch between coordinate-wise adaptation and the coupled valley geometry.
\end{remark}

\paragraph{Why Nesterov outperforms Adam on Rosenbrock.} The Rosenbrock valley has a consistent gradient direction from the starting point to the minimum: gradients consistently point away from the valley walls and slightly towards the far end of the valley. Nesterov's increasing momentum $\beta_k \to \beta_{\max}$ accumulates velocity in this consistent direction, exploiting the persistent alignment of gradient directions along the trajectory. Adam, by contrast, normalises each coordinate independently by $\sqrt{\hat{v}_t + \varepsilon}$. For Rosenbrock, the $x_1$ gradient $g_1 = -2(1-x_1) - 400x_1(x_2 - x_1^2)$ and $x_2$ gradient $g_2 = 200(x_2 - x_1^2)$ are highly correlated (both depend on the residual $x_2 - x_1^2$), and the second-moment estimates $\hat{v}_t$ for both coordinates grow similarly in magnitude. Coordinate-wise normalisation therefore shrinks both gradient components equally, failing to exploit the coupled structure of the valley. Additionally, Adam's $\beta_1$-weighted momentum is less aggressive than Nesterov's increasing-schedule momentum for long valley traversals.

\begin{remark}[The $O(1/k^2)$ vs.\ $O(1/k)$ gap in practice]
On Benchmark~A ($\kappa \approx 1.12$), both GD and Nesterov converge in a similar number of iterations, because the $O(1/k^2)$ advantage matters most when $\kappa \gg 1$. For GD on a strongly convex function, the linear rate is $((\kappa - 1)/(\kappa + 1))^k \approx (0.056)^k$; for Nesterov, the rate is $((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^{2k} \approx (0.028)^{2k}$. With $\kappa \approx 1.12$, both converge within $\sim$20 iterations to machine precision, making the theoretical gap invisible at 150 iterations.

For Benchmark~B ($\kappa \approx 6.9$ at the minimum), the GD linear rate is $((\kappa-1)/(\kappa+1))^k \approx (0.745)^k$ and the Nesterov rate is $((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^{2k} \approx (0.449)^{2k} = (0.202)^k$. After $k = 150$ iterations: GD converges $(0.745)^{150} \approx 10^{-17}$ below the initial gap (past machine precision); Nesterov converges even faster at $(0.202)^{150} \approx 10^{-111}$. Both are well below the function minimum precision at 150 iterations, which is why both converge to $f^{\star}$ in Table~\ref{tab:q2_final}. The gap is most pronounced on ill-conditioned problems: for Rosenbrock ($\kappa \approx 2508$), GD's rate per iteration is $((\kappa-1)/(\kappa+1))^k = (0.99920)^k$, so it takes $\lceil \ln(2)/\ln(1/0.99920) \rceil \approx 866$ iterations to halve the objective gap; Nesterov's rate $(({\sqrt\kappa}-1)/({\sqrt\kappa}+1))^{2k} \approx (0.925)^k$ halves the gap in only $\lceil \ln(2)/\ln(1/0.925) \rceil \approx 9$ iterations --- a $\sim$100$\times$ speedup. This theoretical advantage manifests empirically: in Figure~\ref{fig:q2_fval_C}, Nesterov ($f = 0.479$) makes substantially more progress than GD ($f = 3.388$) within the same 150-iteration budget.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, and 50 epochs with data shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The per-epoch number of gradient steps is $m/b$: with $b = 5$, each epoch performs $200$ updates; with $b = 40$, only $25$ updates. The gradient variance of a mini-batch estimate is $\mathrm{Var}[\hat{g}] = (1/b)\,\mathrm{Var}[g_{\text{single}}]$, so smaller batches introduce higher gradient noise but more frequent updates.

\begin{theorem}[Mini-Batch SGD Convergence for Strongly Convex Loss]
Let $J(\theta) = \frac{1}{m}\sum_{i=1}^m \ell_i(\theta)$ be $\mu$-strongly convex and $L$-smooth. If the mini-batch gradient satisfies $\mathbb{E}[\hat{g}_k] = \nabla J(\theta_k)$ and $\mathbb{E}[\|\hat{g}_k - \nabla J(\theta_k)\|^2] \leq \sigma^2/b$ for batch size $b$, then with step size $\alpha \leq 1/L$:
\begin{equation}
\mathbb{E}[J(\theta_k) - J^{\star}] \leq \underbrace{(1 - \mu\alpha)^k (J(\theta_0) - J^{\star})}_{\text{geometric decay to noise floor}} + \underbrace{\frac{\alpha \sigma^2}{2\mu b}}_{\text{irreducible noise floor}},
\end{equation}
where $J^{\star}$ is the true minimum of $J$ and $\sigma^2 = \frac{1}{m}\sum_i \|\nabla \ell_i(\theta^{\star})\|^2$ is the gradient variance at the optimum. The noise floor $\alpha\sigma^2/(2\mu b)$ is minimised by taking $\alpha \to 0$ (but then convergence rate slows) or $b \to m$ (full batch, no noise). The optimal computational trade-off for a fixed total number of gradient evaluations $T = k \cdot b/m$ epochs is achieved at an intermediate batch size $b$.
\end{theorem}

\noindent For Benchmark~A with $\mu \approx 0.925$, $L \approx 1.033$, $\sigma^2 \approx 1$ (estimated from noise level), $\alpha = 0.06$, the noise floor is $0.06/(2 \times 0.925 \times b)$. For $b = 5$: floor $\approx 0.0065$; for $b = 40$: floor $\approx 0.0008$. Since the true noise floor from observation noise is $J^{\star} \approx 0.4826 \gg 0.0065$, the SGD noise floor is dominated by the irreducible observation noise --- both batch sizes should converge to the same final value (as observed in Figure~\ref{fig:q2_sgd_batch}).

\begin{remark}[Polyak--Ruppert averaging reduces SGD variance]
A classical technique to reduce the variance of the final SGD iterate is \emph{iterate averaging} (Polyak--Ruppert): instead of returning $x_T$, return the time-averaged iterate $\bar{x}_T = \frac{1}{T}\sum_{k=1}^T x_k$. Under appropriate conditions, the averaged iterate achieves $\mathbb{E}[J(\bar{x}_T) - J^{\star}] = O(\sigma^2 / (\mu T b))$, which is $O(1/T)$ vs.\ the geometric-to-noise-floor convergence of the final iterate. Averaging effectively removes the last-iterate variance without requiring a decreasing step size. In the mini-batch experiments here, the final iterate is used directly; averaging would reduce the oscillation visible in Figure~\ref{fig:q2_sgd_batch} near the noise floor.
\end{remark}

\begin{remark}[Variance reduction: SVRG and SAGA]
Iterate averaging reduces \emph{last-iterate} variance but does not eliminate the irreducible SGD noise floor. The \emph{stochastic variance reduced gradient} (SVRG) method (Johnson and Zhang, 2013) periodically computes a full gradient $\tilde{g} = \nabla J(\tilde{x})$ at a snapshot $\tilde{x}$, then uses the variance-reduced estimator $\hat{g}_k = \nabla \ell_k(x_k) - \nabla \ell_k(\tilde{x}) + \tilde{g}$, which satisfies $\mathbb{E}[\hat{g}_k] = \nabla J(x_k)$ and has variance $\mathbb{E}[\|\hat{g}_k - \nabla J(x_k)\|^2] = O(\|x_k - \tilde{x}\|^2) \to 0$ as $x_k \to \tilde{x}$. As a result, SVRG achieves \emph{linear convergence} $\mathbb{E}[J(x_k) - J^{\star}] \leq O(\rho^k)$ for strongly convex $J$ --- the same rate as full gradient descent but with only $O(1)$ gradient evaluations per step on average. The per-epoch cost is $(m/b + m)$ gradient evaluations ($m$ for the snapshot), which is beneficial when $m/b \gg 1$ (small batch) and the problem is strongly convex. SAGA (Defazio et al., 2014) is a memory-efficient alternative that stores per-sample gradient estimates from the previous pass instead of recomputing a full snapshot, achieving the same linear convergence rate at $O(m)$ memory cost vs.\ $O(d)$ for SVRG.
\end{remark}

\subsubsection{Gradient Variance as a Function of Batch Size}

A fundamental theoretical property of mini-batch SGD is that the variance of the stochastic gradient scales inversely with the batch size. For a dataset of $m$ samples with per-sample gradients $g_i = \nabla \ell(\theta; x_i, y_i)$, the mini-batch gradient with batch $\mathcal{B}$ of size $b$ satisfies:
\begin{equation}
\mathbb{E}\left[\left\|\hat{g}_b - \nabla J\right\|^2\right] = \frac{1}{b}\left(1 - \frac{b}{m}\right)\sigma_g^2,
\end{equation}
where $\sigma_g^2 = \frac{1}{m}\sum_{i=1}^m \|g_i - \nabla J\|^2$ is the population gradient variance. For $b \ll m$, this simplifies to $\approx \sigma_g^2 / b$, giving an exact $O(1/b)$ scaling. This means doubling the batch size halves the gradient noise, but also halves the number of gradient steps per epoch: the total variance per epoch remains constant.

By the Central Limit Theorem, for large $b$, the mini-batch gradient $\hat{g}_b$ is approximately Gaussian: $\hat{g}_b \approx \mathcal{N}(\nabla J(\theta), \Sigma_g/b)$ where $\Sigma_g = \frac{1}{m}\sum_i (g_i - \nabla J)(g_i - \nabla J)^T$ is the gradient covariance matrix. This Gaussian approximation is routinely used to analyse SGD dynamics and motivates noise-adapted step size schedules that decay as $\alpha_k \sim 1/\sqrt{k}$ (Robbins-Monro condition: $\sum_k \alpha_k = \infty$, $\sum_k \alpha_k^2 < \infty$) to achieve convergence.

\begin{figure}[H]

\subsection{Local Approximations of $g(x) = x^4$}
\label{subsec:q3_approx}

The Taylor expansion of $g(x) = x^4$ at $x_0 = 0.25$ is:
\begin{align}
g'(x_0) &= 4x_0^3 = 4(0.25)^3 = 0.0625, \\
g''(x_0) &= 12x_0^2 = 12(0.25)^2 = 0.75.
\end{align}

\textbf{First-order (tangent) approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.00391 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.00391 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

Minimising $\hat{g}_2$ gives the Newton update:
\begin{equation}
x^{\star}_{\hat{g}_2} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} \approx 0.167.
\end{equation}
This is closer to the true minimum $x^{\star} = 0$ than a pure gradient step $x_0 - \alpha g'(x_0)$ for any reasonable $\alpha$, illustrating why Newton's direction is powerful.

\subsection{Newton's Method with Damped Steps}

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \leq 1$, regularisation $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised (positive definite) Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: solve linear system}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton direction $p_k = H_k^{-1} g_k$ is the steepest descent direction in the local Hessian-induced metric (also called the \emph{natural gradient}). It adapts the step to the local curvature: large curvature directions receive small steps; shallow directions receive large steps. Tikhonov regularisation ($\lambda > 0$) ensures positive definiteness of $H_k$ even when the Hessian has zero or negative eigenvalues.

\begin{theorem}[Quadratic Convergence of Newton's Method]
If $f$ is twice continuously differentiable, $\nabla^2 f$ is Lipschitz continuous with constant $M$, and $x_k$ is sufficiently close to a non-degenerate local minimum $x^{\star}$ ($\nabla^2 f(x^{\star}) \succ 0$), then the full Newton step ($\alpha = 1$) satisfies:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2,
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^{\star}))$. This \emph{quadratic convergence} means the number of correct significant digits doubles at each iteration.
\end{theorem}

Settings: GD uses 80 iterations; Newton uses only 20 iterations with Tikhonov regularisation $\lambda = 10^{-8}$ to ensure numerical stability.

The \emph{Newton decrement} $\lambda_k = (g_k^T H_k^{-1} g_k)^{1/2} = (p_k^T H_k p_k)^{1/2}$ provides a scale-invariant convergence certificate: for $\alpha = 1$ (pure Newton step) with the Armijo condition, $f(x_k) - f^{\star} \leq \lambda_k^2/2$. A typical stopping criterion is $\lambda_k^2/2 < \varepsilon_{\text{tol}}$, which can be computed without knowing $f^{\star}$. On Benchmark~A, the Newton decrement drops to $\sim 10^{-14}$ after a single step, confirming one-step convergence.

\noindent\textbf{Gradient-norm vs.\ Newton-decrement termination.} For methods without the Hessian, the most common termination criterion is the gradient norm $\|\nabla f(x_k)\| \leq \varepsilon_g$. Under strong convexity ($\mu > 0$), the gradient norm bounds the objective gap via the PL condition (Eq.~\eqref{eq:PL}):
\begin{equation}
f(x_k) - f^{\star} \leq \frac{1}{2\mu}\|\nabla f(x_k)\|^2.
\end{equation}
Thus $\|\nabla f(x_k)\| \leq \sqrt{2\mu\varepsilon}$ guarantees $f(x_k) - f^{\star} \leq \varepsilon$. The Newton decrement is more informative than the gradient norm because it accounts for the local curvature ($\lambda_k^2 = g_k^T H_k^{-1} g_k \leq g_k^T (\mu I)^{-1} g_k = \|g_k\|^2/\mu$), making it tighter by a factor of $\mu$ in the worst case. For Benchmark~B at the starting point $x_0 = (-1, 4)$: $\|\nabla f(x_0)\|^2 = (2(-2)+\cos(-1))^2 + (10 \times 2)^2 \approx (-3.46)^2 + 20^2 \approx 411.9$, so $\varepsilon_g = 0.01$ would require 411.9/(2 \times 2.84) $\approx$ 72 iterations of GD to satisfy the criterion --- consistent with the observed convergence in Figure~\ref{fig:q3_fval_B}.

\begin{remark}[Armijo backtracking line search for Newton's method]
The fixed damping factor $\alpha = 0.22$ used in the assignment is a manual choice. In practice, the step size is selected dynamically via the \emph{Armijo backtracking condition}: starting from $\alpha = 1$ (full Newton step), repeatedly set $\alpha \gets \rho\alpha$ (typically $\rho = 0.5$) until the sufficient decrease condition
\begin{equation}
f(x_k - \alpha p_k) \leq f(x_k) - c\, \alpha\, g_k^T p_k, \quad c \in (0, 1),
\end{equation}
is satisfied. With $c = 10^{-4}$ and $\rho = 0.5$, this guarantees global convergence without requiring manual tuning of the damping factor. For the Rosenbrock function, the Armijo condition would automatically select larger $\alpha$ values when far from the optimum (where the quadratic model is accurate) and smaller $\alpha$ values near the poorly-conditioned valley walls. The damped Newton method with Armijo backtracking achieves \emph{global convergence} from any starting point and \emph{quadratic local convergence} near the optimum --- the best of both worlds. The BFGS quasi-Newton method (Section~\ref{subsec:q3_compare}) combined with Armijo backtracking is the standard workhorse for medium-scale non-linear optimisation.
\end{remark}

\paragraph{Computational cost comparison.}
\label{subsec:q3_compare}
Newton requires computing and inverting the $d \times d$ Hessian at each step, costing $O(d^2)$ for the gradient-Hessian pair and $O(d^3)$ for the solve via Gaussian elimination on $H_k p_k = g_k$. For $d = 2$ this is negligible; for large $d$ (neural networks with $10^6$--$10^9$ parameters), it is completely prohibitive both in memory ($O(d^2)$ to store the Hessian) and computation. There are several practical alternatives:

\begin{itemize}[nosep]
\item \textbf{BFGS:} Broyden--Fletcher--Goldfarb--Shanno maintains a rank-2 update of the inverse Hessian approximation $B_k \approx H_k^{-1}$:
\begin{equation}
B_{k+1} = \left(I - \frac{s_k y_k^T}{y_k^T s_k}\right) B_k \left(I - \frac{y_k s_k^T}{y_k^T s_k}\right) + \frac{s_k s_k^T}{y_k^T s_k},
\end{equation}
where $s_k = x_{k+1} - x_k$ and $y_k = g_{k+1} - g_k$. The update is derived from the \emph{secant condition} $B_{k+1} s_k = y_k$ (the inverse Hessian approximation must satisfy the same relationship as the true inverse Hessian: $H^{-1}(g_{k+1} - g_k) \approx x_{k+1} - x_k$), subject to the constraint that $B_{k+1}$ is the closest positive-definite matrix to $B_k$ in the Frobenius norm. The secant condition encodes observed curvature information from the last gradient difference, allowing $B_k$ to gradually build a global Hessian approximation from local gradient changes. BFGS achieves \emph{superlinear} (but not quadratic) convergence with only $O(d^2)$ storage and cost, bridging the gap between Newton and gradient methods.
\item \textbf{L-BFGS:} Limited-memory BFGS approximates $H_k^{-1}$ using the last $m$ gradient differences $\{s_{k-i}, y_{k-i}\}_{i=0}^{m-1}$. Storage is $O(md)$ and per-iteration cost is $O(md)$; typical choice $m = 10$--$50$. L-BFGS is the standard method for large-scale smooth unconstrained optimisation.
\item \textbf{Gauss-Newton:} For least-squares problems, approximates $H_k \approx J_k^T J_k$ (Jacobian product), which is always positive semi-definite. This is the basis for the Levenberg--Marquardt algorithm widely used in nonlinear regression.
\item \textbf{Truncated Newton:} Approximates the Newton direction $H_k^{-1}g_k$ iteratively via conjugate gradient, terminating early to save computation.
\end{itemize}

For $d = 2$ as in this assignment, full Newton is optimal: quadratic convergence in 20 iterations vs.\ 80+ iterations for GD represents a $4\times$ reduction in computation and a qualitatively different convergence profile.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark~B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces the exact partial derivative with:
\begin{equation}
\label{eq:fd}
\frac{\partial f}{\partial x_i}(x) \approx \hat{\partial}_i f(x; \delta) = \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
The approximation error has two competing components:
\begin{equation}
\left|\hat{\partial}_i f - \frac{\partial f}{\partial x_i}\right| \leq \underbrace{\frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}\right|}_{\text{truncation error}} + \underbrace{\frac{2\varepsilon_{\text{mach}} |f(x)|}{\delta}}_{\text{rounding error}}.
\end{equation}
The total error is minimised at the ``optimal'' perturbation $\delta^{\star} \sim \sqrt{\varepsilon_{\text{mach}}/|f''|} \approx \sqrt{2.2\times10^{-16}/3} \approx 2.7 \times 10^{-8}$ for $g(x) = x^4$ at $x_0 = 0.25$ with $g''(x_0) = 0.75$ and double-precision arithmetic. An improved \emph{central difference} scheme $(\partial f/\partial x_i) \approx (f(x + \tfrac{\delta}{2}e_i) - f(x - \tfrac{\delta}{2}e_i))/\delta$ has truncation error $O(\delta^2)$ instead of $O(\delta)$, so its optimal $\delta$ is larger ($\sim \varepsilon_{\text{mach}}^{1/3} \approx 6\times10^{-6}$) and it achieves higher accuracy. The assignment uses forward differences.

\begin{figure}[H]
\centering
\includegraphics[width=0.88\textwidth]{figures/fd_error_vs_delta.pdf}
\caption{Q4: Finite difference approximation error for $g(x) = x^4$ at $x_0 = 0.25$ vs.\ perturbation size $\delta$, showing the classic two-regime behaviour: for small $\delta$, rounding error $O(\varepsilon_{\text{mach}}/\delta)$ dominates (increasing error as $\delta \to 0$); for large $\delta$, truncation error $O(\delta)$ dominates (increasing error as $\delta$ grows). The optimal $\delta$ for forward differences (vertical blue dashed line) is $\sim 10^{-8}$; for central differences (red dashed), it is $\sim 10^{-5}$. The Q4 choices $\delta = 0.05$ (green) and $\delta = 0.8$ (orange) are both in the truncation-error-dominated regime; $\delta = 0.05$ is near the lower end where truncation is small, while $\delta = 0.8$ has substantial truncation error $O(0.8)$.}
\label{fig:q4_fd_error}
\end{figure}

Each FD gradient evaluation costs $d + 1$ function evaluations (forward differences) or $2d$ for central differences. For exact gradient computation via automatic differentiation (backpropagation), the cost is $O(d)$ but with a small constant; for analytical gradients (as used here), cost is $O(d)$ with essentially zero overhead.

\begin{remark}[Effect of gradient bias on convergence]
The FD gradient with step $\delta = 0.05$ introduces a systematic bias $\epsilon_k = \hat{g}_k - \nabla f(x_k)$ with $\|\epsilon_k\| \leq \delta L_2/2 \approx 0.025 \times 3 = 0.075$ (where $L_2 \approx 3$ is the Lipschitz constant of the Hessian for Benchmark~B). For GD with $\alpha = 0.08$, the iteration $x_{k+1} = x_k - \alpha(\nabla f(x_k) + \epsilon_k)$ converges to an approximate stationary point $x^{\infty}$ satisfying $\|\nabla f(x^{\infty})\| \leq \|\epsilon_{\infty}\| \leq 0.075$. With $L = 10$ and step $\alpha = 0.08$, the FD GD method converges to within $\approx 0.075^2/(2\mu) \approx 0.003$ of the true minimum in objective value --- sufficient precision for this problem. The use of $\alpha = 0.08$ (slightly larger than the exact GD step $\alpha = 0.06$) is appropriate because the FD bias acts as a noise floor, making the very conservative step of 0.06 unnecessarily slow without providing additional accuracy benefit.
\end{remark}

\begin{remark}[Estimating $L$ and $\mu$ from finite differences for step size selection]
In practice, the Lipschitz constant $L$ (needed for step size $\alpha \leq 1/L$) and strong convexity constant $\mu$ (needed for momentum schedules) are estimated directly from the Hessian via finite differences. For forward differences, $H_{ij}(x) \approx (\nabla_j f(x + \delta e_i) - \nabla_j f(x))/\delta$, requiring $(d+1)$ gradient evaluations. Once $\hat{H}$ is available, $L \approx \lambda_{\max}(\hat{H})$ and $\mu \approx \lambda_{\min}(\hat{H})$ (for positive definite $\hat{H}$). On Benchmark~B at the starting point $x_0 = (-1, 4)$: $\hat{H} = \mathrm{diag}(2 - \sin(-1), 10) = \mathrm{diag}(2.84, 10)$, so $L \approx 10$ and $\mu \approx 2.84$, giving a step size constraint $\alpha \leq 0.1$. The chosen $\alpha = 0.06$ is safely below $1/L = 0.1$ with a factor-of-1.67 margin. In higher dimensions where the full Hessian is too expensive, a power iteration estimate of $\lambda_{\max}$ using random gradient probes provides an efficient alternative.
\end{remark}

\subsubsection{Nesterov Random Search (Zeroth-Order)}

\begin{algorithm}[H]
\caption{Nesterov Random Search (1-Point Feedback)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Random unit vector}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method requires only \textbf{two function evaluations} per iteration regardless of dimension, making it dimension-independent in per-iteration cost. The estimator $\hat{g}_k$ satisfies:
\begin{equation}
\mathbb{E}_{u_k}[\hat{g}_k] = \frac{1}{d}\nabla f_{\delta}(x_k) \approx \frac{1}{d}\nabla f(x_k) + O(\delta^2),
\end{equation}
where $f_{\delta}(x) = \mathbb{E}_{u \sim \text{Uniform}(S^{d-1})}[f(x + \delta u)]$ is the Gaussian smoothing of $f$ and $d$ is the dimension. The bias $O(\delta^2)$ arises from the difference between $\nabla f_\delta$ and $\nabla f$; for $\delta \to 0$, the estimator converges to an unbiased gradient estimate, but variance grows as $1/\delta$.

\begin{theorem}[Nesterov Random Search Convergence \cite{nesterov2004introductory}]
For $L$-smooth convex $f$, the random search estimator $\hat{g}_k = (f(x_k + \delta u_k) - f(x_k))/\delta \cdot u_k$ with $u_k$ uniform on $S^{d-1}$ and step size $\alpha = 1/(L d)$ satisfies after $T$ iterations:
\begin{equation}
\mathbb{E}[f(x_T)] - f^{\star} \leq \frac{d\, L\, \|x_0 - x^{\star}\|^2}{T} + O(\delta^2),
\end{equation}
which is $d$ times slower than gradient descent's $O(L\|x_0-x^{\star}\|^2/T)$ rate. For $d = 2$, this means twice as many iterations for the same accuracy; for $d = 100$, one hundred times as many. Choosing $\delta \sim T^{-1/4}$ balances the truncation bias and stochastic variance to give an overall rate $O(d/\sqrt{T})$ for smooth non-convex objectives.
\end{theorem}

The factor $1/d$ in the expected gradient estimate means each step makes only $1/d$ progress compared to a full gradient step, requiring $O(d)$ more iterations to achieve the same accuracy --- the fundamental trade-off of zeroth-order methods.

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ (for $n=2$), iteratively transforming it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\rho_r = 1$): move the worst vertex $x_w$ through the centroid $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + 1 \cdot (\bar{x} - x_w)$.
\item \textbf{Expansion} ($\gamma = 2$): if reflection improved, try $x_e = \bar{x} + 2(\bar{x} - x_w)$.
\item \textbf{Contraction} ($\rho_c = 0.5$): if reflection worsened, try $x_c = \bar{x} + 0.5(\bar{x} - x_w)$ (outside) or $\bar{x} - 0.5(\bar{x} - x_w)$ (inside).
\item \textbf{Shrink} ($\sigma = 0.5$): if all else fails, move all non-best vertices towards the best.
\end{enumerate}
No gradient information is required. Settings: initial simplex step $= 0.35$, 160 iterations.

\begin{remark}[Convergence behaviour of Nelder--Mead]
Nelder--Mead has no general convergence guarantee for $d \geq 2$ \cite{nocedal2006numerical}: McKinnon (1998) constructed a smooth 2D function for which Nelder--Mead converges to a non-stationary point. Despite this, Nelder--Mead performs reliably on smooth low-dimensional problems such as Rosenbrock, and the observed ``staircase'' convergence in Figure~\ref{fig:q4_nm} reflects the discrete simplex operations: each plateau corresponds to a series of reflections/contractions that maintain the same best vertex until a shrink step reorganises the simplex. The absence of a convergence guarantee is an inherent limitation of all direct-search methods that rely only on function evaluations with no derivative information.
\end{remark}

\begin{remark}[Dimension dependence of zeroth-order methods]
For Nesterov random search in dimension $d$, the expected per-step progress towards the optimum is $1/d$ of what a full gradient step achieves. Thus to achieve the same accuracy as $T$ gradient descent steps, NRS requires $O(dT)$ iterations. For $d = 2$, this means $2\times$ more iterations --- manageable. For $d = 1000$, it means $1000\times$ more iterations --- prohibitive compared to any gradient method. This \emph{curse of dimensionality for zeroth-order methods} is why gradient computation (via backpropagation) is essential in high-dimensional machine learning.

Nelder--Mead's convergence deteriorates sharply for $d > 10$: the number of simplex vertices ($d+1$) grows linearly, and the operations (reflection, expansion, contraction) do not scale gracefully to high dimensions. For $d = 2$, Nelder--Mead is effective and intuitive; for $d \geq 20$, random search or evolutionary algorithms are typically preferred.
\end{remark}

\subsection{Problem Formulation}

Minimise Benchmark~B (Eq.~\eqref{eq:bench_B}) subject to the half-space constraint $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$. The feasible set is the closed half-space $\mathcal{X} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}$.

The constrained minimum coincides with the unconstrained minimum at $(x_1^{\star}, x_2^{\star}) = (0.582, 2.0)$ with $f^{\star} \approx 0.7244$, since $(0.582, 2.0) \in \mathcal{X}$ ($0.582 > 0.5$). The constraint is therefore \emph{inactive} at the optimum, but still requires enforcement because the initial point $x_0 = (0.2, 4.0)$ violates it.

\subsection{KKT Optimality Conditions}

The constrained optimisation problem is:
\begin{equation}
\min_{x \in \mathbb{R}^2} f(x) \quad \text{subject to} \quad h(x) = 0.5 - x_1 \leq 0.
\end{equation}
The Karush--Kuhn--Tucker (KKT) necessary conditions for a local minimum $x^{\star}$ are:
\begin{align}
\nabla f(x^{\star}) - \nu^{\star} e_1 &= 0 \quad (\text{stationarity}), \\
h(x^{\star}) &\leq 0 \quad (\text{primal feasibility}), \\
\nu^{\star} &\geq 0 \quad (\text{dual feasibility}), \\
\nu^{\star} h(x^{\star}) &= 0 \quad (\text{complementary slackness}).
\end{align}
For this problem, the unconstrained minimiser $(x_1^{\star}, x_2^{\star}) = (0.582, 2.0)$ satisfies $h(x^{\star}) = 0.5 - 0.582 = -0.082 < 0$ (strict feasibility), so by complementary slackness $\nu^{\star} = 0$. This confirms the constrained and unconstrained minima coincide: the constraint is \emph{inactive} at the solution, and the KKT multiplier is zero. Therefore, methods that eventually reach the unconstrained minimum (PGD, penalty with sufficient $\lambda$) converge to the same point, as observed in Table~\ref{tab:q5_final}.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection operator $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = \begin{bmatrix}\max(0.5,\, z_1) \\ z_2\end{bmatrix}$ \Comment{Project: clip $x_1$ to feasible region}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[PGD Convergence \cite{bertsekas1999nonlinear}]
For convex $f$ with $L$-Lipschitz gradient, PGD with $\alpha \leq 1/L$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{\|x_0 - x^{\star}\|^2}{2\alpha k}.
\end{equation}
This is the same $O(1/k)$ rate as unconstrained GD, with the projection adding only a per-iteration $O(d)$ cost.
\end{theorem}

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $\nabla F_{\lambda}(x) \gets \nabla f(x) - \lambda \mathbf{1}[x_1 < 0.5]\, e_1$ \Comment{Penalised gradient}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ (step sizes $\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected GD} enforces feasibility as a hard constraint at every iteration via the projection oracle. For simple constraint sets such as half-spaces and boxes, projection is a $\max$ or $\text{clip}$ operation costing $O(d)$ per iteration. For complex constraints (e.g.\ second-order cones, nuclear norm balls), projection can be expensive or intractable.

\textbf{Penalty methods} are simpler to implement (no projection oracle required) but enforce feasibility only asymptotically. The weight $\lambda$ controls the hard/soft trade-off. Small $\lambda$ ($= 0.15$) under-penalises violations because the penalty gradient $\lambda\mathbf{1}[x_1 < 0.5]$ is weaker than the objective gradient pulling towards the unconstrained minimiser. Large $\lambda$ ($= 4.5$) enforces feasibility but increases the Lipschitz constant of $\nabla F_\lambda$ from $L_f \approx 10$ to $L_f + \lambda = 14.5$, requiring a smaller step size $\alpha \leq 2/14.5 \approx 0.138$ --- hence the choice $\alpha = 0.03$ for $\lambda = 4.5$.

\paragraph{Log-barrier / Interior-point methods.}
A smooth alternative to the quadratic penalty is the \emph{log-barrier method}, which replaces the inequality constraint $x_1 \geq 0.5$ with a barrier term that is smooth and concave on the feasible interior:
\begin{equation}
F_t(x) = f(x) - \frac{1}{t}\ln(x_1 - 0.5), \quad x_1 > 0.5.
\end{equation}
As $t \to \infty$, $F_t(x) \to f(x)$ for $x_1 > 0.5$ and $F_t(x) \to +\infty$ as $x_1 \to 0.5^+$, so the unconstrained minimiser of $F_t$ approaches the constrained minimiser. Unlike the quadratic penalty (which uses $(\max(0, 0.5 - x_1))^2$), the log-barrier keeps all iterates strictly feasible ($x_1 > 0.5$) throughout the algorithm. Newton's method applied to $F_t$ with a sequence $t_1 < t_2 < \cdots \to \infty$ is the basis of \emph{interior point methods} (Karmarkar, 1984; Boyd and Vandenberghe, 2004), which achieve polynomial-time convergence for convex problems and are the standard approach for large-scale linear and semidefinite programming. The log-barrier Lipschitz constant at the boundary grows as $O(1/t(x_1 - 0.5)^2)$, requiring the iterates to remain at least $O(1/\sqrt{t})$ away from the constraint boundary --- this is the mechanism that ``pushes'' iterates towards the feasible interior and avoids the infeasibility issues of GD with a fixed starting point on the boundary.

\begin{remark}[Augmented Lagrangian avoids the $\lambda \to \infty$ problem]
The pure quadratic penalty requires $\lambda \to \infty$ to achieve exact feasibility, causing increasing ill-conditioning of the penalised gradient (Lipschitz constant $L + \lambda$). The \emph{augmented Lagrangian} (method of multipliers) for the inequality constraint $h(x) = \max(0, 0.5 - x_1) \leq 0$ introduces a multiplier $\nu \geq 0$ alongside a moderate penalty $\rho > 0$:
\begin{equation}
\mathcal{L}_\rho(x, \nu) = f(x) + \nu\, h(x) + \frac{\rho}{2}[h(x)]^2.
\end{equation}
The method of multipliers alternates between:
\begin{enumerate}[nosep]
\item \textbf{Primal update:} $x_{k+1} \gets \arg\min_x \mathcal{L}_\rho(x, \nu_k)$ (approximate, via gradient steps).
\item \textbf{Dual update:} $\nu_{k+1} \gets \max(0,\, \nu_k + \rho\, h(x_{k+1}))$.
\end{enumerate}
The dual update drives $\nu$ towards the Lagrange multiplier $\nu^{\star}$ at the constrained optimum. Once $\nu_k \approx \nu^{\star}$, a single primal solve with a moderate $\rho$ achieves exact feasibility, avoiding the ill-conditioning issue. For the half-space constraint with the unconstrained minimum satisfying the constraint ($h(x^{\star}) = 0$, inactive constraint), the true multiplier is $\nu^{\star} = 0$; the dual update would converge $\nu_k \to 0$, meaning the augmented Lagrangian reduces to the original objective with no penalty needed at convergence. This illustrates why augmented Lagrangian is the preferred approach for problems with multiple constraints of varying activity.

The ADMM (Alternating Direction Method of Multipliers) extends the augmented Lagrangian to separable problems: for $\min_{x,z} f(x) + g(z)$ subject to $Ax + Bz = c$, it decomposes the primal update into separate $x$-minimisation and $z$-minimisation steps, enabling distributed implementation when $f$ and $g$ are evaluated on different compute nodes. ADMM converges at $O(1/k)$ rate for convex objectives and is widely used in large-scale machine learning (Lasso, support vector machines, graph optimisation).
\end{remark}

\begin{remark}[Projection oracle vs.\ linear minimisation oracle: PGD and Frank--Wolfe]
Projected GD (this question) and Frank--Wolfe (Question~6) both achieve $O(1/k)$ convergence for $L$-smooth convex problems over compact constraint sets, but use different first-order oracles:
\begin{itemize}[nosep]
\item \textbf{PGD oracle:} $\Pi_\mathcal{X}(x) = \arg\min_{z \in \mathcal{X}} \|z - x\|^2$ --- the \emph{Euclidean projection}.
\item \textbf{Frank--Wolfe oracle:} $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ --- the \emph{linear minimisation} (LP) subproblem.
\end{itemize}
For the half-space $\mathcal{X} = \{x : x_1 \geq 0.5\}$ (this question): projection is $\max(0.5, x_1)$ ($O(1)$ per coordinate); LP is also $O(1)$. For the Q6 box constraint, both oracles are trivial coordinate-wise operations. The oracle comparison becomes decisive for complex constraint sets:
\begin{center}
\small
\begin{tabular}{lcc}
\toprule
\textbf{Constraint set} & \textbf{Projection cost} & \textbf{LP oracle cost} \\
\midrule
Half-space, box & $O(d)$ & $O(d)$ \\
$\ell^1$ ball $\|x\|_1 \leq r$ & $O(d\log d)$ (soft-threshold) & $O(d)$ (extreme point of $\ell^\infty$ unit ball) \\
Nuclear norm ball $\|X\|_* \leq r$ & $O(d^2 \min(m,n))$ (full SVD) & $O(d\min(m,n))$ (top singular vectors) \\
Spectrahedron $X \succeq 0$, $\mathrm{tr}(X)=1$ & $O(d^3)$ (eigendecomposition) & $O(d^2)$ (top eigenvector) \\
\bottomrule
\end{tabular}
\end{center}
For the nuclear norm ball (key to low-rank matrix completion), the Frank--Wolfe LP oracle requires only the \emph{leading} singular vector pair via power iteration ($O(d\min(m,n))$ per iteration), while PGD requires a \emph{full} SVD ($O(d^2\min(m,n))$) --- a factor-$\min(m,n)$ speedup. This explains why Frank--Wolfe has found renewed interest in matrix optimisation problems.
\end{remark}

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm \hfill [30 marks]}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box polytope with diameter $D = \|(5 - 0.5, 10 - (-5))\| = \|(4.5, 15)\| \approx 15.7$).

\textbf{Linear Programme.} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$. Since $a = [1, 2]^T \succ 0$ componentwise, the minimum is attained at the lower bound of each coordinate:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 1 \times 0.5 + 2 \times (-5) = -9.5.
\end{equation}
This illustrates the fundamental theorem of linear programming: the optimum of a linear objective over a bounded convex polytope is always attained at a \emph{vertex}. For box constraints, the optimal vertex is determined by the sign of each component of $a$: $x_i^{\star} = l_i$ if $a_i > 0$, else $x_i^{\star} = u_i$.

\textbf{LP Duality.} The dual of the box-constrained LP is:
\begin{equation}
\max_{\nu_l, \nu_u \geq 0} -\nu_l^T l - \nu_u^T u \quad \text{s.t.} \quad a - \nu_l + \nu_u = 0,
\end{equation}
with $\nu_l, \nu_u$ being the dual variables for the lower and upper bounds respectively. At the optimal vertex $x^{\star} = (0.5, -5)$: the active constraints are $x_1 = l_1 = 0.5$ and $x_2 = l_2 = -5$, so $\nu_{l,1} = a_1 = 1 > 0$ and $\nu_{l,2} = a_2 = 2 > 0$ with $\nu_{u,1} = \nu_{u,2} = 0$. The dual objective value is $-\nu_l^T l = -1 \times 0.5 - 2 \times (-5) = 9.5 = -f^{\star}$, confirming strong LP duality (primal $=$ dual at the solution). The Frank--Wolfe algorithm exploits this LP subproblem structure in each iteration to provide a descent direction without requiring a projection oracle.

\centering

\caption{Q6-I: Feasible region (blue box) and linear objective contours $f(x) = x_1 + 2x_2$. The contour lines are parallel (slope $-a_1/a_2 = -1/2$), and the minimum occurs at the lower-left vertex $(0.5, -5)$ where the smallest-valued contour line intersects the feasible region.}

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe with Fixed Mixing Parameter}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{LP subproblem (linear oracle)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: maintain feasibility}
\EndFor
\end{algorithmic}
\end{algorithm}

The LP subproblem decomposes per coordinate for box constraints:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0, \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0. \end{cases}
\end{equation}
Since $x_{k+1} = \beta x_k + (1-\beta) z_k$ is a convex combination of two feasible points ($x_k, z_k \in \mathcal{X}$), the iterate $x_{k+1} \in \mathcal{X}$ automatically --- \textbf{no projection oracle is required}. This is the key advantage of Frank--Wolfe over projected gradient descent for problems where projection is expensive.

\begin{theorem}[Frank--Wolfe Convergence \cite{frank1956algorithm,jaggi2013revisiting}]
For $L$-smooth convex $f$ over a compact convex set $\mathcal{X}$ with diameter $D$ and step sizes $\gamma_k = 2/(k+2)$, Frank--Wolfe satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2} = O(1/k).
\end{equation}
This rate is the same as projected GD but without the projection oracle. With a fixed mixing parameter $\beta$ (as used here, $\gamma = 1 - \beta$), the algorithm still converges but the rate depends on $\beta$ and the problem geometry.
\end{theorem}

\noindent The \textbf{Frank--Wolfe gap} (also called the \emph{primal gap} or \emph{duality gap}) is defined as:
\begin{equation}
g_k := \max_{z \in \mathcal{X}}\, \nabla f(x_k)^T(x_k - z) = \nabla f(x_k)^T(x_k - z_k),
\end{equation}
where $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ is the LP vertex selected by the Frank--Wolfe step. By convexity:
\begin{equation}
f(x^{\star}) \geq f(x_k) + \nabla f(x_k)^T(x^{\star} - x_k) \implies f(x_k) - f^{\star} \leq \nabla f(x_k)^T(x_k - x^{\star}) \leq g_k.
\end{equation}
Hence $g_k$ is a valid (and computable) upper bound on the suboptimality $f(x_k) - f^{\star}$. This is particularly valuable: unlike gradient-based methods where $\|\nabla f(x_k)\| \to 0$ certificates require knowing $\mu$ to bound the objective gap, the FW gap $g_k$ directly bounds $f(x_k) - f^{\star}$ without any additional constants. The convergence theorem $f(x_k) - f^{\star} \leq 2LD^2/(k+2)$ can be equivalently stated as $g_k \leq 2LD^2/(k+2)$, meaning the duality gap decays at the same rate as the objective gap.

\subsection{Interior vs Boundary Behaviour}

\paragraph{Interior optimum.}
Frank--Wolfe converges but exhibits a ``zig-zag'' pattern: the LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ at each iteration because $\nabla f(x_k)$ changes sign as $x_k$ crosses the optimum projection. The convex combination gradually averages these jumps, converging at $O(1/k)$. Smaller $\beta$ (larger steps towards $z_k$) is faster initially but introduces more oscillation near the optimum. The \emph{standard step schedule} $\gamma_k = 2/(k+2)$ (corresponding to $\beta_k = k/(k+2)$, increasing from 0 to 1) would improve convergence by taking large steps early and small steps late.

\paragraph{Boundary optimum.}
Frank--Wolfe is naturally well-suited to boundary optima because $z_k$ consistently selects the vertex that minimises the linear objective, which tends to be on the same constraint boundary as the constrained optimum. Iterates approach the boundary without explicit projection, and the algorithm is \emph{projection-free}. The residual gap ($f = 0.36$ vs $f^{\star} = 0.25$ after 140 iterations) reflects the $O(1/k)$ rate: convergence to $\varepsilon$ precision requires $O(1/\varepsilon)$ iterations.

\begin{remark}[Fixed vs.\ decaying step schedule]
The implementation here uses a fixed mixing parameter $\beta$ (equivalently, a fixed step size $\gamma = 1 - \beta$ towards the LP vertex). The convergence theorem for Frank--Wolfe (Theorem above) applies to the \emph{decreasing} schedule $\gamma_k = 2/(k+2)$, which starts large (aggressive movement towards $z_k$) and decays as $O(1/k)$. With fixed $\gamma = 1 - 0.90 = 0.10$, the algorithm converges to an $O(\gamma)$-ball around the optimum rather than achieving arbitrarily precise convergence. The decaying schedule $\gamma_k = 2/(k+2)$ corresponding to $\beta_k = k/(k+2)$ (the Nesterov momentum-like increasing schedule) would improve the interior optimum convergence substantially. However, for the boundary case, the fixed $\beta$ works well because $z_k$ consistently identifies the correct boundary vertex.
\end{remark}

\noindent \textbf{Why Frank--Wolfe over Projected GD?} The LP subproblem $z_k = \arg\min_{x \in \mathcal{X}} g_k^T x$ is trivial for box constraints (coordinate-wise min/max), but for complex constraint sets such as the nuclear norm ball, the spectrotope, or the spectrahedron, Frank--Wolfe's LP subproblem often has faster specialised solvers than the projection oracle required by PGD. This is why Frank--Wolfe has found renewed interest in machine learning problems involving matrix constraints (e.g.\ low-rank matrix completion).

\begin{remark}[Frank--Wolfe and sparse iterates]
A distinctive property of Frank--Wolfe is that iterates $x_k = \sum_{j \leq k} \gamma_j z_j$ (as convex combinations of LP vertices) are automatically \emph{sparse}: if the vertices $z_j$ are vertices of a polytope with at most $r$ non-zero coordinates, then $x_k$ has at most $k$ non-zero coordinates. This sparsity property is exploited in problems such as boosting (where $\mathcal{X}$ is the convex hull of weak learners), low-rank matrix factorisation (where vertices are rank-1 matrices), and compressed sensing (where $\mathcal{X}$ is the $\ell^1$ ball, giving the Lasso). For the box constraint in Q6, the vertices are at most 2D unit vectors, so all iterates remain bounded --- but the sparsity structure is not exploited as usefully as in higher-dimensional problems.
\end{remark}

\begin{remark}[Frank--Wolfe away step: fixing the zig-zag for interior optima]
The zig-zag oscillation observed for the interior optimum (Figure~\ref{fig:q6_evolution}, top row) arises because the LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ as $\nabla f(x_k)$ changes direction across the optimum. The \emph{away-step Frank--Wolfe} variant (Lacoste-Julien and Jaggi, 2015) addresses this by maintaining the \emph{active set} $\mathcal{S}_k \subseteq \mathcal{X}$ of all LP vertices visited so far (i.e., the current convex combination decomposition), and at each iteration choosing between:
\begin{itemize}[nosep]
\item \textbf{FW step:} move towards $z_k^{\mathrm{FW}} = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ (standard step).
\item \textbf{Away step:} move \emph{away} from $z_k^{\mathrm{away}} = \arg\max_{z \in \mathcal{S}_k} \nabla f(x_k)^T z$ (the active vertex most opposed to descent), updating $x_{k+1} = x_k + \gamma_k(x_k - z_k^{\mathrm{away}})$.
\end{itemize}
Away steps enable the iterate to escape the zig-zag by dropping inactive vertices from the convex combination rather than building up oscillatory contributions. Remarkably, the away-step variant achieves \emph{linear convergence} $f(x_k) - f^{\star} \leq \mathcal{O}(\rho^k)$ for strongly convex objectives over polytopes --- a qualitative improvement over the $O(1/k)$ rate of standard Frank--Wolfe. The cost is maintaining the active-set decomposition (up to $O(k)$ vertices), which adds $O(k)$ overhead per iteration for a polytope with many vertices.
\end{remark}

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has systematically investigated six families of optimisation algorithms across three benchmark problems of increasing difficulty. The central theme connecting all questions is that algorithmic performance is fundamentally determined by the interplay between (a) the \emph{information} each method uses about the objective, (b) the \emph{structure} of the optimisation landscape (condition number, convexity, constraint geometry), and (c) the \emph{computational budget} available. The experiments consistently confirm the theoretical predictions: methods that exploit more information converge faster, subject to the stability constraints imposed by the landscape's curvature.

\paragraph{Adaptive step-size methods (Q1).}
The Polyak step size achieves the best convergence when $f^{\star}$ is known exactly (Rosenbrock: $f = 0.009$), but diverges catastrophically when $f^{\star}$ is misspecified --- a severe limitation in practice. Adagrad and Heavy Ball are robust alternatives, consistently matching or exceeding the GD baseline. RMSprop improves stability over Adagrad in non-stationary settings by exponentially forgetting old gradient information.

\paragraph{Momentum methods (Q2).}
Nesterov's accelerated gradient method, with its lookahead evaluation and theoretically optimal $O(1/k^2)$ convergence rate for smooth convex problems, outperforms all first-order methods without oracle knowledge of $f^{\star}$ on Rosenbrock. Adam, while computationally efficient and effective in deep learning, is less suited to the correlated non-quadratic structure of Rosenbrock's valley.

\paragraph{Newton's method (Q3).}
Newton's method demonstrates the power of second-order information: one-step exact convergence on the quadratic problem (Benchmark~A), and dramatically faster convergence on Benchmark~B and~C despite using only 20 iterations. The quadratic convergence rate doubles the number of correct digits per iteration in the neighbourhood of the optimum, but at $O(d^3)$ per-iteration cost that is prohibitive in high dimensions.

\paragraph{Derivative-free methods (Q4).}
Finite differences accurately approximate gradients when $\delta$ is well-chosen ($\delta = 0.05$), nearly matching exact gradient performance. Nesterov random search requires only 2 function evaluations per iteration and is dimension-independent in cost, but converges significantly more slowly due to the $O(1/d)$ factor in the expected gradient estimate. Nelder--Mead adapts its simplex geometry to the local landscape without gradient information, while grid search guarantees global optimum discovery at exponential cost in dimension.

\paragraph{Constrained optimisation (Q5).}
Projected GD immediately enforces feasibility as a hard constraint from the first iteration, at the cost of requiring a projection oracle. Penalty methods achieve feasibility asymptotically: large $\lambda$ enforces the constraint but requires smaller step sizes due to the increased Lipschitz constant of the penalised gradient. The augmented Lagrangian (method of multipliers) combines the dual variable update with a moderate penalty to avoid the ill-conditioning issue. For this problem (half-space constraint with trivial projection), PGD is clearly preferable. For problems where projection is expensive (e.g.\ nuclear norm ball), penalty or augmented Lagrangian methods may be preferred, or the Frank--Wolfe LP oracle (Question~6) should be used instead.

\paragraph{Frank--Wolfe and linear programming (Q6).}
PGD (Q5) and Frank--Wolfe (Q6) are complementary first-order methods: both achieve $O(1/k)$ convergence for $L$-smooth convex objectives over compact feasible sets, but use different oracles. PGD uses a \emph{projection oracle} ($O(d)$ for boxes and half-spaces; $O(d^2\min(m,n))$ for nuclear norm ball); Frank--Wolfe uses a \emph{linear minimisation oracle} ($O(d)$ for boxes; $O(d\min(m,n))$ for nuclear norm ball --- a factor of $\min(m,n)$ cheaper). The fixed mixing parameter $\beta$ leads to vertex oscillation for interior optima; the decaying schedule $\gamma_k = 2/(k+2)$ or the away-step variant (achieving linear convergence on strongly convex objectives) would improve this. For boundary optima, Frank--Wolfe naturally converges to the active constraint boundary without projection.

\paragraph{Overall recommendation.}
For smooth, well-conditioned problems: \textbf{Adam or Nesterov} (fast, robust, no oracle). For quadratic objectives: \textbf{Newton} (one-step). For extremely ill-conditioned problems with known $f^{\star}$: \textbf{Polyak}. For constrained optimisation with easy projections: \textbf{PGD}. For derivative-free settings in low dimension: \textbf{Nelder--Mead}. For high-dimensional derivative-free settings: \textbf{Nesterov random search}.

\paragraph{Practical hyperparameter selection principles.}
The experiments reveal universal principles for step-size selection across methods:
\begin{enumerate}[nosep]
\item \textbf{Stability constraint}: Always start from $\alpha < 2/L$ (GD, Heavy Ball), $\alpha < 1/L$ (Nesterov, PGD). Estimate $L$ from the Hessian trace or by line search. For Rosenbrock, $L \approx 1002$ forces $\alpha < 0.002$.
\item \textbf{Effective step scale}: Adaptive methods (Adagrad, Adam) require $\alpha_0 \sim 1/\|\nabla f(x_0)\|$ to give $O(1)$ effective first steps. Too small $\alpha_0$ leads to very slow initial progress; too large leads to instability before adaptivity kicks in.
\item \textbf{Momentum parameter}: For Heavy Ball on quadratics, $\beta^{\star} = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$. For Nesterov on general problems, $\beta_{\max} \in [0.85, 0.95]$ is robust. Higher $\beta_{\max}$ trades per-step stability for accumulated velocity.
\item \textbf{Adam defaults}: $\beta_1 = 0.9$, $\beta_2 = 0.999$, $\varepsilon = 10^{-8}$ are widely robust. Reducing $\beta_1$ to 0.8 (as used in Q2) provides more gradient responsiveness for structured problems.
\item \textbf{Penalty weight}: For penalty methods, $\lambda \sim L_f / D_{\text{constraint}}$ where $D_{\text{constraint}}$ is the typical constraint violation at the initial point. Here $L_f \approx 10$, $D_{\text{constraint}} \approx 0.3$ gives $\lambda \sim 33$; the used values ($\lambda \in [0.15, 4.5]$) are conservative, explaining why $\lambda = 0.15$ is insufficient.
\end{enumerate}

\paragraph{Theoretical hierarchy revisited.}
The experiments confirm the information hierarchy: \emph{second-order} (Newton) $>$ \emph{oracle} (Polyak with exact $f^{\star}$) $>$ \emph{optimal first-order} (Nesterov) $>$ \emph{adaptive first-order} (Adam, Adagrad) $>$ \emph{momentum} (Heavy Ball) $>$ \emph{plain gradient} (GD) $>$ \emph{zeroth-order} (NRS, Nelder--Mead, FD) for the ill-conditioned Rosenbrock benchmark. The hierarchy collapses for well-conditioned problems (Benchmark~A): all methods reach the noise floor within $O(\kappa \log(1/\varepsilon))$ iterations, confirming that condition number is the key determinant of method choice.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2                          # Monotone accumulation
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2  # Exponential moving average
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball (Polyak Momentum)}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)  # Increasing schedule
        lookahead = x + beta_k * z                   # Evaluate gradient ahead
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # Bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))  # Tikhonov regularisation
        try:
            p = np.linalg.solve(H_reg, g)      # Solve: H_reg p = g
        except np.linalg.LinAlgError:
            p = g                               # Fallback to gradient
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LP subproblem: select box vertex minimising g^T z
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # Convex combination (feasibility preserved)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Mini-Batch SGD}]
def mini_batch_sgd(X, y, theta0, alpha, batch_size, n_epochs):
    m = len(y)
    theta = theta0.copy().astype(float)
    f_hist, theta_hist = [], [theta.copy()]
    for epoch in range(n_epochs):
        perm = np.random.permutation(m)
        X_shuf, y_shuf = X[perm], y[perm]
        for start in range(0, m, batch_size):
            Xb = X_shuf[start:start+batch_size]
            yb = y_shuf[start:start+batch_size]
            g = Xb.T @ (Xb @ theta - yb) / len(yb)  # mini-batch gradient
            theta = theta - alpha * g
        f_hist.append(0.5 * np.mean((X @ theta - y)**2))
        theta_hist.append(theta.copy())
    return np.array(theta_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Projected Gradient Descent (half-space $x_1 \geq 0.5$)}]
def projected_gd(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x_new = x - alpha * grad_fn(x)          # gradient step
        x_new[0] = max(x_new[0], 0.5)           # project: clip x1 to [0.5, inf)
        x = x_new
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Random Search (zeroth-order)}]
def nesterov_random_search(loss_fn, x0, alpha, delta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        u = np.random.randn(len(x))
        u /= np.linalg.norm(u)                   # uniform random unit vector
        f0 = loss_fn(x)
        f1 = loss_fn(x + delta * u)
        g_hat = (f1 - f0) / delta * u            # directional derivative estimate
        x = x - alpha * g_hat
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}
