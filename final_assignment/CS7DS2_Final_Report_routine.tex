\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms across six
methodological categories applied to three benchmark problems: a linear regression quadratic loss
(Benchmark A), a toy neural network loss with sinusoidal perturbation (Benchmark B), and the
Rosenbrock function (Benchmark C). The six categories investigated are: (Q1)~adaptive step-size
methods (Polyak, Adagrad, RMSprop, Heavy Ball); (Q2)~momentum-based and stochastic methods
(Nesterov acceleration, Adam, mini-batch SGD); (Q3)~Newton's method exploiting second-order
curvature; (Q4)~derivative-free approaches (finite differences, Nesterov random search,
Nelder--Mead, grid search); (Q5)~constrained optimisation via projected gradient descent and
penalty methods; and (Q6)~the Frank--Wolfe conditional gradient algorithm for linearly-constrained
problems. Each method is evaluated across convergence speed, trajectory behaviour, theoretical
guarantees, and computational trade-offs. Newton's method achieves the lowest final value on
Rosenbrock in 50 iterations ($f = 4.6 \times 10^{-4}$) due to its quadratic convergence rate, while
Polyak and Nelder--Mead converge to near-zero with sufficient iterations. Projected gradient
descent enforces hard constraints from the first iterate, and Frank--Wolfe avoids projection by
exploiting the linear minimisation oracle. The analysis reveals that second-order information and
the accuracy of the known optimum are the two most influential factors for convergence on
ill-conditioned problems.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) to ensure reproducibility.
Before describing the algorithms, we characterise each benchmark in terms of properties that
determine algorithm performance: smoothness ($L$-Lipschitz gradient), strong convexity ($\mu$),
and condition number $\kappa = L/\mu$.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \in \mathbb{R}^2, \quad
y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss is:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
          = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a convex quadratic with constant Hessian $H = \frac{1}{m}X^T X \succ 0$.
The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$, and the smoothness constant
is $L = \lambda_{\max}(H)$, with $\mu = \lambda_{\min}(H) > 0$.
The noise floor satisfies $J^{\star} = J(\theta_{\text{OLS}}) \approx 0.4826$ (not zero, due to
observation noise). Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is
$H(x) = \operatorname{diag}(2 - \sin(x_1),\ 10)$.
Since $\sin(x_1) \in [-1, 1]$, the Hessian eigenvalues lie in $[1, 3]$ and $\{10\}$,
giving $L = 10$, $\mu \approx 1$, and condition number $\kappa \approx 10$. The function is
strongly convex despite the sinusoidal perturbation, since the Hessian remains positive definite
everywhere. The global minimum satisfies $2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$, yielding
$x_1^{\star} \approx 0.582$ and $x_2^{\star} = 2.0$, with $f^{\star} \approx 0.7244$.
Starting point: $x_0 = [-1, 4]^T$.

\begin{remark}
The contour plots in this report show the minimum marker at the numerically approximate value
$(0.61, 2.0)$ for plotting clarity. The theoretically correct value is $(0.582, 2.0)$.
\end{remark}

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is
\begin{equation}
\nabla^2 f(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $\lambda_1 \approx 1001.6$ and $\lambda_2 \approx 0.4$, giving
$\kappa \approx 2500$. Crucially, Rosenbrock is \emph{not} strongly convex globally: it is convex
only near the optimum, and the function's topology features a narrow curved valley along
$x_2 = x_1^2$ that makes gradient-based methods prone to slow traversal. Starting point:
$x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}

For constrained problems (Questions 5 and 6):
$\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.
This is a convex compact polytope, making the linear minimisation oracle (required for
Frank--Wolfe) solvable in closed form.

\section{Question 1: Adaptive Step-Size Methods \quad\normalfont\textit{[30 marks]}}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations and Theoretical Motivation}

Four adaptive methods are compared against a constant step-size gradient descent baseline for
120 iterations. The key motivation for adaptive methods is that a fixed step size $\alpha$ must
satisfy $\alpha < 2/L$ for convergence, which is conservative when curvature varies across
coordinates. Adaptive methods automatically scale the effective rate to local curvature.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \quad\Comment{Minimise upper bound on distance to optimum}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak Convergence]
For an $L$-smooth convex function with known $f^{\star}$, the Polyak step satisfies
$f(x_k) - f^{\star} \leq \frac{L\|x_0 - x^{\star}\|^2}{2k}$, i.e., an $O(1/k)$ rate.
For strongly convex functions ($\mu > 0$), linear convergence $O((1-\mu/L)^k)$ is achieved.
\end{theorem}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative sum of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The per-coordinate effective step $\alpha_0 / (\sqrt{G_k[i]} + \varepsilon)$ is large initially
(when accumulated gradients are small) and decays as $O(1/\sqrt{k})$ for coordinates with
consistently large gradients. This self-tuning is beneficial for sparse or heterogeneous gradients.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop's exponential moving average with decay $\beta = 0.9$ means the effective window spans
approximately $1/(1-\beta) = 10$ recent gradients. Unlike Adagrad, the denominator does not
diverge, preventing the effective step size from decaying to zero.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum coefficient $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Heavy Ball on Quadratics]
For a strongly convex quadratic with condition number $\kappa = L/\mu$, the Heavy Ball method
with optimal parameters $\alpha^* = 4/(\sqrt{L}+\sqrt{\mu})^2$,
$\beta^* = \bigl((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)\bigr)^2$ achieves linear convergence at
rate $\bigl(1 - 2/(\sqrt{\kappa}+1)\bigr)$, improving over GD's rate $\bigl(1 - 2/(\kappa+1)\bigr)$.
\end{theorem}

\subsection{Comparative Analysis}

\textbf{Why Polyak diverges on A and B but succeeds on C:}
The Polyak step size is $\alpha_k = (f(x_k) - f^{\star}) / (\|\nabla f(x_k)\|^2 + \varepsilon)$.
Near the optimum, $\|\nabla f(x_k)\| \to 0$ (approximately as $O(1/k)$ for strongly convex
functions). On Benchmark A, $f(x_k) \to J^{\star} \approx 0.4826 > 0 = f^{\star}$, so the numerator
remains bounded away from zero while the denominator vanishes, causing $\alpha_k \to \infty$.
On Benchmark C, both numerator and denominator decay proportionally so $\alpha_k$ remains bounded
and decreasing, consistent with the convergence guarantee.

\textbf{Why Adagrad ($f = 2.20$) outperforms GD ($f = 3.51$) on Benchmark C despite its much
larger $\alpha_0 = 0.45$:} Adagrad's per-coordinate adaptation effectively uses different step
sizes for the steep $x_2$-direction ($\approx$ curvature 200 near $(1,1)$) and the shallow
$x_1$-direction ($\approx$ curvature 2). This coordinate-wise scaling reduces the effective
condition number faced by the optimiser. GD must use the same $\alpha = 0.0012$ for both directions,
severely throttling progress in the shallow direction.

\textbf{Why RMSprop ($f = 3.12$) underperforms Adagrad on Benchmark C:} RMSprop requires a small
base rate $\alpha_0 = 0.0035$ for stability (versus Adagrad's $\alpha_0 = 0.45$). The exponential
moving average of RMSprop adapts to \emph{recent} gradients, while Adagrad accumulates all history.
For Rosenbrock, the gradients are consistently large in the same directions, so Adagrad's full
history provides a more stable and accurate per-coordinate scaling.

\begin{itemize}[nosep]
\item \textbf{Correctness of $f^{\star}$:} Polyak's convergence is theoretically guaranteed only
when $f^{\star}$ is specified exactly. A lower bound $f^{\star} < f_{\text{true}}^{\star}$ causes
divergence (Benchmarks A and B); the exact value $f^{\star} = 0$ (Benchmark C) yields the fastest
convergence in Q1. In practice, $f^{\star}$ is rarely known, limiting Polyak's applicability.

\item \textbf{Adagrad vs.\ RMSprop:} Adagrad's monotone accumulation of $G_k$ acts as a
``forgetting-free'' memory, making it uniformly stable but prone to step-size collapse on
non-stationary problems. RMSprop's exponential moving average prevents this collapse, making it
the preferred choice for deep learning where gradients are highly non-stationary. On our static
benchmark problems with consistent gradient structure, Adagrad performs comparably or better.

\item \textbf{Heavy Ball:} Momentum $\beta = 0.88$--$0.90$ effectively accelerates convergence on
moderately ill-conditioned problems. On Benchmark C, momentum helps the iterate sustain progress
along the valley floor between gradient updates: each gradient step points slightly uphill, but
the accumulated momentum from previous steps carries the iterate along the valley.

\item \textbf{GD baseline:} Stable but conservatively slow. Its step size is dictated by the
\emph{largest} curvature, causing slow progress in the directions of small curvature---a fundamental
limitation on anisotropic problems quantified by the condition number $\kappa = L/\mu$.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \quad\normalfont\textit{[30 marks]}}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations and Theoretical Rates}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient Descent}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead / extrapolation step}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball is that the gradient is evaluated at the \emph{extrapolated}
point $\tilde{x}_k = x_k + \beta_k z_k$, not at $x_k$. This ``lookahead'' corrects for the
direction of momentum before taking a step, enabling provably faster convergence.

The schedule $\beta_k = (k-1)/(k+2)$ is the standard Nesterov ``FISTA-style'' schedule. It
starts at $\beta_1 = 0$ (no momentum) and increases towards 1, asymptotically matching the
optimal $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$ schedule for strongly convex functions. The
$\beta_{\max}$ cap prevents the momentum from exceeding a safe threshold, ensuring stability on
non-convex problems like Rosenbrock. The increasing momentum schedule is necessary for the
$O(1/k^2)$ proof: at step $k$, the momentum is approximately $k/(k+3)$, and the proof requires
that the sequence of step sizes and momenta jointly satisfy a specific Lyapunov inequality.

\begin{theorem}[Nesterov Optimal Rate]
For an $L$-smooth convex function, Nesterov's accelerated method with $\alpha = 1/L$ and
$\beta_k = (k-1)/(k+2)$ satisfies
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2},
\end{equation}
i.e., an $O(1/k^2)$ rate. This is \emph{optimal} among first-order methods for smooth convex
functions (Nemirovski--Yudin lower bound). By contrast, gradient descent achieves only $O(1/k)$.
For strongly convex functions, Nesterov achieves linear convergence at rate
$\bigl(1 - \sqrt{\mu/L}\bigr)^k$, compared to GD's $\bigl(1 - \mu/L\bigr)^k$.
\end{theorem}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (biased)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (biased)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines the gradient momentum of Nesterov-style methods (via $m_t$) with per-coordinate
adaptive scaling (via $v_t$, as in RMSprop). The bias correction terms $(1 - \beta_1^t)$ and
$(1 - \beta_2^t)$ compensate for zero-initialisation, ensuring that early iterates are not
spuriously small. While Adam lacks the $O(1/k^2)$ theoretical guarantee of Nesterov (and can
in fact fail to converge for some convex problems, as shown by Reddi et al.\ 2018), it performs
excellently in practice on deep learning tasks due to its robustness to hyperparameter choice.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$,
and runs for 50 epochs. Data is shuffled at every epoch. The SGD gradient estimate is
\begin{equation}
\hat{g}_k = \frac{1}{b}\sum_{i \in \mathcal{B}_k}\nabla f_i(\theta),
\end{equation}
which is an unbiased estimate of the full gradient: $\mathbb{E}[\hat{g}_k] = \nabla J(\theta)$.
The variance of this estimate is
\begin{equation}
\operatorname{Var}(\hat{g}_k) = \frac{\sigma_{\text{grad}}^2}{b}, \quad
\text{where } \sigma_{\text{grad}}^2 = \frac{1}{m}\sum_{i=1}^m \|\nabla f_i(\theta) - \nabla J(\theta)\|^2.
\end{equation}
This variance decreases with batch size $b$: using $b = 40$ gives $8\times$ lower gradient variance
than $b = 5$. A smaller batch increases gradient variance but enables $m/b$ gradient steps per epoch
instead of 1 (for full-batch GD). This fundamental tradeoff means smaller batches converge faster
\emph{per epoch} but exhibit more residual noise near the minimum.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices into $\lfloor m/b \rfloor$ non-overlapping batches
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $\hat{g} \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\includegraphics[width=0.85\textwidth]{figures/q2_sgd_batch.pdf}
\caption{Q2: Mini-batch SGD with batch sizes $b = 5$ and $b = 40$ on Benchmark A. With $b = 5$,

\subsection{Local Approximations of $g(x) = x^4$}

At expansion point $x_0 = 0.25$, the Taylor series gives:
\begin{align}
g(x_0) &= (0.25)^4 = 0.003906, \\
g'(x_0) &= 4(0.25)^3 = 0.0625, \\
g''(x_0) &= 12(0.25)^2 = 0.75.
\end{align}

\textbf{First-order approximation:}
\begin{equation}
\hat{g}_1(x) = 0.003906 + 0.0625\,(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
\hat{g}_2(x) = 0.003906 + 0.0625\,(x - 0.25) + \tfrac{1}{2}(0.75)(x - 0.25)^2.
\end{equation}

The Newton step minimises $\hat{g}_2$, yielding the update rule:
\begin{equation}
x_{\text{new}} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} \approx 0.167.
\end{equation}
For $g(x) = x^4$ with minimum at $x^{\star} = 0$, Newton's method converges \emph{quickly} but
\emph{not in one step}, because $x^4$ is not quadratic. It achieves superlinear (specifically,
quadratic) convergence for smooth functions near the optimum. The subsequent iterations from
$x_0 = 0.25$ are:
\begin{align}
x_1 &= x_0 - \frac{4x_0^3}{12x_0^2} = x_0 - \frac{x_0}{3} = \frac{2}{3}x_0 \approx 0.167, \\
x_2 &= x_1 - \frac{x_1}{3} \approx 0.111, \quad
x_3 \approx 0.074, \quad x_4 \approx 0.049, \ldots
\end{align}
Each step reduces $x_k$ by factor $2/3$ in this case (linear convergence at rate $2/3$, because
$g(x) = x^4$ has a degenerate minimum where $g''(0) = 0$, so the standard quadratic convergence
does not apply here; the rate is slower). In general, for $g(x) = x^p$ with $p \geq 3$, Newton
converges linearly with rate $1 - 1/(p-1)$ at the degenerate minimum $x^{\star} = 0$.

\subsection{Newton's Method with Hessian Regularisation}

\begin{algorithm}[H]
\caption{Newton's Method with Hessian Regularisation (Damped Newton)}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0, 1]$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise to ensure positive definiteness}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Quadratic Convergence of Newton's Method]
Let $f$ be twice continuously differentiable and let $x^{\star}$ be a local minimum with
$\nabla^2 f(x^{\star}) \succ 0$. If $x_0$ is sufficiently close to $x^{\star}$ and $\alpha = 1$,
then the (pure) Newton iterates satisfy
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq C\,\|x_k - x^{\star}\|^2,
\end{equation}
for some constant $C > 0$ depending on $L$ and the minimum eigenvalue of $\nabla^2 f(x^{\star})$.
This \emph{quadratic convergence} means the number of correct decimal digits roughly doubles at
each iteration, compared to linear convergence of first-order methods.
\end{theorem}

Settings: GD runs 80 iterations; Newton uses 20 iterations with Hessian damping
$\lambda = 10^{-8}$ (added to ensure positive definiteness if $\nabla^2 f$ is indefinite).

\subsection{Computational Cost Discussion}

Newton's method requires: (1) computing $\nabla^2 f(x_k)$, which costs $O(d^2)$ in $d$ dimensions;
(2) solving $H_k p_k = g_k$, which costs $O(d^3)$ via Cholesky decomposition. For our 2D
benchmarks this is negligible, but for deep learning with millions of parameters, the $O(d^3)$
cost is prohibitive. Quasi-Newton methods (e.g., BFGS) address this by maintaining a low-rank
approximation $B_k \approx H_k^{-1}$ updated at each iteration:
\begin{equation}
B_{k+1} = B_k + \frac{(s_k - B_k r_k)s_k^T + s_k(s_k - B_k r_k)^T}{s_k^T r_k}
         - \frac{(s_k - B_k r_k)^T r_k}{(s_k^T r_k)^2} s_k s_k^T,
\end{equation}
where $s_k = x_{k+1} - x_k$ and $r_k = \nabla f(x_{k+1}) - \nabla f(x_k)$.
BFGS achieves superlinear convergence at cost $O(d^2)$ per iteration. The limited-memory variant
L-BFGS uses only the last $m$ (e.g., $m = 10$) pairs $(s_i, r_i)$, achieving $O(md)$ cost and
near-Newton performance on large-scale problems without explicitly storing $H_k^{-1}$.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of $\partial f/\partial x_i$:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\label{eq:fd}
\end{equation}
The total error consists of two competing terms: truncation error $O(\delta)$ (from ignoring
higher-order Taylor terms) and numerical precision error $O(\varepsilon_{\text{mach}}/\delta)$
(from floating-point cancellation). Minimising $C_1\delta + C_2\varepsilon_{\text{mach}}/\delta$
yields the optimal step:
\begin{equation}
\delta_{\text{opt}} = \sqrt{\frac{C_2}{C_1}\,\varepsilon_{\text{mach}}} \approx \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}.
\end{equation}
In practice, for smooth functions evaluated with $O(1)$ values, $\delta \approx 10^{-4}$ to
$10^{-3}$ is often more reliable because floating-point cancellation is less severe.
For Benchmark B near $x_0 = (-1, 4)$, the second derivative in the $x_1$ direction is
$g''_{11} = 2 - \sin(-1) \approx 2.84$. The truncation error in the gradient estimate is
$\approx \delta \cdot g''_{11}/2 \approx 0.05 \times 1.42 \approx 0.071$ for $\delta = 0.05$,
and $\approx 0.8 \times 1.42 \approx 1.14$ for $\delta = 0.8$. The gradient magnitude at $x_0$
is $|\nabla f|_1 = |2(-2) + \cos(-1)| \approx 3.46$, so $\delta = 0.05$ gives a relative error
of $0.071/3.46 \approx 2\%$ (acceptable), while $\delta = 0.8$ gives $33\%$ relative error
(unacceptable, causing wrong gradient direction).
The assignment tests $\delta = 0.05$ (practical good choice) and $\delta = 0.8$ (excessively large,
poor).

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Method)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \Comment{Stochastic directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Nesterov random search requires only \emph{two} function evaluations per iteration, independent
of dimension $d$. To compute the expectation of $\hat{g}_k$, let $u \sim \text{Uniform}(S^{d-1})$
(uniformly distributed on the unit sphere). Using the identity
$\mathbb{E}_{u \sim S^{d-1}}[u u^T] = \frac{1}{d} I_d$, the expected gradient estimate satisfies
\begin{equation}
\mathbb{E}[\hat{g}_k] = \mathbb{E}_u\!\left[\frac{f(x_k + \delta u) - f(x_k)}{\delta} u\right]
= \frac{1}{d}\nabla f(x_k) + O(\delta).
\end{equation}
The estimate is thus \emph{approximately} unbiased (up to $O(\delta)$ error); the factor $1/d$
means the effective step is $\alpha/d$ in terms of gradient magnitude. This $d$-fold reduction
in effective step size explains why convergence is $O(d/k)$ rather than $O(1/k)$:
in $d=2$ dimensions, the method converges at approximately half the GD speed per iteration.
In high dimensions ($d = 1000$), NRS would be 1000 times slower per iteration, though its
cost is still only 2 function evaluations regardless of $d$.

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and
iteratively transforms it through four operations:
\begin{enumerate}[nosep,label=\alph*.]
\item \textbf{Reflection} ($\alpha_r = 1.0$): reflect the worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2.0$): if reflection improves on the best, try extending further.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection does not improve, contract towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): as a last resort, shrink all vertices towards the best.
\end{enumerate}
No gradient information is required. Initial simplex step = 0.35, iterations = 160.

\begin{remark}
Nelder--Mead has no general convergence guarantees for $d \geq 2$. In practice, it often works
well on low-dimensional, smooth problems but can stagnate on pathological functions. On Rosenbrock,
the simplex tends to align along the valley, effectively performing a one-dimensional search along
the valley floor once it reaches the valley.
\end{remark}

\section{Question 5: Constrained Optimisation \quad\normalfont\textit{[30 marks]}}
\label{sec:q5}

\subsection{Problem Formulation and KKT Conditions}

We minimise Benchmark B subject to the half-space constraint $x_1 \geq 0.5$:
\begin{equation}
\min_{x \in \mathbb{R}^2}\ f(x_1, x_2) \quad \text{subject to}\quad x_1 \geq 0.5.
\end{equation}
Starting point: $x_0 = (0.2, 4.0)$ (infeasible). The unconstrained minimum of Benchmark B is at
$(0.582, 2.0)$, which satisfies $x_1^{\star} = 0.582 > 0.5$. Therefore, the constrained minimum
\emph{coincides} with the unconstrained minimum, and the constraint is not active at the solution.

The Karush--Kuhn--Tucker (KKT) conditions for this problem are:
\begin{align}
\nabla f(x^{\star}) + \mu(-e_1) &= 0 \quad \text{(stationarity)}, \\
x_1^{\star} - 0.5 &\geq 0 \quad \text{(primal feasibility)}, \\
\mu &\geq 0 \quad \text{(dual feasibility)}, \\
\mu(x_1^{\star} - 0.5) &= 0 \quad \text{(complementarity slackness)}.
\end{align}
Since the constraint is inactive at $x^{\star}$ (i.e., $x_1^{\star} = 0.582 > 0.5$), the
complementarity condition forces $\mu^{\star} = 0$, and the KKT conditions reduce to
$\nabla f(x^{\star}) = 0$---the unconstrained optimality condition.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Projection: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the constraint $\{x : x_1 \geq 0.5\}$, the projection is $\Pi_{\mathcal{X}}(z)_1 = \max(0.5, z_1)$,
which is exact and $O(1)$ to compute. Projected GD converges at the same rate as unconstrained GD
when the optimal point satisfies $\Pi_{\mathcal{X}}(x^{\star} - \alpha\nabla f(x^{\star})) = x^{\star}$
(i.e., the constraint is inactive at the solution, as here).

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $g \gets \nabla f(x_k)$;\quad if $x_{k,1} < 0.5$ then $g_1 \gets g_1 - \lambda$
    \State $x_{k+1} \gets x_k - \alpha\, g$
\EndFor
\end{algorithmic}
\end{algorithm}

Note that the penalty $\lambda\max(0, 0.5 - x_1)$ is a linear (exact penalty) rather than a
quadratic penalty. The gradient is $-\lambda e_1$ when $x_1 < 0.5$, creating a constant force
pushing towards feasibility. For exact quadratic penalty $\frac{\lambda}{2}\max(0, 0.5 - x_1)^2$,
the gradient force grows linearly with violation, but requires $\lambda \to \infty$ for exact
feasibility---the fundamental limitation of penalty methods.

\textbf{Key observations:}
\begin{itemize}[nosep]
\item \textbf{Projected GD} enforces hard feasibility at every iteration.
The projection $\max(0.5, \cdot)$ is $O(1)$ for this simple constraint, but can be expensive
for general constraints (e.g., projection onto a simplex requires $O(d\log d)$; projection onto
a general polytope is an LP).

\item \textbf{Penalty methods} avoid projection at the cost of only \emph{approximate} feasibility.
As $\lambda \to \infty$, feasibility is enforced more precisely, but the penalised gradient
grows large near the constraint, requiring smaller $\alpha$ for stability.

\item Since the true optimum lies in the interior of the feasible set (constraint inactive at
$x^{\star}$), both methods converge to the same solution. The methods would differ significantly
if the optimum lay on or outside the constraint boundary: in that case, projected GD would
converge to the feasibility-optimal point while the penalty method would require $\lambda \to \infty$.

\item \textbf{Augmented Lagrangian (Multiplier Method):} A more sophisticated alternative combines
penalty and dual ascent:
\begin{equation}
\mathcal{L}_{\rho}(x, \mu) = f(x) + \mu(0.5 - x_1) + \frac{\rho}{2}\max(0, 0.5 - x_1)^2,
\end{equation}
where $\mu$ is the Lagrange multiplier (dual variable) updated by $\mu_{k+1} = \max(0, \mu_k +
\rho(0.5 - x_1^k))$. The augmented Lagrangian achieves exact feasibility without $\lambda \to \infty$,
combining the stability of penalty methods with the exactness of dual methods.
\end{itemize}

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm \quad\normalfont\textit{[30 marks]}}
\label{sec:q6}
% ============================================================

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, since $a$ has all
positive components, the minimum over $\mathcal{X}$ is attained at the coordinate-wise lower bounds:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = a^T x^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates the \emph{fundamental theorem of linear programming}: the optimum of a linear
objective over a bounded convex polytope is attained at a \emph{vertex}. For a box constraint,
each vertex corresponds to a point at a corner of the box, and the optimal vertex is the one
that minimises $a^T x$ coordinatewise.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q6_lp.pdf}
\caption{Q6-I: Feasible region (blue box) with linear objective contours $f(x) = x_1 + 2x_2$.
The parallel contour lines have slope $-1/2$ (from $\mathrm{d}x_2/\mathrm{d}x_1 = -a_1/a_2 = -1/2$).
The minimum occurs at the lower-left corner $(0.5, -5)$, where both coordinates are at their
lower bounds because the objective increases in both coordinate directions.}
\label{fig:q6_lp}
\end{figure}

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination $\Rightarrow x_{k+1} \in \mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

The linear minimisation oracle (LMO) replaces the projection oracle of projected GD.
For a box constraint, the LMO decomposes coordinate-wise:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring $O(d)$ operations. Unlike projection, this oracle always returns a \emph{vertex} of
$\mathcal{X}$, and the iterates are always convex combinations of feasible vertices, ensuring
$x_k \in \mathcal{X}$ without explicit projection.

\begin{definition}[Frank--Wolfe Gap]
The Frank--Wolfe gap at iterate $x_k$ is defined as
\begin{equation}
g_{\text{FW}}(x_k) = \nabla f(x_k)^T (x_k - z_k) \geq 0.
\end{equation}
For smooth convex $f$, $g_{\text{FW}}(x_k) = 0$ implies $x_k$ is optimal. It serves as a
\emph{certificate} of approximate optimality and guides the stopping criterion.
\end{definition}

\begin{theorem}[Frank--Wolfe Convergence]
Let $f$ be $L$-smooth and convex over the compact convex set $\mathcal{X}$ with diameter $D =
\max_{x,y \in \mathcal{X}}\|x - y\|$. Frank--Wolfe with step size $\gamma_k = 2/(k+2)$ achieves
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2},
\end{equation}
i.e., an $O(1/k)$ convergence rate. This is the same rate as gradient descent but the iterates
remain in $\mathcal{X}$ by construction, without requiring a projection oracle.
\end{theorem}

In the implementation, we use a \emph{fixed} $\beta$ (retaining $\beta$ fraction of $x_k$ and
$(1-\beta)$ fraction of $z_k$), which corresponds to a constant step size $\gamma = 1-\beta$.

\subsection{Interior Optimum: $f(x) = (x_1 - 1)^2 + (x_2 - 5)^2$}

The unconstrained minimum $(1, 5)$ lies inside $\mathcal{X} = [0.5, 5] \times [-5, 10]$.
Starting point: $x_0 = (1, 1)$. Two mixing parameters tested: $\beta = 0.90$ (larger step
$\gamma = 0.10$) and $\beta = 0.985$ (smaller step $\gamma = 0.015$).

\subsection{Interior vs.\ Boundary Behaviour and Convergence Analysis}

\textbf{Interior optimum:} The characteristic ``zig-zag'' convergence of Frank--Wolfe arises
because $z_k$ jumps between extreme vertices of $\mathcal{X}$ as the gradient changes sign.
For the interior optimum at $(1, 5)$ starting from $(1, 1)$, the gradient
$\nabla f(x_k) = 2(x_k - (1,5)^T)$ points away from $(1,5)$. Initially the gradient's $x_2$
component is large (pointing downward, since $x_{k,2} < 5$), so the LMO selects $z_{k,2} = 10$
(upper bound). As $x_{k,2}$ overshoots 5, the $x_2$ gradient reverses, causing $z_k$ to jump to
the lower bound, hence the oscillation. The FW gap at the starting point is
$g_{\text{FW}}(x_0) = \nabla f(x_0)^T(x_0 - z_0) = (0, -8)^T((1,1)^T - (1,10)^T) = 72$,
providing a certificate that the starting optimality gap is at most $72/L$. Smaller steps
($\beta = 0.985$) reduce oscillation but slow early convergence.

\textbf{Boundary optimum:} Frank--Wolfe is better suited to boundary optima because the LMO
consistently returns the vertex in the direction of the constrained minimum. For $f(x) = \|x\|^2$
with $\nabla f(x) = 2x$, since $x_1 > 0$ the LMO always selects $z_1 = 0.5$ (lower bound),
continuously driving $x_{k,1}$ towards 0.5. The final gap of
$f = 0.364 > f^{\star} = 0.25$ after 140 iterations is partly due to the fixed $\beta = 0.93$
not satisfying the standard $\gamma_k = 2/(k+2)$ schedule. With the theoretically optimal
decaying step size $\gamma_k = 2/(k+2)$, convergence to $f^{\star}$ would be guaranteed at
rate $O(LD^2/(k+2)) = O(1/k)$.

\textbf{Comparison with Projected GD:} Frank--Wolfe is preferred when (1) the LMO is cheaper
than the projection oracle (e.g., for the nuclear norm ball, projection requires SVD at $O(d^3)$
but the LMO is $O(d^2)$), and (2) the domain is a polytope where the solution is inherently sparse
(e.g., the simplex in LASSO, the unit ball in compressed sensing). Projected GD is preferred
when a closed-form projection exists (as in Q5's half-space) and when a linear convergence rate
(achievable by projected GD on strongly convex $f$) is needed. Frank--Wolfe achieves only $O(1/k)$
even for strongly convex objectives, compared to projected GD's $O((1-\mu/L)^k)$ linear rate.

\item \textbf{Ill-conditioned problems} (large $\kappa$, Benchmark C): Newton's method is
the most effective if $d$ is small and the Hessian is available. When gradients are available
but the Hessian is not, Nesterov's acceleration provides the best first-order guarantees.
Polyak step with known $f^{\star}$ is competitive.

\item \textbf{Noisy/stochastic settings:} Mini-batch SGD with moderate batch sizes ($b = 40$)
offers the best balance of convergence speed and gradient variance. Very small batches ($b = 5$)
produce faster early progress but higher variance near the minimum.

\item \textbf{No gradient available:} For $d \leq 5$, Nelder--Mead is highly competitive.
For moderate $d$ with many iterations available, Nesterov random search scales better.
Grid search should only be used for $d \leq 3$.

\item \textbf{Constrained problems with simple constraints:} Projected gradient descent with
its closed-form projection is most efficient. For complex constraints where the projection is
expensive, Frank--Wolfe's LMO may be cheaper.
\end{enumerate}

\subsection{Key Insights from the Comparative Study}

The central insight from comparing methods across all six questions is that \textbf{the quality
of information used at each iterate determines the convergence rate}:
\begin{itemize}[nosep]
\item \emph{Zeroth-order} (function values only, Grid Search, NRS): $O(d/k)$ or $O(N^d)$ total.
\item \emph{First-order, non-adaptive} (GD): $O(1/k)$ or linear, limited by condition number $\kappa$.
\item \emph{First-order, adaptive} (Adagrad, RMSprop, Adam): Effectively reduce $\kappa$ via
coordinate-wise scaling; faster in practice but no theoretical improvement over Nesterov.
\item \emph{First-order, momentum} (Nesterov): $O(1/k^2)$ convex, $O((1-1/\sqrt{\kappa})^k)$ SC.
This is the optimal first-order rate and is achievable without any Hessian information.
\item \emph{Second-order} (Newton): Quadratic convergence $O(C^{2^k})$ near the optimum; requires
Hessian computation and inversion.
\end{itemize}
On the ill-conditioned Rosenbrock function (condition number $\kappa \approx 2500$), the
advantage of Newton's method is most dramatic because it fully exploits the curvature mismatch.
The practical takeaway is: use the \emph{cheapest} method that achieves the required accuracy.

\subsection{Limitations and Future Directions}

Several important aspects are beyond the scope of this study:
\begin{itemize}[nosep]
\item All experiments are in $d = 2$ dimensions; high-dimensional behaviour may differ substantially
(e.g., Adam's advantage over Nesterov typically appears in large-$d$ settings with sparse gradients,
and NRS becomes impractically slow relative to FD methods).
\item The hyperparameters were manually tuned; a fair comparison would require systematic
grid search or Bayesian optimisation over hyperparameter spaces.
\item Stochastic versions of Nesterov (SARAH, SPIDER) and Newton (sub-sampled Newton) may offer
improvements for large-scale problems by reducing gradient variance at lower cost than full-batch.
\item Line-search (e.g., Armijo backtracking) and trust-region variants of Newton and gradient
descent were not explored and would reduce sensitivity to the step-size hyperparameter,
particularly on Rosenbrock where the optimal step size changes dramatically across the domain.
\item Convergence to the \emph{global} minimum is not guaranteed for any of the first-order or
second-order methods on non-convex problems; Rosenbrock is convex (though ill-conditioned), but
for truly non-convex objectives, derivative-free global optimisation (e.g., simulated annealing,
genetic algorithms) or multi-start strategies would be necessary.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient Descent}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Hessian Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (Conditional Gradient)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
