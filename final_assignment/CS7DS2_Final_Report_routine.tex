\begin{abstract}
This report presents a systematic empirical and theoretical investigation of twelve optimisation algorithms across three benchmark problems of increasing difficulty: a linear regression quadratic loss (condition number $\kappa = 1.12$), a toy neural network loss with sinusoidal perturbation ($\kappa = 6.90$), and the ill-conditioned Rosenbrock function ($\kappa = 2508$). Six algorithm families are studied: (1) adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball); (2) accelerated and stochastic methods (Nesterov, Adam, mini-batch SGD); (3) Newton's method with second-order local approximations; (4) derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead simplex, grid search); (5) constrained optimisation via projected gradient descent and quadratic penalty methods; and (6) the Frank--Wolfe conditional gradient algorithm for linear-constrained problems. Each method is evaluated in terms of objective convergence, gradient norm decay, trajectory behaviour, computational cost, and theoretical convergence rate. Key findings: (i) the Polyak step size is highly sensitive to $f^{\star}$ misspecification, diverging on Benchmarks A and B (to $f = 4.77$) but achieving the best result on Rosenbrock ($f = 0.024$, correct $f^{\star} = 0$); (ii) Nesterov's $O(1/k^2)$ rate provides a $7.1\times$ improvement over GD on the ill-conditioned Rosenbrock, confirmed empirically by slope analysis; (iii) Newton's method converges in a single step for quadratics, approximately 5 steps for near-quadratic problems, and achieves $5.4\times$ better final objective on Rosenbrock in $25\%$ of the iterations — with the Hessian eigenvalue analysis confirming that the local condition number $\approx 2508$ persists throughout the trajectory; (iv) finite-difference approximation closely tracks exact gradients for $\delta = 0.05$ (truncation error $O(\delta)$) but degrades for $\delta = 0.8$; central differences improve accuracy to $O(\delta^2)$; (v) KKT analysis confirms the constraint is inactive at the optimum ($\mu^{\star} = 0$), explaining why all constrained methods converge to the same $f^{\star} = 0.7244$; (vi) Frank--Wolfe achieves $O(1/k)$ convergence in the FW gap, providing projection-free optimality certificates applicable to spectahedra, nuclear norm balls, and flow polytopes. All experiments use fixed seed \texttt{np.random.seed(42)} and are fully reproducible.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments are conducted with a fixed random seed (\texttt{np.random.seed(42)}) to ensure full reproducibility across methods. Three benchmark functions of increasing difficulty are employed, together with one constrained feasible set.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
\label{ssec:bench_A}

A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated as:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^{T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\label{eq:bench_A_data}
\end{equation}
The mean squared error loss function is:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\label{eq:bench_A_loss}
\end{equation}
This is a \emph{strongly convex quadratic} with constant Hessian $H = \frac{1}{m}X^T X \succ 0$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$ and the unique minimiser is the ordinary least squares (OLS) solution $\theta^{\star}_{\text{OLS}} = (X^TX)^{-1}X^Ty \approx [3.028,\, 4.017]^T$. The Hessian eigenvalues are approximately $\{0.925, 1.033\}$, giving condition number $\kappa \approx 1.12$ --- an exceptionally well-conditioned problem where any reasonable first-order method converges rapidly. Starting from $\theta_0 = [0, 0]^T$, the initial loss is $J(\theta_0) \approx 13.1$ (dominated by the $\theta^{\star} \cdot x$ term). Due to observation noise $\varepsilon \sim \mathcal{N}(0,1)$, the minimum achievable loss is $J^{\star} = J(\theta_{\text{OLS}}) \approx 0.4826 \approx \sigma^2/2$, an irreducible noise floor that no algorithm can improve upon.

\subsection{Benchmark B: Toy Neural Network Loss}
\label{ssec:bench_B}

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\label{eq:bench_B}
\end{equation}
The gradient and Hessian are:
\begin{align}
\nabla f &= \bigl[2(x_1 - 1) + \cos(x_1),\; 10(x_2 - 2)\bigr]^T, \\
\nabla^2 f &= \operatorname{diag}\bigl(2 - \sin(x_1),\; 10\bigr).
\label{eq:bench_B_hess}
\end{align}
The Hessian is diagonal, with eigenvalues ranging between $[1, 3]$ for the $x_1$-direction (since $2 - \sin(x_1) \in [1, 3]$) and exactly $10$ for the $x_2$-direction. At the minimum $x_1^{\star} \approx 0.582$, the eigenvalues are $(2 - \sin(0.582), 10) \approx (1.450, 10)$, giving condition number $\kappa \approx 6.90$ (moderate anisotropy). The exact minimum satisfies $2(x_1 - 1) + \cos(x_1) = 0$, solved numerically to give $x_1^{\star} = 0.58244$ and $x_2^{\star} = 2.0$, with $f^{\star} = 0.72442$ exactly. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\label{ssec:bench_C}

\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\label{eq:bench_C}
\end{equation}
The gradient is:
\begin{align}
\frac{\partial f}{\partial x_1} &= -2(1 - x_1) - 400x_1(x_2 - x_1^2), \\
\frac{\partial f}{\partial x_2} &= 200(x_2 - x_1^2),
\end{align}
and the Hessian is:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the global minimum $(1, 1)$, the Hessian is $\nabla^2 f(1,1) = \bigl[\begin{smallmatrix}802 & -400 \\ -400 & 200\end{smallmatrix}\bigr]$, with eigenvalues $\{0.3994, 1001.6\}$, giving condition number $\kappa \approx 2508$. This extreme ill-conditioning, combined with a narrow curved valley satisfying $x_2 \approx x_1^2$, makes the Rosenbrock function one of the most challenging standard benchmark problems in unconstrained optimisation. The high curvature across the valley (${\approx}1000$) combined with very low curvature along the valley floor (${\approx}0.4$) forces gradient methods to take tiny steps to avoid oscillating across the valley. Starting point: $x_0 = [-1, 1]^T$.

\textbf{Physical interpretation:} The Rosenbrock valley geometry mimics structures common in machine learning: the cross-valley direction (${\sim}1000\times$ steeper) represents strong coupling between parameters, while the flat valley floor (${\sim}0.4$ curvature) represents a nearly flat loss landscape near solutions. Such anisotropy is characteristic of deep neural networks, where layers interact strongly but the loss surface along solution manifolds is nearly flat — motivating the use of Nesterov momentum and Adam in practice.

\subsection{Condition Number and Convergence Rate}
\label{ssec:kappa}

A fundamental insight connecting all three benchmarks is the relationship between condition number $\kappa$ and first-order convergence rate. For gradient descent on a strongly convex $L$-smooth function:
\begin{equation}
f(x_k) - f^{\star} \leq \left(\frac{\kappa - 1}{\kappa + 1}\right)^{2k}(f(x_0) - f^{\star}).
\end{equation}
The per-iteration contraction factor $\left(\frac{\kappa-1}{\kappa+1}\right)^2$ ranges from $0.003$ (Benchmark A, $\kappa = 1.12$) to $0.998$ (Benchmark C, $\kappa = 2508$), explaining the dramatic difficulty increase across benchmarks: Benchmark A contracts by $>99\%$ per step, while Benchmark C contracts by only $0.16\%$ per step.

\begin{remark}[Nesterov's Lower Bound]
Nesterov proved that for any first-order method applied to an $L$-smooth $\mu$-strongly convex function, at least $k = \Omega(\sqrt{\kappa}\log(1/\epsilon))$ gradient evaluations are necessary to achieve $\epsilon$-accuracy. Gradient descent requires $O(\kappa \log(1/\epsilon))$ evaluations, while Nesterov's accelerated gradient achieves the optimal $O(\sqrt{\kappa}\log(1/\epsilon))$. For $\kappa = 2508$, this is the difference between $2508T$ and $50T$ effective iterations --- a $50\times$ theoretical speedup that motivates acceleration on ill-conditioned problems.
\end{remark}

\subsection{Theoretical Background}
\label{ssec:q1_theory}

Gradient descent with a fixed step size $\alpha \leq 1/L$ satisfies the convergence bound (for $L$-smooth strongly convex functions with modulus $\mu$):
\begin{equation}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k (f(x_0) - f^{\star}),
\end{equation}
where the contraction factor depends on the condition number $\kappa = L/\mu$. For ill-conditioned problems, this linear rate becomes impractically slow, motivating adaptive step-size strategies. The optimal constant step size is $\alpha^{\star} = 2/(L + \mu)$, giving the tightest GD contraction factor $((\kappa - 1)/(\kappa + 1))^k$. For Benchmark C ($\kappa \approx 2508$), the base factor is $(\kappa-1)/(\kappa+1) \approx 0.9992$, giving a bound of $0.9992^{2k}$ after $k$ iterations: achieving a $10^{-3}$ error reduction requires $k \approx 4300$ gradient descent steps. For large-step Polyak ($f^{\star}$ known exactly), the convergence is $O(1/k)$ for convex functions without the condition number dependence --- a qualitatively better rate in the ill-conditioned regime.

\subsection{Algorithm Formulations}
\label{ssec:q1_algorithms}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive step proportional to suboptimality gap}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is theoretically optimal for subgradient methods: it minimises the distance to $x^{\star}$ at each step, provided $f^{\star}$ is known exactly. The update rule follows from minimising $\|x_{k+1} - x^{\star}\|^2 = \|x_k - \alpha g_k - x^{\star}\|^2$ over $\alpha$, yielding $\alpha_k^{\text{opt}} = \langle g_k, x_k - x^{\star} \rangle / \|g_k\|^2$, which is bounded above by $(f(x_k) - f^{\star}) / \|g_k\|^2$ via the subgradient inequality.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's effective per-coordinate step size is $\alpha_0 / \sqrt{\sum_{s=0}^{k} g_{s,i}^2}$, which \emph{decays monotonically}. Coordinates with persistently large gradients receive smaller updates, providing implicit adaptation to the local curvature without requiring problem-specific knowledge of $L$. The monotone decay is both a strength (stability, no divergence) and a weakness: in the online learning interpretation, Adagrad achieves regret bound $O(\sqrt{T})$ over $T$ rounds, matching the minimax optimal rate for stochastic subgradient methods. However, the denominator grows without bound, causing premature step-size decay on non-stationary or long-horizon problems --- the central motivation for RMSprop's exponential forgetting.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's monotone decay by replacing the cumulative sum with an exponential moving average (EMA) controlled by $\beta$. This allows the effective step size to recover when gradient magnitudes decrease, making RMSprop better suited to non-stationary objectives.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Velocity accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method incorporates a momentum term $z_k$ that accumulates a weighted history of past gradients. For strongly convex quadratics with $\mu \leq \nabla^2 f \leq LI$, the theoretically optimal parameters are:
\begin{equation}
\alpha^{\star} = \frac{4}{(\sqrt{L} + \sqrt{\mu})^2}, \quad \beta^{\star} = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2,
\end{equation}
yielding convergence rate $O\!\left(\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^{2k}\right)$, which improves upon GD's $O\!\left(\left(\frac{\kappa-1}{\kappa+1}\right)^k\right)$ by taking the square root of the condition number. For non-quadratic objectives, Heavy Ball lacks global convergence guarantees, explaining why it may oscillate on Benchmark B's sinusoidal component.

\subsection{Comparative Analysis}
\label{ssec:q1_analysis}

\begin{itemize}[nosep]
\item \textbf{Polyak step size:} Theoretically optimal when $f^{\star}$ is exactly known, achieving the best convergence on Rosenbrock ($f = 0.024$). However, even a small error in $f^{\star}$ causes significant instability: on Benchmark A (where $f^{\star}_{\text{true}} = 0.4826$ but $0$ is specified), the method diverges to $f \approx 4.77$ as $\alpha_k \to \infty$. On Benchmark B ($f^{\star}_{\text{true}} = 0.7244$), the large initial function value $f(x_0) \approx 23.2$ keeps step sizes bounded initially, but the method still settles at $f \approx 0.98$, well above the true minimum. In practice, exact $f^{\star}$ is rarely known, severely limiting applicability.

\item \textbf{Adagrad:} Provides uniform improvement over GD on all benchmarks (except Rosenbrock due to the monotone decay of effective step size). The per-coordinate scaling adapts to curvature anisotropy, making it particularly suitable for sparse gradients and problems with heterogeneous feature scales.

\item \textbf{RMSprop:} More robust than Adagrad for non-stationary objectives (e.g., deep learning), but on these smooth benchmarks it underperforms. The fixed forgetting factor $\beta = 0.9$ introduces a trade-off: too large causes slow adaptation; too small loses memory of historical gradients.

\item \textbf{Heavy Ball:} Consistently the best-performing adaptive method (among those with correct $f^{\star}$ handling), matching GD on Benchmarks A and B while significantly outperforming it on Benchmark C ($f = 1.40$ vs $3.51$). The momentum term allows it to build speed along persistent gradient directions (e.g., the Rosenbrock valley).

\item \textbf{Stability hierarchy:} GD $\succ$ Adagrad $\succ$ RMSprop $\succ$ Heavy Ball $\succ$ Polyak (when $f^{\star}$ misspecified). The Polyak method transitions from best to worst depending on knowledge of $f^{\star}$.
\end{itemize}

\begin{theorem}[Adagrad Online Regret Bound (Duchi et al., 2011)]
For a convex objective with subgradients $g_k$ bounded by $\|g_k\|_{\infty} \leq G$, Adagrad with base rate $\alpha_0$ achieves cumulative regret over $T$ rounds:
\begin{equation}
R_T = \sum_{k=1}^T [f(x_k) - f(x^{\star})] \leq \frac{\|x_1 - x^{\star}\|_{\infty}^2}{2\alpha_0} \sum_{i=1}^d \sqrt{\sum_{k=1}^T g_{k,i}^2} \leq \frac{\|x_1 - x^{\star}\|_{\infty}^2\, d\, G\, \sqrt{T}}{2\alpha_0},
\end{equation}
giving per-round average regret $O(\sqrt{T}/T) = O(1/\sqrt{T})$, which is minimax optimal for online convex optimisation under subgradient feedback. The key insight is that Adagrad adapts to the ``effective dimension'' of the gradient sequence: if the gradients are sparse (few coordinates have large $g_{k,i}$), the regret is much smaller than the worst-case bound.
\end{theorem}

\subsection{Theoretical Background}
\label{ssec:q2_theory}

Standard gradient descent on $L$-smooth convex functions achieves an $O(1/k)$ convergence rate: $f(x_k) - f^{\star} \leq 2L\|x_0 - x^{\star}\|^2 / k$. Nesterov's accelerated gradient method achieves the optimal $O(1/k^2)$ rate: $f(x_k) - f^{\star} \leq 2L\|x_0 - x^{\star}\|^2 / (k+1)^2$. This is information-theoretically optimal for first-order methods on convex functions: Nesterov proved that no deterministic first-order algorithm can achieve $f(x_k) - f^{\star} = o(L\|x_0-x^{\star}\|^2/k^2)$ in the worst case. The improvement over GD is most dramatic for smooth functions: GD requires $O(L/\varepsilon)$ iterations while NAG requires only $O(\sqrt{L/\varepsilon})$ --- a square-root speedup.

\subsection{Algorithm Formulations}
\label{ssec:q2_algorithms}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\; \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point (key innovation)}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball is the \emph{lookahead evaluation}: the gradient is evaluated at the predicted future position $\tilde{x}_k = x_k + \beta_k z_k$ rather than the current position $x_k$. This anticipatory gradient allows the algorithm to ``correct its course'' before overshooting, and is responsible for achieving the $O(1/k^2)$ optimal rate. The schedule $\beta_k = (k-1)/(k+2)$ is derived from Nesterov's original sequence $t_{k+1} = (1 + \sqrt{1 + 4t_k^2})/2$, which satisfies $1/t_{k+1}^2 \leq 2/(k+2)^2$.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q2_nesterov_schedule.pdf}
\caption{Q2: Nesterov momentum schedule $\beta_k = \min((k-1)/(k+2), \beta_{\max})$ for $\beta_{\max} = 0.90$. The momentum increases gradually from 0 toward $\beta_{\max}$, reaching saturation around iteration 25. The gradual ramp-up prevents the instability that would arise from applying full momentum from the first iteration, while the asymptotic value provides near-optimal acceleration.}
\label{fig:q2_nesterov_schedule}
\end{figure}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scaling)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction for initialisation at zero}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam unifies momentum ($m_t$, like Heavy Ball) with per-coordinate adaptive scaling ($v_t$, like RMSprop). The bias correction terms $1/(1-\beta_1^t)$ and $1/(1-\beta_2^t)$ are essential in early iterations: without them, both EMA sequences are initialised at zero, so $m_t \approx (1-\beta_1)g_1$ and $v_t \approx (1-\beta_2)g_1^2$ are much smaller than the true first and second moments. The bias-corrected estimates $\hat{m}_t = m_t/(1-\beta_1^t)$ and $\hat{v}_t = v_t/(1-\beta_2^t)$ provide unbiased estimates of $\mathbb{E}[g_t]$ and $\mathbb{E}[g_t^2]$ respectively. As $t \to \infty$, the corrections vanish and Adam reduces to RMSprop with momentum.

\begin{remark}[Adam Convergence]
The original Adam paper (Kingma \& Ba, 2015) claimed an $O(1/\sqrt{T})$ regret bound, but Reddi et al.\ (2018) identified a flaw: Adam can \emph{diverge} on simple convex problems due to non-monotone effective step sizes. The fix (AMSGrad) uses a running maximum of $\hat{v}_t$ in the denominator. In practice, Adam with tuned $\beta_1, \beta_2, \alpha$ is highly effective in deep learning, but its convergence on general convex problems remains an open area. Our results confirm this empirically: Adam performs well on smooth, well-conditioned benchmarks (A, B) but lags behind Nesterov on the highly non-quadratic Rosenbrock function (C), where the second-moment estimate adapts slowly.
\end{remark}

\subsection{Why Nesterov Outperforms Heavy Ball on Rosenbrock}
\label{ssec:q2_nesterov_vs_hb}

A crucial theoretical distinction separates Nesterov (Section~\ref{ssec:q2_algorithms}) from Heavy Ball (Section~\ref{ssec:q1_algorithms}): while both methods use momentum, only Nesterov's accelerated gradient achieves the $O(1/k^2)$ optimality guarantee. The key innovation is the \emph{lookahead gradient evaluation}:
\begin{itemize}[nosep]
\item \textbf{Heavy Ball} evaluates $\nabla f(x_k)$ at the current iterate, then moves with momentum: $x_{k+1} = x_k - \alpha \nabla f(x_k) + \beta(x_k - x_{k-1})$. This can overshoot, and the convergence guarantee applies only to strongly convex quadratics.
\item \textbf{Nesterov} evaluates the gradient at the \emph{predicted} next position $\tilde{x}_k = x_k + \beta_k z_k$, allowing it to ``pre-correct'' before committing to the momentum step. This foresight is precisely why Nesterov achieves $O(1/k^2)$.
\end{itemize}
On Rosenbrock ($\kappa \approx 2508$), Nesterov reaches $f = 0.479$ after 150 iterations while Heavy Ball (Q1, 120 iterations) reaches $f = 1.400$. The Nesterov advantage ($3\times$ better) stems from the lookahead correctly navigating the curved valley rather than oscillating across it.

\subsection{Mini-Batch Stochastic Gradient Descent}
\label{ssec:q2_sgd}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, and 50 epochs with full data shuffling at each epoch. The gradient estimate for mini-batch $\mathcal{B}_t$ is:
\begin{equation}
\hat{g}_t = \frac{1}{b} X_{\mathcal{B}_t}^T (X_{\mathcal{B}_t} \theta_t - y_{\mathcal{B}_t}).
\end{equation}
This is an \emph{unbiased} estimator of the full gradient $\nabla J(\theta)$, with variance $\text{Var}[\hat{g}_t] \propto \sigma^2_{\text{grad}} / b$, which decreases with batch size. The trade-off is: smaller batches perform more parameter updates per epoch but with higher variance.

\begin{remark}[SGD Convergence with Fixed Step Size]
For a strongly convex $\mu$-strongly convex, $L$-smooth objective, mini-batch SGD with constant learning rate $\alpha$ converges to a neighbourhood of the minimum:
\begin{equation}
\mathbb{E}[f(\theta_k) - f^{\star}] \leq (1 - \alpha\mu)^k (f(\theta_0) - f^{\star}) + \frac{\alpha L \sigma^2_{\text{grad}}}{2\mu b},
\end{equation}
where $\sigma^2_{\text{grad}}$ is the gradient noise variance. The irreducible floor $\alpha L \sigma^2_{\text{grad}} / (2\mu b)$ decreases with batch size $b$ and step size $\alpha$, but a smaller $\alpha$ slows the first term's decay. This fundamental bias--variance trade-off motivates \emph{learning rate decay} schedules (e.g., $\alpha_t \propto 1/t$) that drive the floor to zero at the cost of slower convergence speed.
\end{remark}

\begin{algorithm}[H]
\caption{Mini-Batch Stochastic Gradient Descent}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Dataset $(X, y)$ with $m$ samples, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{epoch $e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\pi \gets \text{Permute}(\{1, \ldots, m\})$
    \For{mini-batch $\mathcal{B} = \{\pi_{(j-1)b+1}, \ldots, \pi_{jb}\}$, $j = 1, \ldots, \lfloor m/b \rfloor$}
        \State $\hat{g} \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Mini-batch gradient}
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{Local Approximations of $g(x) = x^4$}
\label{ssec:q3_approx}

Newton's method generalises the idea of finding roots of the first derivative to multidimensional optimisation, using a local quadratic model of the objective. Consider $g(x) = x^4$ at $x_0 = 0.25$:

\textbf{First-order (Taylor) approximation:}
\begin{equation}
g(x) \approx q_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625\,(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{align}
g(x) &\approx q_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 \nonumber \\
&= 0.0039 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2,
\end{align}
where $g'(x) = 4x^3$, $g''(x) = 12x^2$, so at $x_0 = 0.25$: $g'(x_0) = 4(0.25)^3 = 0.0625$ and $g''(x_0) = 12(0.25)^2 = 0.75$, giving the leading quadratic coefficient $\frac{1}{2} \times 0.75 = 0.375$.

The Newton update minimises $q_2(x)$ by setting $q_2'(x) = 0$:
\begin{equation}
x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)}.
\end{equation}
This is equivalent to stepping to the stationary point of the local quadratic model, and is the one-dimensional analogue of the multivariate Newton direction $p_k = H_k^{-1} \nabla f(x_k)$.

\textbf{Successive Newton iterates on $g(x) = x^4$:} Starting from $x_0 = 0.25$, the Newton update $x_{k+1} = x_k - g'(x_k)/g''(x_k) = x_k - \frac{4x_k^3}{12x_k^2} = x_k - x_k/3 = \frac{2}{3}x_k$ (for $x_k \neq 0$). This gives a geometric sequence: $x_k = (2/3)^k \times 0.25$, converging \emph{linearly} (ratio $2/3$) rather than quadratically. This is because $g(x) = x^4$ has $g''(x^*) = 0$ at the minimum, so the quadratic convergence theorem does not apply (the Hessian is degenerate at $x^* = 0$, a fourth-order zero). For functions with non-degenerate Hessian at the minimum (i.e., $\nabla^2 f(x^*) \succ 0$), Newton achieves the super-linear quadratic convergence rate.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$. The first-order tangent line captures only the local slope and diverges rapidly from the true function. The second-order parabola captures the curvature and provides a much better fit in a neighbourhood of $x_0$. The minimum of the parabola ($x = x_0 - g'/g'' = 0.25 - 0.0625/0.75 = 0.167$) is closer to the true minimum $x^{\star} = 0$ than the gradient step $x_0 - \alpha g'(x_0)$ for reasonable $\alpha$. \subsection{Newton's Method with Damping}
\label{ssec:q3_newton}

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0,1]$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised Hessian (ensures positive definiteness)}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Solve linear system for Newton direction}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{Damped Newton step}
\EndFor
\end{algorithmic}
\end{algorithm}

The Tikhonov regularisation $\lambda I$ serves two purposes: (1) it ensures positive definiteness of $H_k + \lambda I$ even when $H_k$ is near-singular or indefinite (e.g., far from the minimum on non-convex problems); (2) for large $\lambda$, it transitions the Newton direction toward the gradient direction $g_k / \lambda$ (pure gradient descent), providing a safety fallback. With $\alpha = 1$ and $\lambda = 0$, the pure Newton step achieves \emph{quadratic convergence} near the minimum:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq C\|x_k - x^{\star}\|^2,
\end{equation}
for some constant $C$ depending on the local curvature and the Lipschitz constant of $\nabla^2 f$. This super-linear convergence property means that near the solution, each iteration roughly doubles the number of accurate digits, explaining Newton's dramatic advantage in iteration count. The damped step ($\alpha < 1$) sacrifices quadratic convergence for global stability, maintaining monotone descent when the Hessian model is inaccurate far from the minimum.

\subsection{Quadratic Convergence: Theoretical Analysis}
\label{ssec:q3_quadratic}

\begin{proposition}[Newton's Quadratic Convergence]
Assume $f$ is twice continuously differentiable with $\nabla^2 f(x^{\star}) \succ 0$ and $\nabla^2 f$ is Lipschitz continuous with constant $M$ near $x^{\star}$. Then for full Newton steps ($\alpha = 1$) starting sufficiently close to $x^{\star}$:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\lambda_{\min}(\nabla^2 f(x^{\star}))}\,\|x_k - x^{\star}\|^2.
\end{equation}
\end{proposition}

\noindent\textit{Proof sketch:} By Taylor expansion of $\nabla f$ around $x^{\star}$:
\begin{align}
\nabla f(x_k) &= \nabla^2 f(x^{\star})(x_k - x^{\star}) + O(\|x_k - x^{\star}\|^2), \nonumber \\
x_{k+1} - x^{\star} &= x_k - [\nabla^2 f(x_k)]^{-1}\nabla f(x_k) - x^{\star} \nonumber \\
&= (I - [\nabla^2 f(x_k)]^{-1}\nabla^2 f(x^{\star}))(x_k - x^{\star}) + O(\|x_k - x^{\star}\|^2).
\end{align}
Near $x^{\star}$, $\nabla^2 f(x_k) \approx \nabla^2 f(x^{\star})$, so the leading term vanishes and the remainder is $O(\|x_k - x^{\star}\|^2)$. \hfill $\square$

This quadratic convergence is observed in our experiments: for Benchmark A (exact quadratic), Newton reaches machine precision in 1 step; for Benchmark B (near-quadratic), it converges in $\sim$5 steps; for Benchmark C (ill-conditioned, rapidly varying Hessian), the damped step $\alpha = 0.22$ slows convergence to approximately linear but still achieves $f = 0.686$ in 20 iterations vs GD's $f = 3.732$ in 80.

\subsection{Computational Cost Analysis}
\label{ssec:q3_cost}

\begin{remark}[Per-Iteration Cost]
For a $d$-dimensional problem:
\begin{itemize}[nosep]
\item GD: $O(d)$ operations per iteration (gradient evaluation).
\item Newton: $O(d^3)$ operations per iteration (Hessian construction $O(d^2)$ + linear solve $O(d^3)$).
\end{itemize}
For $d = 2$ (as here), the per-iteration cost difference is negligible. However, for large-scale problems ($d \gg 100$), Newton becomes computationally infeasible, motivating quasi-Newton methods (e.g., L-BFGS) that approximate $H^{-1}$ using gradient history without explicit Hessian computation. L-BFGS achieves superlinear convergence with $O(md)$ cost per iteration (where $m$ is the history length, typically $3$--$20$), bridging the gap between GD and Newton.
\end{remark}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}
\label{ssec:q4_partA}

\subsubsection{Finite Difference Gradient Approximation}
\label{sssec:q4_fd}

The forward finite-difference approximation replaces each partial derivative:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\label{eq:fd_approx}
\end{equation}
where $e_i$ is the $i$-th standard basis vector. The approximation error has two competing contributions:
\begin{equation}
\left|\frac{\partial f}{\partial x_i}(x) - \frac{f(x + \delta e_i) - f(x)}{\delta}\right| \leq \underbrace{\frac{\delta}{2} \left|\frac{\partial^2 f}{\partial x_i^2}(x)\right|}_{\text{truncation error, } O(\delta)} + \underbrace{\frac{2\varepsilon_{\text{mach}}}{\delta}}_{\text{rounding error, } O(\varepsilon_{\text{mach}}/\delta)}.
\end{equation}
The optimal step size minimising the total error is $\delta^{\star} \approx \sqrt{2\varepsilon_{\text{mach}}/|f''|} \approx 10^{-8}$ for double precision. For $\delta = 0.05$ (good) the error is dominated by $O(\delta) \approx 0.025|f''|$ (truncation), while for $\delta = 0.8$ (poor), the approximation spans a region where $f$ changes nonlinearly, severely biasing the gradient estimate.

\begin{remark}[Central Differences vs Forward Differences]
The \emph{central difference} approximation $\hat{\partial}_i f = \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}$ reduces the truncation error from $O(\delta)$ to $O(\delta^2)$ by cancelling the leading odd Taylor term. The optimal step size improves to $\delta^{\star}_c \approx (3\varepsilon_{\text{mach}}/|f'''|)^{1/3} \approx 10^{-5.3}$ for double precision, giving a total error $O(\varepsilon_{\text{mach}}^{2/3})$ rather than $O(\varepsilon_{\text{mach}}^{1/2})$. The cost is $2d$ function evaluations per gradient (vs $d+1$ for forward differences), which is negligible for the 2D problems studied here but matters at scale.
\end{remark}

\subsubsection{Nesterov Random Search}
\label{sssec:q4_nrs}

\begin{algorithm}[H]
\caption{Nesterov Random (Zeroth-Order) Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, smoothing parameter $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The update direction $\hat{g}_k$ is an estimate of the \emph{gradient of a smoothed surrogate} $f_\delta(x) = \mathbb{E}_u[f(x + \delta u)]$, where the expectation is over uniformly distributed unit vectors $u$. In expectation: $\mathbb{E}[\hat{g}_k] = \frac{d}{\delta} \mathbb{E}[(f(x+\delta u) - f(x)) u] = \nabla f_\delta(x)$, which approximates $\nabla f(x)$ for small $\delta$. The key advantage is that only \emph{two function evaluations} are required per iteration regardless of dimension $d$, making it dimension-independent in per-step cost.

\begin{remark}[Zeroth-Order Complexity]
For an $L$-smooth convex function in $\mathbb{R}^d$, Nesterov's random search with the optimal smoothing $\delta = O(\varepsilon/d)$ achieves $\varepsilon$-accuracy in $O(d^2/\varepsilon)$ function evaluations. By contrast, forward finite differences require $O(d \cdot d/\varepsilon) = O(d^2/\varepsilon)$ evaluations (same asymptotic rate but using gradient approximations). The dimension dependence in both cases arises from the curse of dimensionality in zeroth-order optimisation --- an unavoidable $d$-fold cost relative to first-order methods, which require $O(1/\varepsilon)$ gradient evaluations. For $d = 2$ as here, this distinction is negligible, but for $d = 10^6$ (deep learning), gradient-free methods are typically impractical.
\end{remark}

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}
\label{ssec:q4_partB}

\subsubsection{Nelder--Mead Simplex Method}
\label{sssec:q4_nm}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$. At each iteration, it transforms the simplex through four geometric operations to explore the loss landscape without gradient information:

\begin{enumerate}[nosep]
\item \textbf{Reflection:} Move the worst vertex $x_w$ through the centroid $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_w)$, $\alpha_r = 1$.
\item \textbf{Expansion:} If reflection improves significantly, try a longer step: $x_e = \bar{x} + \gamma(\bar{x} - x_w)$, $\gamma = 2$.
\item \textbf{Contraction:} If reflection does not improve, shrink toward $\bar{x}$: $x_c = \bar{x} + \rho(\cdot - \bar{x})$, $\rho = 0.5$.
\item \textbf{Shrink:} If contraction fails, contract all vertices toward the best vertex $x_b$: $x_i \gets x_b + \sigma(x_i - x_b)$, $\sigma = 0.5$.
\end{enumerate}

Settings: initial simplex step $= 0.35$, 160 iterations. Nelder--Mead requires no gradient information, making it applicable to non-smooth objectives and black-box functions.

\begin{remark}[Convergence of Nelder--Mead]
Despite its widespread use in scientific computing, Nelder--Mead has no general convergence proof. McKinnon (1998) constructed a 2D example where Nelder--Mead converges to a non-stationary point. Powell (1998) showed it can fail to converge in $d \geq 2$ dimensions even for smooth strongly convex functions. Convergence is only guaranteed in 1D (reduces to a bracket search). In practice, the algorithm works well for low-dimensional ($d \leq 10$) problems with smooth objectives, and restarts or multiple initialisations are used to mitigate stagnation.
\end{remark}

\subsection{Problem Formulation}
\label{ssec:q5_problem}

We minimise Benchmark B subject to the constraint $x_1 \geq 0.5$:
\begin{equation}
\min_{x \in \mathbb{R}^2}\; f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad \text{s.t.}\; x_1 \geq 0.5.
\end{equation}
Starting from the infeasible point $x_0 = (0.2, 4.0)$ (where $x_{1,0} = 0.2 < 0.5$). The unconstrained minimiser $(0.582, 2.0)$ satisfies $x_1 = 0.582 > 0.5$, so it is feasible and remains the constrained minimiser.

\subsection{Algorithm Formulations}
\label{ssec:q5_algorithms}

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$ (possibly infeasible), step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project onto feasible set}
\EndFor
\end{algorithmic}
\end{algorithm}

For $\mathcal{X} = \{x : x_1 \geq 0.5\}$, the projection is trivial: $\Pi_{\mathcal{X}}(z) = [\max(0.5, z_1),\; z_2]^T$. This ensures $x_k \in \mathcal{X}$ for all $k \geq 1$, even from an infeasible starting point.

\begin{algorithm}[H]
\caption{Penalty Method (Quadratic Penalty)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute penalised objective: $F_{\lambda}(x_k) = f(x_k) + \lambda \max(0,\; 0.5 - x_1)$
    \State Compute penalised gradient: $\nabla F_{\lambda}(x_k) = \nabla f(x_k) + \lambda (-1) \cdot \mathbf{1}[x_1 < 0.5]\, e_1$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty function $\lambda \max(0, 0.5 - x_1)$ adds a cost proportional to constraint violation. As $\lambda \to \infty$, the penalised minimiser converges to the constrained minimiser, but large $\lambda$ creates an ill-conditioned penalised objective requiring smaller step sizes for stability. Formally, the penalty method solves a sequence of unconstrained problems $\min_x F_\lambda(x)$ for increasing $\lambda$; under constraint qualification (Slater's condition holds here since $(1, 2) \in \text{int}(\{x_1 \geq 0.5\})$), the minimiser $x^*(\lambda) \to x^*_{\text{constrained}}$ as $\lambda \to \infty$, with the Lagrange multiplier $\mu^* = \lim_{\lambda \to \infty} \lambda \max(0, 0.5 - x_1^*(\lambda))$.

\textbf{Hyperparameters:} Projected GD ($\alpha = 0.08$); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$, $\alpha = 0.03$ for $\lambda = 4.5$); Unconstrained GD ($\alpha = 0.07$). All 100 iterations.

\textbf{Projected Gradient Descent} maintains feasibility by construction at every iteration. The projection $\Pi_{\mathcal{X}}$ is the key operation: for simple convex sets (box, simplex, $\ell_2$-ball), it is analytically tractable with $O(d)$ cost. For the half-space $\{x : x_1 \geq 0.5\}$, the projection is simply $x_1 \leftarrow \max(0.5, x_1)$, costing $O(1)$ per iteration.

\textbf{Penalty Methods} offer a trade-off: feasibility is enforced only approximately, but no projection oracle is required. Three observations arise from the experiments:
\begin{enumerate}[nosep]
\item Large $\lambda$ ($= 4.5$) enforces feasibility nearly as well as PGD but requires a smaller step size ($\alpha = 0.03$ vs $0.05$), compensating for the ill-conditioning introduced by the large penalty (Figure~\ref{fig:q5_penalty_analysis} shows the sharp gradient discontinuity at the constraint).
\item The effective condition number of the penalised objective grows as $\lambda \to \infty$, since the penalty Hessian component adds $\lambda$ to the curvature in the constraint-normal direction ($x_1$).
\item Small $\lambda$ ($= 0.15$) results in a weak feasibility force: the penalty gradient ($-\lambda e_1 = -0.15 e_1$ when $x_1 < 0.5$) is much smaller than the objective gradient ($\nabla_1 f \approx -2.3$ at $x = (0.2, 4.0)$), providing insufficient force to drive the iterate toward $x_1 = 0.5$.
\end{enumerate}

\begin{remark}[Convergence of Projected GD]
For a convex objective $f$ with Lipschitz gradient (constant $L$) minimised over a convex set $\mathcal{X}$, projected gradient descent with step size $\alpha \leq 1/L$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{\|x_0 - x^{\star}\|^2}{2\alpha k}.
\end{equation}
For strongly convex $f$ with modulus $\mu$, the rate improves to linear: $\|x_k - x^{\star}\|^2 \leq (1 - \alpha\mu)^k\|x_0 - x^{\star}\|^2$. In our experiment, the constraint is a half-space, which is convex and admits trivial projection, so the projected GD analysis applies directly.
\end{remark}

\begin{remark}[Augmented Lagrangian]
The penalty method can be improved with the \emph{augmented Lagrangian} (method of multipliers), which adds a dual variable $\mu$ updated at each outer iteration: $F_{\lambda,\mu}(x) = f(x) + \mu^T c(x) + \frac{\lambda}{2}\|c(x)\|^2$. This achieves exact constraint satisfaction without requiring $\lambda \to \infty$, avoiding the condition number growth that destabilises the pure penalty method.
\end{remark}

\begin{remark}[KKT Conditions for the Constrained Problem]
The constrained problem $\min_{x_1 \geq 0.5} f(x)$ is a convex program (since $f$ is convex and the constraint is linear), so strong duality holds by Slater's condition. The KKT conditions at the constrained optimum $x^{\star} = (0.582, 2.0)$ are:
\begin{align*}
\nabla f(x^{\star}) + \mu^{\star}(-e_1) &= 0 \quad \text{(stationarity)}, \\
x_1^{\star} - 0.5 &\geq 0 \quad \text{(primal feasibility)}, \\
\mu^{\star} &\geq 0 \quad \text{(dual feasibility)}, \\
\mu^{\star}(x_1^{\star} - 0.5) &= 0 \quad \text{(complementary slackness)}.
\end{align*}
Since $x_1^{\star} = 0.582 > 0.5$, complementary slackness forces $\mu^{\star} = 0$, confirming the constraint is \emph{inactive} at the optimum. Thus $\nabla f(x^{\star}) = 0$, and the constrained and unconstrained optima coincide. This is why all methods (PGD, penalty, unconstrained GD) converge to the same final objective value $f^{\star} = 0.7244$ in Table~\ref{tab:q5_final}.
\end{remark}

\subsection{Linear Programme over the Feasible Set}
\label{ssec:q6_lp}

\textbf{Feasible set:} $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$ (a bounded box with 4 vertices).

\textbf{Linear Programme:} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$. Since the objective is linear and the feasible set is a convex polytope, the minimum is attained at a vertex. For a box constraint with positive components of $a$:
\begin{equation}
x^{\star} = \arg\min_{x \in \mathcal{X}} (x_1 + 2x_2) = \begin{pmatrix}\text{lower bound of } x_1 \\ \text{lower bound of } x_2\end{pmatrix} = (0.5,\; -5), \quad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates the \textbf{fundamental theorem of linear programming}: the optimum of a linear objective over a bounded polytope is always attained at a vertex of the polytope.

\medskip
\textbf{Connection to the Frank--Wolfe LMO:} Observe that this LP is precisely the \emph{linear minimisation oracle} (LMO) that Frank--Wolfe calls at each iteration with $a = \nabla f(x_k)$. Because the feasible set here is a box, the LMO decouples coordinate-wise and costs only $O(d)$ operations. In more complex constraint sets, the LMO can still be efficient:
\begin{itemize}[nosep]
\item \emph{Spectahedron} $\{X \succeq 0 : \operatorname{tr}(X) = 1\}$: LMO = leading eigenvector computation, $O(d^2)$ instead of $O(d^3)$ for projection.
\item \emph{Nuclear norm ball} $\{M : \|M\|_{*} \leq \tau\}$: LMO = rank-1 update via leading singular vectors, enabling scalable matrix completion.
\item \emph{Flow polytope} (network constraints): LMO = shortest path computation, solvable in near-linear time via Dijkstra's algorithm.
\end{itemize}
This projection-free structure is Frank--Wolfe's core advantage in high-dimensional structured optimisation, where projections would require solving the full constraint system at every iteration.

\subsection{Frank--Wolfe Algorithm}
\label{ssec:q6_fw}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient) Algorithm}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\; \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (stays in $\mathcal{X}$)}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Key properties of Frank--Wolfe:}
\begin{itemize}[nosep]
\item \emph{Projection-free:} Iterates are convex combinations of feasible points, so $x_{k+1} \in \mathcal{X}$ by convexity of $\mathcal{X}$ --- no projection oracle needed.
\item \emph{Efficient LMO:} For box constraints, the linear subproblem decouples per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases},
\end{equation}
requiring only $O(d)$ operations --- much cheaper than projections for complex constraint sets.
\item \emph{Convergence:} For $L$-smooth convex objectives, FW achieves an $O(1/k)$ convergence rate:
\begin{equation}
h_k = \langle \nabla f(x_k), x_k - z_k \rangle \leq \frac{2LC^2}{k+2},
\end{equation}
where $C = \max_{x,y \in \mathcal{X}}\|x - y\|$ is the diameter of $\mathcal{X}$, and $h_k \geq f(x_k) - f^{\star}$ is the Frank--Wolfe gap (an efficiently computable upper bound on suboptimality, verifiable without knowing $f^{\star}$).
\item \emph{Step size schedule:} Using the fixed mixing parameter $\beta$ (as implemented here) is equivalent to step size $\gamma_k = 1 - \beta$. The standard decaying schedule $\gamma_k = 2/(k+2)$ achieves the $O(1/k)$ rate theoretically, but the fixed $\beta$ can be better tuned for specific problems.
\end{itemize}

\subsection{Interior Optimum: $f(x) = (x_1-1)^2 + (x_2-5)^2$}
\label{ssec:q6_interior}

The unconstrained minimum $(1, 5) \in \mathcal{X}$ (since $0.5 \leq 1 \leq 5$ and $-5 \leq 5 \leq 10$). This function is $L$-smooth with $L = 2$ (the largest eigenvalue of $\nabla^2 f = 2I$) and the diameter of $\mathcal{X}$ is $C = \|(5, 10) - (0.5, -5)\| = \sqrt{4.5^2 + 15^2} \approx 15.66$. The FW convergence bound gives: $h_k \leq 2LC^2 / (k+2) \approx 980 / (k+2)$.

Two mixing parameters are tested: $\beta = 0.90$ (step size $\gamma = 0.10$ toward $z_k$) and $\beta = 0.985$ (step size $\gamma = 0.015$).

\subsection{Interior vs Boundary Behaviour: Theoretical Analysis}
\label{ssec:q6_analysis}

\textbf{Interior optimum:} The FW gap decreases as $h_k = O(1/k)$, as guaranteed for $L$-smooth convex functions. The ``zig-zag'' oscillation of $z_k$ between vertices is characteristic of Frank--Wolfe on interior optima: since $x_k$ is close to $(1, 5)$, the gradient $\nabla f(x_k) \approx 2(x_k - [1,5]^T)$ is small and alternates direction, causing $z_k$ to jump between adjacent vertices of $\mathcal{X}$.

\textbf{Boundary optimum:} Frank--Wolfe is particularly well-suited to boundary optima because $z_k$ frequently selects the vertex closest to the constrained optimum. For $f(x) = x_1^2 + x_2^2$, the gradient is $2x_k$, so $z_{k,1} = 0.5$ (always, since $x_{k,1} > 0$) and $z_{k,2} = -5$ if $x_{k,2} > 0$ or $z_{k,2} = 10$ if $x_{k,2} < 0$. This drives $x_2 \to 0$ and $x_1 \to 0.5$, naturally reaching the constrained minimum without projection.

\textbf{Slow convergence near boundary:} The FW algorithm achieves only $O(1/k)$ convergence even for strongly convex objectives, whereas projected gradient descent achieves linear $O(\rho^k)$ convergence. For the boundary case ($f(x) = x_1^2 + x_2^2$), our result $f = 0.364$ after 140 iterations (vs $f^{\star} = 0.25$) reflects this: using the standard decaying schedule $\gamma_k = 2/(k+2)$ instead of fixed $\beta = 0.93$ would improve the convergence rate constant. This $O(1/k)$ penalty is the fundamental trade-off of projection-free methods: they pay a slower rate for avoiding the potentially expensive projection oracle, which is acceptable when projections are computationally costly (e.g., nuclear norm ball constraints require SVD).

\begin{proposition}[FW Gap as Optimality Certificate]
At any iteration $k$, the Frank--Wolfe gap $h_k = \langle \nabla f(x_k), x_k - z_k \rangle \geq 0$ satisfies $h_k \geq f(x_k) - f^{\star}$. Thus $h_k < \epsilon$ certifies $\epsilon$-optimality without knowing $f^{\star}$. This is a key practical advantage over methods that require knowledge of the optimal value.
\end{proposition}

\subsection{Practical Recommendations}
\label{ssec:summary_rec}

\begin{itemize}[nosep]
\item \textbf{Well-conditioned problems ($\kappa \leq 10$):} Any first-order method suffices. The well-conditioned Benchmarks A ($\kappa = 1.12$) and B ($\kappa = 6.90$) are solved to high accuracy by GD, Adagrad, and Heavy Ball equally. Adam or Adagrad offer good defaults with minimal hyperparameter tuning.
\item \textbf{Ill-conditioned problems ($\kappa \gg 1$):} Use Nesterov (optimal $O(1/k^2)$ first-order rate) or Newton (quadratic convergence, but requires Hessian). On Rosenbrock ($\kappa = 2508$), Nesterov achieves $7\times$ better $f$ than GD in the same iteration budget.
\item \textbf{Noisy objectives / stochastic settings:} Mini-batch SGD with Adam is the standard choice in machine learning. Batch size $b$ controls the bias--variance trade-off: larger $b$ reduces gradient variance (smoother convergence) but fewer updates per epoch. Under high observation noise ($\sigma = 6$), the irreducible loss floor dominates and batch size differences become negligible.
\item \textbf{Derivative-free settings:} Nelder--Mead for low-dimensional ($d \leq 10$) smooth problems; Nesterov random search for high-dimensional zeroth-order optimisation (only 2 FE/iteration); grid search exclusively for low-dimensional global optimisation with fine accuracy requirements.
\item \textbf{Constrained problems with simple constraints:} Projected GD for box/simplex constraints (trivial $O(d)$ projection); penalty methods when constraint structure is complex but gradients are available. Key difference: PGD maintains hard feasibility at every iteration, while penalty methods are only asymptotically feasible.
\item \textbf{Constrained problems with efficient LMO:} Frank--Wolfe when the linear minimisation subproblem is cheap (box constraints: $O(d)$; spectahedron/nuclear norm ball: $O(d^2)$ vs $O(d^3)$ for projection). FW's $O(1/k)$ rate is acceptable for problems where projections would dominate the computational budget.
\end{itemize}

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has systematically evaluated twelve optimisation algorithms across problems of increasing difficulty, covering unconstrained, stochastic, derivative-free, and constrained settings. The three benchmarks span a $2240\times$ range in condition number ($\kappa = 1.12$ to $\kappa = 2508$), providing a principled experimental framework for evaluating algorithmic properties.

\textbf{Key findings by question:}
\begin{enumerate}[nosep]
\item \textbf{(Q1) Adaptive step-sizes:} The Polyak step size is theoretically optimal under exact $f^{\star}$ knowledge (achieving the best Rosenbrock convergence $f = 0.024$) but diverges catastrophically ($f = 4.77$) when $f^{\star}$ is misspecified. Heavy Ball is the best robust method ($f_C = 1.40$ vs GD's $f_C = 3.51$ on Rosenbrock), confirming the $O(\sqrt{\kappa})$ improvement for strongly convex quadratics. Adagrad achieves minimax-optimal $O(\sqrt{T})$ regret without any problem knowledge.

\item \textbf{(Q2) Momentum and stochastic methods:} Nesterov achieves the information-theoretically optimal $O(1/k^2)$ first-order rate, confirmed empirically: $7.1\times$ better than GD on Rosenbrock ($f = 0.479$ vs $3.39$) after 150 iterations. The verified $O(1/k^2)$ slope in Figure~\ref{fig:q2_conv_rates} confirms the theoretical guarantee holds in practice. Mini-batch SGD noise floor $J^{\star} \propto \sigma^2$ grows $36\times$ when $\sigma$ increases $6\times$, as predicted by the irreducible error formula.

\item \textbf{(Q3) Newton's method:} Quadratic convergence ($\|e_{k+1}\| \leq C\|e_k\|^2$) is confirmed experimentally: one step to machine precision on Benchmark A (exact quadratic), $\sim$5 steps on Benchmark B (near-quadratic), and $5.4\times$ better $f$ in 25\% of GD iterations on Rosenbrock. The Hessian eigenvalue analysis (Figure~\ref{fig:q3_hessian_evol}) shows the local condition number remains $\approx 2508$ throughout the trajectory, explaining why Newton's curvature information is essential for efficient traversal.

\item \textbf{(Q4) Derivative-free methods:} Finite differences match exact GD for $\delta = 0.05$ (truncation error $\approx 0.025|f''|$) but fail catastrophically for $\delta = 0.8$ due to high truncation error. Central differences reduce the truncation error from $O(\delta)$ to $O(\delta^2)$ at $2\times$ the cost. Grid search is the only globally-guaranteed method at $O(N^d)$ total cost; Nelder--Mead works well empirically for low-dimensional problems despite having no general convergence proof.

\item \textbf{(Q5) Constrained optimisation:} Projected GD enforces hard feasibility from iteration 1 via the projection $\Pi_{\mathcal{X}}$. Penalty methods are only asymptotically feasible: the Lagrange multiplier $\mu^{\star} = \lim_\lambda \lambda \cdot v(x^*(\lambda))$ emerges naturally. In this experiment, the KKT conditions show $\mu^{\star} = 0$ (inactive constraint), so all methods converge to the same objective value $f^{\star} = 0.7244$.

\item \textbf{(Q6) Frank--Wolfe:} The FW gap $h_k = \langle \nabla f(x_k), x_k - z_k \rangle$ provides a computable certificate of $\epsilon$-optimality without knowing $f^{\star}$, decaying as $O(1/k)$ (confirmed in Figure~\ref{fig:q6_fw_gap}). The LMO structure enables efficient extension to spectahedra ($O(d^2)$), nuclear norm balls, and flow polytopes — all cases where projection would be computationally prohibitive.
\end{enumerate}

\textbf{Overarching lesson:} The condition number is the single most important problem characteristic for algorithm selection: problems with $\kappa < 10$ are solved equally by any first-order method, while $\kappa \approx 2508$ demands either second-order information (Newton: quadratic convergence) or momentum (Nesterov: $O(\sqrt{\kappa})$ speedup). No single algorithm dominates across all settings (no-free-lunch theorem). The theoretical convergence rates — $O(1/k^2)$ for Nesterov, $O(\sqrt{T})$ regret for Adagrad, $O(1/k)$ FW gap — are not just asymptotic abstractions but accurately predict practical performance differences observed in all six problem categories.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
