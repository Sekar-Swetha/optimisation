This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark A), a toy neural network loss with sinusoidal perturbation (Benchmark B), and the Rosenbrock function (Benchmark C). Six categories of methods are investigated across six questions: (Q1) adaptive step-size methods — Polyak, Adagrad, RMSprop, and the Heavy Ball method; (Q2) momentum-based methods — Nesterov acceleration and Adam — together with mini-batch stochastic gradient descent; (Q3) Newton's method with local quadratic approximation; (Q4) derivative-free optimisation via finite differences, Nesterov random search, Nelder--Mead simplex, and grid search; (Q5) constrained optimisation via projected gradient descent and penalty methods; and (Q6) the Frank--Wolfe (conditional gradient) algorithm for linear-constrained problems. Each method is evaluated in terms of convergence speed, trajectory behaviour, stability, and computational trade-offs, with explicit connections to the theoretical convergence guarantees established in the course lectures.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. We establish notation used throughout: $\nabla f(x)$ denotes the gradient, $\nabla^2 f(x)$ the Hessian, $\|\cdot\|$ the Euclidean norm, and $f^\star$ the optimal value.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
\label{ssec:bench_A}

A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
  \theta^\star = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \sim \mathcal{N}(0, I_2), \quad
  y^{(i)} = {\theta^\star}^T x^{(i)} + \varepsilon^{(i)}, \quad
  \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error (MSE):
\begin{equation}
  J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
            = \frac{1}{2m} \|X\theta - y\|^2.
  \label{eq:loss_A}
\end{equation}
This is a convex quadratic with constant Hessian $H = \frac{1}{m}X^TX \in \mathbb{R}^{2\times 2}$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise $\varepsilon$, the irreducible minimum is
\begin{equation}
  J^\star = J(\hat\theta_{\text{OLS}}) \approx 0.4826,
\end{equation}
where $\hat\theta_{\text{OLS}} = (X^TX)^{-1}X^Ty \approx [3.028,\ 4.017]^T$ is the ordinary least squares solution, close to the true parameter $\theta^\star$ by the law of large numbers. The Lipschitz constant of the gradient is $L = \lambda_{\max}(H) \approx 1.0$ and the strong convexity constant is $\mu = \lambda_{\min}(H) \approx 0.97$; hence the condition number is $\kappa(H) = L/\mu \approx 1.03$, making this benchmark \emph{well-conditioned}. Starting point: $\theta_0 = [0,\ 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\label{ssec:bench_B}

\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
  \label{eq:loss_B}
\end{equation}
The gradient is
\begin{equation}
  \nabla f = \begin{bmatrix} 2(x_1 - 1) + \cos(x_1) \\ 10(x_2 - 2) \end{bmatrix}
\end{equation}
and the Hessian is diagonal:
\begin{equation}
  \nabla^2 f(x) = \mathrm{diag}\!\bigl(2 - \sin(x_1),\ 10\bigr).
\end{equation}
The $x_2$ component has fixed curvature 10, while the $x_2$-independent $x_1$ curvature $2 - \sin(x_1)$ varies in $[1, 3]$, yielding an eigenvalue ratio (condition number) between $10/3 \approx 3.3$ and $10/1 = 10$ depending on the iterate. This creates \emph{moderate anisotropy}. Setting the gradient to zero gives $x_2^\star = 2$ exactly and $x_1^\star$ satisfying $2(x_1^\star - 1) + \cos(x_1^\star) = 0$, which has numerical solution $x_1^\star \approx 0.5826$. The minimum value is $f^\star \approx 0.7244$. Starting point: $x_0 = [-1,\ 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\label{ssec:bench_C}

\begin{equation}
  f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
  \label{eq:loss_C}
\end{equation}
Gradients and Hessian:
\begin{align}
  \nabla f &= \begin{bmatrix} -2(1 - x_1) - 400\,x_1(x_2 - x_1^2) \\ 200(x_2 - x_1^2) \end{bmatrix}, \\
  \nabla^2 f(x) &= \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{align}
The global minimum is at $(1,\,1)$ with $f^\star = 0$. At the optimum the Hessian has eigenvalues approximately $0.5$ and $\approx 801$, giving condition number $\kappa \approx 2000$. This extreme ill-conditioning, combined with the narrow curved valley along $x_2 \approx x_1^2$, makes Rosenbrock one of the canonical hard test functions for optimisation algorithms. Starting point: $x_0 = [-1,\ 1]^T$.

For Questions 5--6: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$, a closed convex box (polytope).

\section{Question 1: Adaptive Step-Size Methods}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}
\label{ssec:q1_theory}

Standard gradient descent $x_{k+1} = x_k - \alpha \nabla f(x_k)$ with constant step size $\alpha$ converges at rate $O(\kappa^k)$ for strongly convex quadratics, where $\kappa$ is the condition number. The optimal constant step is $\alpha^\star = 2/(L+\mu)$, giving convergence rate $(\kappa-1)/(\kappa+1)$. For ill-conditioned problems (large $\kappa$), convergence is slow, motivating \emph{adaptive} step-size strategies.

\begin{definition}[Lipschitz gradient]
$f$ has $L$-Lipschitz gradient if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x-y\|$ for all $x, y$.
\end{definition}

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Gradient Descent (Baseline)}
\label{alg:gd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\,\nabla f(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^\star$, regularisation $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^\star}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive step proportional to suboptimality}
    \State $x_{k+1} \gets x_k - \alpha_k\,g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak step convergence]
\label{thm:polyak}
For a convex function $f$ with $f^\star$ known exactly, the Polyak step satisfies $f(x_k) \to f^\star$. For strongly convex $f$ with constant $\mu > 0$, the iterate-to-optimum distance satisfies $\|x_{k+1} - x^\star\|^2 \leq \left(1 - \mu\,\alpha_k\right)\|x_k - x^\star\|^2$, where the step $\alpha_k = (f(x_k)-f^\star)/\|g_k\|^2$ minimises the worst-case distance decrease.
\end{theorem}

The key insight is that the Polyak step $\alpha_k$ is self-calibrating: when $x_k$ is far from the optimum (large $f(x_k) - f^\star$) the step is large, and as $x_k$ approaches the optimum the step shrinks automatically. However, this requires \emph{exact knowledge of $f^\star$}.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0 > 0$, regularisation $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Monotone accumulated squared-gradient sum}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The effective per-coordinate step for Adagrad is $\alpha_k^{(i)} = \alpha_0 / (\sqrt{\sum_{j=0}^{k} g_j^{(i)2}} + \varepsilon)$, which decays monotonically as $O(1/\sqrt{k})$. This guarantees convergence but can lead to premature slowing on stationary problems. The monotone decay is intentional for convex analysis but limits applicability on deep networks.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0 > 0$, decay rate $\beta \in (0,1)$, regularisation $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\,v_k + (1-\beta)\,g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's diminishing step problem by replacing the cumulative sum with an exponential moving average (EMA) with forgetting factor $\beta$. The effective window length is $1/(1-\beta)$ iterations; with $\beta = 0.9$, recent gradients dominate over a horizon of $\sim 10$ steps, allowing the step size to recover when gradient magnitudes decrease.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, momentum coefficient $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\,z_k + \alpha\,\nabla f(x_k)$ \Comment{Accumulate momentum}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method adds inertia by accumulating a momentum term $z_k$. For $L$-smooth, $\mu$-strongly convex quadratics, the optimal parameters $\alpha^\star = 4/(\sqrt{L}+\sqrt{\mu})^2$ and $\beta^\star = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$ yield convergence rate $\left((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)\right)^{2k}$, which is asymptotically faster than gradient descent by a factor of $\sqrt{\kappa}$ in the exponent. Note, however, that Heavy Ball does \emph{not} carry this guarantee to non-quadratic objectives.

\textbf{Key finding:} The Polyak step size's sensitivity to $f^\star$ is a fundamental limitation. The method is therefore only practically applicable when $f^\star$ is known, such as in Benchmark C where $f^\star = 0$ is exact.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
  \item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$ in 120 iterations). Among methods that do not require $f^\star$, Heavy Ball consistently matches or exceeds GD on all benchmarks, while Adagrad matches GD on well-conditioned problems.
  \item \textbf{Stability:} GD with an appropriate constant step is most stable but slowest. Polyak is only stable when $f^\star$ is accurately specified. Adagrad is uniformly stable due to its monotonically decreasing step, but this guarantees only sub-linear convergence. Heavy Ball can oscillate (evident in slight overshoot in Figure~\ref{fig:q1_contour_B}) but remains stable with appropriate $(\alpha, \beta)$.
  \item \textbf{Coordinate adaptivity:} Adagrad and RMSprop scale each coordinate independently, providing implicit preconditioning that helps on anisotropic problems. Heavy Ball does not adapt per-coordinate but achieves acceleration through momentum.
  \item \textbf{Practical guidance:} RMSprop is preferred over Adagrad in practice when stationarity is not known, as it avoids premature decay. The Polyak step is theoretically elegant but practically limited to known-$f^\star$ settings.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods}
\label{sec:q2}
% ============================================================

\subsection{Theoretical Background}

Nesterov (1983) proved that for smooth convex functions, no first-order method can achieve convergence faster than $O(1/k^2)$. The Nesterov accelerated method achieves exactly this rate, improving upon gradient descent's $O(1/k)$ rate. Adam combines this acceleration idea with per-coordinate adaptation from RMSprop.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum (Accelerated Gradient Method)}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, max momentum cap $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\; \beta_{\max}\right)$
    \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\,z_k$
    \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\,z_k - \alpha\,\nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov acceleration, simplified]
\label{thm:nesterov}
For an $L$-smooth convex function $f$, the Nesterov accelerated method with $\alpha = 1/L$ and the momentum schedule $\beta_k = (k-1)/(k+2)$ satisfies
\begin{equation}
  f(x_k) - f^\star \leq \frac{2L\,\|x_0 - x^\star\|^2}{(k+1)^2} = O\!\left(\frac{1}{k^2}\right).
\end{equation}
This is optimal in the sense that no first-order algorithm can achieve a better worst-case rate for general smooth convex functions.
\end{theorem}

The fundamental difference from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$ itself, which provides a corrective effect that prevents over-shooting. The schedule $\beta_k = (k-1)/(k+2)$ increases the momentum gradually from 0 towards $\beta_{\max}$, matching the Nesterov theoretical recipe.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha > 0$, moments $\beta_1, \beta_2 \in (0,1)$, regularisation $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\,m_{t-1} + (1-\beta_1)\,g_t$
    \Comment{First moment (mean) EMA}
    \State $v_t \gets \beta_2\,v_{t-1} + (1-\beta_2)\,g_t \odot g_t$
    \Comment{Second moment (variance) EMA}
    \State $\hat{m}_t \gets m_t/(1-\beta_1^t)$
    \Comment{Bias correction (important for early $t$)}
    \State $\hat{v}_t \gets v_t/(1-\beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\,\hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam unifies three ideas: (i) first-moment EMA for gradient direction smoothing (like momentum), (ii) second-moment EMA for per-coordinate scaling (like RMSprop), and (iii) bias correction to compensate for the zero initialisation of $m_0$ and $v_0$. Without bias correction, early iterates would be biased towards zero, effectively using a much smaller step than intended.

\begin{remark}[Adam and convergence]
Despite widespread practical success, Adam lacks guaranteed convergence for general non-convex objectives. Reddi et al.\ (2018) showed that Adam can diverge on convex problems and proposed AMSGrad (which maintains the maximum of past second moments) as a provably convergent variant. In our deterministic convex benchmark setting, Adam performs well.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, $E = 50$ epochs, with data shuffled at every epoch (i.e., without replacement sampling within each epoch).

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\sigma$
    \For{each mini-batch $\mathcal{B} = \{\sigma(ib+1), \ldots, \sigma((i+1)b)\}$}
        \State $g \gets \dfrac{1}{b}\,X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \Comment{Unbiased gradient estimate: $\mathbb{E}[g] = \nabla J(\theta)$}
        \State $\theta \gets \theta - \alpha\,g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

Each epoch with batch size $b$ contains $m/b$ parameter updates. With $b = 5$: $1000/5 = 200$ updates per epoch; with $b = 40$: $1000/40 = 25$ updates per epoch.

\section{Question 3: Newton's Method and Local Approximation}
\label{sec:q3}
% ============================================================

\subsection{Theoretical Background: Local Quadratic Approximation}
\label{ssec:q3_theory}

A central idea in optimisation is to replace the objective $f$ by a simpler local model that can be minimised exactly, then use the minimiser of the model as the next iterate. The $k$-th order Taylor expansion around $x_k$ provides the natural local model:
\begin{align}
  f(x) &\approx f(x_k) + \nabla f(x_k)^T(x - x_k) \quad \text{(first order)}, \\
  f(x) &\approx f(x_k) + \nabla f(x_k)^T(x - x_k) + \tfrac{1}{2}(x - x_k)^T\nabla^2 f(x_k)(x - x_k) \quad \text{(second order)}.
\end{align}
Minimising the first-order model yields gradient descent. Minimising the second-order (quadratic) model yields the Newton step.

\subsection{Local Approximations of $g(x) = x^4$}

For $g(x) = x^4$ expanded at $x_0 = 0.25$, with $g(x_0) = 0.0039$, $g'(x_0) = 4(0.25)^3 = 0.0625$, $g''(x_0) = 12(0.25)^2 = 0.75$:

\textbf{First-order (Taylor) approximation:}
\begin{equation}
  g(x) \approx 0.0039 + 0.0625\,(x - 0.25) = 0.0039 + 0.0625x - 0.015625.
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
  g(x) \approx 0.0039 + 0.0625\,(x - 0.25) + \tfrac{1}{2}(0.75)(x - 0.25)^2 = 0.0039 + 0.0625(x-0.25) + 0.375(x-0.25)^2.
\end{equation}

The minimum of the second-order model is found by setting the derivative to zero:
\begin{equation}
  0.0625 + 0.75(x - 0.25) = 0 \implies x = 0.25 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} \approx 0.167.
\end{equation}
This is the Newton step from $x_0 = 0.25$, giving a much better descent direction than the gradient step.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping (Regularised)}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$ (damping factor), Tikhonov regularisation $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise to ensure positive definiteness}
    \State $p_k \gets H_k^{-1}\,g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\,p_k$
    \Comment{$\alpha = 1$: full Newton; $\alpha < 1$: damped Newton}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Quadratic convergence of Newton's method]
\label{thm:newton}
Let $f$ be twice continuously differentiable with $\mu$-strongly convex, $L$-Lipschitz Hessian. Starting from a sufficiently close point to $x^\star$, Newton's method with $\alpha = 1$ satisfies
\begin{equation}
  \|x_{k+1} - x^\star\| \leq \frac{L}{2\mu}\,\|x_k - x^\star\|^2.
\end{equation}
This is \emph{quadratic convergence}: errors are squared at each iteration. In practice, starting far from the optimum requires either line search or damping ($\alpha < 1$) to ensure global convergence.
\end{theorem}

\begin{remark}[Newton for exact quadratics]
For a quadratic $f(\theta) = \frac{1}{2}\theta^T H \theta - b^T\theta + c$ (Benchmark A), $\nabla f = H\theta - b$ and $\nabla^2 f = H$ (constant). The Newton step gives $\theta_{k+1} = \theta_k - H^{-1}(H\theta_k - b) = H^{-1}b = \theta^\star$. Hence Newton converges in \emph{exactly one step} regardless of condition number, directly to the OLS solution $\hat\theta_{\text{OLS}}$.
\end{remark}

Settings used: GD runs 80 iterations; Newton runs 20 iterations with damping $\lambda = 10^{-8}$ (near-zero regularisation, for numerical stability only).

\subsection{Newton vs.\ Quasi-Newton Methods}

The main cost of Newton's method is computing and inverting the $d \times d$ Hessian per iteration: $O(d^3)$ per step (or $O(d^2)$ with specialised solvers). For large $d$, this becomes prohibitive. Quasi-Newton methods (e.g., L-BFGS, BFGS) approximate $H^{-1}$ using secant information from past gradients, achieving super-linear convergence with $O(d^2)$ or $O(d)$ cost per step. For our $d = 2$ benchmarks, exact Newton is affordable; in practice, L-BFGS is the workhorse for medium-scale smooth optimisation.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation}
\label{sec:q4}
% ============================================================

\subsection{Motivation and Context}

Many real optimisation problems involve \emph{black-box} objectives where the gradient is unavailable (e.g., simulation-based objectives, hyperparameter tuning). This section explores methods ranging from gradient approximation (finite differences) to purely derivative-free approaches.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
  \label{eq:fd}
\end{equation}
where $e_i$ is the $i$-th standard basis vector. This requires $d+1 = 3$ function evaluations per gradient estimate (one per coordinate plus one at $x$). The total error is:
\begin{equation}
  \left|\frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right|
  = \underbrace{\frac{\delta}{2}\,f''(\xi)}_{\text{truncation error } O(\delta)} + \underbrace{\frac{2\varepsilon_{\mathrm{mach}}}{\delta}}_{\text{roundoff error } O(\varepsilon/\delta)},
\end{equation}
where the optimal $\delta$ balances truncation and roundoff: $\delta^\star \sim \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$ for double precision. The assignment tests $\delta = 0.05$ (close to good approximation) and $\delta = 0.8$ (large truncation error, poor approximation).

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, perturbation $\delta > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k/\|u_k\|$ \Comment{Uniform direction on sphere}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\,u_k$ \Comment{One-sided directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\,\hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method uses only 2 function evaluations per iteration (1 at $x_k$ and 1 at $x_k + \delta u_k$), regardless of dimension. The directional derivative estimate $\hat{g}_k$ is a noisy projection of the true gradient onto the random direction $u_k$:
\begin{equation}
  \mathbb{E}[\hat{g}_k \mid x_k] = \mathbb{E}\!\left[\frac{f(x_k + \delta u_k) - f(x_k)}{\delta} u_k\right] \approx \frac{1}{d}\,\nabla f(x_k).
\end{equation}
Thus $\hat{g}_k$ is a biased estimator, with bias $O(\delta)$ and high variance due to projection onto a 1D subspace. This gives convergence rates of $O(d/k)$ --- dimension-dependent --- compared to $O(1/k)$ for exact gradient methods.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it using four geometric operations. In iteration order of application:
\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r = 1.0$): Move the worst vertex $x_w$ through the centroid $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_w)$.
  \item \textbf{Expansion} ($\gamma = 2.0$): If reflection improved beyond the best vertex, try further: $x_e = \bar{x} + \gamma(\bar{x} - x_w)$.
  \item \textbf{Contraction} ($\rho = 0.5$): If reflection did not improve, contract towards centroid: $x_c = \bar{x} + \rho(x_r \text{ or } x_w - \bar{x})$.
  \item \textbf{Shrink} ($\sigma = 0.5$): If contraction fails, shrink all vertices towards the best.
\end{enumerate}

The simplex adapts its shape to the local landscape without any gradient information, making it robust for moderately low-dimensional noisy black-box problems. Settings: initial simplex step $= 0.35$, iterations $= 160$.

\subsection{Problem Setup and Theory}

\textbf{Problem:} Minimise Benchmark B, $f(x_1, x_2) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)$, subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$.

The feasible set $\mathcal{C} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}$ is a closed convex half-space. The unconstrained minimum is at $x^\star_{\text{unc}} \approx (0.582, 2.0)$, which satisfies $x_1^\star \approx 0.582 > 0.5$, so the constraint is \emph{inactive} at the unconstrained optimum. Hence the \textbf{constrained and unconstrained minima coincide} at $(0.582, 2.0)$ with $f^\star = 0.7244$.

The \textbf{KKT conditions} for this problem are:
\begin{align}
  \nabla f(x^\star) + \mu^\star \cdot (-e_1) &= 0 \quad \text{(stationarity)}, \\
  x_1^\star &\geq 0.5 \quad \text{(primal feasibility)}, \\
  \mu^\star &\geq 0 \quad \text{(dual feasibility)}, \\
  \mu^\star\,(x_1^\star - 0.5) &= 0 \quad \text{(complementary slackness)}.
\end{align}
Since $x_1^\star \approx 0.582 > 0.5$, complementary slackness gives $\mu^\star = 0$, confirming the constraint is inactive and the constrained and unconstrained solutions agree.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\,\nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1}) = \begin{bmatrix} \max(0.5,\; z_{k+1}^{(1)}) \\ z_{k+1}^{(2)} \end{bmatrix}$ \Comment{Project onto feasible set}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection onto $\mathcal{C} = \{x : x_1 \geq 0.5\}$ is trivially $\Pi_\mathcal{C}(z) = (\max(0.5, z_1), z_2)^T$. For general convex sets the projection can be expensive (e.g., $O(d \log d)$ for norm-ball constraints, $O(d^3)$ for semidefinite constraints), making PGD less attractive than penalty methods in those settings.

\begin{algorithm}[H]
\caption{Penalty Method (Linear / Hinge Penalty)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, penalty weight $\lambda > 0$, iterations $N$
\State Define $F_\lambda(x) = f(x) + \lambda\,\max(0,\; 0.5 - x_1)$ \Comment{Penalise constraint violation}
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute $\nabla F_\lambda(x_k)$:
    \If{$x_1^{(k)} < 0.5$}
        \State $\nabla F_\lambda = \nabla f(x_k) + \lambda(-e_1)$ \Comment{Active penalty gradient}
    \Else
        \State $\nabla F_\lambda = \nabla f(x_k)$ \Comment{Penalty inactive}
    \EndIf
    \State $x_{k+1} \gets x_k - \alpha\,\nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method appends a term that penalises constraint violation. As $\lambda \to \infty$, the penalised minimiser converges to the constrained minimiser, but large $\lambda$ introduces ill-conditioning (Hessian of $F_\lambda$ can be very stiff), requiring smaller step sizes for stability.

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected GD} enforces feasibility exactly and immediately at every iteration. This is its fundamental advantage: feasibility is a hard constraint, unconditionally satisfied from iteration 1. The cost is requiring a projection oracle, which for half-space and box constraints is trivially cheap ($\max$ operations) but can be expensive for general convex sets (e.g., nuclear-norm ball projection requires SVD).

\textbf{Penalty methods} trade exact feasibility for algorithmic simplicity: only the modified gradient $\nabla F_\lambda$ is needed, requiring no projection oracle. The penalty weight $\lambda$ controls the feasibility--optimality trade-off. The hinge penalty $\max(0, -g(x))$ used here is non-smooth at the constraint boundary $x_1 = 0.5$, creating a kink in the gradient. More sophisticated approaches like augmented Lagrangian (ADMM) or interior-point methods combine the advantages of both penalty and projection.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$ (a box / rectangle in $\mathbb{R}^2$).

\textbf{Linear Programme:} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$, where $a = [1, 2]^T$.

Since $f$ is linear and $\mathcal{X}$ is a bounded convex polytope:
\begin{theorem}[Fundamental theorem of LP]
The minimum of a linear objective over a bounded polytope is attained at a \emph{vertex} (extreme point) of the polytope.
\end{theorem}

The vertices of $\mathcal{X}$ are $(0.5, -5)$, $(5, -5)$, $(0.5, 10)$, $(5, 10)$. Evaluating $f$:
\begin{equation}
  f(0.5, -5) = 0.5 + 2(-5) = -9.5, \quad f(5, -5) = 5 - 10 = -5, \quad
  f(0.5, 10) = 0.5 + 20 = 20.5, \quad f(5, 10) = 5 + 20 = 25.
\end{equation}
Hence the minimum is $f^\star = -9.5$ at $x^\star = (0.5, -5)$.

Equivalently, since both components of $a$ are positive ($a_1 = 1 > 0$, $a_2 = 2 > 0$), minimising $a^T x$ requires minimising each coordinate at its lower bound: $x_1 \to 0.5$, $x_2 \to -5$.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing weight $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$
    \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\,x_k + (1-\beta)\,z_k$
    \Comment{Convex combination preserves feasibility}
\EndFor
\end{algorithmic}
\end{algorithm}

The key insight is that $x_{k+1}$ is a convex combination of $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$, hence $x_{k+1} \in \mathcal{X}$ by convexity of $\mathcal{X}$. \emph{No projection is needed.} The linear minimisation oracle (LMO) for a box constraint decomposes per coordinate:
\begin{equation}
  (z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations, compared to $O(d^3)$ for a projection onto a general convex set.

\begin{theorem}[Frank--Wolfe convergence, $O(1/k)$]
\label{thm:fw}
For an $L$-smooth convex function $f$ and the Frank--Wolfe algorithm with step-size schedule $\gamma_k = 2/(k+2)$ (i.e., $1-\beta_k = 2/(k+2)$):
\begin{equation}
  f(x_k) - f^\star \leq \frac{2LC^2}{k+2}, \quad \text{where } C = \max_{x,y \in \mathcal{X}} \|x - y\|.
\end{equation}
Here $C$ is the diameter of the feasible set. This is an $O(1/k)$ convergence rate, slower than projected gradient descent's linear rate for strongly convex $f$, but each Frank--Wolfe step costs only a linear programme over $\mathcal{X}$ instead of a projection.
\end{theorem}

Note that the implementation uses a fixed $\beta$ rather than the theoretically optimal decreasing $\gamma_k = 2/(k+2)$, which explains residual oscillation near the optimum.

\subsubsection{Frank--Wolfe Gap}
A useful stopping criterion is the \emph{Frank--Wolfe gap}:
\begin{equation}
  g_k := \nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - f^\star,
\end{equation}
which provides a computable upper bound on the suboptimality $f(x_k) - f^\star$ and hence can be used to certify convergence without knowing $f^\star$.

\subsection{Interior vs.\ Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits characteristic zig-zagging. The LP vertex $z_k$ alternates between extreme points of $\mathcal{X}$, and the convex combination gradually averages these to reach the interior. The convergence rate $O(1/k)$ from Theorem~\ref{thm:fw} is confirmed empirically. Smaller $\beta$ (larger steps towards $z_k$) accelerates initial convergence but introduces oscillation near the optimum.

\textbf{Boundary optimum:} Frank--Wolfe is naturally suited to boundary optima. When the unconstrained optimum lies outside $\mathcal{X}$, the LMO consistently selects a boundary vertex, and iterates accumulate on that boundary. The residual gap $f = 0.364 > f^\star = 0.25$ after 140 iterations reflects the $O(1/k)$ rate and fixed $\beta$ schedule. Using the theoretically optimal schedule $\gamma_k = 2/(k+2)$ (i.e., $\beta_k = k/(k+2)$) would improve convergence.

\textbf{Projection-free advantage:} Unlike PGD, Frank--Wolfe never requires a projection step. For problems where the LMO is cheap (e.g., box constraints: $O(d)$; simplex: $O(d \log d)$ for sort) but projection is expensive (e.g., nuclear norm ball: requires SVD), Frank--Wolfe can be substantially more efficient. Additionally, Frank--Wolfe produces \emph{sparse iterates}: $x_k$ is always a convex combination of vertices, which has sparsity-promoting properties in structured settings (e.g., sparse recovery, matrix factorisation).

\subsection{Key Insights}

\begin{enumerate}[leftmargin=*]
  \item \textbf{Problem structure drives method choice.} On well-conditioned quadratic Benchmark A, nearly all gradient methods converge rapidly and the choice of algorithm matters little. On ill-conditioned Rosenbrock (condition number $\approx 2000$), method differences are decisive: Newton outperforms all first-order methods despite using far fewer iterations.

  \item \textbf{Adaptivity helps anisotropy.} Adagrad, RMSprop, and Adam provide implicit preconditioning by scaling per-coordinate gradients, which is beneficial for anisotropic problems (Benchmark B) without requiring Hessian computation. However, this per-coordinate scaling does not substitute for full Hessian preconditioning (Newton) on highly nonlinear problems.

  \item \textbf{Acceleration is real but fragile.} Nesterov's $O(1/k^2)$ acceleration over GD's $O(1/k)$ is empirically confirmed on Rosenbrock ($f = 0.48$ vs.\ $3.39$ after 150 iterations). However, the advantage relies on carefully tuned momentum and step parameters; the Heavy Ball method, while theoretically faster for quadratics, may oscillate on non-convex functions.

  \item \textbf{Derivative-free methods are dimension-limited.} Nelder--Mead is competitive on $d = 2$ Rosenbrock, but its $O(n^2)$ simplex operations per step and lack of gradient information make it impractical for $d \gg 10$. Grid search suffers the curse of dimensionality exponentially. For black-box problems at scale, Bayesian optimisation or evolution strategies are preferable.

  \item \textbf{Constrained optimisation: projection vs.\ penalty.} Projected GD enforces hard feasibility at every iteration but requires a tractable projection. Penalty methods are simpler to implement but only achieve asymptotic feasibility as $\lambda \to \infty$. When the constraint is inactive at the optimum (as in Q5), both methods reach the same solution with appropriate parameters.

  \item \textbf{Frank--Wolfe's projection-free advantage.} The Frank--Wolfe algorithm avoids projections entirely, replacing them with cheap linear programmes. Its $O(1/k)$ convergence rate is slower than projected GD but provides sparsity in iterates, which is valuable in applications like sparse recovery and matrix completion. The fixed-$\beta$ step schedule used here is suboptimal; the theoretically optimal decreasing schedule $\gamma_k = 2/(k+2)$ ensures the theoretical $O(1/k)$ rate.
\end{enumerate}

All methods investigated connect directly to material presented in the CS7DS2 lectures. The Polyak step (Week 3b), Adagrad, and RMSprop (Week 3b) extend vanilla gradient descent with adaptive step rules. The Heavy Ball and Nesterov methods (Weeks 4a--4b) introduce momentum, with Nesterov achieving the theoretically optimal first-order rate. Newton's method (Week 6a) uses second-order information for superlinear convergence. Derivative-free methods (Weeks 8a--8b) bypass the need for gradient computations. Projected gradient and penalty methods (Week 9a) handle constraints, while Frank--Wolfe (Week 10a) provides projection-free constrained optimisation. This report's experimental results confirm the theoretical predictions from the lectures across all six families of methods.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum (Accelerated Gradient)}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z       # gradient evaluated at lookahead
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))  # Tikhonov regularisation
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # Linear minimisation oracle: select box vertex component-wise
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # convex combination (feasibility preserved)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
