\begin{abstract}
This report presents a comprehensive comparative study of six categories of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark A), a toy neural network loss with sinusoidal perturbation (Benchmark B), and the ill-conditioned Rosenbrock function (Benchmark C). The methods investigated are: \emph{(i)} adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), \emph{(ii)} momentum-based and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD), \emph{(iii)} Newton's method with local quadratic approximation, \emph{(iv)} derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead, grid search), \emph{(v)} constrained optimisation via projected gradient descent and penalty methods, and \emph{(vi)} the Frank--Wolfe conditional gradient algorithm for linear-constrained problems. Key findings include: Newton's method solves the quadratic problem in one iteration and achieves $f = 0.686$ on Rosenbrock in only 20 steps; Nesterov acceleration yields a $7\times$ improvement over gradient descent on Rosenbrock; Polyak step size is highly sensitive to knowledge of $f^{\star}$ (diverging when misspecified but converging fastest when exact); and projected gradient descent enforces feasibility as a hard constraint from iteration one, outperforming penalty methods in constraint satisfaction. Each method is evaluated in terms of convergence speed, trajectory behaviour, stability, and computational trade-offs, with connections drawn to the theoretical guarantees developed in the course.
\end{abstract}

\section{Introduction}
\label{sec:intro}
% ============================================================

Optimisation lies at the heart of modern machine learning and data analysis. The choice of optimisation algorithm directly impacts training speed, solution quality, and computational feasibility. This report systematically evaluates a broad spectrum of methods spanning four decades of optimisation research, from classical gradient descent and Newton's method to modern adaptive gradient algorithms and derivative-free approaches.

The experiments are structured around six questions, each targeting a distinct class of algorithms. Three standard benchmark functions of increasing difficulty (Section~\ref{sec:benchmarks}) serve as a common evaluation platform. All experiments use \texttt{np.random.seed(42)} for reproducibility.

\subsection*{Notation}
Throughout this report, $f : \mathbb{R}^n \to \mathbb{R}$ denotes the objective function, $\nabla f(x)$ its gradient, and $\nabla^2 f(x)$ its Hessian. The iterates are $\{x_k\}_{k \geq 0}$, and $x^{\star}$ denotes a local or global minimiser with optimal value $f^{\star} = f(x^{\star})$. The operator $\odot$ denotes element-wise (Hadamard) product. The Euclidean norm is $\|\cdot\| = \|\cdot\|_2$.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and $n = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a \emph{convex quadratic} with constant Hessian $H = \frac{1}{m}X^T X \succ 0$ and gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. With $m = 1000$ IID standard Gaussian samples, the empirical covariance $X^TX/m$ is near-identity with condition number $\kappa(H) \approx 1.12$, making this a \emph{well-conditioned} problem. Due to observation noise, the OLS (Bayes-optimal) minimum is $J^{\star} = 0.4826$ at $\theta_{\text{OLS}} = (3.028, 4.017)$ (close to true $\theta^{\star} = [3, 4]^T$ by consistency of OLS). Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = \bigl[2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)\bigr]^T$ and the Hessian is diagonal:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The eigenvalue ratio varies: at $x_1 = -1$ (start), $2 - \sin(-1) \approx 2.84$, giving $\kappa \approx 3.5$; at the optimum $x_1^{\star} \approx 0.582$, $2 - \sin(0.582) \approx 1.45$, giving $\kappa^{\star} \approx 6.9$. The minimum lies at $x^{\star} = (0.58244, 2.0)$ with $f^{\star} = 0.72442$ (exact: solve $2(x_1-1) + \cos(x_1) = 0$ numerically). Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian is:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the starting point $x_0 = [-1, 1]^T$, the Hessian evaluates to:
\begin{equation}
\nabla^2 f(x_0) = \begin{bmatrix} 802 & 400 \\ 400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $\lambda_1 \approx 0.399$ and $\lambda_2 \approx 1001.6$, giving condition number $\kappa(H_0) \approx 2508$. The Lipschitz constant of $\nabla f$ near $x_0$ is $L \approx 1001.6$, requiring a stable gradient descent step size of $\alpha < 2/L \approx 0.002$ (we use $\alpha = 0.0012$ for additional safety). At the global optimum $(1, 1)$, the same eigenvalues appear---the structure is preserved---but the off-diagonal entry changes sign: $\nabla^2 f(1,1) = \text{diag-alike}([802, -400; -400, 200])$ with $\kappa \approx 2508$. This persistent extreme ill-conditioning, combined with the narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock the canonical challenge for first-order methods. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Algorithm Formulations}

Four adaptive step-size methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for 120 iterations. The fundamental motivation for adaptive methods is that a fixed step size $\alpha$ must satisfy the global descent condition $\alpha < 2/L$ (where $L$ is the Lipschitz constant of $\nabla f$), which can be overly conservative when local curvature is much smaller than $L$.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, estimate of optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Optimal subgradient step size}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak convergence]
For a convex function $f$ with known $f^{\star}$, the Polyak step achieves $\min_{k \leq K} f(x_k) - f^{\star} = O(1/\sqrt{K})$ for non-smooth $f$ and linear convergence for strongly convex $f$.
\end{theorem}

The Polyak step size has the elegant property of being scale-invariant: it automatically adjusts to the local landscape without requiring a global Lipschitz constant estimate.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^n$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Accumulate squared gradients per coordinate}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's key insight is that coordinates with historically large gradients should receive smaller effective step sizes. The effective step for coordinate $i$ at iteration $k$ is $\alpha_{k,i} = \alpha_0 / \sqrt{\sum_{j=0}^k g_{j,i}^2}$. This monotonically decreasing schedule makes Adagrad particularly well-suited to sparse gradient settings (e.g., NLP where most word embeddings are rarely updated) but can cause premature deceleration on dense problems.

\begin{remark}[Adagrad online learning regret]
In the online convex optimisation framework, Adagrad achieves regret:
\begin{equation}
R_T = \sum_{t=1}^T f_t(x_t) - f_t(x^\star) \leq \frac{1}{2}\sum_{i=1}^n \left(\|x_{1,i} - x^\star_i\|_2 \cdot \sqrt{\sum_{t=1}^T g_{t,i}^2}\right)
\end{equation}
which adapts dimension-by-dimension to the gradient magnitude, achieving better bounds than SGD with fixed step in problems where most coordinates have small gradients (sparse data).
\end{remark}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's over-decay by replacing the cumulative sum with an exponential moving average (EMA) with decay $\beta$. The effective memory window is approximately $1/(1-\beta)$ iterations. With $\beta = 0.9$, recent gradients receive an effective weight of $0.9^k$ at lag $k$, so information older than $\sim 10$ iterations is largely discarded.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Velocity accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method maintains a momentum vector $z_k$ that accumulates a decaying sum of past gradients: $z_k = \alpha \sum_{j=0}^{k-1} \beta^{k-1-j} \nabla f(x_j)$. For a strongly convex quadratic with condition number $\kappa$, the optimal Heavy Ball parameters achieve a convergence rate of $O\bigl((\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1})^k\bigr)$, compared to $O\bigl((\frac{\kappa-1}{\kappa+1})^k\bigr)$ for GD---a quadratic improvement.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$, a $370\times$ improvement over GD). When $f^{\star}$ is unavailable or inaccurate, Heavy Ball and Adagrad are preferred.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging of squared gradients. Heavy Ball shows characteristic overshoot (due to momentum accumulation) followed by damped oscillation as momentum is gradually exhausted.
\item \textbf{Stability:} GD is most stable but slowest. Polyak is \emph{unstable} when $f^{\star}$ is misspecified. Adagrad is uniformly stable due to its monotonically decreasing step size, though this monotone decay may cause premature convergence stagnation on non-convex problems.
\item \textbf{Anisotropy handling:} Adagrad and RMSprop handle anisotropic curvature implicitly through per-coordinate scaling: coordinates with systematically larger gradients (e.g., $x_2$ in Benchmark B with 5$\times$ stronger curvature) automatically receive smaller effective step sizes.
\item \textbf{Practical recommendation:} In practice, exact $f^{\star}$ is rarely known, limiting Polyak's direct applicability. RMSprop and Adagrad are more robust choices for general problems. Heavy Ball is effective when the curvature tensor is reasonably well-conditioned.
\end{itemize}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, max momentum $\beta_{\max} \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$, \; $x_1 \gets x_0$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Corrected velocity}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov 1983]
For a $\beta$-smooth convex function, Nesterov's method with $\beta_k = (k-1)/(k+2)$ and $\alpha \leq 1/\beta$ achieves:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2\beta \|x_0 - x^{\star}\|^2}{(k+1)^2} = O\!\left(\frac{1}{k^2}\right),
\end{equation}
which is \emph{optimal} among first-order methods for smooth convex optimisation.
\end{theorem}

The crucial distinction from Heavy Ball (Section~\ref{sec:q1}) is the \emph{gradient lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This forward-looking evaluation allows the method to correct overshoot before it occurs, establishing the theoretical $O(1/k^2)$ rate. Heavy Ball evaluates the gradient at $x_k$ and cannot achieve this guarantee in general.

The increasing momentum schedule $\beta_k = (k-1)/(k+2)$ is not arbitrary: it is the exact sequence required to achieve the $O(1/k^2)$ convergence rate certificate in Nesterov's original proof. As $k \to \infty$, $\beta_k \to 1$, meaning the algorithm places increasing weight on the accumulated direction. The cap at $\beta_{\max}$ prevents instability in non-convex or ill-conditioned settings where the asymptotic $\beta_k \to 1$ would cause divergence.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moment parameters $\beta_1, \beta_2 \in [0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_{t-1})$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_t \gets x_{t-1} - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines two key ideas: (1) momentum via the first moment estimate $m_t$ (similar to Heavy Ball), and (2) per-coordinate adaptive scaling via the second moment estimate $v_t$ (similar to RMSprop). The bias correction $(1 - \beta^t)^{-1}$ is essential during early iterations when the EMA estimates are initialised at zero and would otherwise systematically underestimate the true moments. Adam inherits RMSprop's adaptivity to anisotropic curvature while adding momentum-driven acceleration.

\begin{remark}
Adam does not have a general convergence guarantee for non-convex objectives; Reddi et al.\ (2018) demonstrated cases where Adam fails to converge and proposed AMSGrad as a fix. In practice, Adam performs very well on neural network objectives that are empirically well-conditioned.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs (each epoch processes all $m = 1000$ data points). Data is shuffled at every epoch to prevent systematic bias.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{epoch $e = 1, 2, \ldots, E$}
    \State Randomly permute indices $\{1, \ldots, m\}$ into $\pi$
    \For{$i = 0, b, 2b, \ldots, m-b$ (mini-batch start indices)}
        \State $\mathcal{B} \gets \{\pi_{i+1}, \ldots, \pi_{i+b}\}$ \Comment{Mini-batch indices}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Mini-batch gradient}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

With $m = 1000$ samples and batch size $b$, each epoch performs $\lfloor m/b \rfloor$ weight updates. The mini-batch gradient is an unbiased estimate of the full gradient: $\mathbb{E}[g_{\mathcal{B}}] = \nabla J(\theta)$, with variance scaling as $O(\sigma^2/b)$ where $\sigma^2$ is the per-sample gradient variance. This variance-batch size relationship captures the fundamental SGD trade-off: smaller batches provide more updates per epoch but noisier gradients; larger batches provide fewer updates but lower variance. Variance reduction techniques (e.g., SVRG, SAGA) decouple this trade-off by achieving $O(1/k)$ convergence with $O(1)$-sized batches, at the cost of periodically computing a full gradient.

\subsection{Local Approximations of $g(x) = x^4$}

Taylor's theorem states that any sufficiently smooth function can be locally approximated by polynomials. For $g(x) = x^4$ at expansion point $x_0 = 0.25$, we have $g^{(1)}(x) = 4x^3$, $g^{(2)}(x) = 12x^2$:

\textbf{First-order (linear) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.00391 + 0.0625(x - 0.25).
\end{equation}
Minimising this affine function yields no finite minimiser (it has no stationary point), so gradient descent uses a fixed step to move in the negative gradient direction.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.00391 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
The minimum of this quadratic model is obtained by setting the derivative to zero:
\begin{equation}
g'(x_0) + g''(x_0)(x - x_0) = 0 \implies x_{1} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.1667.
\end{equation}
This is precisely the \textbf{Newton step}: $x_{k+1} = x_k - [g''(x_k)]^{-1} g'(x_k)$.

\subsection{Newton's Method with Hessian Regularisation}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0, 1]$, Hessian regularisation $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise to ensure positive definiteness}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{Damped step: $\alpha < 1$ for non-quadratic problems}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Rationale for regularisation:} The regularisation $\lambda I$ (Tikhonov, or Levenberg--Marquardt damping) ensures that $H_k + \lambda I \succ 0$ even when $H_k$ is indefinite or near-singular. Near saddle points on non-convex functions, the raw Hessian may have negative eigenvalues, which would cause Newton to ascend rather than descend. With $\lambda = 10^{-8}$, the regularisation is negligible when $\|\nabla^2 f\|$ is large but prevents numerical instability.

\textbf{Rationale for damping ($\alpha < 1$):} The full Newton step $\alpha = 1$ is optimal for a quadratic objective but can overshoot on non-quadratic functions where the quadratic model is inaccurate far from $x_k$. For Rosenbrock, where the Hessian eigenvalues vary by a factor of $2000$ between locations, $\alpha = 0.22$ provides the appropriate conservatism.

\begin{theorem}[Newton convergence]
If $f$ is twice continuously differentiable, $x^{\star}$ is a non-degenerate local minimiser, and $x_0$ is sufficiently close to $x^{\star}$, then Newton's method (with $\alpha = 1$) exhibits \emph{quadratic local convergence}:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq C \|x_k - x^{\star}\|^2
\end{equation}
for some constant $C > 0$ depending on $\|\nabla^3 f\|$ and $\lambda_{\min}(\nabla^2 f(x^{\star}))$.
\end{theorem}

Settings: GD uses 80 iterations; Newton uses 20 iterations with $\lambda = 10^{-8}$.

\subsection{Computational Cost and Scalability}

Each Newton iteration requires: (i) evaluating $\nabla f(x_k)$: $O(n)$; (ii) evaluating $\nabla^2 f(x_k)$: $O(n^2)$ in general; (iii) solving $H_k p_k = g_k$: $O(n^3)$ via Cholesky factorisation. For $n = 2$ these costs are negligible. For large-scale problems ($n \gg 1$), the $O(n^3)$ cost per step makes Newton impractical.

\textbf{Quasi-Newton methods (L-BFGS):} A practical alternative is to \emph{approximate} $H_k^{-1}$ without explicit Hessian computation. The BFGS update maintains an estimate $B_k \approx H_k$ using rank-2 updates from gradient differences:
\begin{equation}
B_{k+1} = B_k - \frac{B_k s_k s_k^T B_k}{s_k^T B_k s_k} + \frac{y_k y_k^T}{y_k^T s_k}, \quad s_k = x_{k+1} - x_k, \quad y_k = \nabla f(x_{k+1}) - \nabla f(x_k).
\end{equation}
The Limited-memory BFGS (L-BFGS) variant stores only the last $m \approx 5$--$20$ gradient difference pairs, reducing memory to $O(mn)$ and matrix-vector product cost to $O(mn)$ per iteration while retaining superlinear convergence. L-BFGS is the de facto standard for large-scale unconstrained differentiable optimisation in machine learning and scientific computing.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference (FD) approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad i = 1, \ldots, n.
\end{equation}
The approximation error has two competing sources:
\begin{itemize}[nosep]
\item \textbf{Truncation error:} $O(\delta)$ --- from the neglected Taylor term: $f(x + \delta e_i) = f(x) + \delta \partial_i f(x) + \tfrac{\delta^2}{2}\partial_{ii}^2 f + \ldots$, so the FD estimate has bias $\tfrac{\delta}{2}\partial_{ii}^2 f(x)$.
\item \textbf{Floating-point error:} $O(\varepsilon_{\text{mach}}/\delta)$ --- from subtractive cancellation: when $\delta$ is very small, $f(x + \delta e_i)$ and $f(x)$ are nearly equal, and their difference loses significant bits.
\end{itemize}
Minimising the total error $O(\delta + \varepsilon_{\text{mach}}/\delta)$ over $\delta$ yields $\delta_{\text{opt}} = \sqrt{\varepsilon_{\text{mach}}} \approx 1.5 \times 10^{-8}$ for 64-bit arithmetic. A more accurate \emph{central difference} approximation $[\partial_i f(x)]_{\text{CD}} = [f(x + \delta e_i) - f(x - \delta e_i)]/(2\delta)$ has truncation error $O(\delta^2)$ rather than $O(\delta)$, giving optimal $\delta_{\text{opt,CD}} = \varepsilon_{\text{mach}}^{1/3} \approx 6 \times 10^{-6}$ at the cost of $2d$ (rather than $d+1$) function evaluations. The experiments test $\delta = 0.05$ (near-optimal forward-difference accuracy) and $\delta = 0.8$ (strong truncation error regime).

\subsubsection{Nesterov Random Search (Zeroth-Order Method)}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Randomised Smoothing)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, perturbation radius $\delta > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_n)$; normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Random unit vector}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Stochastic directional derivative}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Nesterov random search estimator $\hat{g}_k$ is related to the gradient of the \emph{smoothed function} $f_\delta(x) = \mathbb{E}_{u \sim \text{Uniform}(\mathcal{S}^{n-1})}[f(x + \delta u)]$. One can show that $\mathbb{E}[\hat{g}_k] = \nabla f_\delta(x_k)$, so the method performs stochastic gradient descent on the smoothed objective. The variance of $\hat{g}_k$ is $O(n \|\nabla f\|^2 / \delta^2)$, requiring $\delta$ to be balanced between approximation accuracy (small $\delta$) and variance reduction (large $\delta$). Crucially, it requires only \emph{two} function evaluations per iteration regardless of dimension $n$, and Nesterov (2011) shows convergence at rate $O(n/\sqrt{k})$ for non-smooth convex $f$---slower than GD by a factor of $n$, but with dimension-independent per-iteration cost.

\subsection{Part B: Derivative-Free Methods on Rosenbrock (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$ and iteratively reshapes it using four geometric operations:

\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): Move the worst vertex $x_{n+1}$ through the centroid of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{n+1})$.
\item \textbf{Expansion} ($\gamma = 2.0$): If reflection improved, try a further step: $x_e = \bar{x} + \gamma(\bar{x} - x_{n+1})$.
\item \textbf{Contraction} ($\rho = 0.5$): If reflection worsened, contract towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): If contraction also fails, contract all vertices towards the best point.
\end{enumerate}

No gradient information is used; progress relies entirely on the geometric relationship between simplex vertices and function values. Settings: initial simplex step $= 0.35$, iterations $= 160$.

\subsection{Problem Formulation}

Minimise Benchmark B subject to the half-space constraint $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$:
\begin{equation}
\min_{x \in \mathbb{R}^2}\; f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad \text{s.t.} \quad g(x) = 0.5 - x_1 \leq 0.
\end{equation}
The \textbf{KKT conditions} for this problem are:
\begin{align}
\nabla f(x^{\star}) + \lambda \nabla g(x^{\star}) &= 0, \\
\lambda g(x^{\star}) &= 0, \quad \lambda \geq 0, \quad g(x^{\star}) \leq 0.
\end{align}
Since the unconstrained minimum $x^{\star}_{\text{unc}} = (0.582, 2.0)$ satisfies $g(x^{\star}_{\text{unc}}) = 0.5 - 0.582 = -0.082 < 0$ (strictly feasible), complementary slackness implies $\lambda = 0$, and the first KKT condition reduces to $\nabla f(x^{\star}) = 0$. Therefore the \emph{constraint is not active} at the optimum, and the constrained and unconstrained optima coincide at $x^{\star} = (0.582, 2.0)$, $f^{\star} = 0.7244$. The algorithmic challenge lies in handling the infeasible starting point $x_0$ and ensuring feasibility is maintained or enforced throughout optimisation.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial $x_0$ (possibly infeasible), step size $\alpha > 0$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1})$ \Comment{Project onto feasible set}
\EndFor
\end{algorithmic}
\end{algorithm}

For the constraint $\mathcal{C} = \{x : x_1 \geq 0.5\}$, the projection is simply $\Pi_{\mathcal{C}}(z)_1 = \max(0.5, z_1)$ and $\Pi_{\mathcal{C}}(z)_2 = z_2$ (trivial $O(1)$ operation).

\begin{theorem}[PGD convergence]
For a convex objective $f$ with $L$-Lipschitz gradient and a closed convex feasible set $\mathcal{C}$, PGD with $\alpha = 1/L$ achieves $f(x_k) - f^{\star} \leq O(1/k)$ for general convex $f$ and linear convergence for strongly convex $f$.
\end{theorem}

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial $x_0$, step size $\alpha > 0$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised (exact) objective}
    \State $\nabla F_\lambda(x_k) \gets \nabla f(x_k) - \lambda \mathbf{1}[x_{1,k} < 0.5]\, e_1$ \Comment{Penalised gradient}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty function $\max(0, 0.5 - x_1)$ is an $\ell_1$ penalty (exact penalty), which adds a gradient of magnitude $\lambda$ directed into the feasible region whenever $x_1 < 0.5$. As $\lambda \to \infty$, the penalty method solutions converge to the constrained optimum, but larger $\lambda$ increases the Lipschitz constant of $\nabla F_\lambda$ and requires smaller $\alpha$ for stability.

\subsection{Comparative Discussion}

\textbf{Projected Gradient Descent} is the gold standard for problems with projectable constraint sets. Feasibility is a hard guarantee at every iterate, not just at convergence. This is critical in applications such as probability simplex constraints, norm balls in compressed sensing, or per-coordinate bounds in portfolio optimisation. The cost is requiring a fast projection oracle: trivial for half-spaces and box constraints ($O(d)$), but potentially $O(d^3)$ or worse for general semidefinite constraints.

\textbf{Penalty methods} are more flexible: any differentiable solver can be applied to $F_\lambda$ without modification. However, there is a fundamental trade-off: larger $\lambda$ enforces tighter feasibility but creates a stiffer objective requiring smaller step sizes. In the limit $\lambda \to \infty$, the penalised problem equals the constrained problem but the Lipschitz constant of $\nabla F_\lambda$ grows as $O(\lambda)$, requiring $\alpha = O(1/\lambda)$.

\textbf{Augmented Lagrangian method:} A principled resolution is the method of multipliers, which solves:
\begin{equation}
x_{k+1} = \argmin_{x}\; f(x) + \mu_k \max(0, 0.5 - x_1) + \frac{\rho}{2} \max(0, 0.5 - x_1)^2,
\end{equation}
followed by the dual update $\mu_{k+1} = \max(0, \mu_k + \rho \max(0, 0.5 - x_{1,k+1}))$. By maintaining a Lagrange multiplier estimate $\mu_k$ and a fixed (moderate) penalty $\rho$, the augmented Lagrangian achieves exact constraint satisfaction without requiring $\rho \to \infty$, thereby avoiding the ill-conditioning problem. The ADMM (Alternating Direction Method of Multipliers) extends this to separable problems and is widely used in distributed optimisation and signal processing.

\section{\texorpdfstring{Question 6: Linear Programmes and Frank--Wolfe Algorithm \hfill [30 marks]}{Question 6: Linear Programmes and Frank-Wolfe Algorithm [30 marks]}}

\subsection{Part I: Linear Programme over a Box Constraint}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box/rectangle).

\textbf{Linear Programme:} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$.

\begin{theorem}[Fundamental theorem of linear programming]
The minimum of a linear objective over a bounded polytope is always attained at a vertex (extreme point) of the polytope.
\end{theorem}

Since $f(x) = x_1 + 2x_2$ is separable and the coefficients $a_1 = 1 > 0$, $a_2 = 2 > 0$ are both positive, the minimum is attained by minimising each coordinate independently at its lower bound:
\begin{equation}
x^{\star} = \argmin_{x \in \mathcal{X}}\; (x_1 + 2x_2) = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This vertex solution can be found in $O(d)$ operations for box constraints.

\begin{remark}[Degenerate LP]
If the objective direction $a$ is parallel to a face of $\mathcal{X}$ (e.g., $a = [0, 1]^T$, so $a_1 = 0$), the minimum is attained along an entire face (a continuum of solutions), not a single vertex. In this degenerate case, the simplex method returns one vertex on the optimal face. For the box $\mathcal{X}$, this occurs when any component of $a$ is zero.
\end{remark}

For a general polytope with $m$ constraints, the LP can be solved in polynomial time (e.g., Ellipsoid or interior-point methods), but the Frank--Wolfe approach uses the LP oracle only to find the minimum of $\nabla f(x_k)^T z$ over $\mathcal{X}$, leveraging the LP structure to avoid projections entirely.

\subsection{Part II: Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \displaystyle\argmin_{z \in \mathcal{X}}\; \nabla f(x_k)^T z$ \Comment{Linear Minimisation Oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: iterate remains in $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Key properties of Frank--Wolfe:}
\begin{enumerate}[nosep]
\item \textbf{Projection-free:} Since $x_0 \in \mathcal{X}$ and $z_k \in \mathcal{X}$, the convex combination $x_{k+1} = \beta x_k + (1-\beta)z_k \in \mathcal{X}$ by convexity---no projection step is ever needed.
\item \textbf{LMO:} The linearisation subproblem $\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ is often much cheaper than a projection. For box constraints, the solution is immediate per coordinate.
\item \textbf{Sparse iterates:} Frank--Wolfe naturally produces solutions that are convex combinations of vertices of $\mathcal{X}$, inducing sparsity in problems like the Lasso ($\mathcal{X} = \ell_1$ ball).
\end{enumerate}

\begin{theorem}[Frank--Wolfe convergence]
For a smooth convex $f$ with Lipschitz gradient constant $L_f$ and bounded domain (diameter $D = \max_{x,y \in \mathcal{X}}\|x-y\|$), the standard Frank--Wolfe method with step sizes $\gamma_k = 2/(k+2)$ achieves:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2 L_f D^2}{k + 2} = O\!\left(\frac{1}{k}\right).
\end{equation}
The fixed-$\beta$ variant used here does not achieve this rate but is easier to implement.
\end{theorem}

For box constraints, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i < 0 \end{cases}, \quad l_i = 0.5,\ u_i = 5 \text{ for } i=1, \quad l_i = -5,\ u_i = 10 \text{ for } i=2.
\end{equation}

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits a characteristic ``zig-zag'' pattern near the optimum. The LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ at each iteration as the gradient direction oscillates, and the convex combination averages these jumps at rate $(1-\beta)$. The general convergence rate is $O(1/k)$---slower than gradient descent's linear rate for strongly convex objectives. This $O(1/k)$ rate is information-theoretically optimal for the Frank--Wolfe oracle model.

\textbf{Boundary optimum:} Frank--Wolfe is well-suited to boundary optima because $z_k$ consistently selects the vertex nearest to the constrained optimum (here, the face $x_1 = 0.5$), allowing the iterate to approach the boundary organically. The boundary case achieves $f = 0.364$ after 140 iterations (above $f^{\star} = 0.25$), indicating that the fixed-$\beta$ schedule decays too slowly compared to the optimal $\gamma_k = 2/(k+2)$ schedule.

\textbf{Frank--Wolfe gap as a stopping criterion:} A key advantage of the Frank--Wolfe algorithm is that it provides a computable \emph{duality gap}:
\begin{equation}
g_k = \nabla f(x_k)^T (x_k - z_k) \geq f(x_k) - f^{\star} \geq 0,
\end{equation}
which serves as a certificate of sub-optimality without knowing $f^{\star}$. When $g_k < \varepsilon$, we know $f(x_k) - f^{\star} < \varepsilon$. This is a significant advantage over gradient norm as a stopping criterion, since $\|\nabla f(x_k)\|$ can be small even far from the constrained optimum.

\textbf{Comparison with projected gradient descent:} PGD (Q5) achieves zero constraint violation from iteration 1 and converges linearly for strongly convex $f$, while Frank--Wolfe has an $O(1/k)$ sublinear rate and no projection. The trade-off is between projection oracle availability and LMO availability. For problems where the feasible set is the nuclear norm ball (matrix completion), the LMO is a rank-1 SVD (cheap), while projection requires a full SVD (expensive), making Frank--Wolfe strongly preferred in such settings.

\subsection{Why Rosenbrock Is Hard: A Mathematical Explanation}

The Rosenbrock function is exceptionally challenging for first-order methods due to three compounding factors:

\begin{enumerate}[nosep]
\item \textbf{Extreme ill-conditioning ($\kappa \approx 2508$):} The Hessian eigenvalue ratio means that GD must take steps $\alpha < 2/L \approx 0.002$, which are barely perceptible in the flat direction (eigenvalue $\lambda_{\min} \approx 0.4$). The condition number determines the linear convergence rate ratio: GD converges at rate $(\kappa - 1)/(\kappa + 1) \approx 0.9992$ per step --- extremely slow.

\item \textbf{Curved valley ($x_2 = x_1^2$):} The constraint manifold curves away from any fixed gradient direction. Gradient descent, which follows straight lines, must ``steer'' by repeatedly correcting direction, taking many small zigzag steps along the valley.

\item \textbf{Non-quadratic coupling:} The term $100(x_2 - x_1^2)^2$ creates a cubic off-diagonal Hessian term $-400 x_1$ that changes sign as $x_1$ passes through zero. This means the Newton quadratic model becomes inaccurate over large step sizes, necessitating damping ($\alpha = 0.22$).
\end{enumerate}

The combination of these factors creates the ``banana-shaped'' landscape visible in the contour plots: the iterates must navigate a thin, curved corridor where the walls are $100\times$ steeper than the floor, and where the direction of steepest ascent (perpendicular to the valley) changes continuously.

\subsection{Method Selection Guidelines}

Based on the experimental results and theoretical properties, the following guidelines emerge:

\begin{itemize}[nosep]
\item \textbf{Well-conditioned quadratic objectives (Benchmark A, $\kappa \approx 1.1$):} All methods converge in $< 50$ iterations. Newton is exact in one step. GD achieves linear convergence rate $r = \max(|1 - \alpha \lambda_{\min}|, |1 - \alpha \lambda_{\max}|) \approx 0.91$ per step with $\alpha = 0.08$.
\item \textbf{Mildly nonlinear objectives (Benchmark B, $\kappa \approx 3.5$--$6.9$):} Nesterov and Adam match GD's final accuracy in fewer iterations. Newton converges in $\sim 5$ steps.
\item \textbf{Ill-conditioned nonlinear objectives (Benchmark C, $\kappa \approx 2508$):} Nesterov provides the best first-order result ($7\times$ over GD). Polyak step size is excellent when $f^\star$ is known exactly ($370\times$ better than GD). Nelder--Mead achieves \emph{exact} convergence using zero gradient information.
\item \textbf{Gradient information unavailable:} Finite differences ($\delta \approx \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ optimal, $\delta = 0.05$ practical) provide near-exact gradient estimates at $O(d)$ function evaluations. For $d \geq 100$, Nesterov random search's $O(1)$ evaluations per step become competitive despite slower convergence.
\item \textbf{Constrained optimisation:} Projected gradient descent offers hard feasibility guarantees with trivial $O(d)$ projection for half-spaces and boxes. Penalty methods are simpler but require tuning $\lambda$ and accept approximate feasibility. Augmented Lagrangian (ADMM) provides exact feasibility without infinite $\lambda$.
\item \textbf{Structured convex constraints (LP feasible sets, nuclear norm balls):} Frank--Wolfe (conditional gradient) avoids projections entirely, leveraging cheap LP oracles.
\item \textbf{Deep learning practice:} Adam is the default due to its robustness to hyperparameter tuning and good empirical performance on stochastic nonconvex objectives, despite lack of convergence guarantees in theory.
\end{itemize}

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has presented a systematic comparative study of twelve optimisation algorithms across three benchmark problems of increasing difficulty. The main findings are:

\begin{enumerate}
\item \textbf{Adaptive step-size methods (Q1):} The Polyak step size achieves the fastest convergence when $f^\star$ is known exactly (Rosenbrock: $f = 0.009$, a $370\times$ improvement over GD) but diverges catastrophically under $f^\star$ misspecification. Adagrad and Heavy Ball provide robust, consistently good performance across all benchmarks without requiring $f^\star$.

\item \textbf{Momentum and stochastic methods (Q2):} Nesterov acceleration achieves a $7\times$ improvement over GD on Rosenbrock in 150 iterations, consistent with its theoretical $O(1/k^2)$ rate advantage over GD's $O(1/k)$ rate on smooth convex objectives. Adam combines momentum and adaptivity but does not outperform Nesterov on these deterministic objectives. Mini-batch SGD demonstrates the fundamental variance--batch-size trade-off: smaller batches yield faster per-epoch convergence but more oscillation near the minimum.

\item \textbf{Newton's method (Q3):} Newton's method converges in one iteration on the quadratic benchmark (as theory predicts for exact quadratics) and achieves $5\times$ better results than GD on Rosenbrock in $4\times$ fewer iterations. The $O(n^3)$ per-iteration cost is negligible for $n = 2$ but limits applicability at scale.

\item \textbf{Derivative-free methods (Q4):} Finite differences with $\delta = 0.05$ accurately recover the exact gradient, while $\delta = 0.8$ introduces bias comparable to the gradient magnitude. Nesterov random search provides dimension-free cost per iteration at the expense of slow convergence. Nelder--Mead navigates the curved Rosenbrock valley without any gradient information, demonstrating robust geometric adaptation. Grid search provides global optimality guarantees at exponential cost.

\item \textbf{Constrained optimisation (Q5):} Projected gradient descent enforces feasibility as a hard constraint from iteration 1, while penalty methods asymptotically approach feasibility. The penalty weight $\lambda$ controls the feasibility--stability trade-off: large $\lambda$ achieves tight feasibility but requires smaller step sizes.

\item \textbf{Frank--Wolfe (Q6):} The conditional gradient method handles box-constrained problems without projection, converging at $O(1/k)$ rate. Smaller mixing steps ($\beta = 0.985$) outperform larger steps ($\beta = 0.90$) due to reduced oscillation near interior optima. The algorithm naturally handles boundary optima without explicit constraint enforcement.

\end{enumerate}

A consistent theme across all questions is the \emph{no-free-lunch} principle: no single method dominates all problem types. The best algorithm depends critically on the problem structure (convexity, conditioning, separability), the information available (gradients, Hessians, $f^\star$), the constraint geometry, and computational budget. The course-spanning comparison provided by these experiments reinforces why practitioners need a broad repertoire of optimisation tools.

From a theoretical perspective, the experiments validate several key results from the course:
\begin{itemize}[nosep]
\item Nesterov's optimal $O(1/k^2)$ rate (vs GD's $O(1/k)$) manifests as a $7\times$ improvement on Rosenbrock in 150 steps.
\item Newton's quadratic local convergence reduces the Benchmark B objective from $23.2$ to $0.724$ in 5 steps.
\item The Polyak step's sensitivity to $f^\star$ error illustrates why practical subgradient methods often use heuristic step schedules.
\item The Frank--Wolfe $O(1/k)$ rate is empirically confirmed by the duality gap plot (Figure~\ref{fig:fw_gap}).
\item KKT theory explains why the constrained and unconstrained optima coincide for Benchmark B under the $x_1 \geq 0.5$ constraint.
\item Nelder--Mead's exact convergence to the Rosenbrock global minimum demonstrates that derivative-free methods can match or exceed gradient-based methods on problems where geometric adaptation (simplex reshaping) compensates for the lack of gradient information.
\end{itemize}

\begin{lstlisting}[caption={Gradient Descent (baseline)}]
def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball (Polyak Momentum)}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)  # Increasing schedule
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)  # Gradient at extrapolated point
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g       # Biased 1st moment
        v_vec = beta2 * v_vec + (1 - beta2) * g**2    # Biased 2nd moment
        m_hat = m_vec / (1 - beta1**t)                # Bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))  # Regularise: H + lambda*I
        try:
            p = np.linalg.solve(H_reg, g)     # Newton direction
        except np.linalg.LinAlgError:
            p = g                              # Fallback to GD
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LMO: per-coordinate LP solution for box constraints
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                      for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z  # Convex combination
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
