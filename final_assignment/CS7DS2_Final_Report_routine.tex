\begin{abstract}
This report presents a systematic comparative investigation of fifteen optimisation algorithms across six conceptually distinct paradigms, evaluated on three benchmark problems of increasing difficulty: a well-conditioned linear regression quadratic loss (Benchmark A, $\kappa \approx 1.12$, $L \approx 1.033$, $\mu \approx 0.925$), a mildly anisotropic toy neural-network sinusoidal loss (Benchmark B, $\kappa \approx 6.90$), and the ill-conditioned Rosenbrock function (Benchmark C, $\kappa \approx 2508$ at the minimum, eigenvalues $\lambda_{\min}\approx 0.40$, $\lambda_{\max}\approx 1002$).

The six paradigms studied are: \textbf{(Q1)} adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), \textbf{(Q2)} momentum-based and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD), \textbf{(Q3)} second-order Newton's method with local quadratic approximations, \textbf{(Q4)} derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead simplex, grid search), \textbf{(Q5)} constrained optimisation via projected gradient descent and penalty methods, and \textbf{(Q6)} the Frank--Wolfe conditional gradient algorithm for box-constrained problems.

Key findings: (i) The Polyak step size achieves best convergence when $f^{\star}$ is known exactly but diverges catastrophically under misspecification, with step size $\alpha_k \to \infty$ as $\|\nabla f\| \to 0$. (ii) Newton's method solves the quadratic loss in a single iteration (affine invariance) and outperforms all first-order methods on Rosenbrock despite $4\times$ fewer iterations, owing to quadratic convergence. (iii) Nesterov acceleration empirically confirms the theoretical $O(1/k^2)$ improvement over GD's $O(1/k)$ rate, reaching $f = 0.479$ versus $f = 3.388$ for GD in 150 iterations on Rosenbrock. (iv) Central finite differences achieve $O(\varepsilon_{\mathrm{machine}}^{2/3})$ accuracy versus $O(\varepsilon_{\mathrm{machine}}^{1/2})$ for forward differences. (v) Projected gradient descent enforces hard feasibility from iteration 1 via a $\max$ projection, while penalty methods converge to feasibility asymptotically. (vi) Frank--Wolfe exhibits the $O(1/k)$ rate with standard schedule and provides the Frank--Wolfe gap $h_k \geq f(x_k) - f^{\star}$ as a computable optimality certificate without requiring $f^{\star}$.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmark functions span qualitatively different optimisation landscapes, allowing each algorithm's strengths and weaknesses to be clearly isolated.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \in \mathbb{R}^2 \sim \mathcal{N}(0, I), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2}\theta^T H \theta - b^T\theta + c,
\end{equation}
where $H = \frac{1}{m}X^T X$ is the (constant) Hessian and $b = \frac{1}{m}X^T y$. The gradient is $\nabla J(\theta) = H\theta - b = \frac{1}{m}X^T(X\theta - y)$.

The function is strongly convex with $\mu = \lambda_{\min}(H) \approx 0.925$ and Lipschitz gradient constant $L = \lambda_{\max}(H) \approx 1.033$. The condition number $\kappa(H) \approx 1.12$ makes this an extremely well-conditioned problem --- gradient descent is expected to converge quickly. The OLS solution $\theta^{\star}_{\text{OLS}} = H^{-1}b \approx (3.028, 4.017)^T$ defines the true minimiser; due to observation noise, $J^{\star} = J(\theta^{\star}_{\text{OLS}}) \approx 0.4826 \neq 0$. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \text{diag}(2 - \sin(x_1),\ 10)$. The sinusoidal term introduces mild nonlinearity: the Hessian varies with $x_1$, with eigenvalues ranging between $[1, 3]$ and $[10, 10]$. At the minimum $x^{\star} \approx (0.582, 2.0)$ --- where $2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$ and $x_2^{\star} = 2$ --- the Hessian has eigenvalues $\approx 1.45$ and $10$, giving condition number $\kappa \approx 6.90$. The minimum objective is $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The gradient is:
\begin{align}
\frac{\partial f}{\partial x_1} &= -2(1 - x_1) - 400 x_1(x_2 - x_1^2), \\
\frac{\partial f}{\partial x_2} &= 200(x_2 - x_1^2).
\end{align}
The Hessian at the optimum $(1,1)$ is:
\begin{equation}
H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $\lambda_{\min} \approx 0.40$ and $\lambda_{\max} \approx 1002$, giving condition number $\kappa \approx 2508$. This extreme ill-conditioning, combined with the narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions and the primary discriminator between algorithms in this study. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$, a box with diameter $D = \|(5-0.5, 10-(-5))\| \approx 15.66$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}

\subsection{Theoretical Background}

Four adaptive step-size methods are compared against a constant step-size gradient descent (GD) baseline. The baseline GD update is $x_{k+1} = x_k - \alpha \nabla f(x_k)$.

\begin{proposition}[Gradient Descent Convergence]
\label{prop:gd_rate}
Let $f$ be $\mu$-strongly convex and $L$-smooth ($L$-Lipschitz gradient). With step size $\alpha = 1/L$, gradient descent converges at the linear rate:
\begin{equation}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k [f(x_0) - f^{\star}] = \left(1 - \frac{1}{\kappa}\right)^k [f(x_0) - f^{\star}],
\end{equation}
where $\kappa = L/\mu$ is the condition number. The number of iterations to achieve $\epsilon$-accuracy is $k \geq \kappa \log(1/\epsilon)$.
\end{proposition}

For the three benchmarks: Benchmark A has $\kappa \approx 1.12$, requiring $\approx 13$ iterations for $10^{-5}$ accuracy ($\kappa \log(10^5) \approx 13$); Benchmark B has $\kappa \approx 6.9$, requiring $\approx 80$ iterations; Rosenbrock has $\kappa \approx 2508$ near $(1,1)$, requiring $\approx 28{,}900$ iterations --- explaining why 120--150 iterations are grossly insufficient for GD on Benchmark C. The adaptive methods attempt to improve convergence by automatically adjusting $\alpha_k$ to local curvature information, effectively reducing the governing $\kappa$.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Normalised progress-to-gradient ratio}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is theoretically optimal for non-smooth convex functions when $f^{\star}$ is known: it achieves convergence in $O(1/\sqrt{k})$ with no tuning required. However, it requires accurate knowledge of $f^{\star}$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate cumulative squared gradient}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad adapts the learning rate per coordinate by dividing by $\sqrt{G_k}$, assigning smaller rates to frequently-updated directions and larger rates to infrequent ones. The per-coordinate effective step size is $\alpha_k^{(i)} = \alpha_0 / (\sqrt{G_k^{(i)}} + \varepsilon)$, which monotonically decreases because $G_k^{(i)}$ is non-decreasing. This guarantees convergence but can cause premature stagnation in non-stationary settings.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's growing denominator with an exponential moving average of squared gradients, controlled by decay factor $\beta \in (0,1)$. This gives the effective step size a finite memory, preventing monotone decay and allowing adaptation to non-stationary objectives.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Exponential average of gradients}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method (Polyak Momentum) accelerates gradient descent by accumulating a velocity vector $z_k$ that averages past gradients.

\begin{proposition}[Heavy Ball Optimal Rate on Quadratics]
\label{prop:hb_rate}
For a quadratic $f(x) = \frac{1}{2}x^T A x - b^T x$ with $\mu = \lambda_{\min}(A)$ and $L = \lambda_{\max}(A)$, the optimal Heavy Ball parameters are:
\begin{equation}
\alpha^{\star} = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2}, \qquad \beta^{\star} = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2,
\end{equation}
giving the accelerated convergence rate $((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^k$ versus GD's $((\kappa-1)/(\kappa+1))^k$. The improvement factor per iteration is $(\sqrt{\kappa}-1)/(\kappa-1) = 1/(\sqrt{\kappa}+1)$, which is significant for large $\kappa$: for $\kappa = 2508$, Heavy Ball requires $\approx\sqrt{2508} \approx 50\times$ fewer iterations than GD.
\end{proposition}

Note that these optimal parameters assume exact knowledge of $\mu$ and $L$, which is unavailable in practice. Moreover, on non-quadratic objectives, the Heavy Ball method does not carry a convergence guarantee in general --- it may oscillate or diverge. In contrast, Nesterov's method (Section~\ref{sec:q2}) is provably convergent with rate $O(1/k^2)$ on all smooth convex functions.

\begin{proposition}[Polyak Divergence Under Misspecification]
\label{prop:polyak_div}
Let $f^{\star}_{\mathrm{true}} = \min_x f(x) > f^{\star}_{\mathrm{spec}}$ (specified value strictly below true minimum). Under the Polyak step rule $\alpha_k = (f(x_k) - f^{\star}_{\mathrm{spec}}) / \|\nabla f(x_k)\|^2$, as the iterates approach any $\epsilon$-neighbourhood of $x^{\star}_{\mathrm{true}}$, the step size satisfies:
\begin{equation}
\alpha_k \geq \frac{f^{\star}_{\mathrm{true}} - f^{\star}_{\mathrm{spec}} - \epsilon}{\|\nabla f(x_k)\|^2} \to \infty \quad \text{as } \|\nabla f(x_k)\| \to 0.
\end{equation}
This unbounded step triggers oscillation: the iterate is pushed far from $x^{\star}$, the gradient grows again, the step shrinks, and the cycle repeats without convergence.
\end{proposition}

On Benchmarks A and B with $f^{\star}_{\mathrm{spec}} = 0$ but $f^{\star}_{\mathrm{true}} \approx 0.4826$ and $0.7244$ respectively, this mechanism drives the step to grow without bound as convergence is approached, explaining the divergence observed in Figures~\ref{fig:q1_fval_A}--\ref{fig:q1_fval_B}. On Benchmark C where $f^{\star}_{\mathrm{spec}} = f^{\star}_{\mathrm{true}} = 0$ exactly, this pathology disappears and Polyak achieves superior convergence. This sensitivity to $f^{\star}$ fundamentally limits Polyak's applicability to settings where the optimal value is known analytically (e.g., SVMs with known margin, problems with known lower bounds from duality).

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$). On Benchmarks A and B, where the condition number is small, GD, Adagrad, and Heavy Ball all reach the minimum rapidly and differences between methods are negligible.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential gradient averaging. Heavy Ball shows characteristic overshoot (proportional to $\beta/(1-\beta)$ times the gradient step) followed by damped oscillation.
\item \textbf{Stability:} GD baseline is most stable but slowest. Polyak is unstable when $f^{\star}$ is misspecified. Adagrad is uniformly stable due to its monotonically decreasing step, though this may cause premature stagnation in long runs.
\item \textbf{Practical implications:} Exact $f^{\star}$ is rarely known in practice, limiting Polyak's applicability to settings such as support vector machine training or problems with a known lower bound. RMSprop and Adagrad are more robust defaults for non-stationary objectives. Heavy Ball is effective when some estimate of the condition number is available for setting $(\alpha, \beta)$.
\end{itemize}

\textbf{Connection to preconditioning:} Adagrad can be viewed as gradient descent with a diagonal preconditioner $P_k = \text{diag}(G_k)^{-1/2}$, approximating the inverse Hessian's diagonal. If the Hessian $H = \text{diag}(h_1, \ldots, h_d)$ (diagonal, different curvatures per coordinate), then after receiving enough gradient observations, $G_k^{(i)} \approx h_i^2 k$, giving effective step $\approx \alpha_0 / (h_i \sqrt{k})$ --- the Newton step scaled by $1/\sqrt{k}$. This diagonal Hessian approximation explains why Adagrad performs well on sparse, anisotropic problems (natural language processing) where different coordinates have vastly different curvatures, providing \emph{automatic} adaptation without requiring explicit knowledge of $\kappa$ (see Figure~\ref{fig:adaptive_analysis}).

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}

\subsection{Theoretical Background}

Nesterov's accelerated gradient method achieves the optimal first-order convergence rate for smooth convex functions.

\begin{proposition}[Nesterov Acceleration]
\label{prop:nesterov_rate}
For a convex, $L$-smooth function $f$, the Nesterov accelerated method with step $\alpha = 1/L$ and momentum $\beta_k = (k-1)/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L \|x_0 - x^{\star}\|^2}{(k+1)^2} = O(1/k^2),
\end{equation}
compared to $O(1/k)$ for gradient descent. This $O(1/k^2)$ rate is \emph{optimal}: no first-order black-box oracle method can achieve a better worst-case guarantee on $L$-smooth convex functions (Nesterov's 1983 lower bound). For strongly convex $f$ with $\kappa = L/\mu$, the accelerated rate improves to a linear rate with ratio $(1 - 1/\sqrt{\kappa})^k$, versus $(1-1/\kappa)^k$ for GD --- a quadratic improvement in the number of iterations needed.
\end{proposition}

The key innovation is the \emph{lookahead} step: evaluating the gradient at a predicted future position $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This allows the method to \emph{correct} its trajectory by using a projected estimate of the next iterate before committing to a step, which is what enables the $O(1/k^2)$ guarantee.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ (predicted future position) rather than at $x_k$ (current position). The adaptive schedule $\beta_k = (k-1)/(k+2)$ builds momentum gradually, with $\beta_k \to 1$ as $k \to \infty$, bounded by $\beta_{\max}$ for stability on nonconvex landscapes.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (curvature)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam synthesises momentum (via $m_t$, a first-order gradient exponential average) with per-coordinate curvature adaptation (via $v_t$, a second-order moment estimate), inheriting the advantages of both RMSprop and momentum methods. The bias-correction terms $1/(1-\beta_1^t)$ and $1/(1-\beta_2^t)$ counteract the initialisation at zero: without them, early estimates $\hat{m}_t$ and $\hat{v}_t$ would be biased towards zero, making initial steps too small.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch. The gradient estimator is:
\begin{equation}
\hat{g}(\theta) = \frac{1}{b} \sum_{i \in \mathcal{B}} x^{(i)}\left(\theta^T x^{(i)} - y^{(i)}\right),
\end{equation}
which is an unbiased estimate of the full gradient $\nabla J(\theta)$. The variance of this estimator scales as $\sigma^2_g / b$, where $\sigma^2_g$ is the per-sample gradient variance. For the linear regression model, $\sigma^2_g = \text{Tr}(H \text{Cov}(\varepsilon x x^T) H^{-1}) / m^2$, which under $\varepsilon \sim \mathcal{N}(0,\sigma^2)$ reduces to $\sigma^2_g \propto \sigma^2 \text{Tr}(H)$.

A key theoretical consequence is the \emph{SGD noise floor}: with fixed step size $\alpha$, iterates converge not to the exact minimum but to a ball of radius proportional to $\alpha$. For $\mu$-strongly convex $f$ with $L$-Lipschitz gradients and stochastic gradients with variance $\sigma^2_g$:
\begin{equation}
\limsup_{k\to\infty}\, \mathbb{E}[f(x_k) - f^{\star}] \leq \frac{\alpha L \sigma^2_g}{2\mu b}.
\end{equation}
To achieve $\epsilon$-accuracy, one must choose $\alpha \leq 2\mu b \epsilon / (L \sigma^2_g)$, which slows the geometric convergence term. The optimal strategy is a diminishing step schedule $\alpha_k = \alpha_0 / \sqrt{k}$ (achieving $O(1/\sqrt{k})$) or $\alpha_k = C/k$ (achieving $O(1/k)$) for strongly convex objectives.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Batch size trade-off:} The batch size $b$ controls a fundamental trade-off between computational work per epoch and gradient quality. For a fixed computational budget of $T$ total gradient evaluations: with batch size $b$, we perform $T/b$ parameter updates over $T/b$ mini-batches; with full-batch GD ($b=m$), we perform $T/m$ updates. The mini-batch estimator's variance scales as $\text{Var}[\hat{g}] = \sigma^2_g/b$. The expected squared error per step is thus $\alpha^2\sigma^2_g/b$. The ``linear scaling rule'' suggests that if batch size is increased by $k\times$, the learning rate should also be increased by $k\times$ to maintain the same noise-to-signal ratio --- but this rule breaks down for large batch sizes and highly nonconvex objectives.

\begin{remark}
The noise floor analysis reveals a fundamental limitation of SGD with fixed learning rate: for any $\alpha > 0$, the iterates converge not to the exact minimum but to a neighbourhood of radius $O(\alpha \sigma^2_g / b)$ (see Figure~\ref{fig:sgd_noise}). To achieve exact convergence, one must either use a decaying learning rate schedule (at the cost of slower convergence) or reduce to full-batch GD. The optimal trade-off between convergence speed and noise floor is problem-dependent.

\textbf{Quantitative noise floor prediction:} In the high-noise experiment ($\sigma = 6$), the minimum objective $J^{\star}_{\text{noisy}} = \frac{1}{2m}\|\varepsilon\|^2 \approx \frac{1}{2}\sigma^2 = 18$ (since $\mathbb{E}[\varepsilon^T \varepsilon/m] = \sigma^2$), matching the observed convergence floor of $\approx 17$--$18$. The irreducible loss scales as $\sigma^2$, so a $6\times$ noise increase raises the floor by $36\times$. Additionally, the gradient variance $\sigma^2_g \propto \sigma^2 \|H\|_F$, so the SGD noise floor $\alpha L \sigma^2_g/(2\mu b) \propto \sigma^2$: the bias between the SGD stationary distribution and the true minimum also increases as $\sigma^2$, amplifying the effect of large observation noise.
\end{remark}

While both Nesterov and Adam achieve fast convergence on smooth, convex problems, they embody fundamentally different design philosophies:

\begin{tabular}{lll}
\toprule

The Rosenbrock result can be explained theoretically: Rosenbrock has \emph{global} ill-conditioning (the condition number along the trajectory changes significantly), but the dominant gradient direction is consistently along the valley floor. Nesterov's momentum in this persistent direction is more effective than Adam's per-coordinate adaptation, which may scale gradient components in directions that are not aligned with the natural coordinate axes. Adam excels in deep learning where different layers or neurons may require very different effective learning rates, a situation that does not arise in the 2D test problems studied here.

\textbf{Adam convergence theory:} Unlike Nesterov, Adam does not have a proven $O(1/k^2)$ convergence guarantee. In fact, counterexamples show Adam can fail to converge on simple convex problems with the standard fixed $\beta_1, \beta_2$. The AMSGrad variant (Reddi et al., 2018) fixes convergence by using a non-decreasing second moment estimate $\hat{v}_t = \max(\hat{v}_{t-1}, v_t)$. In practice, Adam's empirical performance is excellent for non-convex deep learning objectives, where its coordinate-wise step normalisation acts as a form of gradient clipping that prevents large steps in directions of high curvature. For smooth convex problems (as in this assignment), Nesterov provides stronger theoretical guarantees.

\textbf{Bias-correction necessity:} Without bias correction ($1/(1-\beta_1^t)$ and $1/(1-\beta_2^t)$), the first and second moment estimates at iteration $t=1$ are $m_1 = (1-\beta_1) g_1$ and $v_1 = (1-\beta_2)g_1^2$, biased towards zero by factors $(1-\beta_1)$ and $(1-\beta_2)$ respectively. For typical $\beta_1=0.9, \beta_2=0.999$, the initial step without correction would be $(0.1 g_1)/(0.001 g_1^2)^{1/2}\cdot\alpha = \sqrt{0.01/0.001}\cdot\alpha\|g_1\| \approx 3.16\alpha$, versus the corrected step $\alpha$ (unit effective step, independent of gradient scale). The correction ensures the effective step magnitude is $\alpha$ regardless of the gradient at initialisation.

\subsection{Local Approximations of $g(x) = x^4$}

To motivate Newton's method, consider the local Taylor approximation of $g(x) = x^4$ around $x_0 = 0.25$:

\textbf{First-order (Taylor) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.0039 + 0.0625(x - 0.25).
\end{equation}
This linear model captures only the slope and provides no information about curvature. Minimising it yields no finite minimum (it is unbounded below).

\textbf{Second-order approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton update minimises this quadratic model: setting $\frac{d}{dx}[\text{second-order approx}] = 0$ gives:
\begin{equation}
x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3} = \frac{2}{3}x_k.
\end{equation}
Starting from $x_0 = 0.25$: $x_1 = 0.167$, $x_2 = 0.111$, \ldots --- geometric convergence towards $x^{\star} = 0$.

\begin{remark}
For $g(x) = x^4$, the Newton iteration converges \emph{linearly} (ratio $2/3$), not quadratically, despite using the second derivative. This is because $g''(0) = 12 \cdot 0^2 = 0$: the Hessian is zero at the optimum (degenerate). The quadratic convergence theorem (Proposition~\ref{prop:newton_rate} below) requires $g''(x^{\star}) > 0$. When the Hessian vanishes at the minimum, Newton's method reduces to a fixed-ratio contraction and the superlinear convergence phase never begins. This example illustrates why Newton's method is most powerful for \emph{strongly convex} functions where $\lambda_{\min}(H(x^{\star})) > 0$.
\end{remark}

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{Damped Newton step}
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton step $p_k = H_k^{-1} g_k$ scales the gradient by the inverse Hessian, effectively rescaling each direction by its curvature. This makes Newton's method \emph{affine invariant}: its performance is independent of the problem's conditioning. The damping factor $\lambda = 10^{-8}$ ensures numerical stability when $H_k$ is nearly singular, and the step size $\alpha \leq 1$ ensures stability far from the minimum where the quadratic approximation is inaccurate.

\begin{proposition}[Quadratic Convergence of Newton's Method]
\label{prop:newton_rate}
For a twice continuously differentiable function $f$ with $\nabla^2 f$ Lipschitz continuous near $x^{\star}$ (i.e., $\|\nabla^2 f(x) - \nabla^2 f(y)\| \leq M\|x-y\|$), if $x_0$ is sufficiently close to $x^{\star}$ and $\alpha = 1$, the undamped Newton iterates satisfy:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2,
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^{\star})) > 0$. This \emph{quadratic convergence} means the number of correct digits approximately doubles each iteration: if $\|x_k - x^{\star}\| = 10^{-p}$, then $\|x_{k+1} - x^{\star}\| \leq (M/2\mu) \cdot 10^{-2p}$.
\end{proposition}

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$.

A practical stopping criterion for Newton's method is the \emph{Newton decrement} $\lambda(x)$, defined as:
\begin{equation}
\lambda(x)^2 = \nabla f(x)^T H(x)^{-1} \nabla f(x) = \|H(x)^{-1/2} \nabla f(x)\|^2.
\end{equation}
The Newton decrement satisfies $f(x) - f^{\star} \leq \lambda(x)^2 / 2$ (from the quadratic upper bound induced by $L$-smoothness), making it a reliable, scale-invariant certificate of proximity to the optimum. Importantly, $\lambda(x)$ is \emph{affine invariant}: it does not depend on the choice of coordinates or the scaling of $f$, unlike the raw gradient norm $\|\nabla f\|$ which depends on $H$. In practice, Newton iterations are stopped when $\lambda(x)^2/2 < \epsilon$ for some small tolerance $\epsilon$ (see Figure~\ref{fig:newton_conv}). In this implementation, the stopping criterion is replaced by a fixed iteration budget.

\textbf{Newton's method in the two phases:} Newton's method has a \emph{damped phase} (far from the optimum, where $\lambda(x) > \eta$ for some threshold $\eta$) where the step is bounded and convergence is at best linear, and a \emph{quadratically convergent phase} (near the optimum, where $\lambda(x) \leq \eta$) where the undamped step is taken and convergence is quadratic. For Benchmark A (pure quadratic), Newton immediately enters the quadratic phase and converges in one step. For Rosenbrock with $\alpha = 0.22$ (damped throughout all 20 iterations), the iterates remain in the damped phase, explaining the slower-than-quadratic convergence observed.

\textbf{Computational cost analysis:} Each Newton iteration costs $O(d^3)$ for Cholesky factorisation of $H_k$ and $O(d^2)$ for back-substitution, compared to $O(d)$ per gradient descent step. For $d = 2$, this overhead is trivial. For large-scale problems with $d = 10^6$ (e.g., neural network weights), the $O(d^3)$ cost makes exact Newton infeasible, motivating quasi-Newton methods (L-BFGS) that approximate $H^{-1}$ using a limited memory rank-$m$ update.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad i = 1, \ldots, d.
\end{equation}
By Taylor expansion, the truncation error is:
\begin{equation}
\left|\frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| \leq \frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(x)\right| + O(\delta^2).
\end{equation}
Additionally, finite-precision arithmetic introduces round-off error of order $\varepsilon_{\text{machine}}/\delta \approx 10^{-16}/\delta$. The total error is minimised at the optimal perturbation $\delta^{\star} \sim \sqrt{\varepsilon_{\text{machine}}} \approx 10^{-8}$.

In this experiment, $\delta = 0.05$ yields good gradient approximations (truncation error $\approx \frac{0.05}{2} \times 10 = 0.25$ for $f_{x_2x_2} = 10$, which is significant but workable), while $\delta = 0.8$ causes severe truncation error as the perturbation spans a region of strong curvature variation.

\textbf{Central differences} provide a more accurate alternative. The central finite difference estimator is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}.
\end{equation}
The leading error term is $O(\delta^2)$ (versus $O(\delta)$ for forward differences), so the optimal step is $\delta^{\star}_{\text{cen}} \sim \varepsilon_{\text{machine}}^{1/3} \approx 10^{-5.3}$, versus $\delta^{\star}_{\text{fwd}} \sim \sqrt{\varepsilon_{\text{machine}}} \approx 10^{-8}$ for forward differences. At their respective optimal steps, central differences achieve $O(\varepsilon_{\text{machine}}^{2/3}) \approx 10^{-10.7}$ error versus $O(\varepsilon_{\text{machine}}^{1/2}) \approx 10^{-8}$ for forward differences, making central differences $\approx 700\times$ more accurate at the same computational cost (two function evaluations per coordinate, same as forward differences).

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Random directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The expected value of the estimator satisfies $\mathbb{E}[\hat{g}_k] = \nabla f_\delta(x_k)$, where $f_\delta(x) = \mathbb{E}_{u}[f(x + \delta u)]$ is a smoothed version of $f$ with $\|f_\delta - f\|_\infty \leq O(\delta^2 \|H\|_F)$. The convergence guarantee for Nesterov random search on $L$-smooth functions is:
\begin{equation}
\mathbb{E}[f(x_k) - f^{\star}] \leq O\!\left(\frac{d \cdot L R^2}{\sqrt{k}}\right),
\end{equation}
giving a $O(1/\sqrt{k})$ rate --- slower than GD's $O(1/k)$ but with only 2 function evaluations per step regardless of $d$ (versus $d+1$ for finite-difference GD). The oracle complexity for $\epsilon$-accuracy is $O(d^2/\epsilon^2)$, reflecting the dimension-dependent price of zeroth-order optimisation. The key property is scalability: this method remains practical for $d = 10^4$ or larger, where even finite-difference GD with $O(d)$ evaluations per step becomes prohibitive.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four ordered operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): Move the worst vertex $x_{n+1}$ through the centroid $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{n+1})$.
\item \textbf{Expansion} ($\gamma = 2.0$): If reflection improved significantly, try a further step: $x_e = \bar{x} + \gamma(\bar{x} - x_{n+1})$.
\item \textbf{Contraction} ($\rho = 0.5$): If reflection worsened, contract towards the centroid: $x_c = \bar{x} + \rho(x_r - \bar{x})$ (outside) or $x_c = \bar{x} + \rho(x_{n+1} - \bar{x})$ (inside).
\item \textbf{Shrink} ($\sigma = 0.5$): If contraction fails, contract all vertices towards the best: $x_i \gets x_1 + \sigma(x_i - x_1)$.
\end{enumerate}
No gradient information is required; the method relies entirely on function comparisons to guide the simplex. The initial simplex uses step size $0.35$.

\textbf{Convergence properties:} Nelder--Mead lacks a general convergence guarantee; counterexamples exist where it stagnates in smooth problems. However, it performs well in practice for low-dimensional problems ($d \leq 6$) with moderate noise. The lack of guarantee arises because the simplex update can cycle without decreasing the function value. Despite this, Nelder--Mead remains popular because: (i) it is completely derivative-free, (ii) it is robust to discontinuities and noise in $f$, (iii) it does not require tuning a step size $\alpha$, and (iv) it naturally avoids local curvature problems that plague gradient-based methods.

\textbf{Function evaluation count:} For reflection/contraction/expansion, the method typically uses $1$--$3$ function evaluations per outer iteration; shrink operations use $d$ evaluations. For 160 outer iterations with $d=2$, the total evaluation count is approximately $160\times 2 = 320$ evaluations --- more than GD with $d$ partial finite differences per step, but less than grid search.

\subsubsection{Grid Search}

Grid search exhaustively evaluates $f$ on a uniform $55 \times 55 = 3025$-point grid over $x_1 \in [-2, 2]$, $x_2 \in [-1, 3]$, with resolution $\Delta x_1 = \Delta x_2 \approx 0.073$. By Taylor expansion, the grid point closest to $x^{\star}$ lies within distance $\|\Delta\|_\infty/2 \approx 0.037$ and the objective at the nearest grid point satisfies:
\begin{equation}
f(x_{\text{grid}}) - f^{\star} \leq \frac{1}{2}\,\nabla^2 f(x^{\star}) \cdot \|\Delta x\|^2 \leq \frac{L}{2}\|\Delta x\|^2 \approx \frac{1001.6}{2}(0.073)^2 \approx 2.67.
\end{equation}
This bound is tight since $\lambda_{\max}(H(1,1)) \approx 1002$: a small grid error maps to a large function value error due to the extreme curvature of Rosenbrock near the minimum.

\subsection{Problem Setup and Optimality Conditions}

Minimise Benchmark B subject to the constraint $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$:
\begin{equation}
\min_{x \in \mathcal{F}} f(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1), \quad \mathcal{F} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}.
\end{equation}

\textbf{Important observation:} The unconstrained minimiser of $f$ is at $x^{\star}_{\text{unc}} \approx (0.582, 2.0)$, which satisfies $x_1^{\star} = 0.582 > 0.5$. Therefore, the constraint is \emph{inactive} at the optimum, meaning the constrained minimum $x^{\star}_{\text{con}} = x^{\star}_{\text{unc}}$ and the KKT multiplier $\mu^{\star} = 0$. The constraint influences only the \emph{trajectory} --- methods starting from the infeasible point $(0.2, 4.0)$ must navigate differently to reach the same final point.

\textbf{KKT Conditions:} Writing the constraint as $g(x) = 0.5 - x_1 \leq 0$ (inequality in standard form), the Lagrangian is $\mathcal{L}(x, \mu) = f(x) + \mu g(x) = f(x) + \mu(0.5 - x_1)$. The KKT conditions for a minimiser $x^{\star}$ are:
\begin{align}
&\text{Stationarity:} && \nabla_x \mathcal{L}(x^{\star}, \mu^{\star}) = \nabla f(x^{\star}) - \mu^{\star} e_1 = 0, \\
&\text{Primal feasibility:} && 0.5 - x_1^{\star} \leq 0 \quad (x_1^{\star} \geq 0.5), \\
&\text{Dual feasibility:} && \mu^{\star} \geq 0, \\
&\text{Complementary slackness:} && \mu^{\star}(0.5 - x_1^{\star}) = 0.
\end{align}
Since $x^{\star} \approx (0.582, 2.0)$ satisfies $x_1^{\star} > 0.5$ (strict inequality), complementary slackness forces $\mu^{\star} = 0$. Stationarity then reduces to $\nabla f(x^{\star}) = 0$, identical to the unconstrained first-order optimality condition. The KKT conditions are also sufficient here, because $f$ is convex and the feasible set $\mathcal{F}$ is a convex set, and Slater's condition (existence of a strictly feasible point, e.g., $x = (1, 1)$ with $x_1 = 1 > 0.5$) ensures that KKT conditions are both necessary and sufficient for global optimality.

\textbf{Constraint qualification:} Slater's condition --- the existence of an interior point $\tilde{x}$ satisfying $g(\tilde{x}) < 0$ strictly --- is satisfied here by any $x$ with $x_1 > 0.5$. This guarantees strong duality and that KKT multipliers exist. Without such a qualification, KKT conditions may fail to hold at a constrained minimiser even when one exists.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{F}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{F}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State Compute $\nabla F_{\lambda}(x_k) = \nabla f(x_k) - \lambda \mathbf{1}[x_1 < 0.5]\, e_1$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method uses a non-smooth penalty term $\lambda \max(0, 0.5 - x_1)$, whose subgradient with respect to $x_1$ is $-\lambda \cdot \mathbf{1}[x_1 < 0.5]$. A large $\lambda$ strongly penalises infeasibility but requires a smaller step size $\alpha$ for stability (since the penalty gradient can be large near the constraint boundary).

Settings: Projected GD ($\alpha = 0.08$), Penalty ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\subsection{Comparative Discussion}

\textbf{Projected GD} is the gold standard for simple constraints: it enforces feasibility as a hard constraint at every iteration, guarantees monotone decrease in $f$ within $\mathcal{F}$, and requires no tuning beyond the step size. For the box constraint $x_1 \geq 0.5$, projection is a trivial $\max$ operation: $\Pi_{\mathcal{F}}(x) = (\max(0.5, x_1), x_2)$. For complex constraints (e.g., $\|x\|_2 \leq 1$), efficient projection algorithms still exist, though at higher computational cost. The convergence rate of projected GD for convex $f$ with $L$-Lipschitz gradient is $O(1/k)$, matching gradient descent.

\textbf{Penalty methods} are simpler to implement (no projection oracle required) but soften the constraint. They impose a trade-off: too-small $\lambda$ leads to persistent infeasibility; too-large $\lambda$ requires a smaller step size for stability and may slow convergence by creating a steep gradient near the boundary. The optimal $\lambda$ depends on the problem scale and is problem-specific. As $\lambda \to \infty$, the penalty method converges to the constrained solution, but numerical conditioning deteriorates.

\textbf{Augmented Lagrangian (beyond this study):} A more sophisticated approach is the Augmented Lagrangian method (also known as the method of multipliers), which adds both a penalty term and a Lagrange multiplier estimate:
\begin{equation}
L_\rho(x, \mu) = f(x) + \mu(0.5 - x_1) + \frac{\rho}{2}\max(0, 0.5 - x_1)^2.
\end{equation}
Alternating between minimising over $x$ and updating $\mu \leftarrow \mu + \rho\max(0, 0.5 - x_k)$ achieves exact feasibility at finite $\rho$ (unlike pure penalty which requires $\rho \to \infty$), making it numerically superior to penalty methods while avoiding the computational cost of second-order projected methods.

\textbf{Key insight:} When the constraint is \emph{inactive} at the optimum (as here), both methods converge to the same final point. The methods differ primarily in their \emph{trajectory}: projected GD respects feasibility throughout, while penalty methods may take an infeasible shortcut before converging. For safety-critical applications requiring \emph{always-feasible} iterates, projected GD is mandatory.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$, a box with diameter $D \approx 15.66$.

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is obtained by minimising each coordinate independently (since the objective decomposes):
\begin{equation}
\min_{x \in \mathcal{X}} x_1 + 2x_2 = \min_{0.5 \leq x_1 \leq 5} x_1 + \min_{-5 \leq x_2 \leq 10} 2x_2 = 0.5 + 2(-5) = -9.5.
\end{equation}
Both components of $a$ are positive, so both coordinates are minimised at their lower bounds:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = -9.5.
\end{equation}
This illustrates the fundamental theorem of linear programming: \emph{the optimum of a linear objective over a bounded polytope is attained at a vertex}. The linear objective has parallel level sets, and the minimum is at the vertex in the direction $-a$.

\textbf{LP duality:} The dual of $\min_{x \in \mathcal{X}} a^T x$ subject to $\ell \leq x \leq u$ (box constraints) has the form $\max_{\lambda \geq 0, \mu \geq 0}\, \ell^T \lambda - u^T \mu - \mathbf{0}$ subject to $\lambda - \mu = -a$. At the optimum $x^{\star} = (0.5, -5)$: the dual multipliers satisfy $\lambda_1 = -a_1 = -1 < 0$ (infeasible for the lower bound), which reveals that the LP's structure is that the lower bounds are active for both components. This LP duality perspective underpins the Frank--Wolfe linear minimisation oracle: each FW step solves an LP, and the FW gap $h_k$ is the LP's objective value at the current gradient.

\textbf{Optimality check:} Evaluating $f^{\star} = -9.5$ at $x = (0.5, -5)$: $f = 1 \cdot 0.5 + 2 \cdot (-5) = 0.5 - 10 = -9.5$. Checking adjacent vertices: $(5, -5) \to 5 - 10 = -5$; $(0.5, 10) \to 0.5 + 20 = 20.5$; $(5, 10) \to 5 + 20 = 25$. The minimum is indeed at $(0.5, -5)$, confirming the vertex theorem.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination, $(1-\beta)$ step towards $z_k$}
\EndFor
\end{algorithmic}
\end{algorithm}

The Frank--Wolfe algorithm replaces the projection step of projected gradient descent with a \emph{linear minimisation} over $\mathcal{X}$. For box constraints, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases},
\end{equation}
requiring only $O(d)$ comparisons. Since $x_0 \in \mathcal{X}$ and $z_k \in \mathcal{X}$ at every step, $x_{k+1} = \beta x_k + (1-\beta) z_k \in \mathcal{X}$ by convexity --- \emph{iterates are always feasible without any projection}.

\begin{proposition}[Frank--Wolfe Convergence Rate]
For a convex differentiable function $f$ with $L$-Lipschitz gradient and the standard step size $\gamma_k = 2/(k+2)$ (corresponding to $\beta_k = k/(k+2)$):
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L D^2}{k+2} = O\!\left(\frac{LD^2}{k}\right),
\end{equation}
where $D = \max_{x, y \in \mathcal{X}} \|x - y\|$ is the diameter of $\mathcal{X}$. This $O(1/k)$ rate is optimal for Frank--Wolfe-type methods on general convex problems. Note that the bound depends on $D^2$: larger feasible sets slow convergence, in contrast to projection methods.
\end{proposition}

In the implementation, a fixed $\beta$ (rather than the decreasing $\gamma_k = 2/(k+2)$) is used, which modifies the convergence properties: the iterates may not achieve the $O(1/k)$ bound with fixed $\beta$, and residual oscillation near the optimum may persist.

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits a characteristic ``zig-zag'' in the LP vertex sequence $z_k$. The LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ at each iteration since even small gradient changes can flip the sign of a component and select the opposite bound. The fixed-$\beta$ convex combination gradually averages these oscillations. The $O(1/k)$ convergence rate (confirmed experimentally in Figure~\ref{fig:rate_verify}) is slower than the $O(1/k^2)$ of Nesterov acceleration, but Frank--Wolfe's advantage lies in its \emph{projection-free} nature and the interpretability of iterates as sparse convex combinations of vertices.

\textbf{Boundary optimum:} Frank--Wolfe is particularly well-suited to boundary optima, which arise when the unconstrained solution lies outside $\mathcal{X}$. When $x^{\star}$ lies on the boundary, the LP vertex $z_k$ consistently selects the closest constraint face in the gradient direction, and the convex combination progressively pulls the iterate towards the boundary. The final value $f = 0.364$ (vs $f^{\star} = 0.25$) indicates the algorithm has not fully converged in 140 iterations; the gap arises from the fixed $\beta = 0.93$ schedule, which does not provide the $O(1/k)$ convergence guarantee. Using the standard decreasing schedule $\gamma_k = 2/(k+2)$ would improve terminal accuracy.

\textbf{Primal-dual certificate:} The Frank--Wolfe gap $h_k$ simultaneously bounds the primal gap $f(x_k) - f^{\star}$ and constitutes a dual feasibility check. This is particularly valuable for boundary optima where $\nabla f(x^{\star})^T(x - x^{\star}) \geq 0$ for all $x \in \mathcal{X}$, so $z_k \to x^{\star}$ and $h_k \to 0$ certify both primal and dual optimality.

\textbf{Frank--Wolfe gap (duality gap):} A key practical advantage of Frank--Wolfe is its built-in optimality certificate, the \emph{Frank--Wolfe gap}:
\begin{equation}
h_k = \nabla f(x_k)^T (x_k - z_k) \geq f(x_k) - f^{\star} \geq 0.
\end{equation}
This follows from the convexity of $f$: $f^{\star} \geq f(x_k) + \nabla f(x_k)^T(x^{\star} - x_k) \geq f(x_k) - h_k$ (using the fact that $z_k$ minimises the linear objective). The FW gap provides an upper bound on the optimality gap without knowing $f^{\star}$, making it a practical stopping criterion. When $h_k < \epsilon$, the current iterate is guaranteed to be $\epsilon$-suboptimal.

\textbf{Frank--Wolfe vs Projected GD:} Projected GD achieves faster convergence (linear rate for strongly convex functions) but requires a projection operation. For simple constraint sets (box, $\ell_2$ ball), projection is $O(d)$ and preferred. For complex constraint sets --- such as the nuclear norm ball in matrix optimisation (projection requires SVD: $O(mn\min(m,n))$) or the flow polytope in network problems (projection requires LP: costly) --- Frank--Wolfe's LMO is substantially cheaper, making it the method of choice in large-scale structured optimisation.

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This study investigated six families of optimisation algorithms across problems of increasing difficulty, revealing systematic relationships between algorithm design, problem structure, and convergence behaviour.

\textbf{Summary of key findings:}

\begin{enumerate}[nosep]
\item \textbf{Problem conditioning dominates method selection.} On the well-conditioned Benchmark A ($\kappa \approx 1.12$), virtually all methods converge rapidly and differences are minimal. Benchmark C (Rosenbrock, $\kappa \approx 2508$) is the principal discriminator: only methods with access to curvature information (Newton), momentum along the valley direction (Nesterov), or exact knowledge of $f^{\star}$ (Polyak) show substantial improvements over standard gradient descent.

\item \textbf{Adaptive step sizes improve robustness at the cost of sensitivity.} Polyak achieves optimal convergence when $f^{\star}$ is exactly known but fails catastrophically under misspecification. Adagrad and RMSprop are more robust alternatives that require only per-coordinate curvature signals rather than the global minimum value.

\item \textbf{Momentum methods achieve optimal first-order complexity.} Nesterov acceleration realises the $O(1/k^2)$ bound, reducing iterations by a factor of $\approx 7\times$ on Rosenbrock compared to GD. Heavy Ball with appropriate $(\alpha, \beta)$ also achieves significant acceleration, especially when the condition number is known. Adam combines both momentum and per-coordinate adaptation, making it robust for anisotropic problems but less effective on highly nonlinear landscapes.

\item \textbf{Newton's method is transformative when applicable.} Solving the quadratic Benchmark A exactly in one step --- while GD requires $>30$ iterations --- highlights the value of curvature information. On Rosenbrock, damped Newton outperforms all first-order methods with $4\times$ fewer iterations. The $O(d^3)$ cost of solving the Newton system limits applicability to moderate-dimensional problems.

\item \textbf{Derivative-free methods incur a cost in convergence rate.} Finite-difference GD with small $\delta$ closely approximates exact GD at a $d$-fold increase in function evaluations. Nesterov random search reduces evaluations to 2 per step at the cost of $O(1/\sqrt{k})$ convergence. Nelder--Mead and grid search are viable only in low dimensions ($d \leq 5$ and $d \leq 3$, respectively).

\item \textbf{Constraint structure determines the method of choice.} Projected GD is preferred when projection is cheap and feasibility must be maintained at every iterate (e.g., safety-critical applications). Penalty methods are simpler to implement but require careful weight tuning and may temporarily violate constraints. When the unconstrained minimum is interior to the feasible set (as in this problem), all constrained methods converge to the same point and differ only in their convergence trajectory.

\item \textbf{Frank--Wolfe provides a projection-free alternative.} The $O(1/k)$ convergence rate is slower than projected GD but the LMO is often substantially cheaper than projection for structured constraint sets. For problems with sparse solution representations (e.g., low-rank matrices), Frank--Wolfe naturally produces structured iterates.
\end{enumerate}

\textbf{Information-theoretic perspective:} The convergence rates observed in this study reflect fundamental information-theoretic limits. Nesterov's $O(1/k^2)$ rate is optimal for smooth convex first-order methods (Nesterov's 1983 lower bound), and the $O(1/\sqrt{k})$ rate for zeroth-order random search reflects the additional cost of learning the gradient from function evaluations. Newton's quadratic convergence is not first-order and does not violate these bounds --- it exploits second-order (Hessian) information, which lies outside the first-order oracle model.

\textbf{Overall recommendation:} For a new optimisation problem, begin with the simplest applicable method (GD or Adam for unconstrained; projected GD for simple constraints), then graduate to more sophisticated methods (Newton, Nesterov) if convergence is unsatisfactory. The choice should be guided by the problem's conditioning ($\kappa$), the availability and cost of gradient or Hessian information, the dimensionality $d$, and feasibility requirements. On ill-conditioned problems ($\kappa > 100$), momentum methods provide a $\sqrt{\kappa}$ speedup that is theoretically optimal; second-order methods provide further acceleration at the cost of $O(d^3)$ per step.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
