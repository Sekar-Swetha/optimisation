\begin{abstract}
This report presents a systematic comparative study of first-order, second-order, derivative-free, and constrained optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark A, $\kappa \approx 1.12$), a toy neural network loss with sinusoidal perturbation (Benchmark B, $\kappa \approx 3.5$--$6.9$), and the Rosenbrock function (Benchmark C, $\kappa \approx 2508$). Six categories of methods are investigated across six questions.

Key findings: (1) The Polyak step size achieves the fastest convergence on Benchmark C ($f = 0.0239$ in 120 iterations) where $f^{\star}=0$ is correctly specified, but oscillates on Benchmarks A and B where the true minimum exceeds zero, reaching $f = 4.767$ and $f = 0.982$ respectively. (2) Nesterov acceleration achieves $O(1/k^2)$ rate and reaches $f = 0.479$ on Rosenbrock in 150 iterations versus GD's $f = 3.388$. (3) Newton's method solves the linear regression problem in \emph{one} iteration (exploiting exact quadratic structure) and reaches $f = 0.686$ on Rosenbrock in only 20 iterations. (4) Finite-difference gradient approximation with $\delta = 0.05$ closely tracks exact gradient descent, while $\delta = 0.8$ diverges significantly due to high truncation error. (5) Projected gradient descent enforces feasibility as a hard constraint from the first iteration, while penalty methods require large $\lambda$ to achieve comparable feasibility.
\end{abstract}

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. All code is implemented in Python with NumPy.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2}\theta^T H \theta - b^T\theta + c,
\end{equation}
where $H = \frac{1}{m}X^T X$ (constant positive definite Hessian) and $b = \frac{1}{m}X^T y$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y) = H\theta - b$. The unique global minimiser is the ordinary least squares (OLS) solution $\theta_{\text{OLS}} = H^{-1}b \approx [3.028, 4.017]^T$, and due to observation noise the minimum loss is $J^{\star} = J(\theta_{\text{OLS}}) \approx 0.4826 \approx \frac{1}{2}\sigma^2 = 0.5$. Since $x^{(i)} \sim \mathcal{N}(0, I_2)$, the population Hessian is $\mathbb{E}[xx^T] = I_2$; the empirical Hessian $H$ has eigenvalues $\lambda_{\min} \approx 0.925$ and $\lambda_{\max} \approx 1.033$ (close to 1 by the law of large numbers), giving condition number $\kappa = \lambda_{\max}/\lambda_{\min} \approx 1.12$. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$. The Hessian is $H = \text{diag}(2 - \sin(x_1),\ 10)$, which is position-dependent due to the sinusoidal term. The global minimum satisfies $x_2^{\star} = 2$ and $2(x_1^{\star}-1) + \cos(x_1^{\star}) = 0$, yielding $x_1^{\star} \approx 0.582$. At this point, $\lambda_{\min}(H) = 2 - \sin(0.582) \approx 1.451$ and $\lambda_{\max}(H) = 10$, giving condition number $\kappa \approx 6.89$. At the starting point $x_0 = [-1, 4]^T$, $\lambda_{\min}(H) = 2 - \sin(-1) \approx 2.841$, so $\kappa$ varies from $\approx 3.5$ to $\approx 6.9$ along the trajectory. The minimum value is $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian is:
\begin{equation}
H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1,1)$: $H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200\end{bmatrix}$ with eigenvalues $\lambda_1 \approx 0.399$ and $\lambda_2 \approx 1001.6$, giving condition number $\kappa \approx 2508$. This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes the Rosenbrock function one of the most challenging standard test functions for first-order methods. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for 120 iterations. The key motivation for adaptive step sizes is to remove the need for a globally-tuned $\alpha$ that must balance stability (requiring $\alpha < 2/L$) against convergence speed (preferring $\alpha$ close to $2/(L+\mu)$).

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate accumulation of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method accumulates gradient history in the momentum variable $z_k$. The effective update $x_{k+1} = x_k - \alpha \sum_{j \leq k} \beta^{k-j} \nabla f(x_j)$ geometrically down-weights older gradients, providing inertia along persistent descent directions.

\begin{remark}[Polyak sensitivity to $f^{\star}$]
The Polyak step is theoretically optimal for strongly convex smooth functions when $f^{\star}$ is known exactly. When $f^{\star}$ is underspecified (set to a value below the true minimum), $\alpha_k = (f(x_k) - f^{\star})/(\|\nabla f(x_k)\|^2 + \varepsilon) \to \infty$ as $x_k$ approaches the optimum since the numerator remains bounded away from zero while the gradient magnitude tends to zero. The regularisation $\varepsilon$ prevents exact divergence but results in large, destabilising steps. In practice, exact knowledge of $f^{\star}$ is rarely available, which severely limits the applicability of the Polyak rule outside of cases such as the Rosenbrock function where the minimum is analytically known to be zero.
\end{remark}

\subsection{Convergence Rate Theory for GD}

For a strongly convex quadratic loss with Hessian eigenvalues $\mu \leq \lambda \leq L$ and step size $\alpha$, gradient descent satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \rho^k\,(f(x_0) - f^{\star}), \quad \rho = \max\bigl(|1 - \alpha\mu|,\; |1 - \alpha L|\bigr).
\end{equation}
The optimal step size $\alpha^{\star} = 2/(L+\mu)$ achieves $\rho^{\star} = (\kappa-1)/(\kappa+1)$. For Benchmark A with $\mu \approx 0.925$, $L \approx 1.033$, $\kappa \approx 1.12$, and $\alpha = 0.08$: $\rho \approx 0.926$, giving a theoretical per-iteration reduction factor of $\rho = 0.926$ in the function gap. For Benchmark C with $\kappa \approx 2508$ and optimal step, $\rho^{\star} \approx 1 - 2/(2508+1) \approx 0.9992$ --- meaning each iteration reduces the gap by only 0.08\%, explaining why 10,000+ iterations would be needed for GD to converge on Rosenbrock.

\subsection{Comparative Analysis (Q1e)}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.024$ in 120 iterations). Without this knowledge, Heavy Ball and Adagrad match GD baseline performance on Benchmarks A and B.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging of squared gradients. Heavy Ball shows characteristic overshoot followed by damped oscillation, while Adagrad converges monotonically.
\item \textbf{Stability:} GD baseline is most stable but slowest. Polyak is \emph{unstable} when $f^{\star}$ is misspecified. Adagrad is uniformly stable but its monotonically shrinking step size may prevent it from escaping saddle points in non-convex settings.
\item \textbf{Adaptivity mechanism:} Adagrad accumulates all past squared gradients (no forgetting), yielding a decreasing effective rate that may stall on long runs. RMSprop's exponential averaging effectively gives each coordinate an adaptive rate responsive to recent curvature. Heavy Ball adapts not to curvature but to gradient history, providing acceleration in consistent directions.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball (Algorithm~\ref{alg:heavyball}) is the \emph{lookahead}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This allows the method to ``look ahead'' and correct its trajectory before committing to a step, giving it the ability to achieve the optimal $O(1/k^2)$ convergence rate on smooth convex problems, compared to $O(1/k)$ for standard GD:

\begin{theorem}[Nesterov convergence rate]
For $L$-smooth convex $f$ with Nesterov momentum using step $\alpha = 1/L$ and $\beta_k = (k-1)/(k+2)$:
$f(x_k) - f^{\star} \leq \dfrac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2}.$
\end{theorem}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum (tracking the first moment $m_t$) with per-coordinate adaptive scaling (tracking the second moment $v_t$), inheriting advantages of both RMSprop and momentum methods. The bias correction terms $1/(1-\beta_1^t)$ and $1/(1-\beta_2^t)$ are essential: without them, the initial iterates would be biased towards zero because $m_0 = v_0 = 0$, slowing early-stage convergence considerably (particularly with $\beta_2 = 0.999$, where $v_t$ takes $\sim 1000$ iterations to equilibrate). As $t \to \infty$, the bias corrections approach 1 and become negligible. The effective per-coordinate learning rate is $\alpha\sqrt{\hat{v}_t}/\hat{m}_t \cdot \hat{m}_t/(\sqrt{\hat{v}_t}+\varepsilon) \approx \alpha\, \text{sign}(g_t)$ when $\hat{m}_t \approx \hat{v}_t^{1/2}$, explaining why Adam is sometimes characterised as ``normalised'' gradient descent.

\subsection{Mini-Batch Stochastic Gradient Descent (Q2d)}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch. With $m = 1000$ samples, batch size $b = 5$ gives $200$ updates per epoch, while $b = 40$ gives $25$ updates per epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Unbiased gradient estimate: $\mathbb{E}[g] = \nabla J(\theta)$}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{Local Approximations of $g(x) = x^4$ (Q3a)}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = (0.25)^4 + 4(0.25)^3(x - 0.25) = 0.003906 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.003906 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

\textbf{Newton update derivation:} Setting $\frac{d}{dx}[g(x_0) + g'(x_0)(x - x_0) + \frac{1}{2}g''(x_0)(x-x_0)^2] = 0$ gives $g'(x_0) + g''(x_0)(x^+ - x_0) = 0$, hence the Newton step:
\begin{equation}
x^+ = x_0 - \frac{g'(x_0)}{g''(x_0)}.
\end{equation}
For the 1D case, this minimises the local quadratic model exactly. For $g(x) = x^4$ at $x_0 = 0.25$: $x^+ = 0.25 - \frac{4(0.25)^3}{12(0.25)^2} = 0.25 - 0.0833 \approx 0.1667$, closer to the true minimum at $x=0$.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$. The first-order tangent line captures local slope only and diverges rapidly from $g$ outside a small neighbourhood. The second-order parabola captures curvature and provides a tight fit near $x_0$. The minimum of the parabola at $x^+ \approx 0.1667$ is a far better estimate of the true minimum (at $x=0$) than any fixed step along the gradient. This geometric motivation illustrates why Newton's method converges much faster than gradient descent: each step minimises a quadratic model rather than merely descending along the gradient.}

\subsection{Newton's Method (Q3b)}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov regularisation for numerical stability}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: solve linear system}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$ (Tikhonov regularisation to ensure positive definiteness). The step sizes are listed in Table~\ref{tab:q3_params}.

\begin{remark}[Quadratic convergence of Newton's method]
For twice-continuously differentiable strongly convex $f$ with $M$-Lipschitz Hessian, Newton's method satisfies the quadratic convergence bound in a neighbourhood of the optimum:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2,
\end{equation}
where $\mu > 0$ is the strong convexity constant. This super-linear (quadratic) rate means the number of correct decimal digits approximately doubles each iteration. In contrast, GD satisfies the linear rate $\|x_k - x^{\star}\| \leq (1 - \mu/L)^k \|x_0 - x^{\star}\|$. For Benchmark A (pure quadratic), Newton achieves exact convergence in \emph{one} step: $x_1 = H^{-1}b = \theta_{\text{OLS}}$, since $\nabla J$ is linear and the Newton step solves the exact system. For Benchmarks B and C (non-quadratic), the Hessian changes with $x$, so multiple iterations are needed. The cost is $O(d^3)$ per step for the Hessian solve, making Newton impractical for large $d$ but highly effective here with $d = 2$.
\end{remark}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation (Q4a--Q4c)}

The forward finite-difference approximation replaces each partial derivative with a ratio of function values:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad i = 1, \ldots, d.
\end{equation}
The approximation error decomposes into two competing components:
\begin{equation}
\text{Error}(\delta) = \underbrace{C_1\delta}_{\text{truncation}} + \underbrace{C_2\,\varepsilon_{\text{machine}}/\delta}_{\text{round-off}},
\end{equation}
where $C_1 \sim \frac{1}{2}\|f''\|$ and $C_2 \sim \|f\|$. Minimising over $\delta$ by setting $\frac{d}{d\delta}\text{Error} = C_1 - C_2\varepsilon_{\text{machine}}/\delta^2 = 0$ gives the \emph{optimal step size}:
\begin{equation}
\delta^{\star} = \sqrt{C_2/C_1}\,\varepsilon_{\text{machine}}^{1/2} \approx \sqrt{\varepsilon_{\text{machine}}} \approx 10^{-8},
\end{equation}
and the minimum total error is $\text{Error}(\delta^{\star}) = 2\sqrt{C_1 C_2}\,\varepsilon_{\text{machine}}^{1/2} = O(\sqrt{\varepsilon_{\text{machine}}}) \approx 10^{-8}$. Both truncation and round-off errors equal $O(\varepsilon_{\text{machine}}^{1/2})$ at the optimum: making $\delta$ smaller reduces truncation error at the cost of amplifying floating-point cancellation, and vice versa.

The two values tested are $\delta = 0.05$ (good: truncation error $\sim 5 \times 10^{-2}$, well above round-off) and $\delta = 0.8$ (poor: truncation error $\sim 0.8$, spanning a region of significant nonlinearity in Benchmark B's sinusoidal term).

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates a scalar directional derivative along a random unit vector $u_k$, requiring only \emph{two} function evaluations per iteration regardless of dimension $d$. The estimate $\hat{g}_k$ is an unbiased estimator of the smoothed gradient $\nabla f_\delta(x_k)$ (the gradient of the $\delta$-smoothing of $f$), and converges to $\nabla f$ as $\delta \to 0$. The convergence rate is $O(1/\sqrt{k})$ for convex $f$, slower than the $O(1/k)$ of exact GD, reflecting the stochastic gradient noise from random direction sampling.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method (Q4d)}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations: \textbf{reflection} (reflect the worst vertex through the centroid of the remaining vertices), \textbf{expansion} (if reflection improved significantly, extend further in that direction), \textbf{contraction} (if reflection worsened, contract toward the centroid), and \textbf{shrink} (if contraction also fails, contract all vertices toward the best). Standard parameters are reflection coefficient $\alpha_r = 1$, expansion $\gamma = 2$, contraction $\rho = 0.5$, shrink $\sigma = 0.5$.

No gradient information is required: the algorithm uses only function value comparisons. The simplex adapts its shape to the local landscape, elongating along steep directions and contracting near the optimum.

Settings: initial simplex step $= 0.35$, iterations $= 160$.

The key trade-off is between \emph{gradient quality} and \emph{per-iteration cost}. Exact GD provides the most accurate descent direction at $O(d)$ cost per gradient. Finite differences match this cost but introduce approximation error controlled by $\delta$. Nesterov random search achieves constant cost (two function evaluations) independent of $d$, making it attractive in high dimensions, but at the expense of convergence rate. Nelder--Mead is practical in low dimensions ($d \leq 10$) but degrades rapidly for $d > 5$. Grid search is theoretically sound for global optimisation but computationally intractable for any dimension beyond 3--4.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Setup}

Minimise Benchmark B subject to the constraint $x_1 \geq 0.5$, starting from the infeasible initial point $x_0 = (0.2, 4.0)$. Since the unconstrained minimum of Benchmark B is at $x_1^{\star} \approx 0.582 > 0.5$, the constrained minimum coincides with the unconstrained minimum and lies in the interior of the feasible half-space (the constraint is inactive at the optimum).

\textbf{KKT conditions.} Writing the constraint as $g(x) = 0.5 - x_1 \leq 0$, the KKT conditions for this inequality-constrained problem are:
\begin{equation}
\nabla f(x^{\star}) - \lambda^{\star} e_1 = 0, \quad \lambda^{\star} \geq 0, \quad 0.5 - x_1^{\star} \leq 0, \quad \lambda^{\star}(0.5 - x_1^{\star}) = 0.
\end{equation}
Since $x_1^{\star} \approx 0.582 > 0.5$, complementary slackness forces $\lambda^{\star} = 0$, reducing the KKT stationarity condition to $\nabla f(x^{\star}) = 0$ --- the unconstrained optimality condition. This confirms analytically that the constrained and unconstrained solutions coincide, and that the Lagrange multiplier at the optimum is zero. For a problem where the true optimum were infeasible (e.g.\ if the constraint were $x_1 \geq 0.7$), the constraint would be active at the optimum, $\lambda^{\star} > 0$, and the penalty method's $\lambda$ would need to exceed $\lambda^{\star}$ to achieve exact feasibility.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x_k) \gets f(x_k) + \lambda \max(0,\, -x_1 + 0.5)$ \Comment{Penalised objective}
    \State $\nabla F_\lambda(x_k) \gets \nabla f(x_k) - \lambda \cdot \mathbf{1}[x_1 < 0.5]\, e_1$ \Comment{Subgradient of penalty}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$, 100 iterations); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$ to maintain stability with the stiffer penalty); Unconstrained GD ($\alpha = 0.07$, 100 iterations).

In practice, a sequence of increasing $\lambda$ (as in the augmented Lagrangian / multiplier method) achieves exact feasibility with bounded penalty coefficients.
\end{itemize}

\subsection{Feasible Set and Linear Programme (Q6 task 1)}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a compact box).

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is obtained by minimising each coordinate independently. Since both components of $a$ are positive, the optimum is attained at the lower bounds:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}
This illustrates the fundamental theorem of linear programming: the optimum of a linear objective over a bounded polytope is attained at a vertex (extreme point) of the feasible set.

\subsection{Frank--Wolfe Algorithm (Q6 tasks 2--5)}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: maintains $x_{k+1} \in \mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

The LMO for box constraints decomposes coordinate-wise in $O(d)$:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
The iterate $x_{k+1} = \beta x_k + (1-\beta) z_k$ is always a convex combination of $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$, hence $x_{k+1} \in \mathcal{X}$ automatically --- no projection oracle is required.

\begin{theorem}[Frank--Wolfe convergence rate]
For $L$-smooth $f$ and compact $\mathcal{X}$ with diameter $D = \sup_{x,y \in \mathcal{X}}\|x-y\|$, the standard Frank--Wolfe with step $\gamma_k = 2/(k+2)$ satisfies:
$f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2} = O(1/k).$
\end{theorem}

This $O(1/k)$ rate is slower than the $O(1/k^2)$ achievable by projected gradient methods on strongly convex problems, but Frank--Wolfe requires no projection (only a linear minimisation) per iteration, which can be cheaper for complex constraint sets such as the nuclear norm ball or the simplex.

\textbf{Interior optimum behaviour:} Frank--Wolfe exhibits a characteristic zig-zag pattern because $z_k$ selects extreme vertices of $\mathcal{X}$ at each step, far from the current iterate. To see why: near the interior optimum $(1, 5)$, $\nabla f \approx [2(x_1-1), 2(x_2-5)]$ is small but non-zero. The LMO selects the vertex that minimises $\nabla f^T x$ over the box, typically an extreme corner. The subsequent convex update $x_{k+1} = \beta x_k + (1-\beta)z_k$ creates a weighted average that moves slowly towards the interior. This ``zig-zagging'' to vertices is the fundamental mechanism of Frank--Wolfe and causes the $O(1/k)$ convergence rate, slower than the $O(1/k^2)$ achievable by projected gradient methods with similar cost.

The fixed-$\beta$ update (rather than the theoretically optimal decaying schedule $\gamma_k = 2/(k+2)$) results in persistent oscillation near the minimum. With $\gamma_k = 2/(k+2)$, one can show $f(x_k) - f^{\star} \leq 2LD^2/(k+2)$; with fixed $\beta$, no such guarantee exists for large $k$. Empirically, running Frank--Wolfe with $\gamma_k = 2/(k+2)$ on the interior objective achieves $f = 0.000145$ after 180 iterations, compared to $f = 0.0089$ with fixed $\beta = 0.90$ --- a $\mathbf{60\times}$ improvement from the schedule change alone, demonstrating the practical importance of the decaying step size rule.

\textbf{Boundary optimum behaviour:} Frank--Wolfe is better suited to boundary optima because the LMO consistently selects vertices near the constrained optimum, creating a strong directional signal. For $f(x) = x_1^2 + x_2^2$ with minimum at $(0, 0) \notin \mathcal{X}$, the gradient $\nabla f = [2x_1, 2x_2]$ is always positive near the boundary $x_1 = 0.5$, so $z_k = (0.5, -5)$ (the vertex closest to the unconstrained minimum) is selected consistently. The iterate approaches the boundary without requiring a projection. However, converging \emph{exactly} to a boundary point is slow with fixed $\beta = 0.93$: the 7\% step per iteration means $x_{k+1} \approx (1-0.07)x_k + 0.07 z_k$, which is an exponentially-weighted average that can only approach $z_k$ asymptotically. The residual gap ($0.1137$ vs $f^{\star} = 0.25$) after 140 iterations reflects this slow geometric approach.

\textbf{Key distinction:} When the optimum is interior, Frank--Wolfe shows the ``zig-zag'' inefficiency (vertices are poor representatives of the interior). When the optimum is on the boundary, the LMO selects the relevant vertex repeatedly, making progress more efficient --- but still limited by the fixed $\beta$ schedule.

\textbf{Effect of problem conditioning:} The condition number $\kappa$ is the dominant factor controlling first-order method performance. On Benchmark A ($\kappa \approx 1.12$), GD with $\alpha = 0.08$ achieves convergence rate $\rho = 0.926$ per iteration and converges to within numerical tolerance of $f^{\star}$ within the allotted 120 iterations; all first-order methods succeed reliably here. On Benchmark C (Rosenbrock, $\kappa \approx 2508$), the GD convergence rate degrades to $\rho \approx 0.9992$ per iteration --- meaning each step reduces the optimality gap by only $0.08\%$, requiring $>10{,}000$ iterations for full convergence. First-order methods without acceleration or curvature information make negligible progress in 80--150 iterations. Only Newton's method and Nesterov acceleration achieve meaningful convergence on Rosenbrock, by exploiting either exact curvature or the $O(1/k^2)$ accelerated rate.

\textbf{Adaptive step sizes:} Polyak's step is theoretically optimal but practically fragile --- its performance is critically sensitive to the accuracy of $f^{\star}$. Adagrad and RMSprop are robust alternatives that automatically adapt to gradient magnitudes without requiring knowledge of $f^{\star}$ or $L$. Adagrad's monotonically decaying step can eventually stall; RMSprop's exponential forgetting prevents this at the cost of some additional tuning.

\textbf{Momentum and acceleration:} Nesterov momentum achieves the optimal $O(1/k^2)$ convergence rate for smooth convex problems, providing the largest single improvement on Benchmark C (factor of 7 over GD baseline). Adam combines momentum with per-coordinate adaptation, offering good practical performance but is suboptimal on strongly nonlinear landscapes like Rosenbrock where gradient second moments are non-stationary.

\textbf{Second-order methods:} Newton's method is unmatched for problems where the Hessian is affordable to compute and invert ($d \leq$ a few hundred): one-step convergence on the quadratic problem, rapid convergence on B and C despite using only 20 iterations versus GD's 80. The computational cost ($O(d^3)$ per step) limits applicability to large-scale problems.

\textbf{Derivative-free methods:} The $\delta = 0.05$ finite-difference approximation nearly matches exact gradient descent at the same asymptotic cost, confirming that gradient-based optimisation remains feasible even when analytical gradients are unavailable, provided $\delta$ is chosen appropriately. Zeroth-order methods (Nesterov random search, Nelder--Mead) are substantially slower but require only function evaluations, making them applicable when even numerical derivatives are unavailable (e.g.\ simulation-based optimisation).

\textbf{Constrained optimisation:} Projected gradient descent enforces feasibility as a hard constraint from iteration 1, achieving the fastest entry into the feasible set and the cleanest convergence. Penalty methods offer greater flexibility (no projection oracle required) at the expense of introducing a hyperparameter $\lambda$ and requiring progressively smaller step sizes for larger penalties. Frank--Wolfe offers a constraint-handling alternative for box and simplex constraints via cheap linear minimisation, converging at $O(1/k)$.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
