\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C). Six categories of methods are investigated: (1)~adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), demonstrating that the Polyak rule achieves the best convergence on Rosenbrock ($f=0.009$) when $f^{\star}$ is known exactly, while diverging ($f=2.72$) when $f^{\star}$ is misspecified; (2)~momentum-based methods (Nesterov acceleration, Adam), where Nesterov achieves its theoretical $O(1/k^2)$ advantage on Rosenbrock ($f=0.479$ vs.\ GD $f=3.39$); (3)~Newton's method, which converges in one step on the quadratic loss and in 20 iterations to $f=0.686$ on Rosenbrock versus $f=3.73$ for gradient descent in 80 iterations; (4)~derivative-free approaches (finite differences with $\delta=0.05$ achieving $f=0.728$ on Benchmark~B vs.\ exact GD's $f=0.724$, while $\delta=0.8$ stalls at $f=1.640$; Nelder--Mead locating the \emph{exact} Rosenbrock minimum $f=0$ in 160 iterations despite using no gradients); (5)~constrained optimisation via projected gradient descent (achieving zero constraint violation from iteration~1) and quadratic penalty methods; and (6)~the Frank--Wolfe conditional gradient algorithm for linear-constrained problems, where the $O(1/k)$ convergence rate and the distinction between interior and boundary optima are analysed. Each method is evaluated in terms of convergence speed, stability, computational cost, and connection to theoretical guarantees.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. This section characterises each benchmark by its curvature, condition number, and relevant theoretical properties.

The three benchmark problems form a deliberate difficulty progression: Benchmark~A (quadratic, $\kappa \approx 1.12$) is nearly trivial for any descent method; Benchmark~B (weakly nonlinear, $\kappa \approx 6.9$) separates momentum from non-momentum methods; Benchmark~C (highly nonlinear, $\kappa \approx 2504$) separates first-order from second-order methods and exposes the fundamental limits of gradient-only algorithms. The optimal GD convergence rate satisfies $\rho^k \approx (0.056)^k$ on A, $(0.737)^k$ on B, and $(0.9992)^k$ on C --- a span of nearly 3 orders of magnitude in effective step.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
\label{sec:benchA}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \in \mathbb{R}^2, \quad y^{(i)} = (\theta^{\star})^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss is:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a \emph{strongly convex} quadratic with constant Hessian $H = \frac{1}{m}X^T X$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. The eigenvalues of $H$ are $\mu \approx 0.925$ and $L \approx 1.033$, giving condition number $\kappa(H) \approx 1.12$ --- a nearly isotropic, well-conditioned problem. The OLS minimiser (which is unique) is $\theta^{\star}_{\text{OLS}} = H^{-1}(X^T y/m) \approx (3.028,\, 4.017)$, with minimum loss $J^{\star} \approx 0.4826$ (nonzero due to observation noise). Starting point: $\theta_0 = [0, 0]^T$.

For gradient descent with constant step $\alpha$, the iteration $\theta_{k+1} = \theta_k - \alpha \nabla J(\theta_k)$ converges at a linear rate. On a quadratic $J(\theta) = \frac{1}{2}\theta^T H\theta - b^T\theta$, the error satisfies $\theta_k - \theta^{\star} = (I - \alpha H)^k(\theta_0 - \theta^{\star})$, so:
\begin{equation}
\|\theta_k - \theta^{\star}\| \leq \|I - \alpha H\|_2^k \|\theta_0 - \theta^{\star}\| = \rho^k\|\theta_0 - \theta^{\star}\|,
\end{equation}
where $\rho = \|I - \alpha H\|_2 = \max_{i}|1 - \alpha\lambda_i| = \max(|1 - \alpha\mu|,\, |1 - \alpha L|)$. Minimising $\rho$ over $\alpha$ gives the optimal step $\alpha^{\star} = 2/(\mu + L) \approx 1.01$ and the optimal rate $\rho^{\star} = (\kappa - 1)/(\kappa + 1) \approx 0.056$. The chosen step $\alpha = 0.08$ satisfies $\alpha < 2/L \approx 1.94$ (stability condition), giving $\rho = |1 - 0.08 \times 0.925| = |1 - 0.074| = 0.926$ --- slower than optimal, but the near-isotropy ($\kappa \approx 1.12$) means even this suboptimal rate gives rapid convergence.

\subsection{Benchmark B: Toy Neural Network Loss}
\label{sec:benchB}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \mathrm{diag}(2 - \sin(x_1),\ 10)$. Since $|\sin(x_1)| \leq 1$, the $(1,1)$-entry of $H$ satisfies $1 \leq 2 - \sin(x_1) \leq 3$, so the eigenvalues range from $[1, 10]$ (at $\sin(x_1) = 1$) to $[3, 10]$ (at $\sin(x_1) = -1$). Near the minimum at $x_1 \approx 0.582$, $\sin(0.582) \approx 0.550$, giving eigenvalues approximately $[1.45, 10]$ and condition number $\kappa \approx 6.9$.

The function is $L$-smooth with $L = \max_{x} \lambda_{\max}(H(x)) = \max\{3, 10\} = 10$ (achieved at $\sin(x_1) = -1$). The gradient descent stability condition requires $\alpha < 2/L = 0.2$. The chosen step $\alpha = 0.06$ satisfies this comfortably.

The global minimum lies at approximately $(0.582, 2.000)$ with $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\label{sec:benchC}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is:
\begin{equation}
H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\quad \text{eigenvalues} \approx 0.40 \text{ and } 1001.6, \quad \kappa \approx 2504.
\end{equation}
This extreme ill-conditioning ($\kappa \approx 2504$), combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions. Figure~\ref{fig:rosen_landscape} provides a multi-panel view of the landscape structure.

The Lipschitz constant of the gradient is not globally bounded (the Hessian grows without bound as $\|x\| \to \infty$), which is why very small step sizes are needed for stability. Near the minimum, $L_{\text{local}} \approx \lambda_{\max}(H(1,1)) \approx 1002$, requiring $\alpha \leq 1/L \approx 0.001$. The linear convergence rate at the minimum would be $\rho = (\kappa - 1)/(\kappa + 1) \approx 0.9992$, giving $\|x_k - x^{\star}\| \leq (0.9992)^k \|x_0 - x^{\star}\|$: reducing the error by half requires $\log(0.5)/\log(0.9992) \approx 866$ iterations. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$  \Comment{Adaptive, oracle-dependent step}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size \cite{polyak1969} is motivated by the subgradient method theory: for any $\alpha_k > 0$,
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 = \|x_k - \alpha_k g_k - x^{\star}\|^2 = \|x_k - x^{\star}\|^2 - 2\alpha_k g_k^T(x_k - x^{\star}) + \alpha_k^2 \|g_k\|^2.
\end{equation}
Using convexity, $g_k^T(x_k - x^{\star}) \geq f(x_k) - f^{\star}$. Minimising the right-hand side over $\alpha_k$ yields $\alpha_k = (f(x_k) - f^{\star})/\|g_k\|^2$, which is exactly the Polyak rule (with $\varepsilon$ regularisation for numerical safety). This guarantees $\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - (f(x_k)-f^{\star})^2/\|g_k\|^2$ \emph{when $f^{\star}$ is known exactly}.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate accumulation of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad adapts per-coordinate learning rates based on the history of all squared gradients. Coordinates with large historical gradients receive smaller steps. The monotonically decreasing effective step $\alpha_k^{(i)} = \alpha_0/\sqrt{\sum_{j=0}^k g_j^{(i)2}}$ provides implicit regularisation but may cause premature stalling on long training runs.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop (Hinton, 2012) replaces Adagrad's cumulative sum with an exponential moving average $v_{k+1} = \beta v_k + (1-\beta)g_k^2$, preventing the effective step size from monotonically decaying to zero. With $\beta = 0.9$, only the most recent gradient history influences the step size.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$  \Comment{Accumulate momentum}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

For a strongly convex quadratic with eigenvalues in $[\mu, L]$, the optimal Heavy Ball parameters are $\alpha^{\star} = 4/(\sqrt{L} + \sqrt{\mu})^2$ and $\beta^{\star} = \bigl((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)\bigr)^2$, yielding the accelerated linear rate $\rho = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$. On Benchmark~A with $\kappa \approx 1.12$, this gives $\rho \approx 0.058$, barely better than vanilla GD ($\rho \approx 0.056$) due to near-isotropy. On Rosenbrock (locally $\kappa \approx 2504$), the optimal $\rho_{\rm HB} \approx 0.961$ vs.\ GD's $\rho \approx 0.999$ --- a substantial speedup if the parameters could be tuned to the local curvature. The practical hyperparameters used here ($\alpha=0.0008$, $\beta=0.86$) are conservative to ensure stability across the entire trajectory where curvature varies significantly.

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$). Otherwise, Adagrad and Heavy Ball match or exceed GD baseline performance.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging. Heavy Ball shows characteristic overshoot followed by damped oscillation.
\item \textbf{Stability:} GD is most stable but slowest on ill-conditioned problems. Polyak is \emph{unstable} when $f^{\star}$ is misspecified. Adagrad is uniformly stable due to its monotonically decreasing effective step.
\item \textbf{Practical implication:} Exact $f^{\star}$ is rarely available in practice, limiting Polyak's applicability. RMSprop and Adagrad are more robust general-purpose choices, and the right choice depends on whether gradient history accumulation (Adagrad) or recent gradient statistics (RMSprop) better describes the problem.
\end{itemize}

\textbf{Relationship between Adagrad and RMSprop.} RMSprop \emph{generalises} Adagrad: with $\beta \to 0$, RMSprop's EMA $v_{k+1} = \beta v_k + (1-\beta) g_k^2$ reduces to $v_{k+1} = g_k^2$ (only the current gradient, no history). With $\beta \to 1$, it accumulates all history equally. Adagrad corresponds to the un-normalised cumulative sum $G_k = \sum_{j=0}^k g_j^2$, which can be written as the limit of an infinite EMA. RMSprop's EMA with $\beta \in (0,1)$ interpolates between these extremes, with effective ``memory window'' of $(1-\beta)^{-1}$ steps --- for $\beta = 0.9$, this is 10 recent gradients dominate.

\textbf{Polyak convergence on convex functions.} For a $\mu$-strongly convex $f$:
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 \leq \left(1 - \frac{(f(x_k)-f^{\star})^2}{\|g_k\|^2 \|x_k - x^{\star}\|^2}\right) \|x_k - x^{\star}\|^2.
\end{equation}
By strong convexity, $\|g_k\| \leq L \|x_k - x^{\star}\|$ and $f(x_k) - f^{\star} \geq \frac{\mu}{2}\|x_k - x^{\star}\|^2$, so the fraction satisfies:
$\frac{(f(x_k)-f^{\star})^2}{\|g_k\|^2\|x_k-x^{\star}\|^2} \geq \frac{(\mu/2)^2 \|x_k-x^{\star}\|^4}{L^2\|x_k-x^{\star}\|^4} = \frac{\mu^2}{4L^2}$.
This gives the linear rate $\rho_{\text{Polyak}} \leq \sqrt{1 - \mu^2/(4L^2)} \approx 1 - \mu^2/(8L^2)$ for $\kappa$ large --- worse than the optimal $(\kappa-1)/(\kappa+1)$, but with no step-size tuning required.

The connection between condition number and convergence speed is illustrated in Figure~\ref{fig:gd_rate_analysis}: Benchmark~A ($\kappa \approx 1.12$) lies in the ``easy'' region where virtually any step size gives fast convergence, while Benchmark~C ($\kappa \approx 2504$) requires adaptive or second-order methods. Figure~\ref{fig:q1_adagrad_analysis} provides a detailed view of Adagrad vs RMSprop on Benchmark~C, highlighting the monotone step decay issue.

\textbf{Adagrad effective step decay.} The per-coordinate Adagrad step for coordinate $i$ after $k$ iterations is:
\begin{equation}
\alpha_k^{(i)} = \frac{\alpha_0}{\sqrt{\sum_{j=0}^{k} (g_j^{(i)})^2} + \varepsilon}.
\end{equation}
If gradients are roughly stationary with $|g^{(i)}| \approx G$, then $\sum_{j=0}^k (g^{(i)}_j)^2 \approx kG^2$, so $\alpha_k^{(i)} \approx \alpha_0/(G\sqrt{k})$. This $O(1/\sqrt{k})$ decay matches the $O(1/\sqrt{k})$ regret bound. The cumulative step length is $\sum_{k=1}^{T} \alpha_k^{(i)} \approx (\alpha_0/G) \sum_{k=1}^T 1/\sqrt{k} = O(\sqrt{T})$, which means Adagrad keeps moving but with diminishing steps. On Figure~\ref{fig:q1_adagrad_analysis}, this decay is visible: for large $k$, the Adagrad step falls \emph{below} the constant GD step $\alpha = 0.0012$, explaining why Adagrad stalls while GD (paradoxically) continues to make slight progress on Benchmark~C in the extended 300-iteration run.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Gradient at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from the Heavy Ball method is the \emph{lookahead}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This correction allows Nesterov momentum to ``look ahead'' and correct its trajectory before overshooting. The lookahead point lies \emph{between} the previous and current iterate (not past the current iterate as in Heavy Ball), providing a stabilising effect.

\textbf{Chebyshev polynomial interpretation.} For strongly convex quadratics, the optimal first-order method corresponds to applying the Chebyshev polynomial of degree $k$ to the eigenvalues. The Chebyshev method with parameters derived from $\mu$ and $L$ achieves error reduction:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2}{(1 + \sqrt{1/\kappa})^{2k}} \cdot \|x_0 - x^{\star}\|^2_H,
\end{equation}
achieving linear rate $\rho = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$ --- the same as the optimal Heavy Ball rate. Nesterov's method with the $\beta_k$ schedule is a \emph{non-stationary} version that achieves $O(1/k^2)$ on general smooth convex functions (not just quadratics), making it more broadly applicable. On quadratics, both Chebyshev and Nesterov achieve the same $O(\rho^k)$ rate.

With the adaptive schedule $\beta_k = (k-1)/(k+2)$, Nesterov's method achieves the \emph{optimal} convergence rate of $O(1/k^2)$ for smooth convex functions \cite{nesterov1983}:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2},
\end{equation}
compared to the $O(1/k)$ rate of gradient descent. This is a \emph{complexity lower bound}: Nesterov (1983) proved that for any first-order method (using only oracle access to $\nabla f$) and any $k \leq (n-1)/2$, there exists an $L$-smooth convex function $f$ for which:
\begin{equation}
f(x_k) - f^{\star} \geq \frac{3L\|x_0 - x^{\star}\|^2}{32(k+1)^2}.
\end{equation}
No first-order method can universally beat $O(1/k^2)$ for smooth convex functions, so Nesterov's method is \emph{optimal}. For strongly convex functions (like Benchmark~A), linear convergence $O(\rho^k)$ is achievable and optimal, matching the Chebyshev lower bound.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam \cite{kingma2015} combines momentum (via first moment $m_t$) with per-coordinate adaptive scaling (via second moment $v_t$), inheriting advantages of both RMSprop and Heavy Ball. The bias-correction terms $1 - \beta_1^t$ and $1 - \beta_2^t$ compensate for the zero initialisation of $m_0$ and $v_0$, ensuring accurate estimates from the first iteration. With default parameters $\beta_1 = 0.9$, $\beta_2 = 0.999$, $\varepsilon = 10^{-8}$, Adam generally converges well across diverse problem types.

\textbf{Bias correction derivation.} Since $m_t = (1-\beta_1)\sum_{i=1}^t \beta_1^{t-i} g_i$, we have $\mathbb{E}[m_t] = (1 - \beta_1^t)\mathbb{E}[g]$ under stationarity assumptions. Thus $\hat{m}_t = m_t/(1-\beta_1^t)$ is an unbiased estimator of $\mathbb{E}[g]$. Similarly for $v_t$. The bias-correction factor $1 - \beta_1^t$ starts near $1 - 0.9^1 = 0.1$ (large correction at $t=1$) and approaches 1 exponentially fast (at $t=50$, $1-0.9^{50} \approx 0.995$), meaning the correction is effectively irrelevant after the first few dozen steps.

Intuitively, the effective step size at iteration $t$ for coordinate $i$ is:
\begin{equation}
\text{eff.\ }\alpha_i^{(t)} = \frac{\alpha \hat{m}_i^{(t)}}{\sqrt{\hat{v}_i^{(t)}} + \varepsilon} \approx \alpha \cdot \text{sign}(\hat{m}_i^{(t)}) \cdot \frac{|\hat{m}_i^{(t)}|/\sqrt{\hat{v}_i^{(t)}}+\varepsilon}{\cdot},
\end{equation}
which normalises the gradient momentum by the RMS of recent gradients. The Adam step is bounded: $|x_{t+1,i} - x_{t,i}| \leq \alpha/(1+\varepsilon)$, providing a per-coordinate ``trust region'' of radius $\approx \alpha$.

\textbf{Convergence caveat.} Despite Adam's popularity, Reddi et al.\ (2018) \cite{reddi2018} showed that Adam \emph{can fail to converge} on simple convex problems with a carefully constructed oscillating sequence of gradients. The root cause is that $v_t$ (exponential moving average) can ``forget'' large past gradients, causing the effective step to increase in directions where it should decrease. The fix --- AMSGrad --- uses $\hat{v}_t = \max(\hat{v}_{t-1}, v_t)$ to ensure non-decreasing second-moment estimates. In practice, Adam works well on the smooth benchmarks studied here, where the gradient magnitude is relatively stable.

Figure~\ref{fig:q2_adam_var} compares Adam (with and without bias correction) and AMSGrad on Benchmark~B. On this smooth problem, all variants converge to the same $f = 0.7244$; differences are mainly visible in early iterations. The right panel shows the bias correction factors: with $\beta_1 = 0.80$, the correction saturates after $\sim$21 steps; with $\beta_2 = 0.999$, the second-moment correction approaches 1 only after $\sim$4600 steps, meaning the second-moment estimate remains significantly biased throughout this 150-iteration run.

The Nesterov momentum schedule $\beta_k = \min\bigl((k-1)/(k+2),\ \beta_{\max}\bigr)$ is shown in Figure~\ref{fig:q2_schedule}. For small $k$, $\beta_k$ grows from 0, building momentum gradually. When $\beta_k$ reaches $\beta_{\max}$, it is capped, preventing momentum from growing without bound. The standard schedule without the cap corresponds to $\gamma_k = 1 - \beta_k = 3/(k+2)$, which decays as $O(1/k)$ --- exactly the step size schedule required for the $O(1/k^2)$ rate proof.

\paragraph{Optimality of $O(1/k^2)$.} Nesterov (1983) \cite{nesterov1983} proved not only that the accelerated gradient method achieves $O(1/k^2)$ but also established a \emph{matching lower bound}: for any first-order algorithm and any $k \leq (n-1)/2$ (dimension-free for $n \to \infty$), there exists a smooth convex function such that:
\begin{equation}
f(x_k) - f^{\star} \geq \frac{3L\|x_0 - x^{\star}\|^2}{32(k+1)^2}.
\end{equation}
This means \textbf{no first-order algorithm can achieve a convergence rate better than $O(1/k^2)$} on $L$-smooth convex functions. Nesterov's method is therefore \emph{optimal in the complexity-theoretic sense}.

\paragraph{Chebyshev polynomial connection.} For the special case of quadratic objectives $f(x) = \frac{1}{2}x^T H x - b^T x$ with $H$ symmetric positive definite, the optimal first-order method on Benchmark~A (eigenvalues $[\mu, L]$) is equivalent to evaluating a sequence of Chebyshev polynomials $\{T_k\}$ on the eigenvalues of $H$. Specifically, the residual after $k$ steps satisfies:
\begin{equation}
\frac{\|x_k - x^{\star}\|}{\|x_0 - x^{\star}\|} \leq 2\left(\frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1}\right)^k = 2\rho_{\rm acc}^k,
\end{equation}
where $\rho_{\rm acc} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$. This is the \emph{Chebyshev} convergence rate, matching the optimal rate achievable by any polynomial (first-order) method. Comparing to GD's $\rho_{\rm GD} = (\kappa-1)/(\kappa+1)$: for $\kappa = 2504$, $\rho_{\rm GD} \approx 0.9992$ but $\rho_{\rm acc} \approx (\sqrt{2504}-1)/(\sqrt{2504}+1) = 49.04/51.04 \approx 0.961$, a dramatic improvement. The ratio of iterations to achieve $\varepsilon$-accuracy scales as $\rho_{\rm GD}/\rho_{\rm acc} \approx \sqrt{\kappa}$ in favour of acceleration.

\textbf{Restart heuristic.} One practical improvement is to \emph{restart} the Nesterov momentum (reset $\beta_k \to 0$) when the objective function increases or when an approximate criterion $f(x_k) > f(x_{k-1})$ triggers. This addresses Nesterov's tendency to occasionally overshoot on non-convex problems. O'Donoghue \& Cand\`{e}s (2015) \cite{odonoghue2015} showed that periodic restarts achieve linear convergence on strongly convex problems (matching $\rho = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$) while retaining $O(1/k^2)$ on convex ones. The capped schedule $\beta_{\max}$ implemented here is a simpler proxy for this idea.

\subsection{Mini-Batch Stochastic Gradient Descent}
\label{sec:q2_sgd}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $\hat{g} \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimator}
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $\hat{g}$ is an unbiased estimator of the full gradient: $\mathbb{E}[\hat{g}] = \nabla J(\theta)$. Its variance is $\mathrm{Var}(\hat{g}) = \sigma^2_{\text{grad}}/b$, so the gradient noise scales inversely with batch size. Each epoch with batch size $b$ performs $m/b$ updates:
\begin{itemize}[nosep]
  \item $b = 5$: $1000/5 = 200$ updates per epoch (faster parameter updates, higher gradient variance).
  \item $b = 40$: $1000/40 = 25$ updates per epoch (lower gradient variance, slower per-epoch progress).
\end{itemize}

\subsection{Variance Reduction and SVRG}

Standard mini-batch SGD has an irreducible variance term that prevents convergence to the exact minimiser with a fixed step size: if $\hat{g}_k$ is an unbiased estimator with variance $\sigma^2_{\text{stoch}}$, then SGD with fixed $\alpha$ converges to a neighbourhood of size $O(\alpha \sigma^2_{\text{stoch}})$. This is the \emph{SGD noise floor}, distinct from the \emph{observation noise floor} $J^{\star}$ discussed in Section~\ref{sec:q2_sgd}.

\textbf{SVRG (Stochastic Variance Reduced Gradient)} \cite{johnson2013} eliminates the stochastic variance via a control variate:
\begin{equation}
\tilde{g}_k = \nabla f_i(x_k) - \nabla f_i(\tilde{x}) + \nabla f(\tilde{x}),
\end{equation}
where $\tilde{x}$ is a snapshot point updated every epoch and $\nabla f(\tilde{x}) = \frac{1}{m}\sum_i \nabla f_i(\tilde{x})$ is the full gradient at the snapshot. The control variate $\nabla f_i(\tilde{x}) - \nabla f(\tilde{x})$ has mean zero and cancels the dominant variance, yielding $\text{Var}(\tilde{g}_k) \to 0$ as $x_k \to x^{\star}$. This allows SVRG to achieve \emph{linear convergence} $O(\rho^k)$ with a constant step size on strongly convex problems --- at the cost of one extra full gradient computation per epoch.

\subsection{SGD Noise Floor: Theoretical Derivation}

The noise floor of SGD on the linear regression objective has an exact closed form. Write the solution as $\theta = \theta^{\star}_{\text{OLS}} + \delta\theta$ where $\theta^{\star}_{\text{OLS}} = (X^TX)^{-1}X^Ty$ is the OLS minimiser. The optimal loss (residual) satisfies:
\begin{equation}
J^{\star} = \frac{1}{2m}\|y - X\theta^{\star}_{\text{OLS}}\|^2 = \frac{1}{2m}\|P_\perp y\|^2,
\end{equation}
where $P_\perp = I - X(X^TX)^{-1}X^T$ is the orthogonal projection onto the null space of $X^T$. Since $y = X\theta^{\star}_{\text{true}} + \varepsilon$ and $P_\perp X = 0$:
\begin{equation}
P_\perp y = P_\perp(X\theta^{\star}_{\text{true}} + \varepsilon) = P_\perp \varepsilon.
\end{equation}
Thus $J^{\star} = \frac{1}{2m}\|P_\perp \varepsilon\|^2$. If $\varepsilon \sim \mathcal{N}(0, \sigma^2 I)$, then $\mathbb{E}[J^{\star}] = \frac{\sigma^2}{2m}\mathrm{tr}(P_\perp) = \frac{\sigma^2(m-n)}{2m}$. For $m = 1000$, $n = 2$, $\sigma = 1$: $\mathbb{E}[J^{\star}] = 998/2000 = 0.499$, consistent with the observed $J^{\star} \approx 0.4826$.

When the noise is scaled to $\sigma' = 6$, the observation vector becomes $y' = X\theta^{\star} + 6\varepsilon$ (using the \emph{same} $\varepsilon$ vector scaled by 6). Then:
\begin{equation}
J^{\star}_{(6)} = \frac{1}{2m}\|P_\perp y'\|^2 = \frac{1}{2m}\|6\,P_\perp \varepsilon\|^2 = 36 \cdot J^{\star}_{(1)}.
\end{equation}
This factor-of-36 scaling is \emph{exact} because the same noise vector is used, and is confirmed numerically: $J^{\star}_{\text{noisy}} \approx 17.37 = 36 \times 0.4826$. The key insight is that the minimum loss $J^{\star}$ is determined entirely by the \emph{irreducible observation noise component} orthogonal to the feature space, and scales exactly as $\sigma^2$.

\subsection{Local Approximations of $g(x) = x^4$}
\label{sec:q3_approx}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.0039 + 0.0625(x - 0.25).
\end{equation}
This is a tangent line that captures only local slope; it has no minimum and provides no useful model for optimisation.

\textbf{Second-order approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
The Newton update minimises this local quadratic model:
\begin{equation}
x_{\text{new}} = \arg\min_{x}\ \text{(quadratic approximation)} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{4 \cdot 0.25^3}{12 \cdot 0.25^2} = 0.25 - \frac{0.25}{3} \approx 0.167.
\end{equation}
Since $g(x) = x^4$ has its minimum at $x^{\star} = 0$, the Newton step takes us closer to the minimum than a gradient step, illustrating the superiority of second-order information.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised (Levenberg-Marquardt) Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton direction $p_k = [\nabla^2 f(x_k)]^{-1} \nabla f(x_k)$ corresponds to the steepest descent direction in the \emph{metric defined by the Hessian}. It fully accounts for curvature and is \emph{scale-invariant}: the Newton step does not depend on the coordinate scaling of $f$. Near a strongly convex minimum, full Newton steps ($\alpha = 1$) achieve \emph{quadratic convergence}:
\begin{equation}
\|\nabla f(x_{k+1})\| \leq \frac{M}{2m^2} \|\nabla f(x_k)\|^2,
\end{equation}
where $m$ is the strong convexity constant and $M$ bounds the Lipschitz constant of the Hessian. Each Newton step requires solving an $n \times n$ linear system at cost $O(n^3)$, compared to $O(n)$ per gradient step. For $n = 2$, this cost is negligible, but in high dimensions Newton's method is prohibitively expensive without approximations.

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian regularisation $\lambda = 10^{-8}$.

\subsection{Newton Decrement and Two-Phase Convergence}

The \textbf{Newton decrement} $\lambda(x)$ is defined by:
\begin{equation}
\lambda(x)^2 = \nabla f(x)^T \bigl[\nabla^2 f(x)\bigr]^{-1} \nabla f(x) = p^T \nabla^2 f(x)\, p,
\end{equation}
where $p = [\nabla^2 f(x)]^{-1}\nabla f(x)$ is the Newton direction. The quantity $\lambda(x)^2/2$ upper-bounds the suboptimality: $f(x) - f^{\star} \leq \lambda(x)^2/2$ when $f$ is self-concordant. This provides a computable, affine-invariant \emph{stopping criterion}: iterate until $\lambda(x)^2/2 \leq \varepsilon$.

Newton's method exhibits two distinct phases:
\begin{enumerate}[nosep]
  \item \textbf{Damped Newton phase} (far from optimum): The Hessian model is inaccurate; full Newton steps may overshoot or even increase $f$. A line search or fixed damping $\alpha < 1$ is required. During this phase, convergence is at most linear: $f(x_{k+1}) - f^{\star} \leq \bigl(1 - \delta\bigr)(f(x_k) - f^{\star})$ for some $\delta > 0$.
  \item \textbf{Quadratic convergence phase} (near optimum): Once $\lambda(x_k)^2/2 \leq \eta$ for some threshold $\eta$ (typically $\eta \approx 0.1$), full Newton steps with $\alpha = 1$ achieve quadratic convergence:
  \begin{equation}
    f(x_{k+1}) - f^{\star} \leq c \cdot (f(x_k) - f^{\star})^2,
  \end{equation}
  for a constant $c > 0$ depending on the self-concordance parameter. This ``doubling of digits'' is why Newton converges in $O(\log\log(1/\varepsilon))$ steps once the quadratic regime is entered.
\end{enumerate}

On Benchmark~A (quadratic), the entire run is in Phase 2 (the Hessian is constant and exact), so Newton converges in exactly 1 step. On Benchmark~B, the mild nonlinearity means Phase 1 lasts $\sim$3--4 iterations before quadratic convergence kicks in. On Rosenbrock (Benchmark~C), the extreme condition number and rapidly varying Hessian mean that the 20 damped Newton steps remain in Phase 1 throughout: the iterate reaches $(0.183, 0.020)$, still far from the quadratic convergence basin around $(1,1)$.

Figure~\ref{fig:q3_phases} demonstrates the two-phase convergence on Benchmark~B, showing both gradient norm decay and the Newton decrement $\lambda^2(x)/2$ as a convergence certificate. Both starting points converge to machine precision ($\|\nabla f\| \approx 10^{-16}$) within 30 iterations, achieving $f = 0.7244$. The Newton decrement drops below the quadratic-phase threshold $\eta \approx 0.01$ within the first 3--5 iterations, after which convergence is superlinear.

Newton's method requires $O(n^2)$ storage (to store the Hessian) and $O(n^3)$ computation (to solve the linear system) per iteration, versus $O(n)$ for gradient-based methods. For the $n = 2$ problems here, this difference is negligible. However, for neural networks with millions of parameters, exact Newton's method is computationally infeasible. The cost--benefit ratio:
\begin{itemize}[nosep]
  \item Benchmark~A: Newton saves $\approx 79$ gradient evaluations (total GD gradient work: 80 evals; Newton: 20 eval + 20 Hessian evaluations + 20 solves). For small $n=2$, the overhead is trivial.
  \item Benchmark~C: Newton achieves 5$\times$ better objective in $4\times$ fewer iterations --- a strong case for second-order methods when the Hessian is affordable.
\end{itemize}

Practical alternatives for large-scale problems include:
\begin{itemize}[nosep]
  \item \textbf{Quasi-Newton (BFGS / L-BFGS):} Approximates $H^{-1}$ using $m$ gradient history pairs, $O(mn)$ per step. Achieves superlinear convergence (between linear and quadratic) without storing the full Hessian.
  \item \textbf{Gauss--Newton:} Approximates the Hessian with the outer product of Jacobians $J^TJ$ (positive semidefinite by construction), appropriate for nonlinear least-squares problems.
  \item \textbf{Natural gradient:} Uses the Fisher information matrix $F = \mathbb{E}[\nabla \log p \cdot (\nabla \log p)^T]$ as a Riemannian metric, popular in probabilistic models and reinforcement learning.
\end{itemize}

\paragraph{L-BFGS in detail.} The BFGS (Broyden--Fletcher--Goldfarb--Shanno) update \citep{nocedal2006} maintains a curvature approximation $B_k \approx H_k$ via a rank-2 correction that satisfies the \emph{secant condition} $B_{k+1} s_k = y_k$, where $s_k = x_{k+1} - x_k$ and $y_k = \nabla f(x_{k+1}) - \nabla f(x_k)$. Working with the inverse approximation $\hat{H}_k = B_k^{-1}$, the BFGS rank-2 update is:
\begin{equation}
\hat{H}_{k+1} = \left(I - \rho_k s_k y_k^T\right)\hat{H}_k\left(I - \rho_k y_k s_k^T\right) + \rho_k s_k s_k^T, \quad \rho_k = \frac{1}{y_k^T s_k}.
\end{equation}
This is a Sherman--Morrison--Woodbury update that preserves positive definiteness whenever the curvature condition $y_k^T s_k > 0$ holds (guaranteed by the Wolfe conditions on the line search). The descent direction is $p_k = -\hat{H}_k \nabla f(x_k)$, requiring only an $O(n^2)$ matrix--vector product rather than an $O(n^3)$ solve.

For large-scale problems, \textbf{L-BFGS} (\emph{limited-memory} BFGS) stores only the $m$ most recent pairs $\{(s_j, y_j)\}_{j=k-m}^{k}$ instead of the full $n \times n$ matrix $\hat{H}_k$, reducing storage from $O(n^2)$ to $O(mn)$. The implicit matrix--vector product $\hat{H}_k g$ is computed in $O(mn)$ via the two-loop recursion (Algorithm~12 of \citealt{nocedal2006}).

On our $n = 2$ Rosenbrock problem, L-BFGS (with $m = 5$) would converge to $f < 10^{-10}$ in $\approx 30$--$50$ iterations --- competitive with Newton but requiring no explicit Hessian computation. For a neural network with $n = 10^7$ parameters, Newton's $O(n^3) = O(10^{21})$ cost is completely infeasible, whereas L-BFGS costs only $O(mn) = O(5 \times 10^7)$ per iteration, making it the standard ``second-order'' method for medium-scale unconstrained optimisation in scientific computing (e.g., \texttt{scipy.optimize.minimize} with \texttt{method='L-BFGS-B'}).

The convergence rate of BFGS satisfies:
\begin{equation}
\lim_{k\to\infty} \frac{\|\theta_{k+1} - \theta^{\star}\|}{\|\theta_k - \theta^{\star}\|} = 0 \quad \text{(superlinear)},
\end{equation}
meaning $\|\theta_k - \theta^{\star}\| = o(\rho^k)$ for \emph{any} $\rho > 0$. In practice this is observed as ``rapidly accelerating'' linear convergence: after an initial phase, each step reduces the error by a larger factor than the last, approaching Newton's quadratic rate without requiring explicit Hessian assembly.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad i = 1, \ldots, n.
\end{equation}
The total approximation error has two competing components whose balance determines the optimal $\delta$.

\paragraph{Truncation error.} Expanding $f(x + \delta e_i)$ via Taylor's theorem:
\begin{equation}
f(x + \delta e_i) = f(x) + \delta \frac{\partial f}{\partial x_i}(x) + \frac{\delta^2}{2}\frac{\partial^2 f}{\partial x_i^2}(x) + O(\delta^3).
\end{equation}
Substituting into the forward difference formula:
\begin{equation}
\frac{f(x + \delta e_i) - f(x)}{\delta} = \frac{\partial f}{\partial x_i}(x) + \frac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x) + O(\delta^2).
\end{equation}
The leading truncation error is $E_T = \frac{\delta}{2}|f''(x)| = O(\delta)$, which grows linearly with $\delta$. The \emph{central difference} formula $[f(x+\delta e_i) - f(x-\delta e_i)]/(2\delta)$ cancels the $O(\delta)$ term, achieving $E_T = O(\delta^2)$ at cost $2n$ function evaluations instead of $n+1$.

\paragraph{Rounding error.} In floating-point arithmetic, $f(x)$ is computed as $f(x)(1 + \varepsilon_1)$ and $f(x+\delta e_i)$ as $f(x+\delta e_i)(1+\varepsilon_2)$ where $|\varepsilon_j| \leq \varepsilon_{\text{machine}} \approx 2.2 \times 10^{-16}$ (double precision). The subtraction $f(x+\delta e_i) - f(x)$ suffers cancellation when both values are close, giving absolute error $\approx 2\varepsilon_{\rm machine}|f(x)|$. The rounding contribution to the gradient approximation is:
\begin{equation}
E_R \approx \frac{2\varepsilon_{\rm machine}|f(x)|}{\delta} = O\!\left(\frac{\varepsilon_{\rm machine}}{\delta}\right),
\end{equation}
which blows up as $\delta \to 0$.

\paragraph{Optimal $\delta$.} Minimising the total error $E(\delta) = E_T + E_R = c_1 \delta + c_2 \varepsilon_{\rm machine}/\delta$ (with $c_1 = |f''|/2$ and $c_2 = 2|f|$) by setting $dE/d\delta = 0$:
\begin{equation}
\delta^{\star} = \sqrt{\frac{c_2}{c_1}\,\varepsilon_{\rm machine}} = \sqrt{\frac{4|f(x)|}{|f''(x)|}}\,\varepsilon_{\rm machine}^{1/2} \approx \varepsilon_{\rm machine}^{1/2} \approx 10^{-8},
\end{equation}
giving minimum total error $E(\delta^{\star}) \approx 2\sqrt{c_1 c_2}\,\varepsilon_{\rm machine}^{1/2} \approx 2\sqrt{|f||f''|}\times 10^{-8}$. For the central difference approximation, the analogous calculation gives $\delta^{\star}_{\rm central} \approx \varepsilon_{\rm machine}^{1/3} \approx 10^{-5}$, a much larger optimal step size. The standard recommendation in numerical analysis \citep{nocedal2006} is to use $\delta = 10^{-8}$ for forward differences and $\delta = 10^{-5}$ for central differences in double precision.

The experiments test $\delta = 0.05$ (moderately small, good accuracy) and $\delta = 0.8$ (large, high truncation error). Each FD gradient evaluation requires $n + 1 = 3$ function evaluations for $n = 2$ dimensions.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Uniform random direction}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates a \emph{directional} derivative along a random direction, requiring only \textbf{two} function evaluations per iteration regardless of dimension $n$. It is \emph{dimension-independent} in cost per step, making it attractive for very high-dimensional black-box optimisation. However, the expected directional derivative satisfies:
\begin{equation}
\mathbb{E}_{u}[\hat{g}_k] = \mathbb{E}_u\!\left[\frac{f(x + \delta u) - f(x)}{\delta} u\right] \approx \mathbb{E}_u[(u^T \nabla f(x)) u] = \frac{1}{n}\nabla f(x),
\end{equation}
meaning the estimator is proportional to the true gradient in expectation but with high variance. In $n = 2$ dimensions, the per-step progress is effectively $1/n = 1/2$ of a true gradient step.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm \cite{nelder1965} maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four geometric operations:
\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r = 1.0$): Move the worst vertex through the centroid of the other vertices.
  \item \textbf{Expansion} ($\gamma = 2.0$): If reflection improved, try a further step in the same direction.
  \item \textbf{Contraction} ($\rho = 0.5$): If reflection worsened, contract toward the centroid.
  \item \textbf{Shrink} ($\sigma = 0.5$): If contraction also fails, contract all vertices toward the best.
\end{enumerate}
No gradient information is used; each iteration requires at most 2 function evaluations (reflection + one of expansion/contraction). The simplex naturally \emph{elongates} along promising directions and \emph{contracts} near the minimum.

Settings: initial simplex step $= 0.35$, iterations $= 160$.

\subsection{Problem Setup and Optimality Conditions}
\label{sec:q5_setup}

Minimise Benchmark~B subject to a half-space constraint, starting from the infeasible point $x_0 = (0.2, 4.0)$:

\textbf{Parametric analysis.} Figure~\ref{fig:q5_parametric} shows how the constrained optimum $f^{\star}(c)$ changes as the constraint level $c$ varies. For $c \leq 0.582$ (the unconstrained $x_1^{\star}$), the constraint is inactive and $f^{\star}(c) = f^{\star}_{\text{unc}} = 0.7244$. For $c > 0.582$, the constraint becomes active ($x_1^{\star}(c) = c$), and $f^{\star}(c)$ increases. At $c = 1.0$, the constrained optimum degrades to $f \approx 0.839$. The experiment uses $c = 0.5 < 0.582$, confirming that the constraint is inactive --- a special case where constrained and unconstrained methods converge to the same point.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q5_parametric.pdf}
\caption{Q5: Parametric constraint analysis. Left: constrained minimum $f^{\star}(c)$ as a function of the lower bound $c$ on $x_1$. For $c \leq 0.582$ (the unconstrained $x_1^{\star}$, green dashed), the constraint is inactive and $f^{\star}$ remains constant at 0.7244. For $c > 0.582$, the constraint is active (red shaded), $x_1^{\star}(c) = c$, and $f^{\star}$ increases. Right: the optimal $x_1^{\star}(c)$ is constant at 0.582 when inactive (blue) and tracks $c$ when active (red). The experiment at $c = 0.5$ is in the inactive region: the constraint is satisfied by the unconstrained optimum.}
\label{fig:q5_parametric}
\end{figure}
\begin{equation}
\min_{x \in \mathcal{C}}\ f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1), \qquad \mathcal{C} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}.
\end{equation}
The \textbf{KKT conditions} for this inequality-constrained problem are:
\begin{align}
\nabla f(x^{\star}) - \mu e_1 &= 0, \quad \mu \geq 0, \label{eq:kkt_stationarity}\\
0.5 - x_1^{\star} &\leq 0, \label{eq:kkt_primal}\\
\mu(0.5 - x_1^{\star}) &= 0. \label{eq:kkt_complementarity}
\end{align}
These three conditions --- stationarity, primal feasibility, and complementary slackness --- are necessary for optimality under the Linear Constraint Qualification (LCQ), which holds trivially for the half-space constraint $x_1 \geq 0.5$ \citep{boyd2004}. Since $f$ is strictly convex near $x^{\star}$ (the Hessian is positive definite with minimum eigenvalue $\mu_f \approx 1.45$ at the optimum), the KKT conditions are also \emph{sufficient}: any KKT point is a global minimiser.

\paragraph{Case 1: Constraint inactive ($c \leq x_1^{\star}_{\rm unc}$).} The unconstrained minimum of $f$ lies at $x^{\star}_{\text{unc}} = (0.582, 2.000)$ with $x_1 = 0.582 > 0.5$, so it satisfies the constraint strictly. Substituting into \eqref{eq:kkt_stationarity}: $\nabla f(x^{\star}_{\rm unc}) = 0$ (since it's the unconstrained minimiser), so $\mu e_1 = 0 \Rightarrow \mu = 0$. Complementary slackness \eqref{eq:kkt_complementarity} is satisfied since $\mu = 0$. All KKT conditions hold with $\mu = 0$, confirming that the constrained solution coincides with the unconstrained solution: $x^{\star}_{\text{con}} = x^{\star}_{\text{unc}} = (0.582, 2.000)$, $f^{\star} = 0.7244$.

\paragraph{Case 2: Constraint active ($c > x_1^{\star}_{\rm unc} \approx 0.582$).} When the lower bound $c$ is moved above $x_1^{\star}_{\rm unc}$, the constraint becomes active at optimality ($x_1^{\star} = c$). From the stationarity condition, the dual variable satisfies:
\begin{equation}
\mu = \frac{\partial f}{\partial x_1}\bigg|_{x^{\star}} = 2(c - 1) + \cos(c).
\end{equation}
For $c = 0.582$: $\mu = 2(0.582-1) + \cos(0.582) = -0.836 + 0.836 = 0$ (consistent with inactive constraint). For $c = 1.0$: $\mu = 2(1.0-1) + \cos(1.0) = 0 + 0.540 = 0.540 > 0$ (active constraint, dual variable positive). The constraint violation $x_1^{\star} = c = 1.0 > 0.5$ satisfies primal feasibility, complementary slackness holds since $\mu > 0$ and $x_1^{\star} - 0.5 = 0.5 > 0$ requires $0.5 - x_1^{\star} = -0.5 \leq 0$ (boundary active), confirming the KKT point.

The current experiment uses $c = 0.5$, which satisfies Case~1: the constraint is inactive, $\mu = 0$, and all methods that converge to the feasible region will find $f^{\star} = 0.7244$ at $(0.582, 2.000)$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1})$ \Comment{Project onto $\mathcal{C}$: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection $\Pi_{\mathcal{C}}(z) = (\max(0.5, z_1), z_2)^T$ is the \emph{Euclidean projection} onto the half-space $\mathcal{C}$. It satisfies $\|\Pi_{\mathcal{C}}(z) - x^{\star}\| \leq \|z - x^{\star}\|$ for all $x^{\star} \in \mathcal{C}$, enabling convergence proofs identical to the unconstrained case. The projected GD iteration corresponds to minimising the first-order Taylor approximation subject to the feasibility constraint and a proximal regulariser.

\begin{algorithm}[H]
\caption{Linear Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Linear (L1) penalty for infeasibility}
    \State Compute $\partial F_\lambda(x_k)$: subgradient of $f$ plus penalty subgradient
    \State $x_{k+1} \gets x_k - \alpha\, \partial F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The code implements the \emph{linear} (L1-type) penalty $F_\lambda(x) = f(x) + \lambda \max(0, 0.5 - x_1)$. The subgradient of the penalty w.r.t.\ $x_1$ is $-1$ when $x_1 < 0.5$ and $0$ when $x_1 > 0.5$ (undefined at $x_1 = 0.5$). Thus:
\begin{equation}
\partial F_{\lambda}(x) = \nabla f(x) + \begin{cases} -\lambda\, e_1 & \text{if } x_1 < 0.5, \\ 0 & \text{if } x_1 \geq 0.5. \end{cases}
\end{equation}
The linear penalty has the advantage that with sufficiently large $\lambda$, it acts as an \emph{exact penalty}: for problems satisfying a constraint qualification, there exists $\lambda^{\star}$ such that for all $\lambda \geq \lambda^{\star}$, the unconstrained minimiser of $F_\lambda$ is exactly the constrained minimiser. For the quadratic penalty $\lambda \max(0, 0.5-x_1)^2$, feasibility is only achieved in the limit $\lambda \to \infty$.

A major limitation of the linear penalty is non-smoothness at $x_1 = 0.5$, requiring subgradient methods rather than gradient descent. In practice, the gradient descent steps still work because the iterates spend most time away from $x_1 = 0.5$, but theoretical guarantees require care. The augmented Lagrangian $\mathcal{L}(x, \mu) = f(x) + \mu(0.5 - x_1) + \frac{\lambda}{2}\max(0, 0.5-x_1)^2$ combines exact penalisation with smooth conditioning.
Settings: projected GD ($\alpha = 0.08$), penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected GD} enforces feasibility as a hard constraint at every iteration --- this is the fundamental advantage over penalty methods. The cost is requiring a \emph{projection oracle}, which is trivial for half-spaces ($\max$ operation) but can be expensive or even intractable for general constraint sets (e.g., semidefinite constraints require $O(n^3)$ projection).

\textbf{Penalty methods} require only a modified gradient and no projection, offering algorithmic simplicity. The trade-off:
\begin{itemize}[nosep]
  \item Small $\lambda$: weak enforcement, residual violations (penalty gradient $-\lambda$ insufficient to overcome the objective gradient near the boundary).
  \item Large $\lambda$: strong enforcement. Unlike the quadratic penalty, the \emph{linear} penalty does not affect the Hessian (the second derivative of $\lambda\max(0, 0.5-x_1)$ is zero), so the conditioning is unchanged. However, the larger penalty gradient $-\lambda$ creates a steep drop at $x_1 = 0.5$, causing oscillation when the iterate crosses the boundary. This is why a smaller step size ($\alpha = 0.03$ for $\lambda = 4.5$ vs.\ $\alpha = 0.05$ for $\lambda \leq 1.8$) is needed for stability.
\end{itemize}
Augmented Lagrangian methods (e.g., ADMM) combine dual multipliers $\mu$ with a quadratic penalty term to achieve exact feasibility at finite $\lambda$. The augmented Lagrangian is:
\begin{equation}
\mathcal{L}_\lambda(x, \mu) = f(x) + \mu(0.5 - x_1) + \frac{\lambda}{2}\max(0,\, 0.5 - x_1)^2.
\end{equation}
The dual update $\mu_{t+1} \leftarrow \max(0,\, \mu_t + \lambda(0.5 - x_t))$ ensures $\mu \to \mu^{\star}$ (the optimal dual multiplier). For this problem, since the unconstrained minimum already satisfies the constraint, $\mu^{\star} = 0$ and the augmented Lagrangian reduces to the original problem at convergence.

\subsection{Feasible Set and Linear Programme}
\label{sec:q6_lp}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box/polytope).

\textbf{Linear Programme:} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$ with $a = [1, 2]^T$. Since the objective is linear, the fundamental theorem of linear programming guarantees that the minimum is attained at a \emph{vertex} of the polytope $\mathcal{X}$. The 4 vertices of the box are $(0.5, -5)$, $(0.5, 10)$, $(5, -5)$, $(5, 10)$. The objective values at these vertices are:
\begin{equation}
f(0.5, -5) = 0.5 - 10 = -9.5 \quad\text{(minimum)}, \quad
f(5, 10) = 5 + 20 = 25 \quad\text{(maximum)}.
\end{equation}
Since both components of $a$ are positive, the minimum is at the lower bounds:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = a^T x^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}

\subsection{Frank--Wolfe Algorithm}
\label{sec:q6_fw}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (feasibility maintained)}
\EndFor
\end{algorithmic}
\end{algorithm}

The Frank--Wolfe method \cite{frankwolfe1956} replaces the projection oracle of projected GD with a \emph{linear minimisation oracle} (LMO): $z_k = \arg\min_{x \in \mathcal{X}} \langle \nabla f(x_k), x \rangle$. For box constraints, this decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations. Since $x_{k+1} = \beta x_k + (1-\beta) z_k$ is a convex combination of two feasible points, \emph{every iterate is automatically feasible} --- no projection is needed.

Key theoretical properties of Frank--Wolfe for $L$-smooth convex $f$ over a compact convex set $\mathcal{X}$ with diameter $D$:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2} = O\!\left(\frac{1}{k}\right),
\end{equation}
using the step size schedule $\gamma_k = 2/(k+2)$ (corresponding to $\beta_k = k/(k+2)$). This rate is \emph{optimal} for the Frank--Wolfe class of methods \cite{jaggi2013} and cannot be improved without stronger assumptions.

\paragraph{Convergence proof sketch.} Starting from the $L$-smoothness inequality applied to the Frank--Wolfe step $x_{k+1} = x_k + \gamma_k(z_k - x_k)$:
\begin{align}
f(x_{k+1}) &\leq f(x_k) + \gamma_k \langle \nabla f(x_k),\, z_k - x_k \rangle + \frac{L\gamma_k^2}{2}\|z_k - x_k\|^2 \nonumber\\
&\leq f(x_k) + \gamma_k \langle \nabla f(x_k),\, x^{\star} - x_k \rangle + \frac{L\gamma_k^2 D^2}{2},
\end{align}
where the second inequality uses $\langle \nabla f(x_k), z_k - x_k \rangle \leq \langle \nabla f(x_k), x^{\star} - x_k \rangle$ (since $z_k$ is the linear minimiser over $\mathcal{X}$, hence no worse than $x^{\star}$), and $\|z_k - x_k\| \leq D$. By convexity, $\langle \nabla f(x_k), x^{\star} - x_k \rangle \leq f(x^{\star}) - f(x_k) = f^{\star} - f(x_k)$. Defining $h_k = f(x_k) - f^{\star}$:
\begin{equation}
h_{k+1} \leq (1 - \gamma_k) h_k + \frac{L\gamma_k^2 D^2}{2}.
\end{equation}
Substituting $\gamma_k = 2/(k+2)$ and applying the recurrence $h_k \leq \frac{C}{k+2}$ with $C = \max(h_0 \cdot 2, 2LD^2)$ proves the $O(1/k)$ rate by induction. For strongly convex $f$, a similar analysis yields a geometric rate $O(\rho^k)$ with the optimal step schedule, but standard Frank--Wolfe cannot achieve this without modifications (away-steps or pairwise steps).

For the interior optimum $f(x) = (x_1-1)^2 + (x_2-5)^2$, the Lipschitz constant of the gradient is $L = 2$ (from $\nabla^2 f = 2I$). The diameter of $\mathcal{X} = [0.5,5] \times [-5,10]$ is $D = \|(4.5, 15)\|_2 \approx 15.66$. The theoretical bound gives:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2 \times 2 \times 15.66^2}{k+2} = \frac{980}{k+2}.
\end{equation}
At $k = 140$ iterations, this gives $f(x_{140}) - f^{\star} \leq 980/142 \approx 6.9$, while the observed suboptimality with $\beta = 0.985$ is only $f - f^{\star} \approx 0.005$ --- three orders of magnitude below the bound. This illustrates that the universal $2LD^2$ constant is highly pessimistic: it assumes the worst-case starting point at distance $D$ from the optimum, whereas the actual distance $\|x_0 - x^{\star}\| = \|(1,1) - (1,5)\| = 4 \ll D = 15.66$.

\subsection{Interior vs.\ Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits the characteristic ``zig-zagging'' between extreme vertices $z_k$ of $\mathcal{X}$. The convex combination $x_{k+1} = \beta x_k + (1-\beta)z_k$ gradually averages these jumps. The general convergence rate is $O(1/k)$ for smooth convex objectives. Fixed $\beta$ schedules are suboptimal compared to the decaying $\gamma_k = 2/(k+2)$, which satisfies $\sum_k \gamma_k = \infty$ and $\sum_k \gamma_k^2 < \infty$ (necessary for convergence).

\textbf{Boundary optimum:} Frank--Wolfe is particularly well-suited to boundary optima in the sense that no projection is required: when the gradient consistently points toward the infeasible region, $z_k$ selects the feasible vertex closest to the unconstrained minimum. However, convergence to a boundary point can be slow with fixed $\beta$ because the algorithm must ``learn'' that the constraint is active. The boundary optimum case here achieves $f = 0.364$ vs.\ $f^{\star} = 0.25$ after 140 iterations, reflecting the $O(1/k)$ rate.

\textbf{Sparsity property of Frank--Wolfe:} Each iterate $x_{k+1} = \beta x_k + (1-\beta)z_k$ is a convex combination of iterates and LP vertices. After $k$ iterations, $x_k$ is a convex combination of at most $k+1$ vertices of $\mathcal{X}$. This naturally produces \emph{sparse} solutions when $\mathcal{X}$ is a polytope with few vertices (e.g., a simplex in probability space, or a norm ball). Projected GD, by contrast, can produce dense iterates even when the constrained optimum is a vertex.

\textbf{Comparison with projected GD:} Frank--Wolfe requires only a linear oracle (cheap for polytopes: $O(d)$ per step) instead of a projection (which can be expensive for general convex sets: $O(n^3)$ for semidefinite constraints). Its $O(1/k)$ rate is slower than the $O(1/k^2)$ rate of projected Nesterov acceleration, but the simpler per-iteration computation and the sparsity property often make it faster in practice for large-scale polytope-constrained problems.

\textbf{Away-step Frank--Wolfe.} A fundamental limitation of the standard Frank--Wolfe method is that it can only move \emph{toward} the LP-optimal vertex $z_k$. If the current iterate $x_k$ already has positive weight on a vertex that is far from the optimum, standard Frank--Wolfe cannot directly remove it. The \emph{away-step} variant solves an additional LP to find the ``worst'' vertex $v_k = \arg\max_{v \in \mathcal{S}_k} \langle \nabla f(x_k), v \rangle$ (where $\mathcal{S}_k$ is the current active set) and moves \emph{away} from $v_k$. Lacoste-Julien \& Jaggi (2015) \cite{lacostejulien2015} showed that away-step Frank--Wolfe achieves \emph{linear convergence} $O(\rho^k)$ on strongly convex objectives over polytopes, closing the gap between Frank--Wolfe and projected gradient methods.

\textbf{LMO complexity across constraint types.}
\begin{itemize}[nosep]
  \item Box $\mathcal{X} = [l, u]$: LMO $z_i = l_i \mathbf{1}[\nabla_i f > 0] + u_i \mathbf{1}[\nabla_i f \leq 0]$. Cost: $O(d)$.
  \item Simplex $\Delta = \{x : x \geq 0, \mathbf{1}^T x = 1\}$: LMO $z = e_{i^{\star}}$ where $i^{\star} = \arg\min_i \nabla_i f$. Cost: $O(d)$.
  \item Spectahedron (trace norm ball): LMO solves a minimum eigenvalue problem. Cost: $O(n^3)$ exact, $O(n^2)$ with power iteration (1 step).
  \item Nuclear norm ball (matrix $\|X\|_{\star} \leq \tau$): LMO requires 1 SVD step. Cost: $O(mn)$ for an $m \times n$ matrix.
\end{itemize}
For the spectahedron and nuclear norm ball, the LMO is \emph{cheaper} than projection ($O(n^3)$ vs $O(n^3)$ for eigendecomposition, but LMO only needs the top eigenvector while projection needs the full decomposition).

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This study compared a broad spectrum of optimisation algorithms across three benchmark problems of increasing difficulty, revealing consistent patterns that connect theoretical properties to empirical behaviour.

\paragraph{Adaptive step sizes (Q1).}
The Polyak rule is theoretically optimal but practically fragile due to its dependence on an exact $f^{\star}$. Adagrad and RMSprop provide robust, parameter-efficient alternatives. Heavy Ball adds momentum to smooth convergence. On well-conditioned problems (Benchmark~A, $\kappa \approx 1.12$), all methods perform similarly. Differences become stark on the ill-conditioned Rosenbrock ($\kappa \approx 2504$).

\paragraph{Momentum and SGD (Q2).}
Nesterov acceleration achieves the $O(1/k^2)$ optimal convergence rate, visibly outperforming GD on Benchmark~C (factor $\approx 7\times$ improvement). Adam is effective across diverse problems but is outperformed by Nesterov on ill-conditioned smooth functions. Mini-batch SGD introduces a fundamental variance--speed trade-off: smaller batches update more frequently but with noisier gradients; the batch size matters less when observation noise dominates.

\paragraph{Newton's method (Q3).}
Newton's method provides dramatically faster convergence at the cost of $O(n^3)$ per-iteration computation. On the quadratic Benchmark~A, it finds the exact solution in a single step. On Rosenbrock, the damped Newton step ($\alpha = 0.22$, 20 iterations) achieves $f = 0.686$ vs.\ GD's $f = 3.73$ in 80 iterations. The Hessian's curvature information is the decisive advantage on ill-conditioned problems.

\paragraph{Derivative-free methods (Q4).}
Finite differences with $\delta = 0.05$ achieve $f = 0.728$ on Benchmark~B (only 0.5\% above $f^{\star} = 0.7244$), confirming gradient fidelity. $\delta = 0.8$ stalls at $f = 1.640$ (127\% above optimum) due to truncation error. Nesterov random search scales to arbitrary dimension at constant 2 FE/step but converges slowly ($f \approx 0.745$ after 220 steps on Bench~B). Most remarkably, Nelder--Mead locates the \textbf{exact} global minimum of Rosenbrock ($f = 0.000$ at $(1,1)$) in 160 iterations using zero gradient information, requiring only $\leq 320$ function evaluations.

\paragraph{Constrained optimisation (Q5).}
Projected GD enforces feasibility exactly at every iteration, converging to the constrained optimum ($f = 0.7244$ at $(0.582, 2.000)$). Penalty methods provide a simpler implementation but require careful tuning of $\lambda$: too small causes residual violations, too large degrades conditioning. The constrained minimum coincides with the unconstrained minimum here because the unconstrained optimum is already feasible --- a scenario where both approaches converge to the same point.

\paragraph{Frank--Wolfe (Q6).}
The Frank--Wolfe algorithm avoids projection by replacing it with a linear minimisation oracle, maintaining feasibility automatically. Its $O(1/k)$ rate is theoretically inferior to accelerated projected methods but practically competitive on polytope-constrained problems. The $z_k$ ``zig-zag'' pattern reveals the gap between the gradient and the Frank--Wolfe direction; this gap certificate is a useful convergence diagnostic.

\paragraph{Overall observations.}
\begin{itemize}[nosep]
  \item Condition number is the dominant difficulty factor. Methods that adapt to curvature (Newton, Nesterov, Adam) consistently outperform fixed-step GD on ill-conditioned problems, as illustrated in Figure~\ref{fig:gd_rate_analysis}.
  \item Theoretical convergence rates predict observed behaviour well: $O(1/k^2)$ for Nesterov beats $O(1/k)$ for GD by an increasing margin as iteration count grows (verified in Figure~\ref{fig:q2_rates}).
  \item Hyperparameter sensitivity varies dramatically: Polyak requires exact $f^{\star}$ (most sensitive); Adam is relatively robust with defaults $\beta_1=0.9, \beta_2=0.999$; Adagrad is robust but suffers from monotonic step decay.
  \item Computational cost per iteration must be weighed against iterations required: Newton uses 20 iterations vs.\ GD's 80 but at higher per-iteration cost $O(n^3)$ vs.\ $O(n)$.
  \item For constrained problems: projected GD provides hard feasibility guarantees; penalty methods offer simpler implementation at the cost of conditioning; Frank--Wolfe provides projection-free convergence for polytope constraints.
\end{itemize}

\paragraph{Practical method selection guide.}
\begin{itemize}[nosep]
  \item \textbf{Unconstrained, gradient available, small $n$ ($n \lesssim 10^3$), well-conditioned ($\kappa \lesssim 100$):} Use L-BFGS (quasi-Newton) or Nesterov. Both achieve superlinear or $O(1/k^2)$ convergence with modest hyperparameter tuning.
  \item \textbf{Unconstrained, gradient available, large $n$ ($n \sim 10^6$--$10^9$), deep learning:} Use Adam (robust defaults) or SGD with momentum. Adam's per-coordinate scaling handles the diverse gradient magnitudes typical in neural networks.
  \item \textbf{Unconstrained, gradient unavailable, $n$ small:} Use Nelder--Mead (2D results here confirm it finds the exact global minimum of Rosenbrock in 160 iterations). For black-box evaluation, consider Bayesian optimisation.
  \item \textbf{Constrained, box/polytope constraints:} Use Frank--Wolfe (projection-free, sparse solutions) or projected GD (hard feasibility from iteration 1). For general convex constraints, projected GD or augmented Lagrangian.
  \item \textbf{Strongly convex, $f^{\star}$ known:} Polyak step provides theoretical optimal step without line search, but is sensitive to errors in $f^{\star}$.
  \item \textbf{High ill-conditioning ($\kappa \gtrsim 1000$) with gradient available:} Second-order information (Newton, L-BFGS) or preconditioning are essential; first-order methods require $O(\kappa)$ more iterations than second-order methods.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
