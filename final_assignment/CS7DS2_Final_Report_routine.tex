\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems of increasing difficulty: a linear regression quadratic loss (Benchmark~A, $\kappa \approx 1.12$), a near-quadratic toy neural network loss with sinusoidal perturbation (Benchmark~B, $\kappa \approx 3.5$--$6.9$), and the ill-conditioned Rosenbrock function (Benchmark~C, $\kappa \approx 2500$). Six categories of methods are investigated across six questions: (1)~adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), (2)~momentum-based methods (Nesterov acceleration, Adam) and stochastic gradient descent, (3)~Newton's method with Hessian-based local approximations, (4)~derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead, grid search), (5)~constrained optimisation via projected gradient descent and penalty methods, and (6)~the Frank--Wolfe algorithm for linear-constrained problems. Key findings include: the Polyak step size achieves the best Rosenbrock convergence ($f = 0.009$) when $f^{\star}$ is known exactly but diverges catastrophically when $f^{\star}$ is misspecified; Newton's method converges in a single step on the quadratic benchmark and outperforms all first-order methods with far fewer iterations; Nesterov momentum ($f = 0.479$) significantly outperforms standard gradient descent ($f = 3.39$) on the ill-conditioned Rosenbrock; and projected gradient descent enforces hard constraint satisfaction from the first iteration while penalty methods only approximate feasibility. Each method is evaluated in terms of convergence speed, trajectory behaviour, stability, and computational trade-offs, with all theoretical claims connected to course material.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \in \mathbb{R}^2, \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
The Hessian is constant: $H = \frac{1}{m}X^T X$, with Lipschitz smoothness constant $L = \lambda_{\max}(H)$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. The condition number $\kappa = \lambda_{\max}/\lambda_{\min} \approx 1.12$ (computed from the data), meaning the level sets are nearly circular. The OLS minimum $\hat{\theta}_{\text{OLS}} \approx (3.03, 4.02)^T$ and noise floor $J^{\star} \approx 0.4826$ due to irreducible observation noise. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H = \text{diag}(2 - \sin(x_1),\ 10)$. The diagonal structure means the two coordinates decouple. The minimum eigenvalue $2 - \sin(x_1)$ varies from $1$ to $3$, giving a condition number $\kappa \in [10/3,\ 10/1] = [3.3,\ 10]$ (tightest near $\sin(x_1) = 1$). The global minimum is at approximately $x^{\star} \approx (0.582,\ 2.0)$ with $f^{\star} \approx 0.7244$, where the first component satisfies $2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$ and the second gives $x_2^{\star} = 2$ exactly. Starting point: $x_0 = [-1, 4]^T$, distance $\|x_0 - x^{\star}\| \approx 2.4$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum and starting point $(-1, 1)$ is:
\begin{equation}
H = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix},
\end{equation}
which at $x_0 = (-1, 1)$ gives eigenvalues $\lambda_{\max} \approx 1002$ and $\lambda_{\min} \approx 0.4$, yielding $\kappa \approx 2500$. This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes the Rosenbrock function one of the most challenging standard test functions for gradient-based methods. Starting point: $x_0 = [-1, 1]^T$, distance $\|x_0 - x^{\star}\| = 2$.

The three benchmarks form a deliberately increasing hierarchy of difficulty: Benchmark~A has nearly circular level sets, Benchmark~B introduces moderate anisotropy and mild nonlinearity, and Benchmark~C tests methods against the pathological banana-shaped valley and near-singular Hessian. Figure~\ref{fig:q_condition_number} illustrates how the condition number directly determines the GD convergence rate.

For a quadratic function $f(x) = \frac{1}{2}x^T H x - b^T x$ with $\mu I \preceq H \preceq LI$ and $\kappa = L/\mu$, the optimal constant GD step $\alpha = 2/(L+\mu)$ achieves the tight convergence bound:
\begin{equation}
\|x_k - x^{\star}\|_H^2 \leq \left(\frac{\kappa - 1}{\kappa + 1}\right)^{2k} \|x_0 - x^{\star}\|_H^2.
\end{equation}
Here $\|v\|_H = \sqrt{v^T H v}$ is the energy norm. For $\kappa = 2500$ (Benchmark~C), this contraction factor is $(2499/2501)^2 \approx 0.99840$, requiring $\approx 4330$ iterations to reduce the error by $10\times$ --- consistent with the observed slow convergence in Figure~\ref{fig:q_condition_number}.

\subsection{Theoretical Background}
\label{sec:q1_theory}

For an $L$-smooth, $\mu$-strongly convex function $f$ (i.e.\ $\mu I \preceq \nabla^2 f(x) \preceq L I$), the standard gradient descent iterate $x_{k+1} = x_k - \alpha \nabla f(x_k)$ satisfies:
\begin{theorem}[GD Linear Convergence]
\label{thm:gd}
With step size $\alpha = 1/L$:
\begin{equation}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k (f(x_0) - f^{\star}).
\end{equation}
\end{theorem}
The contraction factor $(1 - \mu/L) = (1 - 1/\kappa)$ shows that well-conditioned problems ($\kappa$ small) converge quickly while ill-conditioned ones ($\kappa$ large) converge slowly. This motivates the adaptive methods studied in this section.

The four adaptive methods below either estimate or exploit curvature information to overcome the limitation of a fixed step size.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step uniquely adapts to the current sub-optimality: when far from the minimum, $\alpha_k$ is large; as $f(x_k) \to f^{\star}$, $\alpha_k \to 0$, automatically reducing the step without a schedule. Formally:
\begin{theorem}[Polyak Convergence, \cite{polyak1969}]
\label{thm:polyak}
If $f^{\star}$ is known exactly and $f$ is convex, then $\|x_{k+1} - x^{\star}\| \leq \|x_k - x^{\star}\|$ (non-expansive). If additionally $f$ is $L$-smooth and $\mu$-strongly convex, convergence is linear at rate $(1 - \mu/L)$.
\end{theorem}
The critical sensitivity is this: if $f^{\star}$ is underestimated (set too low), then $\alpha_k = (f(x_k) - f^{\star}_{\text{wrong}})/\|g_k\|^2 > (f(x_k) - f^{\star}_{\text{true}})/\|g_k\|^2$ once $x_k$ is near the true minimum, causing the step to grow unboundedly as $\|g_k\| \to 0$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad implements a diagonal preconditioner $P_k = \text{diag}(\sqrt{G_k})^{-1}$. Coordinates with historically large gradients receive smaller effective rates; sparse or rarely-updated coordinates receive larger rates. The monotone accumulation $G_k \nearrow \infty$ ensures the effective step size vanishes, guaranteeing stability at the cost of vanishing updates on long runs. Duchi et al.~\cite{duchi2011} prove that Adagrad achieves regret $O(\sqrt{T})$ in the online convex setting, matching minimax-optimal rates while automatically adapting to feature frequencies.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop~\cite{hinton2012} remedies Adagrad's vanishing step problem by replacing the cumulative sum with an exponential moving average (EMA) with decay $\beta$. The effective memory horizon is $\approx 1/(1-\beta)$ iterations; here $\beta = 0.9$ gives a 10-step window. This allows RMSprop to re-scale its effective step size if the gradient magnitude changes over time, making it suitable for non-stationary optimisation landscapes.

\begin{remark}[RMSprop Convergence Near Minimum]
Near a minimum where $\|\nabla f(x)\| \to 0$, the EMA $v_{k+1} \to 0$, making the effective step $\alpha_0/(\sqrt{v_{k+1}} + \varepsilon) \to \alpha_0/\varepsilon$. For $\alpha_0 = 0.22$ and $\varepsilon = 10^{-5}$, this effective step approaches $22{,}000$, far exceeding the Lipschitz stability bound $2/L$. This explains why RMSprop on Benchmark~A converges to $f = 0.5027 > J^{\star} = 0.4826$: the large effective step near the minimum causes persistent oscillation around the optimum rather than convergence to it. A floor on $v$ (larger $\varepsilon$) or a learning rate schedule would resolve this, but at the cost of reduced adaptivity.
\end{remark}

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method accumulates gradient history via an exponential moving average of gradients. For a quadratic $f$ with Hessian eigenvalues in $[\mu, L]$, the theoretically optimal parameters are $\alpha^{\star} = 4/(\sqrt{L}+\sqrt{\mu})^2$ and $\beta^{\star} = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$, yielding convergence rate $((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$ per iteration --- a factor of $O(\sqrt{\kappa})$ faster than GD's $(1-2/(\kappa+1))$ for large $\kappa$. In practice these optimal parameters are rarely known and must be tuned.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$, 373$\times$ lower than GD's $f = 3.51$ after 120 iterations). Without $f^{\star}$ knowledge, Heavy Ball achieves the next best Rosenbrock result ($f = 1.40$, $2.5\times$ improvement over GD) by exploiting momentum accumulation along the consistent valley direction. Adagrad's coordinate-wise scaling helps moderately ($f = 2.20$, $1.6\times$ improvement).
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging of gradient magnitudes. Heavy Ball shows the characteristic damped oscillation pattern before settling: the momentum accumulates towards the valley but overshoots periodically, producing a series of oscillations with decreasing amplitude --- a signature of under-damped momentum systems ($\beta = 0.86 < \beta^{\star}$).
\item \textbf{Stability:} GD is most stable but slowest (contraction factor $\approx 0.9992$ per step on Rosenbrock). Polyak diverges catastrophically when $f^{\star}$ is misspecified: on Benchmark~A, the step grows to $\alpha_k > 100$ once $\|g_k\| \to 0$, consistent with $\alpha_k = 0.48/\|g_k\|^2 \to \infty$. Adagrad is uniformly stable (effective step $\alpha_0/\sqrt{G_k} \to 0$), ensuring no divergence even with aggressive initial $\alpha_0$, but runs the risk of premature convergence on long horizons.
\item \textbf{Practical implication:} In practice, exact $f^{\star}$ is almost never known, limiting Polyak's applicability to structured problems with known optima (e.g., feasibility problems, where $f^{\star} = 0$). Adagrad is the natural choice for sparse-feature settings (NLP, recommender systems); RMSprop for non-stationary objectives (online learning, deep RL); Heavy Ball for well-conditioned problems with tunable parameters.
\end{itemize}

\subsection{Theoretical Background}

Standard gradient descent achieves the suboptimal rate $O(1/k)$ for convex functions and $O((1-1/\kappa)^k)$ for strongly convex functions. Nesterov~\cite{nesterov1983} showed that these rates can be improved to $O(1/k^2)$ and $O((1-1/\sqrt{\kappa})^k)$ respectively using a ``lookahead'' gradient evaluation, and that these rates are \emph{optimal} among first-order methods --- no deterministic first-order algorithm can do better in general.

\begin{theorem}[Nesterov Acceleration, \cite{nesterov1983}]
\label{thm:nesterov}
For an $L$-smooth convex function, the Nesterov accelerated gradient method with step size $\alpha = 1/L$ and momentum schedule $\beta_k = (k-1)/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\end{equation}
This $O(1/k^2)$ rate is optimal for first-order methods on smooth convex functions.
\end{theorem}

Adam~\cite{kingma2015} combines Nesterov-style momentum with RMSprop-style per-coordinate adaptive scaling, aiming to capture benefits of both approaches.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing schedule, capped at $\beta_{\max}$}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This allows Nesterov to ``look ahead'' in the direction of momentum and correct its trajectory before overshooting, which is the mechanism behind the improved $O(1/k^2)$ rate. The adaptive schedule $\beta_k = (k-1)/(k+2) \to 1$ gradually builds momentum while initially being conservative ($\beta_1 = 0$, so the first step is pure gradient descent). Note: $x_{k+1} = \tilde{x}_k - \alpha \nabla f(\tilde{x}_k)$, i.e., a gradient step from the lookahead point. Figure~\ref{fig:q2_nesterov_sched} shows the $\beta_k$ schedule used in the implementation.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q2_nesterov_schedule.pdf}
\caption{Q2: Nesterov momentum schedule $\beta_k = \min((k-1)/(k+2),\ \beta_{\max})$ vs iteration ($\beta_{\max} = 0.90$). The schedule begins at $\beta_1 = 0$ (pure gradient descent) and increases monotonically, saturating at $\beta_{\max} \approx 30$ iterations. This gradual warm-up is critical: immediately applying high momentum ($\beta \approx 0.9$) would cause instability since the gradient direction has not yet established a consistent trend.}
\label{fig:q2_nesterov_sched}
\end{figure}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (RMSprop)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam's bias correction is essential: at $t=1$, the EMA initialised at zero gives $m_1 = (1-\beta_1)g_1$, which underestimates the true mean. The correction $\hat{m}_1 = m_1/(1-\beta_1) = g_1$ recovers the correct scale. Without bias correction, early iterates would be biased towards zero, slowing initial progress. The effective step size per coordinate is $\alpha/\sqrt{\hat{v}_t + \varepsilon}$, which adapts based on recent gradient history (via $\beta_2 = 0.999$, an $\approx 1000$-step window).

\begin{remark}[Adam Convergence Caveat]
\label{rem:adam_convergence}
Despite its widespread use, Adam does \emph{not} have proven convergence for general smooth convex optimisation. Reddi et al.\ (2018) constructed explicit counterexamples showing Adam can fail to converge even on convex functions with bounded gradients, due to conflicting step-size signals across coordinates when $\beta_2 \to 1$. The fix, \emph{AMSGrad}, replaces $\hat{v}_t$ in the denominator with $\hat{v}_t^{\max} = \max_{s \leq t} \hat{v}_s$, ensuring the effective step is non-increasing and restoring convergence. In practice, Adam works well on deep learning problems (where the objective is highly non-convex anyway) but the theory gap is an important caveat for convex settings.
\end{remark}

\begin{remark}[Why Nesterov Outperforms Adam on Rosenbrock]
Nesterov's advantage on the ill-conditioned Rosenbrock ($\kappa \approx 2500$) over Adam ($f = 0.479$ vs $f = 1.578$) is not immediately obvious since Adam also has momentum. The key difference is that Nesterov's momentum operates in the \emph{original parameter space}, building speed along the consistent valley direction. Adam's per-coordinate scaling, while beneficial for anisotropic separable problems (like Benchmark~B), normalises gradient components independently. On Rosenbrock's curved valley, the off-diagonal curvature ($h_{12} = -400 x_1$ at the starting point) couples the two coordinates, and Adam's diagonal scaling cannot capture this coupling. Furthermore, Adam's $\beta_2 = 0.999$ EMA gives large effective steps in the low-curvature direction, causing oscillation across the narrow valley. Nesterov's lookahead damping is more effective at tracking the curved valley floor.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch. The stochastic gradient is:
\begin{equation}
\hat{g}(\theta; \mathcal{B}) = \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}}),
\end{equation}
which is an unbiased estimate of $\nabla J(\theta)$: $\mathbb{E}[\hat{g}] = \nabla J(\theta)$. The variance is $\text{Var}[\hat{g}] = O(\sigma^2/b)$ where $\sigma^2$ is the variance of the per-sample gradient. Smaller batches have higher variance, leading to noisier updates but performing more parameter updates per epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\centering
\includegraphics[width=0.85\textwidth]{figures/q2_sgd_batch.pdf}
\caption{Q2: Mini-batch SGD with different batch sizes. With $b = 5$: $1000/5 = 200$ updates per epoch, faster per-epoch convergence but higher gradient variance. With $b = 40$: $1000/40 = 25$ updates per epoch, smoother but slower per-epoch progress. The smaller batch reaches lower loss faster in epoch-count due to more frequent parameter updates, but exhibits more oscillation near the minimum because each mini-batch gradient deviates more from the true gradient. \begin{remark}
The noise floor phenomenon is fundamental in stochastic optimisation: SGD cannot converge below $J^{\star}_{\text{true}}$ regardless of batch size or step size, because the noise is irreducible (it comes from the true data-generating process, not the algorithm). This motivates methods with decaying learning rates to \emph{converge} to $J^{\star}$ rather than oscillate around it.
\end{remark}

The formal convergence result under decaying step sizes is:
\begin{theorem}[SGD Convergence, \cite{nocedal2006}]
\label{thm:sgd}
For an $L$-smooth $\mu$-strongly convex $f$ with stochastic gradients satisfying $\mathbb{E}[\hat{g}_k] = \nabla f(x_k)$ and $\mathbb{E}[\|\hat{g}_k - \nabla f(x_k)\|^2] \leq \sigma^2$, mini-batch SGD with step size $\alpha_k = c / (k + 1)$ for $c > 1/(2\mu)$ satisfies:
\begin{equation}
\mathbb{E}[f(x_k)] - f^{\star} \leq \frac{C}{k+1}, \quad C = \max\!\left(\frac{L c^2 \sigma^2}{2\mu c - 1},\ \frac{c(f(x_0)-f^{\star})}{1}\right).
\end{equation}
With fixed step $\alpha_k = \alpha$, convergence stalls at a noise floor: $\liminf_{k\to\infty} \mathbb{E}[f(x_k)] - f^{\star} \leq \frac{\alpha L \sigma^2}{2\mu}$.
\end{theorem}
This confirms the empirical observation: fixed step size SGD converges to a neighbourhood of $f^{\star}$ of size $O(\alpha \sigma^2)$. Reducing batch size $b$ increases $\sigma^2 \propto 1/b$, enlarging the noise floor proportionally. Decaying steps are necessary for exact convergence.

Figure~\ref{fig:q2_sgd_batch_sweep} extends the two-batch comparison by testing six batch sizes from $b=1$ (pure online SGD) to $b=1000$ (full-batch gradient descent), visualising the fundamental batch size--convergence trade-off.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.00391 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.00391 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton update minimises the second-order model: setting $\frac{d}{dx}[\text{quadratic model}] = 0$ gives $x_{k+1} = x_k - g'(x_k)/g''(x_k)$. For $g(x) = x^4$: $x_{k+1} = x_k - (4x_k^3)/(12x_k^2) = x_k - x_k/3 = 2x_k/3$, so starting from $x_0 = 0.25$: $x_1 = 0.167$, $x_2 = 0.111$, etc., converging quickly to $x^{\star} = 0$.

\subsection{Theoretical Background}

\begin{theorem}[Newton Quadratic Convergence]
\label{thm:newton}
Let $f$ be twice continuously differentiable with $\|\nabla^2 f(x) - \nabla^2 f(y)\| \leq L_H\|x-y\|$ (Lipschitz Hessian). If $x_0$ is sufficiently close to a local minimum $x^{\star}$ where $\nabla^2 f(x^{\star}) \succ 0$ (with minimum eigenvalue $\mu > 0$), and the full Newton step $\alpha = 1$ is used, then:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{L_H}{2\mu}\|x_k - x^{\star}\|^2.
\end{equation}
\end{theorem}
This \emph{quadratic convergence} means the number of correct digits doubles at each iteration. Starting with one correct digit, after 10 iterations one has $2^{10} = 1024$ correct digits. In contrast, GD's linear convergence (Theorem~\ref{thm:gd}) requires $O(\kappa \log(1/\varepsilon))$ iterations for $\varepsilon$-accuracy. The practical implication is dramatic: Newton can solve the problem to machine precision in $O(\log \log(1/\varepsilon))$ iterations once in the quadratic convergence regime.

\begin{proposition}[Exact Newton for Quadratics]
\label{prop:newton_quad}
For a strictly convex quadratic $f(\theta) = \frac{1}{2}\theta^T H \theta - b^T\theta + c$, the full Newton step ($\alpha = 1$) converges to the exact minimiser $\theta^{\star} = H^{-1}b$ in \emph{exactly one iteration}, regardless of starting point.
\end{proposition}
\textit{Proof sketch:} $\nabla f(\theta) = H\theta - b$, $\nabla^2 f(\theta) = H$ (constant). Newton step: $\theta_1 = \theta_0 - H^{-1}(H\theta_0 - b) = H^{-1}b = \theta^{\star}$. \hfill$\square$

This explains the observed one-step convergence on Benchmark~A (quadratic loss): the Hessian $H = X^TX/m$ is constant throughout the domain, so the Newton step is exact.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise for numerical stability}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: solve linear system}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$ (for numerical stability, negligible effect). The damped step $\alpha < 1$ is necessary on Benchmark~C because the Rosenbrock Hessian changes rapidly between iterations, violating the local model assumption. The regularisation $\lambda I$ also ensures $H_k \succ 0$ everywhere, preventing the Newton direction from pointing uphill.

\subsection{Limitations and Extensions of Newton's Method}

\begin{remark}[Two-Phase Newton Convergence]
Newton's method on non-quadratic problems exhibits two distinct convergence phases:
\begin{enumerate}[nosep]
\item \textbf{Global phase} (damped Newton): far from $x^{\star}$, the local quadratic model is inaccurate. A damped step $\alpha < 1$ is required, and convergence is linear rather than quadratic. On Rosenbrock, $\alpha = 0.22$ keeps the method stable across the rapidly-changing Hessian landscape.
\item \textbf{Local phase} (pure Newton): once $x_k$ is within the \emph{quadratic convergence basin} (radius $r < 2\mu^2/(L_H)$ from $x^{\star}$), the full step $\alpha = 1$ achieves quadratic convergence as stated in Theorem~\ref{thm:newton}. The transition from the damped to the pure Newton regime is recognised by the Newton decrement $\lambda(x_k) = \sqrt{g_k^T H_k^{-1} g_k}$ falling below $\approx 0.25$.
\end{enumerate}
For our Rosenbrock experiments, the 20 damped-Newton iterations operate primarily in the global phase (the starting point $(-1, 1)$ is far from $x^{\star} = (1,1)$), which is why convergence is superlinear but not yet quadratic.
\end{remark}

Despite its outstanding convergence properties, Newton's method has significant practical limitations:
\begin{enumerate}[nosep]
\item \textbf{Hessian computation cost:} Computing $\nabla^2 f(x) \in \mathbb{R}^{d \times d}$ requires $O(d^2)$ storage and $O(d^3)$ linear solve per iteration. For $d = 10^6$ (typical neural network), this is entirely infeasible.
\item \textbf{Local convergence only:} Theorem~\ref{thm:newton} requires $x_0$ to be in the basin of attraction of $x^{\star}$. On Rosenbrock, the heavily damped step $\alpha = 0.22$ is necessary precisely because the starting point is far from the quadratic convergence regime.
\item \textbf{Non-convex problems:} The Hessian may be indefinite away from local minima, causing the Newton direction to point uphill. Regularisation ($\lambda I$ addition) or trust-region methods are needed.
\end{enumerate}

These limitations motivate \emph{quasi-Newton methods} such as L-BFGS (Limited-memory Broyden--Fletcher--Goldfarb--Shanno), which approximate $H^{-1}$ using a low-rank update from gradient differences across the last $m$ iterates. L-BFGS achieves superlinear convergence while requiring only $O(md)$ storage, making it the method of choice for medium-scale smooth unconstrained problems ($d \leq 10^5$).

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference (FD) approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}. \tag{FD1}
\end{equation}
The approximation error decomposes into two competing terms:
\begin{equation}
\underbrace{\frac{f(x+\delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)}_{\text{total error}} = \underbrace{\frac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x) + O(\delta^2)}_{\text{truncation: }O(\delta)} + \underbrace{O\!\left(\frac{\varepsilon_{\text{mach}}}{\delta}\right)}_{\text{roundoff}},
\end{equation}
where $\varepsilon_{\text{mach}} \approx 2.2 \times 10^{-16}$ is machine epsilon. The total error is minimised at $\delta^{\star} = O(\varepsilon_{\text{mach}}^{1/2}) \approx 1.5 \times 10^{-8}$. The \emph{central difference} formula has a better truncation error of $O(\delta^2)$:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}, \quad \text{error} = O(\delta^2) + O(\varepsilon_{\text{mach}}/\delta), \tag{CD2}
\end{equation}
with optimal $\delta^{\star}_{\text{CD}} = O(\varepsilon_{\text{mach}}^{1/3}) \approx 6 \times 10^{-6}$ and minimum error $O(\varepsilon_{\text{mach}}^{2/3})$, substantially lower than forward differences' $O(\varepsilon_{\text{mach}}^{1/2})$. The cost is $2d$ function evaluations per gradient (vs $d+1$ for forward FD). Figure~\ref{fig:q4_fd_error} visualises this U-shaped error curve for the forward difference used in the assignment.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Method)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Uniform on $\mathbb{S}^{d-1}$}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The estimator $\hat{g}_k$ is a noisy directional derivative. Its expected value satisfies:
\begin{equation}
\mathbb{E}[\hat{g}_k] \approx \mathbb{E}\!\left[\nabla f(x_k)^T u_k \cdot u_k\right] = E[u_k u_k^T]\, \nabla f(x_k) = \tfrac{1}{d}\, \nabla f(x_k),
\end{equation}
using $\mathbb{E}[u u^T] = (1/d)I$ for uniform random unit vectors. In $d = 2$ dimensions, the expected update is $-({\alpha}/{2})\nabla f(x_k)$, effectively a gradient step with half the step size but high variance. The method requires only \emph{two function evaluations} per iteration regardless of $d$, making it uniquely scalable: for high-dimensional problems where $d \gg 1$, the FD cost of $d+1$ evaluations per iteration becomes prohibitive, while NRS remains constant.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm~\cite{neldermead1965} maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively deforms it through four operations:
\begin{itemize}[nosep]
\item \textbf{Reflection} (always attempted): move worst vertex through centroid of remaining vertices: $x_r = c + \alpha_r(c - x_{\text{worst}})$ with $\alpha_r = 1$.
\item \textbf{Expansion} (if reflection improves): try $x_e = c + \gamma(c - x_{\text{worst}})$, $\gamma = 2$.
\item \textbf{Contraction} (if reflection worsens): try $x_c = c + \rho(x_r - c)$ or $c + \rho(x_{\text{worst}} - c)$ with $\rho = 0.5$.
\item \textbf{Shrink} (if contraction fails): contract all vertices towards the best: $x_i \leftarrow x_{\text{best}} + \sigma(x_i - x_{\text{best}})$, $\sigma = 0.5$.
\end{itemize}
The simplex adapts its shape to the local landscape: elongating along valleys and contracting near the minimum. Settings: initial simplex step $= 0.35$, iterations $= 160$.

\subsection{Extended Analysis: Nesterov Random Search Convergence}

The high variance of NRS has a precise characterisation. In $d$ dimensions, the expected update direction is $E[\hat{g}] = (1/d)\nabla f(x)$ (see Section~\ref{sec:q4} preamble), so NRS is equivalent to gradient descent with \emph{effective step size} $\alpha_{\text{eff}} = \alpha/d$ plus gradient noise. The variance of the estimator scales as $\text{Var}[\hat{g}] = O(\|\nabla f\|^2/d)$, meaning that in high dimensions the estimator has large relative noise (signal-to-noise ratio $\propto 1/\sqrt{d}$). For $d = 2$, NRS has a reasonable effective step ($\alpha_{\text{eff}} = \alpha/2$) but still requires roughly $2\times$ more iterations than exact GD at the same $\alpha$, consistent with the observed slower convergence in Figure~\ref{fig:q4_fval_B}.

The key advantage of NRS is its \emph{dimension-free oracle complexity}: achieving $\varepsilon$-accuracy requires $O(1/\varepsilon^2)$ function evaluations regardless of $d$, whereas FD GD requires $O(d/\varepsilon^2)$ evaluations. In $d = 10^5$ (typical large-scale ML problem), NRS has a $10^5\times$ advantage over FD gradient estimation per $\varepsilon$-accuracy target.

\subsection{Nelder--Mead: Convergence Properties and Limitations}

The Nelder--Mead simplex method has several notable theoretical properties:
\begin{itemize}[nosep]
\item \textbf{No convergence guarantee in general}: Lagarias et al.~\cite{lagariasnm1998} proved that Nelder--Mead converges in 1D but showed it can cycle or fail to converge in 2D on certain smooth functions. McKinnon (1999) constructed an explicit 2D counterexample. However, on smooth functions with well-separated local minima it performs well empirically.
\item \textbf{Simplex degeneracy}: The simplex can become degenerate (nearly flat), causing the algorithm to make only tiny steps. The shrink operation recovers from this, but may lose directional diversity.
\item \textbf{Performance vs function evaluation count}: On Rosenbrock, Nelder--Mead with 160 iterations (at most 3 FE each $= \leq 480$ total FE) navigates the valley effectively. Exact GD in 80 iterations uses $80 + 1$ gradient evaluations. Thus Nelder--Mead achieves comparable progress with 6$\times$ more function evaluations but no derivative computation.
\item \textbf{Scaling to higher dimensions}: For $d > 5$, the $n+1$ simplex becomes geometrically complex and the method degrades significantly. Methods like Bayesian optimisation or evolutionary strategies are preferred instead.
\end{itemize}

\subsection{Problem Setup}

Minimise Benchmark~B subject to the constraint $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$:
\begin{equation}
\min_{x \in \mathbb{R}^2} f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad \text{subject to} \quad x_1 \geq 0.5.
\end{equation}

\textbf{KKT Conditions:} The Lagrangian is $\mathcal{L}(x, \mu) = f(x) - \mu(x_1 - 0.5)$ with $\mu \geq 0$. The KKT conditions at a constrained minimum are:
\begin{align}
\nabla_x \mathcal{L} = 0: &\quad \nabla f(x^{\star}) = \mu^{\star} e_1, \text{ i.e., } 2(x_1^{\star}-1) + \cos(x_1^{\star}) = \mu^{\star}, \quad 10(x_2^{\star}-2) = 0 \\
\text{Primal feasibility}: &\quad x_1^{\star} \geq 0.5 \\
\text{Dual feasibility}: &\quad \mu^{\star} \geq 0 \\
\text{Complementary slackness}: &\quad \mu^{\star}(x_1^{\star} - 0.5) = 0.
\end{align}
The unconstrained minimum satisfies $2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$. Numerically, $x_1^{\star} \approx 0.5816$ (root found by Newton's method on the gradient) and $x_2^{\star} = 2.0$ from the second KKT equation. Since $x_1^{\star} \approx 0.5816 > 0.5$, the constraint $x_1 \geq 0.5$ is \emph{inactive}: complementary slackness gives $\mu^{\star}(x_1^{\star} - 0.5) = 0 \Rightarrow \mu^{\star} = 0$ (since the gap $0.0816 \neq 0$). Therefore the constrained and unconstrained optima coincide at $(0.5816, 2.0)$ with $f^{\star} = (0.5816-1)^2 + 5(2-2)^2 + \sin(0.5816) = 0.1751 + 0 + 0.5494 = 0.7245 \approx 0.7244$.

\textbf{Optimality verification:} At $x^{\star} = (0.5816, 2.0)$: $\nabla f = (2(0.5816-1)+\cos(0.5816),\ 0) = (2(-0.4184) + 0.8352,\ 0) = (0,\ 0)$ confirming stationarity. Projected GD should recover this; penalty methods approximate it with accuracy $O(1/\lambda)$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space constraint $x_1 \geq 0.5$, the projection is $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)$, a trivial $O(1)$ operation. More complex constraints (e.g., second-order cone, semidefinite) require more expensive projections, which can dominate the cost of each iteration.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, -x_1 + 0.5)^1$ \Comment{$\ell_1$ exterior penalty}
    \State $\nabla F_\lambda(x) = \nabla f(x) - \lambda \mathbf{1}[x_1 < 0.5]\, e_1$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty $\lambda \max(0, -x_1 + 0.5)$ adds a cost proportional to the constraint violation. As $\lambda \to \infty$, the penalised minimiser $x^{\star}_\lambda \to x^{\star}$, but large $\lambda$ introduces steep gradients near the boundary that require smaller step sizes for stability. Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected GD} enforces feasibility as a hard constraint at every iteration: the projection operator guarantees $x_k \in \mathcal{X}$ for all $k \geq 1$. This is the fundamental advantage. The cost is requiring a projection oracle; for simple box or half-space constraints this is trivial ($O(d)$), but for constraints like the nuclear norm ball or the positive semidefinite cone, projection requires $O(d^3)$ SVD or eigendecomposition operations.

\textbf{Penalty methods} trade exact feasibility for simplicity: no projection oracle is needed, only a modified gradient. The penalty weight $\lambda$ controls the accuracy--stability trade-off. For the $\ell_1$ penalty used here, large $\lambda$ enforces feasibility but the non-smooth gradient at $x_1 = 0.5$ can cause oscillation; this is why a smaller step ($\alpha = 0.03$) is needed for $\lambda = 4.5$. An \emph{augmented Lagrangian} method would improve on this by also updating the dual variable: $\mu^{k+1} = \max(0, \mu^k - \lambda \cdot \text{violation}_k)$, which avoids the need for $\lambda \to \infty$.

The \emph{augmented Lagrangian} (method of multipliers) addresses this by adapting the dual variable $\mu$: starting from a modest $\lambda$, successive outer iterations increase $\lambda$ by a factor and update $\mu^{k+1} = \max(0, \mu^k - \lambda \cdot v_k)$ where $v_k = \max(0, 0.5 - x_1^k)$ is the violation. This systematic adaptation eliminates the need to choose $\lambda$ large enough from the outset.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box / polytope).

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is obtained by minimising each coordinate independently. Since both components of $a$ are positive, the minimum is at the lower bounds:
\begin{equation}
x^{\star} = \arg\min_{x \in \mathcal{X}} x_1 + 2x_2 = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}

\begin{proposition}[LP Vertex Optimality]
The minimum of a linear objective $a^Tx$ over a bounded polytope $\mathcal{X}$ is always attained at a vertex of $\mathcal{X}$. For a box $[l_i, u_i]^d$, the optimal vertex is $x_i^{\star} = l_i$ if $a_i > 0$ and $x_i^{\star} = u_i$ if $a_i < 0$.
\end{proposition}

This is the fundamental theorem of linear programming: the objective is constant along the normal direction to $a$, and any movement along the feasible set in the direction $-a$ (gradient descent on a linear function) will always reach the boundary of the feasible set.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination step towards LMO solution}
\EndFor
\end{algorithmic}
\end{algorithm}

The Frank--Wolfe method~\cite{frankwolfe1956} (also called the conditional gradient method) replaces the projection step of projected GD with a linear minimisation oracle (LMO): instead of projecting a gradient step back onto $\mathcal{X}$, one finds the vertex of $\mathcal{X}$ in the direction of steepest descent. Key properties:
\begin{itemize}[nosep]
\item \textbf{Projection-free:} All iterates $x_k$ are automatically in $\mathcal{X}$ (as convex combinations of feasible points), without any projection computation.
\item \textbf{LMO simplicity:} For box constraints, the LMO decomposes per coordinate: $(z_k)_i = l_i$ if $(\nabla f(x_k))_i > 0$, else $u_i$. This is $O(d)$.
\item \textbf{Sparse iterates:} The iterates are convex combinations of vertices, promoting structured (e.g., sparse or low-rank) solutions.
\end{itemize}

\begin{theorem}[Frank--Wolfe Convergence, \cite{frankwolfe1956}]
\label{thm:fw}
For an $L$-smooth convex function $f$ on a bounded convex set $\mathcal{X}$ with diameter $D = \max_{x,y \in \mathcal{X}}\|x - y\|$, the Frank--Wolfe method with step size $\gamma_k = 2/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L D^2}{k + 2} = O(1/k).
\end{equation}
\end{theorem}
The $O(1/k)$ rate is inferior to projected gradient descent's $O(1/k^2)$ for smooth convex problems, but Frank--Wolfe's projection-free property makes it competitive when projections are expensive (e.g., nuclear norm, simplex constraints).

\begin{remark}[Linear Convergence under Strong Convexity]
When $f$ is additionally $\mu$-strongly convex and $\mathcal{X}$ is a polytope, Frank--Wolfe achieves \emph{linear} convergence $f(x_k) - f^{\star} \leq C \cdot \rho^k$ where $\rho < 1$ depends on $\mu$, $L$, and the \emph{pyramidal width} of $\mathcal{X}$. Intuitively, strong convexity ensures that the optimal face of $\mathcal{X}$ is identified in finitely many steps, after which convergence becomes linear. For the box constraint $\mathcal{X}$ in this assignment, the boundary optimum case illustrates this: once $z_k$ consistently selects the correct boundary vertex, the method converges geometrically towards the boundary minimum.
\end{remark}

The \emph{Frank--Wolfe gap} $g_k = \nabla f(x_k)^T(x_k - z_k)$ is an upper bound on $f(x_k) - f^{\star}$ and serves as a practical convergence certificate and stopping criterion.

For box constraints, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}

Note: The implementation here uses a fixed mixing parameter $\beta$ (step $1-\beta$ towards $z_k$) rather than the theoretically optimal decaying schedule $\gamma_k = 2/(k+2)$. Fixed $\beta$ corresponds to a constant step size and does not satisfy Theorem~\ref{thm:fw}'s hypotheses, but is used here to explore the effect of step size on convergence.

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits the characteristic ``zig-zag'' pattern near interior optima: $z_k$ jumps between extreme vertices of $\mathcal{X}$ at each iteration as the gradient direction oscillates. The convex combination gradually averages these jumps. The theoretical $O(1/k)$ convergence rate (Theorem~\ref{thm:fw}) is suboptimal for smooth strongly convex problems compared to projected GD's $O(1/k^2)$, but Frank--Wolfe achieves this without any projection. Smaller $\beta$ (larger step towards $z_k$) can be faster initially but more oscillatory near the optimum.

\textbf{Boundary optimum:} Frank--Wolfe is naturally suited to boundary optima. When the true minimum is on the boundary, $z_k$ frequently selects the correct boundary vertex, and the iterate converges towards it. With the fixed-$\beta$ schedule used here, the residual oscillation prevents exact convergence; the standard decaying schedule $\gamma_k = 2/(k+2)$ would improve this to match Theorem~\ref{thm:fw}'s guarantee.

\textbf{Frank--Wolfe duality gap:} A key advantage of Frank--Wolfe is that it provides a computable certificate of near-optimality at each iteration. Define the \emph{Frank--Wolfe gap}:
\begin{equation}
g_k := \nabla f(x_k)^T (x_k - z_k) \geq 0.
\end{equation}
Since $z_k = \arg\min_{s \in \mathcal{X}} \nabla f(x_k)^T s$, by convexity: $f(x_k) - f^{\star} \leq \nabla f(x_k)^T(x_k - x^{\star}) \leq \nabla f(x_k)^T(x_k - z_k) = g_k$. Thus $g_k$ is an upper bound on suboptimality that can be evaluated without knowing $f^{\star}$, making it a practical stopping criterion. Theorem~\ref{thm:fw} can be rephrased as: the FW gap satisfies $\min_{j \leq k} g_j \leq 2L D^2/(k+2)$, giving a concrete bound on when the gap certificate is tight.

For the interior optimum case ($\beta = 0.985$, $f = 0.0046$), the FW gap at termination is $g_{140} = \nabla f(x_{140})^T(x_{140} - z_{140})$. With $\nabla f \approx (2(0.997-1), 2(5.067-5)) = (-0.006, 0.134)$ and $z_{140}$ on a boundary vertex, the small gradient magnitude confirms near-optimality. For the boundary case, $g_k = \nabla f(x_k)^T(x_k - z_k)$ with $\nabla f(x_k) = (2(x_1-0.5), 2x_2)$, which decreases as $x_k \to (0.5, 0)$ and the vertex $z_k$ increasingly agrees with the gradient's preferred direction.

\textbf{Comparison with projected GD:} Projected GD would converge faster ($O(1/k^2)$ or linear for strongly convex) but requires the projection $\Pi_{\mathcal{X}}$. For the box constraint $\mathcal{X}$ here, this is $O(d)$, so Frank--Wolfe offers no advantage. Frank--Wolfe's benefit becomes apparent when projection is expensive, such as for the nuclear norm ball (projection requires full SVD, $O(nd^2)$), the matrix spectrahedron, or the flow polytope.

\section{Conclusions}
\label{sec:conclusion}
% ============================================================

This report has investigated six categories of optimisation algorithms across three benchmark functions of increasing difficulty. The following conclusions summarise the key findings and practical implications.

\subsection{Adaptive Step-Size Methods (Q1)}
The Polyak step size is theoretically optimal when $f^{\star}$ is known exactly (best performance on Rosenbrock: $f = 0.009$) but is highly fragile to $f^{\star}$ misspecification, diverging catastrophically on Benchmarks A and B. In practice, $f^{\star}$ is almost never known, limiting Polyak's applicability. Adagrad provides a reliable adaptive method with guaranteed stability (monotonically decreasing step) and is preferred for sparse-gradient problems. RMSprop improves Adagrad's long-horizon behaviour via exponential forgetting. Heavy Ball achieves theoretical speedups of $O(\sqrt{\kappa})$ over GD on quadratics when optimal parameters are used.

\subsection{Momentum and Stochastic Methods (Q2)}
Nesterov acceleration achieves the theoretically optimal $O(1/k^2)$ convergence rate for smooth convex problems, demonstrating a decisive advantage on the ill-conditioned Rosenbrock ($f = 0.479$ vs GD's $f = 3.39$). The lookahead mechanism prevents the oscillation that afflicts Heavy Ball on non-quadratic functions. Adam combines momentum with per-coordinate scaling, performing well on anisotropic benchmarks but showing less advantage than Nesterov on the Rosenbrock's challenging nonlinear curvature. SGD's noise floor is determined by observation noise variance ($J^{\star} \propto \sigma^2$), with batch size controlling the variance of gradient estimates around this floor.

\subsection{Newton's Method (Q3)}
Newton's method demonstrates quadratic convergence (Theorem~\ref{thm:newton}) and, crucially, single-step exact convergence on quadratic functions (Proposition~\ref{prop:newton_quad}). This translates to a $4\times$ reduction in iteration count with dramatically better accuracy on all benchmarks. The cost is computing and inverting the $d \times d$ Hessian at each iteration ($O(d^2)$ gradient + $O(d^3)$ solve), which is prohibitive for high-dimensional problems but decisive for the 2D benchmarks studied. Damping ($\alpha < 1$) is essential for non-quadratic problems (especially Rosenbrock) where the local model assumption breaks down.

\subsection{Derivative-Free Methods (Q4)}
The FD gradient approximation closely matches exact GD when $\delta = 0.05$ (within the truncation-dominated regime near the optimal $\delta^{\star} \approx 1.5 \times 10^{-8}$), confirming the error analysis in Eq.~(13). With $\delta = 0.8$, the gradient is corrupted by $O(\delta) = O(1)$ error, severely degrading convergence. Nesterov random search requires only 2 function evaluations per iteration and scales to arbitrarily high dimensions at the cost of high variance. Grid search guarantees global optimum discovery within grid resolution but suffers from the curse of dimensionality: cost scales as $N^d$.

\subsection{Constrained Optimisation (Q5)}
Projected GD enforces hard constraint feasibility from iteration 1, recovering the constrained minimum exactly. Penalty methods offer a gradient-only alternative at the cost of approximate feasibility, with the accuracy--stability trade-off controlled by $\lambda$. For the simple half-space constraint studied, projection is trivial and projected GD is clearly preferable. For complex constraint sets, the augmented Lagrangian method improves on the penalty method by also updating the dual multiplier, avoiding the need for $\lambda \to \infty$.

\subsection{Frank--Wolfe Algorithm (Q6)}
The Frank--Wolfe method provides a projection-free constrained optimisation approach with $O(1/k)$ convergence for smooth convex functions (Theorem~\ref{thm:fw}). Its natural advantage lies in settings where projection is expensive and the constraint set has cheap linear minimisation oracles (e.g., nuclear norm, flow polytopes). The characteristic zig-zag pattern near interior optima reflects the oscillation of the LP vertex $z_k$ as the gradient direction alternates; a decaying step schedule $\gamma_k = 2/(k+2)$ mitigates this.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z      # gradient evaluated at lookahead
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z                       # equivalent to: x = lookahead - alpha*g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)      # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))   # regularise for stability
        try:
            p = np.linalg.solve(H_reg, g)      # Newton direction
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # Linear minimisation oracle: select boundary vertex
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z    # convex combination
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
