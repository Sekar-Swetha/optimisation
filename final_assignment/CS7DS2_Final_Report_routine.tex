\begin{abstract}
This report presents a comprehensive comparative study of first-order and higher-order
optimisation algorithms applied to three benchmark problems spanning different levels of
difficulty: a linear regression quadratic loss (Benchmark~A, convex quadratic), a toy neural
network loss with sinusoidal perturbation (Benchmark~B, smooth non-quadratic), and the
Rosenbrock function (Benchmark~C, highly ill-conditioned non-convex valley).  Six
categories of methods are investigated across six questions: \textbf{(Q1)} adaptive
step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball / Polyak Momentum);
\textbf{(Q2)} momentum-based and stochastic methods (Nesterov acceleration, Adam,
mini-batch SGD); \textbf{(Q3)} second-order methods (Newton's method with Hessian
damping, motivated by local Taylor approximations); \textbf{(Q4)} derivative-free
approaches (forward finite differences, Nesterov random search, Nelder--Mead simplex,
grid search); \textbf{(Q5)} constrained optimisation via projected gradient descent and
penalty methods; and \textbf{(Q6)} the Frank--Wolfe conditional gradient algorithm for
linear-constrained problems together with a linear programme.

Across all experiments, key quantitative findings are: Newton's method reaches the
Rosenbrock minimum an order of magnitude closer than GD in $\tfrac{1}{4}$ the iterations
($f = 0.686$ in 20 steps vs.\ $f = 3.73$ in 80 steps); Nesterov acceleration achieves
$f = 0.479$ on Rosenbrock versus $f = 3.39$ for constant-step GD at the same iteration
budget; the Polyak step size diverges when $f^{\star}$ is misspecified (Benchmarks A and
B) but converges fastest when exact ($f = 0.009$ on Rosenbrock); and projected gradient
descent enforces constraint feasibility from the very first iteration while penalty methods
require $\lambda \geq 4.5$ for comparable speed.  Each method is evaluated in terms of
convergence speed, trajectory behaviour, stability, and computational trade-offs.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) to ensure
reproducibility across all methods and questions.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
  \theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \in \mathbb{R}^2, \quad
  y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad
  \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
  J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2.
\end{equation}
This is a \emph{strictly convex quadratic} whose Hessian $H = \frac{1}{m}X^T X$ is
constant and positive definite.  Its gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$,
which is globally Lipschitz with constant $L = \lambda_{\max}(H)$.  Because of the
observation noise $\varepsilon^{(i)}$, the true minimum is not zero but rather
$J^{\star} \approx 0.4826$ (the residual sum of squares divided by $2m$), with the
minimiser $\theta^{\star}_{\mathrm{OLS}} \approx (3.03, 4.02)$.  Starting point:
$\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = \bigl[2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)\bigr]^T$ and
the Hessian is $H = \operatorname{diag}(2 - \sin(x_1),\ 10)$.  The Hessian eigenvalues
lie in the ranges $[1, 3]$ and $[10, 10]$, giving a condition number between $\kappa
\approx 3.3$ and $\kappa \approx 10$.  The minimum satisfies $2(x_1^{\star}-1)+\cos
(x_1^{\star})=0$ and $x_2^{\star}=2$, yielding $x^{\star}\approx(0.582,\,2.000)$ with
$f^{\star}\approx 0.7244$.  Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
  f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$.  The Hessian at the optimum has
eigenvalues $\approx 2$ and $\approx 400$, giving condition number $\kappa \approx 200$
for the local quadratic; far from the optimum the effective condition number can exceed
$2000$.  This extreme ill-conditioning, combined with the narrow curved valley along
$x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions for
gradient-based methods.  Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems (Questions 5 and 6):
\begin{equation}
  \mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a compact box (bounded polytope) with 4 vertices, making it amenable to the
Frank--Wolfe linear minimisation oracle.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

All gradient-based methods solve $\min_{x} f(x)$ via iterates of the form
$x_{k+1} = x_k - \alpha_k g_k$ where $g_k = \nabla f(x_k)$.  The choice of step size
$\alpha_k$ critically determines convergence.  A constant step size $\alpha < 2/L$
guarantees convergence for $L$-smooth convex functions at rate $O(1/k)$.  However,
adaptive rules aim to achieve faster or more automatic behaviour without manual tuning.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline
across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is theoretically motivated by the subgradient method convergence
analysis: the exact step $\alpha_k = (f(x_k)-f^{\star})/\|\nabla f(x_k)\|^2$ minimises
the distance $\|x_{k+1} - x^{\star}\|^2$ over all steps of the form $x_{k+1}=x_k - \alpha g_k$,
provided $f$ is convex and $f^{\star}$ is known exactly.  When $f^{\star}$ is
misspecified as $\tilde{f} < f^{\star}_{\mathrm{true}}$, the numerator $f(x_k) -
\tilde{f}$ does not vanish at the true minimiser, causing $\alpha_k \to \infty$ as
$\|\nabla f(x_k)\| \to 0$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate accumulation of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad assigns a smaller effective step size to coordinates with historically large
gradients, promoting equitable learning across dimensions.  Its key property is that the
effective step $\alpha_0/\sqrt{\sum_{j=0}^{k} g_j^2}$ is \emph{monotonically decreasing}:
as gradients accumulate, every coordinate slows down.  This is beneficial early in training
but can cause premature stagnation on non-stationary objectives.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's accumulation problem by replacing the sum with an exponentially
weighted moving average.  The effective step stabilises at a scale inversely proportional
to the recent gradient magnitude, allowing the algorithm to adapt to different curvature
scales without the step decaying to zero.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method adds a memory term $\beta z_k$ that accumulates gradient history,
allowing the iterate to build speed in consistent descent directions.  For strongly convex
quadratics with condition number $\kappa$, the optimal parameters $\alpha^* =
4/(\sqrt{L}+\sqrt{\mu})^2$ and $\beta^* = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$
yield a convergence rate of $O(\exp(-k/\sqrt{\kappa}))$, which is substantially faster
than GD's $O(\exp(-k/\kappa))$.

\subsection{Comparative Analysis and Interpretation}

\begin{itemize}[nosep]
\item \textbf{Polyak Step Size:} Theoretically optimal when $f^{\star}$ is exact.  Its
  adaptive step $\alpha_k = (f(x_k)-f^{\star})/\|g_k\|^2$ is largest when the iterate
  is far from the minimum (large function gap) and small when close (small gradient).
  However, even small misspecification of $f^{\star}$ is catastrophic: on Benchmarks~A
  and~B the method diverges completely.  In practice, $f^{\star}$ is almost never known
  exactly for non-trivial problems, severely limiting Polyak's applicability.

\item \textbf{Adagrad:} Uniform convergence across all benchmarks where $f^{\star}$ is
  reachable.  Its per-coordinate scaling effectively handles moderate anisotropy (Benchmark~B)
  by automatically reducing the step for high-curvature directions.  The monotone decay
  prevents divergence but also prevents the algorithm from recovering from suboptimal
  early choices, which is why its Rosenbrock performance ($f=2.20$) is mediocre --- the
  step decays too quickly to traverse the long valley.

\item \textbf{RMSprop:} The exponential forgetting factor ($\beta=0.9$) remedies
  Adagrad's step collapse, maintaining a non-vanishing effective rate.  However, on
  Benchmarks~A and~B, RMSprop's final values ($0.5027$ and $0.7271$) are slightly above
  the noise floor, indicating residual oscillation near the minimum driven by an effective
  step that does not shrink to zero.

\item \textbf{Heavy Ball:} The momentum term accumulates velocity along the valley,
  providing meaningful acceleration on Rosenbrock ($f=1.40$ vs.\ GD's $f=3.51$).  On
  convex quadratics (Benchmark~A) it matches the GD baseline, as the constant Hessian
  allows perfect momentum tuning.

\item \textbf{Overall takeaway:} Exact knowledge of $f^{\star}$ makes Polyak
  unbeatable; Heavy Ball is the best practical choice on ill-conditioned smooth problems;
  Adagrad and RMSprop are safest for general use, with RMSprop preferred when training
  is long.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Theoretical Background}

Gradient descent with constant step size achieves an $O(1/k)$ rate for convex functions
and $O(\exp(-\mu k/L))$ for $\mu$-strongly convex functions.  Nesterov~(1983) showed that
momentum can accelerate convergence to the theoretically optimal $O(1/k^2)$ for convex
functions using only first-order information --- a result known as Nesterov's optimal
gradient method.  Adam~\citep{KingmaB14} combines this momentum idea with per-coordinate
adaptive scaling, combining the benefits of RMSprop and gradient momentum.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead / extrapolation point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The fundamental distinction from Heavy Ball is the \emph{lookahead}: the gradient is
evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than at
the current iterate $x_k$.  This allows the method to ``correct'' its momentum before
overshooting.  The adaptive schedule $\beta_k = (k-1)/(k+2)$ increases the momentum
coefficient from $0$ to $\beta_{\max}$ gradually, with the sequence of momentum values
related to the sequence of optimal step sizes in the descent lemma proof.  For convex
$f$ with Lipschitz gradient $L$, setting $\alpha \leq 1/L$ achieves:
\begin{equation}
  f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2},
\end{equation}
which is $O(1/k^2)$ --- optimal among first-order methods.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$,
          tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{Biased first moment estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Biased second moment estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum (via $m_t$) with per-coordinate adaptive scaling (via $v_t$),
inheriting the advantages of both RMSprop and Heavy Ball.  The bias-correction terms
$(1-\beta_i^t)$ are critical during the initial iterations: without them, $m_1 = (1-\beta_1)g_1$
and $v_1 = (1-\beta_2)g_1^2$ would be heavily biased towards zero (since $\beta_1, \beta_2$
are close to 1 in typical settings), causing the effective step to be overestimated.  The
correction restores the estimates $\hat{m}_t \approx \mathbb{E}[g_t]$ and
$\hat{v}_t \approx \mathbb{E}[g_t^2]$.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in
\{5, 40\}$, 50 epochs, with data shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute the $m$ data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$ in the permuted order}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Mini-batch gradient}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $g_b$ is an unbiased estimate of the full gradient:
$\mathbb{E}[g_b] = \nabla J(\theta)$.  Its variance is proportional to $\sigma_g^2/b$
where $\sigma_g^2$ is the per-sample gradient variance.  A smaller batch gives a noisier
but cheaper gradient estimate, while a larger batch is more accurate but more expensive.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$.  With $g'(x) = 4x^3$
and $g''(x) = 12x^2$, we compute $g(0.25) = 0.25^4 \approx 0.00391$,
$g'(0.25) = 4(0.25)^3 = 0.0625$, $g''(0.25) = 12(0.25)^2 = 0.75$:
\begin{equation}
  g(x) \approx g(x_0) + g'(x_0)(x - x_0)
  = 0.00391 + 0.0625\,(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
  g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
  = 0.00391 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\end{equation}

\textbf{Newton update derivation.}  Setting the derivative of the quadratic model to
zero:
\begin{equation}
  \frac{d}{dx}\!\Bigl[g(x_0) + g'(x_0)(x-x_0) + \tfrac{1}{2}g''(x_0)(x-x_0)^2\Bigr] = 0
  \;\;\Rightarrow\;\;
  g'(x_0) + g''(x_0)(x-x_0) = 0
  \;\;\Rightarrow\;\;
  x = x_0 - \frac{g'(x_0)}{g''(x_0)}.
\end{equation}
This is the Newton step: $x_{k+1} = x_k - [g''(x_k)]^{-1} g'(x_k)$.  In multiple
dimensions, $g''$ is replaced by the Hessian $\nabla^2 f(x_k)$ and the update becomes
$x_{k+1} = x_k - [\nabla^2 f(x_k)]^{-1} \nabla f(x_k)$.  Critically, Newton's method
is \emph{scale invariant}: applying a linear coordinate transformation does not change
the iterates, unlike gradient descent.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$.  The first-order
         approximation (blue dashed line, tangent) captures only the local slope and
         diverges rapidly from the true function.  The second-order approximation (red
         dash-dot, parabola) captures the local curvature and provides a much better fit
         near $x_0$.  The minimum of the parabola at $x^* = x_0 - g'(x_0)/g''(x_0) =
         0.25 - 0.0625/0.75 \approx 0.167$ is a better approximation to the global
         minimum at $x=0$ than the gradient step $x_0 - \alpha g'(x_0)$ for most
         choices of $\alpha$.  This motivates Newton's method as minimisation of a local
         quadratic model at each step.}
\label{fig:q3_approx}
\end{figure}

\subsection{Newton's Method with Damping}

\begin{algorithm}[H]
\caption{Newton's Method with Hessian Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise Hessian for numerical stability}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Solve linear system for Newton direction}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The regularisation $\lambda I$ (damping) ensures $H_k$ is positive definite even when
the true Hessian has small or negative eigenvalues (e.g., far from a local minimum on
non-convex functions).  With $\lambda = 10^{-8}$ as used here, the modification is
negligible at well-conditioned Hessians but prevents singular or near-singular linear
systems.  The damped Newton step with $\alpha < 1$ is called the \emph{damped Newton
method} and is used when the pure Newton step ($\alpha = 1$) would overshoot --- as
occurs for the highly nonlinear Rosenbrock function.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark~B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector.  The approximation error decomposes into:
\begin{equation}
  \left|\frac{f(x+\delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right|
  \leq \underbrace{\frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(x)\right|}_{\text{truncation error}} + \underbrace{\frac{2\varepsilon_{\text{mach}}}{\delta}\|f\|_{\infty}}_{\text{round-off error}}.
\end{equation}
Balancing these two terms gives an optimal step size $\delta^* \approx \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$.
In practice, larger $\delta$ (e.g., $\delta = 0.8$ here) causes the first (truncation) term to dominate,
producing a badly incorrect gradient direction when the function is highly nonlinear over scale $\delta$.

Settings used:
\begin{itemize}[nosep]
  \item \textbf{Good setting:} $\alpha = 0.08$, $\delta = 0.05$, iterations $= 120$
  \item \textbf{Poor setting:} $\alpha = 0.08$, $\delta = 0.8$, iterations $= 120$
  \item \textbf{Exact GD baseline:} $\alpha = 0.06$, iterations $= 120$
\end{itemize}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Method)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Stochastic directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: $\alpha = 0.025$, $\delta = 0.08$, iterations $= 220$.

This zeroth-order method estimates a directional derivative along a random unit vector,
requiring only \emph{two} function evaluations per iteration regardless of the problem
dimension $d$.  In contrast, the full finite-difference gradient requires $d+1$ evaluations
per step.  For high-dimensional problems, this dimension-independence is the key advantage
of zeroth-order methods.  However, the stochastic directional gradient is a high-variance
estimate: $\mathbb{E}[\hat{g}_k] = \nabla f(x_k)$ (in expectation) but with variance
$O(1)$ rather than $O(1/d)$ per coordinate, so convergence is slow.

\subsection{Part B: Derivative-Free Methods (Benchmark~C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$
and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r=1$): reflect the worst vertex through the centroid of the remaining vertices.
  \item \textbf{Expansion} ($\gamma=2$): if reflection improved beyond the best vertex, try a further extension.
  \item \textbf{Outside/Inside Contraction} ($\rho=0.5$): if reflection worsened, contract towards the centroid.
  \item \textbf{Shrink} ($\sigma=0.5$): if contraction failed, move all vertices towards the best vertex.
\end{enumerate}
No gradient information is required; only function evaluations at simplex vertices.  The
algorithm is well-suited to low-dimensional smooth objectives but has no convergence
guarantees in general and is not recommended for $d > 10$.

Settings: initial simplex step $= 0.35$, iterations $= 160$, starting from $x_0 = (-1, 1)$.

The fundamental trade-off is between per-iteration cost and convergence speed.
Gradient-based methods are efficient in terms of iterations but require either an
analytical gradient or $d+1$ function evaluations per step.  Zeroth-order methods are
cheap per iteration but converge slowly.  Grid search provides global coverage guarantees
at an exponential cost in dimension.  For the two-dimensional benchmarks studied here,
all methods are feasible; at $d=1000$, only exact GD and Nesterov random search remain
practical.

\subsection{Problem Setup and Theoretical Framework}

Minimise Benchmark~B subject to $x_1 \geq 0.5$, starting from the \emph{infeasible}
initial point $x_0 = (0.2, 4.0)$:
\begin{equation}
  \min_{x \in \mathcal{C}}\; f(x_1, x_2) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1),
  \qquad
  \mathcal{C} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}.
\end{equation}

The unconstrained minimum is at $(0.582, 2.0)$.  Since $0.582 > 0.5$, the unconstrained
minimum satisfies the constraint --- the constraint is \emph{inactive} at the optimum, and
the constrained minimum coincides with the unconstrained one.

\textbf{KKT conditions.}  For this inequality-constrained problem, the first-order
optimality conditions (KKT) are:
\begin{align}
  \nabla f(x^*) + \mu \nabla g(x^*) &= 0, \label{eq:kkt1}\\
  \mu g(x^*) &= 0, \label{eq:kkt2}\\
  g(x^*) &\leq 0, \quad \mu \geq 0, \label{eq:kkt3}
\end{align}
where $g(x) = 0.5 - x_1 \leq 0$ (constraint in standard form).  Since the constraint is
inactive at $x^* = (0.582, 2.0)$, complementary slackness~\eqref{eq:kkt2} gives $\mu=0$,
reducing to $\nabla f(x^*) = 0$ --- the unconstrained stationarity condition.  If the
unconstrained minimum had satisfied $x_1^* < 0.5$ (i.e., been infeasible), the constraint
would be active ($g(x^*)=0$, $\mu > 0$) and the KKT conditions would yield a boundary
optimum.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1}) = \bigl[\max(0.5,\, z_1),\; z_2\bigr]^T$ \Comment{Project onto $\mathcal{C}$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection $\Pi_{\mathcal{C}}$ onto the halfspace $\{x_1 \geq 0.5\}$ is simply
$\Pi_{\mathcal{C}}(z)_1 = \max(0.5, z_1)$, $\Pi_{\mathcal{C}}(z)_2 = z_2$.  This $O(1)$
operation makes PGD very efficient for simple constraint sets.

\begin{algorithm}[H]
\caption{Penalty Method (L1 Exterior Penalty)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $\nabla F_{\lambda}(x_k) \gets \nabla f(x_k) + \lambda \cdot [-\mathbf{1}_{x_1 < 0.5},\; 0]^T$ \Comment{Subgradient}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty function $F_\lambda$ adds an $L_1$ penalty proportional to constraint
violation.  As $\lambda \to \infty$, the penalised minimiser converges to the constrained
minimiser.  However, large $\lambda$ creates a non-smooth, steep barrier near $x_1 = 0.5$,
necessitating smaller step sizes for stability.  The subgradient of $\max(0, 0.5-x_1)$
with respect to $x_1$ is $-1$ when $x_1 < 0.5$ and $0$ when $x_1 > 0.5$ (undefined at
$x_1=0.5$, where any value in $[-1,0]$ is a valid subgradient).

Settings: PGD ($\alpha=0.08$); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ (
$\alpha=0.05$ for $\lambda \leq 1.8$; $\alpha=0.03$ for $\lambda=4.5$);
unconstrained GD ($\alpha=0.07$).  All run for 100 iterations.

\textbf{Projected Gradient Descent} enforces feasibility as a hard constraint at every
single iteration via the cheap projection operation ($O(d)$ for a box).  It is
guaranteed to converge to a KKT point for convex objectives with a suitable step size.
The main limitation is that the projection oracle must be efficiently computable; for
complex constraint sets (e.g., semidefinite cone, intersection of many halfspaces), this
can be as hard as the original problem.

\textbf{Penalty methods} are simpler to implement (only a modified unconstrained gradient
is needed) and do not require a projection oracle.  The weight $\lambda$ controls the
trade-off: small $\lambda$ under-penalises violation and fails to enforce feasibility;
large $\lambda$ enforces it but introduces numerical stiffness (the gradient changes
rapidly near the boundary), requiring smaller step sizes.  The exact penalty method
(using $\lambda > 0$ sufficiently large) recovers the constrained solution exactly,
but determining the required $\lambda$ in advance is non-trivial.

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm \hfill [30 marks]}
\label{sec:q6}
% ============================================================

\subsection{Theoretical Background: Polytopes and Linear Programming}

Linear programming (LP) minimises a linear objective over a convex polytope.  The
\emph{fundamental theorem of linear programming} states that the minimum of a linear
function over a bounded polytope is always attained at a vertex (extreme point).  This
follows from the fact that a linear function cannot have a local minimum in the interior
of a polytope (since moving towards any adjacent vertex improves or maintains the
objective).

The Frank--Wolfe algorithm (also called the \emph{conditional gradient method}) exploits
this structure: at each iteration it solves a linear subproblem over the constraint set
$\mathcal{X}$, which always returns a vertex $z_k$.  The new iterate is a convex
combination of $x_k$ and $z_k$, ensuring feasibility is automatically maintained.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme.}  For $f(x) = a^T x$ with $a = [1, 2]^T$, we minimise
$f(x) = x_1 + 2x_2$ over $\mathcal{X}$.  Since the objective is separable and both
components of $a$ are positive, the minimum is achieved at the smallest feasible values
of each coordinate:
\begin{equation}
  x^{\star} = (0.5,\; -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This is a vertex of the polytope $\mathcal{X}$, confirming the fundamental theorem.

\caption{Q6-I: Feasible region $\mathcal{X}$ (blue box) and linear objective contours
         $f(x) = x_1 + 2x_2 = c$.  The contour lines are parallel (constant slope
         $-1/2$) and the objective decreases towards the lower-left corner.  The minimum

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, g_k^T x$ \Comment{Linear minimisation oracle (LP subproblem)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination --- always feasible}
\EndFor
\end{algorithmic}
\end{algorithm}

For box constraints, the LP subproblem decouples per coordinate:
\begin{equation}
  (z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations.  Since $x_{k+1}$ is a convex combination of two
feasible points ($x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$), all iterates
remain feasible by convexity of $\mathcal{X}$ --- no projection is ever needed.

For smooth convex $f$ with Lipschitz gradient, the standard Frank--Wolfe algorithm with
step size $\gamma_k = 2/(k+2)$ converges at rate $O(1/k)$:
\begin{equation}
  f(x_k) - f^{\star} \leq \frac{2L\,\mathrm{diam}(\mathcal{X})^2}{k+2}.
\end{equation}
The fixed $\beta$ schedule used here is a simplified variant; the standard $O(1/k)$ rate
applies with the decreasing step size schedule.

\textbf{Frank--Wolfe duality gap.}  The quantity
$h_k = \nabla f(x_k)^T(x_k - z_k) \geq 0$ is an upper bound on the suboptimality
$f(x_k) - f^{\star}$, providing an efficiently computable certificate of near-optimality.

\subsection{Interior Optimum: $f(x) = (x_1 - 1)^2 + (x_2 - 5)^2$}

The unconstrained minimum $(1, 5)$ lies inside $\mathcal{X}$ (since $1 \in [0.5, 5]$ and
$5 \in [-5, 10]$).  Initial point: $x_0 = (1, 1)$.  Two mixing parameters:
$\beta = 0.90$ and $\beta = 0.985$.

\subsection{Interior vs.\ Boundary Behaviour}

\textbf{Interior optimum:} When $x^{\star} \in \mathrm{int}(\mathcal{X})$, the gradient
$\nabla f(x^{\star}) = 0$, so the LP oracle selects arbitrary vertices (the linear
subproblem is degenerate at the optimum).  As $x_k \to x^{\star}$, the LP vertex $z_k$
oscillates among extreme points of $\mathcal{X}$, and the convex combination weight
$(1-\beta)$ must be small for $x_k$ to ignore these oscillations.  This is the
fundamental limitation of Frank--Wolfe for interior optima: unlike PGD (which converges
at the rate of unconstrained GD once the constraint becomes inactive), Frank--Wolfe
continues to ``consult the boundary'' at every step, creating persistent oscillation.

\textbf{Boundary optimum:} When the optimal $x^{\star}$ lies on $\partial\mathcal{X}$,
Frank--Wolfe is advantageous: the LP oracle consistently selects the vertex closest to
$x^{\star}$, and the iterate naturally approaches the boundary.  The $O(1/k)$ rate is
unavoidable with fixed $\beta$; using the standard $\gamma_k = 2/(k+2)$ schedule would
guarantee $O(1/k)$ convergence with a quantified constant.  The gap at termination
($f=0.364$ vs.\ $f^{\star}=0.25$) is consistent with this rate given 140 iterations.

\subsection{Key Lessons}

\begin{enumerate}
\item \textbf{Second-order information is powerful but expensive.}  Newton's method
  solves quadratic losses in one step and dramatically outperforms GD on
  ill-conditioned problems like Rosenbrock.  Its cost ($O(d^3)$ per step for dense
  Hessians) limits its use to small-to-medium problems.

\item \textbf{Momentum provides nearly free acceleration.}  Nesterov's method matches
  Newton's performance on Rosenbrock at much lower cost (no Hessian required) and
  achieves the theoretically optimal $O(1/k^2)$ convergence rate for convex functions.

\item \textbf{Adaptive methods are robust but not always fast.}  Adagrad and RMSprop
  adapt to problem geometry automatically but are not strictly faster than well-tuned GD;
  their main advantage is reduced hyperparameter sensitivity.

\item \textbf{Exact knowledge of $f^{\star}$ makes Polyak unbeatable.}  In the rare
  case where $f^{\star}$ is known (e.g., Rosenbrock), the Polyak step dominates all
  first-order methods.  Otherwise, misspecification causes catastrophic divergence.

\item \textbf{Projected GD is the cleanest approach for simple constraints.}  For
  halfspace or box constraints, projection is $O(d)$ and enforces feasibility exactly
  at every step.  Penalty methods are more flexible but require careful tuning of
  $\lambda$.

\item \textbf{Frank--Wolfe excels for boundary optima and structured constraint sets.}
  Its projection-free update (only an LP per step) is valuable when projection is
  expensive, but its $O(1/k)$ rate and oscillation near interior optima are
  disadvantages compared to PGD.

\item \textbf{Derivative-free methods are indispensable when gradients are unavailable.}
  Nelder--Mead and grid search require only function values, making them applicable to
  black-box optimisation.  However, they scale poorly with dimension.
\end{enumerate}

\begin{lstlisting}[caption={Gradient Descent}]
def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Hessian Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex}]
def nelder_mead(loss_fn, x0, step, n_iters,
                alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1]); f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1]); f_e = loss_fn(x_e)
            if f_e < f_r: simplex[-1] = x_e; f_vals[-1] = f_e
            else: simplex[-1] = x_r; f_vals[-1] = f_r
        else:
            if f_r < f_vals[-1]:
                x_c = centroid + rho * (x_r - centroid); f_c = loss_fn(x_c)
                if f_c <= f_r: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
            else:
                x_c = centroid + rho * (simplex[-1] - centroid); f_c = loss_fn(x_c)
                if f_c < f_vals[-1]: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
