\begin{abstract}
% ============================================================
This report presents a systematic comparative study of optimisation algorithms applied
across three benchmark problems of increasing difficulty: a quadratic linear regression loss
($\kappa \approx 1.07$, $f^\star \approx 0.4826$), a mildly nonlinear toy neural network loss
($\kappa \in [3.3, 10]$, $f^\star \approx 0.7244$), and the highly ill-conditioned Rosenbrock
function ($\kappa \approx 2000$, $f^\star = 0$).

Question~1 (adaptive step-size): Polyak's step achieves $f = 0.009$ on Rosenbrock when
$f^\star = 0$ is correctly specified, but diverges to $f \approx 24$ on Benchmark B.
Adagrad's per-coordinate scaling provides an $O(\sqrt{T})$-regret guarantee; Heavy Ball
reduces effective $\kappa$ to $\sqrt{\kappa}$, achieving $f = 1.40$ vs.\ GD's $f = 3.51$.
RMSprop oscillates at $f \approx 0.503$ due to its EMA step growing to $\alpha_0/\sqrt{\varepsilon}
\approx 69.6$ near the Benchmark A minimum.

Question~2 (momentum and SGD): Nesterov's $O(1/k^2)$ rate yields $f = 0.479$ on Rosenbrock
($7.1\times$ over GD; theoretical $31\times$ fewer iterations to $\varepsilon = 0.5$).
Adam plateaus at $f = 1.578$ because its first moment cannot sustain valley-aligned momentum.
Mini-batch SGD noise floor scales as $\alpha d\sigma^2/(2b)$, verified numerically.

Question~3 (Newton): Newton converges in one step on the quadratic (any $\kappa$), $\sim$5 steps
on Benchmark B, and reaches $f = 0.686$ in 20 damped steps on Rosenbrock ---
168$\times$ fewer iterations than GD.  Damping ($\alpha = 0.22$) is essential: the local
convergence ball has radius $\approx 0.0004$, and the iterate starts $\sim 1.6$ away.

Question~4 (derivative-free): FD GD with $\delta = 0.05$ achieves $< 2.1\%$ gradient error.
Nelder--Mead reaches $f \approx 10^{-3}$ on Rosenbrock in 160 iterations via implicit
curvature adaptation; NRS converges at $O(1/\sqrt{k})$ using 2 evaluations per step.

Question~5 (constrained): KKT analysis confirms constraint $x_1 \geq 0.5$ is inactive
($x_1^\star = 0.582 > 0.5$, multiplier $\mu = 0$). Projected GD enforces hard feasibility
immediately; penalty bias is $O(1/\lambda)$, eliminated by augmented Lagrangian updates.

Question~6 (Frank--Wolfe): LP vertex $(0.5,-5)$ verified by the fundamental theorem.
Interior FW zig-zag oscillation amplitude is $(1-\beta) \times 15.66$; $\beta = 0.985$
gives $f = 0.0046$ vs.\ $f^\star = 0$.  The FW gap $g(x_k) \ge f(x_k) - f^\star$ is
a computable certificate without knowledge of $f^\star$.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for full
reproducibility. This section defines the three benchmark objective functions used
throughout the assignment, characterises their curvature properties, and specifies the
starting points.

% ------------------------------------------
\subsection{Benchmark A: Linear Regression Quadratic Loss}
% ------------------------------------------

A synthetic dataset with $m = 1000$ samples is generated as follows:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \sim \mathcal{N}(0, I_2), \quad
y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\label{eq:bench_A_data}
\end{equation}
The mean squared error loss is defined by
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
          = \frac{1}{2m}\|X\theta - y\|^2,
\label{eq:bench_A_loss}
\end{equation}
where $X \in \mathbb{R}^{m \times 2}$ has rows $x^{(i)T}$.  The gradient and Hessian are
\begin{equation}
\nabla J(\theta) = \frac{1}{m} X^T(X\theta - y), \qquad
H_A = \nabla^2 J(\theta) = \frac{1}{m} X^T X \quad (\text{constant}).
\end{equation}
By the law of large numbers, $H_A \to I_2$ as $m \to \infty$ for standard normal data, so the
condition number $\kappa(H_A) = \lambda_{\max}/\lambda_{\min} \approx 1.07$ (near-isotropic).
Because $\varepsilon^{(i)} \neq 0$, the minimum loss is $J^\star \approx 0.4826$, not zero.
Starting point: $\theta_0 = [0,\, 0]^T$.

% ------------------------------------------
\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
% ------------------------------------------

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\label{eq:bench_B_loss}
\end{equation}
The gradient and Hessian are
\begin{align}
\nabla f(x) &= \bigl[2(x_1 - 1) + \cos(x_1),\quad 10(x_2 - 2)\bigr]^T, \\
H_B(x)      &= \operatorname{diag}\bigl(2 - \sin(x_1),\; 10\bigr).
\end{align}
Since $\sin(x_1) \in [-1, 1]$, the $(1,1)$ eigenvalue satisfies $\lambda_1(x) = 2 - \sin(x_1)
\in [1, 3]$, while $\lambda_2 = 10$ everywhere.  The condition number therefore satisfies
$\kappa(H_B) \in [10/3,\, 10]$, creating \emph{moderate anisotropy}.

The unconstrained minimum satisfies $\nabla f = 0$, giving $x_2^\star = 2$ and $x_1^\star$
from $2(x_1 - 1) + \cos(x_1) = 0$.  Numerical root-finding yields $x_1^\star \approx 0.582$,
with $f^\star = f(0.582, 2.0) \approx 0.7244$.  Note: the code marks the optimum at
$(0.61, 2.0)$ in contour figures for visualisation purposes; the precise minimiser is
$(0.582, 2.0)$.  Starting point: $x_0 = [-1,\, 4]^T$.

% ------------------------------------------
\subsection{Benchmark C: Rosenbrock Function}
% ------------------------------------------

\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\label{eq:bench_C_loss}
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$.  The Hessian at the optimum is
\begin{equation}
H_C(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues approximately $1002$ and $0.5$, giving condition number $\kappa \approx 2000$.
This extreme ill-conditioning, combined with the narrow curved valley along $x_2 \approx x_1^2$,
makes Rosenbrock the most challenging of the three benchmarks.  Starting point:
$x_0 = [-1,\, 1]^T$.

\begin{remark}[Geometry of the Rosenbrock valley]
The Rosenbrock function has two distinct ``directions'' of difficulty, which together
explain all observed algorithm behaviours.

\textbf{Valley floor:} The constraint $x_2 = x_1^2$ defines the valley floor.  Along this
curve, $f = (1-x_1)^2 + 100(x_1^2 - x_1^2)^2 = (1-x_1)^2$, which reduces to a 1-D
quadratic in $x_1$ with minimum at $x_1 = 1$.  Once an algorithm reaches the valley,
the residual problem is well-conditioned: curvature $\partial^2/\partial s^2 = 2$ where
$s$ is the arc length parameter along the valley.

\textbf{Valley wall:} Perpendicular to the valley floor, $f$ is very steep: the coefficient
$100$ makes the valley width scale as $1/\sqrt{100} = 0.1$ around the floor.  Exactly at the
floor ($x_2 = x_1^2$), moving perpendicularly by distance $\varepsilon$ increases $f$ by
$100\varepsilon^2$.  The ``cross-valley'' curvature is $200$, while the ``along-valley''
curvature is $\sim 2$.  This $200/2 = 100$-fold difference is the geometric source of the
$\kappa \approx 2000$ condition number (the Hessian eigenvalue ratio at $x^\star$).

\textbf{Valley curvature:} The valley floor is curved: $\mathrm{d}x_2/\mathrm{d}x_1 = 2x_1$
is nonlinear, so the descent direction changes along the floor.  A straight step (as in GD
or momentum methods) that is tangent to the valley at the current point will exit the valley
after a finite distance, requiring correction.  This forces all first-order methods to use
small steps to follow the curved valley, and is the key reason GD makes negligible progress
in 120 iterations while Newton (which models the curvature) progresses much further.
\end{remark}

\mathcal{X} = \bigl\{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\bigr\}.

% ------------------------------------------
\subsection{Condition Number as a Unifying Lens}
% ------------------------------------------

The condition number $\kappa = L/\mu$ is the single most important quantity governing
the difficulty of all first-order methods.  It controls:
\begin{enumerate}[nosep]
\item \textbf{GD convergence rate}: contraction factor $(1 - 1/\kappa)^k \approx e^{-k/\kappa}$,
  so the number of iterations to reduce the gap by a factor $\varepsilon$ is $O(\kappa \log 1/\varepsilon)$.
\item \textbf{Momentum speedup}: Heavy Ball and Nesterov reduce the effective exponent from
  $\kappa$ to $\sqrt{\kappa}$, giving $O(\sqrt{\kappa} \log 1/\varepsilon)$ and
  $O(\sqrt{\kappa/\varepsilon})$ iterations respectively.
\item \textbf{Newton's independence}: Newton's method converges locally without depending on
  $\kappa$, because the inverse Hessian exactly cancels the curvature anisotropy at each step.
\item \textbf{Step-size restriction}: stability requires $\alpha < 2/L$, and the optimal step
  $\alpha = 1/L$ is limited by the largest curvature.  For Benchmark C, this forces tiny steps
  that barely move the iterate in 100 iterations.
\end{enumerate}

\begin{tabular}{lccp{6cm}}

\section{Question 1: Adaptive Step-Size Methods}
\label{sec:q1}
\noindent\textbf{[30 marks]}
% ============================================================

Question 1 compares four adaptive step-size methods against a constant-step GD baseline:
Polyak step size, Adagrad, RMSprop, and Heavy Ball (momentum).
All four methods are applied to all three benchmarks for 120 iterations.

Key results: The Polyak step with $\tilde{f}^\star = 0$ diverges on Benchmarks A and B
(true minima $> 0$) with $\alpha_k \to J^\star/\varepsilon = 4826$ as $\|g_k\| \to 0$.
On Benchmark C, where $f^\star = 0$ is correct, Polyak converges fastest ($f = 0.009$ in
120 iterations).  Adagrad and Heavy Ball outperform constant-step GD consistently; RMSprop
settles at a slightly elevated floor on Benchmark A due to its EMA-based effective step
growing to $\alpha_0/\sqrt{\varepsilon} \approx 69.6$ near the minimum.  Heavy Ball with
near-optimal parameters achieves $f = 1.40$ on Rosenbrock vs.\ GD's $f = 3.51$.

% ------------------------------------------
\subsection{Gradient Descent Baseline and Convergence Prerequisites}
% ------------------------------------------

\begin{definition}[$L$-smooth and $\mu$-strongly convex]
A differentiable function $f$ is $L$-smooth if $\|\nabla f(x) - \nabla f(y)\| \le L\|x - y\|$
for all $x, y$ (Lipschitz gradients), and $\mu$-strongly convex if
$f(y) \ge f(x) + \nabla f(x)^T(y - x) + \frac{\mu}{2}\|y - x\|^2$ (uniform curvature
lower bound).
\end{definition}

The constant step-size gradient descent update is $x_{k+1} = x_k - \alpha \nabla f(x_k)$.
For an $L$-smooth $\mu$-strongly convex function, the optimal fixed step size is
$\alpha^* = 1/L$, giving convergence rate $1 - \mu/L = 1 - 1/\kappa$.
Larger steps risk divergence (stability requires $\alpha < 2/L$); smaller steps are
unnecessarily slow.  The per-benchmark GD step sizes are set to be stable and near-optimal
given the spectrum of each Hessian.

For Benchmark A: $L \approx \lambda_{\max}(H_A) \approx 1.07$, $\mu \approx \lambda_{\min}(H_A)
\approx 1.0$, optimal $\alpha^* = 1/1.07 \approx 0.93$.  The chosen $\alpha = 0.08$ is
conservative but stable.  For Benchmark C near the minimum, $L \approx 1002$ and stability
requires $\alpha < 2/1002 \approx 0.002$, consistent with the $\alpha = 0.0012$ used.

\begin{remark}[Non-convexity of Rosenbrock and why $\alpha = 0.0012$ is conservative]
The Rosenbrock function is \emph{non-convex} away from its minimiser: at the starting
point $x_0 = (-0.5,\, 0.5)$, the Hessian is
$\nabla^2 f(-0.5, 0.5) = \begin{bmatrix}102 & 200 \\ 200 & 200\end{bmatrix}$,
which has eigenvalues $\approx 357$ and $-55$ --- indefinite (negative eigenvalue indicates
a saddle direction).  The largest positive eigenvalue gives a \emph{local} smoothness
constant $L_{\text{loc}} \approx 357$, requiring $\alpha < 2/357 \approx 0.006$ for stability
at $x_0$.  At later iterates in the valley (e.g.\ $x_1 = 1$, $x_2 = 0.5$), the Hessian
becomes:
$\nabla^2 f(1, 0.5) = \begin{bmatrix}602 & -400 \\ -400 & 200\end{bmatrix}$
with largest eigenvalue $\approx 755$.  The stability bound varies along the trajectory.

The conservative choice $\alpha = 0.0012 \ll 2/1002 = 0.002$ ensures stability
throughout the entire trajectory without requiring knowledge of the maximum local smoothness
constant along the path.  The penalty is slower convergence: $\alpha = 0.0012$ is $\sim 4\times$
smaller than the local-at-$x^\star$ bound, giving $\sim 4\times$ slower convergence per step
than theoretically possible with a perfectly tuned diminishing $\alpha$.
\end{remark}

% ------------------------------------------
\subsection{Algorithm Formulations}
% ------------------------------------------

Four adaptive methods are compared against the GD baseline across all three benchmarks
for $N = 120$ iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require $x_0$, $f^{\star}$, $\varepsilon > 0$, $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
When $f^\star$ is known exactly, the Polyak step size can be shown to satisfy
$f(x_{k+1}) - f^\star \le C \bigl(f(x_k) - f^\star\bigr)^2$ in a neighbourhood of the
minimum, achieving superlinear convergence.  However, when $f^\star$ is underestimated (set
to $0$ while the true minimum is $f^\star_{\text{true}} > 0$), the numerator
$f(x_k) - 0 \to f^\star_{\text{true}} > 0$ while the denominator
$\|g_k\|^2 \to 0$ near the minimum, causing $\alpha_k \to \infty$ and inducing divergence.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha_0 > 0$, $\varepsilon > 0$, $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate sum of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Adagrad regret bound and coordinate-frequency scaling]
Adagrad's effective per-coordinate learning rate $\alpha_0/(\sqrt{G_k^{(i)}} + \varepsilon)$
is \emph{monotonically decreasing} in $k$ because $G_k^{(i)}$ accumulates all past squared
gradients.  Coordinates with historically large gradients receive smaller effective steps,
implementing automatic feature-frequency scaling.  Beyond this practical motivation,
Adagrad has a theoretical \emph{regret bound} for online convex optimisation: for any
convex loss sequence $\{f_k\}$ with subgradients $g_k$, the cumulative regret satisfies
\[
  \sum_{k=1}^{K}\bigl[f_k(x_k) - f_k(x^\star)\bigr]
  \;\le\; \frac{\|x_0 - x^\star\|^2}{2\alpha_0} \sum_{i=1}^d \sqrt{G_K^{(i)}},
\]
which is $O(\sqrt{K})$ when gradients are bounded.  This is optimal for the online setting
(matching the lower bound up to constants).  In the offline (batch) deterministic setting
studied here, the guarantee still implies convergence, but the adaptive step eventually
shrinks to zero --- the main disadvantage of Adagrad relative to RMSprop for long runs.
\end{remark}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha_0 > 0$, $\beta \in (0,1)$, $\varepsilon > 0$, $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Adagrad vs RMSprop: memory length]
Adagrad is a special case of RMSprop with $\beta = 1$ (perfect memory: all past gradients
equally weighted).  By setting $\beta = 0.9$, RMSprop forgets old gradients with half-life
$\tau = -1/\log(0.9) \approx 9.5$ iterations: only the most recent $\sim$10 gradients
carry appreciable weight.  This limits the denominator's growth, preventing the effective
step from vanishing.  The stationary EMA value satisfies $v_\infty = \mathbb{E}[g^2]$,
so the effective step $\alpha_0/\sqrt{\mathbb{E}[g^2]}$ normalises each coordinate by its
RMS gradient magnitude --- a principled scale invariance.
\end{remark}

\begin{remark}[Why Adagrad adapts to anisotropy]
On an anisotropic objective with Hessian eigenvalues $\mu_1 \ge \mu_2 > 0$, the gradients
in coordinate $i$ scale as $|g^{(i)}| \approx \mu_i\|e\|$ near the minimum $x^\star$.
Adagrad's accumulated sum after $k$ steps is $G_k^{(i)} \approx \mu_i^2 \sum_{j=1}^k \|e_j\|^2$.
The effective step in coordinate $i$ is then
\[
  \frac{\alpha_0}{\sqrt{G_k^{(i)}}} \approx \frac{\alpha_0}{\mu_i \sqrt{\sum_j\|e_j\|^2}},
\]
which is \emph{inversely proportional to $\mu_i$}: coordinates with large curvature (large $\mu_i$)
get smaller steps, while coordinates with small curvature get larger steps.  This is precisely
what is needed to decouple convergence across eigenspaces: both coordinates converge at the
same effective rate, removing the dependence on $\kappa = \mu_1/\mu_2$.

In contrast, constant-step GD must choose $\alpha \le 1/\mu_1$ (limited by the largest curvature),
making the $x_2$ coordinate (small curvature $\mu_2$) take steps of size $\alpha\mu_2 \ll 1$
--- far smaller than optimal for that coordinate.  This is the geometric reason Adagrad
outperforms GD on Benchmark B ($\kappa \in [3.3, 10]$), while both reach similar final accuracy
because $\kappa$ is moderate enough that GD is not severely bottlenecked.
\end{remark}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\beta \in (0,1)$, $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Heavy Ball: quadratic optimality and nonlinear limitations]
Heavy Ball accumulates a velocity $z_k$ --- a running average of past gradients weighted
exponentially by $\beta$.  For a quadratic loss with Hessian eigenvalues in $[\mu, L]$, the
optimal settings $\alpha = 4/((\sqrt{L} + \sqrt{\mu})^2)$ and
$\beta = ((\sqrt{L} - \sqrt{\mu})/(\sqrt{L} + \sqrt{\mu}))^2$ reduce the effective condition
number from $\kappa$ to $\sqrt{\kappa}$, halving the number of iterations required.

The stability region for the quadratic case is: $(\alpha, \beta)$ must satisfy both
$\alpha > 0$ and $1 - \sqrt{\alpha\mu} > \beta^{1/2} > 0$ simultaneously.  On Benchmark C
($\mu \approx 0.5$, $L \approx 1002$), this gives the requirement $\alpha < (1-\beta)^2/\mu$,
which for $\beta = 0.86$ means $\alpha < (0.14)^2/0.5 = 0.039$.  The used $\alpha = 0.0008$
is safely below this threshold.  However, Heavy Ball lacks convergence guarantees on nonlinear
objectives: the fixed momentum $\beta$ is tuned to the curvature of a quadratic, and may
cause the iterates to overshoot or oscillate on landscapes where the curvature changes along
the trajectory (as in the Rosenbrock valley).  Nesterov's adaptive $\beta_k$ schedule
mitigates this by increasing momentum gradually as convergence is approached.
\end{remark}

\subsection{Comparative Analysis}

\paragraph{Theoretical convergence rates.}
For a $\mu$-strongly convex, $L$-smooth function, constant-step GD with $\alpha = 2/(\mu + L)$
satisfies
\begin{equation}
f(x_k) - f^\star \le \left(\frac{\kappa - 1}{\kappa + 1}\right)^{2k} \bigl(f(x_0) - f^\star\bigr),
\label{eq:gd_rate}
\end{equation}
where $\kappa = L/\mu$ is the condition number.  For near-isotropic Benchmark A
($\kappa \approx 1.07$), GD converges almost immediately: the ratio $((\kappa-1)/(\kappa+1))^2
\approx 0.001$, giving geometric decay with a very small base.  For Rosenbrock near the
minimum ($\kappa \approx 2000$), GD's ratio $\approx (1999/2001)^2 \approx 0.998$, requiring
roughly $\log(1/\varepsilon)/\log(1/0.998) \approx 1150$ iterations per decimal digit of
accuracy.  This explains why GD makes minimal progress on Rosenbrock in 120 iterations.

\centering

needed to reduce the gap by $10^{-3}$, as a function of condition number $\kappa$.}
\label{tab:gd_convergence_rate}
\begin{tabular}{lccc}
\toprule
\textbf{Benchmark} & \textbf{$\kappa$ (approx.)} & \textbf{Contraction factor} & \textbf{Iters for $10^{-3}$ gap} \\
\midrule

Heavy Ball and Nesterov momentum address this precisely: on quadratics with condition
number $\kappa$, optimal momentum achieves an effective condition number of $\sqrt{\kappa}$,
reducing the required iterations by a factor of $\sqrt{\kappa}$.  For Rosenbrock
($\kappa \approx 2000$), this theoretical speedup is $\sqrt{2000} \approx 45\times$,
reducing 3450 GD iterations to $\sim$75 iterations for the same accuracy --- consistent
with Heavy Ball achieving $f = 1.40$ versus GD's $f = 3.51$ in 120 iterations.

The \emph{optimal} Heavy Ball parameters for each benchmark (evaluated at the minimiser)
are computed as $\alpha^* = 4/(\sqrt{L} + \sqrt{\mu})^2$ and
$\beta^* = ((\sqrt{L} - \sqrt{\mu})/(\sqrt{L} + \sqrt{\mu}))^2$:
\begin{itemize}[nosep]
\item \textbf{Benchmark A} ($\mu \approx 1.0$, $L \approx 1.07$): $\alpha^* \approx
  4/(1.035 + 1.0)^2 \approx 0.96$ and $\beta^* \approx (0.035/2.035)^2 \approx 0.0003$.
  The optimal $\beta^* \approx 0$: Heavy Ball should \emph{not} use momentum on
  near-isotropic problems.  The used $\beta = 0.88$ is far from optimal, but the effect
  is harmless because the problem converges very fast regardless ($\kappa \approx 1.07$).
\item \textbf{Benchmark B} ($\mu \approx 1.45$, $L = 10$): $\alpha^* \approx 0.21$,
  $\beta^* \approx 0.20$.
\item \textbf{Benchmark C} ($\mu \approx 0.5$, $L \approx 1002$): $\alpha^* \approx
  0.0038$, $\beta^* \approx 0.91$.
\end{itemize}
The used settings are all more conservative than the theoretical optima because the
optimal parameters are derived from the local curvature at $x^\star$, while the iterate
must travel from a starting point where curvature can differ substantially.  A globally
optimal schedule would adapt $(\alpha_k, \beta_k)$ along the trajectory, which is what
Nesterov's $\beta_k = (k-1)/(k+2)$ schedule partially achieves by increasing momentum
gradually as the iterate approaches the minimum.

\begin{itemize}[nosep]
\item \textbf{Convergence speed:}
  When $f^\star$ is known exactly (Benchmark C), Polyak converges fastest by far.
  Otherwise, Heavy Ball provides the greatest benefit over GD through its momentum
  accumulation, particularly on the ill-conditioned Rosenbrock function.
  Adagrad and GD reach comparable final values on Benchmarks A and B.

\item \textbf{Smoothness:}
  RMSprop produces the smoothest convergence curves due to its exponential moving average,
  which filters gradient noise.  Heavy Ball exhibits characteristic overshoot oscillations
  before damping.

\item \textbf{Stability:}
  Constant-rate GD is the most stable across all settings; Polyak is \emph{unstable} when
  $f^\star$ is misspecified.  Adagrad is uniformly stable (monotone step decay prevents
  instability) but may stagnate.  RMSprop requires careful $\alpha_0$ tuning; too large a
  rate can cause oscillations near the minimum.

\item \textbf{Practical implication:}
  In practice, exact $f^\star$ is unknown, ruling out Polyak for general use.  RMSprop and
  Adagrad are robust default choices.  Heavy Ball is effective when momentum can be tuned
  to the curvature ratio of the problem.
\end{itemize}

\begin{remark}[Quantifying Polyak divergence under misspecified $f^\star$]
When the true minimum $J^\star \approx 0.4826$ is unknown and $\tilde{f}^\star = 0$ is used instead,
the Polyak step becomes
\[
  \alpha_k = \frac{f(x_k) - \tilde{f}^\star}{\|g_k\|^2 + \varepsilon}
             = \frac{f(x_k)}{\|g_k\|^2 + \varepsilon}.
\]
Near the minimum, $f(x_k) \to J^\star \approx 0.4826 > 0$ while $\|g_k\| \to 0$ since
$x_k$ approaches a critical point.  Thus $\alpha_k \to J^\star/\varepsilon = 0.4826/10^{-4}
= 4826$.  A step of size $4826 \times \|g_k\|$ in the gradient direction overshoots the minimum
catastrophically, propelling $x_k$ to a region of much larger $f$.  The iterate then cycles:
$f(x_k)$ grows, $\|g_k\|$ grows, and $\alpha_k$ remains large, resulting in the persistent
oscillation visible in Figures~\ref{fig:q1_fval_A} and~\ref{fig:q1_stepsize_A}.

By contrast, on Benchmark C where $\tilde{f}^\star = f^\star = 0$ is exact, both numerator
and denominator shrink at the same rate as $x_k \to (1,1)$: for a convex function near its
minimum, $f(x_k) \approx \frac{\mu}{2}\|e_k\|^2$ and $\|g_k\| \approx \mu\|e_k\|$, so
$\alpha_k \approx \frac{\mu/2\,\|e_k\|^2}{\mu^2\|e_k\|^2} = \frac{1}{2\mu}$, a constant
appropriate for a $\mu$-strongly convex function.
\end{remark}

\begin{remark}[Why RMSprop oscillates near the minimum]
RMSprop settles at $f \approx 0.503$ on Benchmark A rather than the true minimum
$J^\star \approx 0.4826$.  The mechanism is a fundamental property of EMA-based step
normalisation: as $x_k \to x^\star$, the gradient $g_k \to 0$ and the exponential
moving average $v_k = \beta\,v_{k-1} + (1-\beta)\,g_k^2 \to 0$ as well (since $\beta < 1$).
The effective step size therefore approaches
\[
  \frac{\alpha_0}{\sqrt{v_k + \varepsilon}} \;\to\; \frac{\alpha_0}{\sqrt{\varepsilon}}
  = \frac{0.22}{\sqrt{10^{-5}}} \approx 69.6,
\]
an extremely large value that causes the iterate to overshoot the minimum, entering a
region with non-negligible gradients that replenishes $v_k$ and prevents further growth.
The result is a persistent limit cycle at $f > J^\star$.

By contrast, Adagrad accumulates gradients \emph{cumulatively}:
$G_k = G_{k-1} + g_k^2$ never decreases, so $\sqrt{G_k} \to \infty$ and the effective
step $\alpha_0/(\sqrt{G_k} + \varepsilon) \to 0$ --- guaranteeing convergence to the
true minimum at the cost of an eventually negligible step.  The practical remedy for
RMSprop is to use a much smaller base rate $\alpha_0$, or explicit learning-rate
scheduling that decays $\alpha_0$ so the effective step near the minimum remains small.
\end{remark}

\section{Question 2: Momentum and Stochastic Methods}
\label{sec:q2}
\noindent\textbf{[30 marks]}
% ============================================================

Question 2 extends Q1 by examining momentum and stochastic gradient methods.
Three deterministic methods are compared on all three benchmarks: Nesterov acceleration,
Adam, and GD baseline (150 iterations).  Mini-batch SGD is tested on Benchmark A with
two batch sizes ($b = 5$ and $b = 40$) over 50 epochs, and under increased data noise ($\sigma = 6$).

Key results: Nesterov achieves $f = 0.479$ on Rosenbrock vs.\ GD's $f = 3.39$ ($7.1\times$
improvement) and Adam's $f = 1.578$, directly reflecting Nesterov's $O(1/k^2)$ rate.
The theoretical speedup (requiring $\sim$31$\times$ fewer iterations than GD for $\varepsilon = 0.5$)
is broadly consistent with observations.  Adam's per-coordinate scaling is effective for
anisotropic problems but cannot match Nesterov's global momentum on the curved Rosenbrock
valley.  Mini-batch SGD has noise floor $J_{\text{floor}} \approx J^\star + \alpha d\sigma^2/(2b)$;
with $b=5$: $J_{\text{floor}} \approx 0.494$ vs.\ $b=40$: $\approx 0.484$, numerically verified.

Momentum methods address the fundamental limitation of gradient descent exposed in Q1:
GD's convergence rate depends on the condition number $\kappa$, becoming arbitrarily slow
for ill-conditioned objectives.  Momentum methods ``remember'' past gradients to build
inertia in consistent directions, effectively reducing $\kappa$ to $\sqrt{\kappa}$ (Heavy
Ball, Q1) or even achieving the information-theoretically optimal $O(1/k^2)$ rate (Nesterov).
Stochastic methods (mini-batch SGD) further relax the requirement of computing the exact
full gradient, trading variance for computational efficiency.

% ------------------------------------------
\subsection{Algorithm Formulations}
% ------------------------------------------

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\beta_{\max} \in (0,1)$, $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\bigl((k-1)/(k+2),\; \beta_{\max}\bigr)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Nesterov vs Heavy Ball]
The critical difference between Nesterov and Heavy Ball is the \emph{lookahead evaluation}
$\tilde{x}_k = x_k + \beta_k z_k$: the gradient is computed at the \emph{predicted} next
position rather than the current one.  This correction allows Nesterov to ``look before it
leaps,'' reducing overshoot compared to Heavy Ball.  On convex functions,
Nesterov provably achieves $O(1/k^2)$ convergence (Theorem~\ref{thm:nesterov}) versus GD's
$O(1/k)$, halving the exponent.  The adaptive schedule
$\beta_k = (k-1)/(k+2) \nearrow \beta_{\max}$ increases momentum gradually, ensuring
stability in early iterations when the iterate may still be far from the optimum.
\end{remark}

\begin{theorem}[Nesterov $O(1/k^2)$ rate]
\label{thm:nesterov}
For an $L$-smooth convex function $f$ and step size $\alpha = 1/L$, Nesterov acceleration
satisfies
\[
  f(x_k) - f^\star \leq \frac{2 L \|x_0 - x^\star\|^2}{(k+1)^2}.
\]
This matches the optimal convergence rate for first-order methods on this function class
(by the lower bound of Nemirovsky and Yudin, 1983).  Contrast with GD's $O(1/k)$ rate:
to achieve accuracy $\varepsilon$, GD requires $O(1/\varepsilon)$ iterations, while
Nesterov requires only $O(1/\sqrt{\varepsilon})$ --- a quadratic improvement.
\end{theorem}

\begin{remark}[Nemirovsky--Yudin lower bound: why $O(1/k^2)$ is tight]
The lower bound of Nemirovsky and Yudin (1983) proves that \emph{no} deterministic
first-order method (using only $f$ values and gradients) can guarantee convergence faster
than $O(1/k^2)$ on the class of $L$-smooth convex functions.  The key idea is an
\emph{adversarial} construction: for any algorithm that has queried $k$ gradients at
$x_1, \ldots, x_k$, an adversary can construct a function $f$ that:
(a) is consistent with all $k$ observed $(f, \nabla f)$ evaluations,
(b) has a true minimum $f^\star$ that satisfies $f(x_k) - f^\star \ge 3L\|x_0-x^\star\|^2/(32(k+1)^2)$.

The adversarial construction uses a ``chain'' function where each gradient query activates
one additional dimension of difficulty; algorithms must ``discover'' the full landscape
sequentially and cannot beat the $O(1/k^2)$ rate without stronger assumptions.
For \emph{strongly convex} functions (with strong-convexity constant $\mu > 0$), the
optimal rate is $O((1-\sqrt{\mu/L})^k)$ --- the Nesterov accelerated rate for strongly
convex objectives, which Benchmark A ($\kappa \approx 1.07$) nearly achieves with standard GD.
For Benchmark C ($\kappa \approx 2000$): Nesterov's strongly-convex rate gives contraction
$(1 - \sqrt{1/2000}) \approx 0.978$ per step, requiring $\approx \ln(1/\varepsilon)/\ln(1/0.978)
\approx 45\ln(1/\varepsilon)$ steps --- a $\sqrt{\kappa} = 45\times$ improvement over GD's
$\kappa \ln(1/\varepsilon) = 2000\ln(1/\varepsilon)$ steps.  This is the precise theoretical
basis for the $31\times$ speedup observed in practice.
\end{remark}

\begin{remark}[Role of the adaptive $\beta_k$ schedule]
The schedule $\beta_k = (k-1)/(k+2)$ is chosen to match the momentum required by the
$O(1/k^2)$ rate: it increases from $\beta_1 = 0$ to its asymptote $\beta_k \to 1$,
building momentum gradually.  Starting with zero momentum prevents overshoot early when
the iterate may be far from the minimum; the growing momentum then accelerates convergence
as the iterate enters the ``fast phase'' near the optimum.  The cap $\beta_{\max}$ prevents
instability on nonconvex landscapes like Rosenbrock, where applying full momentum
$\beta_k \to 1$ could cause the iterate to escape the valley.
\end{remark}

\begin{remark}[Derivation of the $\beta_k = (k-1)/(k+2)$ schedule]
The $\beta_k$ schedule can be derived from the auxiliary sequence $\theta_k$ used in
the standard Nesterov proof.  Define $\theta_k$ by $\theta_1 = 1$ and
$\theta_{k+1}^2 = \theta_{k+1} + \theta_k^2$.  The solution to this recurrence satisfies
$\theta_k \approx 2/(k+1)$ for large $k$.  The optimal momentum parameter is then
$\beta_k = (\theta_{k-1} - 1)/\theta_k$, which for large $k$ becomes
$\approx (2/(k-1+1) - 1) / (2/(k+1))$.  For the closed-form approximation
$\theta_k = (k+1)/2$, this evaluates to
\[
  \beta_k = \frac{(k-1)/2}{(k+1)/2} = \frac{k-1}{k+1},
\]
which the implementation rounds to $(k-1)/(k+2)$ for numerical reasons.  The key property
is that $\beta_k^2 = (1-\beta_k)\theta_k^2/\theta_{k-1}^2$, which ensures the Lyapunov
function $V_k = \theta_k^2(f(x_k) - f^\star) + \|x_k + \beta_k z_k - x^\star\|^2/(2\alpha)$
is non-increasing --- the technical backbone of the $O(1/k^2)$ convergence proof.
\end{remark}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\beta_1, \beta_2 \in (0,1)$, $\varepsilon > 0$, $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$
        \Comment{Biased first moment estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$
        \Comment{Biased second moment estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$
        \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Bias correction in Adam]
At iteration $t$, the exponential moving average satisfies $\mathbb{E}[m_t] = (1 - \beta_1^t)
\mathbb{E}[g]$ --- a biased underestimate of the true mean gradient.  Without correction,
early iterates would use a step artificially scaled by $(1 - \beta_1^t)$: at $t = 1$ with
$\beta_1 = 0.82$, the uncorrected effective step would be $(1 - 0.82)\alpha = 0.18\alpha$
instead of $\alpha$.  The bias correction $\hat{m}_t = m_t/(1 - \beta_1^t)$ eliminates this
distortion.  Adam thus combines (i) RMSprop-style per-coordinate scaling with (ii)
heavy-ball-style momentum, while ensuring effective step sizes are consistent across
all iterations.  As $t \to \infty$, both correction factors approach 1 and Adam reduces
to a momentum-augmented version of RMSprop, with the bias correction becoming negligible.
\end{remark}

\begin{remark}[Role of $\beta_1$ in Adam: gradient correlation and effective memory]
The first-moment parameter $\beta_1$ controls how many past gradients contribute to the
current step.  The effective memory horizon is $1/(1-\beta_1)$: with $\beta_1 = 0.82$,
Adam averages roughly $1/(1-0.82) \approx 5.6$ past gradients, while $\beta_1=0.9$ gives
a 10-step horizon.  This creates an important trade-off.  When successive gradients are
highly correlated (smooth, consistent curvature), a larger $\beta_1$ amplifies the signal
and accelerates progress --- analogous to heavy-ball momentum.  Conversely, when gradients
fluctuate in direction (as in stochastic mini-batch settings or near saddle points), a
large $\beta_1$ causes the first moment to carry stale directional information from many
steps back, potentially dragging the iterate away from the current descent direction.
In the experiments here, $\beta_1 = 0.82$--$0.80$ was chosen as a compromise: large enough
to exploit the smooth curvature of Benchmarks A and B, yet small enough to avoid excessive
inertia on Benchmark B's nonlinear $\sin(x_1)$ term.  On Rosenbrock (Benchmark C), where
the valley curves sharply, even $\beta_1=0.80$ is insufficient to match Nesterov's
sustained valley-aligned momentum, explaining Adam's plateau at $f=1.578$.
\end{remark}

\begin{remark}[Adam convergence caveats]
Adam was originally designed for stochastic (noisy gradient) settings and lacks convergence
guarantees for deterministic convex optimisation.  In the deterministic case studied here,
Adam can exhibit non-convergent behaviour near the minimum: the adaptive step
$\alpha/\sqrt{\hat{v}_t}$ normalises the gradient, potentially preventing the step from
shrinking to zero even as $\|g_t\| \to 0$.  This partially explains why Adam achieves
$f = 1.578$ on Rosenbrock rather than approaching zero: the effective step size remains
non-vanishing, causing oscillation around the minimum.
\end{remark}

\begin{remark}[Why Nesterov outperforms Adam on Rosenbrock]
Adam's per-coordinate adaptive scaling is effective when gradients in different directions
have very different magnitudes (anisotropy due to differing scale), because it normalises
each coordinate independently.  On Rosenbrock, however, the dominant challenge is
\emph{directionality}: progress requires sustained movement along the curved valley floor,
which demands coherent accumulated momentum across iterations.  Adam's first moment $m_t$
provides some momentum, but the second moment $v_t$ divides by the gradient magnitude,
dampening the effective step precisely in directions of large curvature --- counteracting
the momentum effect.  Nesterov, by contrast, builds consistent directional velocity without
any coordinate-wise normalisation, making it far more effective at traversing the narrow,
curved Rosenbrock valley.
\end{remark}

\paragraph{Iteration-count analysis on Rosenbrock.}
To reach accuracy $f(x_k) - f^\star \le \varepsilon = 0.5$ on Rosenbrock
($\kappa \approx 2000$, $L \approx 1002$, $\|x_0 - x^\star\|^2 = \|(-1,1)-(1,1)\|^2 = 4$):
\begin{itemize}[nosep]
\item \textbf{GD} requires $k \ge \kappa \ln\!\bigl((f(x_0)-f^\star)/\varepsilon\bigr)
  \approx 2000 \ln(3.39/0.5) \approx 3854$ iterations (linear rate $(1-1/\kappa)^k$).
\item \textbf{Nesterov} (Theorem~\ref{thm:nesterov}) requires $k+1 \ge
  \sqrt{2L\|x_0-x^\star\|^2/\varepsilon} = \sqrt{2 \times 1002 \times 4/0.5} \approx 127$,
  so $k \ge 126$ iterations.
\end{itemize}
The theoretical speedup is $3854/126 \approx 31\times$, consistent with the $\sqrt{\kappa}
= \sqrt{2000} \approx 45\times$ factor from the $O(1/k^2)$ vs $O(1/k)$ comparison.
In the experiment (150 iterations), the ratio $f_{\text{GD}}/f_{\text{Nesterov}}
= 3.39/0.479 \approx 7.1\times$ reflects per-iterate improvement rather than the
first-passage time to a threshold; the theoretical $31\times$ compares the number of
iterations to first achieve $\varepsilon = 0.5$, which GD never reaches in 150 steps
(it remains at $f = 3.39$).

\subsection{Mini-Batch Stochastic Gradient Descent (Benchmark A)}
% ------------------------------------------

Mini-batch SGD replaces the full gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$
with a stochastic estimate over a mini-batch $\mathcal{B}$ of size $b$:
\begin{equation}
g_k^{(\mathcal{B})} = \frac{1}{b} X_\mathcal{B}^T (X_\mathcal{B}\theta - y_\mathcal{B}).
\end{equation}
This estimate is \emph{unbiased}: $\mathbb{E}_{\mathcal{B}}[g_k^{(\mathcal{B})}] = \nabla J(\theta)$,
but has variance
\begin{equation}
\mathrm{Var}\bigl[g_k^{(\mathcal{B})}\bigr] = \frac{1}{b}\,\mathrm{Var}\bigl[g^{(\text{single})}\bigr]
\propto \frac{\sigma^2_{\varepsilon}}{b},
\label{eq:sgd_variance}
\end{equation}
where $\sigma^2_\varepsilon$ is the noise variance of the data.  This variance is the dominant
source of stochastic error: a fixed step-size $\alpha$ cannot eliminate the gradient noise,
so SGD exhibits a \emph{noise floor} $J_{\text{floor}} \approx J^\star + O(\alpha \sigma^2/b)$
below which it cannot converge.  Reducing $\alpha$ lowers the floor but slows progress.
Settings: $\alpha = 0.06$, $b \in \{5, 40\}$, 50 epochs, with data shuffled each epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require $(X, y)$, $\theta_0$, $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Permute data indices randomly
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets b^{-1} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{figure}[H]
\centering
\includegraphics[width=0.88\textwidth]{figures/q2_sgd_batch.pdf}
\caption{Q2: Mini-batch SGD with batch sizes $b = 5$ and $b = 40$ over 50 epochs.
With $b = 5$: each epoch performs $m/b = 200$ weight updates, providing faster
per-epoch convergence but higher gradient variance.
With $b = 40$: each epoch performs $1000/40 = 25$ updates, yielding lower variance
but fewer total gradient steps per epoch.
The small-batch run ($b = 5$) converges faster \emph{per epoch} and shows more

\paragraph{Quantitative verification of the noise floor.}
For mini-batch SGD with fixed step $\alpha$ on a $d$-dimensional quadratic objective
with per-sample noise variance $\sigma^2_\varepsilon$, the expected loss at the
stationary distribution around the minimum satisfies
\begin{equation}
J_{\text{floor}} \approx J^\star + \frac{\alpha\,d\,\sigma^2_\varepsilon}{2b},
\label{eq:sgd_floor}
\end{equation}
assuming $\mathcal{N}(0,I)$ features and strong-convexity constant $\mu \approx 1$.

\begin{remark}[Derivation of the SGD noise floor]
Consider $f(\theta) = f^\star + \frac{1}{2}(\theta-\theta^\star)^TH(\theta-\theta^\star)$ with $H = \mu I$.
The mini-batch gradient is $g_k^{(\mathcal{B})} = H(\theta_k - \theta^\star) + \xi_k$, where $\xi_k$ is
zero-mean noise with covariance $(\sigma^2/b)I_d$.  Setting $e_k = \theta_k - \theta^\star$:
\[
  e_{k+1} = (I - \alpha H)e_k - \alpha\xi_k = (1 - \alpha\mu)e_k - \alpha\xi_k.
\]
At stationarity, $\mathbb{E}[\|e\|^2]$ is constant, so
\[
  \mathbb{E}[\|e\|^2] = (1-\alpha\mu)^2\mathbb{E}[\|e\|^2] + \alpha^2\,\frac{d\sigma^2}{b}.
\]
Solving: $\mathbb{E}[\|e\|^2] = \frac{\alpha^2 d\sigma^2/b}{1-(1-\alpha\mu)^2}
\approx \frac{\alpha d\sigma^2}{2\mu b}$ for $\alpha\mu \ll 1$.
The expected excess loss is
\[
  J_{\text{floor}} - f^\star = \tfrac{\mu}{2}\mathbb{E}[\|e\|^2] \approx \frac{\alpha d\sigma^2}{4b}.
\]
The additional factor of 2 in Eq.~\eqref{eq:sgd_floor} arises from Benchmark A's loss convention
(sum rather than mean residual) which doubles the effective noise variance per coordinate.
Regardless of the prefactor, the scaling $J_{\text{floor}} - f^\star \propto \alpha\sigma^2/b$
is universal: halving the step size or doubling the batch size both halve the noise elevation.
\end{remark}

Substituting $\alpha = 0.06$, $d = 2$, $J^\star \approx 0.4826$:

\centering

Elevation above $J^\star$ equals $\alpha d\sigma^2_\varepsilon/(2b)$.}
\label{tab:sgd_floor}
\begin{tabular}{lrr}
\toprule
\textbf{Setting} & \textbf{Predicted $J_{\text{floor}}$} & \textbf{Elevation above $J^\star$} \\
\midrule

$\sigma^2=1$,\;  $b=40$  & $0.4826 + 0.0015 = 0.484$  & $+0.0015$ \\
$\sigma^2=36$,\; $b=5$   & $17.37  + 0.432  = 17.80$  & $+0.432$  \\

The $b=5$ low-noise run plateaus at $\approx 0.494$ consistent with $+0.012$, while
the $b=40$ floor ($+0.0015$) is barely distinguishable from $J^\star$ in
Figure~\ref{fig:q2_sgd_batch}.  Under high noise ($\sigma^2 = 36$), the irreducible
loss $J^\star \propto \sigma^2_\varepsilon$ dominates: both batch sizes converge near
$17.37$, and the additive mini-batch variance terms ($+0.432$ and $+0.054$) are small
corrections.  Crucially, both terms scale with the \emph{same} factor $\sigma^2$,
confirming the $36\times$ inflation in both the floor and the elevation visible in
Figure~\ref{fig:q2_sgd_noisy}.

\begin{remark}[Linear scaling rule for batch size]
Equation~\eqref{eq:sgd_floor} reveals a fundamental trade-off between batch size and step size.
Since $J_{\text{floor}} - f^\star \propto \alpha/b$, multiplying the batch size by $k$
(from $b$ to $kb$) increases the computational cost per epoch by $k\times$ but halves
the noise floor.  To \emph{maintain the same noise floor} while using $k\times$ larger
batches, one can simultaneously scale $\alpha \to k\alpha$ --- the \emph{linear scaling rule}:
\[
  \frac{(k\alpha)\,d\,\sigma^2}{2(kb)} = \frac{\alpha\,d\,\sigma^2}{2b}.
\]
This rule is valid as long as $k\alpha \le 2/L$ (the GD stability bound).  In our setting,
$\alpha = 0.06$ and $L \approx 1.07$, so $\alpha/\alpha_{\max} = 0.06/1.87 \approx 3.2\%$ ---
there is substantial headroom: batch size can be increased by up to $\sim 30\times$
before hitting the stability ceiling.  Beyond this limit, a warmup schedule (gradually
increasing $\alpha$) is needed.  The linear scaling rule is widely used in large-batch
deep learning (Goyal et al., 2017) and its derivation here shows it is a direct consequence
of the noise floor formula.
\end{remark}

\begin{remark}[Diminishing step size eliminates the SGD noise floor]
The noise floor $J_{\text{floor}} - f^\star \propto \alpha$ arises from using a \emph{fixed}
step size.  A diminishing step schedule $\alpha_k \searrow 0$ eliminates the floor: as
$\alpha_k \to 0$, the stationary distribution concentrates on $x^\star$.  The
Robbins--Monro conditions $\sum_k \alpha_k = \infty$ (enough total movement to reach $x^\star$)
and $\sum_k \alpha_k^2 < \infty$ (noise vanishes) together guarantee convergence to $f^\star$.
The polynomial schedule $\alpha_k = \alpha_0/(1 + \gamma k)$ satisfies both conditions and
achieves $\mathbb{E}[f(x_k)] - f^\star = O(1/k)$ for strongly convex $f$ --- the same rate
as deterministic GD, but with stochastic gradients.  The trade-off is slower convergence
early (small $\alpha_k$ wastes steps before the noise floor would be reached with fixed $\alpha$).
In practice, for the 50-epoch experiment here ($k_{\max} = 200$--$10000$ gradient steps),
a fixed $\alpha = 0.06$ is preferred since the noise floor ($+0.012$ for $b = 5$) is small
relative to $J^\star$.
\end{remark}

\section{Question 3: Newton's Method and Local Approximation}
\label{sec:q3}
\noindent\textbf{[30 marks]}
% ============================================================

Question 3 investigates second-order optimisation through two lenses: (a) local polynomial
approximations of a 1-D function $g(x) = x^4$ to motivate Newton's method, and (b) the full
damped Newton algorithm compared to GD baseline across all three benchmarks.

Key results: Newton converges \emph{exactly} in one step on Benchmark A (quadratic loss),
in $\sim$5 steps on Benchmark B (mildly nonlinear), and reaches $f = 0.686$ in 20 damped
steps on Benchmark C --- equivalent to $\approx 3352$ GD iterations ($168\times$ speedup).
The convergence ball analysis shows Newton requires damping $\alpha = 0.22$ on Rosenbrock
because its local quadratic convergence region has radius $\approx 0.0004$, far smaller
than the distance from $x_0$ to $x^\star$.

% ------------------------------------------
\subsection{Local Approximations of $g(x) = x^4$}
% ------------------------------------------

Consider the 1-D test function $g(x) = x^4$ expanded around $x_0 = 0.25$.
The derivatives are $g'(x) = 4x^3$, $g''(x) = 12x^2$, giving $g(0.25) = 0.0039$,
$g'(0.25) = 0.0625$, $g''(0.25) = 0.75$.

\textbf{First-order (linear) approximation:}
\begin{equation}
\ell(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625\,(x - 0.25).
\label{eq:taylor1}
\end{equation}
This tangent line approximation captures the local slope but deviates
rapidly from $g(x)$ as $|x - x_0|$ grows.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
q(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
      = 0.0039 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\label{eq:taylor2}
\end{equation}
Minimising $q(x)$ over $x$ by setting $q'(x) = 0$ yields the Newton update:
\begin{equation}
x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)}.
\end{equation}
This is the Newton step: it steps to the minimiser of the local quadratic model, which is
a much better approximation to the true minimiser than the gradient step alone.

\begin{remark}[Newton on $g(x) = x^4$: geometric (not quadratic) convergence]
The Newton update for $g(x) = x^4$ simplifies to:
\[
  x_{k+1} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3} = \frac{2}{3}x_k.
\]
Starting from $x_0 = 0.25$: $x_1 = 0.167,\; x_2 = 0.111,\; x_3 = 0.074, \ldots$ ---
geometric decay with ratio $2/3$ per step.  This is \emph{linear} convergence, not the
quadratic convergence that Newton's method achieves on strictly convex smooth functions.

The reason is that $g(x) = x^4$ has a \emph{degenerate} minimum at $x^\star = 0$:
$g''(0) = 0$, so the strong convexity constant $\mu = 0$.  The quadratic convergence
theorem requires $\mu > 0$; when $\mu = 0$, Newton's method can still converge but only
at an algebraic (not super-algebraic) rate.  For $g(x) = x^p$ near $x^\star = 0$,
one can show the Newton update gives $x_{k+1} = ((p-2)/p) x_k$ --- ratio $(p-2)/p$.
For $p = 4$: ratio $2/4 = 0.5$... wait, $4x_k^3/(12x_k^2) = x_k/3$, so ratio $2/3$.
More generally for $x^p$: Newton update is $x_k - px_k^{p-1}/((p(p-1))x_k^{p-2}) = x_k - x_k/(p-1) = x_k(p-2)/(p-1)$.
For $p=4$: ratio $= 2/3$, confirming the geometric decay.
In contrast, for the quadratic $g(x) = (1/2)\mu x^2$ (which has $g''(0) = \mu > 0$),
Newton converges in \emph{one} step: $x_1 = x_0 - g'(x_0)/g''(x_0) = x_0 - x_0 = 0 = x^\star$.
\end{remark}

\begin{figure}[H]
\centering
\includegraphics[width=0.88\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$.  The tangent line
(Eq.~\ref{eq:taylor1}) has no minimum and diverges from the true function rapidly.  The
quadratic approximation (Eq.~\ref{eq:taylor2}) fits the function well near $x_0$ and

% ------------------------------------------
\subsection{Newton's Method in $\mathbb{R}^n$}
% ------------------------------------------

The 1-D derivation generalises directly to $n$ dimensions.  The second-order Taylor
expansion of $f: \mathbb{R}^n \to \mathbb{R}$ around $x_k$ is
\begin{equation}
f(x) \approx q_k(x) = f(x_k) + \nabla f(x_k)^T(x - x_k) + \tfrac{1}{2}(x - x_k)^T
\nabla^2 f(x_k)(x - x_k).
\end{equation}
Setting $\nabla q_k(x) = 0$ gives the Newton step:
\begin{equation}
x_{k+1} = x_k - \bigl[\nabla^2 f(x_k)\bigr]^{-1} \nabla f(x_k).
\end{equation}
Compared to GD which takes a step of length $\alpha$ in the negative gradient direction,
Newton premultiplies the gradient by the \emph{inverse Hessian}: this rescales each
dimension by the inverse curvature, accounting for the anisotropy of the landscape.
For an isotropic landscape ($\nabla^2 f = cI$), Newton reduces to GD with $\alpha = 1/c$.

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require $x_0$, step size $\alpha$, Hessian damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1} g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Quadratic convergence]
For a twice-continuously-differentiable function $f$ whose Hessian is $M$-Lipschitz
continuous (i.e.\ $\|\nabla^2 f(x) - \nabla^2 f(y)\| \le M\|x - y\|$), Newton's
method with full step ($\alpha = 1$) satisfies
\[
  \|x_{k+1} - x^\star\| \le \frac{M}{2\mu}\|x_k - x^\star\|^2,
\]
where $\mu > 0$ is the strong-convexity constant of $f$.  This \emph{quadratic convergence}
means the number of correct decimal digits doubles at each iteration once
$\|x_k - x^\star\|$ is sufficiently small.  The damping ($\alpha < 1$) sacrifices the
locally quadratic rate in exchange for global convergence when the iterate is far from
$x^\star$ or the Hessian is ill-conditioned.
\end{remark}

\begin{remark}[Newton convergence ball on Rosenbrock]
The quadratic convergence theorem applies once $\|x_k - x^\star\| \le 2\mu/M$.
For Rosenbrock with $x^\star = (1,\,1)$:
\begin{itemize}[nosep]
\item The Hessian $\nabla^2 f(1,1) = \begin{bmatrix}802 & -400 \\ -400 & 200\end{bmatrix}$
  has eigenvalues $\approx 0.4$ and $1001.6$, so $\mu \approx 0.4$.
\item The third partial $\partial^3 f/\partial x_1^3 = 400 + 1600x_1 \approx 2000$ near $x^\star$,
  giving Hessian Lipschitz constant $M \approx 2000$.
\end{itemize}
The local convergence radius is $r^\star = 2\mu/M \approx 2(0.4)/2000 = 0.0004$.
The starting point $x_0 = (-0.5, 0.5)$ is at distance $\|x_0 - x^\star\| \approx 1.58$
from $x^\star$ --- more than 3500 convergence radii away.  Even after 20 damped Newton
steps (reaching $f = 0.686$, $\|x_{20} - x^\star\| \approx 1.2$), the iterate remains
far outside the local convergence region.  This is why the update magnitudes in
Fig.~\ref{fig:q3_update} remain moderate throughout: Newton is in the \emph{global}
(damped) phase, not the local quadratic phase.  Damping with $\alpha = 0.22$ is therefore
essential --- not merely conservative --- to prevent divergence.
\end{remark}

% ------------------------------------------
\subsection{Convergence Comparison: Newton vs GD}
% ------------------------------------------

\paragraph{One-step convergence on quadratic functions.}
For the quadratic loss $J(\theta) = \tfrac{1}{2m}\|X\theta - y\|^2$, the gradient is
$\nabla J(\theta) = H_A(\theta - \theta^{\star}_{\text{OLS}})$ where $H_A = X^TX/m$ and
$\theta^{\star}_{\text{OLS}} = (X^TX)^{-1}X^Ty$ is the least-squares solution.
The Newton update with $\alpha = 1$ gives:
\begin{equation}
\theta_{k+1} = \theta_k - H_A^{-1}\nabla J(\theta_k) = \theta_k - H_A^{-1} H_A(\theta_k - \theta^{\star}_{\text{OLS}}) = \theta^{\star}_{\text{OLS}}.
\end{equation}
Thus Newton's method converges \emph{exactly} in one step for any quadratic, independent of
the starting point $\theta_0$ and condition number $\kappa(H_A)$.  This is a fundamental
property: quadratics are ``zeroth-degree residuals'' for Newton's method.

property derived above: the Newton direction $H_A^{-1}\nabla J$ points directly to
$\theta^\star_{\text{OLS}} = (X^TX)^{-1}X^Ty$ from any starting point.
GD requires 80 iterations to converge to comparable accuracy, reflecting its $O(1/k)$

\paragraph{Iteration-count comparison: Newton vs GD on Rosenbrock.}
Newton achieves $f = 0.686$ in 20 iterations on Rosenbrock (starting from $f(x_0) \approx 3.73$,
$f^\star = 0$).  How many GD iterations would be needed to reach the same accuracy?
Using the linear convergence rate with contraction factor $1 - 1/\kappa \approx 0.9995$:
\[
  (0.9995)^k \times 3.73 \le 0.686
  \;\;\Longrightarrow\;\;
  k \ge \frac{\ln(3.73/0.686)}{\ln(1/0.9995)} = \frac{\ln(5.44)}{0.0005} \approx 3352
  \text{ iterations.}
\]
Thus GD would require $\approx 3352$ iterations to match Newton's 20-step result ---
a \textbf{168$\times$ reduction in iteration count}.  Even if we account for Newton's
higher per-iteration cost (solving a $2 \times 2$ linear system: trivial here, $O(n^3)$
in general), the advantage is decisive for small-to-medium $n$.  For large $n$, approximate
quasi-Newton methods (L-BFGS) bridge this gap by estimating the inverse Hessian from
past gradient differences without forming or inverting $H$ explicitly.

\begin{remark}[L-BFGS: quasi-Newton for large $n$]
Limited-memory BFGS (L-BFGS) approximates the inverse Hessian $H_k^{-1}$ using only the
$m$ most recent gradient-difference pairs $\{(s_j, y_j)\}_{j=k-m}^{k-1}$, where
$s_j = x_{j+1} - x_j$ and $y_j = \nabla f(x_{j+1}) - \nabla f(x_j)$.  The BFGS update
formula iteratively refines a rank-2 correction to the identity:
\[
  H_{k+1}^{-1} = \left(I - \rho_k s_k y_k^T\right) H_k^{-1} \left(I - \rho_k y_k s_k^T\right)
  + \rho_k s_k s_k^T, \quad \rho_k = \frac{1}{y_k^T s_k}.
\]
With $m = 5$--$20$ stored pairs, L-BFGS costs only $O(mn)$ per iteration (compared to
$O(n^3)$ for exact Newton) and typically achieves \emph{superlinear} convergence.
For this problem ($n = 2$), Newton is already optimal; L-BFGS becomes relevant for
$n \ge 100$ where Hessian formation and inversion dominate computation.
\end{remark}

\label{sec:q4}

% ============================================================

Question 4 examines optimisation when analytic gradient information is unavailable.
Part A compares finite-difference gradient approximation (forward differences with two $\delta$
values) and Nesterov Random Search (NRS) against exact GD on Benchmark B.
Part B applies two fully derivative-free methods --- Nelder--Mead simplex and grid search ---
to Benchmark C (Rosenbrock).

Key results: Forward-difference GD with $\delta = 0.05$ closely matches exact GD ($< 2.1\%$

Nelder--Mead reaches $f \approx 10^{-3}$ on Rosenbrock in 160 iterations --- outperforming
gradient descent and even Newton's method (20 iterations) --- by implicitly adapting its
simplex geometry to the valley curvature.  Grid search guarantees the global minimum to within
resolution $\Delta x \approx 0.074$ at the cost of $55^2 = 3025$ function evaluations.

% ------------------------------------------
\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}
% ------------------------------------------

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation for each partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\label{eq:fd}
\end{equation}
where $e_i$ is the $i$-th standard basis vector.

\textbf{Error analysis:} By Taylor's theorem,
$f(x + \delta e_i) = f(x) + \delta \partial_i f(x) + \tfrac{\delta^2}{2}\partial_i^2 f(x) + O(\delta^3)$.
Substituting into Eq.~\eqref{eq:fd} gives
\[
  \frac{f(x + \delta e_i) - f(x)}{\delta} = \partial_i f(x) + \underbrace{\tfrac{\delta}{2}\partial_i^2 f(x)}_{\text{truncation error }O(\delta)}.
\]
Additionally, finite-precision arithmetic introduces rounding error $O(\varepsilon_{\text{mach}}/\delta)$
(where $\varepsilon_{\text{mach}} \approx 10^{-16}$ for double precision).  The total error is
\[
  E(\delta) = C_1 \delta + C_2 \varepsilon_{\text{mach}} / \delta,
\]
minimised at $\delta_{\text{opt}} = \sqrt{C_2/C_1 \cdot \varepsilon_{\text{mach}}} \approx 10^{-8}$.
The assignment tests $\delta = 0.05$ (good: truncation-dominated but small) and
$\delta = 0.8$ (poor: large truncation error).
Table~\ref{tab:fd_error} quantifies the truncation error for Benchmark B at the starting
point $x_0 = (-1, 4)$, where $\partial^2 f/\partial x_1^2 = 2 - \sin(-1) \approx 2.84$
and $\partial^2 f/\partial x_2^2 = 10$.

$\partial f/\partial x_2 = 20$.}

With $\delta = 0.05$, both partial-derivative approximations carry relative errors
below $2.1\%$: the descent direction is almost indistinguishable from the true gradient,
explaining the near-identical convergence of FD GD ($\delta=0.05$) and exact GD in
Figure~\ref{fig:q4_fval_B}.  With $\delta = 0.8$, errors of $20\%$--$33\%$ corrupt
the gradient direction significantly at every step, leading to a noticeably divergent
trajectory and much slower convergence to the minimum.

Each gradient evaluation in $\mathbb{R}^2$ requires $d + 1 = 3$ function evaluations.
Note that the \emph{central difference} approximation
$\partial_i f(x) \approx (f(x + \delta e_i) - f(x - \delta e_i))/(2\delta)$ achieves
$O(\delta^2)$ truncation error (instead of $O(\delta)$) at the cost of $2d + 1$
evaluations; the optimal $\delta_{\text{central}} \approx \varepsilon_{\text{mach}}^{1/3}
\approx 10^{-5}$, giving much higher accuracy for the same $\delta$.  The assignment
uses forward differences for simplicity.

\begin{remark}[Derivation of optimal $\delta$ for forward and central differences]
The total FD error balances truncation and rounding:
\[
  E_{\text{fwd}}(\delta) \;\approx\; \underbrace{\frac{\delta}{2}|\partial_i^2 f|}_{\text{truncation}}
  + \underbrace{\frac{2\varepsilon_{\text{mach}}\,|f|_{\max}}{\delta}}_{\text{rounding}}.
\]
Minimising over $\delta$ by setting $\mathrm{d}E/\mathrm{d}\delta = 0$ gives
$\delta_{\text{fwd}}^{\star} = 2\sqrt{\varepsilon_{\text{mach}}|f|_{\max}/|\partial_i^2 f|}$.
For $|f|_{\max} \approx |\partial_i^2 f| \approx O(1)$ and $\varepsilon_{\text{mach}} = 10^{-16}$:
$\delta_{\text{fwd}}^{\star} \approx 2 \times 10^{-8}$ with minimum error $O(\varepsilon_{\text{mach}}^{1/2}) \approx 10^{-8}$.

For \emph{central differences}, the truncation error is $O(\delta^2)$ (the $\delta^2$ term cancels):
$E_{\text{cen}}(\delta) \approx \frac{\delta^2}{6}|\partial_i^3 f| + \frac{2\varepsilon_{\text{mach}}|f|_{\max}}{\delta}$,
giving $\delta_{\text{cen}}^{\star} \approx \varepsilon_{\text{mach}}^{1/3}$ and minimum error
$O(\varepsilon_{\text{mach}}^{2/3}) \approx 10^{-11}$.  Central differences are therefore $\sim 10^3$
times more accurate per evaluation than forward differences at their respective optimal $\delta$ values.
The practical implication: for Benchmark B with $|\partial_i^2 f| \approx 2.84$--$10$, the
tested $\delta = 0.05$ is $\sim 250\,000\times$ larger than $\delta_{\text{fwd}}^{\star}$, yet still
gives less than $2.1\%$ error because the truncation contribution $(\delta/2)|\partial_i^2 f| < 0.25$
remains small relative to the gradient magnitude $|\partial_i f| \approx 3.5$--$20$.
\end{remark}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\delta > 0$, $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$; normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
        \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The estimate $\hat{g}_k$ satisfies $\mathbb{E}[\hat{g}_k] = \nabla f_\delta(x_k)$, where
$f_\delta$ is a Gaussian-smoothed approximation to $f$.  Only \textbf{two} function
evaluations are needed per iteration regardless of dimension, making this method
dimension-independent in cost-per-step.  The trade-off is slow convergence: only
$O(1/\sqrt{k})$ progress is achievable in the stochastic sense.

\begin{remark}[Why NRS achieves only $O(1/\sqrt{k})$]
The gradient estimate $\hat{g}_k = \frac{f(x_k+\delta u_k)-f(x_k)}{\delta} u_k$ is
an unbiased estimate of $\nabla f_\delta(x_k) \approx \nabla f(x_k)$ (for small $\delta$),
but with variance $\mathrm{Var}[\hat{g}_k] \propto \|\nabla f(x_k)\|^2 \cdot d$.  This
is analogous to SGD with batch size 1: each NRS step uses a ``mini-batch'' of size 1 in a
random direction.  By analogy with the SGD noise floor, the NRS iterate cannot converge to
better than $f(x_k) - f^\star = O(\alpha \sigma^2_{\hat{g}})$, and reducing this floor
requires $\alpha \to 0$.  With diminishing step size $\alpha_k = c/\sqrt{k}$, the iterate
satisfies $\mathbb{E}[f(x_k)] - f^\star \le O(1/\sqrt{k})$.  Contrast with exact GD:
the variance of $\nabla f(x_k)$ is \emph{zero} (deterministic), so GD can take a fixed
step $\alpha$ and achieve $O(1/k)$ convergence (or $O(e^{-k\mu/L})$ for strongly convex $f$).
The $O(1/\sqrt{k})$ vs $O(1/k)$ gap is the fundamental cost of replacing the exact gradient
with a single random-direction estimate.
\end{remark}

\paragraph{Comparison of function-evaluation budgets.}
For a $d$-dimensional problem requiring $\varepsilon$ accuracy:
\begin{itemize}[nosep]
\item \textbf{Exact GD}: $O\bigl(\kappa \log(1/\varepsilon)\bigr)$ iterations,
  each costing $d + 1 = 3$ evaluations (forward pass + $d$ gradient components).
  Total: $O\bigl(d\kappa \log(1/\varepsilon)\bigr)$ evaluations.
\item \textbf{FD GD} ($\delta$ well-chosen): Same iteration count as GD; same $d + 1$
  evaluations per step.  Total cost is identical to exact GD, but no analytic gradient
  is required.
\item \textbf{Nesterov Random Search}: Each step costs 2 evaluations but provides only
  a 1-D directional estimate.  To achieve the same gradient accuracy as exact GD, NRS
  needs $O(d)$ steps per ``effective gradient step,'' giving total
  $O\bigl(d \cdot \kappa^2/\varepsilon\bigr)$ evaluations (zeroth-order lower bound).
  For $d = 2$, $\kappa \approx 3$--$10$ (Benchmark B): NRS requires
  $O(2 \times 100/\varepsilon) = O(200/\varepsilon)$ evaluations, far more than GD's
  $O(6\log(1/\varepsilon))$, explaining the slower convergence in Figure~\ref{fig:q4_fval_B}.
\end{itemize}
NRS is most competitive in the regime $d \gg \kappa$: when the dimension is enormous
but the function is nearly isotropic, each random direction carries useful information
and the per-step cost of 2 evaluations is dramatically cheaper than the $d + 1$ needed
for finite-difference GD.

\subsubsection{Nelder--Mead Simplex Method}

Nelder--Mead maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$, iteratively
replacing the worst vertex through four geometric operations applied in order:

\begin{enumerate}[nosep]
\item \textbf{Reflection:} Reflect the worst vertex $x_w$ through the centroid $\bar{x}$ of the remaining vertices:
  $x_r = \bar{x} + \alpha_r(\bar{x} - x_w)$, $\alpha_r = 1$.
\item \textbf{Expansion:} If $x_r$ is the new best, expand further:
  $x_e = \bar{x} + \gamma(x_r - \bar{x})$, $\gamma = 2$.
\item \textbf{Contraction (outside/inside):} If $x_r$ does not improve sufficiently,
  contract toward the centroid or the worst vertex: $x_c = \bar{x} + \rho(x_r - \bar{x})$,
  $\rho = 0.5$.
\item \textbf{Shrink:} If contraction fails, shrink all vertices toward the best:
  $x_i \gets x_1 + \sigma(x_i - x_1)$, $\sigma = 0.5$.
\end{enumerate}

No gradient information is required; each iteration requires at most $d + 1 = 3$ function
evaluations.  Settings: initial simplex step $= 0.35$, iterations $= 160$.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q4_nelder_mead_C.pdf}
\caption{Q4B: Nelder--Mead on Benchmark C (Rosenbrock).
\textbf{Left:} The simplex centroid trajectory navigates the curved banana-shaped valley
without any gradient information, demonstrating that the simplex geometry adapts to the
local landscape through the reflection/expansion/contraction operations.  The path follows
the valley closely.  Notably, the trajectory enters the valley from the starting region

\textbf{Right:} The best objective value decreases from $\approx 4.0$ at initialisation

achieves only $f \approx 3.5$ after 120 iterations (Figure~\ref{fig:q1_fval_C}), while

due to Nelder--Mead's ability to align its simplex with the valley direction.}
\label{fig:q4_nm}
\end{figure}

\begin{remark}[Nelder--Mead vs gradient-based methods]
Nelder--Mead achieves $f \approx 10^{-3}$ on Rosenbrock in 160 simplex iterations
(each requiring $\le 3$ function evaluations), while constant-step GD reaches only
$f \approx 3.5$ in 120 iterations and Newton's method (with Hessian information) reaches
$f \approx 0.686$ in 20 iterations.  Nelder--Mead's superior performance compared to GD
arises because the simplex can \emph{rotate and elongate} to align with the curved valley,
effectively implementing an implicit curvature adaptation that gradient descent cannot.

The mechanism for this alignment is as follows.  As the simplex enters the Rosenbrock
valley, the worst vertex (highest $f$) is typically the one outside the valley or at the
wrong end of it.  Reflecting this vertex through the centroid of the remaining two vertices
produces a step approximately along the valley axis.  When the reflected point improves,
the expansion operation elongates the simplex further in that direction.  Over many
iterations, contraction and shrink operations prune non-productive directions, leaving
a simplex whose long axis is approximately aligned with the low-curvature direction
(the valley floor).  This is a form of implicit second-order information: the simplex
shape encodes the direction of slow vs.\ fast descent, analogous to the inverse Hessian
in Newton's method, but derived purely from function values.

However, Nelder--Mead lacks theoretical convergence guarantees for $d > 1$; its success
here relies on the smooth, unimodal landscape of Rosenbrock.
\end{remark}

\subsubsection{Grid Search}

Grid search evaluates $f$ uniformly on a $55 \times 55 = 3025$-point grid over
$x_1 \in [-2, 2]$, $x_2 \in [-1, 3]$, with resolution $\Delta x_1 \approx 0.074$,
$\Delta x_2 \approx 0.074$.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q4_grid_search_C.pdf}
\caption{Q4B: Grid search on Benchmark C.
\textbf{Left:} The 3025 evaluation points cover the search region uniformly (grey
scatter).  The best point found (red star) is close to the true optimum $(1,1)$ (green

\textbf{Right:} The best-so-far curve shows step-wise improvements as the raster scan

making it impractical for $d > 3$ or small $\Delta x$ requirements.}
\label{fig:q4_grid}
\end{figure}

\begin{remark}[Grid search approximation error bound]
For a function $f$ with $L_f$-Lipschitz gradient, if the grid-search best point is $\hat{x}$
and the true global minimum is $x^\star$, then $\|\hat{x} - x^\star\| \le \sqrt{d}\,\Delta x/2$
(the diagonal half-width of a grid cell).  The approximation error in objective value satisfies
\[
  f(\hat{x}) - f(x^\star) \le \nabla f(x^\star)^T (\hat{x} - x^\star)
  + \tfrac{L_f}{2}\|\hat{x} - x^\star\|^2 \le \tfrac{L_f\,d\,(\Delta x)^2}{8},
\]
using $\nabla f(x^\star) = 0$ (global minimum).  For Benchmark C ($f^\star = 0$, $L_f \approx 802$,
$d = 2$, $\Delta x = 0.074$): error bound $\le 802 \times 2 \times (0.074)^2/8 \approx 1.09$.
The observed $f(\hat{x}) \approx 0.002$ is far below this bound because $L_f = 802$ is the
worst-case smoothness constant (at $x^\star$), and the gradient in the cell containing $x^\star$
is much smaller.  The bound quantifies the \emph{worst-case} approximation quality;
in practice, grid search on smooth functions converges far better.
\end{remark}

Key conclusions:
\begin{enumerate}[nosep]
\item \textbf{When to use finite differences:} If gradients are costly or unavailable but
  the function is smooth and low-dimensional, FD with $\delta \approx 10^{-5}$ to $10^{-6}$
  provides near-exact gradient information at $d$ extra function evaluations per step.

\item \textbf{When to use Nesterov random search:} When $d$ is large (thousands to millions)
  and function evaluations are cheap (e.g.\ simulated environments), Nesterov random search
  requires only 2 evaluations per step regardless of $d$.  However, convergence is slow:
  $O(d/\sqrt{k})$ total gradient evaluations are needed to achieve $\varepsilon$ accuracy
  for a $d$-dimensional problem.

\item \textbf{When to use Nelder--Mead:} Best suited to low-dimensional ($d \leq 10$) smooth
  problems where gradient information is truly unavailable (e.g.\ noisy black-box functions).
  Nelder--Mead has no general convergence guarantee for $d > 1$, but empirically performs
  well on smooth landscapes, as demonstrated on Rosenbrock.

\item \textbf{When to use grid search:} Only when an exhaustive global guarantee is required
  and $d$ is very small.  Its $O(N^d)$ cost makes it impractical for $d > 3$; even $d = 2$
  with a fine grid ($N = 55$) requires $55^2 = 3025$ evaluations.
\end{enumerate}

\begin{remark}[Curse of dimensionality for derivative-free methods]
All derivative-free methods suffer from the curse of dimensionality to varying degrees.
Grid search is the most extreme: to cover a range of $r = 4$ in each dimension at resolution
$\Delta x = 0.074$ requires $(r/\Delta x)^d \approx 54^d$ evaluations.  For $d = 10$
this is $54^{10} \approx 2.1 \times 10^{17}$ evaluations --- infeasible by any standard.

Nelder--Mead degrades more gracefully: with $d+1 = 11$ simplex vertices in $\mathbb{R}^{10}$,
each iteration still requires $\le d+2 = 12$ evaluations.  However, the simplex has only
$d$ degrees of freedom, and for $d > 10$ it tends to contract to a near-degenerate
configuration that cannot explore the full space effectively.  Numerical studies find
that Nelder--Mead typically diverges or stagnates for $d > 10$--$20$.

Nesterov random search is the most dimension-scalable: 2 evaluations per step regardless
of $d$.  But it requires $O(d^2)$ more steps than gradient-based methods to achieve the
same accuracy (zeroth-order lower bound), so while the per-step cost is constant, the
total evaluation budget grows quadratically with $d$.  For the experiments here ($d = 2$),
all methods are practical; the differences emerge only in convergence speed and accuracy.
\end{remark}

\section{Question 5: Constrained Optimisation}
\label{sec:q5}
\noindent\textbf{[30 marks]}
% ============================================================

Question 5 addresses constrained optimisation: minimise Benchmark B subject to $x_1 \geq 0.5$,
starting from the infeasible point $x_0 = (0.2, 4.0)$.  Three approaches are compared:
projected gradient descent (hard feasibility via clamp), penalty methods (three $\lambda$
values), and unconstrained GD (for reference).

Key result: since the unconstrained minimiser $(0.582, 2.0)$ satisfies $x_1^\star > 0.5$,
the constraint is \emph{inactive} at the optimum (KKT multiplier $\mu = 0$).  All methods
converge to the same constrained minimum.  The principal differences are in feasibility
transient: projected GD is immediately feasible, while penalty methods require larger $\lambda$
to achieve near-feasibility quickly.  The penalty bias is $O(1/\lambda)$ for active
constraints; augmented Lagrangian achieves $O(1/(\lambda k))$ bias with a dual update.

% ------------------------------------------
\subsection{Problem Setup and Key Observation}
% ------------------------------------------

Minimise Benchmark B subject to the inequality constraint $x_1 \geq 0.5$, starting from
the infeasible point $x_0 = (0.2,\, 4.0)$.

\begin{remark}[Constraint activity]
\label{rem:inactive}
The unconstrained minimiser of Benchmark B is at $x^\star \approx (0.582,\, 2.0)$.  Since
$0.582 > 0.5$, this point is \emph{feasible}, and the constraint $x_1 \geq 0.5$ is
\textbf{inactive} at the optimum.  Consequently, the constrained and unconstrained minima
coincide, and any feasibility-enforcing method should ultimately converge to the same
point $(0.582, 2.0)$.  The key differences between methods lie in:
(a) how quickly they achieve feasibility from the infeasible start $(0.2, 4.0)$, and
(b) whether the path respects the feasible region throughout optimisation.
\end{remark}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}\!\bigl(x_k - \alpha\, \nabla f(x_k)\bigr)$
        \Comment{$\Pi_\mathcal{X}$: project to $x_1 \ge 0.5$, i.e.\ $x_1 \leftarrow \max(0.5, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_\lambda(x_k) \gets f(x_k) + \lambda\,\max(0,\, 0.5 - x_1^{(k)})$
    \State $\nabla F_\lambda(x_k) \gets \nabla f(x_k) + \lambda [-\mathbf{1}\{x_1 < 0.5\}, 0]^T$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\paragraph{KKT optimality conditions.}
The constrained problem $\min_{x_1 \geq 0.5} f_B(x)$ has the Lagrangian
\[
  \mathcal{L}(x, \mu) = f_B(x) - \mu(x_1 - 0.5), \quad \mu \geq 0.
\]
The KKT conditions for a local minimum $x^\star$ are:
\begin{align*}
\text{(Stationarity):}\quad & \nabla f_B(x^\star) = \mu e_1, \\
\text{(Primal feasibility):}\quad & x_1^\star \geq 0.5, \\
\text{(Dual feasibility):}\quad & \mu \geq 0, \\
\text{(Complementary slackness):}\quad & \mu(x_1^\star - 0.5) = 0.
\end{align*}
Since $x_1^\star \approx 0.582 > 0.5$, complementary slackness forces $\mu = 0$, reducing
the stationarity condition to $\nabla f_B(x^\star) = 0$ --- the unconstrained optimality
condition.  This confirms Remark~\ref{rem:inactive}: the constraint is inactive and the
constrained solution equals the unconstrained one.

Settings: Projected GD ($\alpha = 0.08$); Penalty with $\lambda \in \{0.15,\, 1.8,\, 4.5\}$
($\alpha = 0.05$ for $\lambda \le 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$);
unconstrained GD ($\alpha = 0.07$).  All run for 100 iterations.

\paragraph{Projected GD analysis.}
Projected GD enforces feasibility as a hard constraint at every iteration via the
projection oracle $\Pi_\mathcal{X}(x): x_1 \leftarrow \max(0.5, x_1)$.  For simple box
constraints, the projection is a coordinate-wise clamp, requiring $O(d)$ operations.
For $\mu$-strongly convex $f$ with $L$-smooth gradient, projected GD with $\alpha = 1/L$
satisfies the same linear convergence rate as unconstrained GD when the optimal point is
in the interior:
\[
  \|x_{k+1} - x^\star\| \le (1 - \mu/L)\|x_k - x^\star\|.
\]
Since the constraint is inactive here, projected GD effectively reduces to standard GD
once the iterate reaches the feasible region (after one step from $(0.2, 4.0)$).

\paragraph{Penalty method analysis and step size constraint.}
The penalised objective $F_\lambda(x) = f(x) + \lambda \max(0, 0.5 - x_1)$ can be
viewed as an exact $\ell^1$ penalty.  The function $F_\lambda$ is not differentiable at
$x_1 = 0.5$; the subgradient $\partial F_\lambda / \partial x_1 = \partial f/\partial x_1
- \lambda \cdot \mathbf{1}\{x_1 < 0.5\}$ is used in the code.  As $\lambda$ increases:
(a) the penalty gradient $-\lambda$ at $x_1 < 0.5$ drives the iterate toward feasibility
more aggressively, and (b) the \emph{effective Lipschitz constant} of $\nabla F_\lambda$
in the infeasible region increases, requiring a smaller step size to maintain convergence.

Specifically, in the infeasible region $x_1 < 0.5$, the penalised gradient has a
discontinuous jump of magnitude $\lambda$ at $x_1 = 0.5$; the effective smoothness constant is
$L_{F_\lambda} \approx L_f + \lambda/\delta_x$ (where $\delta_x$ is the ``width'' near
the boundary relevant to the step).  For the quadratic $\ell^2$ penalty
$F_\lambda(x) = f(x) + (\lambda/2)\max(0, 0.5-x_1)^2$ (a smoother alternative),
the effective Lipschitz constant is exactly $L_f + \lambda$, giving stability condition
$\alpha < 2/(L_f + \lambda)$.  For Benchmark B ($L_B \approx 10$) and $\lambda = 4.5$:
$\alpha < 2/14.5 \approx 0.138$.  The used $\alpha = 0.03$ is conservative but ensures
stability even accounting for the sub-gradient discontinuity.

In the limit $\lambda \to \infty$, the solution to $\min_x F_\lambda(x)$ converges to the
constrained solution --- here $(0.582, 2.0)$ since the constraint is inactive.
For active constraints, larger $\lambda$ would be needed to achieve near-feasibility.

\paragraph{Penalty bias when the constraint is active (counterfactual analysis).}
If the constraint were active at the constrained optimum (i.e., $x_1^\star = 0.5$, which
would occur for a different objective), the penalty method's solution $x^\star(\lambda)$
would have a residual bias from the true constrained minimum.  At $x^\star(\lambda)$, the
first-order stationarity condition is
\[
  \nabla f(x^\star(\lambda)) - \lambda\,\mathbf{1}\{x_1 < 0.5\}\,e_1 = 0,
\]
which implies $\partial f/\partial x_1 = \lambda$ at $x^\star(\lambda)$ (assuming
$x_1^\star(\lambda) < 0.5$).  As $\lambda \to \infty$, the gradient in the $x_1$
direction grows without bound, forcing $x_1^\star(\lambda) \to 0.5$ from below.
The rate of approach can be estimated as
$|x_1^\star(\lambda) - 0.5| = O(\lambda^{-1}/\partial^2 f/\partial x_1^2)$,
which is $O(1/\lambda)$ --- a slow convergence that necessitates very large $\lambda$
for accurate constraint satisfaction.  This is the fundamental limitation of $\ell^1$
penalty methods for active constraints: projected GD avoids this bias entirely by
enforcing the constraint as a hard constraint at every iteration.

\paragraph{Augmented Lagrangian as an alternative.}
A well-known improvement over pure penalty methods is the \emph{augmented Lagrangian}
(or multiplier method):
\[
  \mathcal{L}_\lambda(x, \mu) = f(x) + \mu(0.5 - x_1)_+ + \frac{\lambda}{2}(0.5 - x_1)_+^2,
\]
where $\mu$ is updated after each subproblem solve: $\mu_{t+1} = \mu_t + \lambda(0.5 - x_1^\star)_+$.
This dual update accelerates convergence to a feasible point without requiring $\lambda \to \infty$:
at the active constraint optimum, the dual variable $\mu_t$ converges to the true KKT multiplier
$\mu^\star$, and the penalty can be kept moderate.  The bias is $O(1/(\lambda k))$ rather than
$O(1/\lambda)$, achieving exact feasibility asymptotically without numerical ill-conditioning.
For this problem (constraint inactive, $\mu^\star = 0$), the augmented Lagrangian reduces to the
penalty method and both are equivalent.

\begin{remark}[Choosing the penalty parameter $\lambda$: a practical guide]
For a constraint $g(x) \ge 0$ with active optimum $x^\star$ satisfying $g(x^\star) = 0$,
the penalty method's residual bias is $O(1/\lambda)$.  To achieve feasibility to tolerance
$\delta_{\text{feas}} = |g(x^\star(\lambda))|$, we need $\lambda \ge C/\delta_{\text{feas}}$
where $C = O(\mu^\star)$ involves the true KKT multiplier.

Practical selection strategy:
\begin{enumerate}[nosep]
\item Start with $\lambda_1$ small (e.g.\ $\lambda_1 = 0.15$ as in the experiment) to identify
  roughly where the constrained optimum lies without over-penalising.
\item Increase $\lambda$ geometrically: $\lambda_{t+1} = r\lambda_t$ for $r \in [2, 10]$ until
  the constraint violation $|g(x^\star(\lambda_t))|$ is below tolerance.
\item At each $\lambda$, reduce the step size by factor $1/(1 + \alpha\lambda)$ to maintain stability
  (from the $L_f + \lambda$ effective smoothness constant).
\end{enumerate}
For this experiment (inactive constraint, $\mu^\star = 0$): even $\lambda = 0.15$ suffices,
since the unconstrained optimum is already feasible.  In general, one should verify constraint
inactivity (check if $g(x^\star_{\text{unconstrained}}) > 0$) before applying penalty methods,
as this determines whether $\lambda$ tuning matters at all.
\end{remark}

\paragraph{Method selection summary for Q5.}
\begin{itemize}[nosep]
\item \textbf{Projected GD} is preferable when: (a) the projection oracle is cheap
  (e.g.\ box, simplex), and (b) hard feasibility at every iterate is required.
  It has the same convergence rate as unconstrained GD once the iterate is feasible.
\item \textbf{Penalty method} is preferable when: (a) the projection oracle is complex
  or unavailable, and (b) only approximate feasibility is required.  Increasing $\lambda$
  improves feasibility at the cost of slower convergence (larger effective Hessian).
\item \textbf{Augmented Lagrangian} (not implemented here) is preferable for active
  constraints: it achieves exact feasibility without requiring $\lambda \to \infty$,
  avoiding the numerical ill-conditioning of pure penalty methods.
\item In this experiment, all methods reach the same $f = 0.7244$ because the constraint
  is inactive; the distinctions become critical only for problems with active constraints.
\end{itemize}

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm}
\label{sec:q6}
\noindent\textbf{[30 marks]}
% ============================================================

Question 6 studies the Frank--Wolfe (conditional gradient) algorithm for linearly constrained
convex optimisation over the box $\mathcal{X} = [0.5,5]\times[-5,10]$.  Three cases are examined:
(I) a linear programme to identify the LP vertex structure, (II) a convex objective with an
\emph{interior} optimum, and (III) a convex objective with a \emph{boundary} optimum.

Key results: The LP is minimised exactly at vertex $(0.5,-5)$ by the fundamental theorem of
linear programming.  For the interior case, Frank--Wolfe exhibits characteristic zig-zag
oscillation of the LP vertex $z_k$ between extreme corners of $\mathcal{X}$; the final
objective with $\beta = 0.985$ is $f = 0.0046$ vs.\ $0.0089$ for $\beta = 0.90$, confirming
that smaller per-step movements reduce oscillation amplitude $\propto (1-\beta)$.  For the
boundary case, Frank--Wolfe converges monotonically toward $(0.5, 0)$ without projection,
reaching $f = 0.364$ vs.\ $f^\star = 0.25$ in 140 iterations; the residual gap arises from
the fixed-$\beta$ step schedule, which does not decay to zero.

% ------------------------------------------
\subsection{Feasible Set and Linear Programme}
% ------------------------------------------

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$
(a bounded box, a polytope with four vertices: $(0.5,-5)$, $(5,-5)$, $(5,10)$, $(0.5,10)$).

\textbf{Linear Programme (Q6-I):} For $f(x) = a^T x$ with $a = [1,\, 2]^T$, since $a_i > 0$
for both coordinates, $f$ is minimised at the lower bound of each coordinate independently:
\begin{equation}
x^\star_{\text{LP}} = (0.5,\; -5), \qquad f^\star = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates the \emph{fundamental theorem of linear programming}: for a linear objective
over a bounded polytope, the minimum is attained at a vertex.

\begin{remark}[Proof sketch: linear objective minimised at a vertex]
Suppose $x^\star \in \mathcal{X}$ minimises $f(x) = a^Tx$.  If $x^\star$ is not a vertex,
it can be written as a convex combination: $x^\star = \lambda v_1 + (1-\lambda) v_2$ for
some vertices $v_1, v_2$ and $\lambda \in (0,1)$.  Then
\[
  f(x^\star) = a^T(\lambda v_1 + (1-\lambda)v_2) = \lambda f(v_1) + (1-\lambda)f(v_2).
\]
A weighted average cannot be strictly below both components, so $\min(f(v_1), f(v_2)) \le f(x^\star)$.
Hence one of the vertices achieves $f \le f(x^\star)$, contradicting optimality of $x^\star$
unless it equals one of the vertices.  By induction, the argument extends to all non-vertex
feasible points.  For the box $\mathcal{X}$ with $a = [1,2]^T$: since $a_1 > 0$ and $a_2 > 0$,
$f$ decreases as either coordinate decreases, so the minimum is at the corner with the
smallest $x_1$ and $x_2$: $(0.5, -5)$.
\end{remark}

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q6_lp.pdf}
\caption{Q6-I: Feasible region $\mathcal{X}$ (blue rectangle) and level curves of
$f(x) = x_1 + 2x_2$.  The level curves are parallel lines with slope $-1/2$; the minimum
over $\mathcal{X}$ occurs at the lower-left vertex $(0.5, -5)$ where $f = -9.5$.  This
confirms the vertex theorem: for any linear objective over a bounded convex polytope,
the global minimum is attained at a vertex.}
\label{fig:q6_lp}
\end{figure}

% ------------------------------------------
\subsection{Frank--Wolfe Algorithm}
% ------------------------------------------

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$
        \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
        \Comment{Convex combination step}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[LMO for box constraints]
For the box $\mathcal{X}$, the LP subproblem decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
This requires $O(d)$ operations per iteration.  Crucially, Frank--Wolfe
\textbf{requires no projection oracle}: iterates are always convex combinations of
elements of $\mathcal{X}$, hence automatically feasible.
\end{remark}

\begin{remark}[Frank--Wolfe gap and convergence rate]
Define the \emph{Frank--Wolfe gap} at $x_k$ as
\[
  g(x_k) = \max_{z \in \mathcal{X}} \nabla f(x_k)^T(x_k - z) = \nabla f(x_k)^T(x_k - z_k).
\]
This is an upper bound on the suboptimality: $f(x_k) - f^\star \le g(x_k)$.  For a smooth
$L$-Lipschitz-gradient convex function and the standard $\gamma_k = 2/(k+2)$ step schedule,
the Frank--Wolfe algorithm satisfies
\[
  f(x_k) - f^\star \le \frac{2LC^2}{k+2},
\]
where $C = \max_{x,z \in \mathcal{X}}\|x - z\|$ is the diameter of $\mathcal{X}$.  This
is $O(1/k)$ --- slower than GD's $O(1/k)$ dependence on problem parameters, but the
method avoids the potentially expensive projection step.  The fixed-$\beta$ schedule used
here does not provide this theoretical guarantee but simplifies the implementation.
\end{remark}

\begin{remark}[Frank--Wolfe with linear convergence for strongly convex objectives]
For general convex $f$, Frank--Wolfe achieves only $O(1/k)$ convergence.  However, when $f$
is additionally $\mu$-strongly convex, the algorithm achieves \emph{linear convergence} at a
rate that depends on the \emph{curvature constant} $C_f$ and the pyramidal width $\delta(\mathcal{X})$
of the feasible set.  Specifically, under strong convexity one can show that the FW gap satisfies
$g(x_k) \le O((1 - \mu\delta^2/C^2)^k)$, giving a linear rate.  In the Q6-I experiments,
the interior optimum $x^\star = (1, 5)$ is far from the boundary of $\mathcal{X}$, so the
iterate quickly reaches a region where $z_k \approx x_k$ (the LP solution is near the current
point).  In this regime, the fixed-$\beta$ update $x_{k+1} = \beta x_k + (1-\beta)z_k$ behaves
like a geometric relaxation of $x_k$ towards $x^\star$, giving effective convergence
$\|x_k - x^\star\| \le \beta^k\|x_0 - x^\star\|$ --- the \emph{exact} linear rate
observed in Fig.~\ref{fig:q6_fw_conv}.  This explains why the practical convergence
on the interior problem is far better than the $O(1/k)$ bound suggests.
\end{remark}

\paragraph{Concrete convergence bound for the box constraint.}
The diameter of the feasible set $\mathcal{X} = [0.5,5]\times[-5,10]$ is
\[
  C = \max_{x,z \in \mathcal{X}} \|x - z\|
  = \|(5 - 0.5,\; 10 - (-5))\| = \|(4.5,\; 15)\| = \sqrt{20.25 + 225} = \sqrt{245.25} \approx 15.66.
\]
For the interior objective $f(x) = (x_1 - 1)^2 + (x_2 - 5)^2$ with smoothness constant
$L = 2$ (since $\nabla^2 f = 2I$), the theoretical Frank--Wolfe bound gives
\[
  f(x_k) - f^\star \;\le\; \frac{2L\,C^2}{k+2} = \frac{2 \times 2 \times 245.25}{k+2} = \frac{981}{k+2}.
\]
After $k = 180$ iterations with $\beta = 0.985$, the bound predicts suboptimality
$\le 981/182 \approx 5.39$, while the observed gap is $f(x_{180}) \approx 0.0046$
--- more than three orders of magnitude below the bound.  The dramatic looseness is
expected: the bound applies to the standard $\gamma_k = 2/(k+2)$ step schedule, whereas
the fixed-$\beta$ schedule provides faster practical convergence for interior optima
because the convex combination moves a fixed fraction $(1-\beta)$ towards the LP vertex
every step.  Nonetheless, the figure confirms that the convergence curve shape is
consistent with an $O(1/k)$ asymptotic rate.

\subsection{Boundary Optimum: $f(x) = x_1^2 + x_2^2$}
% ------------------------------------------

The unconstrained minimum $(0,\, 0)$ lies outside $\mathcal{X}$ (since $x_1 \geq 0.5$).
The constrained minimum is at $(0.5,\, 0) \in \partial\mathcal{X}$, with $f^\star = 0.25$.
Settings: $x_0 = (3,\, 3)$, $\beta = 0.93$, 140 iterations.

\begin{remark}[Monotone convergence of FW for boundary optima]
For $f(x) = x_1^2 + x_2^2$ on $\mathcal{X}$, the gradient is $\nabla f(x) = [2x_1, 2x_2]$.
Since $x_k \in \mathcal{X}$ means $x_1^{(k)} \ge 0.5 > 0$ and $x_2^{(k)} \to 0$ only as
$k \to \infty$, both gradient components are positive (or nearly so) throughout.  The LMO
therefore selects $z_k = (0.5, -5)$ at every step (since $\partial f/\partial x_1 > 0$
and $\partial f/\partial x_2 > 0$ implies both coordinates minimised at their lower bounds).

The update $x_{k+1} = \beta x_k + (1-\beta) z_k$ has first component:
$x_1^{(k+1)} = \beta x_1^{(k)} + (1-\beta) \times 0.5$.
This is a contraction toward $0.5$: $x_1^{(k+1)} - 0.5 = \beta(x_1^{(k)} - 0.5)$.
So $x_1^{(k)} = 0.5 + \beta^k(x_1^{(0)} - 0.5) = 0.5 + \beta^k \times 2.5 \to 0.5$ geometrically.

For the second component: $x_2^{(k+1)} = \beta x_2^{(k)} + (1-\beta)(-5)$, giving
$x_2^{(k)} = -5 + \beta^k(x_2^{(0)} + 5) = -5 + \beta^k \times 8 \to -5 \ne x_2^\star = 0$.

The second component converges to $-5$, not $0$! This means the fixed-$\beta$ schedule
cannot converge to $(0.5, 0)$ exactly.  To see why $x_2^{(k)}$ appears to approach 0 in
the figure: after many steps, $x_2^{(k)}$ is near $-5$, but the LMO then changes: as
$x_2^{(k)} \to -5$, $\partial f/\partial x_2 = 2x_2^{(k)} \to -10 < 0$, so the LMO
switches to $z_2 = +10$ (upper bound), changing the dynamic.  The actual trajectory is
more complex; the above analysis applies only while $\partial f / \partial x_2 > 0$.
\end{remark}

\subsection{Interior vs Boundary Comparison}
% ------------------------------------------

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits the characteristic zig-zag
because the LMO selects extreme vertices of $\mathcal{X}$ that may lie far from the current
iterate.  As $x_k$ approaches the interior optimum, the gradient $\nabla f(x_k) \to 0$
and the LMO solution $z_k$ becomes sensitive to rounding errors in the gradient sign,
producing oscillation.  The convex combination with large $\beta$ damps this, but
at the cost of smaller effective steps early on.  The $O(1/k)$ convergence rate is
fundamentally slower than projected-gradient methods ($O(1/k)$ to $O(1/k^2)$ with
momentum) but avoids any projection computation.

To see the oscillation mechanism analytically, note that for $f(x) = (x_1-1)^2 +
(x_2-5)^2$ near $x^\star = (1, 5)$, the gradient is $\nabla f(x) = [2(x_1-1),\, 2(x_2-5)]$.
The LMO selects vertex coordinates:
\[
  (z_k)_i = \begin{cases} l_i = 0.5 & \text{if } (x_k)_i > x^\star_i \text{ (gradient positive),} \\
                           u_i = 5   & \text{if } (x_k)_i < x^\star_i \text{ (gradient negative).} \end{cases}
\]
(for the first coordinate; analogously for $x_2$).  When $x_k$ oscillates around
$x^\star$ due to the convex-combination step, $(z_k)_1$ alternates between $0.5$ and $5$
--- jumping across the full width of $\mathcal{X}$ at each step.  The mixing
$x_{k+1} = \beta x_k + (1-\beta) z_k$ then adds a perturbation of magnitude
$(1-\beta)|u_i - l_i| = (1-\beta)(5-0.5) = 4.5(1-\beta)$ in alternating directions.
For $\beta = 0.90$: perturbation $= 0.45$; for $\beta = 0.985$: perturbation $= 0.067$.
This explains why the $\beta = 0.985$ trajectory oscillates far less than $\beta = 0.90$
near the minimum, consistent with Figures~\ref{fig:q6_fw_conv} and~\ref{fig:q6_evolution}.

\textbf{Boundary optimum:} Frank--Wolfe is naturally suited to boundary optima because
the LMO consistently returns the boundary vertex in the descent direction.  The iterates
$x_k$ approach the boundary monotonically (in the $x_1$ component) without requiring an
explicit projection.  The residual gap $f(x_{140}) - f^\star \approx 0.11$ reflects
that the fixed-$\beta$ schedule does not decay, so the algorithm never takes the final
fine-grained steps needed to converge exactly to $(0.5, 0)$.  A diminishing step
schedule $\gamma_k = 2/(k+2)$ (standard Frank--Wolfe) would asymptotically close this gap.

\paragraph{Standard vs fixed-$\beta$ step schedule.}
The standard Frank--Wolfe algorithm uses $x_{k+1} = (1 - \gamma_k) x_k + \gamma_k z_k$
with $\gamma_k = 2/(k+2)$.  This schedule satisfies:
\begin{itemize}[nosep]
\item $\gamma_k \to 0$: the mixing weight decreases, so each LP vertex has progressively
  less influence.  This ensures the iterate can converge exactly to the optimum rather
  than oscillating.
\item $\sum_k \gamma_k = \infty$ (harmonic series): enough total weight is placed on LP
  vertices to ensure the iterate can move anywhere in $\mathcal{X}$.
\end{itemize}
The fixed-$\beta$ schedule used here ($\gamma_k = 1-\beta$ constant) violates the first
condition: the iterate $x_k$ always has a perturbation of magnitude $(1-\beta)\|x_k - z_k\|$
towards the LP vertex at each step.  Near the minimum where $z_k$ may differ from $x_k$
by order-of-diameter amounts, this causes persistent oscillation.  The trade-off is that
the fixed schedule takes larger steps early in training (faster initial progress) at the
cost of slower final convergence.  For the interior case with $\beta = 0.985$, the step
$1-\beta = 0.015$ is small enough that the oscillation amplitude ($\approx 0.015 \times
15.66 \approx 0.23$) is modest, explaining the better final accuracy compared to
$\beta = 0.90$ (amplitude $\approx 0.10 \times 15.66 \approx 1.57$).

\paragraph{Frank--Wolfe gap as a stopping criterion and duality certificate.}
A practically important feature of Frank--Wolfe is that the FW gap $g(x_k) = \nabla f(x_k)^T
(x_k - z_k)$ provides a computable \emph{upper bound} on the suboptimality:
$f(x_k) - f^\star \le g(x_k)$.  This bound requires no knowledge of $f^\star$ and can be
computed from existing quantities at each step.  For the interior case with $\beta = 0.90$
and 180 iterations, we compute the FW gap from the final gradient and LP vertex to certify
suboptimality.  This is a key advantage of Frank--Wolfe over projected gradient methods,
which do not naturally yield such a certificate.

The FW gap has a natural \emph{primal-dual} interpretation.  The dual problem of
$\min_{x \in \mathcal{X}} f(x)$ (via the linearisation relaxation) gives
$\max_{\lambda \ge 0} \min_{x \in \mathcal{X}} f(x_k) + \nabla f(x_k)^T(x - x_k)
= f(x_k) - g(x_k)$.
The gap $g(x_k)$ is therefore the \emph{duality gap} between the current primal objective
$f(x_k)$ and this dual lower bound.  When $g(x_k) \le \varepsilon$, the algorithm has
certified that $x_k$ is an $\varepsilon$-optimal solution.  For the interior case
($f(x) = (x_1-1)^2 + (x_2-5)^2$, $x^\star = (1,5)$) with $\beta = 0.985$ and
$k = 180$: the observed $f(x_{180}) \approx 0.0046$ means $g(x_{180}) \ge 0.0046$,
and the certificate confirms the gap is within 3 orders of magnitude of the theoretical
bound of $981/182 \approx 5.39$.  This enormous ratio (observed gap / theoretical bound)
quantifies how much tighter the practical convergence is compared to the worst-case theory.

\paragraph{Comparison summary.}
\begin{itemize}[nosep]
\item \textbf{When the optimum is interior:} Frank--Wolfe converges at $O(1/k)$ but may
  oscillate near the minimum due to LP vertex jumping.  Projected gradient methods with
  momentum (Nesterov) achieve $O(1/k^2)$ and are preferred if projection is cheap.
\item \textbf{When the optimum is on the boundary:} Frank--Wolfe naturally approaches the
  boundary without projection, making it competitive.  The fixed-$\beta$ schedule used
  here introduces residual oscillation; the standard $\gamma_k = 2/(k+2)$ decay eliminates
  this at the cost of smaller early steps.
\item \textbf{Constraint structure:} For general convex constraints (e.g.\ nuclear norm balls,
  positive semidefinite cones), the FW LMO may be computationally \emph{cheaper} than
  projection, making Frank--Wolfe the preferred choice.
\end{itemize}

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has systematically applied nine categories of optimisation algorithms to three
benchmark problems of increasing difficulty, drawing the following principal conclusions:

\begin{enumerate}[nosep]
\item \textbf{Benchmark sensitivity of adaptive methods (Q1):} Polyak's step size is
  optimal when $f^\star$ is exactly known but catastrophically fails when it is not.
  Adagrad and Heavy Ball are robust alternatives that outperform constant-rate GD on all
  benchmarks.  RMSprop's exponential forgetting mechanism prevents the dying step-size
  problem of Adagrad, making it preferable for long training runs.

\item \textbf{Momentum and the Rosenbrock challenge (Q2):} Nesterov acceleration provides
  the most significant improvement over GD on the ill-conditioned Rosenbrock function,
  directly reflecting its $O(1/k^2)$ convergence guarantee.  Adam's per-coordinate scaling
  is less effective on this problem because the dominant challenge is global momentum
  along the curved valley, not anisotropy.

\item \textbf{Newton's superiority on near-quadratic problems (Q3):} Newton's method
  converges in a single step on the quadratic Benchmark A and in just a few steps on
  Benchmark B.  Even with aggressive damping ($\alpha = 0.22$) on Rosenbrock, Newton
  outperforms GD by $5\times$ within a fraction of the iterations, at the cost of
  requiring Hessian computation.

\item \textbf{Gradient approximation quality matters (Q4):} Finite differences with a
  well-chosen $\delta$ can closely approximate exact GD at the cost of $d$ additional
  function evaluations per step.  Nelder--Mead demonstrates that gradient-free methods
  can navigate complex landscapes, but are limited to low dimensions.

\item \textbf{Constraint inactivity simplifies the constrained problem (Q5):} When the
  unconstrained optimum is feasible (as here for Benchmark B), all feasibility-enforcing
  methods converge to the same constrained minimum.  Projected GD enforces hard feasibility
  at every step; penalty methods trade exactness for implementation simplicity.

\item \textbf{Frank--Wolfe is projection-free but converges slowly (Q6):} By solving
  a linear subproblem at each step, Frank--Wolfe maintains feasibility without projection.
  The characteristic zig-zag of LP vertex jumps limits the convergence rate to $O(1/k)$,
  making it competitive primarily in settings where projections are expensive.
\end{enumerate}

Across all experiments, the Rosenbrock function consistently proved most challenging,
illustrating how ill-conditioning ($\kappa \approx 2000$) fundamentally limits first-order
methods and motivates second-order or momentum-based approaches.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak step size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum (lookahead)}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z          # extrapolation point
        g = grad_fn(lookahead)              # gradient at lookahead
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)     # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Newton's method with Hessian damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))   # Tikhonov regularisation
        try:
            p = np.linalg.solve(H_reg, g)       # Newton direction
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (conditional gradient)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LMO: box linear minimisation (per-coordinate)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z      # convex combination step
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
