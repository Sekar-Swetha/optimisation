\begin{abstract}
This report presents a comprehensive comparative study of first-order, second-order, and derivative-free optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark A, condition number $\kappa\approx 2.4$), a toy neural network loss with sinusoidal perturbation (Benchmark B, $\kappa\approx 5$--$7$), and the Rosenbrock function (Benchmark C, $\kappa\approx 2000$).

Six categories of methods are investigated across 180 total marks: (\textbf{Q1}) adaptive step-size methods --- Polyak, Adagrad, RMSprop, and Heavy Ball --- demonstrating that Polyak achieves optimal convergence ($f=0.009$ on Rosenbrock) when $f^\star$ is known exactly but diverges catastrophically otherwise; (\textbf{Q2}) momentum and stochastic methods --- Nesterov acceleration achieves the best Rosenbrock convergence among first-order methods ($f=0.479$, vs GD's $f=3.388$), while mini-batch SGD reveals that smaller batches ($b=5$) provide faster per-epoch convergence at the cost of gradient noise; (\textbf{Q3}) Newton's method with Hessian regularisation --- Newton converges to quadratic precision on Benchmark A in a \emph{single} iteration and outperforms 80 GD steps on Rosenbrock using only 20 iterations; (\textbf{Q4}) derivative-free approaches --- finite differences with $\delta=0.05$ closely tracks exact gradient descent, while Nelder--Mead successfully navigates the Rosenbrock valley without any gradient information; (\textbf{Q5}) constrained optimisation via projected gradient descent and penalty methods --- projected GD enforces feasibility from iteration~1 with zero constraint violation throughout; and (\textbf{Q6}) the Frank--Wolfe conditional gradient algorithm --- demonstrating the distinct convergence behaviour for interior versus boundary optima with its characteristic $O(1/k)$ convergence rate.

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) and are fully reproducible from the accompanying code file \texttt{CS7DS2\_Final\_Code\_25336453.py}.
\end{abstract}

\subsection{Mathematical Notation}

Throughout this report the following notation is used consistently. Let $f : \mathbb{R}^n \to \mathbb{R}$ denote the objective function. The gradient is $\nabla f(x) \in \mathbb{R}^n$ and the Hessian is $\nabla^2 f(x) \in \mathbb{R}^{n\times n}$. The operator $\odot$ denotes element-wise (Hadamard) multiplication. For a symmetric positive definite matrix $H$, $\lambda_{\min}(H)$ and $\lambda_{\max}(H)$ denote its smallest and largest eigenvalues respectively.

\begin{definition}[Smoothness and Strong Convexity]
A differentiable function $f$ is \emph{$L$-smooth} if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x-y\|$ for all $x,y$. It is \emph{$\mu$-strongly convex} if $f(y) \geq f(x) + \nabla f(x)^T(y-x) + \frac{\mu}{2}\|y-x\|^2$. The \emph{condition number} is $\kappa = L/\mu \geq 1$.
\end{definition}

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and $n = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a convex quadratic with constant Hessian $H = \frac{1}{m}X^T X \succ 0$, gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$, and smoothness constant $L = \lambda_{\max}(H)$, strong convexity $\mu = \lambda_{\min}(H)$. Due to observation noise, the irreducible minimum satisfies $J^{\star} \approx \sigma^2/2 = 0.5$ (empirically $J^\star \approx 0.4826$). Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
This function mimics a two-parameter neural network loss: the quadratic part $(x_1-1)^2 + 5(x_2-2)^2$ represents a well-conditioned regression-type loss, while $+\sin(x_1)$ adds non-quadratic (non-convex) perturbation that prevents the Hessian from being constant. The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is
\begin{equation}
H(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The $x_2$-coordinate has fixed curvature $10$, while the $x_1$-coordinate has spatially varying curvature $2 - \sin(x_1) \in [1, 3]$, giving a condition number $\kappa(x) \in [10/3, 10]$ depending on the iterate. This spatial variation in curvature is why Newton's method (which uses the local Hessian) outperforms gradient methods that must commit to a fixed global step size. The minimum satisfies $2(x_1-1) + \cos(x_1) = 0$, yielding $x_1^\star \approx 0.582$, $x_2^\star = 2.0$, with $f^\star \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is $(1, 1)$ with $f^\star = 0$. The Hessian at the optimum is
\begin{equation}
\nabla^2 f(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues approximately $2$ and $1002$ (condition number $\kappa \approx 501$; at the starting point $(-1,1)$, $\kappa \approx 2000$). The narrow curved valley $x_2 \approx x_1^2$ makes gradient-based methods prone to oscillation perpendicular to the valley. Starting point: $x_0 = [-1, 1]^T$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Background and Motivation}

Standard gradient descent with constant step size $\alpha$ satisfies
\begin{equation}
f(x_{k+1}) - f(x_k) \leq -\alpha\left(1 - \frac{\alpha L}{2}\right)\|\nabla f(x_k)\|^2
\end{equation}
for $L$-smooth $f$, requiring $\alpha < 2/L$ for descent. For $\mu$-strongly convex $f$, the optimal constant step $\alpha^\star = 2/(L+\mu)$ yields linear convergence at rate $(\kappa-1)/(\kappa+1)$. The challenge is that $L$ must be estimated or bounded, and using $\alpha \ll 2/L$ (to be safe) slows convergence proportionally. Adaptive methods aim to automatically estimate and apply the locally appropriate step size, sidestepping the need to know $L$ globally.

The practical motivation is clear in the context of deep learning: neural network loss functions have vastly different curvature across layers and weight coordinates. A single global step size is simultaneously too large for some coordinates (causing divergence) and too small for others (causing stagnation). Per-coordinate adaptive methods (Adagrad, RMSprop, Adam) address this by maintaining separate effective step sizes per coordinate.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Normalised subgradient step}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative squared gradient}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{Theoretical Properties}

\begin{proposition}[Polyak step convergence]
\label{prop:polyak}
For a convex function $f$ with known minimum value $f^\star$, the Polyak step $\alpha_k = (f(x_k) - f^\star)/\|\nabla f(x_k)\|^2$ satisfies
$\|x_{k+1} - x^\star\|^2 \leq \|x_k - x^\star\|^2 - (f(x_k) - f^\star)^2 / \|\nabla f(x_k)\|^2$.
Crucially, if the specified $\tilde{f}^\star < f^\star_{\mathrm{true}}$, then $\alpha_k \to \infty$ as $\|g_k\| \to 0$ while $f(x_k) - \tilde{f}^\star \to f^\star_{\mathrm{true}} - \tilde{f}^\star > 0$, causing divergence.
\end{proposition}

\begin{remark}[Adagrad monotone decay]
The effective per-coordinate step in Adagrad, $\alpha_0/(\sqrt{G_k^{(i)}} + \varepsilon)$, is monotonically decreasing in $k$ because $G_k^{(i)} = \sum_{j=0}^{k-1}(g_j^{(i)})^2$ is a non-decreasing accumulation. This guarantees stability but can cause premature stagnation on non-stationary loss surfaces.
\end{remark}

\begin{remark}[RMSprop vs Adagrad]
RMSprop replaces cumulative accumulation with an exponential moving average controlled by $\beta$. The stationary effective step $\alpha_0 / (\sqrt{(1-\beta^k)\,\overline{g^2}} + \varepsilon)$ does not decay to zero, enabling sustained learning on non-stationary problems. The decay rate $1-\beta$ determines the effective memory window: approximately $1/(1-\beta)$ recent gradients dominate.
\end{remark}

For the Heavy Ball method on a quadratic $f(x) = \frac{1}{2}x^T H x - b^T x$ with eigenvalues $\mu \leq \lambda \leq L$, the optimal hyperparameters satisfying the spectral radius condition are
\begin{equation}
\alpha_{\mathrm{opt}} = \left(\frac{2}{\sqrt{L} + \sqrt{\mu}}\right)^2, \qquad \beta_{\mathrm{opt}} = \left(\frac{\sqrt{L} - \sqrt{\mu}}{\sqrt{L} + \sqrt{\mu}}\right)^2,
\end{equation}
yielding a convergence rate of $\rho_{\mathrm{HB}} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$, compared to GD's $\rho_{\mathrm{GD}} = (\kappa-1)/(\kappa+1)$. For Benchmark A with $\kappa \approx 2.4$: $\rho_{\mathrm{GD}} \approx 0.41$, $\rho_{\mathrm{HB}} \approx 0.21$ --- a modest improvement because $\kappa$ is already small. For Rosenbrock near the start ($\kappa \approx 2000$): $\rho_{\mathrm{GD}} \approx 0.9990$, $\rho_{\mathrm{HB}} \approx 0.957$ --- a meaningful improvement, though both are still close to 1.

\subsection{Comparative Analysis and Discussion}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$ vs GD's $f = 3.51$, a $370\times$ improvement). Otherwise, Heavy Ball and Adagrad match or exceed GD. The optimal GD step size for a $\mu$-strongly convex, $L$-smooth function is $\alpha^\star = 2/(L+\mu)$, giving the convergence ratio $\rho = (\kappa-1)/(\kappa+1)$. For Benchmark A with $\kappa \approx 2.4$, $\rho \approx 0.41$, and for Rosenbrock with $\kappa \approx 2000$, $\rho \approx 0.9990$, explaining why GD barely progresses on Rosenbrock even with the optimal step.
\item \textbf{Stability:} GD is most stable but slowest. Polyak is \emph{unstable} when $f^{\star}$ is misspecified (Proposition~\ref{prop:polyak}). Adagrad is uniformly stable due to its monotonically decreasing step size (see the Adagrad Remark in Section~\ref{sec:q1}), though this prevents adaptation to non-stationary objectives. RMSprop's exponential forgetting provides stability without sacrificing adaptivity.
\item \textbf{Anisotropy handling:} Adagrad and RMSprop independently scale each coordinate, naturally handling anisotropic curvature by effectively preconditioning the gradient. GD and Heavy Ball use a single global step size, making them sensitive to ill-conditioning; the step size is limited by the largest curvature direction ($\alpha < 2/L$) even when other directions have much smaller curvature. This is why coordinate-adaptive methods can match GD on well-conditioned Benchmark A but do not separate dramatically on Rosenbrock: the extreme ill-conditioning requires fundamentally different curvature information (as Newton provides).
\item \textbf{Practical implication:} Exact $f^{\star}$ is almost never known in practice, severely limiting Polyak's applicability. RMSprop and Adagrad are more robust choices for machine learning where loss surfaces have heterogeneous curvature across coordinates (e.g., different layers of a neural network).
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This allows Nesterov momentum to ``correct'' its trajectory \emph{before} overshooting, unlike Heavy Ball which corrects \emph{after}. The increasing schedule $\beta_k = (k-1)/(k+2)$ ensures momentum builds up gradually.

\begin{theorem}[Nesterov Optimal Rate, \normalfont Nesterov 1983]
\label{thm:nesterov}
For any $L$-smooth convex function $f$ and step size $\alpha \leq 1/L$, the Nesterov accelerated method with the $(k-1)/(k+2)$ momentum schedule achieves
\begin{equation}
f(x_k) - f(x^\star) \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2} = O\!\left(\frac{1}{k^2}\right).
\end{equation}
This is optimal among first-order methods for smooth convex optimisation, matching the lower bound of Nemirovsky and Yudin.
\end{theorem}

In contrast, gradient descent achieves only $O(1/k)$ on convex functions and $O((1-\mu/L)^k)$ on strongly convex functions. The $O(1/k^2)$ rate represents a fundamental improvement for moderately ill-conditioned problems.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (EMA of gradient)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (EMA of $g^2$)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction for initialisation at 0}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum (via $m_t$) with RMSprop-style per-coordinate adaptive scaling (via $v_t$). The bias-correction factors $(1-\beta_1^t)^{-1}$ and $(1-\beta_2^t)^{-1}$ are essential in early iterations: without them, the EMA estimates are biased towards zero due to the zero initialisation, producing over-large effective step sizes. By iteration $k$, the correction factor for $\beta_2=0.999$ is $(1-0.999^k)^{-1}$, which equals approximately $1000$ at $k=1$ and approaches $1$ for $k \gg 1000$.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$ and batch sizes $b \in \{5, 40\}$ over 50 epochs, with data shuffled at every epoch. Each epoch processes all $m = 1000$ samples once, using $\lfloor m/b\rfloor$ gradient steps per epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}_j \subseteq \{1,\ldots,m\}$ with $|\mathcal{B}_j| = b$}
        \State $\hat{g} \gets \frac{1}{b} X_{\mathcal{B}_j}^T(X_{\mathcal{B}_j}\theta - y_{\mathcal{B}_j})$ \Comment{Stochastic gradient}
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The stochastic gradient $\hat{g}$ is an unbiased estimator of the true gradient: $\mathbb{E}[\hat{g}] = \nabla J(\theta)$, but with variance
\begin{equation}
\mathrm{Var}(\hat{g}) = \frac{1}{b} \cdot \mathrm{Var}\!\left(\frac{1}{m}\sum_{i=1}^{m}(x^{(i)T}\theta - y^{(i)})x^{(i)}\right) = O\!\left(\frac{\sigma^2}{b}\right),
\end{equation}
where $\sigma^2$ is the observation noise variance. Smaller batch sizes $b$ yield higher gradient variance but more updates per epoch. The expected loss after each SGD step satisfies
\begin{equation}
\mathbb{E}[J(\theta_{k+1})] \leq \mathbb{E}[J(\theta_k)] - \alpha\left(1 - \frac{\alpha L}{2}\right)\|\nabla J(\theta_k)\|^2 + \frac{\alpha^2 L \sigma^2}{2b},
\end{equation}
which reveals the fundamental SGD limitation: even as $\|\nabla J(\theta_k)\| \to 0$, the noise term $\alpha^2 L \sigma^2 / (2b)$ prevents convergence below the ``noise floor''. To reach $\varepsilon$-accuracy, one must either use $b = O(\sigma^2/\varepsilon)$ (large batch) or decay the step size $\alpha \to 0$.

\textbf{Learning rate schedules:} The experiments here use a fixed learning rate, which is suboptimal for convergence to the exact minimum. Common schedules for practical SGD include: (i) \emph{step decay} $\alpha_k = \alpha_0 \cdot \gamma^{\lfloor k/T\rfloor}$, (ii) \emph{cosine annealing} $\alpha_k = \alpha_{\min} + \frac{1}{2}(\alpha_0 - \alpha_{\min})(1 + \cos(\pi k/K))$, and (iii) \emph{polynomial decay} $\alpha_k = \alpha_0/(1 + ak)$. With a decaying schedule satisfying $\sum_k \alpha_k = \infty$ and $\sum_k \alpha_k^2 < \infty$ (the Robbins--Monro conditions), SGD converges to the exact minimiser even in the presence of gradient noise, albeit at a slower $O(1/\sqrt{k})$ rate compared to the linear convergence of full-batch GD. Fixed learning rates are used here for clarity of comparison, not as a practical recommendation.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.00391 + 0.0625\,(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
g_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.00391 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\end{equation}

The Newton step minimises $g_2(x)$: setting $g_2'(x) = 0$ gives $x_{k+1} = x_k - g'(x_k)/g''(x_k) = 0.25 - 0.0625/0.75 = 0.25 - 0.0833 = 0.167$. For $g(x) = x^4$, the global minimiser is $x^\star = 0$, so Newton's first step reduces $x$ from $0.25$ to $0.167$ (a $33\%$ reduction in $|x_k|$).

\subsection{Newton's Method with Damping}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, Hessian damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (via \texttt{np.linalg.solve})}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The damping $\lambda = 10^{-8}$ in the regularised Hessian $H_k + \lambda I$ serves two purposes: (i) it ensures positive definiteness when the Hessian is only positive semi-definite, and (ii) it improves numerical conditioning by bounding $\kappa(H_k + \lambda I) \leq \kappa(H_k)/(1 + \lambda/\lambda_{\min}(H_k))$. The step size $\alpha < 1$ (``damped Newton'') is necessary on non-quadratic functions where the quadratic model is only locally accurate.

\begin{theorem}[Local Quadratic Convergence of Newton's Method]
\label{thm:newton_quadratic}
Let $f$ be twice continuously differentiable near $x^\star$ with $\nabla f(x^\star) = 0$ and $\nabla^2 f(x^\star) \succ 0$. Assume the Hessian is Lipschitz continuous: $\|\nabla^2 f(x) - \nabla^2 f(y)\| \leq M\|x-y\|$. Then for any $x_0$ in a sufficiently small neighbourhood of $x^\star$, the full Newton method ($\alpha = 1$) satisfies
\begin{equation}
\|x_{k+1} - x^\star\| \leq \frac{M}{2\lambda_{\min}(\nabla^2 f(x^\star))}\,\|x_k - x^\star\|^2.
\end{equation}
This \emph{quadratic convergence} (doubling of significant digits per iteration) contrasts with the \emph{linear convergence} $\|x_{k+1} - x^\star\| \leq (1 - \mu/L)\|x_k - x^\star\|$ of gradient descent.
\end{theorem}

In terms of iteration complexity, Newton requires $O(\log\log(1/\varepsilon))$ iterations to achieve $\|x_k - x^\star\| \leq \varepsilon$ in the local quadratic convergence regime, versus $O(\kappa \log(1/\varepsilon))$ iterations for gradient descent.

\subsection{Practical Limitations of Newton's Method}

While Newton achieves superior convergence when the local quadratic model is accurate, three practical limitations must be acknowledged:

\begin{enumerate}[nosep]
\item \textbf{Cost per iteration}: Computing and inverting the $n\times n$ Hessian requires $O(n^2)$ storage and $O(n^3)$ arithmetic (via Cholesky factorisation), compared to $O(n)$ per step for gradient descent. For large $n$ (e.g.\ deep neural networks with $n > 10^6$ parameters), this is prohibitive even for a single step.
\item \textbf{Non-convex landscapes}: Newton's direction $p_k = H_k^{-1}g_k$ may be an \emph{ascent} direction if $H_k$ has negative eigenvalues (which occur at saddle points and in non-convex regions). Standard remedies include: (a) Hessian modification via $H_k + \lambda I$ with $\lambda > |\lambda_{\min}(H_k)|$ (used here with $\lambda = 10^{-8}$), (b) taking the absolute value of eigenvalues, or (c) falling back to the gradient direction. Rosenbrock has negative-curvature regions away from the valley, making these modifications important.
\item \textbf{Non-local accuracy}: The quadratic approximation degrades far from the current iterate, as observed on Rosenbrock where $\alpha = 0.22 \ll 1$ is necessary. Line search (e.g., Armijo backtracking) automatically determines a step size satisfying the sufficient decrease condition, providing a principled alternative to fixed damping.
\end{enumerate}

\textbf{Quasi-Newton methods} (notably L-BFGS) address the cost issue by approximating $H_k^{-1}$ from $m$ most recent gradient difference pairs $\{(s_j, y_j)\}_{j=k-m}^{k-1}$ where $s_j = x_{j+1}-x_j$ and $y_j = \nabla f(x_{j+1}) - \nabla f(x_j)$. L-BFGS achieves super-linear convergence at $O(mn)$ cost per iteration (with $m \approx 5$--$20$), making it the method of choice for large-scale smooth optimisation in practice. The implementation used here (full Newton with $O(n^3)$ solve) is appropriate only for small $n$ (here $n=2$).

\textbf{Gauss-Newton for least squares:} For the specific case of Benchmark A (least-squares loss $J(\theta) = \frac{1}{2m}\|X\theta - y\|^2$), the Hessian $\nabla^2 J = X^TX/m$ is constant and exactly the Gauss-Newton matrix $J_r^T J_r$ where $J_r = X/\sqrt{m}$ is the Jacobian of the residual. The Gauss-Newton method (solving $(J_r^T J_r) p = J_r^T r$ at each step) therefore coincides with Newton's method for linear models, explaining the one-step convergence on Benchmark A. For nonlinear least-squares problems, the Gauss-Newton approximation drops the second-order term $\sum_i r_i \nabla^2 r_i$, which is small when residuals are small near the solution, yielding the Levenberg-Marquardt algorithm widely used in nonlinear curve fitting.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad e_i = i\text{-th standard basis vector}.
\end{equation}
By Taylor's theorem, the truncation error is:
\begin{equation}
\left|\frac{\partial f}{\partial x_i}(x) - \frac{f(x + \delta e_i) - f(x)}{\delta}\right| \leq \frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(x)\right| + O(\delta^2).
\end{equation}
Simultaneously, floating-point rounding introduces an error of order $\varepsilon_{\mathrm{mach}}/\delta$ (where $\varepsilon_{\mathrm{mach}} \approx 10^{-16}$ for double precision). Balancing these gives an optimal $\delta \approx \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$ for forward differences. The assignment tests:
\begin{itemize}[nosep]
  \item $\delta = 0.05$: small enough for good accuracy ($\delta/2 \cdot |f''| \approx 0.025 \cdot 2 \approx 0.05$ error in gradient), close to optimal for the problem scale.
  \item $\delta = 0.8$: too large --- truncation error $\approx 0.4 \cdot |f''|$ dominates, producing a biased gradient direction.
\end{itemize}

\subsubsection{Nesterov Random Search (Zeroth-Order Method)}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation radius $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_n)$, normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Uniform on $\mathbb{S}^{n-1}$}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates $\nabla f(x_k)^T u_k \cdot u_k$, a noisy projection of the gradient onto a random direction. Critically, it requires only \emph{two} function evaluations per iteration regardless of dimension $n$, making it dimension-free in per-step cost.

\begin{proposition}[Nesterov Random Search bias]
The estimate $\hat{g}_k = \frac{f(x+\delta u) - f(x)}{\delta} u$ satisfies
$\mathbb{E}_{u}[\hat{g}_k] = \frac{1}{n}\nabla f(x) + O(\delta)$,
since $\mathbb{E}[u_i^2] = 1/n$ for a uniform random direction $u$. Thus $\hat{g}_k$ is a biased estimator of $\nabla f(x)/n$, requiring the step size to be scaled accordingly. The variance is $O(n\|\nabla f\|^2)$, so convergence in $\varepsilon$-optimality requires $O(n/\varepsilon^2)$ function evaluations, scaling linearly with dimension.
\end{proposition}

In contrast, finite-difference GD requires $n+1$ function evaluations per step (one per coordinate plus a base evaluation) but uses a high-fidelity gradient estimate with variance $O(\delta^2)$. For $n=2$, both methods use a similar number of evaluations per step, explaining why NRS is competitive on Benchmark B despite using a random direction.

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four geometric operations:

\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\rho_r = 1.0$): move worst vertex $x_w$ through the centroid $\bar{x}$ of the remaining vertices to $x_r = \bar{x} + \rho_r(\bar{x} - x_w)$.
\item \textbf{Expansion} ($\gamma = 2.0$): if reflection improved, try $x_e = \bar{x} + \gamma(\bar{x} - x_w)$; keep the better of $x_r, x_e$.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection did not improve, contract towards $\bar{x}$.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction failed, contract all vertices towards the best vertex.
\end{enumerate}

These coefficients $(\rho_r, \gamma, \rho, \sigma) = (1.0, 2.0, 0.5, 0.5)$ are the standard Nelder--Mead values used in the implementation. Settings: initial simplex step $= 0.35$, 160 iterations.

\textbf{Why Nelder--Mead works:} The simplex is a geometrically adaptive search pattern. When the reflection of the worst vertex produces improvement, the simplex ``rolls'' along the gradient direction. When no single vertex movement helps, the shrink operation contracts the entire simplex, concentrating it in a region of improvement. On the Rosenbrock valley, the simplex first stretches along the valley axis (reflection and expansion accepted repeatedly) then contracts transversally (contraction accepted as the simplex aligns with the valley). This geometric adaptation achieves gradient-free descent without any finite difference computation.

\textbf{Known limitations:} Nelder--Mead is known to fail (non-convergence or convergence to non-stationary points) on problems with $d > 10$ or with non-smooth objectives, since the simplex can become degenerate (all vertices coplanar) or too small to detect the gradient direction. Theoretical convergence guarantees exist only for $d = 1$; for $d \geq 2$, convergence is not guaranteed in general.

\subsection{Problem Setup}

Minimise $f(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)$ (Benchmark B) subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$. The unconstrained minimiser is $x^\star_{\mathrm{unc}} \approx (0.582, 2.0)$, which satisfies $x_1^\star = 0.582 > 0.5$, so the constrained and unconstrained optima coincide: $x^\star_{\mathrm{con}} = x^\star_{\mathrm{unc}} \approx (0.582, 2.0)$ with $f^\star = 0.7244$.

This is a special case where the constraint is \emph{inactive} at the solution. However, the starting point is infeasible ($x_0^{(1)} = 0.2 < 0.5$), making the initial feasibility behaviour an important test.

\begin{definition}[KKT Conditions]
\label{def:kkt}
For the problem $\min f(x)$ s.t. $g_j(x) \leq 0$, a point $x^\star$ satisfies the Karush--Kuhn--Tucker (KKT) conditions if:
\begin{align}
\nabla f(x^\star) + \sum_j \mu_j \nabla g_j(x^\star) &= 0 \quad \text{(stationarity)}, \\
\mu_j g_j(x^\star) &= 0 \quad \text{(complementary slackness)}, \\
\mu_j &\geq 0, \quad g_j(x^\star) \leq 0 \quad \text{(dual feasibility / primal feasibility)}.
\end{align}
For our problem with $g(x) = 0.5 - x_1 \leq 0$: at $x^\star = (0.582, 2.0)$, $g(x^\star) = -0.082 < 0$ (strictly feasible), so by complementary slackness $\mu = 0$, and the KKT condition reduces to $\nabla f(x^\star) = 0$ --- the constrained optimum coincides with the unconstrained optimum.
\end{definition}

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Gradient step (may leave $\mathcal{X}$)}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = [\max(0.5, z_{k+1}^{(1)}),\; z_{k+1}^{(2)}]^T$ \Comment{Project back to $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Non-expansiveness of Projection]
\label{thm:projection}
The orthogonal projection $\Pi_\mathcal{X}$ onto a closed convex set $\mathcal{X}$ is non-expansive: $\|\Pi_\mathcal{X}(x) - \Pi_\mathcal{X}(y)\| \leq \|x - y\|$ for all $x, y \in \mathbb{R}^n$. Moreover, $\Pi_\mathcal{X}(x)$ is the unique closest point to $x$ in $\mathcal{X}$, and for our halfspace constraint $\{x : x_1 \geq 0.5\}$ it reduces to the trivial operation $[\Pi_\mathcal{X}(z)]_1 = \max(0.5, z_1)$.
\end{theorem}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\State Define $F_\lambda(x) \gets f(x) + \lambda \max(0,\; 0.5 - x_1)$
\For{$k = 0, 1, \ldots, N-1$}
    \State $\nabla F_\lambda(x) \gets \nabla f(x) + \lambda \cdot \mathbf{1}[x_1 < 0.5] \cdot [-1, 0]^T$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{Hyperparameter Settings}

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

The reduction in step size from $\alpha = 0.05$ to $\alpha = 0.03$ for $\lambda = 4.5$ is necessary because the penalty gradient $\nabla F_\lambda$ has a large magnitude near the constraint boundary (approximately $\lambda = 4.5$ more than $\nabla f$ when $x_1 < 0.5$), requiring a smaller step to maintain stability.

\textbf{Projected GD} enforces feasibility as a hard constraint from the first iteration by the non-expansiveness of the projection operator (Theorem~\ref{thm:projection}). \textbf{Penalty methods} trade exact feasibility for algorithmic simplicity. The penalised objective $F_\lambda(x) = f(x) + \lambda g_+(x)$ (where $g_+(x) = \max(0, 0.5-x_1)$) approximates the constrained problem with accuracy $O(1/\lambda)$ --- specifically, the gap between the unconstrained minimiser of $F_\lambda$ and the constrained minimiser of $f$ satisfies $f(x^\star_\lambda) - f(x^\star) = O(1/\lambda)$. \textbf{Augmented Lagrangian (Method of Multipliers):} The augmented Lagrangian method resolves this tension by combining a quadratic penalty with a Lagrange multiplier estimate $\mu$:
\begin{equation}
\mathcal{L}_\lambda(x, \mu) = f(x) + \mu \cdot g_+(x) + \tfrac{\lambda}{2}[g_+(x)]^2.
\end{equation}
Alternating between minimising $\mathcal{L}_\lambda(\cdot, \mu_k)$ and updating $\mu_{k+1} = \mu_k + \lambda \cdot g_+(x_{k+1})$ achieves exact feasibility even at finite $\lambda$ (typically $\lambda = 10$), avoiding the ill-conditioning of pure penalty methods. Augmented Lagrangian is the foundation of modern ADMM (Alternating Direction Method of Multipliers) algorithms widely used in distributed optimisation and signal processing.

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm \hfill [30 marks]}
\label{sec:q6}

\subsection{The Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a bounded box / polytope with 4 vertices).

\textbf{Linear Programme (Q6-I):} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$.

Since $a_1 = 1 > 0$ and $a_2 = 2 > 0$, the gradient $\nabla f = a$ points in the direction of increasing $f$, so the minimum is at the coordinate-wise lower bounds:
\begin{equation}
x^{\star} = \argmin_{x \in \mathcal{X}} (x_1 + 2x_2) = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates the \textbf{fundamental theorem of linear programming}: the optimal solution of a linear objective over a bounded polytope is always attained at a vertex (extreme point). Here the box $\mathcal{X}$ has 4 vertices: $(0.5,-5)$, $(5,-5)$, $(5,10)$, $(0.5,10)$ with objective values $-9.5$, $15$, $25$, $20.5$ respectively, confirming $(0.5,-5)$ is optimal.

\textbf{LP duality:} The dual problem is $\max_{y \geq 0} [-b^T y : A^T y = a]$ where $Ax \leq b$ encodes the box constraints. Strong duality (LP satisfies Slater's condition since the interior of $\mathcal{X}$ is non-empty) guarantees the primal and dual optima are equal. The dual variables $y^\star$ represent the shadow prices of each constraint --- a unit tightening of the active constraints at $x^\star = (0.5, -5)$ changes $f^\star$ by $y^\star_1 \cdot 1 + y^\star_4 \cdot 2 = 1 + 4 = 5$.

The contour lines are perpendicular to the gradient $a = [1,2]^T$, and the minimum occurs at the lower-left corner $(0.5, -5)$ where the contour achieves the smallest value $f = -9.5$ within $\mathcal{X}$. \subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial feasible point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \displaystyle\argmin_{x \in \mathcal{X}}\; \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LP subproblem)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination $\Rightarrow$ $x_{k+1} \in \mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Key property:} Since $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$, and $\mathcal{X}$ is convex, $x_{k+1} = \beta x_k + (1-\beta)z_k \in \mathcal{X}$ for all $k$. This means Frank--Wolfe \textbf{never requires a projection}, instead maintaining feasibility through convex combinations. The trade-off is that iterates can only move towards vertices of $\mathcal{X}$ (the LP solutions $z_k$), never beyond.

For box constraints, the LP subproblem has a closed-form solution:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations per iteration.

\begin{theorem}[Frank--Wolfe Convergence Rate]
\label{thm:fw}
Let $f$ be $L$-smooth and convex, and let $\mathcal{X}$ be a compact convex set with diameter $D = \max_{x,y \in \mathcal{X}}\|x-y\|$. With the Frank--Wolfe step size $\gamma_k = 2/(k+2)$ (equivalent to $\beta_k = k/(k+2)$), the algorithm satisfies
\begin{equation}
f(x_k) - f(x^\star) \leq \frac{2LD^2}{k+2} = O\!\left(\frac{1}{k}\right).
\end{equation}
This $O(1/k)$ rate is \textbf{suboptimal} compared to projected gradient descent's $O(1/k)$ for convex and $O((1-\mu/L)^k)$ for strongly convex objectives, reflecting the constraint imposed by the LP vertex structure on feasible step directions.
\end{theorem}

The \textbf{Frank--Wolfe gap} $g_k = \nabla f(x_k)^T(x_k - z_k)$ provides an upper bound on the optimality gap: $f(x_k) - f(x^\star) \leq g_k$, making it a useful stopping criterion without knowing $f^\star$.

\subsection{Interior vs Boundary Convergence Behaviour}

\textbf{Interior optimum:} When $x^\star \in \mathrm{int}(\mathcal{X})$, Frank--Wolfe exhibits the characteristic ``zig-zag'' oscillation near the optimum. As $x_k \to x^\star$, the gradient $\nabla f(x_k) \to 0$ but the LP oracle still selects a vertex $z_k$ at maximum distance from $x^\star$. The convex combination $\beta x_k + (1-\beta) z_k$ therefore always slightly overshoots $x^\star$ in the direction of $z_k$. This explains why the convergence slows dramatically near the optimum. The convergence rate is $O(1/k)$ by Theorem~\ref{thm:fw}; however, the constant factor can be large due to the vertex oscillation.

\textbf{Boundary optimum:} When $x^\star \in \partial\mathcal{X}$ (on the boundary), Frank--Wolfe is more naturally suited: the LP oracle $z_k$ tends to select the same vertex near $x^\star$ consistently (since $\nabla f$ points consistently away from $\mathcal{X}$), leading to monotone convergence without zig-zagging. The gap between the achieved $f = 0.364$ and $f^\star = 0.25$ after 140 iterations reflects the slow $O(1/k)$ rate with fixed $\beta = 0.93$. Using the optimal schedule $\gamma_k = 2/(k+2)$ (i.e., $\beta_k = k/(k+2)$) would guarantee the theoretical $O(1/k)$ bound of Theorem~\ref{thm:fw}.

\textbf{Projection-free advantage:} Frank--Wolfe's primary advantage over projected gradient descent is the absence of a projection step. For complex constraint sets (e.g., the nuclear norm ball $\{X : \|X\|_* \leq \tau\}$ or the positive semidefinite cone $\{X \succeq 0, \mathrm{tr}(X) \leq 1\}$), projection requires computing a full matrix eigendecomposition at $O(n^3)$ cost, while the corresponding LP subproblem only needs the top singular vector at $O(n^2)$ cost. This makes Frank--Wolfe the algorithm of choice for matrix-valued optimisation problems in machine learning (e.g., low-rank matrix completion, trace norm regularisation).

For the simple box constraint here, both methods have the same per-iteration cost, so Frank--Wolfe's $O(1/k)$ rate versus PGD's faster $O((1-\mu/L)^k)$ rate makes PGD preferable for this specific problem. The \textbf{away-step Frank--Wolfe} variant addresses the slow convergence for interior optima by also allowing steps away from the vertices already in the current active set, achieving a linear convergence rate for strongly convex objectives, though at the cost of maintaining a more complex iterate representation.

\subsection{Key Insights}

\textbf{(1) The importance of curvature information.} The Rosenbrock function, with its $\kappa \approx 2000$ condition number, dramatically separates methods. Gradient descent barely progresses ($f = 3.51$), momentum-based methods improve substantially (Nesterov: $f = 0.479$), and Newton's method reaches $f = 0.686$ in only 20 iterations. The fundamental reason is that ill-conditioning creates a narrow valley where gradient directions are nearly perpendicular to the descent direction, and all gradient-based methods require curvature information (explicit or implicit) to navigate efficiently.

\textbf{(2) Adaptivity is double-edged.} Adaptive methods (Adagrad, RMSprop, Adam) handle heterogeneous curvature well on Benchmark B but do not consistently outperform simpler methods. Adagrad's monotonically decaying step size can be too conservative on ill-conditioned problems (Rosenbrock). Adam's combination of momentum and adaptivity is powerful in practice but theoretically complex, lacking global convergence guarantees without additional assumptions.

\textbf{(3) Prior knowledge is decisive.} The Polyak step size demonstrates that prior knowledge of $f^\star$ can yield the best convergence ($f = 0.009$ on Rosenbrock), but that the same method diverges when the prior is wrong. This parallels the broader principle in optimisation: well-calibrated problem-specific information dramatically outperforms general-purpose methods, but is fragile to misspecification.

\textbf{(4) Derivative-free methods scale poorly.} Grid search's $O(N^d)$ cost makes it infeasible for $d > 3$. Nelder--Mead, while effective in low dimensions, is known to fail on $d > 10$ problems. Nesterov random search scales to high dimensions at the cost of slow ($O(n/\varepsilon^2)$) sample complexity. When gradients are available, they should be used.

\textbf{(5) Constraint handling strategy depends on the projection cost.} Projected gradient descent provides exact feasibility at every iteration at the cost of a projection oracle. For simple constraint sets (boxes, halfspaces, $\ell_1$/$\ell_2$ balls), projection is cheap and PGD is preferred. For structured polytopes where the LP subproblem is cheaper, Frank--Wolfe is competitive despite its $O(1/k)$ rate. Penalty methods are most appropriate when neither projection nor LP is tractable, accepting approximate feasibility in exchange for unconstrained gradient access.

\subsection{Practical Recommendations}

\begin{itemize}[nosep]
\item \textbf{Well-conditioned problems ($\kappa < 10$, large-scale):} Adam or RMSprop --- robust, fast, minimal tuning. Benchmark A demonstrates that even GD converges quickly for small $\kappa$.
\item \textbf{Moderately ill-conditioned problems ($10 \leq \kappa \leq 100$):} Nesterov momentum with the $(k-1)/(k+2)$ schedule --- theoretical $O(1/k^2)$ guarantee, no Hessian needed.
\item \textbf{Severely ill-conditioned or nonlinear problems ($\kappa > 100$, small-scale):} Damped Newton ($n$ small) or L-BFGS ($n$ large) --- curvature information is essential. Rosenbrock demonstrates that all first-order methods fail when $\kappa \approx 2000$.
\item \textbf{When $f^\star$ is known exactly:} Polyak step size provides optimal per-iteration progress without learning rate tuning. Particularly useful for fixed-point iterations and certain structured problems.
\item \textbf{Derivative-free with limited budget:} Nelder--Mead for $d \leq 5$ (robust, no gradient); Nesterov random search for $d > 5$ (dimension-free per-step cost). Grid search only for global search with $d \leq 2$ and coarse resolution requirements.
\item \textbf{Constrained problems (simple convex sets):} Projected GD if projection is cheap (boxes, balls, simplices). Frank--Wolfe if the LP subproblem is cheaper than projection (trace-norm balls, structured polytopes). Penalty methods if neither oracle is available.
\item \textbf{Stochastic/mini-batch setting:} Use Adam with $\beta_1 = 0.9$, $\beta_2 = 0.999$ as the default. Decay the learning rate when approaching convergence. For very large batches or distributed training, consider SGD with momentum and learning rate warmup.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Damped Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (Conditional Gradient)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
