\begin{abstract}
This report presents a rigorous comparative study of first-order, second-order, and derivative-free
optimisation algorithms applied to three benchmark problems of increasing difficulty: a nearly
isotropic linear regression quadratic loss (condition number $\kappa \approx 1.12$), a mildly
anisotropic toy neural network loss with sinusoidal perturbation ($\kappa \approx 6.90$), and the
ill-conditioned Rosenbrock function ($\kappa \approx 2508$). Six algorithm families are investigated:
(1)~adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), which reduce sensitivity to
the global Lipschitz constant; (2)~accelerated and adaptive momentum methods (Nesterov acceleration,
Adam), achieving superior convergence rates; (3)~second-order Newton's method, achieving
one-step convergence on quadratics and quadratic convergence near smooth minima;
(4)~derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead, grid
search), circumventing gradient computation at the cost of convergence rate; (5)~constrained
optimisation via projected gradient descent and quadratic penalty methods; and (6)~the Frank--Wolfe
conditional gradient algorithm for linear-constrained problems. Key findings: Polyak step sizes
fail when $f^\star$ is misspecified, causing overshooting rather than divergence; Newton's method
achieves exact one-step convergence on the quadratic benchmark; Nesterov momentum outperforms Adam
on the highly non-quadratic Rosenbrock function; and the Frank--Wolfe algorithm converges at an
$O(1/k)$ rate with sublinear dependence on feasible-set diameter.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.
Starting function values are: $f_A(\theta_0) = 13.11$, $f_B(x_0) = 23.16$, $f_C(x_0) = 4.00$.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \in \mathbb{R}^2, \quad
y^{(i)} = (\theta^{\star})^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss is:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strongly convex quadratic with constant Hessian $H = \frac{1}{m}X^T X$. The analytical
solution is $\theta^{\star}_{\mathrm{OLS}} = H^{-1}(X^Ty/m) \approx (3.028,\, 4.017)^T$.
Numerical computation with the given seed yields eigenvalues $\lambda_1 \approx 0.9247$ and
$\lambda_2 \approx 1.0325$, giving condition number $\kappa = \lambda_2/\lambda_1 \approx 1.12$
and Lipschitz constant $L = \lambda_2 \approx 1.033$. The near-unity condition number makes
Benchmark A essentially isotropic --- all first-order methods should converge at nearly identical
rates. Due to observation noise, the minimum loss is $J^{\star} \approx 0.4826$ rather than zero.
Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\; 10(x_2 - 2)]^T$ and the Hessian is
$H(x) = \mathrm{diag}(2 - \sin(x_1),\; 10)$. Numerical minimisation yields $x^{\star} \approx
(0.582, 2.000)$ with $f^{\star} \approx 0.7244$. At the minimum, the Hessian eigenvalues are
approximately $\lambda_1 \approx 1.450$ and $\lambda_2 = 10$, giving $\kappa \approx 6.90$. The
sinusoidal term introduces mild non-quadratic curvature (the $x_1$ component of $\nabla^2 f$
varies between $2 - 1 = 1$ and $2 + 1 = 3$ over the domain), making Benchmark B a challenging
test for methods that rely on quadratic approximations. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is
\begin{equation}
\nabla^2 f(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}
\end{equation}
with eigenvalues $\lambda_1 \approx 0.399$ and $\lambda_2 \approx 1001.6$, giving $\kappa \approx
2508$. At the starting point $x_0 = (-1, 1)$, the Hessian has eigenvalues approximately $[0.399,\,
1001.6]$, implying a stable gradient descent step size of $\alpha \leq 1/L \approx 0.001$. This
extreme ill-conditioning, combined with the narrow curved valley $x_2 \approx x_1^2$ along which
the gradient points almost perpendicular to the direction of minimum traversal, makes Rosenbrock
the most stringent of the three benchmarks. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems, the feasible set is the axis-aligned box:
\[
\mathcal{X} = \bigl\{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\bigr\}.
\]

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive step-size methods are compared against a fixed-step gradient descent baseline across
all three benchmarks for 120 iterations. The motivation for adaptive methods is the difficulty of
choosing a fixed step size: too large causes divergence, too small causes slow convergence. Adaptive
methods couple the step size to local gradient information.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step is motivated by the inequality $f(x_{k+1}) \leq f(x_k) - \alpha_k \|g_k\|^2 +
\frac{L}{2}\alpha_k^2 \|g_k\|^2$, which is minimised (over $\alpha_k$) by the \emph{optimal}
step size $\alpha_k^{\mathrm{opt}} = \frac{f(x_k) - f^\star}{L\|g_k\|^2}$. The Polyak rule
approximates this optimal step without knowing $L$, instead requiring knowledge of $f^\star$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$
    \Comment{Cumulative sum of squared gradients (per coordinate)}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad automatically scales the learning rate inversely by the square root of historical gradient
magnitudes. Coordinates with consistently large gradients receive smaller effective steps; rarely
updated coordinates receive larger steps. The cumulative sum $G_k$ is monotonically non-decreasing,
which means the effective step size $\alpha_0/\sqrt{G_k}$ is monotonically non-increasing and
converges to zero --- a known limitation for non-stationary problems. Adagrad achieves $O(1/\sqrt{k})$
regret on convex online problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$
    \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop remedies Adagrad's monotone step-decay by replacing the cumulative sum with an exponential
moving average. The effective memory horizon is approximately $1/(1-\beta)$ iterations; with
$\beta = 0.9$ this corresponds to approximately 10 recent gradient observations. Unlike Adagrad,
the effective step size can \emph{increase} if recent gradients are smaller than historical ones.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

Heavy Ball is equivalent to introducing a ``velocity'' variable $z_k$ that accumulates gradient
information with exponential decay. For strongly convex quadratic functions with Lipschitz constant
$L$ and strong convexity constant $\mu$, the optimal parameters $\alpha^* = 4/({\sqrt{L}+\sqrt{\mu}})^2$
and $\beta^* = ((\sqrt{L}-\sqrt{\mu})/(\sqrt{L}+\sqrt{\mu}))^2$ yield a linear convergence rate
with constant $(({\sqrt{\kappa}-1})/({\sqrt{\kappa}+1}))^2$, dramatically faster than gradient
descent's rate of $(1 - 2/(\kappa+1))^2$ when $\kappa \gg 1$.

\subsection{Comparative Analysis and Theoretical Interpretation}

\begin{itemize}[nosep]
\item \textbf{Polyak:} Achieves the theoretically optimal step for convex problems when $f^\star$
is known exactly (Rosenbrock: $f \approx 0.024$). When $f^\star$ is misspecified, the step grows
unboundedly near the iterate's limiting point, causing persistent oscillation. This makes Polyak
inapplicable to problems where the global minimum is unknown (the common case).

\item \textbf{Adagrad:} Universally stable due to monotonically decreasing effective step sizes.
Matches GD on well-conditioned problems (Benchmarks A and B) and outperforms GD on the
ill-conditioned Rosenbrock despite its slower step decay. The implicit normalisation by cumulative
gradient magnitude provides a form of per-coordinate curvature adaptation. Limitation: premature
step-size decay on long optimisation runs.

\item \textbf{RMSprop:} More flexible than Adagrad through its decay factor $\beta = 0.9$,
maintaining adaptivity over the most recent $\approx 10$ gradient steps. However, the need to use
a small base rate (especially on Rosenbrock, $\alpha_0 = 0.0035$) limits convergence speed.

\item \textbf{Heavy Ball:} Best first-order method on Rosenbrock ($f = 1.40$) thanks to momentum
accumulation along the consistent valley-floor gradient direction. The momentum term $z_k$
effectively accumulates the gradient signal along the principal direction of the valley, partially
overcoming the ill-conditioning.

\item \textbf{Practical recommendation:} Adagrad and Heavy Ball offer the best robustness across
diverse problem types. Polyak is optimal only when $f^\star$ is known precisely.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum (Accelerated Gradient Descent)}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\bigl(\tfrac{k-1}{k+2},\; \beta_{\max}\bigr)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The crucial innovation over Heavy Ball is evaluating the gradient at the \emph{lookahead point}
$\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This ``corrected'' gradient allows
Nesterov acceleration to anticipate the effect of momentum and correct its trajectory before
overshooting. The adaptive schedule $\beta_k = (k-1)/(k+2) \to 1$ gradually increases momentum
as the algorithm progresses, reaching a maximum of $\beta_{\max}$. On smooth convex functions,
this achieves the optimal $O(1/k^2)$ convergence rate, compared to $O(1/k)$ for standard
gradient descent and Heavy Ball.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$
    \Comment{First moment: exponential MA of gradient}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$
    \Comment{Second moment: exponential MA of squared gradient}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$
    \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t / (1 - \beta_2^t)$
    \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t \mathbin{/} (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam unifies the adaptive scaling of RMSprop (through $v_t$) with gradient momentum (through
$m_t$). The bias correction terms $1/(1 - \beta_i^t)$ are essential in early iterations: without
correction, initialising $m_0 = v_0 = 0$ would bias the estimates towards zero. With the standard
defaults $\beta_1 = 0.9$, $\beta_2 = 0.999$, Adam effectively maintains a long gradient history
(1000 steps for second moment) combined with a short momentum memory (10 steps for first moment).
The effective per-coordinate step size is bounded: $|\Delta x_t^{(i)}| \leq \alpha \cdot
(1-\beta_2^t)^{-1/2} / (1-\beta_1^t)$, providing uniform update magnitude bounds.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses a constant learning rate $\alpha = 0.06$ and is run for 50
epochs. At each epoch, the data are randomly shuffled and processed in mini-batches of size $b$.
The gradient estimator for a mini-batch $\mathcal{B}$ is:
\begin{equation}
\hat{g}(\theta) = \frac{1}{b} X_{\mathcal{B}}^T (X_{\mathcal{B}}\theta - y_{\mathcal{B}}).
\end{equation}
The variance of this estimator satisfies $\mathrm{Var}[\hat{g}] \propto \sigma_y^2/b$, where
$\sigma_y^2$ is the observation noise variance. The key trade-off is between \emph{update
frequency} (more frequent updates with small $b$) and \emph{gradient quality} (lower variance with
large $b$).

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices (shuffle)
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q2_sgd_batch.pdf}
\caption{Q2: Mini-batch SGD convergence for batch sizes $b \in \{5, 40\}$. With $b = 5$, each
epoch performs $m/b = 1000/5 = 200$ parameter updates, providing rapid per-epoch progress but

updates, resulting in smoother but slower per-epoch convergence. Both converge to the noise floor

\subsection{Local Approximations of $g(x) = x^4$}

Around an expansion point $x_0 = 0.25$, with $g(x_0) = 0.25^4 \approx 0.00391$,
$g'(x_0) = 4 \times 0.25^3 = 0.0625$, and $g''(x_0) = 12 \times 0.25^2 = 0.75$:

\textbf{First-order (gradient/tangent) approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.00391 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic/Newton) approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.00391 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton step minimises $\hat{g}_2$: setting $\partial \hat{g}_2 / \partial x = 0$ gives
$x_1 = x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 \approx 0.167$. Applying this iteratively
yields quadratic convergence: $|x_{k+1} - x^\star| \leq C |x_k - x^\star|^2$ for some constant
$C > 0$, where $x^\star = 0$ for $g(x) = x^4$.

\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$. The first-order approximation
(tangent line) is accurate only infinitesimally near $x_0$. The second-order approximation
(upward parabola) captures the curvature and remains accurate over a broader neighbourhood. The

\subsection{Newton's Method with Damping}

\begin{algorithm}[H]
\caption{Newton's Method with Regularised Hessian}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, Hessian regularisation $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$
    \Comment{$\lambda I$: regularisation to ensure positive definiteness}
    \State $p_k \gets H_k^{-1}\, g_k$
    \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
    \Comment{$\alpha = 1$: pure Newton; $\alpha < 1$: damped Newton}
\EndFor
\end{algorithmic}
\end{algorithm}

Pure Newton ($\alpha = 1$) on strictly convex quadratic problems is exact in one step. For general
smooth functions, quadratic convergence holds in a neighbourhood of the minimiser: there exist
$\delta, C > 0$ such that if $\|x_0 - x^\star\| \leq \delta$, then $\|x_{k+1} - x^\star\| \leq
C \|x_k - x^\star\|^2$. The damping factor $\alpha < 1$ extends convergence to larger basins of
attraction at the cost of convergence rate. Regularisation $\lambda > 0$ ensures the Hessian is
positive definite even when the true Hessian is indefinite or near-singular. Here $\lambda = 10^{-8}$
(essentially numerical regularisation only).

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}

\subsection{Part A: Finite Differences and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector. For a $C^2$ function, the truncation error is
$O(\delta)$ from Taylor's theorem:
\begin{equation}
\frac{f(x + \delta e_i) - f(x)}{\delta} = \frac{\partial f}{\partial x_i}(x) + \frac{\delta}{2}
\frac{\partial^2 f}{\partial x_i^2}(x) + O(\delta^2).
\end{equation}
The total approximation error is $O(\delta + \varepsilon_{\mathrm{mach}}/\delta)$, minimised at
$\delta_{\mathrm{opt}} \sim \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$. This experiment
tests $\delta = 0.05$ (good: error $\approx 0.025$) versus $\delta = 0.8$ (poor: error $\approx
0.4$). Each gradient computation costs $d + 1 = 3$ function evaluations for $d = 2$ parameters.
The FD gradient descent uses step size $\alpha = 0.08$ for both $\delta$ values.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Randomised Smoothing)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation radius $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Nesterov random search estimates the gradient via a stochastic directional derivative: $\mathbb{E}[\hat{g}_k] = \nabla f_\delta(x_k)$ where $f_\delta$ is the Gaussian-smoothed function. This zeroth-order
method requires only \emph{two} function evaluations per iteration regardless of dimension $d$,
making it uniquely scalable to very high dimensions. However, only a one-dimensional projection of
the gradient is captured per step, so convergence rate scales as $O(d/\sqrt{k})$ --- much slower
than gradient-based methods. The perturbation $\delta = 0.08$ and step size $\alpha = 0.025$ are
used with 220 iterations.

\subsection{Part B: Derivative-Free Methods on Rosenbrock (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$ and
iteratively transforms it using four operations based solely on function values:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\rho_r = 1$): Move the worst vertex $x_{n+1}$ through the centroid
of the remaining vertices to $x_r = \bar{x} + ({\bar{x} - x_{n+1}})$.
\item \textbf{Expansion} ($\gamma = 2$): If reflection improved, extend further to
$x_e = \bar{x} + 2(\bar{x} - x_{n+1})$.
\item \textbf{Contraction} ($\rho = 0.5$): If reflection worsened, contract towards centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): Contract all vertices towards the best vertex.
\end{enumerate}
Initial simplex step size 0.35; 160 iterations.

\subsubsection{Grid Search}

f(\hat{x}) - f(x^\star) \leq \frac{L_f}{2}\|\hat{x} - x^\star\|^2 \leq \frac{L_f}{2} \cdot 2 \cdot (\Delta/2)^2
\end{equation}
where $\Delta \approx 0.074$ is the grid spacing and $L_f$ is the gradient Lipschitz constant.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Setup}

Minimise Benchmark B --- $f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)$ --- subject to
$x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$. Crucially, the
\emph{unconstrained} minimum is at $x^\star \approx (0.582, 2.000)$, which satisfies $x_1^\star
= 0.582 > 0.5$. Therefore, the \textbf{constraint is inactive at the optimum}: the constrained
and unconstrained solutions coincide. This is a fundamental concept from KKT theory --- the
Lagrange multiplier is zero whenever the constraint is inactive.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$
    \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$
    \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$, $x_2$ unchanged}
\EndFor
\end{algorithmic}
\end{algorithm}

For the constraint $x_1 \geq 0.5$, the projection is the element-wise clipping:
$\Pi_{\mathcal{X}}(z)_1 = \max(0.5, z_1)$, $\Pi_{\mathcal{X}}(z)_2 = z_2$. This is $O(1)$
cost for the box constraint but may be expensive for complex constraint sets.

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\; 0.5 - x_1)$
    \Comment{Augmented objective with linear penalty}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
    \Comment{$\nabla F_\lambda(x)_1 = \nabla f(x)_1 - \lambda \mathbf{1}[x_1 < 0.5]$}
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method replaces the constrained problem with an unconstrained penalised objective.
As $\lambda \to \infty$, the penalty method's solution approaches the constrained solution
(exact penalty theorem). For \emph{inactive} constraints, any $\lambda > 0$ ultimately produces
the same solution as the unconstrained problem, since the penalty term vanishes at the optimum.

Settings: PGD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ and step sizes

\subsection{Comparative Discussion}

\textbf{Projected GD} enforces feasibility as a hard constraint at every iterate. Key advantages:
(1) feasibility is never violated; (2) convergence theory for constrained optimisation directly
applies (the iteration is a contraction on the feasible set). Key limitation: requires an efficient
projection oracle. For the half-space $x_1 \geq 0.5$, projection is trivially $O(1)$, but for
general convex sets the projection can be expensive.

\textbf{Penalty methods} trade strict feasibility for simplicity. The penalised objective $F_\lambda$
is unconstrained and smooth (for smooth $f$), requiring only a modified gradient without a
projection oracle. The weight $\lambda$ controls feasibility enforcement speed: larger $\lambda$
achieves faster entry into the feasible region but requires smaller step sizes (here $\alpha = 0.03$
for $\lambda = 4.5$ vs $\alpha = 0.05$ for $\lambda \leq 1.8$) due to increased gradient magnitude
in the infeasible region. For this problem (inactive constraint at the optimum), all values of
$\lambda > 0$ eventually produce the correct solution.

\textbf{Key insight:} When the constraint is inactive at the optimum (which can be verified
retrospectively via KKT conditions), the penalty weight $\lambda$ affects only the transient
convergence path, not the final solution. The correct constrained solution requires merely that
the algorithm converge from an infeasible starting point without violating the constraint
persistently.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$.

For a linear objective over a box constraint, the minimum is attained at a vertex of the box.
Since both components of $a$ are positive, the minimum is at the lower bounds of each coordinate:
\begin{equation}
x^{\star} = (0.5,\; -5), \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}
This illustrates the \textbf{fundamental theorem of linear programming}: the optimum of a linear
objective over a bounded convex polytope is always attained at a vertex of the polytope.

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Fixed Mixing Parameter)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\; \nabla f(x_k)^T x$
    \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
    \Comment{Convex combination: step size $\gamma = 1-\beta$ towards $z_k$}
\EndFor
\end{algorithmic}
\end{algorithm}

The Frank--Wolfe (FW) algorithm replaces the projection step of projected gradient descent with a
\textbf{Linear Minimisation Oracle (LMO)}: at each iteration, find the vertex of $\mathcal{X}$
that minimises a linearisation of $f$ at the current point. The LMO for a box constraint decomposes
per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
This requires $O(d)$ operations per iteration. The update $x_{k+1} = \beta x_k + (1-\beta) z_k$
corresponds to FW with fixed step size $\gamma = 1-\beta$; the standard FW uses the diminishing
schedule $\gamma_k = 2/(k+2)$.

\textbf{Convergence rate:} For $L$-smooth convex $f$ with feasible diameter $D =
\max_{x,y\in\mathcal{X}}\|x-y\|$, FW with $\gamma_k = 2/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f(x^\star) \leq \frac{2LD^2}{k+2}.
\end{equation}
This $O(1/k)$ rate is slower than projected gradient descent's $O(1/k^2)$ for strongly convex
objectives, but FW avoids projection entirely. For this problem, $D = \sqrt{(5-0.5)^2 + (10+5)^2}
\approx 15.65$.

The unconstrained minimum $(1, 5)$ lies inside $\mathcal{X}$ (since $0.5 \leq 1 \leq 5$ and
$-5 \leq 5 \leq 10$). Two step sizes are tested: $\gamma = 1-\beta$ with $\beta \in \{0.90,
0.985\}$, giving $\gamma \in \{0.10, 0.015\}$.

\subsection{Interior vs Boundary Convergence Behaviour}

\textbf{Interior optimum:} When $x^\star \in \mathrm{int}(\mathcal{X})$, the LP vertex $z_k$
oscillates between extreme vertices of $\mathcal{X}$ because $\nabla f(x_k) \neq 0$ for
$x_k \neq x^\star$, and infinitesimally near $x^\star$ the gradient direction fluctuates randomly
due to numerical precision. The Frank--Wolfe gap $g_k = \nabla f(x_k)^T(x_k - z_k)$ is the
primary convergence certificate; here the gap decreases at $O(1/k)$.

\textbf{Boundary optimum:} Frank--Wolfe is especially well-suited to boundary optima because the
LP vertex $z_k$ consistently points towards the constraint face containing $x^\star$. The iterate
approaches the boundary monotonically without projection, and the fixed $\gamma = 0.07$ maintains
steady progress. Residual gap ($f = 0.364$ vs $f^\star = 0.25$) reflects the $O(1/k)$ convergence
rate; more iterations or a diminishing schedule would close this gap.

\textbf{Key advantage of Frank--Wolfe:} The algorithm never performs a projection, requiring
only the LMO. For problems where projection is expensive (e.g., nuclear norm balls, flow
polytopes) but the LMO is cheap, FW is the preferred projection-free constrained optimisation
algorithm. Additionally, since all iterates are convex combinations of vertices of $\mathcal{X}$,
FW naturally produces sparse solutions when $\mathcal{X}$ is a simplex or similar sparse-vertex
polytope.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad and RMSprop}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)

def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball and Nesterov Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z      # gradient at look-ahead point
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser and Newton's Method}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))  # ensure positive definiteness
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # Linear minimisation oracle: select vertex minimising g^T z
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # step size gamma = 1 - beta
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
