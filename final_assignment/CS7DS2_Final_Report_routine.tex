\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three canonical benchmark problems: a linear regression quadratic loss (Benchmark~A), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C). Six categories of methods are systematically investigated: (1)~adaptive step-size methods including Polyak, Adagrad, RMSprop, and Polyak Momentum (Heavy Ball); (2)~momentum-based and second-order moment methods including Nesterov acceleration and Adam; (3)~Newton's method exploiting local second-order approximations; (4)~derivative-free approaches including finite-difference gradient approximation, Nesterov random search, Nelder--Mead simplex, and grid search; (5)~constrained optimisation via projected gradient descent and penalty methods; and (6)~the Frank--Wolfe (conditional gradient) algorithm for linearly-constrained problems. Each method is evaluated on convergence speed, trajectory behaviour, constraint satisfaction, stability, and computational trade-offs. Theoretical convergence guarantees are stated alongside empirical results, connecting experimental observations to course theory. All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for full reproducibility.
\end{abstract}

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and two parameters $\theta \in \mathbb{R}^2$ is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\label{eq:loss_A}
\end{equation}
This is a \emph{strictly convex quadratic} with constant Hessian $H = \frac{1}{m}X^T X \succ 0$ and gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the noise-floor minimum is $J^{\star} \approx 0.4826$ (not zero), attained at the OLS estimator $\hat{\theta} = (X^TX)^{-1}X^Ty \approx (3.028, 4.017)^T$. The Lipschitz constant of the gradient is $L = \lambda_{\max}(H) \approx 2.05$ and the strong convexity constant is $\mu = \lambda_{\min}(H) \approx 0.96$, giving condition number $\kappa = L/\mu \approx 2.1$. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\label{eq:loss_B}
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \text{diag}(2 - \sin(x_1),\ 10)$. The function is a convex quadratic in $x_2$ but nonlinear (though still convex for $|x_1|$ not too large) in $x_1$ due to the $\sin$ term. The eigenvalue ratio varies between $\approx 1:10$ and $\approx 3:10$, creating moderate anisotropy with condition number $\kappa \in [3, 10]$. The global minimum lies at $(x_1^{\star}, x_2^{\star}) \approx (0.582, 2.0)$ (found by solving $2(x_1^{\star}-1)+\cos(x_1^{\star})=0$, $x_2^{\star}=2$) with $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\label{eq:loss_C}
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The gradient is:
\begin{equation}
\nabla f = \begin{bmatrix} -2(1 - x_1) - 400 x_1(x_2 - x_1^2) \\ 200(x_2 - x_1^2) \end{bmatrix}.
\end{equation}
The Hessian at the optimum has eigenvalues $\approx 2$ and $\approx 400$, giving condition number $\kappa \approx 2000$. This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard benchmark functions. It is also non-quadratic (the Hessian varies significantly away from the minimum). Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems (Questions 5 and 6):
\begin{equation}
\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a compact, convex polytope (axis-aligned box). Projection onto $\mathcal{X}$ is a component-wise clipping operation, and the linear minimisation oracle (required by Frank--Wolfe) reduces to a per-coordinate argmin.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

Four adaptive methods are compared against constant step-size gradient descent. The fundamental challenge in gradient descent is choosing a step size $\alpha$: too large causes divergence, too small causes slow convergence. Adaptive methods adjust $\alpha$ automatically based on gradient history, aiming to achieve faster convergence without manual tuning.

For a function with Lipschitz-continuous gradient (constant $L$), gradient descent with step $\alpha \leq 1/L$ achieves:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{\|x_0 - x^{\star}\|^2}{2\alpha k},
\end{equation}
giving an $O(1/k)$ rate for convex functions and linear convergence $\rho = (1 - \mu/L) < 1$ for strongly convex functions. The condition number $\kappa = L/\mu$ governs convergence speed.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Gradient Descent (Baseline)}
\label{alg:gd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive, oracle step}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is theoretically derived: for a convex $f$,
\begin{equation}
f(x_k) - f^{\star} \leq \nabla f(x_k)^T(x_k - x^{\star}) \leq \|g_k\|\cdot\|x_k - x^{\star}\|,
\end{equation}
so the step $\alpha_k = (f(x_k)-f^{\star})/\|g_k\|^2$ minimises the squared distance to $x^{\star}$ along the gradient direction. It achieves $O(1/k)$ convergence without requiring knowledge of $L$ or $\mu$, but \emph{requires exact knowledge of $f^{\star}$}.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative sum of squared gradients (per-coordinate)}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's effective step size for coordinate $i$ is $\alpha_0/\sqrt{\sum_{t=0}^{k}g_{t,i}^2}$. It adapts faster to infrequent (sparse) features (large effective rate) and slower to frequent ones (small effective rate). The monotonically \emph{non-increasing} effective step size is a key property: it ensures stability but can cause premature stagnation on dense problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's monotonically growing denominator with an exponential moving average (EMA) with forgetting factor $\beta$. The effective denominator approximates the root-mean-square of recent gradients, maintaining a non-zero effective step size even after many iterations. The stationary effective step is $\alpha_0/\sqrt{(1-\beta) \cdot \mathbb{E}[g^2] / (1-\beta)} = \alpha_0/\text{RMS}(g)$.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Accumulated gradient (velocity)}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method is equivalent to $x_{k+1} = x_k - \alpha\nabla f(x_k) + \beta(x_k - x_{k-1})$, where $\beta(x_k - x_{k-1})$ is the momentum term. For a quadratic with condition number $\kappa$, the optimal parameters $\alpha^{\star} = 4/(L + \mu)^2$ and $\beta^{\star} = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$ achieve linear convergence with rate $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$, significantly faster than GD's rate $({\kappa}-1)/({\kappa}+1)$.

\begin{theorem}{Polyak Step Size Divergence Condition}{polyak_div}
Let $f$ be $L$-smooth and let $f^{\star}_{\text{true}}$ denote the true minimum. If the specified value satisfies $f^{\star}_{\text{spec}} < f^{\star}_{\text{true}}$, then as $x_k \to x^{\star}$ we have $f(x_k) \to f^{\star}_{\text{true}} > f^{\star}_{\text{spec}}$, so the Polyak step behaves as:
\begin{equation}
\alpha_k = \frac{f(x_k) - f^{\star}_{\text{spec}}}{\|g_k\|^2 + \varepsilon} \;\xrightarrow[x_k \to x^{\star}]{}\; \frac{f^{\star}_{\text{true}} - f^{\star}_{\text{spec}}}{0 + \varepsilon} = \frac{\Delta f^{\star}}{\varepsilon} \to \infty,
\end{equation}
causing the iterate to oscillate away from $x^{\star}$ rather than converge.
\end{theorem}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly (Benchmark C), Polyak converges by far the fastest ($f = 0.009$ vs next-best Heavy Ball at $f = 1.40$). Otherwise, Adagrad and Heavy Ball match or exceed GD baseline performance.

\item \textbf{Adaptivity trade-off:} Adagrad's monotonically decaying effective step is a two-edged sword: it prevents divergence uniformly but causes it to converge to higher function values on non-sparse problems like Rosenbrock, where large early gradients unnecessarily shrink subsequent steps.

\item \textbf{RMSprop vs Adagrad:} RMSprop addresses Adagrad's aggressive decay by forgetting old gradients. This maintains non-vanishing step sizes but requires an additional hyperparameter ($\beta$) and can be sensitive to its choice near the minimum.

\item \textbf{Heavy Ball momentum benefit:} On the ill-conditioned Benchmark C, Heavy Ball ($\beta = 0.86$) provides approximately $2.5\times$ improvement over GD, consistent with the theory predicting momentum convergence rate $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$ vs GD's $(\kappa-1)/(\kappa+1)$.

\item \textbf{Practical guidance:} In practice, exact $f^{\star}$ is rarely known, limiting Polyak's applicability outside specialised settings. RMSprop and Adagrad are more robust defaults. Heavy Ball is preferred when momentum acceleration is desired and hyperparameters can be tuned.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Theoretical Background}

Nesterov acceleration is the canonical method achieving the optimal $O(1/k^2)$ convergence rate for smooth convex functions, proved to be tight for first-order methods (Nesterov's lower bound). Adam combines momentum with per-coordinate adaptive scaling and is the dominant optimiser for neural network training.

\begin{theorem}{Nesterov Convergence Rate}{nesterov_rate}
For $L$-smooth convex $f$, Nesterov's accelerated gradient method with step size $\alpha = 1/L$ achieves:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2},
\end{equation}
compared to $O(L\|x_0 - x^{\star}\|^2 / (2k))$ for gradient descent --- an improvement by a factor of $k$.
\end{theorem}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient Descent}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, maximum momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This allows Nesterov momentum to ``look ahead'' in the direction of motion and correct its trajectory before overshooting. The adaptive schedule $\beta_k = (k-1)/(k+2)$ increases momentum gradually, which is crucial for the $O(1/k^2)$ proof.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in [0,1)$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (EMA of gradient)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (EMA of $g^2$)}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t / (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam is a combination of momentum (via $m_t$, tracking direction history) and per-coordinate adaptive scaling (via $v_t$, tracking per-coordinate curvature). The bias correction in steps 6--7 prevents the estimates from being biased towards zero in early iterations (since $m_0 = v_0 = 0$). At initialisation, $(1-\beta_1^1) = 1-\beta_1$ and $(1-\beta_2^1) = 1-\beta_2$, so the corrections are substantial; they become negligible as $t \to \infty$. The effective per-coordinate step size is $\alpha \cdot \hat{m}_{t,i}/(\sqrt{\hat{v}_{t,i}}+\varepsilon) \approx \alpha \cdot \text{sign}(g_{t,i})$ when gradients are consistent, providing a near-unit step in direction of descent.

Mini-batch SGD on Benchmark A: learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, data shuffled each epoch. Each epoch processes all $m = 1000$ samples in batches.

\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute indices $\pi = \text{shuffle}(\{1,\ldots,m\})$

    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{itemize}[nosep]
\item \textbf{On convex, near-quadratic problems (A, B):} Both Nesterov and Adam converge to the minimum, with Adam often slightly faster initially due to per-coordinate normalisation. Neither has a clear theoretical advantage over the other on smooth strongly convex functions.

\item \textbf{On ill-conditioned, nonlinear problems (C):} Nesterov substantially outperforms Adam. \end{itemize}

\subsection{Local Taylor Approximations of $g(x) = x^4$}

At expansion point $x_0 = 0.25$, we compute $g(x_0) = 0.25^4 = 0.00390625$, $g'(x_0) = 4 \cdot 0.25^3 = 0.0625$, $g''(x_0) = 12 \cdot 0.25^2 = 0.75$.

\textbf{First-order (linear) Taylor approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.003906 + 0.0625(x - 0.25).
\label{eq:first_order}
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.003906 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\label{eq:second_order}
\end{equation}

The Newton step for minimising the second-order model (\ref{eq:second_order}) sets its derivative to zero:
\begin{equation}
g'(x_0) + g''(x_0)(x_{\text{Newton}} - x_0) = 0 \implies x_{\text{Newton}} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\end{equation}
This is indeed closer to the true minimiser $x^{\star} = 0$ than a gradient step of step size $\alpha$ would achieve unless $\alpha$ were chosen very specifically.

The second-order approximation (parabola) fits well near $x_0$ and provides a substantially better prediction of the function's shape. \subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0,1]$, Tikhonov damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised (positive definite) Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{$\alpha = 1$: pure Newton; $\alpha < 1$: damped Newton}
\EndFor
\end{algorithmic}
\end{algorithm}

For the pure Newton step ($\alpha = 1$, $\lambda = 0$), the update minimises the local quadratic model $f(x_k) + g_k^T(x - x_k) + \frac{1}{2}(x-x_k)^T H_k (x-x_k)$. The Tikhonov regularisation $\lambda I$ ensures positive definiteness of $H_k$ even when the Hessian is near-singular (as occurs on Rosenbrock far from the minimum). Settings: GD uses 80 iterations; Newton uses 20 iterations with $\lambda = 10^{-8}$.

\begin{theorem}{Newton's Quadratic Convergence}{newton_quad}
Near a non-degenerate local minimiser $x^{\star}$ (where $\nabla f(x^{\star}) = 0$ and $\nabla^2 f(x^{\star}) \succ 0$), undamped Newton's method converges quadratically:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq C \|x_k - x^{\star}\|^2,
\end{equation}
for some constant $C$ depending on the Lipschitz constant of $\nabla^2 f$. This means the number of correct digits roughly doubles at each iteration.
\end{theorem}

\subsection{Discussion: Newton vs First-Order Methods}

Newton's method achieves superior convergence rates at the cost of computing and inverting the Hessian ($O(n^3)$ for dense $n \times n$ problems). For the 2D benchmarks used here, this is trivial. In practice:
\begin{itemize}[nosep]
\item \textbf{Quadratic problems (A):} Newton is exact in one step. First-order methods always require multiple iterations.
\item \textbf{Near-quadratic problems (B):} Newton converges in $O(\log(1/\varepsilon))$ iterations vs $O(\kappa/\varepsilon)$ for GD.
\item \textbf{Non-quadratic, ill-conditioned problems (C):} Damped Newton still dramatically outperforms GD, but requires careful step-size selection to maintain the accuracy of the local quadratic model.
\item \textbf{Scalability limitation:} For large-scale problems ($n \gg 100$), the $O(n^3)$ Hessian solve is prohibitive, motivating quasi-Newton methods (L-BFGS) that approximate $H^{-1}$ from gradient differences.
\end{itemize}

\subsection{Part A: Finite-Difference Gradient Approximation and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite-Difference Gradient Approximation}

When the analytic gradient is unavailable, it can be approximated using finite differences. The forward difference approximation is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \qquad i = 1, \ldots, n,
\label{eq:fd}
\end{equation}
where $e_i$ is the $i$-th standard basis vector. This requires $n+1$ function evaluations per gradient step (compared to $1$ for automatic differentiation).

\textbf{Error analysis:} The truncation error from the Taylor expansion is:
\begin{equation}
\frac{f(x + \delta e_i) - f(x)}{\delta} = \frac{\partial f}{\partial x_i}(x) + \frac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x) + O(\delta^2).
\end{equation}
So the approximation error is $O(\delta)$ (truncation) plus $O(\varepsilon_{\text{mach}}/\delta)$ (floating-point cancellation in the numerator). Balancing these gives the optimal choice $\delta^{\star} \sim \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ for double precision. In the experiments: $\delta = 0.05$ provides good approximation while $\delta = 0.8$ is far too large.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Gaussian Smoothing / Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation scale $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_n)$, then normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Random unit direction}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Random directional gradient estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates the gradient in a single random direction, requiring only 2 function evaluations per iteration regardless of dimension $n$. The estimate $\hat{g}_k$ is an unbiased estimate of $\frac{n}{\delta}\mathbb{E}_{u}[(f(x_k+\delta u)-f(x_k))u] \approx \nabla f(x_k)$ in expectation (by Stein's lemma for spherical smoothing). The variance is $O(n)$ times larger than the true gradient, explaining the slower convergence.

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ points in $\mathbb{R}^2$ and iteratively updates it through four geometric operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): Move the worst vertex $x_w$ through the centroid $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_w)$.
\item \textbf{Expansion} ($\gamma = 2.0$): If reflection improved, try further: $x_e = \bar{x} + \gamma(\bar{x} - x_w)$.
\item \textbf{Contraction} ($\rho = 0.5$): If reflection worsened, contract towards centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): Contract all vertices towards the best.
\end{enumerate}

Settings: initial simplex step $= 0.35$, 160 iterations. Total function evaluations: $\leq 3 \times 160 = 480$.

\subsection{Problem Statement}

Minimise Benchmark B~(\ref{eq:loss_B}) subject to the constraint $x_1 \geq 0.5$:
\begin{equation}
\min_{x \in \mathcal{C}} f(x), \qquad \mathcal{C} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}.
\end{equation}
Starting from the \emph{infeasible} initial point $x_0 = (0.2, 4.0)$ (which violates $x_1 \geq 0.5$). The unconstrained minimum at $(0.582, 2.0)$ satisfies $0.582 \geq 0.5$, so the constrained and unconstrained minima coincide. This means both projected GD and the penalty method should converge to $f^{\star} = 0.7244$ at $x^{\star} = (0.582, 2.0)$ when correctly tuned.

\textbf{KKT Optimality Conditions:} At the constrained minimum $x^{\star}$, the KKT conditions require:
\begin{equation}
\nabla f(x^{\star}) + \mu \nabla g(x^{\star}) = 0, \quad \mu \geq 0, \quad g(x^{\star}) \leq 0, \quad \mu g(x^{\star}) = 0,
\end{equation}
where $g(x) = 0.5 - x_1$. Since $x_1^{\star} = 0.582 > 0.5$, the constraint is inactive and $\mu = 0$ (complementary slackness), so $\nabla f(x^{\star}) = 0$, confirming the unconstrained optimum is also the constrained optimum.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1})$ \Comment{$\Pi_{\mathcal{C}}(z): x_1 \gets \max(0.5, z_1),\; x_2 \gets z_2$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection $\Pi_{\mathcal{C}}$ onto the half-space $\{x_1 \geq 0.5\}$ is a clipping operation in the first coordinate. It is the exact Euclidean projection: $\Pi_{\mathcal{C}}(z) = \arg\min_{x \in \mathcal{C}} \|x - z\|^2$.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\State Define $F_{\lambda}(x) \gets f(x) + \lambda\,P(x)$ where $P(x) = \max(0,\, 0.5 - x_1)$ \Comment{L1 exterior penalty}
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute $\nabla F_{\lambda}(x_k)$: if $x_1 < 0.5$, add $-\lambda$ to first gradient component; else add $0$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method replaces the constrained problem with an unconstrained one via $F_{\lambda} = f + \lambda P$. As $\lambda \to \infty$, the unconstrained minimiser of $F_{\lambda}$ approaches the constrained minimiser. The L1 penalty $P(x) = \max(0, 0.5 - x_1)$ is exact (achieves the constrained minimum at finite $\lambda$) for convex problems under mild conditions.

\textbf{Settings:} Projected GD ($\alpha = 0.08$); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$, $\alpha = 0.03$ for $\lambda = 4.5$); Unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected GD} enforces feasibility as a hard constraint at every iteration: $x_k \in \mathcal{C}$ for all $k \geq 1$. \textbf{Penalty methods} trade exact feasibility for algorithmic simplicity: only the gradient of the augmented objective $F_\lambda$ is needed. The weight $\lambda$ controls the feasibility-optimality trade-off:
\begin{itemize}[nosep]
\item Small $\lambda$: the penalty gradient $-\lambda$ is dominated by $\nabla f$, so constraint enforcement is weak. The solution can be significantly infeasible.
\item Large $\lambda$: feasibility is enforced aggressively but the penalised objective $F_\lambda$ has a steeper landscape near the constraint boundary, requiring smaller step sizes for stability. Here, $\alpha$ was reduced from $0.05$ to $0.03$ for $\lambda = 4.5$ to prevent oscillation.
\end{itemize}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (axis-aligned box).

\textbf{Linear Programme (Q6-I):} Minimise $f(x) = a^T x = x_1 + 2x_2$ (with $a = [1, 2]^T$) over $\mathcal{X}$.

Since $f$ is linear, its minimum over the convex compact set $\mathcal{X}$ is attained at an extreme point (vertex). For a box constraint with positive objective coefficients, the minimum is at the component-wise lower bounds:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = 0.5 - 10 = -9.5.
\end{equation}

\begin{theorem}{Fundamental Theorem of Linear Programming}{lp_fund}
The minimum of a linear objective $c^Tx$ over a non-empty, bounded polytope $\mathcal{P} = \{x : Ax \leq b\}$ is attained at a vertex (extreme point) of $\mathcal{P}$.
\end{theorem}

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient) Method}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\; \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (stays in $\mathcal{X}$)}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Key property:} Since $\mathcal{X}$ is convex and $x_k, z_k \in \mathcal{X}$, the convex combination $x_{k+1} = \beta x_k + (1-\beta)z_k \in \mathcal{X}$ is always feasible. Frank--Wolfe is \emph{projection-free}: no projection oracle is required, only an LMO.

For the box $\mathcal{X}$, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
This requires $O(n)$ operations (one comparison per coordinate). The standard decaying-step Frank--Wolfe with $\gamma_k = 2/(k+2)$ achieves the following convergence rate for smooth convex $f$:

\begin{theorem}{Frank--Wolfe Convergence Rate}{fw_rate}
For $f$ convex and $L$-smooth over $\mathcal{X}$ with diameter $D = \max_{x,y \in \mathcal{X}}\|x - y\|$, the standard Frank--Wolfe algorithm with step sizes $\gamma_k = 2/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2}.
\end{equation}
This $O(1/k)$ rate is slower than projected gradient descent's $O(1/k^2)$ (for smooth strongly convex $f$), but each iteration requires only an LMO rather than a projection.
\end{theorem}

In the experiments, a fixed $\beta$ (mixing parameter) is used in place of the standard decaying schedule.

\subsection{Interior vs Boundary Optimum: Analysis}

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits a characteristic zig-zag pattern. Near the interior minimum, the gradient $\nabla f(x_k)$ is small and its sign in each coordinate alternates as $x_k$ oscillates around $x^{\star}$. This causes $z_k$ to jump between extreme points (e.g., $z_{1,k}$ alternates between $0.5$ and $5$), and the fixed-$\beta$ convex combination induces sustained oscillation of amplitude $\propto (1-\beta)\cdot D$ where $D$ is the box diameter. This oscillation is fundamental to the $O(1/k)$ convergence rate of Frank--Wolfe and can be reduced by using the standard $\gamma_k = 2/(k+2)$ decaying schedule instead of fixed $\beta$.

\textbf{Boundary optimum:} When the unconstrained minimum is infeasible, $z_k$ more consistently selects the LMO vertex closest to the constrained minimiser (since the gradient direction is consistent --- it always points away from the boundary towards $(0,0)$). The trajectory approaches the boundary naturally, and the final $z_k = (0.5, \text{sgn of gradient})$ stabilises quickly. However, the fixed $\beta = 0.93$ prevents exact convergence: $x_k$ approaches but never reaches $(0.5, 0)$ in finite iterations. The final $f = 0.364 > f^{\star} = 0.25$ reflects the $O(1/k)$ residual.

\subsection{Conclusions by Category}

\textbf{Q1 --- Adaptive Step-Size Methods:}
The Polyak step size is the theoretically optimal adaptive method when $f^{\star}$ is known exactly, achieving the best Rosenbrock result among Q1 methods ($f = 0.009$). However, it is highly sensitive to $f^{\star}$ misspecification and diverges badly on Benchmarks A and B where $f^{\star}_{\text{spec}} = 0 \neq f^{\star}_{\text{true}}$. Adagrad and Heavy Ball are robust alternatives. RMSprop's exponential forgetting provides more stable step sizes than Adagrad's monotone decay, but can oscillate near the minimum on Benchmark A.

\textbf{Q2 --- Momentum Methods:}
Nesterov acceleration achieves the optimal $O(1/k^2)$ convergence rate for smooth convex functions and consistently outperforms all first-order methods on Benchmark C (Rosenbrock, $f = 0.479$). Adam combines momentum with per-coordinate scaling and performs excellently on near-quadratic problems but is less effective on Rosenbrock where directional coherence (not coordinate-wise normalisation) is the key factor. Mini-batch SGD demonstrates the trade-off between update frequency (small batches) and gradient variance (large batches); under high output noise, the irreducible floor dominates.

\textbf{Q3 --- Newton's Method:}
Newton's method is the clear winner for problems with computable Hessians. It solves Benchmark A in one iteration (exact for quadratics), Benchmark B in $\sim$5 iterations, and achieves the lowest Benchmark C value ($f = 0.686$) in just 20 iterations vs GD's $f = 3.73$ in 80 iterations. The second-order information provided by the Hessian fundamentally changes the convergence from linear (first-order methods) to quadratic, at the cost of $O(n^3)$ per iteration for large $n$.

\textbf{Q4 --- Derivative-Free Methods:}
Finite-difference gradient approximation with small $\delta$ closely matches exact gradient descent, confirming the $O(\delta)$ truncation error bound. Large $\delta$ gives poor gradient estimates. Nesterov random search is the most scalable approach (only 2 function evaluations per step) but converges slowly due to the noisy 1D gradient projection. Nelder--Mead navigates Rosenbrock effectively without any gradient information. Grid search provides a guaranteed global solution up to grid resolution but is exponentially expensive in dimension.

\textbf{Q5 --- Constrained Optimisation:}
Projected gradient descent provides hard constraint guarantees at every iteration, with zero violation from iteration 1. Penalty methods offer algorithmic simplicity but require careful $\lambda$ tuning and cannot guarantee feasibility at finite $\lambda$. For this specific problem (the unconstrained minimum is feasible), both methods converge to the same value $f^{\star} = 0.7244$ at $(0.582, 2.0)$ when $\lambda$ is sufficiently large.

\textbf{Q6 --- Frank--Wolfe:}
The Frank--Wolfe algorithm is projection-free, requiring only a cheap linear minimisation oracle per iteration. This makes it attractive for constraint sets where projection is expensive (e.g., nuclear norm ball, traffic polytopes). However, the $O(1/k)$ convergence rate is slower than projected GD's $O(1/k^2)$. The fixed-$\beta$ variant exhibits characteristic zig-zag oscillation near interior optima; the standard decaying $\gamma_k = 2/(k+2)$ schedule would reduce this. Frank--Wolfe naturally handles boundary optima, approaching the constraint boundary without any explicit projection.

\subsection{Recommendations for Practice}

\begin{enumerate}[nosep]
\item \textbf{When Hessian is available and $n$ is small:} Use Newton or quasi-Newton (L-BFGS). Far superior convergence rates.
\item \textbf{For large-scale machine learning:} Adam or RMSprop for non-convex neural network loss, with tuning of learning rate. Nesterov for convex problems.
\item \textbf{When gradient is unavailable:} Nelder--Mead for low $d \leq 10$; random search for high $d$; grid search only for very low $d$ visualisation.
\item \textbf{For constrained problems:} Projected GD for hard constraints with cheap projection oracle; penalty/augmented Lagrangian for complex constraint sets; Frank--Wolfe when LMO is cheap and projection is expensive.
\item \textbf{Polyak step size:} Only use when $f^{\star}$ is known exactly (e.g., feasibility problems, logistic regression minimum at $-\infty$).
\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak step size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad (per-coordinate adaptive step)}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float); G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x); G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop (exponential moving average)}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float); v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float); z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x); z = beta * z + alpha * g; x = x - z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov accelerated gradient descent}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float); z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z       # extrapolation point
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g       # velocity update
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam optimiser (bias-corrected moments)}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's method with Tikhonov regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))   # ensure PD
        try:
            p = np.linalg.solve(H_reg, g)       # Newton direction
        except np.linalg.LinAlgError:
            p = g                                # fallback to GD
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe conditional gradient algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LMO: per-coordinate argmin over box
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z       # stay in X
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
