\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three
benchmark problems: a linear regression quadratic loss (Benchmark A), a toy neural network loss
with sinusoidal perturbation (Benchmark B), and the Rosenbrock function (Benchmark C).  Six
categories of methods are investigated: \emph{adaptive step-size methods} (Polyak, Adagrad,
RMSprop, Heavy Ball), \emph{momentum-based methods} (Nesterov acceleration, Adam),
\emph{second-order methods} (Newton's method with local quadratic approximations),
\emph{derivative-free approaches} (finite differences, Nesterov random search, Nelder--Mead,
grid search), \emph{constrained optimisation} via projected gradient descent and exterior penalty
methods, and the \emph{Frank--Wolfe algorithm} for linear-constrained problems.  Each method is
evaluated in terms of convergence speed, trajectory behaviour, stability, and computational
trade-offs.  Key findings include: (i) the Polyak step diverges whenever the supplied lower bound
$f^{\star}$ is inaccurate; (ii) Nesterov acceleration achieves an optimal $O(1/k^2)$ rate,
dominating plain gradient descent on ill-conditioned surfaces; (iii) Newton's method reaches the
quadratic minimum in a single iteration; (iv) derivative-free methods carry a dimensionality
penalty but are indispensable when gradients are unavailable; (v) projected gradient descent
enforces hard feasibility instantly, whereas penalty methods require careful penalty scheduling;
and (vi) Frank--Wolfe avoids projection but converges at the slower $O(1/k)$ rate.
A cross-benchmark performance table (Table~\ref{tab:summary}) and hierarchy diagram
(Figure~\ref{fig:comparison_C}) provide a unified synthesis of all results.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for full reproducibility.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\label{eq:bench_A_data}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \sim \mathcal{N}(0, I_2), \quad
y^{(i)} = (\theta^{\star})^{\!\top} x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error (MSE):
\begin{equation}
\label{eq:bench_A_loss}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^{\top} x^{(i)} - y^{(i)}\bigr)^2
           = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a \emph{strongly convex quadratic} with constant Hessian $H = \frac{1}{m}X^{\top}X \succ 0$
and gradient $\nabla J(\theta) = \frac{1}{m}X^{\top}(X\theta - y)$.  The condition number
$\kappa(H) = \lambda_{\max}/\lambda_{\min}$ is close to 1 due to the near-isotropic feature
distribution.  Because of observation noise $\varepsilon^{(i)}$, the noise-floor minimum is
$J^{\star} = \frac{1}{2}\sigma^2 \approx 0.4826$, not zero.
Starting point: $\theta_0 = [0,\, 0]^{\top}$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
\label{eq:bench_B}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient and Hessian are
\begin{align}
\nabla f(x) &= \bigl[2(x_1 - 1) + \cos(x_1),\; 10(x_2 - 2)\bigr]^{\top}, \label{eq:grad_B}\\
H(x)        &= \operatorname{diag}\bigl(2 - \sin(x_1),\; 10\bigr). \label{eq:hess_B}
\end{align}
The Hessian eigenvalues range from $2 - \sin(x_1) \in [1, 3]$ (in the $x_1$ direction) to
$10$ (in the $x_2$ direction), creating moderate anisotropy with condition number
$\kappa \in [3.3, 10]$ depending on the iterate's $x_1$ coordinate.  The unique minimum
satisfies the stationarity condition $2(x_1^{\star}-1)+\cos(x_1^{\star})=0$,
which has the numerical solution $x_1^{\star} \approx 0.5754$ (obtained via Newton's
method on the scalar equation), $x_2^{\star} = 2.0$, and $f^{\star} \approx 0.7244$.
Starting point: $x_0 = [-1,\, 4]^{\top}$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
\label{eq:bench_C}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$.  The gradient is
\begin{align}
\label{eq:grad_C}
\frac{\partial f}{\partial x_1} &= -2(1-x_1) - 400\,x_1(x_2 - x_1^2), \\
\frac{\partial f}{\partial x_2} &= 200(x_2 - x_1^2),
\end{align}
and the Hessian is
\begin{equation}
H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1,1)$: $H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}$,
with eigenvalues $\lambda \approx \{0.40,\, 1001.6\}$ and condition number $\kappa \approx 2508$.
At the \emph{starting point} $(-1, 1)$: $H(-1,1) = \begin{bmatrix} 2 + 1200 - 400 & 400 \\ 400 & 200 \end{bmatrix}
= \begin{bmatrix} 802 & 400 \\ 400 & 200 \end{bmatrix}$, with the same spectrum---the
Rosenbrock's condition number is essentially constant along the ridge, always near 2508.
This extreme ill-conditioning creates two challenges:
\begin{enumerate}[nosep]
\item \textbf{Gradient descent step size:} $\alpha \leq 1/L \approx 0.00100$ is required for
  stability (with $L \approx 1001.6 = \lambda_{\max}$ at the optimum), but this step size
  is too small to make progress along the low-curvature direction ($\lambda_{\min} \approx 0.40$),
  where $\|x_k - x^{\star}\|$ decays as $(1 - \alpha\lambda_{\min})^k \approx (1 - 0.0004)^k$---
  requiring about $1/0.0004 = 2500$ iterations to halve the error in the valley direction.
\item \textbf{Curved valley:} The valley $x_2 = x_1^2$ is a quadratic curve, not a straight
  line.  Methods that assume a locally flat landscape (GD, Heavy Ball) must follow the curve
  with small steps; second-order methods (Newton) can compute the curvature of the valley
  and take along-valley steps directly.
\end{enumerate}
Starting point: $x_0 = [-1,\, 1]^{\top}$.

\subsection{Benchmark D: Constrained Feasible Set}
For Questions~5 and~6: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$,
a box-constrained (polytope) feasible set.

\subsection{Experimental Protocol and Stopping}

All experiments run for a fixed number of iterations (specified per question) rather than
a gradient-norm stopping criterion.  In practice, common stopping rules include:
\begin{enumerate}[nosep]
\item $\|\nabla f(x_k)\|_2 \leq \varepsilon_g$ (gradient norm tolerance).
\item $|f(x_{k+1}) - f(x_k)| \leq \varepsilon_f$ (objective change tolerance).
\item $\|x_{k+1} - x_k\|_2 \leq \varepsilon_x$ (iterate change tolerance).
\end{enumerate}
Fixed-iteration budgets are used here to enable fair comparison of convergence \emph{curves}
rather than convergence \emph{points}, and to avoid stopping criteria masking the convergence
behaviour of faster methods.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline
across all three benchmarks for 120 iterations.  The fundamental motivation is that a fixed
step size $\alpha$ must satisfy $\alpha < 2/L$ (where $L$ is the Lipschitz constant of the
gradient) to ensure convergence, but this bound may be overly conservative for most
coordinates.  Adaptive methods remedy this by learning coordinate-specific or
iteration-specific step sizes from gradient history.

\begin{algorithm}[H]
\caption{Gradient Descent (Baseline)}
\label{alg:gd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

For a $\mu$-strongly convex, $L$-smooth objective, GD with $\alpha = 1/L$ converges as
\begin{equation}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k [f(x_0) - f^{\star}],
\end{equation}
i.e.\ a linear rate $O((1-\mu/L)^k)$ governed by the condition number $\kappa = L/\mu$.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, known optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step is the unique step size that minimises a linear local model of $f$ subject to
the constraint that the iterates descend to $f^{\star}$.  When $f^{\star}$ is known exactly and
$f$ is convex with Lipschitz gradient, the Polyak rule achieves $O(1/k)$ convergence without
any knowledge of $L$ or $\mu$:
\begin{equation}
\label{eq:polyak_rate}
f(x_k) - f^{\star} \leq \frac{\|x_0 - x^{\star}\|^2}{2k}, \qquad \text{(Polyak, 1969)}.
\end{equation}
This rate matches GD's $O(1/k)$ for convex functions, but requires zero hyperparameter
tuning---a remarkable property that is unmatched by any other first-order rule.

\begin{remark}[Why $f^{\star} = 0$ fails on Benchmarks A and B]
\label{rem:polyak_fail}
The Polyak step requires the \emph{true} minimum value.  In this assignment, all three
benchmarks are tested with the nominal value $f^{\star} = 0$, which is correct only for
Benchmark C (Rosenbrock, with minimum value $f(1,1) = 0$).  For Benchmarks A and B,
$f^{\star} = 0$ is incorrect:
\begin{itemize}[nosep]
\item \textbf{Benchmark A}: The noise floor is $J^{\star} = \mathbb{E}[\varepsilon^2]/2 = \sigma^2/2 \approx 0.4826$.
  No estimator of $\theta^{\star}$ can achieve $J < J^{\star}$; the error is irreducible.
  Using $f^{\star} = 0$ gives a step $\alpha_k = (J(x_k) - 0) / \|g_k\|^2 \approx 0.48 / \|g_k\|^2$;
  as $x_k \to \theta_{\mathrm{OLS}}$, $\|g_k\| \to 0$ while the numerator stays $\approx 0.48$,
  causing $\alpha_k \to \infty$ and divergence.
\item \textbf{Benchmark B}: The minimum is $f^{\star} = \sin(x_1^{\star}) + 0 + 0 \approx 0.7244$.
  The function $\sin$ introduces a non-zero minimum value.  With $f^{\star} = 0$,
  the step grows as $\alpha_k \approx 0.72/\|g_k\|^2 \to \infty$, again causing divergence.
\end{itemize}
The Polyak step with the \emph{correct} lower bound $f^{\star}$ would converge on all three
benchmarks.  In practice, $f^{\star}$ is obtainable when: (i) the noise floor is known
analytically (as in OLS: $J^{\star} = \sigma^2/2$), (ii) a lower bound from a dual problem
is available, or (iii) the objective represents classification error with achievable zero loss.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate accumulation of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad scales the step size inversely with the \emph{cumulative} squared gradient magnitude in
each coordinate.  Coordinates that have received large historical gradients are given smaller
effective step sizes, which is beneficial for sparse features.  The exact regret bound of Duchi
et al.\ (2011) for the online convex setting is:
\begin{equation}
\label{eq:adagrad_regret}
R_T = \sum_{t=1}^{T} \bigl[f_t(x_t) - f_t(x^{\star})\bigr] \leq
\frac{\|x_1 - x^{\star}\|_{\infty}}{2\alpha_0} \sum_{i=1}^{d} \sqrt{\sum_{t=1}^T g_{t,i}^2}
+ \frac{\alpha_0}{2}\sum_{i=1}^d \sqrt{\sum_{t=1}^T g_{t,i}^2},
\end{equation}
which for bounded gradients $\|g_t\|_\infty \leq G$ simplifies to $R_T \leq O(G\sqrt{T})$---the
optimal $O(\sqrt{T})$ regret for online convex optimisation---matching the minimax lower bound.
In the offline batch setting, the monotonically shrinking step ($\alpha_0/\sqrt{G_k}$) may
cause premature stagnation on smooth strongly convex problems because the learning rate decays
to zero even after convergence.

\begin{remark}[Adaptive methods as diagonal preconditioning]
\label{rem:adagrad_precond}
Writing $P_k = \operatorname{diag}(\sqrt{G_k} + \varepsilon I)^{-1}$ for Adagrad, the update
becomes $x_{k+1} = x_k - \alpha_0 P_k \nabla f(x_k)$---a scaled gradient step with
coordinate-wise learning rates.  The ideal preconditioner for strongly convex $f$ is
$P = (\nabla^2 f(x^{\star}))^{-1}$ (which reduces the Newton step to a unit step), but computing
$P$ requires $O(d^3)$ work.  Adagrad approximates $P$ by its diagonal, reducing cost to $O(d)$
while capturing the most important curvature variation (heterogeneous per-coordinate scales).
For objectives with homogeneous curvature (isotropic $\nabla^2 f$), the diagonal approximation
is exact and Adagrad matches Newton.  For objectives with large off-diagonal curvature (such as
Rosenbrock, where $\partial^2 f / \partial x_1 \partial x_2 = -400 x_1$), diagonal preconditioning
is insufficient, which explains why Adagrad and RMSprop underperform Nesterov on Benchmark C
despite their adaptive scaling.  The full quasi-Newton approach (BFGS/L-BFGS) approximates the
\emph{full} inverse Hessian from gradient history, bridging the gap between first-order and
second-order methods.
\end{remark}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's unbounded accumulation with an exponential moving average with
decay factor $\beta$ (typically $0.9$), allowing the effective step to adapt to \emph{recent}
curvature rather than all historical gradients.  This prevents the step from shrinking
monotonically to zero.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method introduces a \emph{momentum} term $z_k$ that accumulates a running
average of past gradients, accelerating through elongated valleys and dampening oscillations
perpendicular to the descent direction.  For strongly convex quadratics with optimal
parameters
\begin{equation}
\alpha^{\star} = \left(\frac{2}{\sqrt{L} + \sqrt{\mu}}\right)^2, \qquad
\beta^{\star} = \left(\frac{\sqrt{L} - \sqrt{\mu}}{\sqrt{L} + \sqrt{\mu}}\right)^2,
\end{equation}
Heavy Ball achieves the linear rate $O(\exp(-k\sqrt{\mu/L})) = O\!\left((\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1})^k\right)$---the
same asymptotic rate as Nesterov on quadratics, with the key differences that
(i) Nesterov's $O(1/k^2)$ rate holds globally for all smooth convex functions while
Heavy Ball's optimal rate is only guaranteed for quadratics, and (ii) Heavy Ball
can \emph{diverge} on non-convex objectives such as Rosenbrock when the momentum
term carries the iterate across the curved valley floor, whereas Nesterov's
lookahead mechanism provides a corrective force that stabilises the trajectory.
Figure~\ref{fig:q2_hb_diverge} demonstrates this sensitivity: identical step sizes
that cause Heavy Ball to diverge are safely handled by Nesterov acceleration.

Concretely, for Benchmark C with $\kappa \approx 2508$ ($\mu \approx 0.40$, $L \approx 1001.6$
at the optimum), the per-iteration convergence factors compare as:
\begin{align}
\rho_{\text{GD}} &= \left(\frac{\kappa-1}{\kappa+1}\right)^2 \approx 0.9984
  \quad \Rightarrow \quad k_{1/e} \approx \tfrac{1}{|\log \rho|} \approx 625 \text{ iterations to reduce by } e^{-1}, \\
\rho_{\text{HB}} &= \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right) \approx 0.961
  \quad \Rightarrow \quad k_{1/e} \approx \tfrac{1}{|\log \rho|} \approx 25 \text{ iterations},
\end{align}
a \textbf{25$\times$ speedup} from momentum on this ill-conditioned quadratic.  The same
$25\times$ factor applies to Nesterov on the same quadratic.  In practice, the Rosenbrock
function deviates from a quadratic (the valley is curved, not straight), so neither Heavy Ball
nor Nesterov achieves these exact rates---but the theoretical advantage is partially realised,
as seen in the convergence plots.

\begin{remark}[Heavy Ball stability analysis: characteristic polynomial]
\label{rem:hb_stability}
The stability of Heavy Ball on a quadratic $f(x) = \frac{1}{2}\lambda x^2$ (scalar case,
$\lambda \in [\mu, L]$) can be analysed via the characteristic polynomial of the iteration
matrix.  The Heavy Ball iteration
$z_{k+1} = \beta z_k + \alpha\lambda x_k$, $x_{k+1} = x_k - z_{k+1}$
has the recurrence $x_{k+2} = (1 + \beta - \alpha\lambda)x_{k+1} - \beta x_k$, with
characteristic polynomial $p(\xi) = \xi^2 - (1 + \beta - \alpha\lambda)\xi + \beta$.
The roots $\xi_{1,2} = \frac{(1+\beta-\alpha\lambda) \pm \sqrt{(1+\beta-\alpha\lambda)^2 - 4\beta}}{2}$
must satisfy $|\xi_{1,2}| < 1$ for convergence.  For the optimal parameters
$\alpha^{\star} = 4/(\sqrt{L}+\sqrt{\mu})^2$ and $\beta^{\star} = ((\sqrt{L}-\sqrt{\mu})/(\sqrt{L}+\sqrt{\mu}))^2$,
the discriminant is negative for all $\lambda \in [\mu, L]$, giving complex conjugate roots
with $|\xi| = \sqrt{\beta^{\star}} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1) \approx 0.961$
(for $\kappa = 2508$)---explaining the geometric decay at rate $0.961^k$.  For $\lambda$ outside
$[\mu, L]$ (as on Rosenbrock, where the local Hessian eigenvalue varies), the discriminant may
be positive and one root may exit the unit disk, causing divergence.
\end{remark}

\textbf{Key finding on Polyak divergence.}  When the supplied $f^{\star}$ underestimates the
true minimum $f^{\star}_{\mathrm{true}}$, the iterates converge to a neighbourhood of the true
minimiser but the step size satisfies
\begin{equation}
\alpha_k = \frac{f(x_k) - f^{\star}}{\|\nabla f(x_k)\|^2 + \varepsilon}
         \;\xrightarrow[x_k \to x^{\star}_{\mathrm{true}}]{}\;
         \frac{f^{\star}_{\mathrm{true}} - f^{\star}}{\varepsilon} \gg 1,
\end{equation}
causing the iterates to oscillate with growing amplitude.  On Benchmark C, $f^{\star} = 0$ is
exact, so $\alpha_k \to 0$ as $f(x_k) \to 0$, giving the well-behaved convergence observed.

\subsection{Theoretical Step-Size Decay Comparison}

Before examining the empirical step sizes, it is instructive to analyse the theoretical
decay rates.  For a constant gradient magnitude $\|g_k\| = g$ (a simplified model):
\begin{itemize}[nosep]
\item \textbf{Adagrad:} $G_k = kg^2$, so $\alpha_k^{\mathrm{eff}} = \alpha_0/\sqrt{kg^2 + \varepsilon}
  \sim \alpha_0/(g\sqrt{k})$---the effective step decays as $O(1/\sqrt{k})$.
\item \textbf{RMSprop:} $v_k = g^2(1 - \beta^k)/(1-\beta) \to g^2/(1-\beta)$ as $k \to \infty$,
  so $\alpha_k^{\mathrm{eff}} \to \alpha_0\sqrt{1-\beta}/g$---a \emph{constant} effective rate.
\end{itemize}
This explains why Adagrad's step size decreases without bound while RMSprop's stabilises.

\subsection{Backtracking Line Search (Alternative Step-Size Strategy)}

While this question focuses on adaptive step-size rules, it is instructive to contrast them
with \emph{backtracking line search} (Armijo rule), which selects $\alpha_k$ satisfying the
sufficient decrease condition:
\begin{equation}
\label{eq:armijo}
f(x_k - \alpha_k g_k) \leq f(x_k) - c_1 \alpha_k \|g_k\|^2, \qquad c_1 \in (0,1),
\end{equation}
by starting with $\alpha_k = 1$ and repeatedly multiplying by $\rho < 1$ until~\eqref{eq:armijo}
holds.  Unlike adaptive rules, backtracking requires an additional function evaluation per
inner loop iteration but guarantees the \emph{sufficient decrease} property at every step,
which is critical for global convergence guarantees.  The adaptive methods in this question
(Adagrad, RMSprop, Polyak) avoid the inner loop but sacrifice this guarantee---they may not
decrease the objective at every step.

\begin{remark}[Strong Wolfe conditions and their role in quasi-Newton methods]
\label{rem:wolfe}
The Armijo rule~\eqref{eq:armijo} alone (sufficient decrease) does not prevent arbitrarily
small step sizes that stall convergence.  The \emph{strong Wolfe conditions} add a curvature
requirement:
\begin{equation}
\label{eq:wolfe}
f(x_k - \alpha_k g_k) \leq f(x_k) - c_1 \alpha_k \|g_k\|^2
\quad \text{(sufficient decrease)},
\qquad
|\nabla f(x_{k+1})^{\top} g_k| \leq c_2 \|\nabla f(x_k)\|^2
\quad \text{(curvature)},
\end{equation}
with $0 < c_1 < c_2 < 1$ (typically $c_1 = 10^{-4}$, $c_2 = 0.9$).  The curvature condition
ensures the gradient at the new point is not too negative in the search direction, preventing
near-zero step sizes.  Crucially for BFGS: the Wolfe conditions guarantee $y_k^{\top}s_k > 0$
(the secant pair has positive curvature), which is necessary for the rank-2 BFGS update to
maintain positive definiteness of $B_k$.  Without the curvature condition, the inner product
$y_k^{\top}s_k$ may be negative or zero, causing the BFGS update to produce an indefinite
matrix and losing all convergence guarantees.  This is why quasi-Newton methods invariably
use Wolfe line search rather than pure Armijo backtracking.
\end{remark}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\, \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead} step: the gradient is evaluated at
the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$.  This allows the
algorithm to ``correct'' its trajectory before overshooting.  With the $\beta_k = (k-1)/(k+2)$
schedule and step size $\alpha \leq 1/L$, Nesterov acceleration achieves the optimal rate:
\begin{equation}
\label{eq:nesterov_rate}
f(x_k) - f^{\star} \leq \frac{2\|x_0 - x^{\star}\|^2}{\alpha k^2} \leq \frac{2L\|x_0 - x^{\star}\|^2}{k^2}
  = O\!\left(\frac{1}{k^2}\right),
\end{equation}
compared to GD's $O(1/k)$.  This $O(1/k^2)$ rate is known to be \emph{optimal} for
first-order methods on smooth convex functions (Nesterov, 1983)---no deterministic first-order
algorithm can achieve a better dependence on $k$ without additional assumptions (such as
strong convexity).  To achieve accuracy $\varepsilon$, Nesterov requires $O(1/\sqrt{\varepsilon})$
iterations while GD requires $O(1/\varepsilon)$---a quadratic improvement in the number of
gradient evaluations.

\begin{remark}[Optimality of the $O(1/k^2)$ rate and the OGM]
\label{rem:nesterov_optimal}
The $O(1/k^2)$ rate is \emph{information-theoretically optimal}: Nesterov (1983) constructed
a ``worst-case'' function (the Chebyshev-polynomial lower bound function) on which no first-order
method querying $\nabla f$ at each iterate can achieve better than $\Omega(1/k^2)$ convergence.
Nesterov's specific momentum schedule $\beta_k = (k-1)/(k+2)$ matches this lower bound, making
it minimax optimal for the class of smooth convex functions.  More recently, Kim \& Fessler
(2016) derived the \emph{Optimized Gradient Method} (OGM), which achieves the same $O(1/k^2)$
rate with a coefficient exactly half that of Nesterov---the tightest known bound for this class.
For strongly convex problems (where the Chebyshev rate is $O((\sqrt{\kappa}-1)^k/(\sqrt{\kappa}+1)^k)$),
Nesterov's schedule is sub-optimal and a restarted variant (restarting every $O(\sqrt{\kappa})$
steps) recovers the optimal linear rate.
\end{remark}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\quad v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (biased)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (biased)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam simultaneously maintains:
\begin{enumerate}[nosep]
\item An exponential moving average of gradients ($m_t$), providing momentum (like Nesterov or
  Heavy Ball).
\item An exponential moving average of squared gradients ($v_t$), providing per-coordinate
  adaptive scaling (like RMSprop).
\end{enumerate}
The bias corrections $1/(1-\beta_j^t)$ prevent the first few iterates from being biased toward
zero due to zero initialisation of $m_0$ and $v_0$.  Standard parameters are $\beta_1 = 0.9$,
$\beta_2 = 0.999$, $\varepsilon = 10^{-8}$.

\begin{remark}[Adam convergence: theory vs practice]
While Adam is extensively used in deep learning, its convergence properties are nuanced.
Reddi et al.\ (2018) showed that Adam can fail to converge on simple convex problems when
$\beta_2$ is close to 1---the issue is that $v_t$ can shrink (when recent gradients are small)
even if historical large gradients would have required a small step, allowing the step size
to increase at the wrong moment.  The fix (AMSGrad) replaces $v_t$ with $\hat{v}_t =
\max(\hat{v}_{t-1}, v_t)$, ensuring the effective step size is non-increasing, which
restores the $O(1/\sqrt{T})$ convergence guarantee.  For the convex benchmarks in this
assignment, Adam converges well because the gradient scale is relatively stable and
$\beta_2 = 0.999$ provides sufficient averaging.  However, on Benchmark C (Rosenbrock),
Adam's per-coordinate scaling based on recent squared gradient magnitudes is less effective
than Nesterov's global momentum accumulation: the Rosenbrock valley is a curved parabola,
and the optimal descent direction couples both coordinates ($\partial^2 f/\partial x_1
\partial x_2 = -400x_1$), making coordinate-independent scaling suboptimal.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}
\label{sec:q2_sgd}

Mini-batch SGD is applied to Benchmark A with learning rate $\alpha = 0.06$, batch sizes
$b \in \{5, 40\}$, and 50 epochs.  Data indices are randomly permuted each epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}_j$ of size $b$}
        \State $\hat{g} \gets \dfrac{1}{b}\, X_{\mathcal{B}_j}^{\top}(X_{\mathcal{B}_j}\theta - y_{\mathcal{B}_j})$
          \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $\hat{g}$ is an unbiased estimator of the full gradient $\nabla J(\theta)$,
i.e.\ $\mathbb{E}[\hat{g}] = \nabla J(\theta)$, with variance:
\begin{equation}
\label{eq:sgd_var}
\operatorname{Var}(\hat{g}) = \frac{m - b}{mb} \Sigma_g, \qquad
\Sigma_g = \frac{1}{m}\sum_{i=1}^m [\nabla \ell_i(\theta) - \nabla J(\theta)][\nabla \ell_i(\theta) - \nabla J(\theta)]^{\top},
\end{equation}
where $\ell_i(\theta) = \frac{1}{2}(\theta^{\top}x^{(i)} - y^{(i)})^2$ is the per-sample loss.
The finite-population correction $(m-b)/(mb) \to 1/b$ for $b \ll m$, so variance scales as
$O(1/b)$.  There is a fundamental \textbf{bias--variance tradeoff} in batch size selection:
\begin{itemize}[nosep]
\item \emph{Small $b$:} Low compute per update, high variance, many updates per epoch.
  The stochastic noise can help escape shallow local minima in non-convex problems.
\item \emph{Large $b$:} High compute per update, low variance, few updates per epoch.
  For convex problems like Benchmark A, larger batches give smoother and more reproducible
  convergence.
\end{itemize}
Convergence theory for SGD with constant step size $\alpha$ shows that for $\mu$-strongly
convex objectives, the expected error converges to a \emph{ball}:
\begin{equation}
\label{eq:sgd_ball}
\mathbb{E}[f(x_k) - f^{\star}] \leq \left(1 - \alpha\mu\right)^k [f(x_0) - f^{\star}]
  + \frac{\alpha \sigma^2}{2\mu},
\end{equation}
where $\sigma^2 = \operatorname{tr}(\operatorname{Var}(\hat{g}))$ is the gradient variance.
The first term decays geometrically; the second is an irreducible noise floor of radius
$O(\alpha\sigma^2/\mu)$.  To achieve exact convergence, a \emph{decaying} schedule
$\alpha_k = \alpha_0/(1 + \gamma k)$ (Robbins--Monro) satisfies $\sum_k \alpha_k = \infty$
(sufficient for convergence) and $\sum_k \alpha_k^2 < \infty$ (sufficient for noise
suppression), at the cost of $O(1/k)$ rate rather than the linear rate of deterministic GD.
The mini-batch correction $(m-b)/(mb)$ in Eq.~\eqref{eq:sgd_var} shows that increasing
batch size $b$ reduces $\sigma^2$ proportionally, shifting the noise floor downward without
changing the geometric decay rate.

\begin{remark}[Optimal batch size and the linear scaling rule]
\label{rem:optimal_batch}
For a fixed total compute budget of $T$ gradient evaluations, the number of SGD steps is
$K = T/b$, and the expected error after $K$ steps with step size $\alpha = c/(b\mu)$ (scaled
with batch size) is approximately:
\begin{equation}
\mathbb{E}[f(\theta_K) - f^{\star}] \approx (f(x_0) - f^{\star})\exp(-cT/b^2)
  + \frac{c L \sigma_1^2}{\mu b},
\end{equation}
where $\sigma_1^2$ is the per-sample gradient variance.  Balancing the two terms gives
an optimal batch size $b^{\star} \propto \sigma_1 \sqrt{T\mu/(L(f(x_0)-f^{\star}))}$---increasing
with the compute budget $T$.  This ``linear scaling rule'' (Goyal et al., 2017) states that
when doubling the batch size, the learning rate should also double to maintain the same
convergence trajectory, up to $b \leq m$ (full-batch limit).  In practice, very large batches
($b > m/\sqrt{\kappa}$) provide diminishing returns because the noise floor $\sigma_1^2/b$
becomes negligible compared to the bias from using a constant step size---requiring a learning
rate decay rather than batch size increase to make further progress.  For Benchmark A with
$m = 1000$ and $\kappa \approx 1$, the critical batch size is $b^{\star} \approx m = 1000$,
so mini-batch with $b < 40$ is already in the variance-limited regime.
\end{remark}

\begin{remark}[Variance reduction: beyond mini-batch]
The irreducible noise floor~\eqref{eq:sgd_ball} of $O(\alpha\sigma^2/\mu)$ prevents constant
step-size SGD from reaching the exact optimum.  \emph{Variance reduction} methods such as
SVRG (Johnson \& Zhang, 2013) and SARAH (Nguyen et al., 2017) periodically compute a full
gradient and use it to correct the stochastic estimate, reducing the noise floor to zero.
On finite-sum objectives $f(\theta) = \frac{1}{m}\sum_{i=1}^m f_i(\theta)$ (as in Benchmark A),
SVRG achieves \emph{linear convergence} $O((1-\mu/L)^k)$ with $O(m/\mu\varepsilon)$ total
function evaluations---the best of both worlds: cheap per-step cost and exact convergence.
For Benchmark A with $m = 1000$, this would achieve the GD convergence rate while requiring
only one full gradient pass per $O(1/\mu\varepsilon)$ iterations, making it approximately
$\kappa$ times more efficient than full GD for ill-conditioned problems.
\end{remark}

\begin{remark}[Polyak--Ruppert iterate averaging]
\label{rem:polyak_ruppert}
A subtle variant of SGD returns the \emph{average} iterate $\bar{x}_T = \frac{1}{T}\sum_{k=0}^{T-1} x_k$
rather than the last iterate $x_T$.  \emph{Polyak--Ruppert averaging} (Polyak, 1992; Ruppert, 1988)
shows that with a constant step size $\alpha$, the averaged iterate satisfies:
\begin{equation}
\label{eq:polyak_ruppert}
\mathbb{E}[f(\bar{x}_T) - f^{\star}] \leq \frac{\|x_0 - x^{\star}\|^2}{2\alpha T} + \frac{\alpha\sigma^2}{2\mu}.
\end{equation}
Setting $\alpha = \|x_0-x^{\star}\|/(\sigma\sqrt{T})$ gives $O(\sigma\|x_0-x^{\star}\|/\sqrt{T})$---the
optimal $O(1/\sqrt{T})$ rate for stochastic convex optimisation without requiring a decaying
schedule.  This is noteworthy: the \emph{last} iterate of SGD with constant $\alpha$ converges
only to the noise ball~\eqref{eq:sgd_ball}, while the \emph{average} iterate achieves the
optimal statistical rate.  However, the averaging introduces a ``momentum-like'' inertia that
slows convergence near the optimum, making it less useful for problems with well-controlled
noise (large $b$) where the last iterate converges more precisely.  For Benchmark A with
$m = 1000$ and small $b$, Polyak--Ruppert averaging would reduce oscillation near the noise
floor $J^{\star} \approx 0.4826$, but the averaged $\bar{\theta}_T$ converges to the same OLS
minimiser as the last iterate $\theta_T$.
\end{remark}

\begin{remark}[Nesterov acceleration for composite objectives: FISTA]
\label{rem:fista}
Many practical objectives have the \emph{composite} form $F(x) = f(x) + g(x)$, where
$f$ is convex and $L$-smooth but $g$ is convex and \emph{non-smooth}---for example,
$g(x) = \lambda\|x\|_1$ (Lasso), $g(X) = \lambda\|X\|_*$ (nuclear norm for matrix completion),
or $g(x) = \lambda\|\nabla x\|_1$ (total variation for image denoising).  Gradient steps
on $g$ are undefined, but the \emph{proximal operator}
$\operatorname{prox}_{\alpha g}(u) = \arg\min_x \bigl\{g(x) + \tfrac{1}{2\alpha}\|x-u\|^2\bigr\}$
admits closed forms---\emph{soft-thresholding} $\operatorname{sign}(u_i)\max(|u_i|-\alpha\lambda,0)$
for $\ell_1$, and rank-$r$ SVD truncation for the nuclear norm.  FISTA (Fast Iterative
Shrinkage-Thresholding Algorithm, Beck \& Teboulle, 2009) applies Nesterov's lookahead
to the \emph{proximal gradient} step:
\begin{equation}
\label{eq:fista}
y_{k+1} = x_k + \beta_k(x_k - x_{k-1}), \quad
x_{k+1} = \operatorname{prox}_{\alpha g}\!\bigl(y_{k+1} - \alpha \nabla f(y_{k+1})\bigr),
\end{equation}
with $\beta_k = (t_k - 1)/t_{k+1}$, $t_{k+1} = (1 + \sqrt{1 + 4t_k^2})/2$.  FISTA achieves
the same $O(1/k^2)$ convergence rate~\eqref{eq:nesterov_rate} for composite objectives---the
same improvement from $O(1/\varepsilon)$ to $O(1/\sqrt{\varepsilon})$ evaluations of
$\nabla f$ and $\operatorname{prox}_g$ as Nesterov provides for smooth problems.
\end{remark}

\begin{remark}[Katyusha: optimal stochastic momentum for strongly convex finite sums]
\label{rem:katyusha}
Can Nesterov's $O(\sqrt{\kappa})$ improvement be combined with SVRG's variance reduction?
\emph{Katyusha} (Allen-Zhu, 2017) achieves this by applying a ``negative momentum'' technique
to SVRG, reaching the optimal complexity
\begin{equation}
\label{eq:katyusha}
O\!\left(\bigl(n + \sqrt{n\kappa}\bigr)\log\tfrac{1}{\varepsilon}\right)
\end{equation}
gradient evaluations for $\mu$-strongly convex, $L$-smooth finite-sum objectives
$f = \frac{1}{n}\sum_i f_i$.  This matches the lower bound (Woodworth \& Srebro, 2016),
making it the \emph{first optimal} stochastic algorithm for this problem class.
The improvement over SVRG's $O(n + \kappa)\log(1/\varepsilon)$ is significant for
ill-conditioned problems: when $\kappa = 2508$ and $n = 1000$, Katyusha needs roughly
$O(\sqrt{n\kappa}) = O(\sqrt{2.5\times10^6}) \approx 1580$ gradient evaluations to halve
the error, whereas SVRG needs $O(n + \kappa) \approx 3508$ evaluations.  For Benchmark A,
the near-isotropic Hessian gives $\kappa \approx 1$ and the improvement is negligible;
the benefit appears for large-scale ill-conditioned regression problems.
\end{remark}

The irreducible noise floor $J^{\star} \approx \sigma^2/2$ cannot be eliminated by any

algorithm.  Concretely, the Bias--Variance decomposition of the OLS estimator gives

model complexity and Var$(\hat{\theta})$) or data collection (to reduce $\sigma^2$ or
increase $m$) rather than more aggressive optimisation.
\end{remark}

\subsection{Effect of Condition Number on Convergence Speed}
\label{sec:condition_number}

The condition number $\kappa = L/\mu$ fundamentally limits first-order convergence rates.
For strongly convex, $L$-smooth objectives, GD with step $\alpha = 2/(L+\mu)$ satisfies
\begin{equation}
\label{eq:gd_kappa}
f(x_k) - f^{\star} \leq \left(\frac{\kappa - 1}{\kappa + 1}\right)^{2k} [f(x_0) - f^{\star}].
\end{equation}
As $\kappa \to \infty$, this rate approaches $1$---GD stagnates.  Nesterov acceleration
replaces $(\kappa-1)/(\kappa+1)$ with $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$, requiring only
$O(\sqrt{\kappa})$ iterations versus $O(\kappa)$ for GD.

The ratio $(\kappa-1)/(\kappa+1)$ arises naturally from Chebyshev polynomial theory: the
convergence factor of GD on a quadratic $f(x) = \frac{1}{2}x^{\top}Hx$ is determined by
how well a polynomial $p_k(\lambda)$ of degree $k$ can uniformly approximate zero on the
spectrum $[\lambda_{\min}, \lambda_{\max}]$ while satisfying $p_k(0) = 1$.  The optimal
polynomial is the Chebyshev polynomial of degree $k$, which achieves
$\max_{\lambda \in [\mu, L]} |1 - \lambda p_{k-1}(\lambda)| = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^k$---
exactly the Nesterov rate.  In this sense, Nesterov's momentum schedule is the
\emph{polynomial preconditioner} that concentrates the spectrum onto the Chebyshev nodes,
explaining both the $O(\sqrt{\kappa})$ improvement and why no first-order method can do
better on this problem class without additional structure.

\centering
\includegraphics[width=\textwidth]{figures/condition_number_effect.pdf}
\caption{Effect of condition number $\kappa$ on convergence rates.  Left: GD convergence
  (semi-log) for quadratics with $\kappa \in \{1, 5, 25, 100, 500\}$ using the optimal

\subsection{Local Approximations of $g(x) = x^4$}

The Taylor expansion of $g(x) = x^4$ around $x_0 = 0.25$ provides the theoretical
justification for Newton's method.

\textbf{First-order (affine) approximation:}
\begin{equation}
\label{eq:taylor_first}
g_1(x) = g(x_0) + g'(x_0)(x - x_0)
        = 0.25^4 + 4 \cdot 0.25^3(x - 0.25)
        = 0.0039 + 0.0625(x - 0.25).
\end{equation}
This is the tangent line and minimises only to $x^{\star} \to -\infty$ (unbounded), so a
gradient step uses a fixed step size rather than minimising $g_1$.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
\label{eq:taylor_second}
g_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
        = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
Minimising $g_2$ with respect to $x$ yields the \emph{Newton step}:
\begin{equation}
x_{\mathrm{Newton}} = x_0 - \frac{g'(x_0)}{g''(x_0)}
                    = 0.25 - \frac{0.0625}{0.75}
                    = 0.25 - 0.0833 = 0.1667.
\end{equation}
This places the iterate at the minimum of the local parabola, using curvature information
that a first-order method ignores.

\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3 Local approximations of $g(x) = x^4$ at $x_0 = 0.25$.  The first-order
  approximation (tangent line) captures only the local slope and deviates rapidly from the
  true function.  The second-order approximation (parabola) captures the curvature and fits

\subsection{Newton's Method with Hessian Regularisation}

\begin{algorithm}[H]
\caption{Newton's Method (Damped / Regularised)}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, Hessian regularisation $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise to ensure positive definiteness}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: solve $H_k p_k = g_k$}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\paragraph{Cost per iteration.}
Each Newton step requires: (i) computing $\nabla^2 f(x_k)$---$O(d^2)$ entries for a dense
Hessian; (ii) solving a $d \times d$ linear system---$O(d^3)$ using Cholesky factorisation.
This compares unfavourably to GD's $O(d)$ per step, but Newton requires far fewer iterations.

\begin{remark}[Damped Newton and the trust-region interpretation]
\label{rem:trust_region}
The implementation uses a \emph{damped Newton step} $x_{k+1} = x_k - \alpha p_k$ with fixed
step size $\alpha < 1$ rather than a full Newton step ($\alpha = 1$).  This is motivated by the
fact that Newton's local quadratic model $m_k(p) = f(x_k) + g_k^{\top}p + \frac{1}{2}p^{\top}H_k p$
may be inaccurate for large $\|p_k\|$ when the Hessian changes rapidly (as on Rosenbrock, where
$\|\nabla^3 f\|$ is large).  An alternative to damping is the \emph{trust-region method}, which
explicitly limits the step norm:
\begin{equation}
\label{eq:trust_region}
p_k = \arg\min_{\|p\| \leq \Delta_k} m_k(p)
     = -(H_k + \lambda_k I)^{-1} g_k,
\end{equation}
where $\Delta_k$ is an adaptively adjusted trust-region radius and $\lambda_k \geq 0$ is chosen
so that $\|p_k\| = \Delta_k$ (if the unconstrained minimum exceeds the radius).  Trust-region
methods have global convergence guarantees independent of the initial point, and locally achieve
the same quadratic rate as Newton when the quadratic model is accurate ($\Delta_k$ large) and
linear rate when the model is poor ($\Delta_k$ small).  The Hessian regularisation
$\nabla^2 f + \lambda I$ used in this implementation is equivalent to trust-region with a fixed
$\lambda$ rather than an adaptive $\Delta_k$, providing robustness at the cost of not adapting
the step size to the local model quality.
\end{remark}

\paragraph{Convergence theory.}
Near a strongly convex minimum $x^{\star}$, Newton's method converges \emph{quadratically}:
\begin{equation}
\label{eq:newton_quadratic}
\|x_{k+1} - x^{\star}\| \leq C\,\|x_k - x^{\star}\|^2
\end{equation}
for some constant $C > 0$, provided the Hessian is Lipschitz continuous.  This means the
number of correct decimal digits roughly doubles at each step.

\paragraph{Affine invariance.}
Newton's method is \emph{affine invariant}: if $f(x)$ is replaced by $f(Ax)$ for invertible
$A$, the Newton iterates are identical up to the linear change of variables.  This property
is crucial for ill-conditioned problems---unlike GD, Newton's convergence rate is independent
of $\kappa(H)$.  Concretely, on Benchmark A (where $\kappa(H) \approx 1$) and Benchmark C
(where $\kappa(H) \approx 2508$ at the optimum), Newton's \emph{local} convergence behaviour
is identical: both require the same number of iterations once the quadratic regime is reached.

\begin{proposition}[Exact one-step convergence on quadratics]
\label{prop:newton_quadratic}
Let $f(x) = \tfrac{1}{2}x^{\top}Hx - b^{\top}x$ be a strictly convex quadratic with $H \succ 0$.
A single \emph{undamped} Newton step ($\alpha = 1$) from any $x_0$ reaches $x^{\star} = H^{-1}b$
exactly: $x_1 = x_0 - H^{-1}\nabla f(x_0) = x_0 - H^{-1}(Hx_0 - b) = H^{-1}b = x^{\star}$.
\end{proposition}
This is confirmed empirically on Benchmark A (Figure~\ref{fig:q3_fval_A}): the objective
drops from $J(x_0)$ to the noise floor $J^{\star} = 0.4826$ in exactly one Newton step.

\paragraph{Quasi-Newton methods.}
The $O(d^3)$ cost of Newton steps motivates \emph{quasi-Newton methods}, which maintain a
matrix $B_k \approx H_k^{-1}$ (inverse Hessian approximation) updated using only gradient
differences $y_k = \nabla f(x_{k+1}) - \nabla f(x_k)$ and step vectors $s_k = x_{k+1} - x_k$.
The \textbf{BFGS} (Broyden--Fletcher--Goldfarb--Shanno) rank-2 update is:
\begin{equation}
\label{eq:bfgs}
B_{k+1} = \left(I - \rho_k s_k y_k^{\top}\right) B_k \left(I - \rho_k y_k s_k^{\top}\right)
         + \rho_k s_k s_k^{\top},
\quad \rho_k = \frac{1}{y_k^{\top} s_k},
\end{equation}
where the two rank-1 terms maintain the secant condition $B_{k+1} y_k = s_k$ and positive
definiteness (when $y_k^{\top} s_k > 0$, guaranteed by the Wolfe line search conditions).
This rank-2 update achieves \emph{superlinear} convergence (between linear and quadratic)
while costing $O(d^2)$ per step.  L-BFGS further reduces cost to $O(dm)$ (where $m \leq 20$ is the history length),
making it the dominant method for large-scale unconstrained optimisation.  In practice, L-BFGS
with $m = 10$--$20$ history pairs achieves near-Newton accuracy on smooth problems while
requiring only gradient evaluations---no Hessian storage or factorisation.  This makes it
the default method in \texttt{scipy.optimize} and modern machine learning frameworks for
smooth, moderately-sized problems ($d \lesssim 10^6$).

\begin{remark}[Dennis--Moré theorem: why BFGS achieves superlinear convergence]
\label{rem:dennis_more}
BFGS achieves superlinear convergence ($\|e_{k+1}\|/\|e_k\| \to 0$) under the
\emph{Dennis--Moré condition} (1974): the quasi-Newton matrix $B_k$ satisfies
\begin{equation}
\label{eq:dennis_more}
\lim_{k \to \infty} \frac{\|(B_k - H^{\star})(x_{k+1} - x_k)\|}{\|x_{k+1} - x_k\|} = 0,
\end{equation}
where $H^{\star} = \nabla^2 f(x^{\star})$.  Remarkably, $B_k$ need not converge to $H^{\star}$
globally---it only needs to approximate $H^{\star}$ accurately \emph{along the step direction}.
BFGS satisfies this because the secant condition $B_{k+1}s_k = y_k$ ensures $B_{k+1}$
exactly matches the curvature along the most recently taken direction, and the rank-2 update
is minimal (in a weighted Frobenius norm) among all matrices satisfying the secant condition
and symmetry---concentrating accuracy where it is most needed.  For Rosenbrock ($\kappa = 2508$),
BFGS eventually learns the valley-floor curvature ($\partial^2 f/\partial x_1 \partial x_2$
coupling) through accumulated secant information and takes near-Newton steps---without ever
forming $\nabla^2 f$ explicitly.  The L-BFGS variant truncates the history to $m$ pairs,
which means the approximation quality degrades for directions not recently visited, but
empirically $m = 10$--$20$ pairs suffice to maintain superlinear convergence in practice.
\end{remark}

Another important alternative is the \emph{conjugate gradient method} (CG), which solves
the Newton system $H_k p_k = g_k$ iteratively using only matrix-vector products $H_k v$---
enabling $O(d)$ cost per CG iteration when the Hessian has structure.  Truncated-Newton
methods use CG to approximately solve the Newton system, combining Newton's curvature
exploitation with first-order computational cost.  For Benchmark A (quadratic), exact CG
terminates in at most $d = 2$ steps (i.e., 2 matrix-vector products) and achieves the
same one-step convergence as full Newton---a direct consequence of the Krylov space
exactly spanning the solution in $d$ conjugate directions.

\begin{remark}[CG and BFGS: equivalent on quadratics, complementary on non-quadratics]
\label{rem:cg_bfgs}
On a strictly convex quadratic $f(x) = \frac{1}{2}x^{\top}Hx - b^{\top}x$, the Fletcher--Reeves
variant of nonlinear CG and BFGS with exact line search are \emph{equivalent}: both generate
the same iterates (Theorem 5.2, Nocedal \& Wright, 2006).  After $k$ steps, both methods
span the same Krylov subspace $\mathcal{K}_k(H, g_0) = \operatorname{span}\{g_0, Hg_0, \ldots, H^{k-1}g_0\}$,
and the optimal solution within this space is found by the same polynomial of degree $k$ applied
to $g_0$.  For $d = 2$ (Benchmark A), both methods terminate in exactly 2 steps.  On non-quadratics
(Benchmarks B and C), they differ: CG uses the local gradient direction and a conjugacy update
(Polak--Ribière or Fletcher--Reeves) to maintain approximate conjugacy, while BFGS builds a
global Hessian approximation from all historical secant pairs.  For Rosenbrock ($\kappa = 2508$),
BFGS outperforms CG because its global inverse Hessian approximation captures the valley-floor
curvature coupling ($H_{12} = -400x_1$), whereas CG's local conjugacy update loses this
information at each restart.
\end{remark}

\subsection{Convergence Order Verification}

\begin{remark}[Newton on degenerate zeros: $f(x) = (x-1)^4$]
\label{rem:newton_degenerate}
The function $f(x) = (x-1)^4$ has a \emph{degenerate} minimum at $x^{\star} = 1$: the
Hessian $f''(x^{\star}) = 12(x^{\star}-1)^2 = 0$ vanishes.  The Newton step is
\begin{equation}
x_{k+1} = x_k - \frac{f'(x_k)}{f''(x_k)} = x_k - \frac{4(x_k-1)^3}{12(x_k-1)^2}
         = x_k - \frac{x_k - 1}{3},
\end{equation}
so $e_{k+1} = x_{k+1} - 1 = \tfrac{2}{3}(x_k - 1) = \tfrac{2}{3}e_k$.  This is \emph{linear}
convergence with rate $2/3$---\emph{not} quadratic.  The standard quadratic rate
$\|e_{k+1}\| \leq C\|e_k\|^2$ requires the Hessian at $x^{\star}$ to be \emph{non-singular},
which fails here.  The empirical log--log slope lying between 1 and 2 at early iterations is
an artefact of the transient: for large $|e_k|$ the nonlinear terms give apparent superlinearity,
but asymptotically the slope converges to exactly 1 (linear convergence at rate $2/3$).
For GD on $f(x) = (x-1)^4$, the gradient $f'(x) = 4(x-1)^3$ is not Lipschitz continuous near $x^{\star}$
(since $f''(x^{\star}) = 0$), so standard GD theory does not directly apply---but empirically,
GD still converges linearly with a smaller rate, visible in the figure.
\end{remark}

\subsection{Part A: Finite Difference Gradient Approximation and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Approximation}

The forward finite-difference approximation replaces each partial derivative with a
difference quotient:
\begin{equation}
\label{eq:fd}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector.  Taylor expanding $f(x + \delta e_i)$ gives:
\begin{equation}
\label{eq:fd_error}
\frac{f(x + \delta e_i) - f(x)}{\delta} = \frac{\partial f}{\partial x_i}(x)
  + \underbrace{\frac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x) + O(\delta^2)}_{\text{truncation error}}
  + \underbrace{O\!\left(\frac{\varepsilon_{\mathrm{mach}}}{\delta}\right)}_{\text{rounding error}},
\end{equation}
where $\varepsilon_{\mathrm{mach}} \approx 2.2 \times 10^{-16}$ for IEEE double precision.
The total error is minimised when truncation $\sim$ rounding:
$\delta/2 \approx \varepsilon_{\mathrm{mach}}/\delta$, giving
$\delta^{\star} = \sqrt{2\varepsilon_{\mathrm{mach}}} \approx 2 \times 10^{-8}$.
The assignment tests $\delta = 0.05$ (near-optimal for the function scale, where typical
gradient magnitudes are $O(1)$) and $\delta = 0.8$ (truncation error dominates: the finite
difference spans a region where $\partial^2 f/\partial x_i^2$ is non-negligible, causing
the gradient estimate to reflect a smoothed version of $f$).
Cost: $d+1$ function evaluations per gradient estimate ($d = 2$ here); for $d$-dimensional
problems, cost scales linearly with $d$, making FD impractical for $d \gg 10^3$.

\begin{remark}[Central difference vs forward difference]
\label{rem:central_diff}
The \emph{central difference} scheme
$\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}$
has truncation error $O(\delta^2)$ (the odd-order terms cancel in the Taylor expansion) rather
than $O(\delta)$ for the forward scheme.  Balancing against rounding error $O(\varepsilon_{\mathrm{mach}}/\delta)$ gives
optimal $\delta_{\mathrm{CD}}^{\star} = (6\varepsilon_{\mathrm{mach}})^{1/3} \approx 8.6 \times 10^{-6}$---roughly
100 times larger than $\delta_{\mathrm{FD}}^{\star}$, and correspondingly more tolerant of rounding.
However, central differences require $2d$ evaluations per gradient (vs $d+1$ for forward), doubling
the evaluation budget.  For the typical function scale in this assignment
(gradient $O(1)$, $\partial^2 f/\partial x_i^2 \sim O(5)$), the optimal forward-difference
$\delta^{\star} \approx 2 \times 10^{-8}$ and optimal central-difference $\delta^{\star} \approx 10^{-5}$ both
yield errors near machine precision; the choice $\delta = 0.05$ is deep in the truncation-dominated
regime for both schemes, where central differences would give roughly
$0.05^2 \times 5 / 2 \approx 0.006$ error (vs $0.05 \times 5 / 2 = 0.125$ for forward).
\end{remark}

\subsubsection{Nesterov Random Search (Zeroth-Order Method)}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, then normalise: $u_k \gets u_k/\|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
      \Comment{Stochastic directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates a directional derivative along a uniformly random direction
on the unit sphere.  Each step requires only \emph{two} function evaluations regardless of
dimension---a crucial advantage in high dimensions.

The estimator $\hat{g}_k = \frac{f(x+\delta u) - f(x)}{\delta}u$ can be understood as a
gradient estimator via the Gaussian smoothing identity:
\begin{equation}
\label{eq:nrs_expectation}
\mathbb{E}_u[\hat{g}_k] = \nabla f_\delta(x), \qquad
f_\delta(x) = \mathbb{E}_u[f(x + \delta u)] \approx f(x) + \frac{\delta^2}{2d}\Delta f(x),
\end{equation}
where $f_\delta$ is a $\delta$-smoothed version of $f$.  Thus NRS is a gradient descent
method on the smoothed objective $f_\delta$, with bias $O(\delta^2)$ and variance
$O(L^2\delta^2)$.  Convergence bounds on the original $f$ are $O(d/\sqrt{k})$ for convex
problems---slower than GD's $O(1/\sqrt{k})$ by a factor of $\sqrt{d}$ reflecting the
additional uncertainty from random direction sampling.

\begin{remark}[Dimension dependence of zeroth-order methods: evaluation budget perspective]
\label{rem:nrs_dim}
The $\sqrt{d}$ iteration penalty for NRS over GD translates to a larger \emph{function
evaluation} advantage than it may first appear.  Each GD step requires $1$ gradient
evaluation ($O(d)$ operations if the gradient has closed form) or $d+1$ function
evaluations via forward FD.  Each NRS step uses only $2$ evaluations regardless of $d$.
For a budget of $T$ total function evaluations:
\begin{itemize}[nosep]
\item \textbf{FD-gradient GD:} performs $T/(d+1)$ steps, achieving accuracy $O(d\sqrt{L}/\sqrt{T})$ for convex $f$.
\item \textbf{NRS:} performs $T/2$ steps, achieving accuracy $O(d\sqrt{L}/\sqrt{T})$ for convex $f$.
\end{itemize}
Asymptotically both achieve the \emph{same} $O(1/\sqrt{T})$ evaluation complexity---the NRS
iteration factor $\sqrt{d}$ penalty is exactly compensated by its $(d+1)/2$-fold cheaper
per-step cost.  The key difference is memory: NRS stores only $O(d)$ (current iterate),
while FD GD stores the same; but NRS can be implemented as a \emph{streaming} algorithm
with one function call per random direction, making it suitable for online/distributed
settings.  For $d = 2$ (as in this assignment), NRS costs $3/2 \approx 1.5\times$ more
function evaluations than FD for the same iteration count---barely visible in the convergence
plots.  The true advantage of NRS emerges for $d \gg 10^3$, where FD's $d+1$ evaluations per
gradient become prohibitive while NRS maintains its 2-evaluation-per-step budget.
\end{remark}

\subsection{Part B: Derivative-Free Methods on Benchmark C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$ and
iteratively transforms it via four geometric operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): Move the worst vertex through the centroid of the
  best $n$ vertices.
\item \textbf{Expansion} ($\gamma = 2.0$): If reflection improved, attempt a further step.
\item \textbf{Contraction} ($\rho = 0.5$): If reflection did not improve, contract towards
  the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): If contraction also fails, contract all vertices towards
  the best vertex.
\end{enumerate}
No gradient information is required; only function evaluations.  Settings: initial simplex step
$= 0.35$, 160 iterations.

\begin{remark}
Nelder--Mead has no general convergence guarantee for $d \geq 2$ (McKinnon, 1998).  However,
it works well in practice on smooth, low-dimensional functions and is widely used for
hyperparameter tuning.  The simplex adaptively contracts towards the optimum, effectively
learning local geometry without derivatives.  The intuition for its empirical success is
that on smooth functions, the simplex ``senses'' the local curvature through function
evaluations at its $d+1$ vertices, providing implicit first- and second-order information
at the cost of $O(d+1)$ function evaluations per simplex transformation.  For $d = 2$,
this gives 3 function evaluations per step---the same as FD gradient estimation---but with
an adaptive geometry that contracts and expands based on local progress rather than using
a fixed perturbation size.  For ill-conditioned problems like Rosenbrock ($\kappa \approx 2508$),
the simplex naturally elongates along the low-curvature valley direction, providing implicit
preconditioning that a fixed $\delta$ perturbation cannot achieve.
\end{remark}

\subsubsection{Grid Search}

Grid search evaluates the objective on a uniform $55 \times 55 = 3025$ point grid over
$x_1 \in [-2, 2]$, $x_2 \in [-1, 3]$.  The grid spacing is $\Delta x_1 = 4/54 \approx 0.074$
and $\Delta x_2 = 4/54 \approx 0.074$, so the best-found point is within $\sqrt{2}\Delta/2 \approx 0.052$
of the true minimum in $\ell^\infty$ norm.

\begin{remark}[Grid search function value error and condition number]
\label{rem:grid_error}
The $\ell^\infty$ distance $\Delta/2 \approx 0.037$ from the grid point to $(1,1)$ translates
to a function value error bounded by Taylor's theorem:
\begin{equation}
\label{eq:grid_error}
f(\hat{x}) - f(x^{\star}) \leq \tfrac{1}{2}\lambda_{\max}(H(x^{\star}))\|\hat{x} - x^{\star}\|^2
\leq \tfrac{1}{2} \times 1001.6 \times (0.052)^2 \approx 1.35.
\end{equation}
Concretely, the grid search finds a point with $f \approx 1.35$ above the minimum---not
particularly accurate for this level of evaluation budget.  Conversely, to achieve $f - f^{\star}
\leq 0.01$, the required grid spacing satisfies
$\Delta \leq \sqrt{2 \times 0.01/\lambda_{\max}} = \sqrt{0.02/1001.6} \approx 0.0045$,
requiring a grid of size $N = (4/0.0045)^2 \approx 790{,}000$ points in $d = 2$---far more
than the 3025-point grid used.  This illustrates the \emph{condition number effect} on
grid search: high $\lambda_{\max}$ means the function value changes rapidly, requiring very
fine grids even in $d = 2$.  Preconditioning (aligning the grid with the eigenvectors of
$H(x^{\star})$) would reduce the required grid points by a factor of $\kappa \approx 2508$.
\end{remark}

\begin{remark}[Bayesian optimisation: sample-efficient alternative to grid search]
\label{rem:bayesian_opt}
For expensive black-box objectives (e.g.\ hyperparameter tuning where each evaluation
requires hours of computation), neither grid search nor Nelder--Mead is appropriate---the
former requires $O(N^d)$ evaluations and the latter needs many function calls to contract.
\emph{Bayesian optimisation} (Snoek et al., 2012) fits a probabilistic surrogate model
(Gaussian Process) to the observed function values and uses an \emph{acquisition function}
(Expected Improvement, UCB) to select the next evaluation point, balancing exploration of
the space and exploitation of the current best.  Bayesian optimisation typically finds
near-optima in $O(10)$--$O(100)$ evaluations, far fewer than grid search's $O(N^d)$, making
it the method of choice for hyperparameter tuning in deep learning.  For the Rosenbrock
function in $d = 2$, Bayesian optimisation would typically find a solution to within
$\varepsilon = 0.01$ in fewer than 50 evaluations---compared to grid search's 3025.
The key requirement is the ability to compute the GP posterior efficiently, which limits
Bayesian optimisation to $d \lesssim 20$ dimensions (GP regression scales as $O(n^3)$ in
the number of observations $n$).
\end{remark}

The key insight is the \textbf{dimension penalty} of derivative information:
\begin{itemize}[nosep]
\item Finite-difference GD costs $d+1$ evaluations per step---the same scaling as exact GD
  with the additional overhead of approximation error.  For $d = 1000$, this means 1001
  function evaluations per gradient.
\item Nesterov random search costs 2 evaluations regardless of $d$.  For $d = 1000$, this
  is 500$\times$ cheaper per step---but convergence requires $O(d/\varepsilon^2)$ steps
  (vs.\ $O(\log(1/\varepsilon))$ for Newton), so the total evaluation budget may be higher.
\item Grid search suffers the \emph{curse of dimensionality}: $N^d$ evaluations for a grid
  of $N$ points per dimension.  A 10-point grid in $d = 10$ requires $10^{10}$ evaluations,
  making it infeasible for $d > 4$ in practice.
\end{itemize}
The fundamental result (Nemirovski--Yudin, 1983) is that any zeroth-order (function-only)
method requires $\Omega((L/\varepsilon)^{d/2})$ evaluations to find an $\varepsilon$-optimal
point for a $d$-dimensional problem, regardless of the specific algorithm.  This exponential
lower bound cannot be circumvented without gradient information, quantifying the fundamental
difficulty of derivative-free optimisation.

\begin{remark}[Practical implications of the oracle lower bound]
For the Rosenbrock function ($d = 2$, $L \approx 1600$) with target accuracy
$\varepsilon = 0.01$, the lower bound predicts
$\Omega((1600/0.01)^{2/2}) = \Omega(1.6 \times 10^5)$ function evaluations
for any derivative-free method.  In contrast, Newton's method (first-order plus Hessian)
requires $O(\log\log(1/\varepsilon))$ iterations in the quadratic convergence regime---orders
of magnitude fewer.  For $d = 100$ and the same parameters, the bound becomes
$\Omega((1600/0.01)^{50}) \approx \Omega(10^{265})$, rendering derivative-free methods
computationally infeasible.  This underscores why gradient information is not merely
convenient but \emph{computationally essential} for high-dimensional optimisation.
\end{remark}

\subsection{Problem Formulation and Optimality Conditions}

Minimise Benchmark B subject to the inequality constraint $x_1 \geq 0.5$:
\begin{equation}
\label{eq:q5_problem}
\min_{x \in \mathcal{F}} f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1), \qquad
\mathcal{F} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}.
\end{equation}
The unconstrained minimiser $x^{\star}_{\mathrm{unc}} \approx (0.5754, 2.0)$ satisfies $x_1^{\star} > 0.5$,
so the constraint is \emph{inactive} at the optimum---the constrained and unconstrained minima
coincide at $f^{\star} \approx 0.7244$.  However, the starting point $x_0 = (0.2, 4.0)$ is
\emph{infeasible} ($x_1 = 0.2 < 0.5$), making the initial approach to feasibility the key
challenge.

\paragraph{KKT optimality conditions.}
For the problem~\eqref{eq:q5_problem}, the Karush--Kuhn--Tucker (KKT) conditions require:
\begin{align}
\nabla f(x^{\star}) - \mu^{\star} e_1 &= 0, \label{eq:kkt_stationarity}\\
x_1^{\star} - 0.5 &\geq 0, \label{eq:kkt_primal}\\
\mu^{\star} &\geq 0, \label{eq:kkt_dual}\\
\mu^{\star}(x_1^{\star} - 0.5) &= 0, \label{eq:kkt_complementarity}
\end{align}
where $e_1 = [1, 0]^{\top}$ and $\mu^{\star} \geq 0$ is the dual variable.  Since the
unconstrained minimum $(0.5754, 2.0)$ is feasible, the complementarity
condition~\eqref{eq:kkt_complementarity} implies $\mu^{\star} = 0$ (constraint is inactive),
and Eq.~\eqref{eq:kkt_stationarity} reduces to $\nabla f(x^{\star}) = 0$ as in the
unconstrained case.  This confirms that the projected GD and penalty methods converge to
the same point as unconstrained GD.

\paragraph{Constraint qualification.}
The KKT conditions are \emph{necessary} for optimality when a \emph{constraint qualification}
(CQ) holds.  For the half-space constraint $g(x) = 0.5 - x_1 \leq 0$, the
\emph{Linear Independence Constraint Qualification} (LICQ) requires that the active constraint
gradients are linearly independent at $x^{\star}$.  Since the only constraint is $g(x) \leq 0$
with $\nabla g = -e_1 = [-1, 0]^{\top}$, LICQ trivially holds whenever the constraint is active
(one constraint, one gradient, trivially independent).  When the constraint is \emph{inactive}
($\mu^{\star} = 0$), the KKT conditions are equivalent to the unconstrained stationarity
$\nabla f(x^{\star}) = 0$, and are both necessary and sufficient for a local minimum of the
convex objective.  For convex $f$ and convex $\mathcal{F}$, KKT conditions are globally sufficient.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{F}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{F}}(z_{k+1}) = \bigl[\max(0.5,\, z_1),\ z_2\bigr]^{\top}$
      \Comment{Project onto half-space}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space $\{x : x_1 \geq 0.5\}$, the projection is simply a clipping operation:
$\Pi_{\mathcal{F}}(z)_1 = \max(0.5, z_1)$, $\Pi_{\mathcal{F}}(z)_2 = z_2$.  This costs
$O(1)$ per iteration.

\begin{theorem}[Projected GD convergence rate]
\label{thm:pgd_convergence}
Let $f : \mathbb{R}^d \to \mathbb{R}$ be $\mu$-strongly convex and $L$-smooth, and let
$\mathcal{F}$ be a non-empty closed convex set with constrained minimiser
$x^{\star} = \arg\min_{x \in \mathcal{F}} f(x)$.  For PGD with step size $\alpha \in (0, 1/L]$:
\begin{equation}
\label{eq:pgd_convergence}
\|x_k - x^{\star}\|^2 \leq (1 - \alpha\mu)^k \|x_0 - x^{\star}\|^2.
\end{equation}
With optimal $\alpha = 1/L$, this gives the rate $(1 - \mu/L)^k = (1 - 1/\kappa)^k$---identical
to unconstrained GD on strongly convex functions.
\end{theorem}
\begin{proof}[Proof sketch]
Since $x^{\star} \in \mathcal{F}$ and $\Pi_{\mathcal{F}}$ is a non-expansive projection
($\|\Pi_{\mathcal{F}}(u) - \Pi_{\mathcal{F}}(v)\| \leq \|u - v\|$):
\begin{align*}
\|x_{k+1} - x^{\star}\|^2
&= \bigl\|\Pi_{\mathcal{F}}(x_k - \alpha\nabla f(x_k)) - \Pi_{\mathcal{F}}(x^{\star})\bigr\|^2 \\
&\leq \|(x_k - \alpha\nabla f(x_k)) - x^{\star}\|^2 \\
&= \|x_k - x^{\star}\|^2 - 2\alpha\langle\nabla f(x_k),\, x_k - x^{\star}\rangle
   + \alpha^2\|\nabla f(x_k)\|^2.
\end{align*}
Strong convexity gives $\langle\nabla f(x_k), x_k - x^{\star}\rangle \geq \mu\|x_k - x^{\star}\|^2$,
and $L$-smoothness gives $\|\nabla f(x_k)\|^2 \leq 2L[f(x_k) - f(x^{\star})] \leq L^2\|x_k - x^{\star}\|^2$
(via descent lemma).  Substituting and using $\alpha \leq 1/L$ yields $(1-\alpha\mu)$.
Recursing from $k = 0$ gives~\eqref{eq:pgd_convergence}.
\end{proof}

\begin{algorithm}[H]
\caption{Exterior Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\State Define $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$
  \Comment{Penalise infeasibility}
\For{$k = 0, 1, \ldots, N-1$}
    \State $\nabla F_{\lambda}(x_k) \gets \nabla f(x_k) + \lambda [-\mathbf{1}_{x_1 < 0.5},\, 0]^{\top}$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty gradient adds a constant force $-\lambda$ pushing the iterate towards $x_1 = 0.5$
whenever $x_1 < 0.5$.  As $\lambda \to \infty$, the penalty method solution converges to the
true constrained optimum, but large $\lambda$ makes the penalised objective ill-conditioned
(condition number scales as $\kappa \sim \lambda/\mu$), requiring correspondingly smaller step sizes.

\paragraph{Augmented Lagrangian (method of multipliers).}
The \emph{augmented Lagrangian} overcomes the ill-conditioning limitation of pure penalty methods
by combining a penalty with a multiplier estimate $\hat{\mu}$:
\begin{equation}
\label{eq:augmented_lagrangian}
\mathcal{L}_{\rho}(x, \hat{\mu}) = f(x) + \hat{\mu}\,h(x) + \frac{\rho}{2}\,h(x)^2,
\end{equation}
where $h(x) = \max(0, 0.5 - x_1) \geq 0$ is the constraint violation and $\rho > 0$ is a
fixed penalty parameter.  After minimising over $x$, the multiplier is updated as
$\hat{\mu} \leftarrow \hat{\mu} + \rho\,h(x_k)$.  This dual ascent keeps $\rho$ fixed (avoiding
ill-conditioning) while $\hat{\mu}$ converges to the true dual variable $\mu^{\star} = 0$
(since the constraint is inactive here).  The augmented Lagrangian method is the theoretical
foundation of \emph{ADMM} (Alternating Direction Method of Multipliers), widely used in
distributed optimisation and statistical learning.

\begin{remark}[KKT conditions and the projected GD fixed-point]
\label{rem:kkt_pgd}
At the projected GD fixed point $x^{\star} = \Pi_{\mathcal{F}}(x^{\star} - \alpha\nabla f(x^{\star}))$,
the iterate satisfies the KKT optimality conditions for the constrained problem
$\min_{x : g(x) \leq 0} f(x)$ with $g(x) = 0.5 - x_1$.  These are:
\begin{equation}
\label{eq:kkt}
\nabla f(x^{\star}) + \mu^{\star}\nabla g(x^{\star}) = 0, \quad
\mu^{\star} \geq 0, \quad
\mu^{\star} g(x^{\star}) = 0 \quad \text{(complementary slackness)}.
\end{equation}
Since the unconstrained minimiser $x_1^{\star} \approx 0.576 > 0.5$ renders the constraint
inactive ($g(x^{\star}) < 0$), complementary slackness forces $\mu^{\star} = 0$, and the
unconstrained condition $\nabla f(x^{\star}) = 0$ applies.  Geometrically, the PGD
fixed-point condition $x^{\star} = \Pi_{\mathcal{F}}(x^{\star} - \alpha g)$ is equivalent
to the \emph{variational inequality}: $\nabla f(x^{\star})^{\top}(y - x^{\star}) \geq 0$
for all $y \in \mathcal{F}$---the gradient $\nabla f(x^{\star})$ lies in the normal cone
$N_{\mathcal{F}}(x^{\star})$.  For an active constraint ($x_1^{\star} = 0.5$), the normal
cone at the boundary is $\{v : v_1 \leq 0\}$, and $\mu^{\star} = -(\partial f/\partial x_1)|_{x_1=0.5}$
gives the shadow price of the constraint---the rate at which the optimal value would
increase if the constraint were tightened.
\end{remark}

\subsection{Feasible Set and Linear Programme}

The feasible set is the axis-aligned box $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\
-5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} For $f(x) = a^{\top}x$ with $a = [1, 2]^{\top}$, the minimum over
$\mathcal{X}$ is obtained by minimising each coordinate independently:
\begin{equation}
\label{eq:lp}
x^{\star} = \arg\min_{x \in \mathcal{X}} (x_1 + 2x_2)
           = \arg\min (x_1) + 2 \arg\min(x_2)
           = (0.5,\; -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates the \textbf{fundamental theorem of linear programming}: the minimum (and
maximum) of a linear objective over a bounded polytope is always attained at a vertex.  The
feasible set $\mathcal{X}$ is a box with four vertices; since both components of $a$ are
positive, the minimum is at the lower-left vertex $(0.5, -5)$.

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial feasible point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^{\top} z$
      \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
      \Comment{Convex combination: $x_{k+1} \in \mathcal{X}$ since $\mathcal{X}$ is convex}
\EndFor
\end{algorithmic}
\end{algorithm}

\paragraph{Key properties.}
\begin{itemize}[nosep]
\item \textbf{Projection-free:} Frank--Wolfe replaces the projection step $\Pi_{\mathcal{X}}$
  of PGD with a \emph{linear minimisation oracle} (LMO)---solving $\min_{z \in \mathcal{X}}
  \nabla f(x_k)^{\top}z$.  For box constraints, this decomposes per coordinate:
  \begin{equation}
  \label{eq:fw_lmo}
  (z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
  \end{equation}
  requiring only $O(d)$ operations---far cheaper than projection for complex polytopes (e.g.\ spectahedra, flow polytopes).

\item \textbf{Feasibility:} Since $x_0 \in \mathcal{X}$, $z_k \in \mathcal{X}$, and $\mathcal{X}$
  is convex, the convex combination $x_{k+1} = \beta x_k + (1-\beta) z_k \in \mathcal{X}$ for
  all $k$.  Iterates are always feasible without any explicit enforcement.

\item \textbf{Convergence rate:} For $L$-smooth convex $f$, Frank--Wolfe with step size
  $\gamma_k = 2/(k+2)$ converges as
  \begin{equation}
  \label{eq:fw_rate}
  f(x_k) - f^{\star} \leq \frac{2LC^2}{k+2} = O\!\left(\frac{1}{k}\right),
  \end{equation}
  where $C = \mathrm{diam}(\mathcal{X})$---slower than the $O(1/k^2)$ rate of projected
  Nesterov acceleration.  This $O(1/k)$ rate is \emph{tight}: Lan (2013) showed a
  matching lower bound, proving that no variant of Frank--Wolfe using only LMO calls can
  achieve a better dependence on $k$ in the worst case.  This fundamental gap between
  Frank--Wolfe ($O(1/k)$) and projected gradient ($O(1/k^2)$ with Nesterov) cannot be
  closed without additional oracle access---it is the price of using LMO instead of projection.

\item \textbf{Convergence certificate (FW gap):}
  At each iteration, the quantity $g_k = \nabla f(x_k)^{\top}(x_k - z_k) \geq 0$ serves as
  an upper bound on $f(x_k) - f^{\star}$.  When $g_k = 0$, $x_k$ is optimal.
\end{itemize}

\subsection{Interior Optimum: $f(x) = (x_1 - 1)^2 + (x_2 - 5)^2$}

The unconstrained minimum $(1, 5)$ lies inside $\mathcal{X}$ (since $0.5 \leq 1 \leq 5$ and
$-5 \leq 5 \leq 10$).  Two mixing parameters are tested: $\beta = 0.90$ (10\% step towards LP
vertex per iteration) and $\beta = 0.985$ (1.5\% step).

When the optimum is at an \emph{interior} point, the FW iteration has a distinctive behaviour:
near $x^{\star} = (1, 5)$, the gradient $\nabla f(x_k) = 2(x_k - (1,5))$ is small and rotates
as $x_k$ orbits $x^{\star}$.  The LMO selects
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{otherwise} \end{cases},
\end{equation}
so $z_k$ alternates between box vertices as the gradient sign changes (each time $x_k$ crosses
the line $x_i = x_i^{\star}$ for some coordinate $i$).  The resulting trajectory is a convex
combination average of alternating vertices, which converges to $x^{\star}$ only because the
weight $(1-\beta)$ applied to each new vertex is small.  The FW gap at an interior optimum
satisfies $g_k = \nabla f(x_k)^{\top}(x_k - z_k) \to 0$ as $\nabla f(x_k) \to 0$---FW
``naturally'' certifies the interior optimum once the gradient is negligible.

\subsection{Frank--Wolfe Gap as Convergence Certificate}

\begin{proposition}[FW gap as certificate]
\label{prop:fw_gap}
For any convex $f$ and any $x_k \in \mathcal{X}$, let $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^{\top}z$.
Then the \emph{Frank--Wolfe gap}
\begin{equation}
\label{eq:fw_gap}
g_k = \nabla f(x_k)^{\top}(x_k - z_k) \geq 0
\end{equation}
satisfies $f(x_k) - f^{\star} \leq g_k$.
\end{proposition}
\begin{proof}
By convexity, $f(x^{\star}) \geq f(x_k) + \nabla f(x_k)^{\top}(x^{\star} - x_k)$, so
$f(x_k) - f^{\star} \leq \nabla f(x_k)^{\top}(x_k - x^{\star}) \leq \max_{z \in \mathcal{X}} \nabla f(x_k)^{\top}(x_k - z) = g_k$,
where the last inequality uses $x^{\star} \in \mathcal{X}$.
\end{proof}
The gap $g_k$ can be computed without knowing $f^{\star}$---only the current gradient and LMO are needed.
When $g_k \leq \varepsilon$, we have a certified $\varepsilon$-optimal solution.

\begin{theorem}[Frank--Wolfe convergence rate]
\label{thm:fw_convergence}
Let $f : \mathbb{R}^d \to \mathbb{R}$ be convex and $L$-smooth on the compact convex set
$\mathcal{X}$ with diameter $C = \max_{x,y \in \mathcal{X}} \|x - y\|$.  Frank--Wolfe with
step sizes $\gamma_k = 2/(k+2)$ satisfies
\begin{equation}
\label{eq:fw_theorem_rate}
f(x_k) - f^{\star} \leq \frac{2LC^2}{k+2}.
\end{equation}
\end{theorem}
\begin{proof}[Proof sketch]
By $L$-smoothness, $f(x_{k+1}) \leq f(x_k) + \nabla f(x_k)^{\top}(x_{k+1}-x_k)
+ \frac{L}{2}\|x_{k+1}-x_k\|^2$.
Substituting $x_{k+1} = (1-\gamma_k)x_k + \gamma_k z_k$ and $\|x_{k+1}-x_k\| = \gamma_k\|z_k-x_k\| \leq \gamma_k C$:
\begin{equation}
f(x_{k+1}) \leq f(x_k) + \gamma_k \nabla f(x_k)^{\top}(z_k - x_k) + \tfrac{L\gamma_k^2 C^2}{2}.
\end{equation}
Since $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^{\top}z$, we have
$\nabla f(x_k)^{\top}z_k \leq \nabla f(x_k)^{\top}x^{\star}$, so by convexity:
\begin{equation}
\nabla f(x_k)^{\top}(z_k - x_k) \leq \nabla f(x_k)^{\top}(x^{\star} - x_k) \leq f(x^{\star}) - f(x_k) = -(f(x_k) - f^{\star}).
\end{equation}
Letting $h_k = f(x_k) - f^{\star}$, we obtain the recurrence
$h_{k+1} \leq (1-\gamma_k)h_k + \frac{L\gamma_k^2 C^2}{2}$.
An inductive argument with $h_0 \leq LC^2/2$ and $\gamma_k = 2/(k+2)$ verifies
$h_k \leq 2LC^2/(k+2)$ for all $k \geq 0$.
\end{proof}

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe converges and the LP vertex $z_k$ alternates between
extreme points of $\mathcal{X}$.  The convex combination progressively averages these extreme
points, producing a zig-zag convergence towards the interior optimum.  The general $O(1/k)$
rate predicts, for example, that reducing the gap from 0.1 to 0.01 requires roughly 10 times
more iterations.

\textbf{Boundary optimum:} Frank--Wolfe is particularly well-suited to this case because the
LMO consistently selects the vertex closest to the constrained optimum direction ($z_1 = 0.5$
here).  The iterate slides naturally towards the boundary without requiring an explicit
projection.  However, the fixed $\beta$ schedule does not achieve the optimal convergence rate;
using the standard line-search or $\gamma_k = 2/(k+2)$ step schedule would improve the rate
to match the $O(1/k)$ theoretical bound more tightly.

\textbf{Standard vs fixed-$\beta$ step schedule.}
The implementation uses a fixed convex combination parameter $\beta$, which is equivalent to
a fixed step size $\gamma = 1 - \beta$ towards the LP vertex.  The standard Frank--Wolfe
schedule uses $\gamma_k = 2/(k+2)$, which gives the tight $O(1/k)$ bound~\eqref{eq:fw_rate}.
A fixed $\gamma$ does not decay to zero, which prevents convergence to the exact optimum:
the iterates converge in expectation only to a neighbourhood of size
$O\!\left(\gamma L C^2\right)$ around the optimum, where $C = \mathrm{diam}(\mathcal{X})$
and $L$ is the smoothness constant.  For $\gamma = 1 - \beta = 0.015$ and $C \approx 15.5$
(box diameter), $L = 2$ (for $f(x) = (x_1-1)^2 + (x_2-5)^2$), this gives a residual
gap of $O(0.015 \times 2 \times 240) \approx 7.2$---much larger than the observed
$f(x_{\text{final}}) - f^{\star} = 0.0046$.  The fixed $\beta$ still performs well here
because the minimum lies at an interior point where the gradient direction stabilises,
but the theoretical guarantee is weaker than for the decaying schedule.
This explains the residual gap observed in Table~\ref{tab:q6_final}.

\textbf{Frank--Wolfe vs Projected GD:}
\begin{itemize}[nosep]
\item PGD requires an efficient projection oracle but achieves $O(1/k)$ rate with standard GD
  and $O(1/k^2)$ with Nesterov acceleration.  For box constraints,
  $\Pi_{\mathcal{X}}$ costs $O(d)$ per coordinate, the same as the LMO.
\item Frank--Wolfe requires only a linear minimisation oracle---typically much cheaper for
  structured polytopes (e.g., computing the minimum eigenvalue for spectahedra is
  $O(d^3)$ but projection is $O(d^{3.5})$).
\item Frank--Wolfe is limited to $O(1/k)$ convergence even with optimal step sizes.
  Projected Nesterov achieves $O(1/k^2)$---a strictly faster asymptotic rate.
\item For large-scale problems with structured constraint sets (nuclear norm ball, flow
  polytopes, permutation matrices), Frank--Wolfe is the method of choice because
  the LP subproblem admits efficient closed-form solutions unavailable for projection.
\end{itemize}

\begin{remark}[Away-steps Frank--Wolfe and linear convergence on polytopes]
\label{rem:away_steps_fw}
Standard Frank--Wolfe cannot achieve better than $O(1/k)$ convergence even for strongly
convex objectives on polytopes.  The \emph{away-steps} variant (Wolfe, 1970; restated
rigorously by Lacoste-Julien \& Jaggi, 2015) overcomes this by considering, at each
iteration, both the standard FW step (towards a new atom $z_k$) and a step \emph{away}
from the current worst active vertex $v_k$:
\begin{equation}
\label{eq:away_step}
v_k = \arg\max_{s \in \mathcal{A}_k} \nabla f(x_k)^{\top} s,
\qquad x_{k+1} = x_k - \gamma_k \underbrace{(x_k - v_k)}_{\text{away direction}},
\end{equation}
where $\mathcal{A}_k$ is the current active set (the vertices with positive weight in the
current iterate's representation as a convex combination).  By moving \emph{away} from
atoms that increase the objective, the active set contracts, and the algorithm can achieve
\emph{linear convergence} $O(\rho^k)$ with $\rho < 1$ for $\mu$-strongly convex objectives
on polytopes.  For the box constraint in this question, the active set has at most $2d$
vertices (one per coordinate bound), making the bookkeeping $O(d)$ per iteration---the
same cost as the standard LMO.  The price is a more complex step-selection rule and the
need to track the active set.
\end{remark}

\subsection{Key Findings}

\paragraph{1. The role of second-order information.}
Newton's method is transformative on well-conditioned problems (1 iteration on Benchmark A)
and substantially superior on ill-conditioned problems (0.69 vs 3.73 on Rosenbrock in fewer
iterations).  The quadratic convergence rate~\eqref{eq:newton_quadratic} is fundamentally
superior to any first-order rate when close to the optimum.  The cost---$O(d^3)$ per step---is
prohibitive for $d \gg 100$, motivating quasi-Newton methods (L-BFGS) that approximate $H^{-1}$
using only gradient history.

\paragraph{2. Condition number as a fundamental obstacle.}
All first-order methods are adversely affected by high condition number $\kappa = L/\mu$.
On Benchmark C ($\kappa \approx 2508$), the GD convergence rate per iteration is
$\rho = ((\kappa-1)/(\kappa+1))^2 \approx (1 - 2/\kappa)^2 \approx 1 - 4/\kappa \approx 0.9984$.
To reduce the objective gap from $f(x_0) - f^{\star} = 4$ (since $f(-1,1) = 4$)
to $\varepsilon = 0.1$, GD requires
$k \geq \log(0.1/4)/\log(0.9984) \approx 2313$ iterations.
In just 120 iterations, GD achieves only a factor $0.9984^{120} \approx 0.826$ reduction---barely
17\% progress.  Nesterov's $O(\sqrt{\kappa})$ requirement reduces this to approximately
$\sqrt{2508} \approx 50$ iterations for the same reduction, consistent with the
factor-7 advantage observed on Benchmark C.  Adaptive methods (Adagrad, RMSprop, Adam)
provide coordinate-wise preconditioning that partially addresses this.
Newton's method is immune to $\kappa$ via affine invariance (Proposition~\ref{prop:newton_quadratic}).

\paragraph{3. Polyak's step: elegance at a cost.}
The Polyak rule is provably optimal when $f^{\star}$ is known, requiring no step-size tuning
and achieving $O(1/k)$ convergence.  However, its catastrophic failure when $f^{\star}$ is
misspecified (as demonstrated on Benchmarks A and B) severely limits its practical utility,
restricting it to settings where a tight lower bound on $f$ is available.

\paragraph{4. Derivative-free methods: the oracle hierarchy.}
There is a fundamental hierarchy of information:
\emph{full gradient} $\succ$ \emph{finite differences} $\approx$ \emph{Nelder--Mead}
$\succ$ \emph{random search} $\succ$ \emph{grid search}.
Each step down the hierarchy reduces per-step information, increasing the number of
iterations needed.  For high-dimensional problems, random search's dimension-free cost
(2 evaluations per step) makes it the only scalable option when derivatives are unavailable.

\paragraph{5. Constraint handling: hard vs soft enforcement.}
Projected GD enforces feasibility as a hard constraint from the very first iteration,
making it the preferred method when feasibility is non-negotiable.  Penalty methods
are simpler to implement but require careful scheduling of $\lambda$---too small and
feasibility is violated; too large and the problem is ill-conditioned.  Frank--Wolfe
avoids both problems for linear-constrained problems by keeping iterates feasible
through convex combinations.

\paragraph{6. Practical recommendations.}
\begin{itemize}[nosep]
\item \emph{Smooth convex, no constraints, $d$ large:} Use Adam (robust, well-tuned defaults)
  or Nesterov acceleration (optimal theoretical rate $O(1/k^2)$).
\item \emph{Strongly convex quadratic, $d$ small ($\leq 1000$):} Newton's method for exact
  minimum in $O(1)$ iterations.  For $d$ large, use conjugate gradient or L-BFGS.
\item \emph{Constrained, simple projection (box, simplex):} Projected GD with Nesterov
  acceleration for $O(1/k^2)$ rate.
\item \emph{Constrained, structured polytope (nuclear norm ball, flow polytope):}
  Frank--Wolfe---the LMO often admits an efficient closed-form solution while projection
  does not.
\item \emph{No derivative access, $d \leq 10$:} Nelder--Mead (robust, widely used).
  $d \leq 100$: finite differences.  $d \gg 100$: random directional search.
\item \emph{Known $f^{\star}$ (e.g., classification with achievable zero loss):} Polyak
  step size---optimal $O(1/k)$ rate with no step-size tuning.
\end{itemize}

The hierarchy Hessian $\succ$ gradient $\succ$ stochastic gradient $\succ$ function value

\begin{remark}[Matching upper and lower bounds: the complete oracle complexity picture]
\label{rem:lower_bounds}
Each row in Table~\ref{tab:oracle_complexity} represents both an upper bound (achieved by
a specific algorithm) and, where indicated, a \emph{matching lower bound} showing no
algorithm in that oracle class can do better in the worst case:
\begin{enumerate}[nosep]
\item \textbf{First-order smooth convex}: Nesterov (1983) showed $\Omega(1/k^2)$ is tight
  for gradient oracles, matching the upper bound of his own accelerated method.
\item \textbf{First-order strongly convex}: The rate $O(\exp(-k/\sqrt{\kappa}))$ for
  Nesterov is matched by an $\Omega(\exp(-k/\sqrt{\kappa}))$ lower bound (Nesterov, 2004).
\item \textbf{Zeroth-order (function-value oracle)}: Nemirovski \& Yudin (1983) proved
  $\Omega((L/\varepsilon)^{d/2})$ is tight, matching the grid search upper bound.
\item \textbf{Stochastic first-order}: Agarwal et al.\ (2012) established $\Omega(1/\sqrt{T})$
  for non-strongly-convex objectives, matched by SGD with decaying learning rate.
\end{enumerate}
These matching bounds mean the algorithms in this report are not merely good heuristics---they
are provably optimal (up to constants) within their oracle class.  The only way to improve
beyond these limits is to change the oracle: adding a projection oracle enables Frank--Wolfe's
feasibility guarantees; adding a Hessian oracle enables Newton's quadratic convergence.  This
motivates the practical question: ``what is the cheapest oracle that achieves the required
accuracy within the computation budget?''
\end{remark}

\subsection{Limitations and Future Work}

Several extensions and limitations merit acknowledgement:
\begin{enumerate}[nosep]
\item \textbf{Two-dimensional benchmarks.} All experiments use $d = 2$, which allows
  trajectory visualisation but does not reveal high-dimensional scaling behaviour.  In
  practice, the relative advantages of Newton's method (cost $O(d^3)$ per step) versus
  Nesterov (cost $O(d)$) reverse as $d$ grows beyond $\sim$100.  L-BFGS bridges this
  gap, achieving near-Newton convergence at $O(dm)$ cost per step.
\item \textbf{Hyperparameter sensitivity.} Results depend on the specific hyperparameter
  settings in Table~\ref{tab:q1_params} etc.  A full comparison should include automated
  hyperparameter tuning (grid search, Bayesian optimisation, or backtracking line search).
  The Polyak step size is particularly notable for being \emph{hyperparameter-free} given $f^{\star}$.
\item \textbf{Non-convex extensions.} All methods are analysed under convexity assumptions
  where applicable.  For non-convex problems (typical in deep learning), convergence
  guarantees weaken to stationary point conditions ($\|\nabla f(x_k)\| \leq \varepsilon$
  rather than $f(x_k) - f^{\star} \leq \varepsilon$) and momentum methods like Heavy Ball
  may diverge, as noted in Section~\ref{sec:q1}.
\item \textbf{Stochastic setting.} Mini-batch SGD was only analysed on Benchmark A.  Full
  analysis across all benchmarks with various batch schedules would provide a more complete
  picture of stochasticity's effect, particularly for Benchmark C where the non-convexity
  of the loss landscape makes local minima a concern.
\item \textbf{Constraint geometry.} The box constraint in Questions 5--6 admits a
  particularly simple projection and LMO.  For general polytopes (e.g., probability
  simplices, permutation matrices), the relative cost of projection vs.\ LMO determines
  whether projected GD or Frank--Wolfe is preferred.
\item \textbf{Line search integration.} All experiments use fixed or scheduled step sizes.
  Backtracking Armijo or Wolfe line search would improve step-size robustness without
  requiring knowledge of $L$---particularly beneficial for Newton (where the unit step
  may overshoot on Benchmark C) and Heavy Ball (where the step/momentum pair $(\alpha, \beta)$
  governs stability together).  The Barzilai-Borwein step rule provides a computationally
  cheap alternative that uses gradient information to estimate $L_k$ locally.
\end{enumerate}

\subsection{Connections to Machine Learning Practice}

The algorithms studied in this assignment directly underpin modern machine learning:
\begin{itemize}[nosep]
\item \textbf{Adam/RMSprop} are the default optimisers for neural network training (PyTorch,
  TensorFlow, JAX), where $d \sim 10^6$--$10^{10}$ and the loss landscape is highly non-convex.
  Their adaptive scaling compensates for heterogeneous layer-wise curvature.
\item \textbf{L-BFGS} (limited-memory BFGS) is the default for logistic regression and small
  neural networks in \texttt{scipy.optimize} and \texttt{sklearn}---it achieves near-Newton
  convergence at $O(dm)$ cost, with $m = 10$--$20$ history pairs.
\item \textbf{Frank--Wolfe} finds application in \emph{matrix completion} (collaborative
  filtering) where the nuclear norm ball $\{X : \|X\|_* \leq \tau\}$ serves as a convex
  relaxation of the rank constraint.  The FW linear subproblem on the nuclear norm ball
  is solved by a rank-1 SVD---$O(dm)$ for $d \times m$ matrices---making FW the method of
  choice when $d, m \sim 10^4$.
\item \textbf{Projected Gradient Descent} is used in Support Vector Machines (dual
  formulation: project onto the simplex $\{\alpha : \sum_i \alpha_i = 1, 0 \leq \alpha_i \leq C\}$)
  and in Lasso regularisation (project onto the $\ell_1$ ball).
\item \textbf{Mini-batch SGD with Nesterov momentum} (``SGD + Nesterov'') is one of the
  best-performing methods for image classification (ResNet training on ImageNet), combining
  stochastic mini-batch updates with Nesterov's lookahead for acceleration.
\end{itemize}
The theoretical framework established in this report---oracle complexity, condition number
analysis, convergence rate hierarchies---forms the analytical foundation for understanding
and improving these practical algorithms.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}, label=lst:polyak]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}, label=lst:adagrad]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}, label=lst:rmsprop]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}, label=lst:hb]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}, label=lst:nesterov]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)  # momentum schedule
        lookahead = x + beta_k * z                  # extrapolate
        g = grad_fn(lookahead)                       # gradient at lookahead
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}, label=lst:adam]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g       # biased first moment
        v_vec = beta2 * v_vec + (1 - beta2) * g**2    # biased second moment
        m_hat = m_vec / (1 - beta1**t)                # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method}, label=lst:newton]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))          # regularise Hessian
        try:
            p = np.linalg.solve(H_reg, g)             # Newton direction
        except np.linalg.LinAlgError:
            p = g                                      # fall back to GD
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # Linear minimisation oracle for box constraints
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z               # convex combination
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
