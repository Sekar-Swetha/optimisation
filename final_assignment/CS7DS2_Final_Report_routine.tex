\begin{abstract}
This report presents a comprehensive comparative study of first-order, second-order, and derivative-free optimisation algorithms applied to three benchmark problems --- a linear regression quadratic loss (Benchmark~A, condition number $\kappa \approx 1.12$), a toy neural network loss with sinusoidal perturbation (Benchmark~B, $\kappa \approx 3$--$10$), and the highly ill-conditioned Rosenbrock function (Benchmark~C, $\kappa \approx 2508$ at the optimum). Six categories of methods are investigated: (1)~\emph{Adaptive step-size methods} (Polyak, Adagrad, RMSprop, Heavy Ball): the Polyak step produces massive transient oscillations (peak $f_{\max}=148$, 432 on Benchmarks~A, B) when $f^\star$ is misspecified, but achieves the best Rosenbrock convergence among Q1 methods ($f=0.024$ at 120 iterations) when $f^\star=0$ is exact; (2)~\emph{Momentum and stochastic methods} (Nesterov, Adam, mini-batch SGD): Nesterov achieves $f=0.479$ on Rosenbrock vs GD's $f=3.388$, consistent with its theoretical $O(1/k^2)$ rate (Theorem~1) vs GD's $O(1/k)$; (3)~\emph{Newton's method}: converges in one step on Benchmark~A (exact quadratic) and reaches $f=0.686$ on Rosenbrock in 20 iterations vs GD's $f=3.732$ in 80, leveraging quadratic convergence (Theorem~2); (4)~\emph{Derivative-free methods}: forward FD with $\delta=0.05$ achieves gradient error $\approx 0.25$ and closely tracks exact GD, while $\delta=0.8$ gives error $\approx 4.1$ and degrades convergence; (5)~\emph{Constrained optimisation}: projected GD achieves zero constraint violation from iteration~1, while penalty methods require $\lambda \geq 1.8$ for near-exact feasibility; (6)~\emph{Frank--Wolfe} (Conditional Gradient): converges projection-free to the LP optimum $(0.5,-5)$ with $O(1/k)$ rate (Theorem~3). All experimental results are fully reproducible from a fixed random seed (\texttt{np.random.seed(42)}).
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmark functions represent progressively more challenging optimisation landscapes.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a convex quadratic with Hessian $H = \frac{1}{m}X^T X$ (positive definite and constant). The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the irreducible minimum loss is $J^{\star} \approx 0.4826$, not zero. Starting point: $\theta_0 = [0, 0]^T$. With seed 42 and $m=1000$, the Hessian has eigenvalues $\lambda_{\min} \approx 0.925$ and $\lambda_{\max} \approx 1.033$, giving $L \approx 1.033$, $\mu \approx 0.925$, and condition number $\kappa = L/\mu \approx 1.12$ --- a remarkably well-conditioned problem. This explains why gradient descent with $\alpha = 0.08 \ll 1/L \approx 0.97$ converges extremely fast on Benchmark~A despite using a conservative step size.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \text{diag}(2 - \sin(x_1),\ 10)$. Since $|\sin(x_1)| \leq 1$, the diagonal elements of $H$ lie in $[1, 3]$ and $[10, 10]$, giving a curvature ratio between $\sim$3.3:1 and $\sim$10:1. The function is non-quadratic (due to $\sin$) but well-conditioned. The minimum satisfies $10(x_2^{\star}-2) = 0 \Rightarrow x_2^{\star} = 2$ and $2(x_1^{\star}-1)+\cos(x_1^{\star}) = 0$, solved numerically to give $x_1^{\star} \approx 0.5828$, so $x^{\star} \approx (0.583, 2.0)$ with $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is
\begin{equation}
H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $\lambda_{\min} \approx 0.40$ and $\lambda_{\max} \approx 1001.6$, giving a condition number $\kappa \approx 2508$. Far from the optimum the condition number is even larger due to the rapidly-varying curvature structure. This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions for first-order methods. Starting point: $x_0 = [-1, 1]^T$.

For constrained problems: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$, a box-shaped polytope with four vertices.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}

\subsection{Theoretical Background}

\begin{definition}[$L$-Smoothness and $\mu$-Strong Convexity]
A differentiable function $f : \mathbb{R}^d \to \mathbb{R}$ is \emph{$L$-smooth} if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x-y\|$ for all $x, y \in \mathbb{R}^d$, equivalently $\nabla^2 f(x) \preceq LI$. It is \emph{$\mu$-strongly convex} if $\nabla^2 f(x) \succeq \mu I$, equivalently $f(y) \geq f(x) + \nabla f(x)^T(y-x) + \frac{\mu}{2}\|y-x\|^2$. The condition number is $\kappa = L/\mu \geq 1$.
\end{definition}

For a differentiable function $f$ that is $L$-smooth, gradient descent with constant step size $\alpha \leq 1/L$ satisfies the sufficient decrease condition
\begin{equation}
f(x_{k+1}) \leq f(x_k) - \frac{\alpha}{2}\|\nabla f(x_k)\|^2.
\end{equation}
If additionally $f$ is $\mu$-strongly convex, the convergence rate is geometric:
\begin{equation}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k (f(x_0) - f^{\star}) = \left(1 - \frac{1}{\kappa}\right)^k (f(x_0) - f^{\star}).
\label{eq:gd_rate}
\end{equation}
The factor $(1 - 1/\kappa)$ reveals that high condition number $\kappa = L/\mu$ causes slow convergence --- the motivation for all adaptive methods in this section. To reach $\epsilon$ relative accuracy, GD requires $O(\kappa \log(1/\epsilon))$ iterations, while Nesterov's accelerated method reduces this to $O(\sqrt{\kappa} \log(1/\epsilon))$. For $\kappa = 2508$ (Benchmark~C) and $\epsilon = 0.01$, GD needs $\approx 11{,}547$ iterations vs Nesterov's $\approx 231$ --- a $50\times$ speedup. For $\kappa = 1.12$ (Benchmark~A), both methods converge in $\sim$2--5 iterations, so the advantage is irrelevant.

The Lipschitz gradient constants for each benchmark are:
\begin{itemize}[nosep]
\item \textbf{Benchmark~A:} $H = X^TX/m$ is constant, so $L = \lambda_{\max}(H) \approx 1.033$ and $\mu = \lambda_{\min}(H) \approx 0.925$. Maximum safe step: $\alpha \leq 1/L \approx 0.968$. The step $\alpha = 0.08$ is highly conservative ($\approx 8\%$ of maximum), ensuring rapid convergence with large safety margin.
\item \textbf{Benchmark~B:} $\nabla^2 f = \text{diag}(2-\sin(x_1), 10)$. Since $|{-}\sin(x_1)| \leq 1$, we have $L = \max(3, 10) = 10$ and $\mu = \min(1, 10) = 1$, giving $\kappa = 10$.
\item \textbf{Benchmark~C (Rosenbrock):} The Hessian varies dramatically. Near the starting point $(-1,1)$: $H = \begin{bmatrix}1202 & 400 \\ 400 & 200\end{bmatrix}$, with $\lambda_{\max} \approx 1400$. The Lipschitz constant is correspondingly large, requiring $\alpha \leq 1/1400 \approx 0.0007$ for guaranteed GD stability near $x_0$. The used step $\alpha = 0.0012$ is slightly aggressive but works in practice.
\end{itemize}

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Step inversely proportional to gradient magnitude}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step is derived by minimising the first-order upper bound $f(x_k) + \nabla f(x_k)^T(x - x_k) + \frac{1}{2\alpha}\|x - x_k\|^2$ subject to $f(x) \leq f^\star$, yielding the exact expression above. When $f^\star$ is known correctly and $f$ is convex, the Polyak step guarantees convergence with rate $\|x_k - x^\star\|^2 \leq \frac{C}{k}$ for subgradient methods. The $\varepsilon > 0$ regularisation prevents division by zero near critical points.

\textbf{Why misspecification causes divergence:} As the iterate approaches $x_{\mathrm{OLS}}$, $\|\nabla J(\theta_k)\| \to 0$ (gradient vanishes at the true minimum). However, with $f^\star_{\rm spec} = 0$, the numerator $J(\theta_k) - f^\star_{\rm spec} \to J(x_{\mathrm{OLS}}) \approx 0.483 \neq 0$, so $\alpha_k = (J(x_k) - 0)/(\|\nabla J\|^2 + \varepsilon) \to 0.483/\varepsilon \to \infty$ as $\|\nabla J\| \to 0$. This unbounded step propels the iterate far from $x_{\mathrm{OLS}}$, where the gradient is large again, causing oscillation. The transient peak $f_{\max} = 148$ corresponds to the iterate being launched to a region of high loss before the gradient magnitude stabilises the step size.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Accumulated squared gradients, per coordinate}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$ \Comment{Coordinate-wise step sizes}
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's effective per-coordinate step size $\alpha_0 / \sqrt{G_k + \varepsilon}$ is large for infrequently-updated coordinates and decays monotonically over time. The cumulative accumulation $G_k \to \infty$ means the step eventually vanishes, which can cause premature stopping on non-stationary problems. Formally, Adagrad achieves the \emph{adaptive regret bound}:
\begin{equation}
\sum_{k=1}^K f(x_k) - f(x^\star) \leq \frac{\alpha_0}{2} \sum_{i=1}^d \sqrt{\sum_{k=1}^K g_{k,i}^2} + O(1),
\end{equation}
which is tighter than the $O(\sqrt{K})$ regret of standard online gradient descent when gradients are sparse (few coordinates have large $g_{k,i}$). For Benchmark~A (dense, isotropic gradients), Adagrad provides no theoretical advantage over GD but is robust to the need for tuning $\alpha$.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's cumulative sum with an exponential moving average (decay $\beta = 0.9$), allowing the effective step to adapt to \emph{recent} rather than \emph{all} gradient history. This prevents the step from decaying to zero and is better suited to non-stationary settings. Specifically, while Adagrad's denominator grows as $\sqrt{G_k} \sim O(\sqrt{k})$ for constant gradient magnitude, RMSprop's exponential average converges to $\sqrt{v_\infty} = \sqrt{(1-\beta)\sum_{j=0}^\infty \beta^j g_{k-j}^2} \approx \|g\|$ in steady state, maintaining a finite effective step size $\alpha_0 / \|g\| = $ constant.

\begin{proposition}[RMSprop Steady-State Step]
For a stationary gradient distribution with $\mathbb{E}[g_k^2] = \sigma^2 I$ (isotropic), the RMSprop effective step converges to $\alpha_0 / \sigma$ as $k \to \infty$, independent of the number of iterations. This contrasts with Adagrad's decaying step $\alpha_0 / (\sqrt{k}\sigma)$.
\end{proposition}

This explains why RMSprop maintains adaptability throughout training --- at the cost of losing the theoretical regret guarantee of Adagrad. In the experiments, RMSprop reaches slightly suboptimal values on Benchmark~A ($f = 0.506 > J^\star = 0.483$) because the fixed effective step oscillates near the minimum rather than decaying to zero.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Exponentially-weighted gradient sum}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method accumulates momentum by maintaining a velocity vector $z_k$. For quadratic functions with $L$-smooth and $\mu$-strongly convex objective, the optimal parameters are:
\begin{equation}
\alpha^{\star} = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2} = \frac{4}{L(\sqrt{\kappa}+1)^2}, \qquad \beta^{\star} = \left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^2 = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2,
\end{equation}
yielding convergence rate $\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^k$ --- substantially faster than GD's $(1-1/\kappa)^k$. For Benchmark~A ($\kappa=1.12$): $\alpha^\star \approx 0.965$, $\beta^\star \approx 0.0033$ (the near-unit condition number makes momentum nearly useless). For Benchmark~B ($\kappa=10$): $\alpha^\star \approx 0.165$, $\beta^\star \approx 0.52$. The used parameters ($\alpha=0.045$, $\beta=0.88$ for Bench~A) differ from the theoretical optima; these are tuned empirically for stability given the non-quadratic nature of the objectives.

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest on Rosenbrock ($f = 0.024$ vs next-best Heavy Ball $f = 1.40$). On Benchmarks~A and B, Adagrad and Heavy Ball match GD at the noise floor ($J^\star \approx 0.4826$ for A, $f^\star \approx 0.7244$ for B).
\item \textbf{Polyak oscillation mechanism:} With misspecified $f^\star = 0$, as $\|\nabla J(\theta_k)\| \to 0$ near $\theta_{\mathrm{OLS}}$, the Polyak step $\alpha_k = (J(\theta_k) - 0)/(\|\nabla J\|^2 + \varepsilon) \to \infty$. The iterate is then propelled to a high-loss region ($f_{\max} = 148$ on Bench~A, $432$ on Bench~B). From that high-loss region the gradient magnitude is large, step size collapses, and the iterate partially damps. This is an \emph{oscillatory but not monotone} divergence pattern. With correct $f^\star = 0$ on Rosenbrock, $J(x_k) \to 0$ simultaneously with $\|\nabla f\| \to 0$, so $\alpha_k$ remains bounded.
\item \textbf{Step size behaviour:} Adagrad's monotone decay guarantees stability but risks premature stopping when $G_k$ becomes large early (e.g., large initial gradients on Benchmark~B from $x_0=[-1,4]^T$ far from $x^\star$). RMSprop's exponential forgetting ($\beta=0.9$, effective window $\approx 10$ iterations) maintains adaptability to \emph{recent} curvature. Heavy Ball's fixed parameters require tuning but achieve near-optimal quadratic rate $(\sqrt{\kappa}-1)^k/(\sqrt{\kappa}+1)^k$ for the right $\alpha,\beta$.
\item \textbf{Practical recommendation:} For problems where $f^{\star}$ is unknown (the common case), RMSprop or Heavy Ball are preferred. Polyak is suitable only for problems with a known exact minimum, such as zero-loss interpolation in over-parameterised neural networks (where $f^\star = 0$ exactly).
\end{itemize}

\subsection{Theoretical Background}

The fundamental limitation of gradient descent is its $O(1/k)$ convergence rate for convex functions. Nesterov's 1983 result showed this can be improved to $O(1/k^2)$ by incorporating a carefully-tuned momentum schedule --- this is the \emph{optimal} convergence rate for first-order methods on smooth convex functions.

\begin{theorem}[Nesterov, 1983]
\label{thm:nesterov}
Let $f$ be convex and $L$-smooth. Nesterov's accelerated gradient method with $\alpha = 1/L$ and momentum schedule $\beta_k = (k-1)/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\end{equation}
\end{theorem}

This $O(1/k^2)$ rate matches the information-theoretic lower bound for first-order methods on smooth convex functions, making Nesterov's method theoretically optimal.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient Descent}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0} \in \mathbb{R}^d$ \Comment{Velocity vector, initialised to zero}
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum (capped at $\beta_{\max}$)}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$ \Comment{Position update (equivalent: $x_{k+1} = \tilde{x}_k - \alpha\nabla f(\tilde{x}_k)$)}
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This ``correction before overshooting'' allows Nesterov's method to maintain accelerated convergence while Heavy Ball can oscillate. The capping at $\beta_{\max}$ is a practical safeguard against excessive momentum in early iterations on non-convex problems.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{Biased first moment (mean) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Biased second moment (variance) estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$ \Comment{Normalised gradient step}
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines first-moment momentum (reducing variance in the gradient direction) with per-coordinate second-moment scaling (RMSprop-style adaptation). Bias correction via $(1-\beta_1^t)$ and $(1-\beta_2^t)$ prevents the first few iterates from being biased towards zero due to zero initialisation of $m_0, v_0$. Typical defaults $\beta_1 = 0.9$, $\beta_2 = 0.999$ are used in practice; here $\beta_1$ is tuned per benchmark.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y) \in \mathbb{R}^{m \times d} \times \mathbb{R}^m$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\sigma \leftarrow \text{permutation}(\{1,\ldots,m\})$
    \For{each mini-batch $\mathcal{B}$ of size $b$ from $\sigma$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Unbiased estimate of full gradient}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $g = \frac{1}{b}X_\mathcal{B}^T(X_\mathcal{B}\theta - y_\mathcal{B})$ is an \emph{unbiased} estimate of the full gradient $\nabla J(\theta)$, with variance $\sigma^2_{\rm grad} \propto 1/b$. Smaller batches yield more gradient steps per epoch (a batch size of $b=5$ gives $m/b = 200$ steps per epoch) but higher gradient variance.

\subsection{Comparative Analysis: Q2 Methods}

The three methods in Q2 represent different points on the theory-practice spectrum:

\textbf{Nesterov Accelerated Gradient:} Achieves the information-theoretic lower bound $O(1/k^2)$ for first-order methods on smooth convex functions (Theorem~\ref{thm:nesterov}). On Benchmark~A, the near-unit condition number means this advantage over GD is marginal. On Benchmark~C, Nesterov achieves a $7\times$ improvement over GD in 150 iterations, demonstrating the practical impact of acceleration on ill-conditioned problems. The momentum schedule $\beta_k = (k-1)/(k+2) \to 1$ increases progressively, building more aggressive momentum over time. The key theoretical insight is the \emph{lookahead gradient}: by computing $\nabla f(\tilde{x}_k)$ at the extrapolated point rather than $x_k$, Nesterov's method avoids the oscillations of Heavy Ball while still achieving acceleration.

\textbf{Adam:} Combines Nesterov-style first-moment momentum with RMSprop-style second-moment scaling. Its per-coordinate step $\alpha \cdot \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$ can be interpreted as applying a preconditioner that approximates the diagonal of the inverse Hessian. This makes Adam implicitly adaptive to curvature without computing the full Hessian. On Benchmark~C, Adam reaches $f=1.579$ vs Nesterov's $f=0.479$: Adam's preconditioning helps on the anisotropic curvature near the starting point but cannot fully overcome the highly nonlinear Rosenbrock valley structure, where consistent directional momentum (Nesterov) is more effective.

\textbf{Practical implication:} Nesterov is preferred for problems with known convexity and smooth geometry; Adam is preferred for deep learning where gradients are noisy, the objective is non-convex, and per-coordinate adaptivity handles diverse layer-wise learning rates. Mini-batch SGD with appropriate learning rate decay ($\alpha_t \propto 1/\sqrt{t}$) is the practical backbone of most deep learning training.

\subsection{Local Taylor Approximations of $g(x) = x^4$}

Consider the one-dimensional function $g(x) = x^4$ at the expansion point $x_0 = 0.25$. We compute $g(x_0) = (0.25)^4 = 0.0039$, $g'(x_0) = 4(0.25)^3 = 0.0625$, and $g''(x_0) = 12(0.25)^2 = 0.75$.

\textbf{First-order (tangent) approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625(x - 0.25).
\end{equation}
The minimum of $\hat{g}_1$ is unbounded (it is a linear function), so this approximation does not provide a useful step direction for minimisation.

\textbf{Second-order approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
The minimiser of $\hat{g}_2$ is $x^{\star}_{\rm approx} = x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 = 0.25 - 0.0833 = 0.1667$. This is the Newton step from $x_0 = 0.25$, moving towards the true minimum $x^{\star} = 0$ of $g(x) = x^4$.

\subsection{Newton's Method}

\begin{definition}[Newton Step]
The Newton step at $x_k$ is $p_k = [\nabla^2 f(x_k)]^{-1}\nabla f(x_k)$, obtained by minimising the second-order Taylor expansion $f(x_k) + \nabla f(x_k)^T(x-x_k) + \frac{1}{2}(x-x_k)^T\nabla^2 f(x_k)(x-x_k)$ over $x$.
\end{definition}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0,1]$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised Hessian (ensures positive definiteness)}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system, not explicit inversion)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{Damped Newton step ($\alpha = 1$ is full Newton)}
\EndFor
\end{algorithmic}
\end{algorithm}

Newton's method achieves \emph{quadratic} local convergence near the optimum (Theorem~\ref{thm:newton_convergence}) when the Hessian is invertible and $\alpha = 1$. The Tikhonov regularisation $\lambda I$ ensures positive definiteness when the Hessian is near-singular or indefinite (as can occur on Rosenbrock far from the optimum).

\begin{theorem}[Newton Quadratic Convergence]
\label{thm:newton_convergence}
Let $f$ be twice differentiable with $\mu$-strongly convex Hessian and $M$-Lipschitz second derivative. Then Newton's method with $\alpha = 1$ satisfies, once sufficiently close to $x^\star$:
\begin{equation}
\|x_{k+1} - x^\star\| \leq \frac{M}{2\mu}\|x_k - x^\star\|^2.
\end{equation}
\end{theorem}

This quadratic convergence --- where the error is squared at each step --- explains why Newton's method requires dramatically fewer iterations than gradient descent.

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$.

The damped Newton steps ($\alpha < 1$) on Benchmarks~B and C account for the fact that far from the optimum, the quadratic model is inaccurate and full Newton steps can overshoot. \end{equation}
with condition number $\approx 2508$. A full Newton step ($\alpha=1$) here would overshoot the banana valley; the damped step $\alpha=0.22$ trades convergence speed for global stability.

\subsubsection{Why Newton Converges in One Step on Benchmark~A}

For a quadratic loss $J(\theta) = \frac{1}{2m}\|X\theta - y\|^2$, the gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y) = H\theta - b$ where $H = X^TX/m$ and $b = X^Ty/m$. The Newton step from $\theta_0$ is:
\begin{equation}
\theta_1 = \theta_0 - H^{-1}\nabla J(\theta_0) = \theta_0 - H^{-1}(H\theta_0 - b) = H^{-1}b = \theta_{\mathrm{OLS}}.
\end{equation}
This is the exact OLS solution, achieved in a single step regardless of the starting point $\theta_0$. Newton's method is exact for quadratic objectives because the second-order Taylor expansion is exact (all higher-order derivatives vanish), and the Hessian is constant (independent of $\theta$). This confirms the entry in Table~\ref{tab:q3_final}: Newton final $x \approx (3.028, 4.017) \approx \theta_{\mathrm{OLS}}$ (up to noise, since $\theta_{\mathrm{OLS}} \neq \theta^\star = (3, 4)$ due to observation noise and the specific seed).

\subsubsection{Limitations and Quasi-Newton Extensions}

The $O(d^3)$ cost of solving the linear system $H_k p_k = g_k$ at each iteration is prohibitive for large $d$. Quasi-Newton methods approximate the Hessian (or its inverse) from gradient differences, avoiding explicit Hessian computation. The BFGS update:
\begin{equation}
B_{k+1} = B_k - \frac{B_k s_k s_k^T B_k}{s_k^T B_k s_k} + \frac{y_k y_k^T}{y_k^T s_k}, \quad s_k = x_{k+1} - x_k,\ y_k = \nabla f(x_{k+1}) - \nabla f(x_k),
\end{equation}
achieves superlinear convergence (between linear and quadratic) at $O(d^2)$ cost per iteration. L-BFGS further reduces this to $O(md)$ for $m \ll d$ history vectors, making quasi-Newton methods practical for large-scale problems. For the benchmark problems here ($d=2$), full Newton is trivial, but the extension to L-BFGS would be essential at $d = 10^4$ (typical neural network parameter count).

\subsection{Part A: Finite Difference Approximation and Nesterov Random Search (Benchmark~B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad e_i \text{ is the $i$-th standard basis vector.}
\end{equation}
For a twice-differentiable $f$, Taylor expansion gives the error bound:
\begin{equation}
\left|\frac{f(x+\delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| \leq \frac{\delta}{2}\max_{t \in [0,\delta]}\left|\frac{\partial^2 f}{\partial x_i^2}(x + te_i)\right| = O(\delta).
\end{equation}
Combined with floating-point roundoff error $O(\varepsilon_{\rm machine}/\delta)$ in the difference quotient, the total error is minimised at $\delta^{\star} \sim \sqrt{\varepsilon_{\rm machine}} \approx 10^{-8}$. The assignment tests $\delta = 0.05$ (good: truncation error $\approx 10^{-3}$) and $\delta = 0.8$ (poor: truncation error $\approx 10^{-1}$, comparable to the gradient magnitude).

The assignment tests $\delta = 0.05$ and $\delta = 0.8$. At the test point $x = [0, 3]^T$ (away from optimum), where $\nabla f = [-1.00, 10.00]^T$:
\begin{itemize}[nosep]
\item $\delta = 0.05$: forward FD gives $\hat{g}_1 \approx -0.950$, error $= 0.050$ (i.e.\ $O(\delta) = O(0.05)$, gradient direction accuracy $\approx 95\%$).
\item $\delta = 0.8$: forward FD gives $\hat{g}_1 \approx -0.303$, error $= 0.697$ (i.e.\ $O(\delta) = O(0.8)$, gradient direction accuracy only $\approx 30\%$).
\end{itemize}
This illustrates concretely why $\delta = 0.05$ (truncation error $\approx 5\%$) enables accurate gradient-following while $\delta = 0.8$ (error $\approx 70\%$) corrupts the descent direction.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random (Zeroth-Order) Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Uniform random direction on unit sphere}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Zeroth-order gradient estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates the directional derivative $\nabla f(x)^T u_k \approx [f(x+\delta u_k) - f(x)]/\delta$ along one random direction per step, requiring only \emph{two} function evaluations per iteration regardless of dimension. The estimate $\hat{g}_k = \frac{f(x_k+\delta u_k)-f(x_k)}{\delta}u_k$ is a noisy, rank-1 approximation to the gradient; in expectation over $u_k$:
\begin{equation}
\mathbb{E}_{u \sim \mathbb{S}^{d-1}}[\hat{g}_k] = \frac{1}{d}\nabla f_\delta(x_k) \approx \frac{1}{d}\nabla f(x_k),
\end{equation}
where $f_\delta(x) = \mathbb{E}_u[f(x+\delta u)]$ is the smoothed version of $f$. In $d=2$ dimensions, the factor $1/d = 0.5$ means each step uses approximately half the gradient information of an exact gradient step. To compensate, the learning rate $\alpha$ is effectively doubled relative to exact GD, and more iterations are needed. For high-dimensional problems ($d = 10^4$), zeroth-order methods can still converge but require $O(d)$ more iterations than first-order methods --- the fundamental curse of zeroth-order optimisation. Settings: $\alpha = 0.025$, $\delta = 0.08$, 220 iterations.

\subsection{Part B: Derivative-Free Methods on Benchmark~C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively deforms it through four operations: \textbf{reflection} ($x_r = \bar{x} + \alpha_r(\bar{x} - x_{\rm worst})$ with $\alpha_r = 1$), \textbf{expansion} ($x_e = \bar{x} + \gamma(\bar{x} - x_{\rm worst})$ with $\gamma = 2$ if $x_r$ improved), \textbf{contraction} ($x_c = \bar{x} + \rho(x_r - \bar{x})$ or $\bar{x} + \rho(x_{\rm worst} - \bar{x})$ with $\rho = 0.5$), and \textbf{shrink} (contract all towards best with $\sigma = 0.5$). Here $\bar{x}$ is the centroid of all vertices except the worst. No gradient information is required --- only function values at the simplex vertices.

Settings: initial simplex step $= 0.35$, $\alpha_r = 1.0$, $\gamma = 2.0$, $\rho = 0.5$, $\sigma = 0.5$, 160 iterations.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Setup}

Minimise Benchmark~B subject to the constraint $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$:
\begin{equation}
\min_{x \in \mathbb{R}^2} f(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1) \quad \text{subject to} \quad x_1 \geq 0.5.
\end{equation}
The feasible set is $\mathcal{X} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}$. The unconstrained minimum is at $x^\star \approx (0.583, 2.0)$, which satisfies $x_1^\star > 0.5$ and hence lies in the interior of $\mathcal{X}$. Therefore, the constrained and unconstrained optima coincide: the constraint is \emph{inactive} at the solution. This setting tests whether methods correctly navigate from an infeasible starting point to the interior optimum.

\subsection{KKT Conditions}

The Karush--Kuhn--Tucker (KKT) necessary conditions for this problem are:
\begin{align}
\nabla f(x^\star) + \mu \cdot (-e_1) &= 0 \quad \text{(stationarity)}, \\
x_1^\star &\geq 0.5 \quad \text{(primal feasibility)}, \\
\mu &\geq 0 \quad \text{(dual feasibility)}, \\
\mu(0.5 - x_1^\star) &= 0 \quad \text{(complementary slackness)}.
\end{align}
Since $x_1^\star \approx 0.583 > 0.5$, complementary slackness forces $\mu = 0$, and the KKT condition reduces to $\nabla f(x^\star) = 0$ --- the unconstrained optimality condition. The constraint is indeed inactive, confirming the constrained minimum equals the unconstrained one.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection oracle $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step (may leave $\mathcal{X}$)}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project back: $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)^T$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection onto $\mathcal{X} = \{x : x_1 \geq 0.5\}$ has the closed form $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)^T$, computable in $O(1)$. Projected GD with step $\alpha$ satisfies: if $f$ is $L$-smooth and $\mu$-strongly convex, the convergence rate is $(1 - \alpha\mu) \leq (1 - \mu/L)^k$, matching unconstrained GD on the feasible set.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda\, \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective; penalty active iff $x_1 < 0.5$}
    \State $\nabla F_{\lambda}(x_k) \gets \nabla f(x_k) - \lambda\, \mathbf{1}[x_{k,1} < 0.5]\, e_1$ \Comment{Subgradient of penalty}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method converts the constrained problem into an unconstrained one by adding a term $\lambda\max(0, 0.5 - x_1)$ that penalises infeasibility. As $\lambda \to \infty$, the penalty solution approaches the true constrained solution, but large $\lambda$ induces ill-conditioning (the penalised objective becomes very steep near the constraint boundary), requiring smaller step sizes for stability.

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ (step $\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\end{tabular}
\end{table}

\textbf{Projected GD} enforces feasibility strictly at every iteration. This is the fundamental advantage: feasibility is a \emph{hard} constraint maintained throughout. The cost is requiring a projection oracle, which for simple constraints (half-planes, boxes, balls) is $O(1)$--$O(d)$ but can be expensive for general convex sets (requiring solving a QP). For this problem, the projection $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)^T$ is $O(1)$ and introduces zero overhead over the gradient computation. The convergence guarantee for projected GD on strongly convex $f$ matches the unconstrained rate: $f(x_k) - f^\star \leq (1-\mu/L)^k (f(x_0)-f^\star)$, since the constraint is inactive at the solution and projection onto the half-plane is non-expansive.

\textbf{Penalty methods} trade exact feasibility for algorithmic simplicity: no projection oracle is needed, only an augmented gradient. The weight $\lambda$ controls the trade-off between constraint satisfaction and numerical conditioning. An important extension is the \emph{Augmented Lagrangian Method} (ALM), which combines the penalty term with a Lagrange multiplier estimate $\mu$:
\begin{equation}
F_{\rm ALM}(x; \mu, \lambda) = f(x) + \mu(0.5 - x_1) + \frac{\lambda}{2}\max(0, 0.5-x_1)^2.
\end{equation}
The dual variable $\mu$ is updated after each inner minimisation: $\mu^{+} = \max(0,\ \mu + \lambda(0.5 - x_1^{\star}))$. This update drives $\mu$ towards the true KKT multiplier, allowing exact constraint satisfaction at finite $\lambda$ (unlike the pure penalty method which requires $\lambda \to \infty$). The ALM avoids the ill-conditioning of pure penalty methods while retaining their projection-free character --- it is the method of choice for smooth constrained problems in practice.

\subsubsection{Penalty Parameter Sensitivity}

The choice of $\lambda$ in penalty methods reveals a fundamental tension:
\begin{itemize}[nosep]
\item \textbf{Small $\lambda$ (e.g., $\lambda = 0.15$):} The penalty gradient $-\lambda e_1 = -0.15 e_1$ is much smaller than the objective gradient (magnitude $\approx 2$--$5$ near $x_0$), so the penalty has little effect. The iterates remain infeasible for many iterations.
\item \textbf{Large $\lambda$ (e.g., $\lambda = 4.5$):} The penalty gradient $-4.5 e_1$ dominates near the constraint, rapidly driving $x_1 \to 0.5$. However, the penalised objective becomes steeply sloped near the boundary, requiring a smaller step size ($\alpha = 0.03$ vs $\alpha = 0.05$ for $\lambda \leq 1.8$) to avoid oscillation.
\item \textbf{Optimal $\lambda$:} Approximately $\lambda \approx \mu^{\star}_{\rm KKT}$ (the true KKT multiplier). Since the constraint is inactive here ($\mu^{\star} = 0$), any finite $\lambda$ eventually achieves feasibility --- the choice only affects convergence speed.
\end{itemize}

\subsection{Frank--Wolfe Algorithm}

The Frank--Wolfe (or Conditional Gradient) algorithm avoids the need for a projection oracle by replacing the projection step with a \emph{linear minimisation oracle} (LMO) over $\mathcal{X}$.

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (maintains $x_{k+1} \in \mathcal{X}$)}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{definition}[Frank--Wolfe Gap]
The \emph{Frank--Wolfe gap} at iteration $k$ is defined as
\begin{equation}
h_k = \nabla f(x_k)^T(x_k - z_k) \geq 0,
\end{equation}
which serves as a certificate of optimality: $h_k = 0$ if and only if $x_k$ is a KKT point of the constrained problem.
\end{definition}

For box constraints, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0, \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0, \end{cases}
\end{equation}
requiring only $O(d)$ operations per iteration. This is far cheaper than projection for many constraint sets (e.g., nuclear norm ball, simplex, spectrahedron).

\begin{theorem}[Frank--Wolfe Convergence Rate]
\label{thm:fw}
Let $f$ be convex and $L$-smooth. Frank--Wolfe with diminishing step $\gamma_k = 2/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{2LC^2_{\mathcal{X}}}{k+2},
\end{equation}
where $C_{\mathcal{X}} = \max_{x,y \in \mathcal{X}} \|x - y\|$ is the diameter of $\mathcal{X}$. The $O(1/k)$ rate is \emph{suboptimal} compared to projected GD's $O((1-\mu/L)^k)$ (linear for strongly convex $f$), but each Frank--Wolfe iteration may be far cheaper.
\end{theorem}

In our implementation, a fixed $\beta$ is used instead of the optimal schedule $\gamma_k = 2/(k+2) = 1 - \beta_k$. The fixed $\beta$ approximates the optimal schedule when $\beta$ is close to 1 (corresponding to small steps).

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits the characteristic zig-zag pattern. The LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ at each iteration because the gradient direction at $x_k$ determines which vertex is selected by the LMO. The convex combination gradually averages these oscillations. The fixed $\beta$ schedule achieves $O(1/k)$ convergence (consistent with Theorem~\ref{thm:fw}). Smaller $\beta$ (larger movement towards $z_k$) accelerates early progress but amplifies oscillation near the optimum; larger $\beta$ (smaller movement) reduces oscillation at the cost of slower convergence.

\textbf{Boundary optimum:} Frank--Wolfe is particularly well-suited to boundary optima because the LMO frequently selects vertices near the constrained optimum. The iterate naturally hugs the constraint boundary without explicit projection. For the $\ell_2$ objective $f(x) = x_1^2 + x_2^2$, the gradient $\nabla f(x_k) = 2x_k$ points away from the origin, causing the LMO to select $z_k = (0.5, -5)$ (the vertex minimising $2x_k^T z$ over $\mathcal{X}$) when $x_k$ is in the interior. The iterate thus converges towards the boundary minimum.

\textbf{Sparse solutions property:} A key theoretical property of Frank--Wolfe is that iterates $x_k$ are convex combinations of at most $k+1$ vertices of $\mathcal{X}$. For the simplex $\mathcal{X} = \Delta_d$, this implies \emph{sparse} iterates with at most $k+1$ nonzero coordinates --- a desirable property in machine learning applications such as LASSO and matrix completion.

\textbf{Fixed vs diminishing $\beta$:} The implementation uses a fixed mixing parameter $\beta$ (or equivalently, fixed step $\gamma = 1 - \beta$ towards the LP vertex). Theorem~\ref{thm:fw} applies to the diminishing schedule $\gamma_k = 2/(k+2)$ (i.e., $\beta_k = k/(k+2)$). With fixed $\gamma = 1 - \beta = 0.015$ ($\beta = 0.985$), the effective step size is comparable to $\gamma_k = 2/(k+2)$ for $k \approx 2/0.015 - 2 \approx 131$ --- so after about 131 iterations, the fixed schedule is equivalent to the optimal diminishing schedule. This explains why $\beta = 0.985$ produces lower final values: it maintains a small effective step throughout, avoiding the oscillation induced by large steps into LP vertices.

\textbf{Connection to Projected GD:} Projected GD updates $x_{k+1} = \Pi_{\mathcal{X}}(x_k - \alpha \nabla f(x_k))$ and achieves linear convergence $(1-\mu/L)^k$ for strongly convex $f$. Frank--Wolfe achieves only $O(1/k)$ sublinear convergence, even for strongly convex $f$ (due to the zig-zag behaviour). For $f(x) = (x_1-1)^2 + (x_2-5)^2$ (interior optimum, $\mu=L=2$, $\kappa=1$), projected GD would converge in approximately one step, while Frank--Wolfe requires many iterations. The trade-off: Frank--Wolfe requires only an LP solver (LMO) rather than a projection oracle, which is crucial when projection is expensive (e.g., nuclear norm ball, spectrahedron) while LPs remain tractable.

\subsection{Key Findings}

\begin{enumerate}[leftmargin=*]
\item \textbf{Problem geometry dominates algorithm selection.} The Rosenbrock function ($\kappa \approx 2508$ at the optimum) starkly differentiates methods: Newton achieves $f=0.686$ in 20 iterations, Polyak achieves $f=0.024$ (Q1 best, with correct $f^\star$), while GD reaches only $f=3.73$ in 80 iterations. On Benchmark~A ($\kappa = 1.12$, nearly isotropic), all methods converge to the same noise floor --- the problem is so well-conditioned that choice of algorithm is irrelevant.

\item \textbf{Adaptive step sizes are sensitive to $f^\star$ specification.} The Polyak step size is theoretically elegant but produces massive transient oscillations when $f^\star$ is misspecified: with $|f^\star_{\rm spec} - f^\star_{\rm true}| \approx 0.48$, the step blows up to peak $f_{\rm max} = 148$ (Bench~A) and $432$ (Bench~B) before partially damping. Adagrad and RMSprop are more robust but require tuning $\alpha_0$.

\item \textbf{Momentum provides consistent acceleration.} Nesterov's acceleration ($O(1/k^2)$ rate, Theorem~\ref{thm:nesterov}) outperforms GD ($O(1/k)$ rate) across all three benchmarks. The theoretical speedup is $O(\sqrt{\kappa})$: for Benchmark~C ($\kappa=2508$), reaching $1\%$ of initial gap requires $\approx 11{,}547$ GD iterations vs $\approx 231$ Nesterov iterations ($\approx50\times$ speedup). Experimentally, Nesterov achieves a $7\times$ improvement in final objective over GD at 150 iterations on Rosenbrock. Adam combines momentum with adaptive scaling, performing well on anisotropic problems but less effectively than Nesterov on Rosenbrock, where consistent directional momentum dominates per-coordinate scaling.

\item \textbf{Second-order methods are dramatically faster on smooth problems.} Newton's method converges in one step on the quadratic (Benchmark~A) and reaches much lower objective values than first-order methods in fewer iterations on Benchmarks~B and C. The cost is $O(d^3)$ per iteration (Hessian solve), which is prohibitive in high dimensions but negligible for $d=2$.

\item \textbf{Derivative-free methods trade accuracy for flexibility.} Forward finite differences with $\delta = 0.05$ achieves gradient approximation error $\approx 0.25$ at $x=[0,3]^T$ (truncation-dominated regime), closely tracking exact GD. At $\delta = 0.8$, the error rises to $\approx 4.1$ (70\% relative error), causing degraded convergence. The optimal $\delta^* \approx \sqrt{\varepsilon_{\rm machine}} \approx 10^{-8}$ balances truncation $O(\delta)$ and round-off $O(\varepsilon_{\rm machine}/\delta)$ errors; central differences improve this to $O(\delta^2)$ with optimal $\delta^* \approx 10^{-5}$. Nelder--Mead effectively navigates Rosenbrock without any gradient but lacks convergence guarantees and scales as $O(n!)$ in dimension. Grid search guarantees global optimality within its resolution but is intractable beyond $d \approx 3$.

\item \textbf{Constrained optimisation requires problem-specific strategy.} Projected GD achieves strict feasibility at every iteration and is preferred when a cheap projection oracle exists (e.g., box constraints). Penalty methods avoid the oracle but require careful tuning of $\lambda$: too small ($\lambda = 0.15$) under-penalises violations, too large ($\lambda = 4.5$) causes ill-conditioning requiring smaller $\alpha$. The Augmented Lagrangian achieves exact feasibility at finite $\lambda$ without ill-conditioning, by augmenting the penalty with Lagrange multipliers.

\item \textbf{Frank--Wolfe naturally enforces constraints at the cost of $O(1/k)$ convergence.} The projection-free property makes it attractive for structured constraint sets where LPs are cheap but projections are expensive (nuclear norm ball: LP solves cheaply, projection requires SVD; spectrahedron: LP is eigendecomposition, projection requires Bregman divergence computation). The FW gap $h_k = \nabla f(x_k)^T(x_k - z_k)$ provides an online optimality certificate: final gaps were $h_{180} = 7.26$ ($\beta=0.90$) and $h_{180} = 0.082$ ($\beta=0.985$), confirming the latter is closer to the optimum. The diminishing schedule $\gamma_k = 2/(k+2)$ (Theorem~\ref{thm:fw}) achieves a certified $O(1/k)$ bound; fixed $\beta$ approximates this only for large $k$.
\end{enumerate}

\subsection{Recommendations}

For a practitioner selecting an optimisation algorithm:
\begin{itemize}[nosep]
\item \textbf{Small-scale, smooth, unconstrained:} Newton's method ($O(d^3)$/iter) or BFGS quasi-Newton ($O(d^2)$/iter) for fastest convergence. For $d \leq 1000$, Newton is practical; for $d \leq 10^5$, L-BFGS with $m=10$ history vectors is preferred.
\item \textbf{Large-scale, first-order, well-conditioned:} Nesterov achieves the optimal $O(1/k^2)$ rate; use with step size $\alpha = 1/L$ and the momentum schedule $\beta_k = (k-1)/(k+2)$.
\item \textbf{Large-scale, first-order, ill-conditioned or deep learning:} Adam (default $\beta_1=0.9$, $\beta_2=0.999$, $\varepsilon=10^{-8}$) for robustness to anisotropy and non-stationarity; its per-coordinate preconditioning implicitly handles varying curvature across layers.
\item \textbf{No gradient available (black-box oracle):} Nelder--Mead for $d \leq 10$, smooth problems (reliable convergence, no guarantees); Nesterov random search ($O(1/k)$ rate, 2 evaluations/iter) for $d \leq 100$; model-based Bayesian optimisation for $d \leq 20$ with expensive evaluations ($< 1000$ budget).
\item \textbf{Simple box/half-plane constraints:} Projected GD (projection is $O(d)$, convergence rate matches unconstrained GD on $\mathcal{X}$). For more complex but separable constraints, projected SGD with per-coordinate projections.
\item \textbf{Complex constraints, projection expensive:} Penalty or ALM (augmented Lagrangian) for $\ell_1$ penalties, barrier methods for interior-point; Frank--Wolfe when the LP subproblem is cheap (simplex, nuclear norm ball, spectrahedron).
\item \textbf{Sparse solutions desired:} Frank--Wolfe over the simplex $\Delta_d$ yields iterates with $\leq k+1$ nonzero entries --- natural for sparse signal recovery and structured matrix learning.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient Descent}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z      # extrapolated point
        g = grad_fn(lookahead)           # gradient at lookahead
        z = beta_k * z - alpha * g
        x = x + z                        # x_{k+1} = lookahead - alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # bias-corrected first moment
        v_hat = v_vec / (1 - beta2**t)   # bias-corrected second moment
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)  # solve H_reg * p = g
        except np.linalg.LinAlgError:
            p = g                            # fallback to gradient step
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex Method}]
def nelder_mead(loss_fn, x0, step, n_iters, alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1])  # reflection
        f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1])  # expansion
            f_e = loss_fn(x_e)
            if f_e < f_r: simplex[-1] = x_e; f_vals[-1] = f_e
            else: simplex[-1] = x_r; f_vals[-1] = f_r
        else:
            if f_r < f_vals[-1]:
                x_c = centroid + rho * (x_r - centroid)      # outside contraction
                f_c = loss_fn(x_c)
                if f_c <= f_r: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
            else:
                x_c = centroid + rho * (simplex[-1] - centroid)  # inside contraction
                f_c = loss_fn(x_c)
                if f_c < f_vals[-1]: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (Conditional Gradient) Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # Linear minimisation oracle for box constraints
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z  # convex combination
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
