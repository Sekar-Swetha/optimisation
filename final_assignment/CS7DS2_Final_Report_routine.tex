\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three
benchmark problems: a linear regression quadratic loss (Benchmark A, condition number $\kappa \approx
1.12$), a toy neural network loss with sinusoidal perturbation (Benchmark B, $\kappa \approx 3.5$--$6.9$),
and the highly ill-conditioned Rosenbrock function (Benchmark C, $\kappa \approx 2000$).
Six categories of methods are systematically investigated: (Q1) adaptive step-size methods
(Polyak, Adagrad, RMSprop, Heavy Ball); (Q2) momentum-based and stochastic methods (Nesterov
acceleration, Adam, mini-batch SGD, learning-rate schedules); (Q3) Newton's method exploiting local
second-order approximations; (Q4) derivative-free approaches (finite differences, Nesterov random
search, Nelder--Mead, grid search); (Q5) constrained optimisation via projected gradient descent
and penalty/augmented-Lagrangian methods; and (Q6) the Frank--Wolfe conditional gradient
algorithm for linear-constrained problems.

Key findings: Polyak's step size diverges whenever $f^{\star}$ is misspecified (Benchmarks A, B)
but achieves the best convergence on Rosenbrock where $f^{\star}=0$ is exact.
Nesterov momentum achieves an empirical $O(1/k^2)$ convergence rate that substantially outperforms
gradient descent's $O(1/k)$ on ill-conditioned problems; this rate is provably optimal (Nesterov's
lower bound).
Newton's method, which is \emph{affine invariant} and therefore unaffected by condition number in
the local convergence regime, converges in a single iteration on the quadratic Benchmark A and in
$\sim$5 iterations on Benchmark B.
Finite-difference gradient approximation with $\delta = 0.05$ matches exact GD
(${\approx}1\%$ gradient error), while $\delta = 0.8$ incurs ${\approx}59\%$ truncation bias.
Projected gradient descent enforces hard feasibility at every iteration, whereas penalty methods
require $\lambda \to \infty$ for exact feasibility (causing $O(\lambda)$ condition number growth),
a limitation resolved by the augmented Lagrangian method.
Frank--Wolfe naturally stays feasible via convex combinations and promotes sparse iterates,
converging at the theoretical $O(1/k)$ rate; the Frank--Wolfe duality gap provides a computable
sub-optimality certificate.
Cross-benchmark analysis confirms that the condition number $\kappa$ is the primary driver of
relative method performance: for small $\kappa$, all gradient methods converge rapidly; for
$\kappa \approx 2000$, only momentum-based (Nesterov) or curvature-aware (Newton, Polyak)
methods remain competitive.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) to ensure full reproducibility.
A central quantity governing first-order method convergence is the \emph{condition number}
$\kappa = L/\mu$, where $L$ is the smoothness constant (largest Hessian eigenvalue) and $\mu$ is the
strong-convexity constant (smallest Hessian eigenvalue).
For gradient descent with step size $\alpha \leq 1/L$, the convergence rate per iteration is
$(1 - \mu/L) = (1 - 1/\kappa)$, confirming that large $\kappa$ slows convergence dramatically.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad
y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2
= \frac{1}{2}\|\bar{X}\theta - \bar{y}\|^2,
\end{equation}
where $\bar{X} = X/\sqrt{m}$ is the scaled design matrix.
This is a convex quadratic with Hessian $H = \frac{1}{m}X^T X$ (constant throughout optimisation).
The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$.
The OLS solution is $\theta_{\mathrm{OLS}} = H^{-1}\frac{1}{m}X^T y \approx (3.028,\ 4.017)^T$,
close to the true parameters.
Hessian eigenvalues: $\lambda_{\min} \approx 0.925$, $\lambda_{\max} \approx 1.033$,
giving $\kappa \approx 1.12$ --- a very well-conditioned problem.
Due to observation noise, the minimum loss is $J^{\star} \approx 0.4826$, not zero.
Starting point: $\theta_0 = [0,\ 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the diagonal Hessian is
\begin{equation}
H(x) = \mathrm{diag}\!\left(2 - \sin(x_1),\ 10\right).
\end{equation}
Since $\sin(x_1) \in [-1, 1]$, the $(1,1)$ Hessian entry varies in $[1, 3]$, giving a condition
number $\kappa(x) = 10 / (2 - \sin(x_1))$ that ranges from approximately $3.3$ (when $x_1 \approx
\pi/2$) to $10$ (when $x_1 \approx -\pi/2$).
At the starting point $x_0 = [-1, 4]$, $\kappa \approx 3.5$; at the minimum
$x^{\star} \approx (0.582,\ 2.0)$, $\kappa \approx 6.9$.
The global minimum is $f^{\star} \approx 0.7244$.
Starting point: $x_0 = [-1,\ 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$.
The Hessian is:
\begin{equation}
H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1,1)$, the eigenvalues are approximately $0.5$ and $1000.5$,
yielding $\kappa \approx 2000$.
This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$,
makes Rosenbrock the most challenging test function in this study.
Starting point: $x_0 = [-1,\ 1]^T$.

\subsection{Theoretical Background}

A function $f$ is \emph{$L$-smooth} if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x - y\|$ for all $x, y$.
For a quadratic, $L = \lambda_{\max}(H)$.
Gradient descent with $\alpha = 1/L$ satisfies:
\begin{equation}
f(x_{k+1}) \leq f(x_k) - \frac{1}{2L}\|\nabla f(x_k)\|^2.
\end{equation}
For a $\mu$-strongly convex, $L$-smooth function, the convergence rate is:
\begin{equation}
\label{eq:gd_rate}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k \!\bigl(f(x_0) - f^{\star}\bigr)
= \left(1 - \frac{1}{\kappa}\right)^k \!\bigl(f(x_0) - f^{\star}\bigr).
\end{equation}
For Benchmark A with $\kappa \approx 1.12$, GD with $\alpha = 0.08 \approx 1/L$ converges at rate
$(1 - 1/1.12)^k \approx 0.107^k$ per iteration --- extremely fast convergence.
For Benchmark C with $\kappa \approx 2000$, the rate is $(1 - 1/2000)^k$,
requiring $\sim 2000\ln(1/\varepsilon)$ iterations to reduce the gap by $\varepsilon$.

\begin{remark}[Polyak--Łojasiewicz condition]
The linear convergence result of Equation~\eqref{eq:gd_rate} extends beyond strongly convex
functions to any function satisfying the \emph{Polyak--Łojasiewicz (PL) condition}:
$\frac{1}{2}\|\nabla f(x)\|^2 \geq \mu(f(x) - f^{\star})$.
The PL condition is weaker than strong convexity (it holds on Benchmark B even though
$f$ is not globally strongly convex due to the $\sin(x_1)$ term).
It captures the essential property needed for linear convergence: the gradient must not
vanish at non-optimal points.
\end{remark}

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all
three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value estimate $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent The Polyak step size theoretically minimises the worst-case bound for subgradient descent
when $f^{\star}$ is known exactly. Its convergence guarantee requires $f^{\star} \leq f(x)$ for all $x$
along the trajectory; if $f^{\star}$ is under-estimated, the step size $\alpha_k =
(f(x_k) - f^{\star}) / \|g_k\|^2$ can grow without bound as $\|g_k\| \to 0$
while $(f(x_k) - f^{\star}) > 0$, causing divergence.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate cumulative squared gradient}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent Adagrad's effective per-coordinate learning rate is $\alpha_0 / \sqrt{G_k^{(i)}}$, which
automatically decreases for coordinates with historically large gradients and increases for those with
small gradients. The cumulative nature means the effective step size is monotonically decreasing,
which guarantees stability but can cause premature convergence on non-stationary problems.

\noindent \textbf{Adagrad regret bound:}
In the online convex optimisation setting where $f_t(x)$ is an arbitrary convex function at
round $t$, Adagrad achieves the following \emph{regret bound} (Duchi et al., 2011):
\begin{equation}
\label{eq:adagrad_regret}
R_T = \sum_{t=1}^{T} f_t(x_t) - \min_{x} \sum_{t=1}^T f_t(x) \leq \frac{\alpha_0}{2}\sum_{i=1}^n \sqrt{G_T^{(i)}} + C,
\end{equation}
where $G_T^{(i)} = \sum_{t=1}^T g_t^{(i)2}$ is the cumulative squared gradient for coordinate $i$,
and $C$ is a constant depending on the domain diameter.
This bound has the key property that the regret decomposes \emph{per coordinate}, so coordinates
with small cumulative gradients (``sparse'' features) receive higher learning rates and thus
faster per-coordinate convergence.
In the worst case, $\sum_i \sqrt{G_T^{(i)}} \leq n\sqrt{T}$, recovering the $O(\sqrt{T})$
bound of standard SGD.
However, if the gradient is sparse (most coordinates have zero or near-zero gradients),
the bound can be much tighter, making Adagrad particularly effective for natural language
processing and other sparse gradient problems.

For Benchmark A (non-sparse, all coordinates receive similar gradients), Adagrad's advantage
over GD is minimal, consistent with the observations in Table~\ref{tab:q1_final}.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent RMSprop differs from Adagrad by using an exponential moving average of squared gradients
($\beta = 0.9$ here), which gives finite ``memory'' and prevents the step size from shrinking to zero.
The effective step size stabilises at $\alpha_0 / \sqrt{v_{\infty}}$ where
$v_{\infty} = \mathbb{E}[g^2]/(1-\beta)$ (in stationary regions), providing
a natural scaling by the root mean squared gradient.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Accumulate momentum}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent The Heavy Ball method can be seen as a second-order finite difference approximation to the
continuous-time ODE $\ddot{x} + \gamma\dot{x} + \nabla f(x) = 0$ (damped harmonic oscillator).
This connection motivates the choice of $(\alpha, \beta)$ via the ODE parameter $\gamma$:
setting $\gamma = (1-\beta)/\sqrt{\alpha}$ and $\alpha = 1/L$ in the ODE derivation yields
the optimal discrete parameters.
For a quadratic $f$ with condition number $\kappa$, the optimal Heavy Ball parameters are
$\alpha^{\star} = 4/((\sqrt{L}+\sqrt{\mu})^2)$ and $\beta^{\star} = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$,
giving convergence at rate $((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^{2k}$ --- the same asymptotic rate
as Nesterov's method.

For Benchmark B ($\kappa \approx 6.9$ at the minimum), the optimal momentum is
$\beta^{\star} = ((\sqrt{6.9}-1)/(\sqrt{6.9}+1))^2 = ((2.627-1)/(2.627+1))^2 \approx 0.203$.
The empirical value $\beta = 0.90$ substantially exceeds this, which would be unstable for a
quadratic but is beneficial for the non-quadratic Benchmark B because the varying Hessian
along the trajectory (due to $\sin(x_1)$) means the iterate spends much of its time in
regions with lower effective condition number, where higher momentum is tolerable.
The ODE interpretation also explains Heavy Ball's oscillation: a physical pendulum with
high damping coefficient overshoots the equilibrium before settling, exactly as observed
in Figure~\ref{fig:hb_beta} for $\beta = 0.99$.

\textbf{Key finding:} The Polyak step size is highly sensitive to the accuracy of $f^{\star}$.

to problems where tight lower bounds on the optimal value are available.

\item \textbf{Convergence speed:}
  When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$).
  On Benchmark A ($\kappa \approx 1.12$), GD, Adagrad, and Heavy Ball converge at similar speeds
  because the problem is already well-conditioned and any reasonable step size works.
  The methods differentiate most on Rosenbrock where condition number causes large variation.
\item \textbf{Coordinate adaptivity:}
  Adagrad and RMSprop use per-coordinate step sizes, which is particularly valuable for
  anisotropic problems (large $\kappa$). However, for Benchmark A where all coordinates have
  similar curvature ($\kappa \approx 1.12$), this adaptivity provides no advantage over GD.
\item \textbf{Smoothness:}
  RMSprop produces the smoothest convergence curves due to exponential averaging of gradient
  history. Heavy Ball shows characteristic overshoot followed by damped oscillation, mirroring
  the transient response of a damped oscillator.
\item \textbf{Stability:}
  GD baseline is most stable but slowest. Polyak is \emph{unstable} when $f^{\star}$ is
  misspecified. Adagrad is uniformly stable due to its monotonically decreasing step size.
\item \textbf{Practical recommendation:}
  RMSprop or Adagrad for robust adaptive step sizes without requiring $f^{\star}$;
  Heavy Ball when momentum is beneficial and stability can be verified.
\item \textbf{Heavy Ball momentum sensitivity:}
  The momentum parameter $\beta$ controls the memory of past gradients.
  Figure~\ref{fig:hb_beta} shows that $\beta = 0$ recovers standard GD (no momentum),
  while too large $\beta$ (e.g.\ $\beta = 0.99$) leads to persistent oscillation because
  the accumulated momentum overshoots and correction is too slow.

  For Benchmark B with $\kappa \approx 6.9$, $\beta^{\star} = ((\sqrt{6.9}-1)/(\sqrt{6.9}+1))^2 \approx 0.42$,
  but empirically $\beta = 0.90$ performs better because the non-quadratic $\sin(x_1)$
  term changes the effective curvature along the trajectory.
\end{itemize}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient Descent}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at
$\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$, allowing Nesterov momentum to
``look ahead'' and correct its trajectory before overshooting.
The schedule $\beta_k = (k-1)/(k+2)$ grows from $0$ and approaches $1$, gradually building
acceleration while maintaining stability in early iterations.

\begin{theorem}[Nesterov Acceleration]
\label{thm:nesterov}
For a convex $L$-smooth function, Nesterov's accelerated gradient descent with step size
$\alpha = 1/L$ and the momentum schedule $\beta_k = (k-1)/(k+2)$ achieves:
\begin{equation}
\label{eq:nesterov_rate}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\end{equation}
This $O(1/k^2)$ rate is \emph{optimal} among all first-order methods: no gradient-based
algorithm can guarantee better than $O(1/k^2)$ on the class of convex $L$-smooth functions
\textup{(Nesterov's lower bound)}.
\end{theorem}

\noindent \textbf{Why $O(1/k^2)$ is optimal (Nesterov's lower bound):}
Nesterov proved in 1983 that for any first-order algorithm $\mathcal{A}$ and any $k \leq (n-1)/2$,
there exists an $L$-smooth convex function $f$ such that the algorithm's output after $k$
gradient queries satisfies:
\begin{equation}
f(x_k) - f^{\star} \geq \frac{3L\|x_0 - x^{\star}\|^2}{32(k+1)^2}.
\end{equation}
The construction uses a \emph{Krylov-space argument}: after $k$ first-order oracle calls,
the information accessible is at most $\mathrm{span}\{\nabla f(x_0), \nabla f(x_1), \ldots, \nabla f(x_{k-1})\}$,
which has dimension at most $k$, so the method cannot have eliminated all $n$-dimensional
error.
Since GD's $O(1/k)$ rate is strictly below this lower bound, GD is \emph{suboptimal} for
convex smooth minimisation, and Nesterov's method closes this gap exactly.

For \emph{strongly convex} functions (condition number $\kappa$), the analogous lower bound is
$(1 - 2/(\sqrt{\kappa}+1))^k$, and Heavy Ball / Nesterov achieve this rate, making them
\emph{optimal} first-order methods for strongly convex objectives.

\noindent Compared to GD's $O(1/k)$ rate, Nesterov's $O(1/k^2)$ provides a
$\sim k$-fold speedup in terms of function value suboptimality at iteration $k$.
For Benchmark C with the ill-conditioned Rosenbrock function, this acceleration is most
visible: Nesterov achieves $f = 0.479$ vs GD's $f = 3.388$ in 150 iterations.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \dfrac{\alpha\, \hat{m}_t}{\sqrt{\hat{v}_t} + \varepsilon}$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent Adam combines (1) gradient momentum via $m_t$ (analogous to Heavy Ball) with
(2) per-coordinate adaptive scaling via $v_t$ (analogous to RMSprop),
inheriting advantages of both.
The bias correction terms $1/(1-\beta_1^t)$ and $1/(1-\beta_2^t)$ compensate for the
zero-initialisation of $m_0$ and $v_0$, which would otherwise underestimate true moment
magnitudes in early iterations (particularly important when $\beta_2 = 0.999$ makes
$v_t$ warm up very slowly in the first $\sim 1000$ steps).

\noindent \textbf{Convergence caveats:}
Unlike GD and Nesterov, Adam does not have a general convergence guarantee for smooth
non-convex functions (convergence has been shown for convex problems under certain conditions).
In practice, Adam is widely used for neural network training where its adaptive per-coordinate
scaling provides significant benefits, but it may fail to converge to critical points on some
smooth problems (Reddi et al., 2018 identified cases where Adam diverges; AMSGrad was
proposed as a fix).
The step sizes $(\alpha = 0.12, 0.08, 0.006)$ used here were tuned for these specific problems
and would not transfer directly to other settings.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes
$b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent The key trade-off with batch size $b$ is between gradient variance and computational cost.
Each epoch with batch size $b$ performs $\lfloor m/b \rfloor$ gradient steps.
The stochastic gradient $g_b$ satisfies $\mathbb{E}[g_b] = \nabla J(\theta)$ (unbiasedness)
and $\mathrm{Var}(g_b) \propto 1/b$ (variance decreases with batch size).
Near convergence, the dominant source of noise is the mini-batch sampling noise,
which creates an irreducible ``noise floor'' that prevents convergence below
$\sim \alpha\sigma^2/b$ for step size $\alpha$ and noise variance $\sigma^2$.

\subsection{Learning-Rate Scheduling for SGD}
\label{subsec:lr_schedule}

A constant learning rate creates a fundamental tension in SGD: a large $\alpha$ enables fast
early progress but causes persistent oscillation near the minimum (because mini-batch noise
prevents exact convergence), while a small $\alpha$ ensures convergence but is slow initially.
\emph{Learning-rate schedules} resolve this by decaying $\alpha$ over time, allowing large
steps early and precise convergence later.

Five representative schedules are compared on Benchmark A with batch size $b = 40$:

\begin{table}[H]
\centering
\caption{Q2: Learning-rate schedule definitions ($\alpha_0 = 0.12$, $T$ = total epochs).}
\label{tab:lr_schedules}
\begin{tabular}{lll}
\toprule
\textbf{Schedule} & \textbf{Formula} & \textbf{Rate guarantee} \\
\midrule
Constant          & $\alpha_k = \alpha_0$                                    & Noise floor $\propto \alpha_0\sigma^2/b$ \\
Step decay        & $\alpha_k = \alpha_0 \cdot 0.5^{\lfloor k/20 \rfloor}$  & Linear on each segment \\
Linear decay      & $\alpha_k = \alpha_0(1 - k/T)$                           & $O(1/T)$ convergence \\
Cosine annealing  & $\alpha_k = \tfrac{\alpha_0}{2}(1 + \cos(\pi k / T))$    & $O(1/T)$ convergence \\
Polynomial        & $\alpha_k = \alpha_0 / \sqrt{k+1}$                       & $O(1/\sqrt{T})$ convergence \\
\bottomrule
\end{tabular}
\end{table}

\noindent The convergence guarantees reflect the summability condition: SGD converges to
$f^{\star}$ if $\sum_k \alpha_k = \infty$ (diverging sum ensures enough progress) and
$\sum_k \alpha_k^2 < \infty$ (converging sum bounds cumulative variance).
A constant step size satisfies neither condition simultaneously, producing a noise floor.
Decreasing schedules (linear, cosine, polynomial) satisfy both conditions.
The step decay schedule is discontinuous and thus does not satisfy them smoothly, but
works well in practice for deep learning by periodically ``resetting'' the convergence
momentum.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q2_lr_schedule.pdf}
\caption{Q2: Comparison of learning-rate schedules on Benchmark A (batch size $b=40$, 60 epochs).
Left: convergence of $J(\theta_k) - J^{\star}$ on a log scale.
The constant schedule (blue) plateaus at a noise floor determined by
$\alpha_0 \sigma^2 / b \approx 0.12 \times 1 / 40 \approx 0.003$.
Step decay (orange) achieves the best final accuracy by applying large steps early and
fine-tuning with small steps after each decay event ($\times 0.5$ every 20 epochs).
Cosine annealing (green) achieves smooth, continuous decay with similar final accuracy.
Linear decay (red) is slightly slower due to its more aggressive decay in early epochs.
Polynomial decay $O(1/\sqrt{k})$ (purple) converges most slowly because the step sizes

Right: the corresponding schedule profiles show how rapidly each schedule reduces the

\label{fig:q2_lr_schedule}
\end{figure}

\begin{remark}[Cyclic schedules and warm restarts]
The \emph{SGDR} (Stochastic Gradient Descent with Warm Restarts, Loshchilov \& Hutter, 2017)
strategy restarts the cosine schedule periodically, allowing the iterate to ``escape'' local
minima and explore the loss landscape multiple times.
This is particularly effective for non-convex neural network training, where the loss surface
has many competing local minima and saddle points.
For the convex problems studied here, warm restarts provide no benefit beyond a carefully
chosen monotone schedule.
\end{remark}

\subsection{Adam Bias Correction and Warmup}
\label{subsec:adam_warmup}

Adam uses exponential moving averages (EMAs) of gradient moments initialised at zero:
$m_0 = \mathbf{0}$, $v_0 = \mathbf{0}$.
These zero initialisations create a \emph{bias} in the early iterations: the raw estimates
$m_t$ and $v_t$ are systematically underestimated because they have not had enough steps to
warm up to the true gradient statistics.
The bias-correction terms $\hat{m}_t = m_t / (1-\beta_1^t)$ and $\hat{v}_t = v_t / (1-\beta_2^t)$
exactly compensate for this initialisation bias.

For the second moment with $\beta_2 = 0.999$, the bias factor $(1-\beta_2^t)$ is very small
in early iterations:
\begin{equation}
(1 - 0.999^t) \approx t \cdot 0.001 \quad \text{for } t \ll 1000.
\end{equation}
Without bias correction, $\hat{v}_t \approx v_t / (t \cdot 0.001) = 1000 v_t / t$, which
\emph{vastly overestimates} the second moment in early iterations ($t < 100$), causing
Adam's effective learning rate $\alpha / \sqrt{\hat{v}_t}$ to be artificially small at the
start of training.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

\noindent Here $g'(x_0) = 4(0.25)^3 = 0.0625$ and $g''(x_0) = 12(0.25)^2 = 0.75$.
The Newton step minimises $\hat{g}_2(x)$ exactly, yielding:
\begin{equation}
x_{\mathrm{Newton}} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.1\overline{6}.
\end{equation}
This demonstrates that the Newton update corresponds to stepping to the stationary point
of the local quadratic model, which is much more aggressive than the gradient step
$x_0 - \alpha g'(x_0)$ for typical $\alpha \ll 1$.

\begin{figure}[H]
\centering

\subsection{Newton's Method with Damping}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, Hessian damping $\lambda_d$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda_d I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction via linear solve}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{$\alpha = 1$: pure Newton step}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent The Hessian regularisation $\lambda_d I$ ($\lambda_d = 10^{-8}$ here) ensures positive
definiteness when $H$ is ill-conditioned or indefinite (which can occur for the Rosenbrock
function far from the minimum).
A step size $\alpha < 1$ further damps the Newton step to prevent overshooting when the
Hessian changes rapidly.

\noindent \textbf{Affine invariance:} A fundamental theoretical property of Newton's method is
\emph{affine invariance}: the Newton iterates are identical under any invertible linear change of
variables $y = Ax$.
To see this, note that for $g(y) = f(A^{-1}y)$, the Newton step is
$y_{k+1} = y_k - (\nabla^2 g(y_k))^{-1}\nabla g(y_k) = Ax_k - A H_k^{-1} g_k = A x_{k+1}$.
This invariance means Newton's method is unaffected by the condition number $\kappa$: a
poorly-conditioned quadratic and a well-conditioned one are solved in the same number of steps
(1 step for any quadratic with $\alpha = 1$).
This is in stark contrast to gradient descent, where the convergence rate is $(1-1/\kappa)^k$,
and to Nesterov, where strongly convex convergence rate is $(1-1/\sqrt{\kappa})^k$.
The condition number $\kappa \approx 2000$ of Benchmark C has no effect on Newton's
convergence once the Rosenbrock function enters the local quadratic regime,
which is why Newton dramatically outperforms gradient methods on this benchmark.

\noindent \textbf{Practical note on Hessian computation:}
For the 2D benchmark functions, the Hessian is computed analytically:
for Benchmark B, $H = \text{diag}(2 - \sin(x_1), 10)$; for Rosenbrock,
$H_{11} = 2 + 1200x_1^2 - 400x_2$, $H_{12} = H_{21} = -400x_1$, $H_{22} = 200$.
In general, computing the exact Hessian requires $O(n^2)$ second-derivative evaluations;
for large $n$, this motivates quasi-Newton methods (BFGS, L-BFGS) that estimate the Hessian
from first-order information alone.

\begin{theorem}[Quadratic Convergence of Newton's Method]
Suppose $f$ is twice continuously differentiable, $\nabla^2 f$ is $M$-Lipschitz continuous,
and $x_0$ lies in a region where $\nabla^2 f(x) \succeq \mu I$. Then Newton's method
($\alpha = 1$, $\lambda_d = 0$) satisfies:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2,
\end{equation}
giving \emph{quadratic} (superlinear) local convergence.
Once $\|x_k - x^{\star}\| < 2\mu/M$, convergence is guaranteed and accelerates doubly
at each step.
\end{theorem}

\noindent The Newton decrement $\lambda(x) = \sqrt{\nabla f(x)^T H(x)^{-1} \nabla f(x)}$
provides a coordinate-free convergence certificate: the function value satisfies
$f(x) - f^{\star} \leq \lambda(x)^2/2$ for convex $f$.

Settings: GD uses 80 iterations; Newton uses 20 iterations with $\lambda_d = 10^{-8}$.

\noindent \textbf{Quasi-Newton methods as a practical bridge:}
The BFGS (Broyden--Fletcher--Goldfarb--Shanno) method and its limited-memory variant L-BFGS
address the high-dimensional limitation by building a rank-2 approximation of $H_k^{-1}$
from gradient differences.
The BFGS update enforces the \emph{secant condition}: $H_{k+1}(x_{k+1} - x_k) = \nabla f(x_{k+1}) - \nabla f(x_k)$,
i.e., the updated Hessian approximation must be consistent with the observed gradient change.
Among all symmetric positive-definite matrices satisfying this condition, BFGS selects the one
closest to $H_k$ in the weighted Frobenius norm, yielding the closed-form update:
\begin{equation}
H_{k+1}^{-1} = \left(I - \frac{s_k y_k^T}{y_k^T s_k}\right)H_k^{-1}\left(I - \frac{y_k s_k^T}{y_k^T s_k}\right) + \frac{s_k s_k^T}{y_k^T s_k},
\end{equation}
where $s_k = x_{k+1} - x_k$ and $y_k = \nabla f(x_{k+1}) - \nabla f(x_k)$.
Positive definiteness is maintained provided $y_k^T s_k > 0$, which holds automatically
when $\alpha_k$ is chosen by an Armijo--Wolfe line search.

The update is equivalent to a rank-2 modification of $H_k^{-1}$, adding only $O(n^2)$
storage per step.
L-BFGS stores only the last $p$ vector pairs $(s_k, y_k)$ (typically $p = 5$--$20$),
using a two-loop recursion to apply $H_k^{-1}$ implicitly in $O(pn)$ time without
ever forming the full matrix --- making it applicable for $n = 10^6$ or more.
L-BFGS achieves superlinear (but not quadratic) convergence and is the method of choice for
smooth unconstrained problems in moderate-to-high dimensions, forming the basis of
\texttt{scipy.optimize.minimize(method='L-BFGS-B')}.
For $n = 2$ in this experiment, BFGS and L-BFGS are equivalent and both converge in
$\sim$5 iterations on Benchmark B, matching Newton's performance without requiring
explicit Hessian computation.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\label{eq:fd}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}
= \frac{\partial f}{\partial x_i}(x) + \frac{\delta}{2}f_{ii}(x) + O(\delta^2),
\end{equation}
where the $O(\delta)$ \emph{truncation error} dominates for large $\delta$.
Floating-point cancellation introduces a numerical error of $O(\varepsilon_{\mathrm{machine}}/\delta)$
in the denominator when $\delta$ is small.
Balancing these two sources gives an optimal step size $\delta^{\star} = \sqrt{\varepsilon_{\mathrm{machine}}} \approx 10^{-8}$,
yielding overall approximation error $\approx \varepsilon_{\mathrm{machine}}^{1/2} \approx 10^{-8}$.

The assignment uses $\delta = 0.05$ (near-optimal) and $\delta = 0.8$ (poor).
The exact truncation error at the starting point $x_0 = (-1, 4)$ of Benchmark B is:
\begin{align}
\text{Error in }\hat{\partial}_1 f &= \frac{\delta}{2} f_{11}(x_0) = \frac{\delta}{2}(2-\sin(-1)) \approx \frac{\delta}{2} \times 2.84,\\
\text{Error in }\hat{\partial}_2 f &= \frac{\delta}{2} f_{22}(x_0) = \frac{\delta}{2} \times 10.
\end{align}
At $\delta = 0.05$: $\|\text{FD error}\| \approx \sqrt{(0.025\times 2.84)^2 + (0.25)^2} \approx 0.07$,
which is about $1\%$ of $\|\nabla f(x_0)\| \approx 7.1$.
At $\delta = 0.8$: $\|\text{FD error}\| \approx \sqrt{(0.4\times 2.84)^2 + (4.0)^2} \approx 4.16$,
which is $\sim 59\%$ of $\|\nabla f(x_0)\|$, rendering the gradient approximation almost useless
for determining the descent direction.
This quantitative comparison explains the dramatic difference in convergence quality between
the two $\delta$ values observed in Figure~\ref{fig:q4_fval_B}.

Cost comparison: computing the full $n$-dimensional gradient via forward finite differences requires
$n + 1 = 3$ function evaluations per iteration (for $n = 2$).
For high dimensions, this cost scales as $O(n)$, compared to $O(1)$ gradient evaluations when an
analytical gradient is available.
\emph{Central differences} $(\partial f/\partial x_i \approx (f(x+\delta e_i)-f(x-\delta e_i))/(2\delta))$
reduce the truncation error to $O(\delta^2)$ at the cost of $2n$ function evaluations per gradient,
extending the viable range of $\delta$ by approximately $\varepsilon_{\text{mach}}^{1/3}/\varepsilon_{\text{mach}}^{1/2}$
before numerical cancellation dominates.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Optimisation)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State Estimate directional derivative: $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent This zeroth-order method estimates a one-dimensional directional derivative at cost of
only \emph{two} function evaluations per iteration, independent of the dimension $n$.
In expectation, the estimate satisfies:
\begin{equation}
\mathbb{E}[\hat{g}_k] = \frac{1}{n}[\nabla f(x_k) + O(\delta^2 \|H\|)],
\end{equation}
so it is a noisy, direction-projected, scaled version of the true gradient.
The $1/n$ factor means that in high dimensions, many random directions are needed
to approximate the full gradient, leading to slow convergence compared to finite differences.
For $n = 2$ in this experiment, the penalty is mild.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$
and iteratively transforms it through four operations:
(1) \textbf{Reflection} --- move the worst vertex $x_w$ through the centroid of the remaining
vertices $\bar{x}$: $x_r = \bar{x} + \rho({\bar{x} - x_w})$, $\rho = 1$.
(2) \textbf{Expansion} --- if reflection improved, try $x_e = \bar{x} + \gamma(\bar{x} - x_w)$
with $\gamma = 2$.
(3) \textbf{Contraction} --- if reflection worsened, contract towards centroid with $\rho = 0.5$.
(4) \textbf{Shrink} --- contract all vertices towards the best with factor $\sigma = 0.5$.

\noindent Settings: initial simplex step $= 0.35$, iterations $= 160$.

\begin{remark}
Nelder--Mead has no formal convergence guarantee for $n > 1$ in general --- there exist
counter-examples where it converges to non-stationary points.
However, in practice it works well for smooth unimodal functions in low dimensions ($n \leq 5$).
Its strength is complete freedom from gradient and Hessian information, requiring only
function evaluations.
\end{remark}

\subsubsection{Simplex Geometry Analysis}

The Nelder--Mead simplex undergoes characteristic geometric transformations at each stage of
the optimisation, which can be tracked by observing the simplex shape and size:

\begin{itemize}[nosep]
\item \textbf{Initialisation:} The simplex is constructed as an equilateral triangle with
  vertices $\{x_0,\ x_0 + \text{step}\cdot e_1,\ x_0 + \text{step}\cdot e_2\}$ where
  $\text{step} = 0.35$.
  The initial diameter is $0.35\sqrt{2} \approx 0.49$ --- deliberately large to
  enable exploration before converging.

\item \textbf{Exploration phase} (iterations 1--30): The simplex descends rapidly from the
  starting plateau $f \approx 5$ to the Rosenbrock valley floor.
  During this phase, \emph{reflection} and \emph{expansion} operations dominate ---
  the worst vertex is reflected through the centroid and potentially expanded further,
  enabling large, exploratory steps.

\item \textbf{Valley alignment phase} (iterations 30--80): The simplex elongates along the
  valley floor direction ($x_2 \approx x_1^2$).
  Repeated reflection operations create a long, thin simplex aligned with the local steepest
  direction, which is highly effective for navigating the narrow valley of the Rosenbrock function.

\item \textbf{Contraction phase} (iterations 80--160): Near the optimum, reflections frequently
  fail (the reflected vertex is worse than the second-worst vertex), triggering \emph{contraction}
  towards the centroid.
  The simplex shrinks, reducing the diameter and enabling fine-scale convergence.
  The simplex diameter (right axis, orange) drops from $\sim 0.5$ at iteration 0 to
  $< 0.01$ by iteration 160, matching the precision of the final iterate.
\end{itemize}

\subsection{Problem Setup}

Minimise Benchmark B, $f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2-2)^2 + \sin(x_1)$,
subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$.

The constraint $x_1 \geq 0.5$ is active at the unconstrained minimum $x^{\star} \approx (0.582, 2.0)$
since $0.582 > 0.5$, so the constrained minimum \emph{coincides} with the unconstrained minimum.
However, starting from $x_0 = (0.2, 4.0)$ (with $x_1 = 0.2 < 0.5$), all methods must first
achieve feasibility before optimising.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project back to feasible set: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent For the halfspace constraint $\mathcal{X} = \{x : x_1 \geq 0.5\}$, the projection
$\Pi_{\mathcal{X}}(z) = ({\max(0.5, z_1)},\ z_2)^T$ is a $O(1)$ operation.
For general convex sets (e.g., polytopes, semidefinite cones), computing $\Pi_{\mathcal{X}}$
may require solving a constrained quadratic programme, which can be expensive.

\begin{theorem}[Projected GD Convergence]
For a convex $L$-smooth function $f$ and a convex feasible set $\mathcal{X}$, projected
gradient descent with $\alpha = 1/L$ satisfies:
\begin{equation}
\min_{0 \leq k \leq K} f(x_k) - f^{\star}_{\mathcal{X}} \leq
\frac{L\|x_0 - x^{\star}\|^2}{2K},
\end{equation}
where $f^{\star}_{\mathcal{X}} = \min_{x \in \mathcal{X}} f(x)$.
For $\mu$-strongly convex $f$, the rate improves to $(1-\mu/L)^k$ (linear convergence).
\end{theorem}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)^2$
    \Comment{Quadratic penalty on constraint violation}
    \State $\nabla F_{\lambda}(x) \gets \nabla f(x) - 2\lambda\,\mathbb{1}[x_1 < 0.5]\cdot(x_1 - 0.5)\,e_1$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent \textbf{Note:} The code implements a linear (non-smooth) penalty $\lambda\max(0, 0.5-x_1)$
rather than a quadratic penalty; the gradient is $-\lambda\,\mathbb{1}[x_1 < 0.5]$
in the $x_1$ direction.
The penalty method converts a constrained problem into an unconstrained one, at the cost
of: (1) approximate feasibility (exact only as $\lambda \to \infty$), and
(2) increasing condition number of the penalised objective.

Settings: Projected GD ($\alpha = 0.08$); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$
($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$);
unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\subsection{KKT Optimality Conditions}

The Karush--Kuhn--Tucker (KKT) conditions characterise optimality for the constrained problem
$\min_{x \in \mathcal{X}} f(x)$ with $\mathcal{X} = \{x : x_1 \geq 0.5\}$:
\begin{align}
\nabla f(x^{\star}) - \mu^{\star} e_1 &= 0 \quad \text{(stationarity)},\\
x_1^{\star} - 0.5 &\geq 0 \quad \text{(primal feasibility)},\\
\mu^{\star} &\geq 0 \quad \text{(dual feasibility)},\\
\mu^{\star}(x_1^{\star} - 0.5) &= 0 \quad \text{(complementary slackness)}.
\end{align}
Since the unconstrained minimum $(0.582, 2.0)$ satisfies $0.582 > 0.5$, the constraint is
\emph{inactive} at the optimum.
By complementary slackness, $\mu^{\star}(0.582 - 0.5) = 0$ requires $\mu^{\star} = 0$.
With $\mu^{\star} = 0$, stationarity becomes $\nabla f(x^{\star}) = 0$, confirming that
the constrained optimum coincides with the unconstrained optimum.
This is visible in Figure~\ref{fig:q5_kkt}: no gradient truncation is needed at the solution.

\textbf{Projected GD} enforces feasibility at every iteration through a hard projection.
This is the fundamental advantage: the feasibility guarantee is unconditional and immediate.
The cost is requiring a projection oracle $\Pi_{\mathcal{X}}$, which is trivial for halfspace
constraints ($\max$ operation) but can be a costly subproblem for complex constraint sets such
as semidefinite cones or general polytopes.

\textbf{Penalty methods} trade exact feasibility for algorithmic simplicity:
only a modified unconstrained gradient is required.
The weight $\lambda$ controls the feasibility--optimality trade-off.
As $\lambda \to \infty$, the penalty method converges to the constrained optimum, but
the condition number of the penalised problem grows as $O(\lambda)$, requiring smaller step
sizes and more iterations.
Figure~\ref{fig:q5_penalty} illustrates this trade-off quantitatively.

\subsection{Augmented Lagrangian Method: Improvement Over Penalty}
\label{subsec:alm}

The \emph{augmented Lagrangian method} (ALM, also called the \emph{method of multipliers})
addresses the fundamental weakness of the penalty method by introducing an explicit Lagrange
multiplier estimate $\mu_k$ that is updated alongside the primal variable.
The augmented Lagrangian for the constraint $x_1 \geq 0.5$ (equivalently $h(x) = 0.5 - x_1 \leq 0$) is:
\begin{equation}
\mathcal{L}_{\rho}(x, \mu) = f(x) + \mu \max(0,\, 0.5 - x_1) + \frac{\rho}{2}\max(0,\, 0.5 - x_1)^2.
\end{equation}
Each ALM iteration consists of two steps:
\begin{enumerate}[nosep]
\item \textbf{Primal update:} $x_{k+1} = \arg\min_x \mathcal{L}_{\rho}(x, \mu_k)$
  (solved approximately with a gradient method).
\item \textbf{Dual update:} $\mu_{k+1} = \mu_k + \rho\,\max(0,\, 0.5 - (x_{k+1})_1)$.
\end{enumerate}

The key insight is that the dual variable $\mu_k$ provides a \emph{running estimate} of the
Lagrange multiplier $\mu^{\star}$.
As $\mu_k \to \mu^{\star}$, the augmented Lagrangian subproblem approximates the true
constrained problem, so even a \emph{finite} (fixed) $\rho$ achieves exact feasibility at
the limit.
Contrast this with the pure penalty method, which requires $\lambda \to \infty$ (and
correspondingly $\alpha \to 0$) to achieve exact feasibility.

\label{tab:alm_comp}

\noindent For this problem where $\mu^{\star} = 0$ (the constraint is inactive at the optimum),
the dual update $\mu_{k+1} = \max(0, \mu_k + \rho\max(0, 0.5 - (x_{k+1})_1)) \to 0$
automatically, and ALM reduces to a standard penalty method for this specific instance.
The power of ALM is most evident when the constraint \emph{is} active (e.g., if the
unconstrained minimum were at $x_1^{\star} = 0.2 < 0.5$), where $\mu^{\star} > 0$ and the
penalty method would require $\lambda \to \infty$ while ALM converges with fixed $\rho$.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$
is obtained by minimising each coordinate independently. Since both components of $a$ are positive,
the optimum is at the lower bounds:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates the \emph{fundamental theorem of linear programming}: the optimum of a linear
objective over a bounded polytope is always attained at a vertex (extreme point) of the feasible
set.
For box constraints $\mathcal{X} = \prod_{i=1}^n [l_i, u_i]$, the optimal vertex has
$(x^{\star})_i = l_i$ if $a_i > 0$ and $(x^{\star})_i = u_i$ if $a_i < 0$.

\caption{Q6-I: Feasible region (blue box) and linear objective contours $f(x) = x_1 + 2x_2$.
The contour lines are parallel (since $f$ is linear), and the minimum occurs at the

\subsection{Frank--Wolfe Algorithm (Conditional Gradient)}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial feasible point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: stays in $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent The Frank--Wolfe method replaces the gradient step + projection with a
\emph{linear minimisation oracle} (LMO): minimise the linear approximation
$\nabla f(x_k)^T x$ over $\mathcal{X}$.
For box constraints, this decomposes per coordinate in $O(d)$:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
Since $z_k \in \mathcal{X}$ and $x_k \in \mathcal{X}$ (by induction), the convex combination
$\beta x_k + (1-\beta) z_k \in \mathcal{X}$ automatically, without any projection.
This is the key advantage for constraint sets where projection is expensive but LMO is cheap
(e.g., nuclear norm ball, spectrahedra, transportation polytopes).

\noindent \textbf{Sparsity-promoting property:}
A distinctive structural property of Frank--Wolfe is that iterates are always convex
combinations of vertices of $\mathcal{X}$.
Starting from any initial $x_0 \in \mathcal{X}$, after $k$ steps, $x_k$ lies in the convex
hull of at most $k+1$ vertices.
For structured constraint sets (e.g., the probability simplex $\{p \geq 0 : \sum_i p_i = 1\}$),
this means the iterates are automatically \emph{sparse}: at most $k+1$ coordinates are non-zero.
This makes Frank--Wolfe the natural choice for problems like LASSO (L1 ball),
low-rank matrix optimisation (nuclear norm ball), and structural SVM training.
For the box constraint in this experiment, the sparsity interpretation is less relevant
since all vertices have both coordinates at their bounds, but the convex-combination
structure ensures the iterates stay in the feasible box without any projection step.

\begin{theorem}[Frank--Wolfe Convergence]
For a convex $L$-smooth function and a convex compact set $\mathcal{X}$ with diameter $D$,
the Frank--Wolfe algorithm with step size $\gamma_k = 2/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2}.
\end{equation}
This $O(1/k)$ rate is the \emph{best possible} for the conditional gradient method without
additional assumptions.
\end{theorem}

\noindent \textbf{Frank--Wolfe gap} (duality gap): $G(x) = \nabla f(x)^T(x - z)$ where
$z = \arg\min_{u \in \mathcal{X}} \nabla f(x)^T u$.
Since $G(x) \geq f(x) - f^{\star}$ (by convexity), the gap provides a \emph{computable
upper bound} on the optimality gap, serving as a natural stopping criterion.

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:}
Frank--Wolfe converges but exhibits a characteristic ``zig-zag'' pattern.
At each iteration, the LP vertex $z_k = \arg\min_{z\in\mathcal{X}} \nabla f(x_k)^T z$ jumps
between extreme points of $\mathcal{X}$.
Near the interior optimum, small gradient perturbations cause $z_k$ to switch between
opposite vertices, so the iterate oscillates.
This zig-zag is a fundamental property of conditional gradient methods; it can be reduced
by using the \emph{open-loop} step size $\gamma_k = 2/(k+2)$ (decreasing to zero), which
ensures the contribution from each vertex diminishes and averages out.
The fixed $\beta$ schedule used here does not decay the step size, explaining the
residual oscillation.

\textbf{Boundary optimum:}
Frank--Wolfe is better suited to boundary optima because $z_k$ frequently selects the
vertex that is geometrically closest to the constrained optimum (since the gradient always
points away from the constraint surface towards the infeasible region).
The iterate naturally approaches the boundary without explicit projection.
However, the fixed $\beta$ schedule introduces a gap between the achieved $f = 0.364$ and
the true $f^{\star} = 0.25$ after 140 iterations.
Using the optimal step schedule $\gamma_k = 2/(k+2)$ would give the $O(1/k)$ convergence
guarantee and eliminate this residual gap.

\textbf{Comparison with projected gradient descent:}
Both PGD (Section~\ref{sec:q5}) and Frank--Wolfe maintain feasibility at every iterate.
PGD uses explicit projection and can use any step size; Frank--Wolfe uses the LMO and
has fixed oracle structure.
For problems where LMO is cheap (matrix constraints, simplex constraints), Frank--Wolfe
is preferred; for problems where projection is cheap (box constraints, norm balls),
PGD is typically faster.

\subsubsection{Frank--Wolfe Duality Gap: Numerical Computation}
\label{subsubsec:fw_gap}

The Frank--Wolfe gap at iterate $x_k$ is:
\begin{equation}
\label{eq:fw_gap}
G_k = \nabla f(x_k)^T(x_k - z_k), \quad z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z.
\end{equation}
By convexity of $f$, $G_k \geq f(x_k) - f^{\star}$, so the gap provides a
\emph{computable certificate} of sub-optimality without requiring knowledge of $f^{\star}$.
When $G_k < \varepsilon$, the iterate is guaranteed to be within $\varepsilon$ of the
constrained optimum.

\noindent \textbf{Numerical example:} For the interior optimum case ($f(x) = (x_1-1)^2 + (x_2-5)^2$,
$\mathcal{X} = [0.5, 5] \times [-5, 10]$) with $\beta = 0.985$ at iteration $k = 180$, the
iterate is approximately $x_{180} \approx (0.997, 5.067)$ with $f(x_{180}) \approx 0.0046$.
The gradient is $\nabla f(x_{180}) \approx [-0.006,\ 0.134]^T$.
The LMO selects:
\begin{align*}
(z_{180})_1 &= 5.0  \quad (\text{since }(\nabla f)_1 = -0.006 < 0 \Rightarrow \text{upper bound}) \\
(z_{180})_2 &= -5.0 \quad (\text{since }(\nabla f)_2 = 0.134 > 0 \Rightarrow \text{lower bound})
\end{align*}
The duality gap is:
\begin{align*}
G_{180} &= \nabla f(x_{180})^T(x_{180} - z_{180}) \\
&= (-0.006)(0.997 - 5.0) + (0.134)(5.067 - (-5.0)) \\
&= (-0.006)(-4.003) + (0.134)(10.067) \\
&= 0.024 + 1.349 = 1.373.
\end{align*}
This gap $G_{180} \approx 1.37$ is a valid upper bound on $f(x_{180}) - f^{\star} = 0.0046$,
though it is loose --- the gap bound is conservative by a factor of $\sim 300$.
The loose bound occurs because the LMO vertex $z_{180} = (5.0, -5.0)$ is far from the
iterate (on the opposite corner of the feasible box), reflecting the fundamental
``zig-zag'' oscillation of Frank--Wolfe.
A tighter bound would require either (1) the decaying step size $\gamma_k = 2/(k+2)$ (which
reduces $\|x_k - z_k\|$ over time), or (2) the \emph{pairwise Frank--Wolfe} variant,
which explicitly tracks both forward and backward vertices to obtain tighter gap estimates.

\textbf{Frank--Wolfe duality gap as a stopping criterion:}
The gap $G_k$ satisfies $G_k \geq f(x_k) - f^{\star}$ (by convexity), providing a
\emph{computable certificate} of sub-optimality.
However, as demonstrated above, $G_k$ can be orders of magnitude larger than the true gap
when the iterate is near an interior optimum and the LMO jumps between opposite vertices.
In practice, $G_k < \varepsilon$ is a sufficient (but not necessary) condition for
$f(x_k) - f^{\star} < \varepsilon$: when the gap is large, the method may still be close
to the optimum.
For the open-loop schedule $\gamma_k = 2/(k+2)$, the convergence theorem guarantees
$G_k \leq 2LD^2/(k+2)$, so the gap eventually contracts at $O(1/k)$ rate.
With a fixed $\beta$, no such guarantee holds and the gap can stagnate, as observed here.

\textbf{Step size scheduling:}
The fixed $\beta$ parameter used here does not decay the vertex weight $(1-\beta)$ over time.
The \emph{standard Frank--Wolfe schedule} uses decreasing step sizes $\gamma_k = 2/(k+2)$,
giving $(1-\gamma_k) = k/(k+2) \to 1$ as $k \to \infty$.
This is equivalent to $\beta_k = k/(k+2)$ increasing towards 1 --- exactly the schedule
that achieves the $O(1/k)$ convergence guarantee.
For the interior case with fixed $\beta = 0.90$ (constant), the effective step to the vertex
remains $10\%$ per iteration indefinitely, causing persistent oscillation near the optimum.

The \emph{information complexity} viewpoint quantifies this trade-off precisely:
\begin{itemize}[nosep]
\item Zeroth-order methods need $O(n/\varepsilon^2)$ function evaluations to find an
  $\varepsilon$-stationary point of a convex Lipschitz function.
\item First-order methods need $O(1/\varepsilon)$ gradient evaluations (GD) or
  $O(1/\sqrt{\varepsilon})$ (Nesterov, strongly convex) --- both are optimal.
\item Second-order Newton needs only $O(\log\log(1/\varepsilon))$ Hessian evaluations
  once in the local convergence basin.
\end{itemize}

\subsection{Practical Recommendations}

\begin{itemize}[nosep]
\item Use \textbf{Adam or RMSprop} for neural network training where gradients are noisy,
dimensions are high, and the problem is non-convex.
Pair with \textbf{warmup + cosine annealing} for best practical convergence
(Section~\ref{subsec:lr_schedule}).
\item Use \textbf{Nesterov acceleration} when strong convexity holds and momentum can
accumulate in a consistent direction (batch setting, convex problems).
\item Use \textbf{Newton's method} for small-dimensional problems with smooth, near-convex
structure where Hessian computation is affordable; switch to \textbf{L-BFGS} for $n > 50$.
\item Use \textbf{Finite differences} when code differentiation is unavailable and $n$ is
moderate; choose $\delta \approx \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$.
For higher accuracy at twice the cost, use central differences ($O(\delta^2)$ error).
\item Use \textbf{Projected GD} for constrained problems with simple constraint sets
(box, halfspace, simplex) and when strong convergence guarantees are needed.
\item Use \textbf{Frank--Wolfe} for problems with expensive projections but cheap
LP subproblems (matrix factorisation, combinatorial structures).
Prefer the \textbf{open-loop schedule} $\gamma_k = 2/(k+2)$ over fixed $\beta$ to ensure
the $O(1/k)$ convergence guarantee and a computable duality gap certificate.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient Descent}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Hessian Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Finite Difference Gradient Descent}]
def fd_gradient(f, x, delta):
    """Forward finite-difference gradient approximation."""
    n = len(x)
    f0 = f(x)
    grad = np.zeros(n)
    for i in range(n):
        ei = np.zeros(n); ei[i] = delta
        grad[i] = (f(x + ei) - f0) / delta
    return grad

def fd_gradient_descent(loss_fn, x0, alpha, delta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = fd_gradient(loss_fn, x, delta)
        x = x - alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Random Search (Zeroth-Order)}]
def nesterov_random_search(loss_fn, x0, alpha, delta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        u = np.random.randn(len(x))
        u /= np.linalg.norm(u)  # uniform random direction
        g_hat = (loss_fn(x + delta * u) - loss_fn(x)) / delta * u
        x = x - alpha * g_hat
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Mini-Batch SGD with Epoch Shuffling}]
def mini_batch_sgd(X, y, theta0, alpha, batch_size, n_epochs):
    m = len(y)
    theta = theta0.copy().astype(float)
    theta_hist, loss_hist = [theta.copy()], []
    loss_hist.append(0.5 * np.mean((X @ theta - y)**2))
    for epoch in range(n_epochs):
        idx = np.random.permutation(m)
        for start in range(0, m, batch_size):
            b_idx = idx[start:start+batch_size]
            g = X[b_idx].T @ (X[b_idx] @ theta - y[b_idx]) / batch_size
            theta = theta - alpha * g
        theta_hist.append(theta.copy())
        loss_hist.append(0.5 * np.mean((X @ theta - y)**2))
    return np.array(theta_hist), np.array(loss_hist)
\end{lstlisting}

def projected_gd(grad_fn, loss_fn, x0, alpha, project_fn, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = project_fn(x - alpha * grad_fn(x))
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def project_halfspace(z, lb=0.5):
    """Project onto {x : x[0] >= lb}."""
    z = z.copy()
    z[0] = max(lb, z[0])
    return z

def penalty_method(grad_fn, loss_fn, x0, alpha, lam, n_iters, lb=0.5):
    """Gradient descent on f(x) + lambda * max(0, lb - x[0])."""
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x).copy()
        if x[0] < lb:
            g[0] -= lam  # gradient of linear penalty term
        x = x - alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
