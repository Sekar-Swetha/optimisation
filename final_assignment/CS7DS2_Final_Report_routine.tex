\begin{abstract}
This report presents a rigorous comparative study of first-order, second-order, derivative-free, and constrained optimisation methods applied to three benchmark problems: a linear regression quadratic loss (Benchmark A), a toy neural network loss with sinusoidal perturbation (Benchmark B), and the Rosenbrock function (Benchmark C). Six categories of methods are investigated: (i)~adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball) evaluated over 120 iterations; (ii)~momentum-based and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD) over 150 iterations; (iii)~Newton's method with Hessian-based local approximation over 20 iterations; (iv)~derivative-free methods (finite differences, Nesterov random search, Nelder--Mead, grid search); (v)~constrained optimisation via projected gradient descent and penalty methods; and (vi)~the Frank--Wolfe conditional gradient algorithm. Key findings include: Polyak step size achieves the best convergence on the Rosenbrock function ($f = 0.009$) when $f^\star$ is exactly known but diverges ($f = 2.72$ on Benchmark~A, $f = 24.07$ on Benchmark~B) when $f^\star$ is misspecified; Nesterov acceleration achieves $f = 0.479$ on Rosenbrock versus GD's $f = 3.39$; Newton's method solves the quadratic loss in a single iteration; and projected gradient descent achieves strict feasibility from iteration 1 whereas penalty methods require tuning $\lambda$ to balance constraint satisfaction against optimality. All experiments use \texttt{numpy.random.seed(42)} for reproducibility.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmark functions are chosen to exhibit qualitatively different landscapes: a well-conditioned quadratic (A), a mildly nonlinear function with moderate anisotropy (B), and a highly nonlinear, ill-conditioned function with a narrow curved valley (C).

\subsection{Notation}

Throughout this report, the following notation is used consistently:
\begin{itemize}[nosep]
  \item $f : \mathbb{R}^d \to \mathbb{R}$: objective function ($d = 2$ in all experiments).
  \item $f^\star = \min_{x \in \mathbb{R}^d} f(x)$ or $\min_{x \in \mathcal{X}} f(x)$: optimal value.
  \item $x^\star$: optimal point. $\nabla f(x)$: gradient at $x$. $\nabla^2 f(x)$ or $H(x)$: Hessian at $x$.
  \item $\alpha$ (or $\alpha_k$): step size / learning rate.
  \item $\beta$ (or $\beta_k$): momentum parameter.
  \item $\odot$: elementwise (Hadamard) product.
  \item $L$: Lipschitz constant of $\nabla f$ (smoothness). $\mu$: strong convexity parameter.
  \item $\kappa = L/\mu$: condition number.
  \item $\varepsilon_{\text{mach}} \approx 2.2 \times 10^{-16}$: machine epsilon for float64.
  \item $\mathcal{X} \subseteq \mathbb{R}^d$: feasible (constraint) set. $\Pi_\mathcal{X}$: Euclidean projection.
\end{itemize}

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
  \theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \in \mathbb{R}^2 \sim \mathcal{N}(0,I), \quad
  y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad
  \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
  J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2
            = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a \emph{strongly convex} quadratic with constant Hessian $H = \frac{1}{m}X^T X \succ 0$ and constant gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to irreducible observation noise, the minimum loss is $J^{\star} \approx 0.4826$ (not zero). The OLS solution $\theta^{\star}_{\mathrm{OLS}} = (X^TX)^{-1}X^Ty$ satisfies $J(\theta^{\star}_{\mathrm{OLS}}) = J^{\star} > 0$.  Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is
$H(x) = \mathrm{diag}(2 - \sin(x_1),\ 10)$.
Since $\sin(x_1) \in [-1,1]$, the $(1,1)$ Hessian entry satisfies $H_{11} \in [1, 3]$, so the condition number $\kappa(H) = 10/H_{11}$ varies between $10/3 \approx 3.3$ (when $\sin(x_1) = -1$) and $10/1 = 10$ (when $\sin(x_1) = 1$). At the starting point $x_0 = (-1, 4)$, $H_{11} = 2 - \sin(-1) \approx 2.84$, giving $\kappa \approx 3.5$. This creates \emph{moderate anisotropy} that challenges isotropic methods. The unique minimum lies at approximately $(x_1^\star, x_2^\star) \approx (0.582, 2.0)$, where $2(x_1^\star - 1) + \cos(x_1^\star) = 0$ and $10(x_2^\star - 2) = 0$, yielding $f^\star \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
  f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$. The Hessian is
\begin{equation}
  H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum, $H(1,1) = \bigl[\begin{smallmatrix}802 & -400 \\ -400 & 200\end{smallmatrix}\bigr]$ with eigenvalues $\lambda_{1,2} = \frac{1002 \pm \sqrt{1002^2 - 4 \cdot 400}}{2} \approx 1001.6,\; 0.4$, giving a condition number $\kappa \approx 2500$. This extreme ill-conditioning, combined with the narrow curved valley $x_2 \approx x_1^2$, makes the Rosenbrock function one of the most challenging standard test functions for gradient-based methods. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained optimisation problems (Sections~\ref{sec:q5}--\ref{sec:q6}), the box-constrained feasible set is:
\begin{equation}
  \mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a convex polytope, which is closed and bounded, guaranteeing that both projected gradient descent and Frank--Wolfe remain within $\mathcal{X}$ at every iteration.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive step-size methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak Step-Size Convergence]
For a convex function $f$ with $f^\star = \min_x f(x)$, the Polyak step size satisfies $f(x_k) - f^\star \to 0$ as $k \to \infty$, and for $L$-smooth functions converges at rate $O(1/k)$.
\end{theorem}
\begin{remark}
The rate degrades to divergence when $f^\star$ is misspecified: if the supplied value $\tilde{f}^\star < f_{\mathrm{true}}^\star$, then $\alpha_k \to \infty$ as $\|g_k\| \to 0$ while $f(x_k) - \tilde{f}^\star$ stays bounded away from zero.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0 > 0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate accumulated squared gradient}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Adagrad Regret Bound \cite{duchi2011}]
For a convex loss and any sequence of iterates, Adagrad achieves regret $R(T) = \sum_{t=1}^T [f(x_t) - f(x^\star)] \leq \frac{\alpha_0}{\varepsilon}\sum_{j=1}^d \|g_{1:T,j}\|_2$, where $\|g_{1:T,j}\|_2 = (\sum_{t=1}^T g_{t,j}^2)^{1/2}$ is the $\ell_2$ norm of gradients in coordinate $j$. This gives an $O(\sqrt{T}/T) = O(1/\sqrt{T})$ average regret, which adapts to the geometry of the gradients: coordinates with sparse (infrequent large) gradients receive larger effective learning rates.
\end{theorem}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0 > 0$, decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop differs from Adagrad in that the denominator uses an \emph{exponential moving average} (controlled by $\beta$) rather than the full historical sum. This allows the effective step size to recover from early large gradients, making RMSprop suitable for non-stationary problems where the gradient scale changes over time.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Heavy Ball Convergence for Quadratics \cite{polyak1964}]
For a quadratic $f(x) = \frac{1}{2}x^T H x - b^T x$ with eigenvalues $0 < \mu \leq \lambda_1 \leq \cdots \leq L$, the optimal parameters
\[
  \alpha^\star = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2}, \quad \beta^\star = \left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^2
\]
yield the asymptotic linear rate $\rho = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$ where $\kappa = L/\mu$. This matches Nesterov's rate $O((\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1})^k)$ for strongly convex quadratics, but without the theoretical guarantees on general convex functions. GD's contraction factor is $\frac{\kappa-1}{\kappa+1}$, which is always larger (worse) than Heavy Ball's $\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}$.
\end{theorem}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$ in 120 iterations). Otherwise, Adagrad and Heavy Ball match or exceed the GD baseline on convex problems.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential smoothing of the effective step. Heavy Ball shows characteristic overshoot followed by damped oscillation, consistent with the underdamped regime of a second-order linear system.
\item \textbf{Stability:} GD baseline is most stable but slowest. Polyak is \emph{conditionally stable}: stable when $f^\star$ is correctly specified but catastrophically unstable otherwise. Adagrad is uniformly stable due to its monotonically decreasing effective step, though this can cause premature stagnation on long optimisation horizons.
\item \textbf{Practical implications:} In machine learning contexts, exact knowledge of $f^\star$ is almost never available, fundamentally limiting Polyak's applicability. Adagrad and RMSprop are preferable for practical problems; Adagrad excels on sparse gradient settings, while RMSprop is better suited to dense, non-stationary gradients.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Accelerated Gradient Descent}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, max momentum $\beta_{\max} \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Optimal Convergence Rate \cite{nesterov1983}]
For an $L$-smooth convex function $f$, Nesterov's accelerated gradient method achieves
\begin{equation}
  f(x_k) - f(x^\star) \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2},
\end{equation}
which is \emph{optimal} in the sense that no first-order method can improve upon $O(1/k^2)$ without additional structural assumptions. This represents a quadratic improvement over vanilla gradient descent's $O(1/k)$ rate for convex functions.
\end{theorem}

The key distinction from the Heavy Ball method (Q1) is the \emph{lookahead (extrapolation)}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$ itself. This ``look before you leap'' strategy allows Nesterov momentum to correct its trajectory before overshooting, providing both faster convergence and better stability on convex functions. On quadratics, both Heavy Ball and Nesterov achieve the same asymptotic rate $O((\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1})^k)$; on general convex functions, only Nesterov carries the $O(1/k^2)$ guarantee, while Heavy Ball can diverge for poor step size choices. The schedule $\beta_k = (k-1)/(k+2) \nearrow 1$ guarantees the $O(1/k^2)$ rate; capping at $\beta_{\max}$ prevents oscillation on nonlinear problems where the quadratic approximation is inaccurate.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha > 0$, moments $\beta_1, \beta_2 \in [0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (mean) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (variance) estimate}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$ \Comment{Bias correction for mean}
    \State $\hat{v}_t \gets v_t / (1 - \beta_2^t)$ \Comment{Bias correction for variance}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam~\cite{kingma2015} combines \emph{gradient momentum} (via $m_t$, analogous to Heavy Ball) with \emph{per-coordinate adaptive scaling} (via $v_t$, analogous to RMSprop~\cite{tieleman2012}). The bias-correction terms $(1 - \beta_1^t)$ and $(1 - \beta_2^t)$ prevent the early iterates from being biased toward zero, which is especially important in the first few iterations when the exponential averages have not yet warmed up. With default parameters $\beta_1 = 0.9$, $\beta_2 = 0.999$, $\varepsilon = 10^{-8}$, Adam is robust across a wide range of problems and is the de-facto standard in deep learning.

\begin{remark}
Despite its practical success, Adam does not have convergence guarantees as strong as SGD with decreasing step sizes for convex problems. The non-convergence example of \cite{reddi2018} in simple convex settings motivates the AMSGrad variant, which replaces $\hat{v}_t$ with $\max(\hat{v}_{t-1}, \hat{v}_t)$ to ensure a non-increasing effective step size, though this instability rarely manifests in practice for the smooth objectives considered here.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD is applied to Benchmark A with learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, data shuffled at every epoch. At each epoch, the $m = 1000$ samples are randomly permuted and processed in mini-batches; within each batch, the gradient estimate is
\begin{equation}
  \hat{g}_k = \frac{1}{b} X_{\mathcal{B}}^T\!\left(X_{\mathcal{B}}\theta_k - y_{\mathcal{B}}\right).
\end{equation}

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha > 0$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $\hat{g}$ is an \emph{unbiased estimator} of the full gradient, $\mathbb{E}[\hat{g}] = \nabla J(\theta)$, with variance proportional to $1/b$. Smaller batches provide more frequent updates ($m/b$ per epoch) but higher gradient variance; larger batches provide smoother estimates but fewer updates per epoch.

\subsection{Local Approximations of $g(x) = x^4$}

Taylor expansion of $g(x) = x^4$ around $x_0 = 0.25$:

\textbf{First-order (linear) approximation:} Using $g'(x) = 4x^3$ and $g'(0.25) = 4(0.25)^3 = 0.0625$,
\begin{equation}
  g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625(x - 0.25).
\end{equation}
The gradient step minimises a linear function, but a linear function has no finite minimum --- the Newton step cannot be defined from the first-order approximation alone.

\textbf{Second-order (quadratic) approximation:} Using $g''(x) = 12x^2$ and $g''(0.25) = 12(0.25)^2 = 0.75$,
\begin{equation}
  g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
Minimising this quadratic in $x$ yields the Newton update:
\begin{equation}
  x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\end{equation}
The Newton step directly targets the minimum of the local quadratic model, using curvature ($g''$) to determine the appropriate step magnitude.

\subsection{Newton's Method with Damping}

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Newton's Quadratic Convergence \cite{nocedal2006}]
If $f$ is twice continuously differentiable with $L$-Lipschitz Hessian and $\nabla^2 f(x^\star) \succ 0$, then for $x_0$ sufficiently close to $x^\star$, Newton's method ($\alpha = 1$, $\lambda = 0$) satisfies
\begin{equation}
  \|x_{k+1} - x^\star\| \leq \frac{L}{2\mu}\|x_k - x^\star\|^2,
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^\star))$. This is \emph{quadratic convergence}: the error is squared at each iteration. Starting from $\|x_0 - x^\star\| \leq \mu/L$, the method achieves $\epsilon$-accuracy in $O(\log\log(1/\epsilon))$ iterations, compared to $O(\kappa \log(1/\epsilon))$ for gradient descent.
\end{theorem}

\begin{remark}
For a quadratic $f(x) = \frac{1}{2}x^T H x - b^T x$, the Hessian is constant ($\nabla^2 f = H$), so Newton's method ($\alpha = 1$) achieves \emph{exact convergence in a single step}: $x_1 = x_0 - H^{-1}\nabla f(x_0) = H^{-1}b = x^\star$. This is confirmed experimentally on Benchmark~A.
\end{remark}

Settings: GD uses 80 iterations; Newton uses 20 iterations with Tikhonov damping $\lambda = 10^{-8}$ (ensures positive definiteness).

\subsection{Part A: Gradient Approximation Methods (Benchmark B)}

\subsubsection{Forward Finite Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad i = 1, \ldots, d.
\end{equation}
This introduces two sources of error:
\begin{enumerate}[nosep]
  \item \textbf{Truncation error:} $O(\delta)$ from the linear Taylor expansion $f(x + \delta e_i) = f(x) + \delta\partial_i f + \tfrac{\delta^2}{2}\partial_{ii} f + \cdots$.
  \item \textbf{Cancellation error:} $O(\varepsilon_{\mathrm{mach}}/\delta)$ from subtracting two nearly-equal floating-point numbers.
\end{enumerate}
Balancing these errors gives an optimal step size $\delta^\star \approx \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$, yielding an error of $O(\sqrt{\varepsilon_{\mathrm{mach}}}) \approx 10^{-8}$. This experiment tests two deliberately sub-optimal choices: $\delta = 0.05$ (truncation dominates, but still acceptable) and $\delta = 0.8$ (large truncation error, gradient direction corrupted).

\begin{remark}
Central differences $[f(x + \delta e_i) - f(x - \delta e_i)]/(2\delta)$ yield $O(\delta^2)$ truncation error, with optimal $\delta^\star \approx \varepsilon_{\mathrm{mach}}^{1/3} \approx 10^{-5}$. The forward difference is used here as the simpler baseline; it requires $d+1$ function evaluations (one reference evaluation reused), while central differences require $2d$.
\end{remark}

\begin{figure}[H]
\centering
\includegraphics[width=0.88\textwidth]{figures/q4_fd_error_analysis.pdf}
\caption{Q4: Gradient approximation error $\|\hat{g} - \nabla f\|$ as a function of step size $\delta$ (evaluated at $x_0 = (-1, 4)$ on Benchmark~B). Forward differences (blue) show the characteristic U-shaped error curve: truncation error $O(\delta)$ dominates for large $\delta$, cancellation error $O(\varepsilon_{\text{mach}}/\delta)$ dominates for small $\delta$, with the optimal at $\delta^\star \approx \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$. Central differences (red) achieve $O(\delta^2)$ truncation error with optimal $\delta^\star \approx 10^{-5}$, providing a $10^{5\times}$ improvement in accuracy. The two experiment values $\delta = 0.05$ (green) and $\delta = 0.8$ (orange) are marked; $\delta = 0.05$ achieves acceptable accuracy while $\delta = 0.8$ incurs significant truncation error, explaining the convergence differences observed in Figure~\ref{fig:q4_fval_B}.}
\label{fig:q4_fd_error}
\end{figure}

\subsubsection{Nesterov Random Search (Zeroth-Order Method)}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Gradient Estimator)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, perturbation $\delta > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \text{Uniform}(\mathbb{S}^{d-1})$ (unit sphere) \Comment{Random direction}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Nesterov random search estimator $\hat{g}_k = \frac{f(x_k+\delta u_k)-f(x_k)}{\delta}u_k$ is a \emph{biased} estimate of the gradient (unlike mini-batch SGD). However, $\mathbb{E}[d \hat{g}_k] \approx \nabla f_\delta(x_k)$ where $f_\delta$ is a Gaussian-smoothed version of $f$, so the method can be viewed as gradient descent on a smoothed objective. Only \emph{two} function evaluations are required per iteration (one at $x_k$, one at $x_k + \delta u_k$), regardless of dimension $d$. This makes zeroth-order methods attractive when $d$ is large or gradients are unavailable (e.g., reinforcement learning, simulation optimisation).

The expected convergence rate of random search is $O(d/k)$ for smooth convex functions, which is $d$ times slower than gradient descent's $O(1/k)$, reflecting the $d$-dimensional averaging cost.

\subsection{Part B: Derivative-Free Methods on Benchmark C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm~\cite{neldermead1965} maintains a \emph{simplex} of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations. Letting the vertices be ordered $f(x_1) \leq f(x_2) \leq f(x_3)$ (best to worst) and $c = \frac{1}{2}(x_1 + x_2)$ the centroid of the $n$ best vertices:

\begin{enumerate}[nosep]
  \item \textbf{Reflection:} $x_r = c + \alpha_r(c - x_3)$, $\alpha_r = 1$.
  \item \textbf{Expansion:} If $f(x_r) < f(x_1)$, try $x_e = c + \gamma(c - x_3)$, $\gamma = 2$.
  \item \textbf{Contraction:} If reflection was worse, contract: $x_c = c + \rho(x_r - c)$ or $c + \rho(x_3 - c)$, $\rho = 0.5$.
  \item \textbf{Shrink:} If contraction fails, shrink all vertices towards $x_1$: $x_i \gets x_1 + \sigma(x_i - x_1)$, $\sigma = 0.5$.
\end{enumerate}

Settings: initial simplex step $= 0.35$, iterations $= 160$. No gradient information is used at any step.

\subsection{Problem Setup}

The constrained problem is:
\begin{equation}
  \min_{x \in \mathbb{R}^2} f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad \text{subject to}\quad x_1 \geq 0.5.
\end{equation}
The starting point $x_0 = (0.2, 4.0)$ is \emph{infeasible} ($x_1 = 0.2 < 0.5$). The unconstrained minimum at $x^\star_{\text{unc}} \approx (0.582, 2.0)$ \emph{is feasible} (since $0.582 > 0.5$), so the constrained minimum coincides with the unconstrained one: $x^\star_{\text{con}} \approx (0.582, 2.0)$ with $f^\star \approx 0.7244$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Feasible initial point $x_0 \in \mathcal{X}$ (or project $x_0$ first), step size $\alpha > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Euclidean projection: $x_1 \gets \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space constraint $\{x : x_1 \geq 0.5\}$, the projection is the closed-form operation $\Pi_\mathcal{X}(z) = (\max(0.5, z_1), z_2)^T$, requiring $O(1)$ computation per iteration.

\begin{theorem}[PGD Convergence \cite{beck2009,nocedal2006}]
For a convex $L$-smooth function and a closed convex set $\mathcal{X}$, projected gradient descent with step size $\alpha = 1/L$ satisfies
\[
  f(x_k) - f(x^\star) \leq \frac{L\|x_0 - x^\star\|^2}{2k} = O(1/k).
\]
With the Nesterov momentum extension (FISTA), this improves to $O(1/k^2)$.
\end{theorem}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$ (possibly infeasible), step size $\alpha > 0$, penalty $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x_k) \gets f(x_k) + \lambda \max(0,\, -x_1^{(k)} + 0.5)$ \Comment{Penalised objective}
    \State Compute gradient: $\nabla_x F_\lambda = \nabla f(x_k) + \lambda \cdot [-\mathbf{1}_{x_1 < 0.5}, 0]^T$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$, 100 iterations); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$); Unconstrained GD ($\alpha = 0.07$). All run for 100 iterations. Note that a smaller step size is required for larger $\lambda$ because the penalty function has a larger Lipschitz constant for the gradient.

\textbf{Projected Gradient Descent} enforces feasibility as a hard constraint at every iteration via the projection operator. Its advantages are: exact feasibility throughout, the standard convergence analysis applies immediately for convex problems, and the step size can be tuned independently of constraint strength. Its limitation is requiring a tractable projection oracle; for complex constraint sets (e.g., semidefinite constraints, second-order cone constraints), projection can be as expensive as solving the original problem.

\textbf{Penalty Methods} avoid explicit projection by modifying the objective. The key trade-off is:
\begin{itemize}[nosep]
  \item \emph{Small $\lambda$:} The penalty gradient $\lambda \mathbf{1}_{x_1<0.5}$ is weak, allowing the iterate to remain infeasible. The method converges to the wrong point.
  \item \emph{Large $\lambda$:} The penalty forces feasibility but introduces a sharp (non-differentiable) gradient discontinuity at the constraint boundary, requiring smaller step sizes. As $\lambda \to \infty$, the penalised problem approaches the constrained problem but becomes increasingly ill-conditioned.
\end{itemize}
The \emph{augmented Lagrangian method} addresses the ill-conditioning by combining a quadratic penalty with a dual variable update, achieving exact penalisation at finite $\lambda$ while preserving problem conditioning.

\textbf{Why both methods reach the same optimum here:} Since the constraint is inactive at the unconstrained minimum ($x_1^\star \approx 0.582 > 0.5$), the KKT conditions at $x^\star$ require the Lagrange multiplier $\mu^\star = 0$ (complementary slackness: $\mu^\star(0.5 - x_1^\star) = 0$ with $x_1^\star > 0.5$), so the stationarity condition reduces to $\nabla f(x^\star) = 0$. Both PGD and sufficiently strong penalty methods converge to $x^\star \approx (0.582, 2.0)$.

\subsection{Part I: Linear Programme over Box Constraints}

The feasible set is $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} Minimise $f(x) = a^T x = x_1 + 2x_2$ with $a = [1, 2]^T$ over $\mathcal{X}$.

Since $f$ is linear and $\mathcal{X}$ is a box (product of intervals), the minimum over each coordinate is achieved at the lower bound when the corresponding coefficient is positive:
\begin{equation}
  x^\star = \arg\min_{x \in \mathcal{X}} (x_1 + 2x_2) = (0.5,\, -5), \qquad f^\star = 0.5 + 2(-5) = -9.5.
\end{equation}

\begin{theorem}[Fundamental Theorem of Linear Programming]
The minimum of a linear objective over a bounded polytope is attained at a \emph{vertex} (extreme point) of the polytope. For a box constraint $\{x : l_i \leq x_i \leq u_i\}$, the vertices are all $2^d$ corner points, and the optimal vertex is found by independently setting each coordinate to $l_i$ (if $a_i > 0$) or $u_i$ (if $a_i < 0$).
\end{theorem}

\centering
\includegraphics[width=0.85\textwidth]{figures/q6_lp.pdf}
\caption{Q6-I: Feasible region $\mathcal{X}$ (blue rectangle) and linear objective contours $f(x) = x_1 + 2x_2 = c$ (parallel lines). The minimum objective value is achieved at the vertex $(0.5, -5)$ (lower-left corner), where both $x_1$ and $x_2$ are at their lower bounds. This is consistent with the Fundamental Theorem of LP: the level sets move toward decreasing $c$ in the direction $-a = (-1, -2)$, which points towards the lower-left corner.}
\label{fig:q6_lp}
\end{figure}

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient) with Fixed Mixing Parameter}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial feasible point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: move $(1-\beta)$ toward $z_k$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Frank--Wolfe Convergence Rate \cite{frankwolfe1956,jaggi2013}]
For an $L$-smooth convex function $f$ with $D = \mathrm{diam}(\mathcal{X})$ and the standard diminishing step schedule $\gamma_k = 2/(k+2)$ (i.e., $1-\beta_k = \gamma_k$), Frank--Wolfe achieves
\begin{equation}
  f(x_k) - f(x^\star) \leq \frac{2LD^2}{k+2} = O(1/k).
\end{equation}
The $O(1/k)$ rate matches projected gradient descent in order but with a different constant; importantly, Frank--Wolfe avoids projection entirely. The rate cannot be improved to $O(1/k^2)$ for general convex functions, as the zig-zag behaviour near interior optima is a fundamental limitation \cite{jaggi2013}.
\end{theorem}

\begin{remark}
The Frank--Wolfe algorithm maintains \emph{feasibility by construction}: since $x_0 \in \mathcal{X}$, $z_k \in \mathcal{X}$, and $\mathcal{X}$ is convex, every iterate $x_{k+1} = \beta x_k + (1-\beta)z_k$ is a convex combination of feasible points and hence in $\mathcal{X}$. No projection oracle is needed.
\end{remark}

For the box constraint, the LMO decomposes per coordinate in $O(d)$:
\begin{equation}
  (z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0. \end{cases}
\end{equation}
This makes Frank--Wolfe especially attractive when the constraint set has a cheap LMO (e.g., $\ell_1$ ball, nuclear norm ball) but an expensive projection (e.g., semidefinite cone).

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} When $x^\star \in \mathrm{int}(\mathcal{X})$, Frank--Wolfe exhibits a characteristic ``zig-zag'' convergence pattern. The LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ (i.e., corners of the box), and the convex combination gradually averages these jumps. This leads to the $O(1/k)$ rate, which is slower than the accelerated rates achievable with gradient-based methods that can exploit strong convexity near interior optima. The constant $\beta$ schedule introduces an effective step size of $1-\beta$ that does not decay, causing residual oscillation proportional to $(1-\beta) \cdot \mathrm{diam}(\mathcal{X})$.

\textbf{Boundary optimum:} Frank--Wolfe is particularly well-suited to boundary optima because the LP oracle frequently selects the vertex closest to the constrained optimum. When the unconstrained optimum is outside $\mathcal{X}$, the gradient consistently points away from the boundary, causing $z_k$ to reliably select the boundary vertex (here $(0.5, -5)$). The iterate is then steadily pulled toward $(0.5, 0)$. The gap between achieved ($f = 0.364$) and optimal ($f^\star = 0.25$) reflects the finite number of iterations; using the theoretically recommended schedule $\gamma_k = 2/(k+2)$ would guarantee $f(x_{140}) - f^\star \leq \frac{2LD^2}{142}$ convergence.

  \item Standard GD stagnates at $f \approx 3.5$--$3.7$ across all Q1/Q2/Q3 baselines.
\end{itemize}

The Rosenbrock function is the decisive discriminator: the condition number $\kappa \approx 2500$ means that gradient descent requires $O(\kappa) = O(2500)$ iterations for each order of magnitude of error reduction, whereas Nesterov requires $O(\sqrt{\kappa}) = O(50)$ and Newton requires $O(\log\log(1/\epsilon))$ once in the locally quadratic regime.

\textbf{The gradient information hierarchy:} Methods using more information converge faster on smooth functions:
\begin{equation}
  \text{GD} \; < \; \text{Momentum/Adaptive} \; < \; \text{Newton} \qquad \text{(convergence rate on smooth functions)}.
\end{equation}
Derivative-free methods occupy a special niche: they are applicable when gradients are unavailable but pay an $O(d)$ factor in iteration complexity.

\textbf{Constrained optimisation (Q5--Q6):} When the unconstrained optimum lies in the interior of the feasible set, both PGD and strong-penalty methods recover it exactly. Frank--Wolfe is structurally different: it enforces constraints implicitly via the LP oracle, making it especially powerful for structured constraint sets (nuclear norm, $\ell_1$ ball) where projection is expensive but the LP is cheap.

% ============================================================
\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has investigated six families of optimisation algorithms across three benchmark functions of increasing difficulty, arriving at the following conclusions.

\textbf{Adaptive step-size methods (Q1).} Adagrad and Heavy Ball are robust across well-conditioned and moderately nonlinear problems, matching GD's accuracy with minimal additional computation. Polyak's step size is theoretically elegant --- it achieves the fastest convergence on Rosenbrock when $f^\star = 0$ is correctly known --- but is highly sensitive to misspecification of $f^\star$, rendering it impractical when the true optimum is unknown (as is typically the case). RMSprop's exponential forgetting makes it preferable over Adagrad for long-horizon or non-stationary problems.

\textbf{Momentum and stochastic methods (Q2).} Nesterov's acceleration provides the most dramatic improvement on ill-conditioned problems: it reduces the Rosenbrock objective from $3.39$ (GD) to $0.48$ in the same 150 iterations. Adam combines momentum and adaptivity, performing well on all benchmarks, though its per-coordinate scaling is less effective than Nesterov's accumulated momentum along the valley direction for highly ill-conditioned nonlinear problems. Mini-batch SGD demonstrates the fundamental noise--efficiency trade-off: smaller batches update more frequently but with higher gradient variance; when observation noise dominates, the batch size becomes irrelevant to the limiting accuracy.

\textbf{Newton's method (Q3).} The most computationally powerful method tested, Newton achieves quadratic convergence near the optimum and solves the quadratic loss in a single iteration. The need for a regularised Hessian solve ($O(d^3)$ per iteration) limits scalability to high dimensions, but for the $d = 2$ benchmarks studied, Newton's 20-iteration advantage over GD's 80 iterations is substantial. Damping is essential on highly nonlinear problems (Rosenbrock), where the Hessian changes rapidly and the full Newton step overshoots.

\textbf{Derivative-free optimisation (Q4).} For problems where gradients are unavailable, finite differences with a well-chosen $\delta \approx \sqrt{\varepsilon_\mathrm{mach}}$ closely approximates the true gradient with minimal overhead. Nelder--Mead is effective in low dimensions and can navigate curved valleys (such as Rosenbrock) without gradient information. Grid search guarantees global optimality within resolution but is limited to $d \leq 3$ due to its exponential scaling.

\textbf{Constrained optimisation (Q5).} Projected gradient descent is the gold standard when a cheap projection oracle exists: it enforces strict feasibility from iteration 1 and has the same convergence theory as unconstrained GD. Penalty methods offer simplicity but require careful $\lambda$-tuning; too small penalises infeasibility too weakly, while too large requires proportionally smaller step sizes to maintain stability.

\textbf{Frank--Wolfe (Q6).} The Frank--Wolfe algorithm is uniquely suited to problems where the constraint set has a cheap LP oracle but an expensive projection (e.g., trace-norm constrained problems in machine learning). Its $O(1/k)$ convergence is no better than PGD in theory, but it avoids all projection computations and naturally stays feasible. The zig-zag behaviour for interior optima reflects the fundamental limitation of only moving toward extreme points of $\mathcal{X}$.

In summary, the choice of optimisation algorithm should be guided by:
\begin{itemize}[nosep]
  \item \emph{Problem conditioning}: use Newton/quasi-Newton for well-structured ill-conditioned problems; use Nesterov for moderately ill-conditioned ones.
  \item \emph{Gradient availability}: use exact gradients when available; use finite differences or random search otherwise.
  \item \emph{Constraint structure}: use PGD for simple projectable constraints; use Frank--Wolfe for structured LP-oracle constraints; use penalty methods as a last resort.
  \item \emph{Stochastic setting}: use Adam or SGD with momentum; prefer larger mini-batches when gradient variance is the limiting factor and smaller mini-batches when iteration count (wall-clock time) matters.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball (Polyak Momentum)}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex}]
def nelder_mead(loss_fn, x0, step, n_iters,
                alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1])
        f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1])
            f_e = loss_fn(x_e)
            if f_e < f_r: simplex[-1] = x_e; f_vals[-1] = f_e
            else: simplex[-1] = x_r; f_vals[-1] = f_r
        else:
            if f_r < f_vals[-1]:
                x_c = centroid + rho * (x_r - centroid)
                f_c = loss_fn(x_c)
                if f_c <= f_r: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
            else:
                x_c = centroid + rho * (simplex[-1] - centroid)
                f_c = loss_fn(x_c)
                if f_c < f_vals[-1]: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
