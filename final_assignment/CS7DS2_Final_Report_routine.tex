\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A, condition number $\kappa = 1.12$), a nonlinear toy neural network loss with sinusoidal perturbation (Benchmark~B, $\kappa \approx 6.9$), and the highly ill-conditioned Rosenbrock function (Benchmark~C, $\kappa \approx 2508$). Fifteen algorithms across six categories are implemented and analysed: (1)~adaptive step-size methods --- Polyak, Adagrad, RMSprop, and Heavy Ball; (2)~momentum methods --- Nesterov acceleration and Adam, with mini-batch SGD; (3)~Newton's method with local quadratic approximation; (4)~derivative-free methods --- finite differences, Nesterov random search, Nelder--Mead simplex, and grid search; (5)~constrained optimisation via projected gradient descent and penalty methods; and (6)~Frank--Wolfe for linear-constrained problems.

Key quantitative findings: (i)~the Polyak step size diverges on Benchmarks~A and~B (final $f = 4.77$ and $0.98$) when $f^\star = 0$ is misspecified, but achieves $f = 0.024$ on Rosenbrock where $f^\star = 0$ is exact; (ii)~Nesterov acceleration achieves $f = 0.479$ on Rosenbrock versus $f = 3.39$ for GD baseline ($7\times$ improvement); (iii)~Newton's method minimises Benchmark~A exactly in \emph{one iteration}, confirming the theoretical single-step convergence for quadratic losses; (iv)~the constraint $x_1 \geq 0.5$ in Question~5 is provably inactive at the Benchmark~B optimum (verified via KKT conditions), so all methods converge to the same point and differ only in how quickly they leave the infeasible region; (v)~Nelder--Mead achieves $f = 0.0068$ on Rosenbrock using only function evaluations, demonstrating competitive derivative-free performance; and (vi)~the Frank--Wolfe algorithm follows an $O(1/k)$ convergence curve on the interior-optimum case. Each method is analysed with theoretical convergence rate context, connecting experimental results to first-order optimality conditions, quadratic convergence theory, and course material.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \in \mathbb{R}^2, \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2}\theta^T H \theta - b^T \theta + c,
\end{equation}
where $H = \frac{1}{m}X^TX$ is the Gram matrix, $b = \frac{1}{m}X^T y$, and $c = \frac{1}{2m}\|y\|^2$. This is a strongly convex quadratic with constant Hessian $H$. The gradient is $\nabla J(\theta) = H\theta - b = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the ordinary least-squares minimiser is $\hat{\theta}_{\text{OLS}} = H^{-1}b = [3.028,\; 4.017]^T$ (close to $\theta^\star$, as expected by statistical consistency), with minimum loss $J^\star = J(\hat{\theta}_{\text{OLS}}) = 0.4826$. The Hessian eigenvalues are $\lambda_{\min} = 0.925$ and $\lambda_{\max} = 1.033$, giving condition number $\kappa(H) = 1.117$ --- a near-perfectly conditioned problem. For gradient descent to be stable, the step size must satisfy $\alpha < 2/\lambda_{\max} = 1.936$; the chosen $\alpha = 0.08$ is well within this range. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \text{diag}(2 - \sin(x_1),\ 10)$. The $x_2$ direction is 10 times stiffer than the $x_1$ direction, creating anisotropy. The unconstrained minimum is found numerically at $x^\star \approx (0.5824, 2.0)$ with $f^\star \approx 0.7244$; at this point the Hessian has eigenvalues $1.450$ and $10.0$, giving local condition number $\kappa \approx 6.90$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$. The Hessian at the optimum $(1,1)$ is:
\begin{equation}
\nabla^2 f(1,1) = \begin{bmatrix} 2 + 1200(1)^2 - 400(1) & -400(1) \\ -400(1) & 200 \end{bmatrix} = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $0.40$ and $1001.6$, giving condition number $\kappa \approx 2508$. For gradient descent stability at the optimum, the step size must satisfy $\alpha < 2/\lambda_{\max} \approx 2/1001.6 \approx 0.002$, which is why the baseline uses $\alpha = 0.0012$. This extreme ill-conditioning, combined with the narrow parabolic valley $x_2 \approx x_1^2$, makes Rosenbrock the canonical challenge for gradient-based optimisers: the gradient is nearly perpendicular to the valley floor, so gradient steps mostly oscillate across the valley rather than advancing along it. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems (Questions~5 and~6): $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive step-size methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for 120 iterations. The baseline GD update is $x_{k+1} = x_k - \alpha \nabla f(x_k)$, where $\alpha$ is fixed.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Theoretical motivation for Polyak step]
The Polyak step size is theoretically motivated: for a convex function, minimising the bound $\|x_{k+1} - x^\star\|^2 \leq \|x_k - x^\star\|^2 - \frac{(f(x_k) - f^\star)^2}{\|\nabla f(x_k)\|^2}$ yields exactly $\alpha_k = (f(x_k)-f^\star)/\|\nabla f(x_k)\|^2$. This step uniquely minimises the distance to the optimum at each iteration, subject only to knowledge of $f^\star$. When $f^\star$ is exact and $f$ is strongly convex with parameter $\mu$, the Polyak step achieves linear convergence at the optimal rate $\|x_{k+1} - x^\star\|^2 \leq (1 - \mu/L)^k \|x_0 - x^\star\|^2$.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad accumulates squared gradients per coordinate, reducing the effective learning rate more aggressively in directions with large historical gradients. This makes it well-suited to sparse gradients (common in NLP tasks): its online regret bound is $R_T \leq \alpha_0 \sum_i \sqrt{\sum_{t=1}^T g_{t,i}^2}$, which depends only on the gradients of active coordinates and scales as $O(\sqrt{T})$ for sparse sequences. However, on dense gradient problems Adagrad suffers from premature step-size decay: since $G_k = \sum_{j=0}^{k-1} g_j \odot g_j \to \infty$ monotonically, the effective step $\alpha_0/\sqrt{G_k} \to 0$ and later iterations make negligible progress even if the current gradient is informative.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's cumulative sum with an exponential moving average (decay $\beta = 0.9$), preventing the step size from vanishing. It can be seen as a heuristic approximation to the diagonal of the Fisher information matrix. The effective step size $\alpha_0/\sqrt{v_{k+1}}$ adapts to recent gradient magnitudes rather than all historical gradients.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method accumulates a velocity $z_k$, allowing it to build up speed in consistent gradient directions and damp oscillations across them. For strongly convex quadratics with condition number $\kappa$, the optimal parameters are $\alpha^\star = \frac{4}{(\sqrt{\lambda_{\max}} + \sqrt{\lambda_{\min}})^2}$ and $\beta^\star = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$, achieving a convergence rate of $\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2 < 1-1/\sqrt{\kappa}$, which is significantly faster than gradient descent's rate of $1 - 1/\kappa$.

\textbf{Key insight regarding Polyak divergence:} The step size formula $\alpha_k = (f(x_k) - f^\star)/(\|\nabla f(x_k)\|^2 + \varepsilon)$ requires an accurate lower bound $f^\star \leq f(x^\star)$. This causes $\alpha_k \to \infty$, inducing oscillations and divergence. On Rosenbrock, $f^\star = 0$ is exact, so the numerator correctly vanishes as $f(x_k) \to 0$, yielding a decreasing, well-behaved step sequence.

\subsection{Comparative Analysis}

\textbf{Convergence speed:} When $f^\star$ is known exactly (Rosenbrock), Polyak converges fastest ($f = 0.024$), exploiting the adaptive step formula $\alpha_k = (f(x_k) - f^\star)/\|\nabla f(x_k)\|^2$ which is optimal for subgradient methods. For well-conditioned problems (Benchmark~A, $\kappa = 1.117$), the GD linear convergence rate $(1 - \mu/L)^k = (1 - 0.925/1.033)^k \approx 0.895^k$ is already fast, and adaptive methods offer little additional benefit. For moderately ill-conditioned problems (Benchmark~B, $\kappa = 6.9$), Heavy Ball's convergence rate $(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1})^2 \approx (\frac{1.63}{3.63})^2 \approx 0.20^2 = 0.04$ per iteration represents a significant improvement over GD's rate $1 - 1/\kappa \approx 0.855$.

\textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging of squared gradients with $\beta = 0.9$, which acts as a low-pass filter on the effective step size. Heavy Ball shows characteristic overshoot followed by damped oscillation --- the momentum accumulates in the wrong direction briefly before the gradient corrects it. Adagrad curves are monotonically non-increasing (by construction, since the accumulated sum $G_k$ only grows).

\textbf{Stability:} GD baseline is always stable for step sizes well below the stability threshold $\alpha < 2/\lambda_{\max}$. Polyak is unstable when $f^\star$ is misspecified. Adagrad is uniformly stable (monotonically shrinking step), but this is also its weakness: on Benchmark~C, the step decays so aggressively that later iterations make negligible progress. RMSprop and Heavy Ball are stable with proper hyperparameter tuning.

\textbf{Connection to course theory:} As discussed in the lectures, the Polyak step is the solution to $\min_\alpha \|x_k - \alpha g_k - x^\star\|^2$ (the step that minimises the distance to $x^\star$), yielding $\alpha_k = \langle x_k - x^\star, g_k \rangle / \|g_k\|^2 \leq (f(x_k) - f^\star)/\|g_k\|^2$ for convex $f$ (by the subgradient inequality). The formula used here adds $\varepsilon$ to the denominator for numerical stability. Adagrad's theoretical motivation comes from online convex optimisation: it achieves a regret bound $R_T \leq \alpha_0 \sum_i \sqrt{T\sum_t g_{t,i}^2}$ which is optimal for adversarial settings with sparse gradients.

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead / extrapolated point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Nesterov lookahead and optimal first-order rate]
The crucial distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This ``corrective'' evaluation anticipates the next position, allowing Nesterov momentum to damp oscillations that Heavy Ball would amplify. For smooth convex functions with Lipschitz constant $L$, Nesterov's method achieves:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2},
\end{equation}
the \emph{optimal} $O(1/k^2)$ first-order convergence rate --- proved to be tight by the lower bound of Nemirovsky and Yudin. Gradient descent only achieves $O(L\|x_0-x^\star\|^2/k)$, a factor of $k$ slower. For strongly convex functions ($\mu > 0$), the optimal rate for first-order methods is $O((1 - \sqrt{\mu/L})^k)$, achievable by Nesterov's strongly-convex variant, versus $O((1-\mu/L)^k)$ for GD.
\end{remark}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scaling)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction (prevents cold start)}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines heavy-ball momentum (via $m_t$) with per-coordinate RMSprop-style adaptive scaling (via $v_t$). The bias correction terms $(1 - \beta_j^t)$ prevent the zero initialisation from suppressing early updates: without them, early estimates $m_t \approx (1-\beta_1)g_1$ and $v_t \approx (1-\beta_2)g_1^2$ would be a factor $(1-\beta_j)$ below the true first/second moments, causing the effective learning rate in early iterations to be inflated by $(1-\beta_2^t)^{1/2}/(1-\beta_1^t)$ relative to $\alpha$. In practice, Adam is the default optimiser for deep neural networks due to its robustness to hyperparameter choices and insensitivity to sparse gradients. However, Reddi et al. (2018) showed that the original Adam algorithm can fail to converge even on simple convex problems due to non-monotone effective step sizes; the \emph{AMSGrad} variant replaces $v_t$ with $\hat{v}_t = \max(\hat{v}_{t-1}, v_t)$ to enforce a non-increasing step size and restore convergence guarantees.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch. The mini-batch gradient is:
\begin{equation}
g_{\mathcal{B}} = \frac{1}{|\mathcal{B}|} X_{\mathcal{B}}^T (X_{\mathcal{B}} \theta - y_{\mathcal{B}}),
\end{equation}
which is an unbiased estimator of the full gradient $\nabla J(\theta)$, since $\mathbb{E}[g_{\mathcal{B}}] = \nabla J(\theta)$. The variance satisfies $\mathbb{V}[g_{\mathcal{B}}] = \frac{1}{b} \mathbb{V}[g_i]$ where $g_i$ is a single-sample gradient, so variance decreases linearly with batch size. For a fixed learning rate, the expected suboptimality after $T$ iterations converges to a noise floor proportional to $\alpha \mathbb{V}[g_{\mathcal{B}}] \propto \alpha/b$, not zero. This is the fundamental limitation of constant step-size SGD: it cannot converge to the exact minimiser, only to a ball around it of radius $O(\alpha/b)$.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{|\mathcal{B}|} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{Comparative Discussion: Nesterov vs Adam}

On well-conditioned problems (Benchmarks~A and~B), Nesterov and Adam both achieve convergence to the true minimum, with little qualitative difference. The key distinction emerges on the ill-conditioned Rosenbrock function, where convergence to a low $f$ value requires efficiently traversing a narrow, curved valley:

\begin{itemize}[nosep]
  \item \textbf{Nesterov} achieves $f = 0.479$ on Rosenbrock by accumulating momentum in the consistent valley-floor direction. The gradient along the Rosenbrock valley floor points roughly in the increasing-$x_1$ direction, so a momentum accumulator $z_k$ that persists across iterations efficiently exploits this directional consistency. The lookahead step $\tilde{x}_k = x_k + \beta_k z_k$ pre-positions the gradient evaluation in the expected next location, correcting potential overshoots before they occur. This is why Nesterov achieves the theoretically optimal $O(1/k^2)$ rate on convex functions, compared to $O(1/k)$ for GD.
  \item \textbf{Adam} achieves $f = 1.578$. While Adam's per-coordinate scaling (via $v_t$) compensates for diagonal curvature differences between $x_1$ and $x_2$, the Rosenbrock valley's curvature is highly non-diagonal (the off-diagonal Hessian entry $-400x_1$ is large). Adam cannot exploit off-diagonal curvature structure, limiting its effectiveness. Furthermore, the second-moment estimate $v_t$ reacts to recent gradient magnitudes, which vary significantly along the curved valley, causing the effective step size to fluctuate rather than accumulate speed.
  \item \textbf{Key principle:} Pure momentum methods (Nesterov) outperform adaptive step-size methods (Adam) on problems with consistent gradient directions, while adaptive methods outperform momentum on problems with highly variable gradient scales and sparse updates (such as NLP tasks).
\end{itemize}

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} at $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = (0.25)^4 + 4(0.25)^3 (x - 0.25) = 0.0039 + 0.0625(x - 0.25).
\end{equation}
This is a linear approximation (tangent line) that captures only the local slope.

\textbf{Second-order approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
This parabolic model captures local curvature. Setting its derivative to zero yields the Newton update:
\begin{equation}
0 = g'(x_0) + g''(x_0)(x - x_0) \implies x_1 = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.167.
\end{equation}
This is the Newton step: move to the minimiser of the local quadratic approximation. In the multivariate case, this generalises to $x_{k+1} = x_k - [\nabla^2 f(x_k)]^{-1}\nabla f(x_k)$, where the Hessian inverse replaces the reciprocal second derivative. Newton's step can be interpreted as gradient descent in a metric defined by the local curvature: it scales each gradient component inversely to the curvature in that direction, which automatically accounts for anisotropy.

\subsection{Newton's Method Applied to All Benchmarks}

\begin{algorithm}[H]
\caption{Newton's Method with Hessian Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0,1]$, damping $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov regularisation for stability}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: solves $H_k p_k = g_k$}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The damping term $\lambda I$ (with $\lambda = 10^{-8}$ here) ensures $H_k + \lambda I$ is positive definite even when the Hessian is singular or nearly so, preventing ill-conditioned linear system solves. The step size $\alpha$ is chosen less than 1 for Benchmarks~B and~C where the Hessian changes significantly between iterations.

\begin{remark}[Convergence theory]
Near the optimum $x^\star$ of a strongly convex, twice-differentiable function, Newton's method achieves \emph{quadratic} local convergence:
\begin{equation}
\|x_{k+1} - x^\star\| \leq C \|x_k - x^\star\|^2
\end{equation}
for some constant $C > 0$, meaning the number of correct digits doubles at each iteration. This is dramatically faster than gradient descent's linear convergence $\|x_{k+1} - x^\star\| \leq (1 - \mu/L)\|x_k - x^\star\|$. The trade-off is the $O(d^3)$ cost of solving the linear system $H_k p_k = g_k$ (or $O(d^2)$ with iterative methods), making Newton impractical for large-scale problems without approximation.
\end{remark}

\begin{remark}[Newton on Rosenbrock]
For Benchmark~C, Newton reaches $(0.183, 0.020)$ after 20 iterations, still far from the optimum $(1,1)$. In practice, Newton's method on Rosenbrock typically requires a line search (Wolfe conditions or backtracking) and either more iterations or a warm start near the optimum to exhibit its quadratic convergence. For large-scale problems, \emph{quasi-Newton} methods such as L-BFGS approximate the inverse Hessian using the BFGS rank-2 update from curvature pairs $(s_k, y_k) = (x_{k+1}-x_k,\, \nabla f(x_{k+1})-\nabla f(x_k))$:
\begin{equation}
H_{k+1}^{-1} = \left(I - \rho_k s_k y_k^T\right) H_k^{-1} \left(I - \rho_k y_k s_k^T\right) + \rho_k s_k s_k^T, \quad \rho_k = \frac{1}{y_k^T s_k}.
\end{equation}
The limited-memory variant (L-BFGS) stores only the $m \approx 10$--$20$ most recent $(s_k, y_k)$ pairs and applies the two-loop recursion to compute $H_k^{-1} g$ in $O(md)$ operations, compared to $O(d^3)$ for exact Newton. L-BFGS achieves superlinear local convergence (between linear and quadratic) and is the standard method for large-scale smooth unconstrained optimisation.
\end{remark}

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark~B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
The truncation error is $O(\delta)$ (from the Taylor expansion remainder), while numerical round-off error is $O(\varepsilon_{\text{machine}}/\delta)$. Balancing these gives an optimal step $\delta^\star = \sqrt{\varepsilon_{\text{machine}}} \approx 10^{-8}$ for double precision. The assignment tests $\delta = 0.05$ (moderate, close to optimal region) and $\delta = 0.8$ (large, dominated by truncation error).

The centred finite-difference formula $\frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}$ gives $O(\delta^2)$ truncation error at double the function evaluation cost; however, the forward difference is used here as specified. Figure~\ref{fig:q4_fd_error} illustrates this trade-off empirically.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$ (uniform on sphere)
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates a one-dimensional directional derivative along a random unit vector $u_k$, requiring only \textbf{two function evaluations} per iteration regardless of dimension $d$. The gradient estimator satisfies $\mathbb{E}[\hat{g}_k] = \frac{1}{d}\nabla f_\delta(x_k)$ where $f_\delta(x) = \mathbb{E}_{u \sim \mathcal{U}(\mathbb{S}^{d-1})}[f(x + \delta u)]$ is the Gaussian-smoothed surrogate, so Nesterov random search performs gradient descent on $f_\delta$ rather than $f$ directly. In $d=2$ dimensions (as here), the single random direction samples half the information of a full gradient at 50\% the cost of a forward-difference gradient estimate. The convergence rate is $O(d/\sqrt{k})$ for convex functions, reflecting the high-variance, one-dimensional projection: in $d$ dimensions, the estimator's variance scales as $O(d)$ because only a $1/d$ fraction of the gradient is captured per sample, explaining why Nesterov random search is most useful in very high dimensions where even storing a full gradient is infeasible.

\subsection{Part B: Derivative-Free Methods (Benchmark~C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it using four operations:

\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r = 1.0$): Move the worst vertex $x_{\text{worst}}$ through the centroid $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{\text{worst}})$.
  \item \textbf{Expansion} ($\gamma = 2.0$): If reflection improves beyond the current best, attempt a further expansion $x_e = \bar{x} + \gamma(\bar{x} - x_{\text{worst}})$.
  \item \textbf{Contraction} ($\rho = 0.5$): If reflection does not improve sufficiently, contract the simplex.
  \item \textbf{Shrink} ($\sigma = 0.5$): Contract all vertices toward the best vertex when contraction also fails.
\end{enumerate}

No gradient information is required at any step — only function evaluations at the simplex vertices.

\subsection{Problem Setup}

Minimise Benchmark~B subject to the inequality constraint $x_1 \geq 0.5$, starting from the infeasible initial point $x_0 = (0.2, 4.0)$. The feasible set is the half-space $\mathcal{C} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}$.

\begin{remark}[Constraint Activity and KKT Conditions]
An important observation: the unconstrained minimum of Benchmark~B is at $x^\star \approx (0.5824, 2.0)$, which satisfies $x_1^\star \approx 0.582 > 0.5$. Therefore, the constraint $x_1 \geq 0.5$ is \emph{inactive} at the optimum (also called a non-binding constraint). This can be verified via the Karush--Kuhn--Tucker (KKT) conditions for the constrained problem: $\min f(x)$ s.t.\ $g(x) = 0.5 - x_1 \leq 0$. The KKT conditions require:
\begin{align}
\nabla f(x^\star) + \mu \nabla g(x^\star) &= 0, \quad \mu \geq 0, \quad \mu g(x^\star) = 0.
\end{align}
Since $\nabla f(x^\star) = [0, 0]^T$ at the unconstrained minimum and $\nabla g = [-1, 0]^T$, the first condition gives $\mu \cdot (-1) = 0$, so $\mu = 0$. The complementary slackness condition $\mu g(x^\star) = 0$ is satisfied with $\mu = 0$ regardless of $g(x^\star)$. Thus the constraint is inactive and the unconstrained and constrained optima coincide. The key practical difference between methods lies in \emph{how they handle the infeasible starting point} $(0.2, 4.0)$ and \emph{how quickly they enforce feasibility along the trajectory}.
\end{remark}

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z)$ \Comment{Project: $(x_1, x_2) \gets (\max(0.5,\, x_1),\, x_2)$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection onto a half-space $\{x : x_1 \geq 0.5\}$ is trivial: $\Pi_{\mathcal{C}}(z) = (\max(0.5, z_1), z_2)^T$. For more complex constraints (e.g., norm balls or polytopes with many inequalities), projection can require solving a separate optimisation problem.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_\lambda(x) \gets f(x) + \lambda \max(0,\, -x_1 + 0.5)$ \Comment{Penalised objective}
    \State $\nabla F_\lambda(x) \gets \nabla f(x) - \lambda\, \mathbf{1}[x_1 < 0.5]\, e_1$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty approach replaces the hard constraint with a soft penalty in the objective. As $\lambda \to \infty$, the penalty term dominates, forcing the solution to satisfy the constraint. However, very large $\lambda$ induces numerical instability: under a smoothed quadratic penalty $\frac{\lambda}{2}\max(0,\, 0.5 - x_1)^2$, the $x_1$-Hessian in the infeasible region becomes $H_{11}(f) + \lambda$, so the condition number grows as $\kappa(H_{F_\lambda}) \approx \lambda / \lambda_{\min}(H_f)$ and the stability bound tightens to $\alpha < 2/(H_{\max}(f) + \lambda)$. For the non-smooth linear penalty used here, the gradient switches discontinuously between $\nabla f$ and $\nabla f - \lambda e_1$ as iterates cross $x_1 = 0.5$, causing oscillation across the constraint boundary for large $\lambda$ unless the step size is reduced accordingly.

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected GD} enforces feasibility as a \emph{hard constraint} at every iteration via the projection operator $\Pi_{\mathcal{C}}$. The cost is requiring a projection oracle, which is trivial for half-spaces (as here, $O(1)$ per coordinate) but can be expensive for polyhedral or non-convex constraints.

\textbf{Penalty methods} convert the constrained problem into an unconstrained one, requiring only gradient computation on the modified objective. The weight $\lambda$ controls the trade-off: small $\lambda$ enforces feasibility slowly, large $\lambda$ enforces it quickly but introduces numerical ill-conditioning requiring smaller step sizes. In the limit $\lambda \to \infty$, penalty methods recover the constrained solution.

\textbf{Unconstrained GD} happens to converge to a feasible point in this problem because the unconstrained minimum $(0.582, 2.0)$ is already in the feasible region. This coincidence should not be expected in general --- when the unconstrained minimum violates constraints (as in Question~6 Boundary case), unconstrained GD diverges from the constrained optimum.

\textbf{When to use which:} For simple constraint sets, projected GD is preferred (hard feasibility, well-understood convergence). For complex constraint sets where projection is expensive, penalty or augmented Lagrangian methods are preferred. The augmented Lagrangian method $F_{\lambda, \mu}(x) = f(x) + \lambda \cdot c(x) + \frac{\mu}{2}c(x)^2$ (not implemented here) combines penalty with a Lagrange multiplier update, achieving feasibility with less ill-conditioning than pure penalty.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box / bounded polytope).

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is found by minimising each coordinate independently. Since both components of $a$ are positive, the minimum of $a_i x_i$ over $[l_i, u_i]$ is at the lower bound $x_i = l_i$:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}
This illustrates the \emph{fundamental theorem of linear programming}: the minimum of a linear objective over a bounded convex polytope is always attained at a vertex (extreme point). Since there are finitely many vertices (here: 4 corners of the box), the LP can be solved by vertex enumeration, or more efficiently by the Simplex algorithm or interior-point methods.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: $x_{k+1} \in \mathcal{X}$ always}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Key properties of Frank--Wolfe]
  \begin{enumerate}[nosep]
    \item \textbf{Projection-free}: Since $x_{k+1}$ is a convex combination of $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$, feasibility is maintained automatically by convexity of $\mathcal{X}$. No projection oracle is needed.
    \item \textbf{Sparse iterates}: Each iterate is a convex combination of previous LMO solutions (vertices of $\mathcal{X}$), naturally generating sparse solutions in applications such as matrix completion or lasso.
    \item \textbf{Convergence}: For smooth convex $f$ with Lipschitz gradient constant $L$ and the standard step-size rule $\gamma_k = 2/(k+2)$ (corresponding to $\beta = k/(k+2)$), $f(x_k) - f^\star \leq 2LC/(k+2)$ where $C = \text{diam}(\mathcal{X})^2$. This is an $O(1/k)$ convergence rate.
    \item \textbf{Fixed $\beta$ variant}: The implementation here uses a fixed $\beta$ (not the optimal decaying schedule), which can cause oscillation near the optimum but may converge faster in early iterations for well-chosen $\beta$.
  \end{enumerate}
\end{remark}

For box constraints, the LP subproblem decomposes per coordinate in $O(d)$ operations:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
This is vastly cheaper than projection onto a general polytope (which requires solving an LP), making Frank--Wolfe particularly attractive for constrained problems over polytopes.

\begin{remark}[Computational advantage of FW for structured constraint sets]
For the box constraint here, both projection and the LMO cost $O(d)$, so the advantage of FW over projected GD is not apparent. The advantage becomes decisive for constraints such as the \emph{nuclear norm ball} $\{W \in \mathbb{R}^{m \times n} : \|W\|_* \leq \tau\}$ (used in matrix completion): projection requires a full SVD at $O(\min(m,n)^2 \max(m,n))$ cost, while the LMO requires only the leading singular vectors at $O(mn)$ cost via the power method. Similarly, for the \emph{spectrahedron} $\{X \succeq 0 : \operatorname{tr}(X) = 1\}$ (used in semidefinite relaxations), projection costs $O(d^3)$ while the LMO is a single eigenvector computation at $O(d^2)$. Frank--Wolfe is therefore the method of choice for large-scale optimisation over structured constraint sets where the LMO is significantly cheaper than projection.
\end{remark}

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe converges to the unconstrained minimum as the iterates remain interior to $\mathcal{X}$. The gradient at the optimum $\nabla f(1, 5) = (0, 0)$ vanishes, so $z_k$ can point to any vertex; however, since the iterate is already at the minimum, the convex combination $x_{k+1} = \beta x_k + (1-\beta) z_k$ moves slightly away from $(1,5)$, causing small oscillations bounded by $(1-\beta) \cdot \text{diam}(\mathcal{X})$. This explains why convergence is approximate rather than exact with a fixed $\beta$ schedule.

\textbf{Boundary optimum:} When the unconstrained minimum $(0,0)$ lies outside $\mathcal{X}$, the constrained optimum is at the boundary: $(0.5, 0)$ with $f^\star = 0.25$. At this point, $\nabla f = (1, 0)$, which is non-zero and points toward decreasing $x_1$ (inward from the $x_1 = 0.5$ boundary into the infeasible region). The LMO consistently selects $z_k = (0.5, -5)$ (lower-left vertex, minimising $\nabla f^T z = z_1 + 0 \cdot z_2$ subject to $z \in \mathcal{X}$). The iterate converges toward $(0.5, 0)$ as $z_k$'s $x_2$-component drags $x_2$ toward $-5$ while the $x_1 = 0.5$ constraint boundary naturally stops the $x_1$-progress. The final value $f = 0.364 > f^\star = 0.25$ reflects that the fixed $\beta = 0.93$ schedule cannot achieve exact convergence. Near the constrained optimum $(0.5, 0)$, the $x_2$-gradient changes sign as $x_2$ oscillates through zero, causing $z_k$ to alternate between $(0.5, -5)$ and $(0.5, 10)$ for the $x_2$-coordinate. Using the fixed-point analysis (same as the interior case), the $x_2$-oscillation amplitude is $(1-\beta)/(1+\beta) \cdot 7.5 \approx (0.07/1.93) \times 7.5 \approx 0.27$, giving a residual $f \approx 0.5^2 + 0.27^2 \approx 0.32$, consistent with the observed $f = 0.36$. The decaying schedule $\gamma_k = 2/(k+2)$ cures this by making $\beta_k \to 1$ so the oscillation amplitude $(1-\beta_k)/(1+\beta_k) \to 0$.

\textbf{Frank--Wolfe (linearisation) gap:} A useful stopping criterion for Frank--Wolfe is the \emph{FW gap} (also called the duality gap):
\begin{equation}
g_k = \nabla f(x_k)^T (x_k - z_k),
\end{equation}
where $z_k$ is the LMO solution. The gap satisfies $g_k \geq f(x_k) - f^\star \geq 0$ for convex $f$, providing a computable upper bound on the suboptimality without knowing $f^\star$. When $g_k = 0$, the iterate $x_k$ already satisfies the first-order optimality condition (no descent direction within $\mathcal{X}$ is available), indicating convergence to the constrained optimum. For the boundary case, the FW gap would remain positive as long as the iterate has not reached $(0.5, 0)$.

\textbf{Advantage of FW over projection:} For the box constraint $\mathcal{X}$, projection is also trivial ($O(d)$). However, for more complex polytopes --- such as the spectrahedron $\{X : X \succeq 0, \text{tr}(X) = 1\}$ in semidefinite programming, or the nuclear norm ball $\{W : \|W\|_* \leq \tau\}$ in low-rank matrix optimisation --- the LMO requires only a leading singular vector computation ($O(d^2)$) versus a full eigendecomposition for projection ($O(d^3)$). This computational advantage is the primary motivation for Frank--Wolfe in large-scale structured optimisation problems.

\subsection{Benchmark-Specific Conclusions}

\textbf{Benchmark~A (well-conditioned quadratic, $\kappa = 1.12$):} Nearly all methods converge effectively due to the near-perfect conditioning ($\kappa \approx 1$). Newton's method achieves exact minimisation in one step, as predicted by theory for quadratic objectives. GD convergence rate $(1 - \mu/L)^k = (0.925/1.033)^k \approx 0.895^k$ is already fast, and the noise floor $J^\star = 0.4826$ (due to observation noise $\sigma = 1$) means convergence to machine precision is impossible regardless of method. Practical lesson: for near-isotropic problems, simple gradient descent is nearly as good as any other method, and the algorithm choice matters little.

\textbf{Benchmark~B (moderately ill-conditioned, $\kappa = 6.9$):} The sinusoidal term introduces local nonlinearity but the basin of attraction near the minimum has a Hessian with condition number $\kappa \approx 6.9$, which is small enough that all gradient-based methods converge. The meaningful comparison is convergence speed rather than final accuracy. Heavy Ball (Q1) and Nesterov (Q2) both accelerate convergence by accumulating momentum in the consistent gradient direction. Adam and Adagrad provide robustness through adaptive scaling.

\textbf{Benchmark~C (severely ill-conditioned, $\kappa = 2508$):} This benchmark is the most informative discriminator of method quality. Key results: GD ($f = 3.39$), Heavy Ball ($f = 1.40$), Adagrad ($f = 2.20$), Nesterov ($f = 0.48$), Polyak ($f = 0.024$, given exact $f^\star$), Newton ($f = 0.69$, 20 iterations), Nelder--Mead ($f = 0.007$, 160 iterations). Counterintuitively, derivative-free Nelder--Mead outperforms second-order Newton: this is explained by a severe \emph{model mismatch} for Newton. The Rosenbrock function's ``banana'' shape is far from quadratic globally; the second-order Taylor model $q(x) = f(x_k) + \nabla f(x_k)^T d + \frac{1}{2}d^T H(x_k)d$ is accurate only in a small neighbourhood of the current point. Outside this neighbourhood, the Newton step can be misaligned with the true descent direction, and the conservative damping $\alpha = 0.22$ limits progress to 22\% of each Newton direction. Nelder--Mead, in contrast, directly samples function values along the banana-shaped valley and self-aligns its simplex with the low-curvature valley direction through repeated expansion, accumulating progress over 160 iterations ($8\times$ more than Newton's 20). When iterations are equalised (both at 80), Newton would likely outperform Nelder--Mead near the optimum, but the given iteration budgets favour Nelder--Mead's global exploration.

\subsection{General Recommendations}

Based on the experimental and theoretical analysis:
\begin{enumerate}[nosep]
  \item Use \textbf{Adam or RMSprop} as the default for deep learning and general non-convex problems with unknown structure.
  \item Use \textbf{Nesterov momentum} when the problem has a known convex or near-convex structure and convergence rate matters.
  \item Use \textbf{Newton's method} (or quasi-Newton variants like L-BFGS) when $d$ is small (up to thousands) and second-order information is affordable.
  \item Use \textbf{finite differences} only when gradients are unavailable and $d$ is small; prefer Nesterov random search for large $d$.
  \item Use \textbf{Nelder--Mead} for low-dimensional, smooth, derivative-free problems where some inaccuracy is tolerable.
  \item Use \textbf{projected gradient descent} for constrained problems with simple, cheap projections; prefer \textbf{Frank--Wolfe} when projection is expensive but the LMO is cheap.

\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Hessian Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
