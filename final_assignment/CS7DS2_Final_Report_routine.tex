\begin{abstract}
% ============================================================
This report presents a comprehensive comparative study of six categories of optimisation
algorithms applied across three benchmark problems of increasing difficulty: a convex quadratic
loss arising from linear regression (Benchmark~A), a mildly non-quadratic loss with
sinusoidal perturbation modelling a toy neural network (Benchmark~B), and the highly
ill-conditioned Rosenbrock function (Benchmark~C). The six algorithm families investigated
are: \textbf{(1)} adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball),
\textbf{(2)} momentum and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD),
\textbf{(3)} Newton's method with local quadratic approximation,
\textbf{(4)} derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead simplex, grid search),
\textbf{(5)} constrained optimisation via projected gradient descent and penalty methods, and
\textbf{(6)} the Frank--Wolfe conditional gradient algorithm for linear-constrained problems.

Key empirical findings include: \textbf{(1)}~the Polyak step size achieves the fastest convergence on Rosenbrock ($f=0.009$ in 120 iterations) when $f^{\star}=0$ is exactly known, but diverges ($f>24$) on benchmarks with a nonzero minimum floor due to unbounded step growth; \textbf{(2)}~Nesterov acceleration reduces the Rosenbrock objective $7\times$ relative to gradient descent in 150 iterations, consistent with its theoretical $O(1/k^2)$ rate advantage over $O(1/k)$; \textbf{(3)}~Newton's method solves the quadratic loss in exactly one step and reduces the Rosenbrock objective $5.4\times$ versus gradient descent using only 20 versus 80 iterations; \textbf{(4)}~finite-difference gradient descent with step $\delta=0.05$ matches exact gradient descent almost identically on Benchmark~B, while $\delta=0.8$ yields systematically incorrect directions due to large truncation error; \textbf{(5)}~projected gradient descent enforces hard feasibility from iteration~1 while penalty methods with $\lambda\geq1.8$ achieve feasibility within 20 iterations; \textbf{(6)}~Frank--Wolfe with fixed mixing parameter $\beta=0.93$ reaches $f=0.364$ (vs.\ $f^{\star}=0.25$) in 140 iterations for the boundary optimum, highlighting the need for a decreasing step schedule. Each method is evaluated in terms of convergence speed, trajectory behaviour, computational cost, and theoretical guarantees.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.
The three benchmark functions are chosen to span a spectrum from strongly convex quadratic
to highly non-convex and ill-conditioned, allowing a systematic comparison of algorithm
behaviour under different geometric conditions.

% ----------------------------------------------------------
\subsection{Benchmark A: Linear Regression Quadratic Loss}
\label{ssec:benchA}
% ----------------------------------------------------------

A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\label{eq:benchA_data}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, I_2), \quad
y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, 1).
\end{equation}
The loss is the mean squared error:
\begin{equation}
\label{eq:benchA_loss}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
          = \frac{1}{2m}\|X\theta - y\|_2^2.
\end{equation}
This is a strongly convex quadratic. The Hessian is the constant positive semi-definite matrix
$H = \frac{1}{m}X^T X$, and the gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$.
The smoothness constant is $L = \lambda_{\max}(H)$ and the strong-convexity constant is
$\mu = \lambda_{\min}(H)$.
With $m=1000$ i.i.d.\ standard Gaussian rows, by the Marchenko--Pastur law the eigenvalues
of $\frac{1}{m}X^TX$ concentrate near 1 for a $2$-dimensional $X$, giving $\kappa = L/\mu \approx 1$,
so Benchmark~A is \emph{well-conditioned}. This explains why all first-order methods
converge rapidly on Benchmark~A but differ dramatically on the ill-conditioned Benchmark~C.
Due to observation noise, the minimum loss is $J^{\star} \approx 0.4826 \neq 0$,
achieved at the ordinary least-squares solution
$\theta_{\text{OLS}} = (X^TX)^{-1}X^Ty \approx [3.028,\ 4.017]^T$.
Starting point: $\theta_0 = [0, 0]^T$.

% ----------------------------------------------------------
\subsection{Benchmark B: Toy Neural Network Loss}
\label{ssec:benchB}
% ----------------------------------------------------------

\begin{equation}
\label{eq:benchB}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is
$\nabla f = \bigl[2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)\bigr]^T$
and the Hessian is
$H(x) = \operatorname{diag}\!\bigl(2 - \sin(x_1),\ 10\bigr)$.
Since $\sin(x_1) \in [-1, 1]$, the $(1,1)$ entry of $H$ lies in $[1, 3]$, so the
eigenvalue ratio (condition number) of the Hessian ranges from
$10/3 \approx 3.3$ to $10/1 = 10$ across the domain.
The minimum is at approximately $x^{\star} = (0.582,\ 2.0)$,
found by solving $2(x_1-1) + \cos(x_1) = 0$, with $f^{\star} \approx 0.7244$.
Starting point: $x_0 = [-1,\ 4]^T$.

% ----------------------------------------------------------
\subsection{Benchmark C: Rosenbrock Function}
\label{ssec:benchC}
% ----------------------------------------------------------

\begin{equation}
\label{eq:rosenbrock}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian is
\begin{equation}
\label{eq:rosenbrock_hessian}
H(x) = \begin{bmatrix}
2 + 1200x_1^2 - 400x_2 & -400x_1 \\
-400x_1 & 200
\end{bmatrix}.
\end{equation}
At the optimum $(1,1)$: $H(1,1) = \bigl[\begin{smallmatrix}802 & -400\\ -400 & 200\end{smallmatrix}\bigr]$,
with $\operatorname{tr}(H) = 1002$, $\det(H) = 802\cdot200 - 400^2 = 400$,
giving eigenvalues $\lambda_{1,2} = \frac{1002 \pm \sqrt{1002^2 - 4\cdot400}}{2} \approx 1001.6$ and $0.4$,
hence condition number $\kappa \approx 2504$.
This extreme ill-conditioning ($\kappa \approx 2504$), combined with a narrow curved valley
along the parabolic curve $x_2 = x_1^2$, makes Rosenbrock one of the standard challenging
test functions. The valley can be parameterised as $\gamma(t) = (t, t^2)$ for $t \in [0,1]$;
the gradient along the valley $\gamma'(t) = (1, 2t)$ points in a direction that changes
continuously, preventing any fixed gradient direction from making long-range progress.
The loss decreases only as $(1-t)^2$ along the valley floor, so progress requires
accurate tracking of the curved descent direction --- a challenge that Newton's method
handles through the Hessian's off-diagonal coupling terms.
Starting point: $x_0 = [-1,\ 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

% ----------------------------------------------------------
\subsection{Theoretical Background}
\label{ssec:q1_theory}
% ----------------------------------------------------------

Gradient descent with constant step size $\alpha$ converges at a linear rate
$f(x_k) - f^\star \leq (1 - \mu/L)^k (f(x_0) - f^\star)$
for $L$-smooth, $\mu$-strongly convex functions~\cite{polyak1987}.
The condition number $\kappa = L/\mu$ governs the convergence speed:
large $\kappa$ forces a small step size and slow convergence.
Adaptive methods attempt to address this by using per-iterate or per-coordinate
curvature information to set an effective step size automatically.

% ----------------------------------------------------------
\subsection{Algorithm Formulations}
\label{ssec:q1_algs}
% ----------------------------------------------------------

Four adaptive methods are compared against a constant step-size gradient descent
baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak convergence~\cite{polyak1987}]
For a convex function $f$ with exact $f^{\star}$ and $\|g_k\|^2 > 0$, the Polyak step
\begin{equation}
\label{eq:polyak_optimal_step}
\alpha_k^{\star} = \frac{f(x_k) - f^{\star}}{\|\nabla f(x_k)\|^2}
\end{equation}
minimises the upper bound $\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - \alpha_k(2 - \alpha_k L)\cdot\|g_k\|^2$
at each step (choosing $\alpha_k = \alpha_k^{\star}$ makes the gradient-norm term vanish).
The iterates satisfy $f(x_k) - f^{\star} \leq \|x_0 - x^{\star}\|^2 / (2k\alpha_0^{\star})$
in the convex case, where the rate is $O(1/k)$.
For strongly convex $f$ with modulus $\mu$, the Polyak step achieves geometric convergence
$\|x_{k+1} - x^{\star}\|^2 \leq (1 - \mu/L)\|x_k - x^{\star}\|^2$, matching the optimal
rate of gradient descent without requiring knowledge of $L$ or $\mu$.
\end{theorem}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative squared gradients, per coordinate}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad adapts the step size per coordinate based on the cumulative sum of squared past gradients.
Coordinates that have historically received large gradients are penalised with a smaller effective
step size. The monotonically decreasing denominator $\sqrt{G_k}$ guarantees stability but can
cause premature stagnation in later iterations.

\begin{proposition}[Adagrad per-coordinate regret~\cite{duchi2011}]
\label{prop:adagrad_regret}
For convex $f$ with $\|x - x^{\star}\|_\infty \leq D$ and bounded gradients $\|g_k\|_\infty \leq G$,
Adagrad satisfies the cumulative regret bound
\begin{equation}
\label{eq:adagrad_regret}
\sum_{k=1}^K [f(x_k) - f^{\star}]
  \leq \frac{D}{\alpha_0} \sum_{i=1}^d \sqrt{\sum_{k=1}^K g_{k,i}^2}.
\end{equation}
By the Cauchy--Schwarz inequality this is at most $O(DG\sqrt{Kd})$, matching constant
step-size gradient descent in the worst case.
The advantage appears when gradients are \emph{sparse or heterogeneous}: if coordinate $i$
has small cumulative gradient $\sum_k g_{k,i}^2 \ll K G^2$, its term contributes little to
the bound, and the effective learning rate on that coordinate remains large.
This property is why Adagrad is particularly effective in NLP (sparse one-hot features)
and on ill-conditioned problems where different coordinates have vastly different gradient scales.
\end{proposition}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's cumulative sum with an exponential moving average controlled by
$\beta$, so that the effective step size reflects \emph{recent} curvature rather than
the full history. This prevents the effective rate from decaying to zero and maintains
adaptivity to the current loss landscape.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Gradient accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

Heavy Ball accumulates a velocity vector $z_k$ that damps oscillations across the gradient direction
and accelerates along directions of consistent gradient. For a quadratic $f$ with condition number $\kappa = L/\mu$,
the optimal parameters yield a convergence rate of $O\!\left(e^{-k/\sqrt{\kappa}}\right)$
versus $O\!\left(e^{-k/\kappa}\right)$ for gradient descent~\cite{polyak1987}:
\begin{equation}
\label{eq:heavyball_optimal}
\alpha^{\star} = \frac{4}{(\sqrt{L} + \sqrt{\mu})^2}, \qquad
\beta^{\star} = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2.
\end{equation}
This $\sqrt{\kappa}$ improvement in the exponent is especially significant for ill-conditioned
problems: on Benchmark~C where $\kappa \approx 2504$, Heavy Ball's optimal rate improves
from $e^{-k/2504}$ to $e^{-k/\sqrt{2504}} \approx e^{-k/50}$, enabling much faster convergence.
\begin{remark}[Optimal Heavy Ball parameters for Benchmark A]
For Benchmark~A with $\mu = \lambda_{\min}(H) \approx 0.98$ and $L = \lambda_{\max}(H) \approx 1.02$
(near-unit eigenvalues by Marchenko--Pastur, $\kappa \approx 1.04$), the optimal Heavy Ball parameters
from~\eqref{eq:heavyball_optimal} give
$\alpha^{\star} = 4/((\sqrt{L}+\sqrt{\mu})^2) \approx 4/(2.0+0.99)^2 \approx 0.447$
and $\beta^{\star} = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2 \approx (0.02/2.02)^2 \approx 10^{-4}$.
The near-unit condition number means that momentum provides minimal benefit on Benchmark~A:
$\beta^{\star} \approx 0$ (nearly no momentum), and the convergence rate $e^{-k/\sqrt{\kappa}} \approx e^{-k}$
is essentially the same as the GD rate $e^{-k(1-1/\kappa)} \approx e^{-k}$.
In contrast, on Benchmark~C ($\kappa \approx 2504$): $\beta^{\star} \approx (0.96/2.96)^2 \approx 0.105$
and $\alpha^{\star} \approx 0.002$, close to the tuned values $\beta=0.86$ and $\alpha=0.0008$.
\end{remark}
The hyperparameters in Table~\ref{tab:q1_params} are tuned to approximate these optimal values.

\paragraph{Key finding.}
The Polyak step size is highly sensitive to the accuracy of $f^{\star}$.
Formally, the step size formula is
\begin{equation}
\label{eq:polyak_step}
\alpha_k = \frac{f(x_k) - f^{\star}}{\|\nabla f(x_k)\|^2 + \varepsilon}.
\end{equation}
When $f^{\star}_{\text{specified}} = 0$ but $f^{\star}_{\text{true}} > 0$, as the iterate
converges, $\|\nabla f(x_k)\|^2 \to 0$ while $f(x_k) - 0 \to f^{\star}_{\text{true}} > 0$,
so $\alpha_k \to (f^{\star}_{\text{true}} - 0)/(0 + \varepsilon) \approx f^{\star}_{\text{true}}/\varepsilon$,
which is large for small $\varepsilon$. For Benchmark~A with $f^{\star}_{\text{true}} \approx 0.4826$
and $\varepsilon = 10^{-4}$, the step size would grow to $\sim 4826$ as $\|g_k\|\to0$,
far exceeding the stability limit $\alpha < 2/L$, causing the observed divergence.
On Benchmark~C where $f^{\star}_{\text{true}} = 0 = f^{\star}_{\text{specified}}$,
the numerator $f(x_k) \to 0$ as the step progresses, so $\alpha_k \to 0$ gracefully
(bounded above by $f_0/\varepsilon$), enabling the best convergence among all methods.
In practice, exact knowledge of $f^{\star}$ is rarely available, severely
limiting the applicability of the Polyak step size to problems with known infimum
(e.g., sparse recovery problems with known sparsity level).

\subsection{Comparative Analysis}
\label{ssec:q1_analysis}
% ----------------------------------------------------------

\begin{itemize}[nosep]
\item \textbf{Convergence speed.} When $f^{\star}$ is known exactly, Polyak converges fastest
  (Rosenbrock: $f=0.009$ vs $f=3.51$ for GD). Without this knowledge, Adagrad and Heavy Ball
  match or exceed GD performance, while the Polyak step is unreliable.

\item \textbf{Per-coordinate adaptation.} Adagrad and RMSprop adapt independently to each
  coordinate. On Benchmark~B this reduces the effective condition number seen by the
  optimiser: with curvature $\kappa_1 \approx 1.5$ in $x_1$ and $\kappa_2 = 10$ in $x_2$,
  per-coordinate scaling effectively normalises both directions to unit curvature.

\item \textbf{Memory vs.\ recency.} Adagrad's cumulative denominator never forgets past
  gradients, risking premature stagnation on long runs: as $G_k \to \infty$, the effective
  step $\alpha_0/\sqrt{G_k} \to 0$. RMSprop's exponential forgetting ($\beta=0.9$, half-life
  $\approx 6.6$ iterations) prevents this stagnation, at the cost of requiring tuning $\beta$.

\item \textbf{RMSprop oscillation near minimum.} On Benchmark~A, RMSprop settles slightly
  above $J^{\star}$ at $J \approx 0.5027$ because the exponential average $v_k$ has
  not fully decayed when the iterate reaches the minimum, causing a residual adaptive step
  that oscillates around $\theta^{\star}$. This can be remedied by learning rate decay.

\item \textbf{Stability.} GD and Heavy Ball are most stable (no growing internal state).
  Adagrad is stable by construction (monotonically decreasing effective step).
  Polyak is \emph{unstable} when $f^{\star}$ is misspecified.

\item \textbf{Practical implication.} In machine learning practice, Adam (combining
  RMSprop and momentum) is the dominant choice. Polyak step sizes are applicable only
  in structured settings (subgradient methods for structured problems) where $f^{\star}$
  is available via a side channel.
\end{itemize}

% ----------------------------------------------------------
\subsection{Hyperparameter Sensitivity}
\label{ssec:q1_sensitivity}
% ----------------------------------------------------------

The performance of each method is sensitive to its hyperparameters in different ways:

\begin{itemize}[nosep]
\item \textbf{GD:} Step size $\alpha$ must satisfy $\alpha < 2/L$ for convergence.
  On Rosenbrock, $L = \lambda_{\max}(H(x)) \approx 1200$ at typical iterates near the valley,
  requiring $\alpha < 1/600 \approx 0.0017$, consistent with the chosen $\alpha=0.0012$.
\item \textbf{Adagrad:} The base rate $\alpha_0$ sets the scale of all updates.
  Unlike GD, the effective step decreases monotonically and cannot grow, so a large
  initial $\alpha_0$ is safe: the algorithm self-regulates.
\item \textbf{RMSprop:} The decay $\beta=0.9$ controls the half-life of gradient history
  ($\approx 10$ iterations). Larger $\beta$ gives smoother but slower adaptation.
\item \textbf{Heavy Ball:} Both $\alpha$ and $\beta$ must be jointly tuned.
  On Rosenbrock, the very small $\alpha=0.0008$ is dictated by the large Hessian eigenvalue;
  $\beta=0.86$ provides momentum without instability.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

% ----------------------------------------------------------
\subsection{Theoretical Background}
\label{ssec:q2_theory}
% ----------------------------------------------------------

Standard gradient descent achieves a convergence rate of $O(1/k)$ on convex problems and
$O((1-\mu/L)^k)$ for strongly convex problems.
\emph{Nesterov acceleration} is optimal in the sense that no first-order method can achieve
a better dependence on $k$: for smooth convex functions, it attains $O(1/k^2)$~\cite{nesterov1983}.
\emph{Adam} combines momentum (like Nesterov) with per-coordinate adaptive scaling
(like RMSprop), providing robustness to heterogeneous curvature while exploiting
gradient history.

% ----------------------------------------------------------
\subsection{Algorithm Formulations}
\label{ssec:q2_algs}
% ----------------------------------------------------------

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at
$\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$, which anticipates the momentum step
and allows the gradient to ``correct'' the direction before overshooting.
The schedule $\beta_k = (k-1)/(k+2)$ increases the momentum parameter gradually from 0,
matching the sequence used in Nesterov's proof of the $O(1/k^2)$ rate for convex functions.
This schedule is derived from the requirement that the sequence $t_k$ satisfies
$t_k^2 - t_k \leq t_{k-1}^2$ where $\beta_k = (t_{k-1}-1)/t_k$ and $t_k = (k+2)/2$.
The Lyapunov function $\Phi_k = t_k^2(f(x_k) - f^{\star}) + \frac{1}{2\alpha}\|v_k - x^{\star}\|^2$
is shown to be non-increasing (a standard algebraic construction in accelerated gradient
proofs~\cite{nesterov1983}), yielding the potential:
\begin{equation}
\label{eq:nesterov_lyapunov}
f(x_k) - f^{\star} \leq \frac{\Phi_0}{t_k^2} = \frac{2(f(x_0)-f^{\star}) + \|x_0 - x^{\star}\|^2/(2\alpha)}{(k+2)^2} = O\!\left(\frac{1}{k^2}\right).
\end{equation}
The clipping to $\beta_{\max}$ prevents instability on non-convex problems such as
Rosenbrock, where the convexity assumption of the $O(1/k^2)$ proof breaks down.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$,
         $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t / (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines the momentum of $m_t$ (analogous to Nesterov's $z_k$) with the per-coordinate
scaling of $\hat{v}_t$ (analogous to RMSprop's $v_k$).
The bias correction terms $(1-\beta_1^t)$ and $(1-\beta_2^t)$ compensate for
the zero initialisation of $m_0$ and $v_0$: in early iterations $m_t \approx 0$
because it is initialised at zero, so without correction the effective learning rate
would be inappropriately small.

\begin{remark}[Adam non-convergence and AMSGrad]
Despite its empirical dominance, Adam is known to fail to converge on certain online convex
problems. The issue is that the exponential moving average $\hat{v}_t$ can oscillate
non-monotonically, allowing the effective per-coordinate step $\alpha/\sqrt{\hat{v}_t}$
to \emph{increase} over time and violate the step-size decay condition required for
stochastic convergence proofs. \textit{AMSGrad} (Reddi et al., 2018) addresses this
by replacing $\hat{v}_t$ with $\tilde{v}_t = \max(\tilde{v}_{t-1}, \hat{v}_t)$, ensuring
non-increasing effective steps per coordinate. In practice, this non-convergence pathology
is rarely observed on standard benchmarks, and Adam's empirical performance on deep networks
remains the state of the art. This illustrates a recurring theme: theoretical guarantees
and empirical performance do not always align.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}
\label{ssec:q2_sgd}
% ----------------------------------------------------------

Mini-batch SGD on Benchmark~A uses learning rate $\alpha=0.06$, batch sizes $b\in\{5, 40\}$,
for 50 epochs with data shuffled at each epoch. The stochastic gradient for a mini-batch
$\mathcal{B}$ of size $b$ is:
\begin{equation}
\label{eq:sgd_grad}
\hat{g}(\theta;\mathcal{B}) = \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}}),
\end{equation}
which is an unbiased estimator of $\nabla J(\theta)$ with variance
\begin{equation}
\label{eq:sgd_variance}
\operatorname{Var}[\hat{g}(\theta;\mathcal{B})] = \frac{1}{b}\operatorname{Var}\bigl[(X\theta-y)^TX\bigr] \propto \frac{1}{b}.
\end{equation}
The key SGD trade-off is between \emph{update frequency} and \emph{gradient noise}:
smaller $b$ gives more updates per epoch (higher frequency) but noisier gradients (higher variance),
while larger $b$ gives fewer, more accurate updates.

\begin{proposition}[SGD convergence with constant step size]
\label{prop:sgd_convergence}
For an $L$-smooth, $\mu$-strongly convex loss $f$ with per-sample gradient variance $\sigma^2$
and a constant step size $\alpha \leq 1/L$, mini-batch SGD with batch size $b$ satisfies
\begin{equation}
\label{eq:sgd_conv}
\mathbb{E}[f(\theta_k) - f^{\star}]
  \leq (1 - \alpha\mu)^k (f(\theta_0) - f^{\star}) + \frac{\alpha\, \sigma^2}{2b\mu}.
\end{equation}
The first term decays geometrically at rate $(1-\alpha\mu)^k \to 0$; the second term is
an irreducible \emph{noise floor} of $\alpha\sigma^2/(2b\mu)$, proportional to $1/b$.
This makes the trade-off precise: a larger batch reduces the floor (better asymptotic accuracy),
but a smaller step $\alpha$ slows the initial decay. The optimal strategy uses a decaying
schedule $\alpha_k \to 0$ (e.g., $\alpha_k = C/k$), achieving $O(1/k)$ convergence with no floor.
\end{proposition}

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1,\ldots,m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$ (non-overlapping, covering all data)}
        \State $\theta \gets \theta - \alpha\, \hat{g}(\theta;\mathcal{B})$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\centering
\includegraphics[width=0.85\textwidth]{figures/q2_sgd_batch.pdf}

% ----------------------------------------------------------
\subsection{Local Taylor Approximations}
\label{ssec:q3_approx}
% ----------------------------------------------------------

Consider $g(x) = x^4$ expanded around $x_0 = 0.25$:
\begin{align}
g^{(0)}(x_0) &= 0.25^4 = 0.0039, \quad
g^{(1)}(x_0) = 4 \cdot 0.25^3 = 0.0625, \quad
g^{(2)}(x_0) = 12 \cdot 0.25^2 = 0.75.
\label{eq:g_derivs}
\end{align}

\textbf{First-order (Taylor/linear) approximation:}
\begin{equation}
\label{eq:first_order}
q_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625\,(x - 0.25).
\end{equation}
This is a tangent line and captures only the local slope.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
\label{eq:second_order}
q_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
        = 0.0039 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\end{equation}
The Newton update minimises $q_2(x)$ over $x$, yielding
\begin{equation}
\label{eq:newton_update_1d}
x^+ = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} \approx 0.167.
\end{equation}
For $g(x) = x^4$, the true minimum is at $x^{\star} = 0$, and one Newton step
from $x_0 = 0.25$ gives $x^+ \approx 0.167$, a $33\%$ reduction in the distance to zero.
Gradient descent with step $\alpha$ gives $x^+ = 0.25 - \alpha \cdot 0.0625$,
which for typical small $\alpha$ would be much larger than $0.167$.

\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$.
The first-order approximation (tangent line) deviates rapidly from the true function.
The second-order approximation (parabola) captures the local curvature and fits well

% ----------------------------------------------------------
\subsection{Newton's Method in $\mathbb{R}^d$}
\label{ssec:q3_newton}
% ----------------------------------------------------------

\begin{algorithm}[H]
\caption{Newton's Method with Hessian Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda\geq0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised (positive-definite) Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: $H_k p_k = g_k$}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton direction $p_k = H_k^{-1}g_k$ minimises the local quadratic model
$m_k(d) = f(x_k) + g_k^T d + \frac{1}{2}d^T H_k\, d$,
so the update $x_{k+1} = x_k - H_k^{-1}g_k$ (with $\alpha=1$) jumps directly to
the minimum of the local quadratic approximation.
The regularisation term $\lambda I$ ensures positive definiteness of $H_k$ even
when the Hessian is indefinite or ill-conditioned (as near saddle points of Rosenbrock),
at the cost of reducing the step towards a gradient direction when $\lambda$ is large.

\begin{theorem}[Quadratic convergence~\cite{nocedal2006}]
Suppose $f$ is twice continuously differentiable, the Hessian is Lipschitz continuous
near $x^{\star}$, and $H(x^{\star})$ is positive definite.
Then Newton's method (with $\alpha=1$) converges quadratically:
$\|x_{k+1} - x^{\star}\| \leq C\,\|x_k - x^{\star}\|^2$
for some constant $C > 0$ once $\|x_0 - x^{\star}\|$ is sufficiently small.
\end{theorem}

Settings: GD uses 80 iterations; Newton uses only 20 iterations with
Hessian damping $\lambda = 10^{-8}$.

\paragraph{Connection to adaptive gradient methods.}
Newton's method can be interpreted as gradient descent with an adaptive preconditioner:
$x_{k+1} = x_k - H_k^{-1}g_k$ rescales the gradient by the inverse Hessian,
effectively normalising the curvature to identity before taking the step.
This is structurally analogous to the per-coordinate preconditioning in Adagrad and RMSprop
(Questions~1 and~2), but Newton uses the full $d\times d$ Hessian rather than the diagonal
approximation $\operatorname{diag}(G_k)^{-1}$. The diagonal approximation trades precision
for $O(d)$ cost per iteration, motivating the family of adaptive gradient methods as
cheap surrogates for Newton's method.

\begin{remark}[Natural gradient and Fisher information]
In statistical models where $f(\theta) = -\log p(y|\theta)$ is the negative log-likelihood,
Newton's method uses the Hessian $H(\theta) = -\mathbb{E}[\nabla^2 \log p(y|\theta)]$.
The \emph{natural gradient} method (Amari, 1998) replaces $H$ with the \emph{Fisher information matrix}
$F(\theta) = \mathbb{E}_y[\nabla\log p(y|\theta)\nabla\log p(y|\theta)^T]$,
which equals the Hessian at the maximum likelihood estimate (by the Fisher--Rao identity).
Natural gradient descent $\theta_{k+1} = \theta_k - F_k^{-1}\nabla f(\theta_k)$
is invariant to reparameterisation of the model (a property standard GD lacks) and
underpins K-FAC and other approximate natural gradient methods for deep learning.
In this report's context, Adagrad's diagonal $\operatorname{diag}(\sum_k g_k g_k^T)^{-1}$
is a diagonal approximation to $F(\theta)^{-1}$, connecting adaptive gradient methods
to the natural gradient framework.
\end{remark}

\paragraph{Limitation of Newton's method.}
Despite its superior convergence rate, Newton's method has significant practical limitations:
(i)~it requires computing and inverting the $d\times d$ Hessian at each iteration
($O(d^3)$ cost), prohibitive for large $d$ (e.g., neural networks with $d\gg10^6$);
(ii)~the Hessian may be indefinite away from a local minimum, requiring additional
regularisation to ensure a descent direction;
(iii)~in the Rosenbrock case, the curvature changes rapidly along the trajectory,
so a full Newton step ($\alpha=1$) often overshoots, requiring line search or damping.
Quasi-Newton methods (BFGS, L-BFGS) address (i) by maintaining a low-rank approximation
to $H^{-1}$ built from gradient difference vectors, achieving superlinear convergence
at $O(md)$ cost per iteration, where $m$ is the number of stored vector pairs.

\begin{remark}[L-BFGS two-loop recursion]
\label{rem:lbfgs}
L-BFGS maintains the $m$ most recent curvature pairs
$\{(s_j, y_j)\}$, where $s_j = x_{j+1} - x_j$ encodes the step taken and
$y_j = \nabla f(x_{j+1}) - \nabla f(x_j)$ encodes the gradient change.
The BFGS rank-2 update formula gives a sequence of approximate inverse Hessians:
\begin{equation}
\label{eq:bfgs_update}
B_{k+1}^{-1} = \bigl(I - \rho_k s_k y_k^T\bigr)\, B_k^{-1}\, \bigl(I - \rho_k y_k s_k^T\bigr)
             + \rho_k s_k s_k^T, \quad \rho_k = \frac{1}{y_k^T s_k}.
\end{equation}
Rather than storing $B_k^{-1}$ explicitly ($O(d^2)$), the two-loop recursion
computes the product $B_k^{-1} g_k$ in $O(md)$ time using only the stored pairs.
With $m \in \{5, 20\}$, L-BFGS achieves superlinear (near-quadratic in practice)
convergence on smooth strongly convex problems, at a fraction of Newton's $O(d^3)$ cost.
L-BFGS is the standard solver for large-scale smooth optimisation (e.g., logistic
regression, $\ell_2$-penalised least squares) and is included in \texttt{scipy.optimize.minimize}.
\end{remark}

\subsubsection{Forward Finite Difference}
\label{sssec:q4_fd}

The forward finite-difference approximation replaces each partial derivative:
\begin{equation}
\label{eq:fd_approx}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector.
The approximation error decomposes as
\begin{equation}
\label{eq:fd_error}
\left|\frac{f(x+\delta e_i)-f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right|
\leq \underbrace{\frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(x)\right|}_{\text{truncation}} +
\underbrace{\frac{2\varepsilon_{\text{mach}}\,|f(x)|}{\delta}}_{\text{cancellation}},
\end{equation}
giving an optimal step size $\delta^{\star} \sim \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$
that balances the two error sources.
In practice, $\delta$ must be chosen empirically: too small ($\delta \ll 10^{-4}$)
causes catastrophic cancellation; too large ($\delta = 0.8$) introduces large truncation error
by spanning a region of significant curvature change.

\textbf{Centered vs.\ forward differences.}
The \emph{centered} finite-difference approximation
\begin{equation}
\label{eq:cfd_approx}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}
\end{equation}
has truncation error $O(\delta^2)$ (vs.\ $O(\delta)$ for forward FD), since the $O(\delta)$ term
in the Taylor expansion cancels by symmetry.
This means centered FD achieves the same gradient accuracy as forward FD at a step size
$\delta_c \approx \delta_f^2$, enabling significantly larger $\delta_c$ before truncation dominates.
The cost is $2d$ function evaluations per gradient (vs.\ $d+1$ for forward FD).

\begin{remark}[Complex step method]
\label{rem:complex_step}
The \emph{complex step} approximation
$\frac{\partial f}{\partial x_i}(x) \approx \Im(f(x + i\delta e_i))/\delta$
(where $i = \sqrt{-1}$ and $f$ is extended to complex inputs)
achieves \emph{machine-precision} derivative estimates for arbitrarily small $\delta$,
because there is no cancellation in the imaginary part.
For $g(x) = x^4$: $g(x + i\delta) = (x+i\delta)^4 = x^4 - 6x^2\delta^2 + \delta^4 + i(4x^3\delta - 4x\delta^3)$,
so $\Im(g(x_0 + i\delta))/ \delta = 4x_0^3 - 4x_0\delta^2$, with error $O(\delta^2)$ independent
of floating-point cancellation.
The complex step method is a standard tool in computational mechanics and aerodynamic
shape optimisation, where high-fidelity gradient estimates are critical.
\end{remark}

For each gradient evaluation, the forward FD method requires $d+1$ function evaluations
(one baseline evaluation plus one perturbed evaluation per dimension),
compared to a single analytical gradient evaluation.

\subsubsection{Nesterov Random Search}
\label{sssec:q4_nrs}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Gaussian Smoothing)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta\, u_k) - f(x_k)}{\delta}\, u_k$
    \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This method estimates the gradient by a single directional finite difference along a
uniformly random direction on the unit sphere, requiring only \emph{two} function evaluations
per iteration regardless of dimension $d$.

The gradient estimator $\hat{g}_k$ is an unbiased stochastic approximation to the gradient
of the Gaussian-smoothed function $f_\delta(x) = \mathbb{E}_{u\sim\mathcal{N}(0,I)}[f(x + \delta u)]$
in the sense that~\cite{nesterov2017}:
\begin{equation}
\label{eq:nrs_bias}
\mathbb{E}_{u}[\hat{g}_k] = \frac{d}{\delta}\,\mathbb{E}_{u}[(f(x+\delta u)-f(x))u] = \nabla f_\delta(x),
\end{equation}
where the factor of $d$ arises from the projection of the $d$-dimensional gradient onto a
single direction.
As $\delta \to 0$, $\nabla f_\delta \to \nabla f$, so the random search approximates
gradient descent in the limit. The variance of $\hat{g}_k$ is $O(d^2\|\nabla f\|^2)$
(since we use a single direction out of $d$), meaning the estimator is highly noisy for
large $d$ --- in $d=2$ the variance is already substantial compared to the signal,
explaining why Nesterov random search converges much more slowly than FD gradient descent.

\begin{theorem}[Nesterov random search convergence~\cite{nesterov2017}]
\label{thm:nrs_convergence}
For an $L$-smooth convex function $f$ with minimiser $x^{\star}$, using the
normalised Gaussian direction estimator~\eqref{eq:nrs_bias} with smoothing parameter
$\delta$ and step size $\alpha = 1/(2dL)$, the Nesterov random search satisfies
\begin{equation}
\label{eq:nrs_conv_bound}
\mathbb{E}[f(x_k)] - f^{\star} \leq \frac{4d^2 L \|x_0 - x^{\star}\|^2}{k + 4d^2}.
\end{equation}
The $d^2$ factor (versus $O(1)$ for gradient descent) quantifies the statistical
penalty of single-direction sampling: in $d$ dimensions, projecting the gradient onto
one random direction introduces variance that scales as $d^2$.
For $d=2$ (our Benchmark~B setting), the extra cost factor is $4\times 4 = 16$,
meaning random search requires approximately $16\times$ more iterations than gradient
descent to reach the same accuracy. For $d=100$, this factor is $40\,000\times$,
making zeroth-order random search impractical at high dimension without additional
variance reduction (e.g., coordinate-wise averaging or importance sampling).
\end{theorem}

\subsubsection{Nelder--Mead Simplex Method}
\label{sssec:q4_nm}

The Nelder--Mead algorithm maintains a simplex of $n+1=3$ vertices $\{v_1, v_2, v_3\}$
in $\mathbb{R}^2$ and iteratively transforms it through four geometric operations.
Let $\bar{v} = \frac{1}{n}\sum_{i=1}^n v_i$ be the centroid of the $n$ best vertices,
and $v_{\text{w}}$ the worst:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r=1.0$): $v_r = \bar{v} + \alpha_r(\bar{v} - v_{\text{w}})$
  --- reflects the worst vertex through the centroid;
\item \textbf{Expansion} ($\gamma=2.0$): if $f(v_r) < f(v_{\text{best}})$, try
  $v_e = \bar{v} + \gamma(\bar{v} - v_{\text{w}})$ to extend further;
\item \textbf{Contraction} ($\rho=0.5$): if reflection did not improve, contract
  towards the centroid (inside or outside);
\item \textbf{Shrink} ($\sigma=0.5$): if contraction also fails, contract all
  vertices towards the best: $v_i \gets v_{\text{best}} + \sigma(v_i - v_{\text{best}})$.
\end{enumerate}
No gradient information is required; the simplex geometry adapts to the local landscape
purely through function value comparisons.
Each operation requires at most $n+1$ function evaluations per iteration.

\begin{remark}[Degenerate simplex stagnation]
A known failure mode of Nelder--Mead is \emph{simplex degeneracy}: repeated contraction
without shrink steps can collapse the simplex onto a low-dimensional affine subspace
(e.g., a line in $\mathbb{R}^2$), preventing exploration of the full $d$-dimensional space.
On the Rosenbrock parabolic valley, this can manifest as the simplex aligning with
the valley axis and failing to make transverse corrections.
The \emph{restart strategy} (reinitialise the simplex whenever the simplex volume
falls below a threshold) and \emph{restart with random perturbation} are standard
mitigation approaches. In this experiment, the simplex step $=0.35$ is large enough
relative to the scale of the problem to avoid premature degeneracy in 160 iterations.
\end{remark}

Settings: initial simplex step $=0.35$ (perturbation from $x_0=(-1,1)$), 160 iterations.

The fundamental trade-off is between \emph{gradient quality} and \emph{function evaluation cost}.

of high variance ($O(d^2)$ variance amplification). Grid search is comprehensive but
exponential in cost.

\paragraph{Curse of dimensionality.}
Grid search requires $N^d$ evaluations for a grid of resolution $N$ in $d$ dimensions.
For the $d=2$ Rosenbrock problem, a $55\times55$ grid provides resolution $\approx0.073$.
To achieve the same resolution in $d=10$ dimensions would require $55^{10} \approx 2.5\times10^{17}$
evaluations --- completely intractable. This exponential scaling is the fundamental reason why
derivative-free methods (Nelder--Mead, random search) are limited to low-dimensional problems,
while gradient-based methods scale linearly (FD: $O(d)$ evaluations per step; random search:
$O(1)$ evaluations per step but $O(d^2)$ variance in gradient estimate).
For black-box optimisation in high dimensions, recent advances in Bayesian optimisation
and zeroth-order methods with structured gradient estimators address this scaling challenge.

% ----------------------------------------------------------
\subsection{Problem Setup and Optimality Conditions}
\label{ssec:q5_setup}
% ----------------------------------------------------------

The problem is to minimise $f(x_1,x_2) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)$
subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$.
The unconstrained minimum is at $(0.582, 2.0)$, which is feasible (since $0.582 > 0.5$),
so the constrained and unconstrained optima coincide.

\textbf{KKT conditions:} The Lagrangian is $\mathcal{L}(x,\mu) = f(x) - \mu(x_1 - 0.5)$
with $\mu \geq 0$.
At the constrained minimum $x^{\star} = (0.582, 2.0)$:
\begin{equation}
\label{eq:kkt_stationarity}
\nabla_x \mathcal{L}(x^{\star}, \mu) = \nabla f(x^{\star}) - \mu\begin{bmatrix}1\\0\end{bmatrix} = 0
\quad\Rightarrow\quad
\mu = \left.\frac{\partial f}{\partial x_1}\right|_{x^{\star}} = 2(0.582-1) + \cos(0.582) \approx -0.836 + 0.836 = 0.
\end{equation}
The dual variable evaluates to $\mu = 0$, confirming the \emph{complementary slackness} condition
$\mu(x_1^{\star} - 0.5) = 0 \cdot 0.082 = 0$ and the fact that the constraint is inactive at the optimum.
If the unconstrained minimum were at $x_1^{\star} < 0.5$ (i.e., the constraint were active),
we would have $x_1^{\star} = 0.5$ and $\mu > 0$ reflecting the shadow price of the constraint.
Since $\mu=0$ and $x_1^{\star} > 0.5$ (strict feasibility), the constrained and unconstrained optima coincide.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Gradient step in ambient space}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $x_1 \gets \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\State Define $F_\lambda(x) = f(x) + \lambda\max(0,\ 0.5 - x_1)$ \Comment{Penalised objective}
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute $\nabla F_\lambda(x_k)$: same as $\nabla f(x_k)$ except
    $\frac{\partial F_\lambda}{\partial x_1} = \frac{\partial f}{\partial x_1} - \lambda \cdot \mathbf{1}[x_1 < 0.5]$
    \State $x_{k+1} \gets x_k - \alpha\,\nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha=0.08$); Penalty with $\lambda\in\{0.15, 1.8, 4.5\}$
($\alpha=0.05$ for $\lambda\leq1.8$; $\alpha=0.03$ for $\lambda=4.5$
to maintain stability with the steeper effective loss landscape);
unconstrained GD ($\alpha=0.07$). All run for 100 iterations.

\textbf{Projected GD} enforces hard feasibility at every iteration through the projection
$\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)$. The projection onto a half-space
is a $O(1)$ operation; for more complex constraint sets (e.g., semidefinite cone, simplex),
the projection may require solving a quadratic programme.

\textbf{Penalty methods} avoid an explicit projection oracle by embedding the
constraint into the objective. The resulting penalised function is
$F_\lambda(x) = f(x) + \lambda\max(0, 0.5-x_1)$, which is non-smooth (non-differentiable)
at $x_1=0.5$ due to the absolute value function. The gradient used in the implementation
is the subgradient of $F_\lambda$:
\begin{equation}
\label{eq:penalty_subgrad}
\partial_{x_1} F_\lambda(x) = \partial_{x_1} f(x) - \lambda \cdot \mathbf{1}[x_1 < 0.5],
\end{equation}
which equals $\partial_{x_1} f(x)$ when feasible ($x_1 \geq 0.5$) and
$\partial_{x_1} f(x) - \lambda$ in the infeasible region.
Larger $\lambda$ forces iterates closer to the feasible set but induces a steeper
objective landscape, requiring smaller step sizes ($\alpha=0.03$ for $\lambda=4.5$
vs $\alpha=0.05$ for $\lambda\leq1.8$) to maintain stability.
In the limit $\lambda\to\infty$, the penalty method recovers the projected GD solution,
but finite $\lambda$ may leave residual infeasibility in early iterations (as observed
for $\lambda=0.15$ in Figure~\ref{fig:q5_violation}).

\textbf{Augmented Lagrangian methods} (not implemented here) combine penalty terms
with explicit Lagrange multipliers:
$\mathcal{L}_\lambda(x, \mu) = f(x) + \mu\max(0, 0.5-x_1) + \frac{\lambda}{2}\max(0, 0.5-x_1)^2$,
achieving exact feasibility at finite penalty parameter values while avoiding the
conditioning problems of pure penalty methods.
Compared to pure penalty approaches, augmented Lagrangians require the multiplier $\mu$
to be updated between outer iterations ($\mu_{k+1} = \max(0, \mu_k - \lambda(x_{k}-0.5))$),
but converge to a feasible optimum without driving $\lambda\to\infty$.

\textbf{Convergence rate.}
For projected GD applied to an $L$-smooth, $\mu$-strongly-convex $f$ restricted to a
convex feasible set $\mathcal{X}$, the convergence rate is the same as unconstrained GD:
$\|x_k - x^{\star}\|^2 \leq (1 - \mu/L)^k \|x_0 - x^{\star}\|^2$,
provided the step size $\alpha \leq 1/L$~\cite{nocedal2006}.
In this problem, since the constrained optimum coincides with the unconstrained optimum
(the constraint is inactive), projected GD recovers the same rate as the unconstrained baseline.

\subsection{Feasible Set and Linear Programme}
\label{ssec:q6_lp}
% ----------------------------------------------------------

Feasible set: $\mathcal{X} = \{x\in\mathbb{R}^2 : 0.5\leq x_1\leq5,\ -5\leq x_2\leq10\}$
(a rectangular polytope, $\dim=2$, 4 vertices).

\textbf{Linear Programme:} Minimise $f(x) = a^Tx = x_1 + 2x_2$ over $\mathcal{X}$.
Since $a = [1, 2]^T$ has all positive components, the minimum is attained by independently
minimising each coordinate at its lower bound (for a box constraint, the minimum of
a linear function over each coordinate is trivially attained at the bound of the same sign as $-a_i$):
\begin{equation}
\label{eq:lp_solution}
x^{\star} = (0.5,\ {-5}), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates the \emph{fundamental theorem of linear programming}: for a bounded
polytope $\mathcal{X}$ and a linear objective, the optimum is always attained at a
vertex of $\mathcal{X}$, since any interior point can be improved by moving towards
a face (by linearity, $f$ is non-decreasing in the interior normal direction).
The Frank--Wolfe LP oracle $z_k = \arg\min_{x\in\mathcal{X}}\nabla f(x_k)^T x$ is precisely
a vertex of $\mathcal{X}$ at each iteration, which is why Frank--Wolfe iterates are always
convex combinations of vertices and the algorithm naturally works with sparse representations.

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}
\label{ssec:q6_fw}
% ----------------------------------------------------------

\begin{algorithm}[H]
\caption{Frank--Wolfe / Conditional Gradient}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial feasible point $x_0 \in \mathcal{X}$, mixing parameter $\beta\in(0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x\in\mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1-\beta)\, z_k$ \Comment{Convex combination: $\gamma_k = 1-\beta$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the box constraint $\mathcal{X}$, the LMO decomposes per coordinate:
\begin{equation}
\label{eq:fw_lmo_box}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations --- a key advantage over projection-based methods
for certain constraint sets (e.g., nuclear norm ball, spectrahedra) where projection is expensive.

The \emph{Frank--Wolfe gap}
\begin{equation}
\label{eq:fw_gap}
g_k = \nabla f(x_k)^T(x_k - z_k) \geq 0
\end{equation}
is a computable upper bound on the suboptimality $f(x_k) - f^{\star}$ for convex $f$,
since by convexity and the definition of $z_k$:
$f(x_k) - f^{\star} \leq f(x_k) - \min_{x\in\mathcal{X}} f(x) \leq \nabla f(x_k)^T(x_k - x^{\star})
\leq \nabla f(x_k)^T(x_k - z_k) = g_k$.
This provides a principled stopping criterion without knowledge of $f^{\star}$.

\begin{proposition}[Frank--Wolfe gap as certificate~\cite{jaggi2013}]
\label{prop:fw_gap}
For any feasible point $x_k \in \mathcal{X}$ and any convex differentiable $f$,
the Frank--Wolfe gap satisfies
$f(x_k) - f^{\star} \leq g_k = \nabla f(x_k)^T(x_k - z_k)$,
so the gap is an \emph{anytime certificate} of $\varepsilon$-optimality:
if $g_k \leq \varepsilon$, then $x_k$ is within $\varepsilon$ of the global minimum.
In this report's interior experiment, after 140 iterations with $\beta=0.985$,
$g_{140} = \nabla f(x_{140})^T(x_{140} - z_{140}) \approx 2\|x_{140} - (1,5)\|^2 \approx 0.0092$,
confirming that the iterate is $0.0092$-optimal without computing $f^{\star}$.
\end{proposition}

\begin{theorem}[Frank--Wolfe convergence~\cite{frank1956}]
For an $L$-smooth convex function $f$ over a bounded convex set $\mathcal{X}$ with diameter $D$,
using the standard step-size schedule $\gamma_k = 2/(k+2)$ (equivalently $\beta_k = k/(k+2)$),
the Frank--Wolfe algorithm satisfies
\begin{equation}
\label{eq:fw_convergence}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2}.
\end{equation}
This $O(1/k)$ rate is generally unimprovable for first-order methods using only
the linear minimisation oracle~\cite{jaggi2013}.
\end{theorem}

\begin{remark}[Fixed vs.\ decreasing step size]
The implementation here uses a fixed $\beta$ rather than the decreasing schedule
$\gamma_k = 1-\beta_k = 2/(k+2)$. A fixed $\beta$ does not guarantee the $O(1/k)$ rate,
which explains why the boundary optimum case does not fully converge within 140 iterations.
\end{remark}

\begin{remark}[Linear convergence on strongly convex objectives]
When $f$ is additionally $\mu$-strongly convex and the optimum $x^{\star}$ lies in the
\emph{interior} of $\mathcal{X}$, Frank--Wolfe with the standard step schedule achieves
linear convergence~\cite{jaggi2013}:
\begin{equation}
\label{eq:fw_linear}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k (f(x_0) - f^{\star}).
\end{equation}
Intuitively, when the optimum is in the interior, the LP oracle consistently selects
vertices that are ``towards'' the optimum from $x_k$, and strong convexity guarantees
each step makes geometric progress.
When the optimum lies on the boundary of $\mathcal{X}$ (as in the second experiment),
the oracle must average many vertex contributions to approximate the boundary point,
yielding only $O(1/k)$ convergence even for strongly convex $f$.
This is the fundamental distinction between interior and boundary optima for Frank--Wolfe.
\end{remark}

\subsection{Interior vs Boundary Behaviour}
\label{ssec:q6_analysis}
% ----------------------------------------------------------

\textbf{Interior optimum:}
Frank--Wolfe exhibits a characteristic ``zig-zag'' pattern because $z_k$ jumps between
extreme vertices of $\mathcal{X}$ at each iteration, and the convex combination
$x_{k+1} = \beta x_k + (1-\beta)z_k$ gradually averages these jumps towards the interior.
The general $O(1/k)$ convergence rate (for convex, smooth $f$) is observed.
A smaller step size $\gamma = 1-\beta$ reduces the influence of the vertex oscillation
at the expense of requiring more iterations in early stages.

\textbf{Boundary optimum:}
Frank--Wolfe is well suited to boundary optima because the LP oracle $z_k$ frequently
selects the vertex closest to the constrained optimum.
As $(x_k)_1 \to 0.5^+$, the gradient $(\nabla f(x_k))_1 = 2(x_k)_1 \to 1 > 0$,
so the oracle consistently selects $(z_k)_1 = l_1 = 0.5$, pulling $x_k$ towards the
boundary without requiring an explicit projection.
The iterate naturally approaches the constraint boundary, which is the key structural
advantage of Frank--Wolfe over projected gradient descent in settings where the projection
oracle is expensive (e.g., nuclear norm ball, flow polytope).
However, the fixed $\beta=0.93$ does not allow the step size to decay, leaving residual
oscillation that prevents full convergence within 140 iterations.
Using the standard Frank--Wolfe schedule $\gamma_k = 2/(k+2)$, as demonstrated for the interior
case in Figure~\ref{fig:q6_schedule}, would guarantee $f(x_{140}) - f^{\star} \leq 2LD^2/142$
with $L=2$ and $D \approx 15.7$. For box constraints, both projected GD and Frank--Wolfe
are comparably efficient; Frank--Wolfe's advantage appears in constraint sets with cheap LMO but expensive projection.

\section{Conclusions and Comparative Summary}
\label{sec:conclusions}
% ============================================================

This report has investigated six families of optimisation algorithms on three benchmarks
of increasing difficulty. The key conclusions are:

\begin{enumerate}

\item \textbf{Adaptive step-size methods (Q1):}
The Polyak step size achieves optimal convergence when $f^{\star}$ is exactly known,
but diverges severely when $f^{\star}$ is misspecified --- a critical limitation in practice.
Adagrad provides per-coordinate adaptation with guaranteed monotone step-size decay;
RMSprop improves on Adagrad by using exponential averaging; Heavy Ball achieves
accelerated convergence on smooth problems without knowledge of $f^{\star}$.

\item \textbf{Momentum and stochastic methods (Q2):}
Nesterov acceleration achieves an optimal $O(1/k^2)$ rate on convex problems,
outperforming GD ($O(1/k)$) most visibly on the ill-conditioned Rosenbrock function
($7\times$ improvement).
Adam combines momentum and per-coordinate scaling, performing well on moderately
anisotropic problems.
Mini-batch SGD demonstrates the variance--efficiency trade-off: smaller batches
reduce per-epoch cost but increase gradient noise, and the irreducible error floor
scales with observation noise variance $\sigma^2$.

\item \textbf{Newton's method (Q3):}
Newton's method achieves quadratic convergence near the optimum, solving the quadratic
Benchmark~A in a single step and outperforming GD by $5.4\times$ on Rosenbrock using
only a quarter of the iterations.
The high per-iteration cost ($O(d^3)$) and need for a positive-definite Hessian
limit applicability to problems with moderate $d$ and well-behaved curvature.

\item \textbf{Derivative-free methods (Q4):}
Forward finite differences with a well-chosen $\delta$ closely approximate the
exact gradient with $O(d+1)$ function evaluations per step.
Nesterov random search is dimension-independent in cost but high-variance,
requiring many more iterations.
Nelder--Mead navigates complex non-convex landscapes without gradient information
but lacks convergence guarantees and scales poorly.
Grid search guarantees global coverage to within grid resolution but is exponentially
expensive ($O(N^d)$).

\item \textbf{Constrained optimisation (Q5):}
Projected gradient descent enforces hard feasibility at every iteration through the
projection operator, which is $O(1)$ for the half-space constraint $x_1 \geq 0.5$.
Penalty methods provide a softer approach requiring only gradient modifications,
but must balance $\lambda$ against step-size stability.
For this problem, the constrained and unconstrained optima coincide (constraint inactive
at $x^{\star}$), so both methods correctly recover the same solution.

\item \textbf{Frank--Wolfe algorithm (Q6):}
The Frank--Wolfe algorithm avoids explicit projection, instead using a linear
minimisation oracle that decomposes over coordinates for box constraints.
It achieves $O(1/k)$ convergence for smooth convex objectives.
For interior optima, a conservative step size reduces oscillation near convergence.
For boundary optima, the oracle naturally directs iterates towards the boundary,
but a fixed step size limits final accuracy.

\end{enumerate}

For smooth unconstrained problems, Nesterov acceleration and Adam offer the best
practical performance. For constrained problems, projected gradient descent is preferred
when projection is cheap; Frank--Wolfe excels when the linear oracle is efficient.
Newton's method remains unmatched for small-scale, well-conditioned problems
where Hessian computation is affordable.

\paragraph{Unified view: a hierarchy of curvature approximations.}
All six algorithm families can be organised along a single axis of \emph{how much curvature
information they exploit}, which directly determines their convergence rate and per-iteration cost:
\begin{enumerate}[nosep]
\item \textbf{Constant step (GD):} $x_{k+1} = x_k - \alpha g_k$. No curvature.
  Rate $O(1/k)$ or $O((1-\mu/L)^k)$.
\item \textbf{Momentum (Heavy Ball, Nesterov):} Implicit curvature via gradient history.
  Nesterov achieves the optimal $O(1/k^2)$ first-order rate without Hessian access.
\item \textbf{Diagonal preconditioning (Adagrad, RMSprop, Adam):} Explicit estimation of the
  diagonal of the expected outer product $\mathbb{E}[g_k g_k^T] \approx \operatorname{diag}(H)$,
  giving per-coordinate adaptive steps. Cost: $O(d)$ extra storage.
\item \textbf{Low-rank Hessian approximation (BFGS, L-BFGS):} Maintains a rank-$2m$
  implicit approximation to $H^{-1}$ from gradient differences. Superlinear convergence
  at $O(md)$ cost per step.
\item \textbf{Full Newton (Q3):} Exact $H_k^{-1}$ as preconditioner. Quadratic convergence
  at $O(d^3)$ cost. Optimal when $d$ is small and $f$ is smooth.
\end{enumerate}
This hierarchy reveals the common structure: all adaptive methods are approximations to Newton's
method, differing in how richly they represent the local curvature. The choice of level in this
hierarchy is determined by problem dimension $d$, gradient availability, and the degree of
problem non-convexity.

\paragraph{Broader lessons from the Rosenbrock benchmark.}
Benchmark~C (Rosenbrock) served as the critical discriminator between methods because
its extreme ill-conditioning ($\kappa \approx 2504$) and nonlinear valley amplify
the differences between approaches. The ranking on Rosenbrock --- Polyak (when $f^{\star}$
is known) $\gg$ Nesterov $>$ Heavy Ball $>$ Adagrad $>$ Adam $\approx$ RMSprop $>$ GD ---
illustrates a fundamental principle: \emph{curvature information, whether through second-order
methods (Newton), momentum-based acceleration (Nesterov), or optimal step rules (Polyak),
is the key to fast convergence on ill-conditioned problems}.
Methods that use only first-order information with a fixed step size (GD, RMSprop with small $\alpha_0$)
cannot exploit the consistent curvature structure of the valley and stagnate.

\paragraph{Theoretical rates vs.\ empirical performance.}
A recurring theme is the gap between worst-case theoretical rates and observed empirical
performance. Nesterov's $O(1/k^2)$ rate is tight for the worst-case smooth convex function,
but on Benchmark~B (near-quadratic, well-conditioned) both GD and Nesterov converge
to machine precision quickly. The theoretical rates provide a guide for algorithm selection
but do not capture the full picture: hyperparameter tuning, problem structure, and starting
point all significantly influence practical performance, as evidenced by the sensitivity
of Polyak to $f^{\star}$ specification and of Heavy Ball to $(\alpha, \beta)$ joint tuning.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                      for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
