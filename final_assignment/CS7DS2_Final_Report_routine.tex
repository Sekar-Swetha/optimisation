\begin{abstract}
This report presents a comprehensive comparative study of 14 first-order, second-order, and derivative-free optimisation algorithms applied to three benchmark functions of increasing difficulty: a linear regression quadratic loss (strongly convex, $\kappa \approx 1$, well-conditioned), a toy neural network loss with sinusoidal perturbation (mildly non-quadratic, $\kappa \in [10/3, 10]$), and the Rosenbrock function (highly non-convex, $\kappa \approx 2000$). Six categories of methods are studied, with all convergence claims supported by theoretical bounds and empirical results.

\textbf{Question 1} compares adaptive step-size methods --- Polyak step, Adagrad, RMSprop, and Heavy Ball --- against a constant-step gradient descent baseline. The Polyak step achieves the best convergence on Rosenbrock ($f = 0.0094$) when $f^{\star}$ is known exactly but diverges on the regression problem where $f^{\star}$ is misspecified. \textbf{Question 2} examines Nesterov acceleration, Adam, and mini-batch SGD; Nesterov achieves a theoretical $O(1/k^2)$ convergence rate and reaches $f = 0.479$ on Rosenbrock versus $f = 3.388$ for vanilla gradient descent. \textbf{Question 3} demonstrates that Newton's method solves the quadratic problem in \emph{one iteration} and reduces Rosenbrock loss by a factor of 5$\times$ compared to gradient descent in only 20 iterations. \textbf{Question 4} analyses derivative-free methods, showing that finite-difference approximation with $\delta = 0.05$ closely tracks exact gradient descent while $\delta = 0.8$ degrades severely, and Nelder--Mead navigates the Rosenbrock valley without any gradient information. \textbf{Question 5} compares constrained optimisation strategies: projected gradient descent enforces feasibility from the very first iteration, while penalty methods require $\lambda \geq 1.8$ to achieve feasibility within 100 iterations. \textbf{Question 6} analyses the Frank--Wolfe algorithm, demonstrating how the linear minimisation oracle naturally enforces box constraints and how the optimum location (interior vs.\ boundary) fundamentally changes convergence behaviour.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. Each benchmark is chosen to stress-test different algorithm properties.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\label{eq:loss_A}
\end{equation}
This is a \emph{strongly convex} quadratic with constant Hessian $H = \frac{1}{m}X^T X \in \mathbb{R}^{2\times 2}$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the minimum loss is $J^{\star} \approx 0.4826$ (the irreducible noise floor), \emph{not} zero. The starting point is $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\label{eq:loss_B}
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H = \mathrm{diag}(2 - \sin(x_1),\ 10)$. Since $\sin(x_1) \in [-1, 1]$, the $(1,1)$ Hessian entry varies in $[1, 3]$, giving a condition number $\kappa = 10/(2-\sin(x_1)) \in [10/3,\, 10]$, i.e.\ moderate anisotropy. The minimum satisfies $x_2 = 2$ and $2(x_1 - 1) + \cos(x_1) = 0$, giving $x_1 \approx 0.582$ and $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\label{eq:loss_C}
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is:
\[
\nabla^2 f(1,1) = \begin{bmatrix} 2 + 1200 - 400 & -400 \\ -400 & 200 \end{bmatrix} = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\]
with eigenvalues approximately $0.5$ and $1001.5$, giving condition number $\kappa \approx 2003$. This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions in numerical optimisation. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$. This is a convex box constraint set, for which projection and linear minimisation both admit closed-form solutions.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}

\subsection{Theoretical Background}

\begin{definition}
A differentiable function $f$ is \emph{$L$-smooth} if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x-y\|$ for all $x, y$. It is \emph{$\mu$-strongly convex} if $f(y) \geq f(x) + \nabla f(x)^T(y-x) + \frac{\mu}{2}\|y-x\|^2$. The condition number is $\kappa = L/\mu$.
\end{definition}

Standard gradient descent with a fixed step size $\alpha \leq 1/L$ applied to an $L$-smooth, $\mu$-strongly convex function satisfies $f(x_k) - f^{\star} \leq (1 - \mu\alpha)^k (f(x_0) - f^{\star})$, a linear (geometric) convergence rate with ratio $(1-\mu/L)$. For ill-conditioned problems with large $\kappa = L/\mu$, this rate is close to 1 and convergence is slow. Adaptive methods aim to exploit local curvature information to escape this worst-case bound on a per-coordinate or per-iteration basis.

For Benchmark A: The Hessian $H = \frac{1}{m}X^TX$ has eigenvalues $L = \lambda_{\max}(H)$ and $\mu = \lambda_{\min}(H)$. By the law of large numbers, $H \to I_2$ as $m \to \infty$ (since $\mathbb{E}[x^{(i)} x^{(i)T}] = I_2$). For $m = 1000$, the eigenvalues concentrate tightly around 1 (finite-sample fluctuations are $O(d/m) = O(2/1000)$), yielding $\kappa \approx 1$, enabling near-optimal convergence of all methods. This explains why all Q1 methods converge to $J^{\star}$ on Benchmark A despite their otherwise differing convergence rates. For Benchmark C: $L \approx 1002$ at the starting point, so GD requires $\alpha \leq 1/1002 \approx 0.001$, matching the used value of $\alpha = 0.0012$. The theoretical convergence rates for GD, Heavy Ball, and Nesterov are visualised in the conclusion (Figure~\ref{fig:theory_rates}).

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak step convergence]
\label{thm:polyak}
Let $f$ be $\mu$-strongly convex and $L$-smooth. The Polyak step $\alpha_k = (f(x_k)-f^{\star})/\|g_k\|^2$ satisfies:
\[
f(x_{k+1}) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)(f(x_k) - f^{\star}).
\]
\end{theorem}
\begin{remark}
To see why: by $L$-smoothness, $f(x_{k+1}) \leq f(x_k) - \alpha_k \|g_k\|^2 + \frac{L}{2}\alpha_k^2 \|g_k\|^2$. Substituting $\alpha_k = (f(x_k)-f^{\star})/\|g_k\|^2$ and using $\|g_k\|^2 \geq 2\mu(f(x_k)-f^{\star})$ (from strong convexity) gives the stated bound. Thus the Polyak step is an \emph{adaptive} step automatically tuned to the remaining sub-optimality: it is large when $f(x_k) - f^{\star}$ is large (far from optimum) and shrinks as the optimum is approached. When $f^{\star}$ is misspecified (set below the true minimum), $\alpha_k \to \infty$ as $\|g_k\| \to 0$, causing divergence.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0 > 0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad assigns a smaller effective step to coordinates with large historical gradients, automatically normalising coordinates by their curvature. The monotone accumulation $G_k \to \infty$ ensures $\alpha_{\text{eff}} \to 0$, guaranteeing convergence but potentially slowing progress in later iterations.

\begin{theorem}[Adagrad regret bound]
For a convex objective and any sequence $g_1, \ldots, g_T$, Adagrad achieves the online regret bound:
\[
\sum_{t=1}^T f(x_t) - f(x^{\star}) \leq \frac{1}{2\alpha_0}\sum_{i=1}^d \sqrt{\sum_{t=1}^T g_{t,i}^2}\cdot\|x^{\star} - x_0\|^2,
\]
which adapts to the \emph{observed gradient structure}: if coordinate $i$ has small historical gradients ($G_{T,i}$ small), then a larger effective step is allowed for that coordinate. This contrasts with standard GD, where the step is constrained by the worst-case coordinate. The regret bound is $O(\sqrt{T})$ in the worst case, giving a convergence rate $O(1/\sqrt{T})$ on average.
\end{theorem}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0 > 0$, decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop fixes Adagrad's monotone decay by using an \emph{exponential moving average} of squared gradients, allowing the effective step size to increase after a period of small gradients. The parameter $\beta = 0.9$ means the effective memory window is approximately $1/(1-\beta) = 10$ recent gradients.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method can be interpreted as a discretised damped harmonic oscillator: $z$ is the ``velocity'' that accumulates gradient information, and $\beta$ controls the damping. On quadratic functions with $\mu$-strong convexity and $L$-smoothness, the optimal $(\alpha, \beta)$ yields convergence rate $\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^k$, which is the square root of the GD rate.

\begin{remark}[Lyapunov argument for Heavy Ball]
Consider the Lyapunov function $E_k = \|x_k - x^{\star}\|_H^2 + \frac{\alpha}{1-\beta}\|z_k\|^2$, where $\|\cdot\|_H$ denotes the $H$-weighted norm. On a quadratic $f(x) = \frac{1}{2}(x-x^{\star})^T H (x-x^{\star})$ with $H = \text{diag}(\mu_1,\ldots,\mu_d)$, expanding the Heavy Ball recursion shows:
$E_{k+1} \leq \rho_{\text{HB}}^2\, E_k$ where $\rho_{\text{HB}} = \frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}$ at the optimal $\alpha^{\star} = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2}$, $\beta^{\star} = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$.
Compared to GD where the Lyapunov function $E_k = f(x_k) - f^{\star}$ decreases as $(1-\mu\alpha)^k \approx e^{-\mu k/L} = e^{-k/\kappa}$, Heavy Ball achieves $\rho_{\text{HB}}^k \approx e^{-2k/\sqrt{\kappa}}$, an exponential improvement in the exponent by $\sqrt{\kappa}$. This is exact for quadratics; for non-quadratic $f$, Heavy Ball loses this guarantee (the optimal $\alpha^{\star}$ and $\beta^{\star}$ depend on the unknown $\mu$ and $L$ locally), which partly explains its Rosenbrock underperformance relative to the theoretical prediction.
\end{remark}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly (Rosenbrock, Q1C), Polyak converges fastest ($f = 0.0094$, $87\%$ better than GD). Heavy Ball achieves $60\%$ better than GD on Rosenbrock through momentum, and matches GD's final accuracy on the well-conditioned problems A and B. Adagrad matches GD in final accuracy but reaches the noise floor in fewer iterations on Benchmark A due to its large initial rates.

\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to the exponential moving average, which filters out gradient noise. Heavy Ball introduces damped oscillations characteristic of under-damped second-order dynamics. Polyak is smooth when $f^{\star}$ is correct (Rosenbrock) but erratic otherwise.

\item \textbf{Stability:} GD baseline is most stable but constrained by the global Lipschitz constant $L$. Adagrad is uniformly stable due to its monotonically decreasing step, guaranteed not to overshoot once initialised. Polyak is \emph{unstable} under $f^{\star}$ misspecification: the effective step size is theoretically unbounded.

\item \textbf{Practical implication:} In practice, the true $f^{\star}$ is rarely known, severely limiting Polyak's applicability to idealised settings. RMSprop and Adagrad are preferred for problems where a good $\alpha_0$ is unknown, as they automatically calibrate effective step sizes per coordinate. Heavy Ball is effective on strongly convex problems but can oscillate on non-convex landscapes.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}

\subsection{Theoretical Background}

Standard gradient descent achieves an $O(1/k)$ convergence rate for convex functions. Nesterov (1983) proved that no first-order method can achieve better than $O(1/k^2)$ on smooth convex functions, and his acceleration exactly attains this bound:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2},
\end{equation}
a quadratic improvement over GD's $O(L\|x_0 - x^{\star}\|^2 / k)$. Adam combines gradient momentum (via $m_t$) with per-coordinate adaptive scaling (via $v_t$), making it robust to heterogeneous gradient magnitudes at the cost of a weaker $O(1/\sqrt{k})$ convergence guarantee in the non-convex setting.

\begin{theorem}[Nesterov lower bound (informal)]
\label{thm:nesterov_lb}
For any first-order algorithm (i.e., any method whose iterate $x_{k+1}$ lies in the affine span of $\{x_0, \nabla f(x_0), \ldots, \nabla f(x_k)\}$), there exists an $L$-smooth convex function $f$ such that:
\[
f(x_k) - f^{\star} \geq \frac{3L\|x_0 - x^{\star}\|^2}{32(k+1)^2}.
\]
\end{theorem}
This means Nesterov's $O(1/k^2)$ rate is \emph{information-theoretically optimal}: no first-order method can achieve a better asymptotic rate on the class of smooth convex functions. The matching of upper and lower bounds establishes Nesterov acceleration as the gold standard for first-order convex optimisation. Heavy Ball matches this rate on quadratics but lacks the $O(1/k^2)$ guarantee for general convex functions, which is why Nesterov's adaptive $\beta_k$ schedule is essential for provable acceleration.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, max momentum $\beta_{\max} \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Gradient evaluated at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball (Section~\ref{sec:q1}) is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$, allowing the method to ``look ahead'' and correct its trajectory before overshooting. Heavy Ball uses a \emph{fixed} $\beta$ and gradient at the current point $x_k$, making it equivalent to a second-order recurrence $x_{k+1} - \beta x_k = x_k - \beta x_{k-1} - \alpha \nabla f(x_k)$, whereas Nesterov's adaptive $\beta_k = (k-1)/(k+2) \to 1$ schedule is carefully chosen so that the \emph{Lyapunov energy} $E_k = (f(x_k) - f^{\star}) + \frac{1}{2\alpha}\|v_k\|^2$ decreases at the rate $O(1/k^2)$. Heavy Ball is optimal for quadratic $f$ but can diverge on non-quadratic functions; Nesterov's adaptive schedule provides a provable $O(1/k^2)$ guarantee for all smooth convex $f$.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha > 0$, moments $\beta_1, \beta_2 \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (exponential moving average of gradient)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (exponential moving average of $g^2$)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction to account for zero initialisation}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

The bias correction is essential: without it, early estimates $m_1 = (1-\beta_1)g_1$ and $v_1 = (1-\beta_2)g_1^2$ are heavily biased towards zero. Dividing by $(1-\beta_1^t)$ rescales these to be unbiased estimators of the true first and second moments of the gradient. For $t \gg 1$, the bias correction factor approaches 1 and has negligible effect.

\subsection{Mini-Batch Stochastic Gradient Descent}
\label{sec:q2_sgd}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch. The mini-batch gradient is:
\begin{equation}
g_{\mathcal{B}} = \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}}), \qquad |\mathcal{B}| = b.
\end{equation}
This is an \emph{unbiased} estimator of the full gradient: $\mathbb{E}[g_{\mathcal{B}}] = \nabla J(\theta)$. The variance is $\mathrm{Var}[g_{\mathcal{B}}] = \frac{m - b}{m - 1} \cdot \frac{\sigma^2_g}{b}$, where $\sigma^2_g$ is the per-sample gradient variance, decreasing as $O(1/b)$. Each epoch performs $\lfloor m/b \rfloor$ gradient updates: $m/5 = 200$ for $b=5$ vs.\ $m/40 = 25$ for $b=40$.

The total progress per epoch scales as (updates per epoch) $\times$ (step quality): small batches make many noisy steps; large batches make few accurate steps. For linear regression near convergence, the expected per-step progress is $\propto \alpha \mu / 2 \cdot \mathbb{E}[\|x_k - x^{\star}\|^2] - \alpha^2/(2\mu b)\cdot \sigma^2_g$, revealing a fundamental speed-accuracy trade-off: smaller batches are faster per epoch but introduce a noise floor that prevents convergence below $O(\alpha \sigma^2_g / (b\mu))$.

More precisely, for an $L$-smooth $\mu$-strongly-convex function, a single SGD step with mini-batch $\mathcal{B}$ satisfies:
\begin{equation}
\mathbb{E}\!\left[\|x_{k+1} - x^{\star}\|^2\right] \leq (1 - \alpha\mu)\,\mathbb{E}\!\left[\|x_k - x^{\star}\|^2\right] + \frac{\alpha^2 \sigma^2_g}{b},
\label{eq:sgd_recurrence}
\end{equation}
where $\sigma^2_g = \frac{1}{m}\sum_{i=1}^m \|\nabla f_i(x^{\star})\|^2$ is the per-sample gradient variance at the optimum. Unrolling \eqref{eq:sgd_recurrence} over $K$ steps yields:
\begin{equation}
\mathbb{E}\!\left[\|x_K - x^{\star}\|^2\right] \leq (1 - \alpha\mu)^K\|x_0 - x^{\star}\|^2 + \frac{\alpha \sigma^2_g}{\mu b}.
\label{eq:sgd_bound}
\end{equation}
The two terms in \eqref{eq:sgd_bound} trade off: the first term (optimisation bias) decays geometrically and is reduced by large $\alpha$; the second term (statistical noise floor) grows with $\alpha$ and shrinks with batch size $b$. The optimal fixed $\alpha$ balances these as $\alpha^{\star} = \sqrt{b \|x_0 - x^{\star}\|^2 / (\sigma^2_g K)}$, giving a convergence rate $O(1/\sqrt{bK})$, which is faster than GD ($O(1/K)$) only when the variance term dominates. In practice, learning rate schedules $\alpha_k = c/k$ can achieve $O(1/k)$ at the cost of slower early-phase convergence.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$ \Comment{Ensures unbiasedness each epoch}
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{First- and Second-Order Local Approximations}

For a smooth function $g : \mathbb{R} \to \mathbb{R}$, the Taylor expansion around $x_0$ is:
\begin{equation}
g(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 + O((x-x_0)^3).
\end{equation}

For the test function $g(x) = x^4$ around $x_0 = 0.25$, with $g'(x) = 4x^3$, $g''(x) = 12x^2$:

\textbf{First-order approximation} (tangent line):
\begin{equation}
g(x) \approx g(0.25) + g'(0.25)(x - 0.25) = 0.0039 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order approximation} (osculating parabola):
\begin{equation}
g(x) \approx 0.0039 + 0.0625(x - 0.25) + \tfrac{1}{2}(0.75)(x - 0.25)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

\textbf{Deriving the Newton update:} To minimise the second-order approximation $\hat{g}(x) = g(x_0) + g'(x_0)(x-x_0) + \frac{1}{2}g''(x_0)(x-x_0)^2$, set $\hat{g}'(x) = g'(x_0) + g''(x_0)(x-x_0) = 0$ and solve:
\begin{equation}
x^{\star}_{\hat{g}} = x_0 - \frac{g'(x_0)}{g''(x_0)}.
\end{equation}
This is the \emph{Newton update}. Numerically for $g(x) = x^4$ at $x_0 = 0.25$:
\[
x_1^{\text{Newton}} = 0.25 - \frac{g'(0.25)}{g''(0.25)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667,
\]
which is close to (but not equal to) the true minimum at $x^{\star} = 0$ because $g(x) = x^4$ is not quadratic. A second Newton step from $x_1 = 0.1667$ gives $x_2 = 0.1667 - 4(0.1667)^3/(12(0.1667)^2) = 0.1667 - 0.1667/3 = 0.1111$, demonstrating the pattern of superlinear convergence $\|x_{k+1}\| = \frac{1}{3}\|x_k\|$ towards zero.

In multiple dimensions, it generalises to $x_{k+1} = x_k - \alpha [\nabla^2 f(x_k)]^{-1} \nabla f(x_k)$, where $\alpha \in (0, 1]$ is a damping factor that ensures global convergence by preventing overshoot when far from the optimum.

\subsubsection*{Newton Decrement and Convergence Phases}

The \emph{Newton decrement} at $x_k$ is defined as:
\begin{equation}
\lambda_N(x_k) = \sqrt{\nabla f(x_k)^T [\nabla^2 f(x_k)]^{-1} \nabla f(x_k)},
\label{eq:newton_decrement}
\end{equation}
and provides a theoretically-motivated stopping criterion. It measures the predicted decrease of the local quadratic model: $f(x_k) - \inf_x \hat{q}_{x_k}(x) = \frac{1}{2}\lambda_N(x_k)^2$. Two convergence phases exist:
\begin{itemize}[nosep]
  \item \textbf{Damped phase} ($\lambda_N \geq \eta$ for some threshold $\eta$): each step decreases $f$ by at least a constant $\gamma > 0$, giving linear convergence.
  \item \textbf{Quadratic phase} ($\lambda_N < \eta$, close to optimum): full Newton steps ($\alpha = 1$) give \emph{quadratic} convergence $\lambda_N(x_{k+1}) \leq \frac{L_H}{2\mu^2}\lambda_N(x_k)^2$, where $L_H$ is the Hessian Lipschitz constant.
\end{itemize}
In our experiments, Benchmarks A and B quickly enter the quadratic phase (explaining rapid convergence), while Benchmark C stays in the damped phase throughout the 20 iterations (the Hessian varies too rapidly to activate quadratic convergence). For $g(x) = x^4$ at $x_0 = 0.25$: $\lambda_N(0.25) = |g'(0.25)|/\sqrt{g''(0.25)} = 0.0625/\sqrt{0.75} = 0.072$, indicating we are moderately close to the optimum, and the Newton step moves to $x_1 = 0.167$.

\subsection{Newton's Method on All Benchmarks}

\begin{algorithm}[H]
\caption{Newton's Method with Hessian Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0,1]$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian (ensures PD)}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The damping $\lambda = 10^{-8}$ ensures $H_k + \lambda I \succ 0$ even when $H_k$ is nearly singular, preventing ill-conditioned system solves. The per-iteration cost is $O(d^3)$ for the Hessian solve (or $O(d^2)$ with Cholesky factorisation), compared to $O(d)$ for gradient descent --- a significant overhead that is justified only when second-order convergence is achieved.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}
% ============================================================

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\label{eq:fd}
\end{equation}
By Taylor expansion, $f(x + \delta e_i) = f(x) + \delta \frac{\partial f}{\partial x_i}(x) + \frac{\delta^2}{2}\frac{\partial^2 f}{\partial x_i^2}(x) + O(\delta^3)$, so the approximation error is:
\begin{equation}
\left|\frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| = O(\delta) + O\!\left(\frac{\varepsilon_{\text{mach}}}{\delta}\right),
\label{eq:fd_error}
\end{equation}
where the $O(\delta)$ term is \emph{truncation error} (from higher-order terms) and $O(\varepsilon_{\text{mach}}/\delta)$ is \emph{rounding error} (from floating-point cancellation when $\delta$ is too small). The total error is approximately $E(\delta) \approx \frac{\delta}{2}|f''(x)| + \frac{\varepsilon_{\text{mach}}\,|f(x)|}{\delta}$. Minimising over $\delta$ by setting $\frac{dE}{d\delta} = 0$:
\begin{equation}
\frac{1}{2}|f''(x)| = \frac{\varepsilon_{\text{mach}}\,|f(x)|}{\delta^2} \implies \delta^{\star} = \sqrt{\frac{2\,\varepsilon_{\text{mach}}\,|f(x)|}{|f''(x)|}}.
\label{eq:fd_optimal_delta}
\end{equation}
For typical problems where $|f(x)| \approx |f''(x)| \sim O(1)$, this gives $\delta^{\star} \sim \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$.
At the minimum total error, $E(\delta^{\star}) = O(\varepsilon_{\text{mach}}^{1/2})$, meaning finite differences can recover the gradient to about half the floating-point digits of precision. Each gradient evaluation requires $d + 1 = 3$ function evaluations for a $d=2$ problem, scaling linearly with dimension. An alternative is \emph{central differences}:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta},
\label{eq:cfd}
\end{equation}
which has truncation error $O(\delta^2)$ (second-order accurate) at the cost of $2d$ evaluations instead of $d+1$. The optimal $\delta$ for central differences satisfies $\delta^{\star}_{\text{CD}} \sim \varepsilon_{\text{mach}}^{1/3} \approx 10^{-5.3}$ (from $O(\delta^2) = O(\varepsilon_{\text{mach}}/\delta)$), yielding minimum error $O(\varepsilon_{\text{mach}}^{2/3}) \approx 10^{-10.7}$, versus $O(\varepsilon_{\text{mach}}^{1/2}) \approx 10^{-8}$ for forward differences. Central differences are preferred in practice when gradient accuracy is more important than evaluation count.

To quantify the error for Benchmark B at the starting point $x_0 = [-1, 4]$: the true gradient component $\frac{\partial f}{\partial x_1} = 2(-1-1) + \cos(-1) = -4 + 0.540 = -3.460$. With $\delta = 0.05$: $\frac{f(x_0 + 0.05 e_1) - f(x_0)}{0.05}$; since the second derivative is $g''(x_1) = 2 - \sin(x_1)$, the truncation error is $\frac{0.05}{2}|g''(-1)| = \frac{0.05}{2}(2+0.841) = 0.071$, a relative error of $2\%$ --- acceptable. With $\delta = 0.8$: truncation error $\approx \frac{0.8}{2}|g''(-1)| = 1.136$, a $33\%$ relative error --- unacceptable for gradient-based optimisation.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, perturbation $\delta > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Random unit vector on sphere}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates $\nabla f(x_k)^T u_k / \|u_k\|$ (the directional derivative along $u_k$) and uses it as a gradient surrogate. Crucially, only \emph{two} function evaluations are needed per iteration regardless of dimension $d$, making it dimension-independent in evaluation cost per step. However, the estimator is noisy (variance $\sim \|\nabla f\|^2$ in the worst case), leading to slower convergence than gradient-based methods.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations: \textbf{reflection} ($x_r = \bar{x} + \alpha_r(\bar{x} - x_{\text{worst}})$, $\alpha_r = 1$), \textbf{expansion} ($x_e = \bar{x} + \gamma(\bar{x} - x_{\text{worst}})$, $\gamma = 2$), \textbf{contraction} ($x_c = \bar{x} \pm \rho(\bar{x} - x_{\text{worst}})$, $\rho = 0.5$), and \textbf{shrink} (all vertices contracted towards the best by factor $\sigma = 0.5$). The centroid $\bar{x}$ is computed over all vertices except the worst. No gradient information is required.

\subsection{Problem Setup and Optimality Conditions}

Minimise $f(x_1, x_2) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)$ subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$. The constraint is $g(x) = 0.5 - x_1 \leq 0$.

\subsubsection*{Constraint Qualification: Slater's Condition}

Before applying the KKT conditions, we verify that a \emph{constraint qualification} holds to ensure KKT conditions are necessary for optimality. \textbf{Slater's condition} requires the existence of a \emph{strictly feasible} point $\bar{x}$ satisfying $g(\bar{x}) < 0$. For our constraint $g(x) = 0.5 - x_1 \leq 0$, any point with $x_1 > 0.5$ (e.g., $x_1 = 1.0$) gives $g(1.0, \cdot) = 0.5 - 1.0 = -0.5 < 0$. Slater's condition holds, so the KKT conditions are \emph{necessary and sufficient} for optimality (given convexity of $f$ and $g$).

The KKT optimality conditions for this problem are:
\begin{align}
\nabla f(x^{\star}) + \mu \nabla g(x^{\star}) &= 0 \quad \text{(stationarity)}, \label{eq:kkt1}\\
g(x^{\star}) = 0.5 - x_1^{\star} &\leq 0 \quad \text{(primal feasibility)}, \label{eq:kkt2}\\
\mu &\geq 0 \quad \text{(dual feasibility)}, \label{eq:kkt3}\\
\mu\, g(x^{\star}) &= 0 \quad \text{(complementary slackness)}. \label{eq:kkt4}
\end{align}

Since the unconstrained minimum of $f$ lies at $x_1^{\star} \approx 0.582 > 0.5$, the constraint $g(x^{\star}) = 0.5 - 0.582 = -0.082 < 0$ is \emph{strictly inactive} at the unconstrained optimum. By complementary slackness \eqref{eq:kkt4}, this requires $\mu = 0$. Substituting into \eqref{eq:kkt1} gives $\nabla f(x^{\star}) = 0$, i.e.\ the constrained and unconstrained problems share the same optimal solution. This is why the unconstrained GD trajectory naturally converges to the feasible optimum: the constraint is never binding.

The penalty parameter $\lambda$ must be large enough that the gradient of the penalty term $\lambda$ (in the $x_1$ direction, when $x_1 < 0.5$) exceeds the gradient of $f$ pulling the iterate away from feasibility. Near $x_1 = 0.2$: $|\partial f/\partial x_1| = |2(0.2-1) + \cos(0.2)| \approx |-1.6 + 0.980| = 0.62$. So $\lambda \geq 0.62$ should be sufficient to push $x_1$ towards feasibility, explaining why $\lambda = 0.15$ fails and $\lambda = 1.8$ succeeds.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1})$ where $\Pi_{\mathcal{C}}(z)_1 = \max(0.5,\, z_1)$, $\Pi_{\mathcal{C}}(z)_2 = z_2$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x_k) \gets f(x_k) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $[\nabla F_{\lambda}(x_k)]_1 \gets [\nabla f(x_k)]_1 - \lambda \cdot \mathbf{1}[x_1 < 0.5]$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$, 100 iterations), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$, 100 iterations).

\textbf{Projected GD} guarantees \emph{immediate} feasibility from the first iteration. This is its fundamental advantage: the projection operator $\Pi_{\mathcal{C}}$ enforces $x \in \mathcal{C}$ as a hard constraint at every step. The computational cost of projection is $O(1)$ for this simple constraint (a single $\max$ operation), though for complex constraint sets (e.g., the nuclear norm ball in matrix completion) it may require an $O(d^3)$ SVD.

\textbf{Penalty methods} convert the constrained problem into an unconstrained one, avoiding the projection oracle entirely. The trade-off is that feasibility is enforced \emph{approximately}, depending on $\lambda$. As established in the KKT analysis above, the constraint gradient at the starting point has magnitude $\approx 0.62$, requiring $\lambda > 0.62$ for the penalty to overcome the objective gradient pulling towards the infeasible region. The experiments confirm: $\lambda = 0.15 < 0.62$ fails to enforce feasibility, while $\lambda = 1.8 > 0.62$ and $\lambda = 4.5 \gg 0.62$ both succeed.

Increasing $\lambda$ tightens constraint enforcement but causes ill-conditioning: the Lipschitz constant of $F_{\lambda}$ is $L_{F_{\lambda}} = L_f + \lambda$ (the penalty adds an extra gradient term), requiring smaller step sizes $\alpha \leq 1/L_{F_{\lambda}}$ for stability. This explains why $\alpha$ must be reduced from $0.05$ to $0.03$ for $\lambda = 4.5$: $1/(L_f + 4.5) \approx 0.03$ versus $1/(L_f + 1.8) \approx 0.05$.

\subsubsection*{Augmented Lagrangian Remedy}

The \emph{augmented Lagrangian} (method of multipliers) resolves the large-$\lambda$ ill-conditioning by introducing a dual variable $\nu \geq 0$:
\begin{equation}
F_{\text{AL}}(x;\,\nu,\lambda) = f(x) + \nu\, g(x) + \tfrac{\lambda}{2}\!\left[\max(0,\, g(x))\right]^2,
\end{equation}
where $g(x) = 0.5 - x_1 \leq 0$. After each primal minimisation $x^{(j+1)} = \arg\min_x F_{\text{AL}}(x;\nu^{(j)}, \lambda)$, the dual variable is updated as:
\begin{equation}
\nu^{(j+1)} = \max\!\left(0,\ \nu^{(j)} + \lambda\, g(x^{(j+1)})\right).
\label{eq:al_dual_update}
\end{equation}
Under mild conditions, this achieves exact feasibility ($g(x^{\star}) = 0$ or $g(x^{\star}) < 0$, $\nu^{\star} g(x^{\star}) = 0$ by complementarity) even for \emph{finite} $\lambda$, because the dual variable $\nu$ absorbs the Lagrange multiplier, eliminating the need for $\lambda \to \infty$. For our problem: the optimal dual variable is $\nu^{\star} = 0$ (since the constraint is inactive), so the dual update \eqref{eq:al_dual_update} will converge to $\nu^{(j)} \to 0$ and the augmented Lagrangian approach reduces to pure penalty for this problem --- but in general (active-constraint cases) the augmented Lagrangian maintains a moderate $\lambda$ while achieving machine-precision feasibility.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, minimising over $\mathcal{X}$. Since $f$ is linear and $\mathcal{X}$ is a convex polytope, the \emph{fundamental theorem of linear programming} guarantees that the minimum is attained at a \emph{vertex} of $\mathcal{X}$. To see why: suppose $x^{\star}$ is an interior minimum with $\nabla f(x^{\star}) = a \neq 0$. Then $x^{\star} - \varepsilon a \in \mathcal{X}$ for small $\varepsilon > 0$ (interior points have a neighbourhood within $\mathcal{X}$), and $f(x^{\star} - \varepsilon a) = f(x^{\star}) - \varepsilon \|a\|^2 < f(x^{\star})$, contradicting minimality. Hence any minimum must lie on the boundary, and by induction on dimension, at a vertex. Intuitively: a linear function has no local minima in the interior (its Hessian is zero), so the gradient $\nabla f = a = [1, 2]^T$ always points away from the minimum, and the minimum is at the extreme point of $\mathcal{X}$ most opposed to $a$.

For a box constraint, this decouples per coordinate: since $a_1 = 1 > 0$, minimise $x_1$ at its lower bound $l_1 = 0.5$; since $a_2 = 2 > 0$, minimise $x_2$ at its lower bound $l_2 = -5$. This yields the closed-form solution:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}
This is precisely the LMO solution \eqref{eq:lmo_box} applied to the linear objective: $z = (l_1, l_2) = (0.5, -5)$ since both gradient components are positive. The Frank--Wolfe algorithm on a linear objective would converge in one step, since $z_0 = x^{\star}$ and $x_1 = \beta x_0 + (1-\beta)x^{\star}$ with $\beta \to 0$ as iterations increase.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination stays in $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Key property:} Since $\mathcal{X}$ is convex and $x_0, z_k \in \mathcal{X}$, all iterates $x_{k+1} = \beta x_k + (1-\beta)z_k \in \mathcal{X}$ by convexity --- \emph{no projection oracle is required}. This is the fundamental advantage of Frank--Wolfe over projected gradient descent.

For box constraints, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases},
\label{eq:lmo_box}
\end{equation}
requiring only $O(d)$ operations per iteration (vs.\ $O(d^3)$ for projected GD with general constraints).

\begin{theorem}[Frank--Wolfe convergence, $L$-smooth convex case]
For an $L$-smooth convex function $f$ over a convex compact set $\mathcal{X}$ with diameter $D = \max_{x,y \in \mathcal{X}}\|x-y\|$, the Frank--Wolfe method with step-size $\gamma_k = 2/(k+2)$ satisfies:
\[
f(x_k) - f^{\star} \leq \frac{2L D^2}{k+2} = O(1/k).
\]
This $O(1/k)$ rate is the \emph{same} as projected gradient descent but does not require projections. For our box $\mathcal{X}$: $D = \sqrt{(5-0.5)^2 + (10-(-5))^2} = \sqrt{20.25 + 225} = \sqrt{245.25} \approx 15.66$.
\end{theorem}

\begin{definition}[Frank--Wolfe gap]
The \emph{Frank--Wolfe gap} at $x_k$ is $h_k = \nabla f(x_k)^T(x_k - z_k)$, where $z_k$ solves the LMO. This quantity satisfies $h_k \geq f(x_k) - f^{\star} \geq 0$ for convex $f$ and provides a \emph{computable} upper bound on sub-optimality without knowing $f^{\star}$.
\end{definition}

A numerical example of the FW gap: for the \textbf{interior} case $f(x) = (x_1-1)^2 + (x_2-5)^2$ starting at $x_0 = (1, 1)$, the gradient is $\nabla f(x_0) = [2(1-1), 2(1-5)] = [0, -8]$. The LMO selects $z_0 = (\arg\min_{x_1} 0\cdot x_1, \arg\min_{x_2} -8x_2) = (0.5, 10)$ (lower bound in $x_1$ since coefficient zero picks the smaller; upper bound in $x_2$ since coefficient $-8 < 0$). The FW gap is:
\[
h_0 = \nabla f(x_0)^T(x_0 - z_0) = [0, -8]^T[1 - 0.5,\ 1 - 10] = [0, -8]^T[0.5, -9] = 0 + 72 = 72.
\]
Meanwhile, the true sub-optimality is $f(x_0) - f^{\star} = 0 + 16 = 16$. So $h_0 = 72 \geq 16 = f(x_0) - f^{\star}$, confirming the FW gap overestimates sub-optimality (as guaranteed), with a factor of $72/16 = 4.5$ at this initial point.

The fixed-$\beta$ scheme used here (rather than the $\gamma_k = 2/(k+2)$ optimal schedule) does not guarantee $O(1/k)$ convergence, but provides a simple, stable update. With fixed $\beta$, the effective step size towards $z_k$ is $(1-\beta)$, analogous to a fixed learning rate in gradient descent. The key trade-off is: larger $(1-\beta)$ enables faster initial progress but causes more oscillation near the optimum due to large discrete jumps to constraint vertices; smaller $(1-\beta)$ slows initial convergence but produces smoother settling behaviour. Note that our Frank--Wolfe implementation uses the update $x_{k+1} = \beta x_k + (1-\beta)z_k$, where $\beta = 1 - \gamma_k$ in the standard $\gamma$-parametrisation.

\subsection{Interior vs.\ Boundary Behaviour Comparison}

\textbf{Interior optimum:} Frank--Wolfe exhibits the characteristic ``zig-zag'' pattern near interior optima. The Frank--Wolfe gap $h_k = \nabla f(x_k)^T(x_k - z_k)$ bounds the sub-optimality; near the interior optimum $(1,5)$, the gradient $\nabla f \approx 0$, so $h_k \to 0$ but the LMO solution $z_k$ becomes highly sensitive to the gradient direction and jumps between extreme vertices (e.g., $x_2 \in \{-5, +10\}$ alternating) as the tiny gradient fluctuates. The convex averaging $x_{k+1} = \beta x_k + (1-\beta)z_k$ damps these vertex jumps but cannot eliminate oscillation with a fixed $(1-\beta)$. The standard remedy is the decaying schedule $\gamma_k = 2/(k+2)$, ensuring $(1-\beta_k) \to 0$ and provably achieving $O(1/k)$ convergence without oscillation.

\textbf{Boundary optimum:} When the unconstrained optimum $(0,0)$ lies outside $\mathcal{X}$, Frank--Wolfe is particularly efficient: the gradient of $f = x_1^2 + x_2^2$ points away from the origin, so the LMO consistently selects the corner of $\mathcal{X}$ closest to the unconstrained optimum, namely $(0.5, -5)$ (the lower-left corner). The convex averaging then steadily pulls $x_k$ towards this vertex, automatically tracking the boundary. The iterate converges to $(0.5, 0)$ rather than $(0.5, -5)$ because as $x_{2,k} \to 0$ the gradient component $\partial f/\partial x_2 = 2x_2 \to 0$, making the LMO indifferent between $x_2 = -5$ and $x_2 = 10$; the averaging then pulls $x_2$ towards the average of the boundary values, and the objective is minimised at $x_2 = 0$. Unlike projected gradient descent (which would explicitly project the GD step onto $x_1 = 0.5$), Frank--Wolfe achieves boundary convergence without any projection, at the cost of the slow $O(1/k)$ rate rather than the $O(1/k)$ projected GD rate with larger effective constants.

\subsection{Key Lessons}

\begin{enumerate}[nosep]
\item \textbf{Problem structure determines method choice.} On the strongly convex quadratic Benchmark A, Newton's method solves the problem in one iteration while GD requires 80; on the ill-conditioned Rosenbrock ($\kappa \approx 2000$), even Newton requires many damped iterations and only Nesterov with momentum achieves meaningful progress beyond the early rapid descent. No single method dominates across all benchmarks. The theoretical choice should be guided by: (i) availability of gradient/Hessian, (ii) condition number, (iii) dimension $d$, and (iv) convexity structure.

\item \textbf{Adaptive step sizes are robust but not universally optimal.} Adagrad and RMSprop require no knowledge of $f^{\star}$ or $L$ and are stable across all benchmarks. However, their theoretical rates --- Adagrad's $O(1/\sqrt{k})$ for non-convex problems and RMSprop's heuristic adaptation --- do not match the optimal $O(1/k^2)$ Nesterov rate for convex problems. In practice, Adam (which combines both moment estimates) is the dominant choice for stochastic neural network training, where the gradient noise and non-convexity make exact convergence rates irrelevant and robustness paramount.

\item \textbf{$f^{\star}$ misspecification is catastrophic for Polyak.} The Polyak step is theoretically optimal when $f^{\star}$ is exact, achieving the best convergence on Rosenbrock ($f = 0.009$, 87\% better than GD). However, even a moderate error in $f^{\star}$ (Benchmarks A and B use $f^{\star}=0$ vs.\ true minima of $0.4826$ and $0.7244$) causes the effective step size $\alpha_k \to (f^{\star}_{\text{true}} - f^{\star}_{\text{used}})/\varepsilon \to \infty$ as $\|g_k\| \to 0$, inducing divergence. This dependence on oracle knowledge of $f^{\star}$ limits Polyak's practical applicability beyond settings where $f^{\star}$ is analytically known (e.g., classification with known Bayes error).

\item \textbf{Second-order methods offer dramatic speedups on near-quadratic problems.} Newton's method reduces iterations by $4\times$ on Benchmark B and achieves $5.4\times$ lower loss than GD on Benchmark C with $4\times$ fewer iterations. However, the per-iteration cost is $O(d^3)$ (vs.\ $O(d)$ for first-order methods) due to Hessian formation and linear system solving. This cost is justified in low dimensions ($d \leq 10^3$) but becomes completely impractical for modern deep learning ($d \sim 10^7$--$10^9$). Quasi-Newton methods (L-BFGS) bridge this gap by maintaining a low-rank Hessian approximation at $O(md)$ cost for $m$ history vectors.

\item \textbf{Feasibility enforcement depends on method design.} Projected GD provides hard constraint satisfaction from the very first iteration at $O(1)$ projection cost (for box constraints). Penalty methods are simpler to implement but require careful $\lambda$ tuning: our experiments show $\lambda \geq 0.62$ (matching the constraint gradient magnitude at $x_0$) is necessary for feasibility enforcement in this problem, consistent with the KKT analysis. The augmented Lagrangian method ($F_{\lambda, \nu}(x) = f(x) + \nu g(x) + \frac{\lambda}{2}\max(0, g(x))^2$) addresses the ill-conditioning of large $\lambda$ by updating dual variables $\nu$, achieving exact feasibility at moderate penalty values.

\item \textbf{Derivative-free methods trade gradient quality for evaluation simplicity.} Finite differences with small $\delta$ ($= 0.05$, $\approx 2\%$ relative error) recover near-exact gradient information at $d+1 = 3$ function evaluations per step. Random search (Nesterov method, 2 evaluations per step) is dimension-independent but provides only a noisy 1D gradient projection. Grid search guarantees global optimality up to grid resolution ($\Delta x \approx 0.073$ here) but requires $N^d$ evaluations --- $3025$ here vs.\ $\sim 10^{17}$ for $d=10$. Frank--Wolfe avoids the projection oracle entirely through the LMO, making it preferable when the constraint set has exploitable structure (e.g., nuclear norm balls in matrix completion, polytopes in combinatorial optimisation).
\end{enumerate}

\subsection{Limitations of This Study}

Several aspects were not investigated but are relevant in practice: (1) \emph{Line search}: all methods here use fixed step sizes or simple schedules; backtracking line search would eliminate the need for manual $\alpha$ tuning and could substantially improve convergence. (2) \emph{Stochasticity}: Questions 1--3 and 6 use full-batch gradient evaluations; the stochastic variants (SGD, mini-batch Adam, stochastic L-BFGS) are more relevant for large-scale machine learning. (3) \emph{Non-convex landscapes}: Rosenbrock is the only non-quadratic benchmark; deep learning optimisation involves saddle points, local minima, and flat plateaus that these benchmarks do not capture. (4) \emph{Second-order convergence certification}: Newton's method and Nesterov are not run to convergence on Rosenbrock; more iterations would reveal whether they ultimately achieve the true minimum or stagnate in the curved valley.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]

        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
