\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C). Six categories of methods are investigated across six questions: (Q1)~adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), (Q2)~momentum-based and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD), (Q3)~Newton's method with second-order local approximations, (Q4)~derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead, grid search), (Q5)~constrained optimisation via projected gradient descent and penalty methods, and (Q6)~the Frank--Wolfe algorithm for linear-constrained problems.

Key quantitative findings include: the Polyak step achieves the best Rosenbrock convergence ($f=0.0094$ in 120 iterations) when $f^{\star}$ is known exactly, but diverges catastrophically when $f^{\star}$ is misspecified; Newton's method solves the quadratic loss exactly in \emph{one} iteration ($f=0.4826$) and reaches $f=0.686$ on Rosenbrock in only 20 iterations vs GD's $f=3.73$ in 80 iterations; Nesterov momentum achieves $f=0.479$ on Rosenbrock (150 iterations) vs GD's $f=3.39$; and Frank--Wolfe with conservative mixing ($\beta=0.985$) converges to $f=0.0046$ for the interior optimum. Each method is evaluated against theoretical convergence guarantees from course lectures, with particular attention to how problem geometry (curvature, condition number) interacts with algorithmic properties.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. Starting points are fixed throughout: $\theta_0 = [0,0]^T$ for Benchmark~A, $x_0 = [-1,4]^T$ for Benchmark~B, and $x_0 = [-1,1]^T$ for Benchmark~C.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and $d=2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
The Hessian is constant: $H = \frac{1}{m}X^T X \succ 0$. With $m \gg d$, $H$ is well-conditioned by the law of large numbers: eigenvalues concentrate near 1, giving condition number $\kappa(H) \approx 1$--$2$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. The noise floor is $J^{\star} = J(\theta_{\text{OLS}}) \approx \frac{1}{2}\sigma^2 = 0.4826$ (noise variance, not zero), so $\theta_{\text{OLS}} \neq \theta^{\star}$ in general.

\textbf{Smoothness and strong convexity.} The smallest and largest eigenvalues of $H$ satisfy $\mu \leq \lambda_{\min}(H) \leq \lambda_{\max}(H) \leq L$, giving a condition number $\kappa = L/\mu$. The Lipschitz constant of the gradient is $L = \lambda_{\max}(H)$. For the near-isotropic design used here, $\kappa \approx 1.04$.

\subsection{Benchmark B: Toy Neural Network Loss}

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \operatorname{diag}(2 - \sin(x_1),\ 10)$. Since $\sin(x_1) \in [-1,1]$, the $x_1$-curvature varies in $[1, 3]$, while $x_2$-curvature is fixed at 10. The condition number satisfies $\kappa \in [10/3, 10]$ depending on location. The unconstrained minimum satisfies $2(x_1^{\star}-1) + \cos(x_1^{\star}) = 0$, giving $x_1^{\star} \approx 0.582$ and $x_2^{\star} = 2.0$ with $f^{\star} \approx 0.7244$.

\subsection{Benchmark C: Rosenbrock Function}

\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is:
\begin{equation}
H(1,1) = \begin{bmatrix} 2 + 1200\cdot 1 - 400\cdot 1 & -400 \\ -400 & 200 \end{bmatrix} = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $\lambda_1 \approx 1001.6$ and $\lambda_2 \approx 0.40$, giving $\kappa \approx 2508$. Away from the optimum, the condition number grows substantially. This extreme ill-conditioning, combined with the narrow curved valley $x_2 \approx x_1^2$, makes Rosenbrock a canonical stress test for optimisation algorithms.

\subsection{Benchmark D: Constrained Feasible Set}

For constrained problems (Questions~5 and~6):
\begin{equation}
\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

For a function $f$ that is $L$-smooth (gradient Lipschitz with constant $L$) and $\mu$-strongly convex, constant-step gradient descent with $\alpha = 1/L$ satisfies the linear convergence bound:
\begin{equation}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k (f(x_0) - f^{\star}).
\end{equation}
The ratio $\mu/L = 1/\kappa$ determines convergence speed: well-conditioned problems ($\kappa \approx 1$) converge rapidly, while ill-conditioned problems ($\kappa \gg 1$, as in Rosenbrock) converge slowly. Adaptive methods seek to mitigate the effect of curvature variation by adjusting the step size per-coordinate or per-iteration.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive step derived from quadratic model}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{proposition}[Polyak convergence]
For a convex $L$-smooth function with known $f^{\star}$, the Polyak step satisfies $f(x_{k+1}) - f^{\star} \leq \frac{1}{2}\left(1 - \frac{1}{2L} \cdot \frac{\|g_k\|^2}{f(x_k)-f^{\star}}\right) (f(x_k) - f^{\star})$, achieving finite-time convergence for strongly convex functions. Crucially, \emph{exact knowledge of $f^{\star}$ is required}.
\end{proposition}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative squared gradient, per-coordinate}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's per-coordinate effective step size is $\alpha_0 / \sqrt{\sum_{j=0}^{k} g_{j,i}^2}$: coordinates with large historical gradients receive smaller steps. This is particularly beneficial for sparse gradients. However, the monotonically growing denominator causes the effective step to vanish, eventually halting progress.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's cumulative sum with an exponential moving average (decay $\beta = 0.9$), preventing the effective step from vanishing. The $k$-step effective window has approximate length $1/(1-\beta) = 10$ iterations, so the method adapts to recent curvature rather than the full history.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Velocity accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

For a quadratic objective with Hessian eigenvalues in $[\mu, L]$, the Heavy Ball method with optimal parameters $\alpha^{\star} = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2}$ and $\beta^{\star} = \left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^2$ achieves convergence rate $\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$, which is asymptotically faster than GD's rate $\left(\frac{\kappa-1}{\kappa+1}\right)$. For non-quadratic objectives, Heavy Ball lacks global convergence guarantees.

\begin{remark}[Optimal Heavy Ball parameters for Benchmark~C]
For Benchmark~C at the optimum: $L \approx 1001.6$, $\mu \approx 0.40$, $\kappa \approx 2508$. The theoretically optimal Heavy Ball parameters are:
\begin{equation}
\alpha^{\star} = \frac{4}{(\sqrt{1001.6}+\sqrt{0.40})^2} = \frac{4}{(31.65 + 0.632)^2} \approx \frac{4}{1042.1} \approx 0.00384,
\qquad
\beta^{\star} = \left(\frac{31.65 - 0.632}{31.65 + 0.632}\right)^2 \approx (0.961)^2 \approx 0.924.
\end{equation}
The experiment uses $\alpha = 0.0008$, $\beta = 0.86$, which are more conservative---chosen empirically to avoid instability in the highly non-quadratic Rosenbrock valley. On strictly quadratic problems the optimal parameters would give convergence rate $\approx 0.924$ per iteration (vs GD's $(1-1/2508) \approx 0.9996$), a $\sim 50\times$ speedup.
\end{remark}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$ in 120 iters). Otherwise, Adagrad and Heavy Ball match or exceed GD baseline performance, consistent with their theoretical advantages on ill-conditioned problems.

\item \textbf{Stability:} GD baseline is most stable (guaranteed by $\alpha < 2/L$). Polyak is \emph{conditionally} stable: stable only when $f^{\star}$ is correctly specified. Adagrad is unconditionally stable due to its monotonically decreasing step, though this may cause premature stagnation. RMSprop can oscillate if $\alpha_0$ is too large. Heavy Ball \emph{lacks global convergence guarantees on non-quadratic objectives}: Polyak and Lessard et al.\ showed it can cycle or diverge on non-convex functions, despite its optimality on quadratics. This theoretical limitation does not manifest here since Benchmarks B and C are ``approximately quadratic'' near the optimum, but would be relevant on more complex landscapes.

\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging that tracks recent curvature. Heavy Ball exhibits characteristic damped oscillation (``ringing'') from momentum accumulation near the optimum.

\item \textbf{Practical implication:} In practice, $f^{\star}$ is rarely known, severely limiting Polyak's applicability. RMSprop and Adagrad are more robust for general problems. The Heavy Ball method with well-tuned momentum achieves near-optimal rates on Benchmark~C without knowledge of $f^{\star}$.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Gradient at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov, 1983]
For a convex $L$-smooth function, the accelerated gradient method with step $\alpha = 1/L$ and momentum schedule $\beta_k = (k-1)/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} = O(1/k^2),
\end{equation}
which is optimal among first-order methods for smooth convex optimisation.
\end{theorem}

\begin{remark}[Iteration complexity comparison]
To reduce the optimality gap to $\varepsilon$, GD requires $k = O(\kappa \log(1/\varepsilon))$ iterations while Nesterov requires $k = O(\sqrt{\kappa} \log(1/\varepsilon))$. For Benchmark~C ($\kappa \approx 2508$) and $\varepsilon = 0.01$:
\begin{align}
k_{\mathrm{GD}} &\approx \kappa \log(1/\varepsilon) = 2508 \times \log(100) \approx 11\,550,\\
k_{\mathrm{Nest}} &\approx \sqrt{\kappa} \log(1/\varepsilon) = \sqrt{2508} \times \log(100) \approx 231.
\end{align}
The $\sqrt{\kappa}$-vs-$\kappa$ iteration ratio gives a $\sqrt{2508} \approx 50$-fold speedup---explaining why Nesterov achieves $f = 0.479$ in 150 iterations while GD reaches only $f = 3.39$.
\end{remark}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at the tentative next point $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This ``corrective'' gradient step enables the $O(1/k^2)$ rate, compared to $O(1/k)$ for standard GD.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (RMSprop)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam (Kingma \& Ba, 2015) unifies momentum ($m_t$) with RMSprop-style per-coordinate adaptive scaling ($v_t$). The bias correction terms $(1 - \beta_1^t)^{-1}$ and $(1 - \beta_2^t)^{-1}$ counteract the initialisation at zero, which would otherwise bias early iterates towards zero. The effective step size per coordinate is approximately $\alpha \cdot \hat{m}_t / \sqrt{\hat{v}_t} \approx \alpha \cdot \text{sign}(g_t)$ for large $t$, giving near-constant magnitude updates per coordinate. In convex settings, Adam has $O(\sqrt{T}/T) = O(1/\sqrt{T})$ convergence in the online (stochastic) case; in deterministic smooth convex settings, the rate analysis is less clean than for Nesterov (no $O(1/k^2)$ guarantee is known for general Adam). In practice, Adam often outperforms Nesterov on noisy (deep learning) problems due to its per-coordinate normalisation, but Nesterov's theoretical $O(1/k^2)$ advantage is visible on the smooth deterministic benchmarks used here.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $g_{\mathcal{B}}$ is an unbiased estimator of the full gradient: $\mathbb{E}[g_{\mathcal{B}}] = \nabla J(\theta)$. Its variance satisfies $\operatorname{Var}[g_{\mathcal{B}}] = \frac{1}{b} \operatorname{Var}[g_{\text{single}}]$, so larger batches produce lower-variance gradient estimates. The total number of gradient steps per epoch is $m/b$: with $b = 5$, each epoch has 200 updates; with $b = 40$, only 25 updates.

\subsection{Local Approximations of $g(x) = x^4$}

Consider approximating $g(x) = x^4$ at $x_0 = 0.25$ using Taylor expansions. We have $g'(x) = 4x^3$ and $g''(x) = 12x^2$, so $g(0.25) = 0.0039$, $g'(0.25) = 0.0625$, $g''(0.25) = 0.75$.

\textbf{First-order (tangent line) approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625(x - 0.25).
\end{equation}
This captures only the local slope; it is unbounded and has no finite minimiser, illustrating why first-order methods must use bounded step sizes.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
This has a unique minimiser at $x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 = 0.1667$, which constitutes the Newton update from $x_0$.

\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$. The first-order approximation (tangent) captures only the local slope and diverges rapidly. The second-order approximation (parabola) captures local curvature and provides a much better fit in a neighbourhood of $x_0$. The Newton update minimises this parabola, yielding a more direct step towards the optimum than a fixed-step gradient descent update.}

\subsection{Newton's Method with Damping}

\begin{algorithm}[H]
\caption{Newton's Method with Hessian Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov regularisation ensures $H_k \succ 0$}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (descent direction when $H_k \succ 0$)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Quadratic convergence of Newton's method]
\label{thm:newton_quad}
Let $f$ be twice continuously differentiable with $\nabla^2 f$ $M$-Lipschitz near $x^{\star}$, and let $\mu = \lambda_{\min}(\nabla^2 f(x^{\star})) > 0$. Then for $x_0$ sufficiently close to $x^{\star}$ and $\alpha = 1$:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu} \|x_k - x^{\star}\|^2.
\end{equation}
This quadratic convergence rate implies that the number of correct decimal digits roughly doubles at each iteration.
\end{theorem}

\begin{remark}[Quadratic convergence radius (Kantorovitch)]
The quadratic convergence in Theorem~\ref{thm:newton_quad} holds only within a neighbourhood of $x^{\star}$ where $\|x_0 - x^{\star}\| \leq r^{\star} = \mu^2 / M$ (here $M$ is the Lipschitz constant of $\nabla^2 f$). Outside this neighbourhood, damped Newton ($\alpha < 1$) is required for global convergence. For Rosenbrock, the Hessian changes rapidly away from $(1,1)$, so the effective quadratic convergence region is small---hence the heavily damped step $\alpha = 0.22$ used in the experiment. Near the minimum, full Newton steps ($\alpha = 1$) would recover quadratic convergence.
\end{remark}

\begin{remark}[One-step convergence on quadratics]
For a quadratic objective $f(\theta) = \frac{1}{2m}\|X\theta - y\|^2 = \frac{1}{2}\theta^T H\theta - b^T\theta + c$, where $H = X^TX/m$ and $b = X^Ty/m$, the gradient is $\nabla f(\theta) = H\theta - b$. The Newton update with $\alpha = 1$ gives:
\begin{equation}
\theta_1 = \theta_0 - H^{-1}\nabla f(\theta_0) = \theta_0 - H^{-1}(H\theta_0 - b) = H^{-1}b = \theta_{\text{OLS}},
\end{equation}
reaching the exact minimiser in a single iteration, \emph{regardless of condition number or starting point}. This is because the quadratic model is globally exact (not just locally accurate), so Newton's exact quadratic minimisation yields the global solution immediately.
\end{remark}

Settings: GD uses 80 iterations; Newton uses 20 iterations with $\lambda = 10^{-8}$ (negligible regularisation).

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark~B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta} = \frac{\partial f}{\partial x_i}(x) + \underbrace{\frac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x)}_{\text{truncation error}} + O(\delta^2).
\end{equation}
The total error has two competing components:
\begin{itemize}[nosep]
\item \emph{Truncation error}: $O(\delta)$---decreases as $\delta \to 0$.
\item \emph{Rounding error}: $O(\varepsilon_{\text{mach}}/\delta)$---increases as $\delta \to 0$.
\end{itemize}
The optimal $\delta$ balances these: $\delta^{\star} \approx \sqrt{2\varepsilon_{\text{mach}} |f(x)|/|f''(x)|} \approx \sqrt{\varepsilon_{\text{mach}}} \approx 1.5 \times 10^{-8}$ for typical function values. The experiments test $\delta = 0.05$ (moderately good; truncation dominates but is manageable) and $\delta = 0.8$ (poor; large truncation error distorts the gradient direction). Both use step size $\alpha = 0.08$.

\begin{remark}[Central differences give higher-order accuracy]
The \emph{central difference} approximation $\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}$ has truncation error $O(\delta^2)$ (the $O(\delta)$ term cancels by symmetry) at the cost of $2d$ function evaluations instead of $d+1$. Its total error is $O(\delta^2) + O(\varepsilon_{\text{mach}}/\delta)$, with optimal $\delta^{\star}_{\text{cd}} \approx \varepsilon_{\text{mach}}^{1/3} \approx 6 \times 10^{-6}$---larger and thus less susceptible to rounding, and achieving a minimum error of $O(\varepsilon_{\text{mach}}^{2/3}) \approx 10^{-11}$ vs forward-differences' $O(\varepsilon_{\text{mach}}^{1/2}) \approx 10^{-8}$. For the same $\delta = 0.05$, central differences would reduce gradient error by a factor of $\sim \delta = 0.05 / \delta^{\star}_{\text{fd}} \approx 3\times 10^6$, making them strongly preferable when $f$ evaluations are cheap.
\end{remark}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Gaussian smoothing interpretation]
The Nesterov random search estimate can be interpreted as the gradient of a Gaussian-smoothed objective:
$\hat{g}_k \approx \nabla f_{\delta}(x_k)$ where $f_{\delta}(x) = \mathbb{E}_{u \sim \mathcal{N}(0,I)}[f(x + \delta u)]$. As $\delta \to 0$, $f_{\delta}(x) \to f(x)$ and $\nabla f_{\delta}(x) \to \nabla f(x)$. The method requires only \emph{two} function evaluations per iteration regardless of dimension $d$, making it dimension-free in per-iteration cost (at the expense of higher variance per estimate). For convex $L$-smooth objectives, the expected convergence rate is $O(d/(\delta^2 k))$---the dimension $d$ appears explicitly because each step estimates only a 1D projection of the gradient, requiring $O(d)$ steps to approximate a full gradient descent step. In $d = 2$, this overhead is modest; it grows prohibitively in high dimensions.
\end{remark}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1=3$ vertices in $\mathbb{R}^2$ and iteratively modifies it through four operations using parameters $\alpha_r=1$ (reflection), $\gamma=2$ (expansion), $\rho=0.5$ (contraction), $\sigma=0.5$ (shrink):

\begin{enumerate}[nosep]
\item \textbf{Reflect:} $x_r = \bar{x} + \alpha_r(\bar{x} - x_{\text{worst}})$ where $\bar{x}$ is centroid of best $n$ vertices.
\item \textbf{Expand:} if $x_r$ is the best point, try $x_e = \bar{x} + \gamma(\bar{x} - x_{\text{worst}})$.
\item \textbf{Contract:} if $x_r$ is not better than second-worst, try contraction towards centroid.
\item \textbf{Shrink:} if contraction fails, contract all vertices towards the best vertex.
\end{enumerate}

No gradient information is required. Settings: initial simplex step $= 0.35$, 160 iterations from $x_0 = [-1,1]^T$.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Setup and Optimality Conditions}

Minimise $f(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)$ subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$. The feasible set is $\mathcal{X} = \{x : x_1 \geq 0.5\}$.

\textbf{KKT conditions.} For the constrained problem $\min_x f(x)$ s.t.\ $-x_1 + 0.5 \leq 0$, the Karush--Kuhn--Tucker (KKT) necessary conditions at a minimiser $x^{\star}$ are:
\begin{align}
\nabla f(x^{\star}) + \mu^{\star} \nabla(-x_1 + 0.5) &= 0 \quad \Rightarrow \quad \nabla f(x^{\star}) - \mu^{\star} e_1 = 0, \\
\mu^{\star}(-x_1^{\star} + 0.5) &= 0 \quad \text{(complementary slackness)}, \\
-x_1^{\star} + 0.5 &\leq 0 \quad \text{(primal feasibility)}, \\
\mu^{\star} &\geq 0 \quad \text{(dual feasibility)}.
\end{align}
Since the unconstrained minimiser $x^{\star}_{\text{unc}} = (0.582, 2.0)$ satisfies $x_1^{\star} = 0.582 > 0.5$, it lies strictly inside the feasible set, so $\mu^{\star} = 0$ (inactive constraint) and the constrained and unconstrained minima coincide at $f^{\star} = 0.7244$.

\textbf{Second-order sufficiency.} Since the constraint is inactive at $x^{\star}$ ($\mu^{\star} = 0$), the Lagrangian reduces to $\mathcal{L}(x,0) = f(x)$, and the second-order sufficiency condition requires $\nabla^2 f(x^{\star}) \succ 0$ on $\mathbb{R}^2$. At $x^{\star} = (0.582, 2.0)$:
\begin{equation}
\nabla^2 f(x^{\star}) = \begin{pmatrix} 2 - \sin(0.582) & 0 \\ 0 & 10 \end{pmatrix} = \begin{pmatrix} 2 - 0.550 & 0 \\ 0 & 10 \end{pmatrix} = \begin{pmatrix} 1.45 & 0 \\ 0 & 10 \end{pmatrix} \succ 0.
\end{equation}
Both eigenvalues are positive, confirming that $x^{\star}$ is a strict local (and global, since $f$ is strongly convex) minimiser of the constrained problem. The Linear Independence Constraint Qualification (LICQ) also holds trivially since there is a single active constraint with gradient $\nabla(-x_1+0.5) = -e_1 \neq 0$.

\subsection{Algorithms}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $(z_1, z_2) \to (\max(0.5, z_1), z_2)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space $\{x : x_1 \geq 0.5\}$, the projection is $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)$---a simple clipping operation. For more complex constraint sets (general polytopes, second-order cones), the projection itself requires solving a QP, adding significant per-iteration cost.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda\, \max(0,\, 0.5 - x_1)$ \Comment{$L_1$ exact penalty}
    \State Compute $\nabla_x F_{\lambda}(x_k)$: gradient of $f$ plus $-\lambda e_1$ if $x_1 < 0.5$, else $0$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty term $\lambda \max(0, 0.5 - x_1)$ is an $L_1$ (non-differentiable) penalty. For sufficiently large $\lambda > \mu^{\star}$ (the dual optimal multiplier), the unconstrained minimiser of $F_{\lambda}$ coincides with the constrained minimiser---this is the \emph{exact penalty} property. Since $\mu^{\star} = 0$ here (inactive constraint), any $\lambda > 0$ suffices for exactness in theory, though larger $\lambda$ induces steeper gradients requiring smaller $\alpha$.

Settings: Projected GD ($\alpha = 0.08$, 100 iters); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ and $\alpha = 0.05$ ($\lambda \leq 1.8$) or $\alpha = 0.03$ ($\lambda = 4.5$); Unconstrained GD ($\alpha = 0.07$, 100 iters).

\textbf{Discussion.} Since the unconstrained optimum is feasible (KKT multiplier $\mu^{\star} = 0$), all methods converge to the same point. The key distinction is how the infeasible starting point is handled:

\begin{itemize}[nosep]
\item \textbf{Projected GD} enforces feasibility as a hard constraint every iteration. This is the fundamental advantage: the iterate $x_k \in \mathcal{X}$ for all $k \geq 1$. The cost is a projection oracle, trivial here but potentially expensive for complex $\mathcal{X}$.
\item \textbf{Penalty methods} trade exact feasibility for algorithmic simplicity (only a modified gradient is needed). The weight $\lambda$ controls the trade-off: small $\lambda$ under-penalises infeasibility; large $\lambda$ enforces feasibility faster but requires smaller $\alpha$ for stability (here $\alpha$ is reduced from 0.05 to 0.03 at $\lambda=4.5$). In the limit $\lambda \to \infty$, the penalty method approaches the constrained problem, but the conditioning of the penalised problem worsens.
\item \textbf{Unconstrained GD} is equivalent to a penalty method with $\lambda = 0$ and converges to the feasible optimum here only because $\mu^{\star} = 0$.
\item \textbf{Augmented Lagrangian} (not implemented) would improve on the pure penalty method by also maintaining a dual variable estimate $\mu_k$, solving $\min_x \mathcal{L}_{\rho}(x, \mu_k)$ at each outer iteration and updating $\mu_{k+1} = \mu_k + \rho \cdot \max(0, 0.5 - x_1)$. This dual variable ``warm starts'' the constraint enforcement and avoids the ill-conditioning associated with large $\rho$ (equivalent to $\lambda$ in pure penalty), achieving faster practical convergence.
\end{itemize}

\subsection{Linear Programme over the Feasible Set}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a bounded rectangle, hence a polytope).

\begin{proposition}[Fundamental theorem of LP]
The minimum of a linear objective $\min_{x \in \mathcal{X}} a^T x$ over a bounded polytope $\mathcal{X}$ is always attained at a vertex of $\mathcal{X}$.
\end{proposition}

\textbf{Problem:} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$. Since $a_1 = 1 > 0$ and $a_2 = 2 > 0$, the objective decreases in both the $-x_1$ and $-x_2$ directions, so the minimum is at the lower-left vertex:
\begin{equation}
x^{\star} = \begin{pmatrix} 0.5 \\ -5 \end{pmatrix}, \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}
The four vertices of $\mathcal{X}$ have objective values $\{-9.5, -4.5, 25, 30\}$, confirming $x^{\star}$ is the global minimum.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (fixed step)}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Frank--Wolfe convergence]
For a convex $L$-smooth function on a compact convex set $\mathcal{X}$ with diameter $D = \max_{x,y \in \mathcal{X}} \|x - y\|$, and using the standard step size $\gamma_k = 2/(k+2)$ (i.e., $\beta_k = k/(k+2)$), Frank--Wolfe satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2} = O(1/k).
\end{equation}
This $O(1/k)$ rate is slower than projected GD's $O(1/k^2)$ for strongly convex objectives, but Frank--Wolfe avoids the projection oracle entirely, replacing it with a linear minimisation oracle (LMO).
\end{theorem}

For box constraints, the LMO decomposes coordinate-wise in $O(d)$:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}

The experiments use a fixed $\beta$ (not the standard $\gamma_k$ schedule): $x_{k+1} = \beta x_k + (1-\beta) z_k$. This corresponds to a fixed step of size $(1-\beta)$ towards the LP vertex at each iteration. Smaller $(1-\beta)$ means more conservative mixing.

\begin{remark}[Duality gap and diameter]
A natural stopping criterion for Frank--Wolfe is the \emph{Frank--Wolfe gap}: $g_k = \nabla f(x_k)^T (x_k - z_k) \geq 0$. Since $z_k$ minimises the linear approximation over $\mathcal{X}$, the gap upper-bounds the primal optimality gap: $f(x_k) - f^{\star} \leq g_k$. The gap is computable from the LMO output without knowledge of $f^{\star}$. For the box $\mathcal{X}$, the diameter is $D = \|(5 - 0.5, 10 - (-5))\| = \|(4.5, 15)\| \approx 15.66$, so the FW convergence bound gives $f(x_k) - f^{\star} \leq 2LD^2/(k+2) \approx 2L \cdot 245/(k+2)$.
\end{remark}

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits characteristic zig-zag behaviour: $z_k$ repeatedly jumps between extreme vertices of $\mathcal{X}$, and the convex combination averages these jumps towards the interior. This ``zig-zag'' is a signature of FW on polytopes; the $O(1/k)$ theoretical rate reflects the slow averaging. The fixed-$\beta$ schedule used here differs from the theoretically optimal $\gamma_k = 2/(k+2)$ schedule: for an interior optimum, the fixed schedule can oscillate indefinitely if $\beta$ is too small.

\textbf{Boundary optimum:} Frank--Wolfe is better suited to boundary optima because $z_k$ frequently selects the vertex closest to the constrained optimum, actively pushing the iterate towards the boundary. The method naturally exploits the LP structure: whereas projected GD would need to project explicitly, FW implicitly steers iterates towards the boundary through vertex selection. The residual gap ($f = 0.364$ vs $f^{\star} = 0.25$ after 140 iterations) would decrease with more iterations or with the standard $\gamma_k = 2/(k+2)$ step schedule.

\textbf{Comparison of step schedules.} The standard FW step size is $\gamma_k = 2/(k+2)$ (i.e., $\beta_k = k/(k+2)$), which satisfies the $O(1/k)$ convergence guarantee. The fixed-$\beta$ schedule used here corresponds to a constant step $(1-\beta)$ towards the LP vertex at every iteration. For $\beta = 0.985$, $(1-\beta) = 0.015$ is smaller than $\gamma_k$ for $k \lesssim 130$ and larger for $k \gtrsim 130$, explaining why the fixed schedule can match or outperform the standard schedule in the early-to-mid phase but may stagnate later. For long runs, the standard decaying schedule is preferred to guarantee the $O(1/k)$ rate.

\textbf{Comparison with projected GD:} Frank--Wolfe trades convergence rate ($O(1/k)$ vs $O(1/k^2)$ for strongly convex) for projection-oracle-free operation. This is advantageous when projections are expensive (e.g., nuclear norm ball, flow polytopes) but is slower when projections are cheap (e.g., box constraints). In the experiments here, the box projection is $O(d)$, so projected GD would be preferable for accuracy; Frank--Wolfe is chosen to demonstrate the LP subproblem structure.

\begin{remark}[Away-step Frank--Wolfe and linear convergence]
The standard FW algorithm is limited to $O(1/k)$ convergence even for strongly convex objectives because the iterate can become ``stuck'' as a convex combination of vertices. The \emph{away-step} Frank--Wolfe variant (Guélat \& Marcotte, 1986; Lacoste-Julien \& Jaggi, 2015) additionally considers a step \emph{away} from the vertex with the largest weight: $x_{k+1} = x_k + \gamma_{\text{away}}(x_k - v_k)$, where $v_k = \arg\max_{v \in S_k} \nabla f(x_k)^T v$ over the current active vertex set $S_k$. For strongly convex objectives over polytopes, away-step FW achieves \emph{linear} convergence $O\!\left((1-\mu/(LD^2 \psi))^k\right)$, where $\psi$ is the pyramidal width of $\mathcal{X}$. This closes the gap with projected GD at the cost of tracking the active vertex set.
\end{remark}

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

\subsection{Summary of Results}

This report investigated six classes of optimisation algorithms across three benchmark problems of increasing difficulty---from well-conditioned quadratics (Benchmark~A, $\kappa \approx 1$) through moderately anisotropic nonlinear functions (Benchmark~B, $\kappa \leq 10$) to the severely ill-conditioned Rosenbrock function (Benchmark~C, $\kappa \approx 2508$).

\textbf{Question 1 (Adaptive step-size methods):} The Polyak step achieves the best performance when $f^{\star}$ is exactly known, reaching $f = 0.0094$ on Rosenbrock in 120 iterations, but catastrophically diverges under misspecification ($f = 2.72$ on Benchmark~A vs true $f^{\star} = 0.4826$). This motivates the use of robust adaptive methods: Adagrad and Heavy Ball achieve comparable accuracy to GD on well-conditioned problems while offering modest advantages on ill-conditioned ones.

\textbf{Question 2 (Momentum and stochastic methods):} Nesterov acceleration achieves a 7$\times$ improvement over GD on Rosenbrock ($f = 0.479$ vs $3.39$ in 150 iterations), consistent with its theoretical $O(1/k^2)$ convergence advantage for convex objectives. Adam is effective on anisotropic problems (Benchmarks~A, B) due to per-coordinate normalisation, but its coordinate-aligned scaling does not exploit the curved Rosenbrock valley geometry. Mini-batch SGD reveals the variance-batch size trade-off: smaller batches converge faster per epoch but exhibit higher oscillation near the noise floor.

\textbf{Question 3 (Newton's method):} Second-order information provides dramatic advantages. On Benchmark~A, Newton solves the problem exactly in one iteration, a theoretical property of full Newton steps on quadratic objectives. On Rosenbrock, Newton achieves $f = 0.686$ in only 20 iterations (vs GD's $f = 3.73$ in 80 iterations) using a heavily damped step ($\alpha = 0.22$). The damping is essential: the Rosenbrock Hessian varies rapidly, invalidating the quadratic model over large steps.

\textbf{Question 4 (Derivative-free methods):} Finite-difference gradient descent with small $\delta = 0.05$ closely approximates exact gradient descent on Benchmark~B, confirming the $O(\delta)$ truncation error is acceptable at moderate scales. Nesterov random search converges slowly due to high-variance one-dimensional gradient projections but scales well to high dimensions. Nelder--Mead navigates the Rosenbrock valley without gradient information by adapting its simplex geometry; grid search provides global coverage at the cost of exponential scaling in dimension.

\textbf{Question 5 (Constrained optimisation):} For the half-space constraint $x_1 \geq 0.5$, projected GD enforces hard feasibility from iteration 1, while penalty methods gradually steer iterates toward feasibility with speed proportional to $\lambda$. Since the unconstrained optimum is interior to the feasible set ($x_1^{\star} = 0.582 > 0.5$, KKT multiplier $\mu^{\star} = 0$), all methods converge to the same optimum; the distinction is purely in how the infeasible starting point is handled.

\textbf{Question 6 (Frank--Wolfe):} The FW algorithm avoids projection by solving a linear minimisation subproblem at each iteration. For the interior optimum, $\beta = 0.985$ (conservative mixing) outperforms $\beta = 0.90$ by avoiding oscillatory vertex-jumping. For the boundary optimum, FW naturally steers iterates to the boundary through LP vertex selection, converging to $f = 0.364$ in 140 iterations (gap from $f^{\star} = 0.25$ reducible with more iterations or the standard $\gamma_k = 2/(k+2)$ schedule).

The experiments corroborate several fundamental results from the course:
\begin{enumerate}[nosep]
\item \textbf{Condition number dominates convergence:} GD's step size on Rosenbrock must satisfy $\alpha < 2/L \approx 2/1001.6 \approx 0.002$, yielding a convergence rate $(1-\mu/L)^k \approx (1 - 1/2508)^k$---requiring $\sim 5780$ iterations for one order of magnitude of reduction. \item \textbf{Nesterov's optimal rate:} The $O(1/k^2)$ theory translates to measurable advantage over GD ($O(1/k)$) on Rosenbrock, consistent with the $O(\sqrt{\kappa})$ vs $O(\kappa)$ iteration complexity comparison.
\item \textbf{Newton's affine invariance:} Newton's method is invariant to linear changes of coordinates, making its convergence independent of condition number near the optimum---explaining its dramatic advantage on ill-conditioned problems.
\item \textbf{Exact penalty principle:} For the constrained problem in Q5, the $L_1$ penalty with any $\lambda > 0$ is exact (since $\mu^{\star} = 0$), so all penalty weights eventually converge to the true optimum, as observed experimentally.
\item \textbf{Frank--Wolfe $O(1/k)$ rate:} The sub-linear convergence of FW is visible in the evolution plots, and the fixed-$\beta$ schedule's oscillatory behaviour is consistent with the theoretical requirement for a decreasing step schedule for $O(1/k)$ guarantees.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
