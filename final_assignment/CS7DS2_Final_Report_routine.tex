\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms across six
method families, evaluated on three benchmark problems of increasing difficulty: a linear-regression
quadratic loss (Benchmark A), a sinusoidally perturbed quadratic (Benchmark B), and the
Rosenbrock function (Benchmark C). The methods investigated are: (Q1) adaptive step-size
methods --- Polyak, Adagrad, RMSprop, and Polyak-Momentum (Heavy Ball); (Q2) accelerated and
stochastic methods --- Nesterov momentum and Adam, together with mini-batch SGD; (Q3)
Newton's method with Hessian regularisation; (Q4) derivative-free approaches --- finite
differences (forward and central), Nesterov random search, Nelder--Mead simplex, and grid
search; (Q5) constrained optimisation via projected gradient descent and quadratic penalty
methods; and (Q6) the Frank--Wolfe (conditional gradient) algorithm for box-constrained
problems together with an analysis of a linear programme.  Key findings include: (i) Polyak
step size achieves superior convergence when $f^{\star}$ is exactly known but diverges when
it is misspecified; (ii) Nesterov acceleration achieves the theoretically optimal $O(1/k^2)$
rate and dominates all first-order methods on the challenging Rosenbrock problem; (iii)
Newton's method converges in a single iteration on the quadratic benchmark, confirming its
quadratic convergence guarantee; (iv) finite-difference quality degrades for $\delta \gg
\sqrt{\varepsilon_{\mathrm{mach}}}$; (v) projected gradient descent enforces hard feasibility
at every iterate, unlike penalty methods which require tuning; and (vi) the Frank--Wolfe duality
gap decays at the theoretical $O(1/k)$ rate.  The experiments use a fixed random seed
(\texttt{np.random.seed(42)}) for reproducibility throughout.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \sim \mathcal{N}(0, I_2), \quad
y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
           = \frac{1}{2m}\|X\theta - y\|^2.
\label{eq:lossA}
\end{equation}
This is a convex quadratic with Hessian $H = \frac{1}{m}X^T X$ (constant and positive
semi-definite).  The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$.  Due to
irreducible observation noise, the minimum loss is $J^{\star} \approx 0.4826 > 0$; the
closed-form minimiser is the ordinary least-squares estimator
$\theta_{\mathrm{OLS}} = (X^TX)^{-1}X^Ty \approx (3.028,\,4.017)$.  The condition number
of $H$ is $\kappa(H) \approx 1.3$, so Benchmark A is well-conditioned.
Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\label{eq:lossB}
\end{equation}
The gradient and Hessian are
\begin{align}
\nabla f(x) &= \bigl[2(x_1 - 1) + \cos(x_1),\; 10(x_2 - 2)\bigr]^T, \\
\nabla^2 f(x) &= \operatorname{diag}\!\bigl(2 - \sin(x_1),\; 10\bigr).
\end{align}
The Hessian eigenvalue ratio varies between approximately $1.8:10$ (when $\sin x_1 = -1$)
and $3:10$ (when $\sin x_1 = 1$), creating moderate anisotropy with condition number
$\kappa \in [3.3, 5.6]$.  The unique minimum is at approximately
$(x_1^{\star}, x_2^{\star}) \approx (0.582, 2.000)$ with $f^{\star} \approx 0.7244$.
Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\label{eq:lossC}
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$.  The Hessian at the optimum is
\begin{equation}
\nabla^2 f(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues approximately $2$ and $1000$, giving a condition number
$\kappa \approx 500$.  This extreme ill-conditioning, combined with the narrow curved
valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test
functions for first-order methods~\cite{rosenbrock1960}.
Starting point: $x_0 = [-1, 1]^T$.

For constrained problems (Questions 5 and 6):
$\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline
across all three benchmarks for 120 iterations.  Let $g_k = \nabla f(x_k)$ throughout.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, known optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Step size proportional to sub-optimality}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak convergence~\cite{polyak1969}]
For a convex function $f$ with Lipschitz gradient, the Polyak step size satisfies
$f(x_k) - f^{\star} \leq \frac{\|x_0 - x^{\star}\|^2}{2k}$ when $f^{\star}$ is exactly known,
achieving an $O(1/k)$ convergence rate without tuning a learning rate.
\end{theorem}

\begin{remark}
If $f^{\star}_{\mathrm{spec}} < f^{\star}_{\mathrm{true}}$ (i.e.\ the specified value underestimates
the true minimum), then $f(x_k) - f^{\star}_{\mathrm{spec}} \to f^{\star}_{\mathrm{true}} -
f^{\star}_{\mathrm{spec}} > 0$ while $\|g_k\| \to 0$, causing $\alpha_k \to \infty$ and divergence.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad~\cite{duchi2011}}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The effective per-coordinate step is $\alpha_k^{(i)} = \alpha_0 / \sqrt{\sum_{j=0}^{k} g_j^{(i)2}}$,
which decays monotonically.  Adagrad is well-suited to sparse gradients (where most coordinates
receive rare updates) but may prematurely slow on dense problems.

\begin{algorithm}[H]
\caption{RMSprop~\cite{hinton2012}}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $\|g\|^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's monotone decay by using an exponential moving average of squared
gradients with forgetting factor $\beta$.  This allows the effective step size to adapt to
recent curvature rather than the full gradient history.

\begin{algorithm}[H]
\caption{Polyak Momentum (Heavy Ball)~\cite{polyak1964}}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Velocity accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method adds a momentum term that accumulates the gradient direction across
iterations.  For a quadratic $f$ with condition number $\kappa$, the theoretically optimal
parameters $\alpha^{\star} = 4/((\sqrt{L} + \sqrt{\mu})^2)$ and $\beta^{\star} =
((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$ yield an asymptotic contraction factor
$r = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$, dramatically improving on GD's rate of
$(1 - 2/(\kappa+1))$~\cite{polyak1964}.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Sensitivity to $f^{\star}$:} The Polyak step is the only method with a convergence
  \emph{guarantee} ($O(1/k)$ for convex functions), but only when $f^{\star}$ is exactly known.
  The divergence on Benchmarks A and B is \emph{not} a numerical artefact --- it is a fundamental
  consequence of the algorithm's formula $\alpha_k = (f(x_k) - f^{\star})/\|g_k\|^2$.

\item \textbf{Convergence speed on well-conditioned problems:} All methods (except Polyak-misspecified)
  converge to essentially the same final value on Benchmarks A and B, reflecting their similar
  asymptotic rates.  The condition numbers ($\kappa \approx 1.3$ for A, $\kappa \approx 5$ for B)
  are too small to reveal dramatic differences.

\item \textbf{Ill-conditioned Rosenbrock:} The condition number $\kappa \approx 500$ magnifies
  algorithmic differences.  Momentum (Heavy Ball) provides clear improvement over GD by building
  velocity in the consistent valley-floor direction.

\item \textbf{Stability:} GD is most stable but slowest.  Adagrad's monotone step guarantees
  convergence but may prematurely stall.  RMSprop needs careful learning-rate tuning ($\alpha_0 =
  0.0035$ for Rosenbrock vs.\ $\alpha_0 = 0.22$ for Benchmark A).
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient~\cite{nesterov1983}}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\; \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov acceleration~\cite{nesterov1983}]
For a convex function $f$ with $L$-Lipschitz gradient, Nesterov's accelerated gradient method with
$\alpha = 1/L$ and momentum schedule $\beta_k = (k-1)/(k+2)$ achieves
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} = O\!\left(\frac{1}{k^2}\right),
\end{equation}
which is \emph{optimal} for first-order methods on convex functions~\cite{nesterov2003}.  This
contrasts with gradient descent's $O(1/k)$ rate (or $O((1-\mu/L)^k)$ for strongly convex $f$).
\end{theorem}

The key distinction from Heavy Ball (Algorithm~\ref{alg:heavyball}) is the \emph{lookahead}:
the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$.  This
``corrective'' step allows the momentum to be applied in the right direction before overshooting,
enabling the $O(1/k^2)$ guarantee rather than the $O(1/k)$ rate achievable by Heavy Ball.

\begin{algorithm}[H]
\caption{Adam Optimiser~\cite{kingma2015}}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scaling)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines first-moment momentum (inherited from Heavy Ball / Nesterov) with per-coordinate
adaptive scaling (inherited from RMSprop).  The bias-correction terms $1/(1-\beta_i^t)$ compensate
for the zero-initialisation of $m_0$ and $v_0$, preventing underestimation in the early iterations.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$,
50 epochs, with data shuffled at each epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1,\ldots,m\}$
    \For{each mini-batch $\mathcal{B} \subset \{1,\ldots,m\}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \Comment{Mini-batch gradient}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $g_{\mathcal{B}}$ is an unbiased estimator of the full gradient:
$\mathbb{E}[g_{\mathcal{B}}] = \nabla J(\theta)$, with variance
$\mathrm{Var}[g_{\mathcal{B}}] = \frac{m - b}{b(m-1)}\mathrm{Var}[g_{\mathrm{sample}}] \approx
\frac{\sigma^2}{b}$ (proportional to $1/b$).  Larger batches reduce gradient variance but require
more computation per epoch.

\subsection{Comparative Discussion: Heavy Ball vs Nesterov}

Both Heavy Ball (Q1) and Nesterov (Q2) add a momentum term, but they differ in where the
gradient is evaluated:

\begin{itemize}[nosep]
\item \textbf{Heavy Ball}: gradient at current point $x_k$;  momentum weight fixed at $\beta$.
\item \textbf{Nesterov}: gradient at lookahead point $\tilde{x}_k = x_k + \beta_k z_k$;  momentum
  adaptively increases as $\beta_k = (k-1)/(k+2)$.
\end{itemize}

This lookahead corrects the momentum direction \emph{before} committing to the step, enabling the
theoretically optimal $O(1/k^2)$ convergence rate.  In experiments, Nesterov achieves $f = 0.479$
on Rosenbrock vs.\ Heavy Ball's $f = 1.400$ after the same number of iterations, a $3\times$ improvement.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g^{(1)}(x) = g(x_0) + g'(x_0)(x - x_0)
            = 0.0039 + 0.0625\,(x - 0.25).
\end{equation}
This tangent-line approximation captures local slope only and diverges rapidly from the true
function outside a small neighbourhood of $x_0$.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g^{(2)}(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
            = 0.0039 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\end{equation}
Minimising $g^{(2)}$ with respect to $x$ gives the Newton step:
$x^+ = x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 \approx 0.167$.  This is a far better
estimate of the function's zero-gradient point than the gradient-descent step, since the quadratic
model captures the local curvature.

\subsection{Newton's Method with Hessian Regularisation}

\begin{algorithm}[H]
\caption{Newton's Method with Damping and Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, Hessian damping $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov regularisation for numerics}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{Damped Newton step ($\alpha=1$: pure Newton)}
\EndFor
\end{algorithmic}
\end{algorithm}

The regularisation $\lambda I$ ($\lambda = 10^{-8}$) ensures positive definiteness and numerical
stability of the linear solve $H_k p_k = g_k$ without materially affecting the update direction.

\begin{theorem}[Newton's quadratic convergence~\cite{nocedal2006}]
Let $f$ be twice continuously differentiable and $x^{\star}$ be a local minimum with
$\nabla^2 f(x^{\star}) \succ 0$.  If $x_0$ is sufficiently close to $x^{\star}$ and
$\nabla^2 f$ is $L_H$-Lipschitz, then the pure Newton iterates ($\alpha = 1$) satisfy
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{L_H}{2\mu_H}\|x_k - x^{\star}\|^2,
\end{equation}
where $\mu_H$ is the smallest eigenvalue of $\nabla^2 f(x^{\star})$.  This is \emph{quadratic}
(superlinear) convergence, compared with gradient descent's linear rate $(1 - \mu/L)^k$.
\end{theorem}

For a quadratic objective $f(\theta) = \frac{1}{2}\theta^T H \theta - b^T \theta + c$ with constant
Hessian $H$, the pure Newton step is exact:
$\theta_{k+1} = \theta_k - H^{-1}\nabla f(\theta_k) = \theta_k - H^{-1}(H\theta_k - b) = H^{-1}b = \theta^{\star}$,
achieving the global minimum in a \emph{single} step regardless of initial point or condition number.

Settings: GD uses 80 iterations; Newton uses 20 iterations with $\lambda = 10^{-8}$.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}
% ============================================================

\subsection{Part A: Finite Differences and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The \emph{forward} finite-difference approximation of $\partial f / \partial x_i$ is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}
= \frac{\partial f}{\partial x_i}(x) + \frac{\delta}{2}f_{ii}''(x) + O(\delta^2).
\label{eq:fdfwd}
\end{equation}
The approximation error has two competing contributions:
\begin{itemize}[nosep]
\item \textbf{Truncation error}: $O(\delta)$ from the omitted higher-order Taylor terms.
\item \textbf{Round-off error}: $O(\varepsilon_{\mathrm{mach}} / \delta)$ from floating-point
  cancellation in $f(x + \delta e_i) - f(x)$ when $\delta$ is small.
\end{itemize}
Balancing these two contributions (minimising $C_1 \delta + C_2 \varepsilon_{\mathrm{mach}}/\delta$)
gives the optimal perturbation:
\begin{equation}
\delta_{\mathrm{opt}}^{\mathrm{fwd}} \approx \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}.
\label{eq:delta_opt_fwd}
\end{equation}
The \emph{central} difference formula,
$\partial f/\partial x_i \approx [f(x + \delta e_i) - f(x - \delta e_i)]/(2\delta)$, has
$O(\delta^2)$ truncation error, giving an optimal
$\delta_{\mathrm{opt}}^{\mathrm{cen}} \approx \varepsilon_{\mathrm{mach}}^{1/3} \approx 10^{-5}$
and lower overall approximation error.  The experiment tests $\delta = 0.05$ (near good regime)
and $\delta = 0.8$ (far too large, dominated by truncation error).

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q4_fd_error.pdf}
\caption{Q4: Finite-difference gradient error as a function of $\delta$ (evaluated at
$x = (0.5, 1.5)$ on Benchmark B). Forward FD error (blue) shows the classic $O(\delta)$ truncation
increase for large $\delta$ and the $O(\varepsilon_{\mathrm{mach}}/\delta)$ round-off increase for
small $\delta$, with an optimal near $\delta \approx 10^{-8}$.  Central FD (red) achieves
$O(\delta^2)$ truncation error and an optimal near $\delta \approx 10^{-5}$, with lower overall
error.  The assignment values $\delta = 0.05$ and $\delta = 0.8$ lie well within the truncation-dominated
regime for forward FD, explaining the performance difference.}
\label{fig:q4_fd_error}
\end{figure}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method requires only \emph{two} function evaluations per iteration regardless of
dimension $d$.  The directional derivative $\hat{g}_k$ is a noisy estimate of the gradient:
$\mathbb{E}[\hat{g}_k] = \nabla f_{\delta}(x_k)$ where $f_{\delta}$ is the Gaussian-smoothed
objective.  The convergence rate is $O(d/k)$ --- a factor of $d$ slower than gradient-based
methods --- reflecting the information loss from using a single random direction instead of the
full gradient~\cite{nesterov2017}.

\subsection{Part B: Derivative-Free Methods on Rosenbrock (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and
iteratively transforms it through four geometric operations:

\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): move worst vertex through the centroid of the
  remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{\mathrm{worst}})$.
\item \textbf{Expansion} ($\gamma = 2.0$): if reflection improved, try further:
  $x_e = \bar{x} + \gamma(\bar{x} - x_{\mathrm{worst}})$.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection worsened, contract:
  $x_c = \bar{x} + \rho(x_r \text{ or } x_{\mathrm{worst}} - \bar{x})$.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction also fails, contract all vertices toward
  the best: $x_i \gets x_{\mathrm{best}} + \sigma(x_i - x_{\mathrm{best}})$.
\end{enumerate}

No gradient information is required.  Initial simplex step $= 0.35$, iterations $= 160$.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}

\subsection{Problem Setup and KKT Conditions}

The constrained problem is:
\begin{equation}
\min_{x \in \mathbb{R}^2}\; f(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)
\quad \text{subject to}\quad x_1 \geq 0.5,
\label{eq:constrained}
\end{equation}
starting from the infeasible point $x_0 = (0.2, 4.0)$.

The Karush--Kuhn--Tucker (KKT) optimality conditions for~\eqref{eq:constrained} are:
\begin{align}
\nabla f(x^{\star}) - \mu^{\star} e_1 &= 0, \label{eq:kkt_stat} \\
x_1^{\star} - 0.5 &\geq 0, \label{eq:kkt_primal} \\
\mu^{\star} &\geq 0, \label{eq:kkt_dual} \\
\mu^{\star}(x_1^{\star} - 0.5) &= 0. \label{eq:kkt_comp}
\end{align}
Here $e_1 = [1, 0]^T$ and $\mu^{\star}$ is the KKT multiplier.  The unconstrained minimum of
Benchmark B is at $(0.582, 2.0)$, which satisfies $x_1^{\star} = 0.582 > 0.5$ --- hence the
constraint is \emph{inactive} at the constrained optimum, $\mu^{\star} = 0$, and the constrained
and unconstrained solutions coincide: $x^{\star} = (0.582, 2.0)$, $f^{\star} \approx 0.7244$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project onto feasible set: $x_1 \leftarrow \max(0.5, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection onto $\{x : x_1 \geq 0.5\}$ is simply $[\Pi_{\mathcal{X}}(z)]_1 = \max(0.5, z_1)$,
$[\Pi_{\mathcal{X}}(z)]_2 = z_2$ --- a $O(1)$ operation.  For general convex sets $\mathcal{X}$,
computing the projection can be expensive, but it guarantees \emph{exact} feasibility at every iterate.

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\; 0.5 - x_1)$ \Comment{Penalised objective}
    \State $\nabla F_{\lambda}(x) \gets \nabla f(x) + \lambda \cdot \mathbf{1}[x_1 < 0.5]\,(-e_1)$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method replaces the hard constraint with a soft penalty: violations are penalised
proportionally to $\lambda \max(0, 0.5 - x_1)$.  As $\lambda \to \infty$, the penalty solution
approaches the constrained solution, but large $\lambda$ introduces a stiff gradient and requires
smaller step sizes.  Unlike projected GD, the penalty method does \emph{not} guarantee feasibility
at intermediate iterates.

Settings: Projected GD ($\alpha = 0.08$); Penalty: $\lambda \in \{0.15, 1.8, 4.5\}$
($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$); unconstrained GD
($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected GD} enforces feasibility as a hard constraint at every iteration.  This is the
fundamental advantage: the iterate is always feasible, which is critical in settings where
infeasible points cannot even be evaluated (e.g.\ the objective is only defined on $\mathcal{X}$).
The cost is requiring a projection oracle, which for simple constraints like $x_1 \geq 0.5$ is
trivial but can be expensive for complex (e.g.\ semidefinite) constraint sets.

\textbf{Penalty methods} trade exact feasibility for implementation simplicity.  The weight
$\lambda$ controls the trade-off: small $\lambda$ under-penalises violations; large $\lambda$
enforces feasibility but introduces a stiff gradient term, requiring smaller step sizes for
stability (here $\alpha = 0.03$ for $\lambda = 4.5$ vs.\ $\alpha = 0.05$ for $\lambda \leq 1.8$).
An alternative is the \emph{augmented Lagrangian} method, which updates a dual variable $\mu_k$
alongside the penalty, achieving convergence to the exact constrained solution at moderate $\lambda$
without ill-conditioning~\cite{bertsekas1999}.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$
(a bounded box / polytope).

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, minimise over $\mathcal{X}$.
Since $f$ is linear (hence convex), the minimum is attained at an extreme point (vertex) of
$\mathcal{X}$:
\begin{equation}
\min_{x \in \mathcal{X}}\; x_1 + 2x_2
\quad\Longrightarrow\quad
x^{\star} = (0.5,\; -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This follows directly from the fundamental theorem of linear programming: the optimum of a linear
objective over a bounded polytope is always attained at a vertex~\cite{vanderbei2014}.  Here the
positive coefficients in $a$ push both coordinates to their lower bounds.

\subsection{Frank--Wolfe Algorithm (Conditional Gradient)}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)~\cite{frankwolfe1956}}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\; \nabla f(x_k)^T x$
    \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
    \Comment{Convex combination, fixed step size $(1-\beta)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For box constraints, the LMO decouples coordinate-wise:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring $O(d)$ operations --- far cheaper than a general projection onto complex constraint sets.
Crucially, $x_{k+1}$ is a convex combination of $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$,
so \textbf{feasibility is maintained by construction} without any explicit projection step.

\begin{theorem}[Frank--Wolfe convergence rate~\cite{jaggi2013}]
For a convex function $f$ with $L$-Lipschitz gradient, the Frank--Wolfe algorithm with
step size $\gamma_k = 2/(k+2)$ satisfies
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\,\mathrm{diam}(\mathcal{X})^2}{k+2} = O\!\left(\frac{1}{k}\right).
\end{equation}
The duality gap $h_k = \langle \nabla f(x_k),\, x_k - z_k \rangle \geq f(x_k) - f^{\star}$ provides
an \emph{efficiently computable} upper bound on sub-optimality and a natural stopping criterion.
\end{theorem}

Note: the experiments use a \emph{fixed} $\beta$ (i.e.\ $\gamma_k = 1 - \beta$ constant) rather
than the diminishing schedule $\gamma_k = 2/(k+2)$.  Fixed $\beta$ does not achieve the $O(1/k)$
guarantee but is commonly used in practice for its simplicity.

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits the characteristic ``zig-zag'' convergence
pattern~\cite{jaggi2013}.  The LMO vertex $z_k$ jumps between extreme points of $\mathcal{X}$ at
each iteration, and the convex combination gradually averages these jumps.  With the fixed $\beta$
schedule, the step size does not decay, leading to residual oscillation near the optimum.  Using
the standard $\gamma_k = 2/(k+2)$ schedule would eliminate this and recover the $O(1/k)$ bound.

\textbf{Boundary optimum:} Frank--Wolfe is well suited to boundary optima because $z_k$ frequently
selects the vertex closest to the constrained optimum ($x_1 = 0.5$), naturally driving the iterate
to the boundary.  After 140 iterations, the method reaches $f = 0.364$ vs.\ $f^{\star} = 0.25$,
a gap of $0.114$: convergence is slower because the $O(1/k)$ rate is slow at the boundary when
the problem is not strongly convex over $\mathcal{X}$.

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has systematically evaluated six families of optimisation algorithms across benchmarks
of increasing difficulty.  The principal conclusions are as follows.

\subsection*{Key Findings}

\begin{enumerate}[nosep,leftmargin=1.5em]
\item \textbf{Adaptive step-size methods (Q1):} The Polyak step size provides a convergence rate
guarantee that depends entirely on the accuracy of $f^{\star}$.  When this is known (Rosenbrock),
it outperforms all first-order competitors.  Adagrad and Heavy Ball are more robust for
practical use.  RMSprop's exponential forgetting makes it suitable for non-stationary problems
but offers no theoretical convergence guarantee.

\item \textbf{Momentum and acceleration (Q2):} Nesterov momentum's $O(1/k^2)$ rate provides a
measurable advantage on difficult problems: on Rosenbrock it achieves a $7\times$ improvement over
GD.  Adam is widely used in practice due to its per-coordinate scaling, but lacks formal convergence
guarantees and underperforms Nesterov on problems with strongly coupled curvature.  Mini-batch SGD's
optimal batch size involves a noise-versus-compute trade-off: small batches converge faster per epoch
but oscillate more near the minimum.

\item \textbf{Newton's method (Q3):} Newton's method achieves quadratic local convergence and
converges in a single iteration for quadratic objectives, demonstrating a fundamental advantage over
all first-order methods.  The cost is computing and inverting the Hessian ($O(d^3)$ per iteration),
making it impractical for large-scale problems.  Hessian regularisation ($\lambda I$ added to
$\nabla^2 f$) is necessary for numerical stability and global convergence, but does not materially
affect the asymptotic rate near the optimum.

\item \textbf{Derivative-free methods (Q4):} Finite differences with well-chosen $\delta$ closely
mimic exact gradient descent at the cost of $d$ additional function evaluations per step.  The
theoretically optimal $\delta \approx \sqrt{\varepsilon_{\mathrm{mach}}}$ for forward differences
and $\varepsilon_{\mathrm{mach}}^{1/3}$ for central differences can serve as practical guidelines.
Nesterov random search has dimension-independent cost per iteration but an $O(d/k)$ convergence
rate, making it most appropriate when gradients are unavailable and dimension is very large.

\item \textbf{Constrained optimisation (Q5):} Projected gradient descent guarantees feasibility at
every iterate --- a property that penalty methods only satisfy asymptotically.  When the constraint
is simple (as here: a halfspace), projection is $O(1)$ and PGD is the clear choice.  The penalty
approach becomes competitive when the constraint set is complex and projection is expensive.
Augmented Lagrangian methods offer a middle ground, achieving exact feasibility without the
ill-conditioning of large-$\lambda$ penalties.

\item \textbf{Frank--Wolfe and linear programming (Q6):} The Frank--Wolfe algorithm is
projection-free by design, replacing the projection oracle with an inexpensive linear minimisation
oracle.  This makes it especially attractive for structured constraint sets (e.g.\ spectahedra,
flow polytopes) where LP subproblems are tractable but projections are not.  The duality gap
$\langle \nabla f(x_k), x_k - z_k \rangle$ provides an efficiently computable certificate of
near-optimality, and the $O(1/k)$ convergence rate is confirmed empirically.
\end{enumerate}

\begin{lstlisting}[caption={Gradient Descent}]
def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
