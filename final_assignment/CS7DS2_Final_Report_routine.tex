\begin{abstract}
This report presents a comprehensive comparative study of first-order and second-order optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C). Six families of methods are investigated: (1)~adaptive step-size methods including Polyak, Adagrad, RMSprop, and Heavy Ball; (2)~momentum-based acceleration methods including Nesterov accelerated gradient and Adam; (3)~Newton's method with local second-order approximations; (4)~derivative-free approaches including finite differences, Nesterov random search, Nelder--Mead, and grid search; (5)~constrained optimisation via projected gradient descent and penalty methods; and (6)~the Frank--Wolfe conditional gradient algorithm for linearly constrained problems. Each method is implemented from scratch and evaluated on convergence speed, trajectory behaviour, numerical stability, and computational complexity. Key findings include: the Polyak step diverges catastrophically when $f^{\star}$ is misspecified; Nesterov acceleration achieves the best Rosenbrock convergence among first-order methods; Newton's method converges in a single iteration on the quadratic benchmark; and Frank--Wolfe naturally respects constraints without projection.
\end{abstract}

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss is:
\begin{equation}
J(\theta) = \frac{1}{2m} \|X\theta - y\|^2 = \frac{1}{2m}\sum_{i=1}^{m}\!\left({\theta}^T x^{(i)} - y^{(i)}\right)^2,
\end{equation}
with gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$ and constant Hessian $H_A = \frac{1}{m}X^T X$.

\textbf{Spectral properties.} The eigenvalues of $H_A$ are $\mu = 0.9247$ and $L = 1.0325$, giving condition number $\kappa(H_A) = L/\mu \approx 1.117$. This near-unity condition number makes Benchmark~A an ideal testbed for first-order methods: the landscape is isotropic and any gradient direction is nearly optimal. Starting point: $\theta_0 = [0,\, 0]^T$. Due to observation noise $\varepsilon^{(i)} \sim \mathcal{N}(0,1)$, the OLS minimiser is $\hat{\theta} = [3.028,\, 4.017]^T$ (not exactly $\theta^{\star}$), and the noise floor is $J^{\star} \approx 0.4826$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
\label{eq:loss_B}
f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient and Hessian are:
\begin{align}
\nabla f(x) &= \bigl[2(x_1 - 1) + \cos(x_1),\; 10(x_2 - 2)\bigr]^T, \\
H_B(x) &= \operatorname{diag}\!\bigl(2 - \sin(x_1),\; 10\bigr).
\end{align}
The sinusoidal term introduces position-dependent curvature in the $x_1$ direction. At the starting point $x_0 = [-1,\, 4]^T$: $H_B = \operatorname{diag}(2.841,\, 10)$, condition number $\kappa \approx 3.52$. At the minimum $x^{\star} \approx (0.5824,\, 2.0)$: $H_B = \operatorname{diag}(1.450,\, 10)$, condition number $\kappa \approx 6.90$. The minimum satisfies the stationarity condition $2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$ (no closed form), giving $f^{\star} \approx 0.7244$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
\label{eq:loss_C}
f(x) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The gradient and Hessian are:
\begin{align}
\nabla f(x) &= \bigl[-2(1 - x_1) - 400x_1(x_2 - x_1^2),\; 200(x_2 - x_1^2)\bigr]^T, \\
H_C(x) &= \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{align}
The global minimum is at $(1,\, 1)$ with $f^{\star} = 0$. At the optimum, $H_C(1,1) = \bigl[\begin{smallmatrix}802 & -400 \\ -400 & 200\end{smallmatrix}\bigr]$ has eigenvalues $\approx 0.399$ and $1001.6$, giving condition number $\kappa \approx 2508$. This extreme ill-conditioning, combined with the narrow curved valley along $x_2 = x_1^2$, makes the Rosenbrock function a severe test of curvature exploitation. Starting point: $x_0 = [-1,\, 1]^T$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}
\label{subsec:q1_algs}

Four adaptive methods are compared against constant step-size GD across all benchmarks for 120 iterations. The GD update is $x_{k+1} = x_k - \alpha \nabla f(x_k)$, with convergence rate $O(\rho^k)$ where $\rho = 1 - \alpha\mu$ for a $\mu$-strongly convex, $L$-smooth function and $\alpha \leq 1/L$.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value estimate $f^{\star}$, regulariser $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive step}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Polyak Step Convergence]
For a convex function $f$ with unique minimiser $x^{\star}$ and $f^{\star} = f(x^{\star})$ known exactly, the Polyak step satisfies $\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - \frac{(f(x_k) - f^{\star})^2}{\|g_k\|^2}$, guaranteeing monotone decrease. When $f^{\star}$ is misspecified (too small), $\alpha_k \to \infty$ as $\|g_k\| \to 0$, causing divergence.
\end{theorem}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base rate $\alpha_0$, regulariser $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$ \Comment{Per-coordinate adaptive rate}
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's effective per-coordinate rate is $\alpha_k^{(i)} = \alpha_0 / (\sqrt{\sum_{j=0}^{k} g_j^{(i)2}} + \varepsilon)$, which decays monotonically. This property provides \emph{implicit regularisation}: coordinates with historically large gradients receive smaller steps. Adagrad is well-suited for sparse gradients but can prematurely stall on dense problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, regulariser $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $\|g\|^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's monotonic decay by replacing the full accumulation with an exponential moving average controlled by $\beta$. This allows the effective step size to recover if the gradient magnitudes decrease, making it suitable for non-stationary objectives.

\begin{algorithm}[H]
\caption{Polyak Momentum (Heavy Ball)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Accumulated momentum}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

This is equivalent to $x_{k+1} = x_k - \alpha\,\nabla f(x_k) + \beta(x_k - x_{k-1})$: a gradient step augmented by a fraction of the previous displacement. For quadratics, the optimal parameters $\alpha^{\star} = \frac{4}{(\sqrt{L} + \sqrt{\mu})^2}$ and $\beta^{\star} = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$ yield linear convergence at rate $\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}$, the same asymptotic rate as Nesterov.

\subsection{Comparative Analysis and Discussion}

\begin{itemize}[nosep]
\item \textbf{Polyak:} Theoretically optimal when $f^{\star}$ is known exactly, but the requirement for accurate $f^{\star}$ is a fundamental practical limitation. On Benchmark~C it achieves $f = 0.0094$ in 120 iterations, a result no other first-order method in this study achieves.

\item \textbf{Adagrad:} Uniformly stable across all benchmarks due to its monotonically decreasing effective rate. It matches GD on Benchmarks~A and B (where GD already converges fully) and outperforms GD on Benchmark~C by adapting per-coordinate step sizes to the anisotropic Hessian. The limitation is premature stalling in later iterations when the accumulated denominators are large.

\item \textbf{RMSprop:} The exponential moving average $(\beta = 0.9)$ maintains a responsive effective step size, but requires careful base rate selection. On the very small curvatures of Rosenbrock, the required $\alpha_0 = 0.0035$ is too conservative to make substantial progress.

\item \textbf{Heavy Ball:} The most practically valuable adaptive method here. Momentum allows it to accumulate consistent descent directions along the Rosenbrock valley, achieving $f = 1.40$ versus GD's $f = 3.51$. It correctly converges on all three benchmarks when $f^{\star}$ is not needed.

\item \textbf{Stability vs.\ speed:} GD is most stable but slowest on ill-conditioned problems. Adaptive methods trade some stability for speed, with Polyak being the extreme case: maximum speed when $f^{\star}$ is exact, catastrophic failure otherwise.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}

\subsection{Algorithm Formulations}
\label{subsec:q2_algs}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_1$, step size $\alpha$, maximum momentum $\beta_{\max}$, iterations $N$
\State $z_1 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\; \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Momentum update at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This ``correct-before-overshoot'' mechanism allows the gradient to predict the direction of future iterates, suppressing oscillation. The adaptive schedule $\beta_k = (k-1)/(k+2)$ is capped at $\beta_{\max}$ to prevent instability on non-convex functions.

\begin{theorem}[Nesterov Convergence Rate]
For a convex, $L$-smooth function, Nesterov accelerated gradient with step $\alpha = 1/L$ and the schedule $\beta_k = (k-1)/(k+2)$ achieves:
\[
f(x_k) - f(x^{\star}) \leq \frac{2L\|x_1 - x^{\star}\|^2}{(k+1)^2} = O(1/k^2),
\]
which is optimal among first-order methods (matching the lower bound on gradient oracle complexity). This is a quadratic speedup over GD's $O(1/k)$ rate on convex functions.
\end{theorem}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_1$, learning rate $\alpha$, decay rates $\beta_1, \beta_2 \in (0,1)$, regulariser $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (mean)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (uncentred variance)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected mean}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected variance}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines first-moment (momentum) and second-moment (adaptive scaling) estimation. The bias correction $(1 - \beta_j^t)^{-1}$ is essential in early iterations: without it, $m_t$ and $v_t$ are biased towards zero because they are initialised at zero, making the effective step size artificially large. As $t$ increases, the bias corrections converge to one and have negligible effect.

\begin{remark}
The effective per-coordinate step at iteration $t$ is $\alpha\,|\hat{m}_t^{(i)}|/(\sqrt{\hat{v}_t^{(i)}} + \varepsilon)$. Since $|\hat{m}_t^{(i)}| \leq \sqrt{\hat{v}_t^{(i)}}$ (by the Cauchy--Schwarz inequality in expectation), the effective step size is bounded by $\alpha / \varepsilon$. This provides implicit step-size normalisation, explaining Adam's robustness to learning rate scaling.
\end{remark}

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}

    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{Local Taylor Approximations of $g(x) = x^4$}

The first- and second-order Taylor approximations of $g(x) = x^4$ at $x_0 = 0.25$ are:
\begin{align}
g_1(x) &= g(x_0) + g'(x_0)(x - x_0) \nonumber \\
 &= (0.25)^4 + 4(0.25)^3(x - 0.25) = 3.906 \times 10^{-3} + 0.0625\,(x - 0.25), \\[4pt]
g_2(x) &= g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 \nonumber \\
 &= 3.906 \times 10^{-3} + 0.0625\,(x - 0.25) + \tfrac{1}{2}(0.375)(x - 0.25)^2.
\end{align}
The Newton step minimises $g_2$ by moving to its stationary point:
\begin{equation}
x_{\mathrm{Newton}} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{4(0.25)^3}{12(0.25)^2} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\end{equation}
One can verify $g(0.1667) = 0.000772 \ll g(0.25) = 0.00391$, confirming the Newton step produces a large improvement relative to the gradient step.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0, 1]$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised (positive-definite) Hessian}
    \State Solve $H_k\, p_k = g_k$ for the Newton direction $p_k$ \Comment{$p_k = H_k^{-1} g_k$}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Newton's direction $p_k = H_k^{-1} g_k$ is the minimiser of the local quadratic model $f(x_k) + g_k^T d + \frac{1}{2}d^T H_k d$. The damping $\lambda I$ ensures $H_k + \lambda I \succ 0$ even when the Hessian has small or negative eigenvalues (as occurs in non-convex regions of Rosenbrock). The step size $\alpha < 1$ further globalises convergence.

\begin{theorem}[Newton Quadratic Convergence]
For a twice-continuously-differentiable function $f$ with $H(x^{\star}) \succ 0$ and $\alpha = 1$ (full Newton step), if $x_0$ is sufficiently close to $x^{\star}$ and the Hessian is Lipschitz continuous, then:
\[
\|x_{k+1} - x^{\star}\| \leq C\,\|x_k - x^{\star}\|^2,
\]
for some constant $C > 0$ depending on $\nabla^2 f$. This \emph{quadratic} convergence implies the number of correct decimal digits roughly doubles at each iteration.
\end{theorem}

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$.

\subsection{Computational Cost Analysis}

Newton's method requires solving an $n \times n$ linear system at each iteration, costing $O(n^3)$ (LU factorisation) plus $O(n^2)$ for the Hessian. For $n = 2$ this is trivial, but the cost becomes prohibitive for large-scale problems (e.g.\ deep neural networks with $n = 10^8$ parameters). Quasi-Newton methods (L-BFGS) approximate $H_k^{-1}$ with rank-1 or rank-2 updates at $O(n)$ cost per step, trading convergence rate for scalability.

\section{Question 4: Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}

\subsection{Part A: Derivative Approximation Methods (Benchmark B)}

\subsubsection{Forward Finite Difference Gradient}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
\label{eq:fd}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector. The approximation error decomposes into:
\begin{itemize}[nosep]
\item \textbf{Truncation error:} $O(\delta)$ from the first-order Taylor expansion.
\item \textbf{Round-off error:} $O(\varepsilon_{\mathrm{mach}} / \delta)$ from finite-precision arithmetic.
\end{itemize}
The optimal $\delta$ that balances these is $\delta^{\star} = \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$ for double precision. The experiments use $\delta = 0.05$ (good: truncation $O(0.05)$, round-off $O(10^{-14})$) and $\delta = 0.8$ (poor: truncation $O(0.8)$), both far from the optimal $\delta^{\star}$.

Each gradient evaluation at a $d$-dimensional point requires $d + 1$ function evaluations (one at $x$ and one per coordinate), compared to $O(d)$ arithmetic operations for exact gradients. For $d = 2$, this is 3 evaluations per step.

\subsubsection{Nesterov Random Search (Zeroth-Order)}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation scale $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Uniform random direction on $\mathbb{S}^{d-1}$}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The directional estimate $\hat{g}_k$ approximates $(\nabla f(x_k)^T u_k) u_k$, which is an unbiased estimate of $\frac{1}{d} \nabla f(x_k)$ when $u_k$ is uniform on the sphere (after appropriate normalisation). The \emph{critical advantage} is dimension-independence: only \textbf{2} function evaluations per iteration regardless of $d$. This makes it uniquely valuable when evaluating $\nabla f$ is computationally expensive (e.g.\ black-box simulators).

\subsection{Part B: Derivative-Free Methods on Benchmark C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices and iteratively updates it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): move the worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2.0$): if reflection improved, try a further step.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection worsened, contract towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): contract all vertices towards the best vertex.
\end{enumerate}
The algorithm requires no gradient or Hessian information: only function evaluations at simplex vertices. The simplex geometry naturally adapts to the local landscape through scaling and rotation.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Setup and Optimality Conditions}

The constrained problem is:
\begin{equation}
\min_{x \in \mathbb{R}^2}\; f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1), \quad \text{subject to}\; g(x) = 0.5 - x_1 \leq 0.
\label{eq:q5_problem}
\end{equation}
The starting point $x_0 = (0.2, 4.0)$ is infeasible ($g(x_0) = 0.3 > 0$).

\textbf{KKT conditions.} The KKT stationarity conditions require: $\nabla f(x^{\star}) + \mu \nabla g(x^{\star}) = 0$, $g(x^{\star}) \leq 0$, $\mu \geq 0$, $\mu g(x^{\star}) = 0$. Since the unconstrained minimum is at $x^{\star}_{\text{unc}} \approx (0.5824, 2.0)$ with $x_1^{\star} = 0.5824 > 0.5$, the constraint is inactive at the constrained minimum, meaning $\mu = 0$ and $x^{\star} = x^{\star}_{\text{unc}}$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{C}} : x_1 \gets \max(0.5, x_1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}\!\left(x_k - \alpha\, \nabla f(x_k)\right)$ \Comment{Gradient step then project}
\EndFor
\end{algorithmic}
\end{algorithm}

PGD enforces feasibility as a \textbf{hard constraint}: the projection $\Pi_{\mathcal{C}}(x) = \max(0.5, x_1)$ is a valid proximity operator for $\mathcal{C} = \{x : x_1 \geq 0.5\}$. For this constraint, the projection costs $O(1)$.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\State Define $F_\lambda(x) = f(x) + \lambda\, \max(0,\, 0.5 - x_1)$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$ \Comment{GD on augmented objective}
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty gradient is: $\nabla F_\lambda = \nabla f + \lambda \nabla(\max(0, 0.5 - x_1)) = \nabla f - \lambda e_1\,\mathbf{1}[x_1 < 0.5]$. As $\lambda \to \infty$, the penalty solution approaches the constrained solution. The penalty $\max(0, \cdot)$ is non-smooth, meaning the gradient is discontinuous at the constraint boundary.

Settings: PGD ($\alpha = 0.08$); Penalty $\lambda = 0.15$ ($\alpha = 0.05$), $\lambda = 1.8$ ($\alpha = 0.05$), $\lambda = 4.5$ ($\alpha = 0.03$); Unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Key trade-offs:}
\begin{itemize}[nosep]
\item \textbf{PGD} enforces hard feasibility at every step, making it the preferred method when feasibility is a safety requirement (e.g.\ power grid constraints, control bounds). The limitation is the need for an efficient projection oracle; for general convex sets this can be as expensive as solving a quadratic programme.

\item \textbf{Penalty methods} require only the gradient of the augmented objective, making them easy to implement. However, increasing $\lambda$ stiffens the objective, requiring smaller step sizes and potentially slower convergence. The sequence of problems $\min_x F_{\lambda_k}(x)$ with $\lambda_k \to \infty$ (sequential unconstrained minimisation) converges to the constrained solution in theory, but numerical instability arises for very large $\lambda$.

\item \textbf{Augmented Lagrangian} methods combine penalty and dual variable estimation, avoiding the ill-conditioning of pure penalty methods. The method would add a multiplier update $\mu_{k+1} = \mu_k + \lambda (g(x_k))_+$ and augment the objective accordingly, achieving better conditioning than fixed-$\lambda$ penalty.
\end{itemize}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$ (axis-aligned box / polytope).

\textbf{Linear Programme (Task I).} Minimise $f(x) = x_1 + 2x_2 = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$.

Since $f$ is linear over the convex polytope $\mathcal{X}$, the fundamental theorem of linear programming guarantees the minimum is attained at a \emph{vertex}. Since $a_1 = 1 > 0$ and $a_2 = 2 > 0$, the minimum is at the vertex with the smallest $x_1$ and $x_2$ simultaneously:
\begin{equation}
x^{\star} = (0.5,\; -5), \qquad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe with Fixed Mixing Parameter}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial feasible $x_0 \in \mathcal{X}$, fixed mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\;\nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1-\beta)\, z_k$ \Comment{Convex combination step toward LMO solution}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
The standard Frank--Wolfe algorithm uses a decaying step $\gamma_k = 2/(k+2)$, yielding the $O(1/k)$ convergence guarantee. The implementation here uses a \textbf{fixed} mixing parameter $\beta$ (equivalently, fixed $\gamma = 1 - \beta$), which can converge faster initially but may oscillate near the optimum for interior solutions where $z_k$ jumps between vertices.
\end{remark}

For the box constraint $\mathcal{X}$, the LMO decomposes per coordinate: $(z_k)_i = l_i$ if $(\nabla f(x_k))_i > 0$, else $(z_k)_i = u_i$. This costs $O(d)$ per iteration and requires no projection oracle --- a key advantage over projected methods for complex constraint sets.

\begin{theorem}[Frank--Wolfe Convergence]
For a convex, $L$-smooth function $f$ over a compact convex set $\mathcal{X}$ with diameter $D = \max_{x,y \in \mathcal{X}}\|x-y\|$, the standard Frank--Wolfe with $\gamma_k = 2/(k+2)$ satisfies:
\[
f(x_k) - f(x^{\star}) \leq \frac{2LD^2}{k+2} = O(1/k).
\]
This is slower than gradient descent's $O(1/k)$ constant (which depends on $1/\mu$ rather than $D^2$) for strongly convex objectives, but projection-free. For the box $\mathcal{X}$: $D = \sqrt{(5-0.5)^2 + (10-(-5))^2} \approx 15.66$.
\end{theorem}

\subsection{Interior vs.\ Boundary Behaviour and Analysis}

\textbf{Interior optimum:} Near the interior minimum, $\nabla f \approx 0$, so even small perturbations in gradient direction cause $z_k$ to jump to different vertices. The fixed-$\beta$ FW oscillates because the iterate never truly ``settles'' at the interior: each step takes a fixed fraction towards a (possibly different) vertex. This oscillation is a known limitation of Frank--Wolfe for strongly convex functions with interior optima. Methods that allow decreasing step sizes ($\gamma_k = 2/(k+2)$) or away-steps (corrective FW) converge more reliably.

\textbf{Boundary optimum:} Frank--Wolfe excels here. When $x^{\star}$ lies on the boundary, the gradient $\nabla f(x^{\star})$ points towards the interior of the constraint set, and the LMO consistently selects the same vertex adjacent to $x^{\star}$. After 140 iterations, the boundary method reaches $f = 0.364$ vs $f^{\star} = 0.25$ --- still above the true optimum because the fixed $\beta = 0.93$ step does not decay, preventing tight convergence. Additional iterations would close this gap.

\textbf{Projection-free advantage:} The key practical virtue of Frank--Wolfe is that it avoids projections. For the box constraint $\mathcal{X}$, projection is $O(d)$ and equivalent to the LMO; but for structured constraint sets like the nuclear norm ball, the trace norm ball, or probability simplices, the LMO is computationally much cheaper than the projection (which requires full SVD). This is why Frank--Wolfe is preferred in sparse approximation and matrix optimisation problems.

\section{Summary and Conclusions}
\label{sec:conclusion}
% ============================================================

This study investigated 14 distinct optimisation algorithms across three benchmark functions covering well-conditioned quadratic, mildly non-convex, and severely ill-conditioned landscapes. The following conclusions are drawn:

\subsection{Adaptive Step-Size Methods (Question 1)}
Polyak achieves optimal performance when $f^{\star}$ is precisely known, but fails catastrophically on misspecified $f^{\star}$ --- a fundamental practical limitation. Adagrad provides robust monotone convergence due to its guaranteed step-size decay. RMSprop maintains responsive step sizes but requires careful $\alpha_0$ selection. Heavy Ball offers the best balance of convergence speed and stability without requiring $f^{\star}$.

\subsection{Momentum and Stochastic Methods (Question 2)}
Nesterov acceleration is the most effective first-order method on ill-conditioned problems, achieving the optimal $O(1/k^2)$ rate. Adam provides a practical default for neural network training by combining momentum with per-coordinate normalisation. Mini-batch SGD with small batches converges faster per epoch but exhibits more variance; the batch-size advantage vanishes in high-noise regimes.

\subsection{Newton's Method (Question 3)}
Newton's method dramatically outperforms first-order methods by exploiting second-order curvature information. On quadratic objectives, it converges in a single iteration. On non-convex, ill-conditioned objectives (Rosenbrock), a damped step is essential for global convergence: full Newton steps overshoot the narrow valley, while a damped step $\alpha = 0.22$ achieves $5.4\times$ better results than GD with $4\times$ fewer iterations.

\subsection{Derivative-Free Methods (Question 4)}
Finite differences provide high-fidelity gradients when $\delta$ is chosen appropriately; the error is $O(\delta)$ and a step $\delta \approx 0.05$ is practical. Nesterov random search trades gradient fidelity for dimension-independence. Nelder--Mead is the best derivative-free method for smooth low-dimensional problems ($\leq 10$ dimensions), outperforming GD on Rosenbrock due to its geometric simplex adaptation. Grid search provides global guarantees but scales catastrophically with dimension.

\subsection{Constrained Optimisation (Question 5)}
Projected gradient descent enforces hard feasibility constraints at every iteration at the cost of requiring a projection oracle. Penalty methods are simpler to implement but converge to the constrained solution only approximately (for finite $\lambda$). Increasing $\lambda$ improves constraint satisfaction but stiffens the objective, requiring smaller step sizes and risking numerical instability.

\subsection{Frank--Wolfe Algorithm (Question 6)}
Frank--Wolfe's projection-free design makes it the preferred method for linearly-constrained problems where the LMO is cheaper than the projection. It performs well for boundary optima, where the LMO naturally aligns with the constrained minimiser. For interior optima, oscillation of the LMO vertex solution is a limitation that can be addressed with decaying step sizes or away-step variants.

\subsection{Overall Recommendation}
For convex, well-conditioned problems: standard GD or Adagrad. For ill-conditioned convex problems: Nesterov or Adam. When Hessians are available: Newton (or quasi-Newton for large scale). When gradients are unavailable: Nelder--Mead (small $d$) or random search (large $d$). For constrained problems: PGD (simple constraints) or Frank--Wolfe (linear constraints).

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g   # z accumulates momentum
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z       # Evaluate gradient at lookahead
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # Bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Damped Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))   # Tikhonov regularisation
        try:
            p = np.linalg.solve(H_reg, g)       # Newton direction
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe with Fixed Mixing Parameter}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LMO: solve min_{z in X} g^T z (box constraint decomposes per coord)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z  # Fixed-step convex combination
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
