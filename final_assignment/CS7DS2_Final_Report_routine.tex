\begin{abstract}
This report presents a systematic comparative study of twelve optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark A, condition number $\kappa \approx 1.12$), a toy neural network loss with sinusoidal perturbation (Benchmark B, $\kappa \in [3.5, 6.9]$), and the Rosenbrock function (Benchmark C, $\kappa \approx 2508$). Six categories of methods are investigated: (Q1) adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), (Q2) momentum-based acceleration (Nesterov) and Adam with mini-batch stochastic gradient descent; (Q3) Newton's method with Tikhonov damping; (Q4) derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead, grid search); (Q5) constrained optimisation via projected gradient descent and penalty methods; (Q6) the Frank--Wolfe conditional gradient algorithm. Key findings: (1) Polyak step size achieves $374\times$ improvement over GD on Rosenbrock when $f^{\star}$ is known, but diverges when it is misspecified (Proposition~\ref{prop:polyak-div}); (2) Nesterov acceleration achieves $7\times$ improvement over GD on Rosenbrock consistent with the theoretical $\sqrt{\kappa}$ speedup; (3) Newton's method shows empirically confirmed quadratic convergence ($d_{k+1}/d_k^2 \approx 0.29$, consistent with $M/(2\mu) \approx 0.345$) reaching machine precision in 5 steps on Benchmark B; (4) projected gradient descent enforces the constraint $x_1 \geq 0.5$ exactly from iteration 1, while penalty methods require $\lambda \geq 4.5$ for feasibility; (5) the Frank--Wolfe duality gap decays empirically at $O(1/k)$, confirming the theoretical convergence certificate, with the gap overestimating suboptimality near interior optima due to vertex oscillation.
\end{abstract}

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \in \mathbb{R}^2 \sim \mathcal{N}(0, I), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2}\theta^T H\theta - b^T\theta + c,
\end{equation}
where $H = \frac{1}{m}X^T X$, $b = \frac{1}{m}X^T y$. This is a \emph{strongly convex quadratic}. The gradient is $\nabla J(\theta) = H\theta - b = \frac{1}{m}X^T(X\theta - y)$, and the Hessian $H = \frac{1}{m}X^T X$ is constant (independent of $\theta$).

Since $x^{(i)} \sim \mathcal{N}(0, I)$, by the law of large numbers $H \approx I$ for large $m$, giving eigenvalues $\lambda_{\min} \approx 0.925$, $\lambda_{\max} \approx 1.033$ and condition number $\kappa(H) \approx 1.12$. This near-uniform curvature means the loss surface is nearly circular, so gradient descent converges rapidly regardless of step size. The OLS minimiser is $\theta^{\star}_{\text{OLS}} = H^{-1}b \approx (3.028, 4.017)$, and the irreducible noise floor is $J^{\star} \approx 0.4826$ (due to observation noise $\sigma=1$). Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \operatorname{diag}(2 - \sin(x_1),\ 10)$. Since $\sin(x_1) \in [-1, 1]$, the diagonal elements are $h_{11}(x_1) \in [1, 3]$ and $h_{22} = 10$ (constant). The condition number therefore ranges from $10/3 \approx 3.3$ (when $\sin(x_1) = 1$) to $10/1 = 10$ (when $\sin(x_1) = -1$). Near the minimum ($x_1^* \approx 0.582$), $h_{11} \approx 1.45$, giving $\kappa \approx 6.9$.

The minimum satisfies $x_2^* = 2$ and $2(x_1^*-1)+\cos(x_1^*)=0$, yielding (by Newton's method) $x_1^* \approx 0.5824$ with $f^* \approx 0.7244$. The sinusoidal perturbation makes this function non-convex but unimodal near the minimum. Starting point: $x_0 = [-1, 4]^T$ with $f(x_0) = (-1-1)^2 + 5(4-2)^2 + \sin(-1) \approx 4 + 20 - 0.841 \approx 23.16$.

\textbf{Note on visualisation:} In the contour trajectory figures for Benchmark B (Figures~\ref{fig:q1_contour_B}, \ref{fig:q2_contour_B}, \ref{fig:q3_contour_B}, \ref{fig:q4_contour_B}), the code marks the ``optimum'' star at $(0.61, 2.0)$ rather than the more precise $(0.5824, 2.0)$. \subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at any point is:
\begin{equation}
H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1,1)$: $H = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}$ with eigenvalues $\lambda_{\min} \approx 0.399$ and $\lambda_{\max} \approx 1001.6$, giving condition number $\kappa \approx 2508$. This extreme ill-conditioning arises from the interplay between the steep $x_1$-direction gradient and the gentle valley floor. Combined with the non-convex, narrow curved valley along $x_2 \approx x_1^2$, Rosenbrock is one of the most challenging standard test functions. Starting point: $x_0 = [-1, 1]^T$ with $f(x_0) = (1-(-1))^2 + 100(1-(-1)^2)^2 = 4 + 100 \cdot 0 = 4$. Note the starting point lies exactly on the valley floor ($x_2 = x_1^2$: $1 = (-1)^2 = 1$ ✓), meaning all methods begin at the parabolic valley and must progress along it toward $(1,1)$.

\begin{remark}[Rosenbrock Valley Floor and Why GD is Slow]
At any point $(t, t^2)$ on the valley floor ($x_2 = x_1^2$), the gradient evaluates to:
\[
\nabla f(t, t^2) = \Bigl(-2(1-t) - 400t(t^2-t^2),\; 200(t^2-t^2)\Bigr) = \bigl(2(t-1),\ 0\bigr).
\]
The gradient has \emph{zero} transverse (x₂) component on the valley floor --- only an along-valley (x₁) component pointing toward $x_1^* = 1$. A GD step from $(t, t^2)$ moves $x_1 \to t + 2\alpha(1-t)$ but leaves $x_2 = t^2$ unchanged, immediately taking the iterate off the curved valley floor. At the perturbed point $(t', {t^2+\epsilon})$ with $t' = t + 2\alpha(1-t)$ and $\epsilon > 0$ small, the transverse gradient appears: $\partial f/\partial x_2 = 200\epsilon$, a strong restoring force pulling the iterate back toward the valley. This interplay --- small along-valley gradient $\propto |t-1|$, large transverse restoring force $\propto L_\perp = 1001.6$ --- is the precise mechanism behind Rosenbrock's difficulty. The minimum eigenvalue of $H(1,1)$ (along-valley curvature $\approx 0.399$) forces a tiny step size $\alpha < 2/L = 0.002$, while progress along the valley is bounded by $\|x_{k+1}-x^*\|_{\text{along}} \leq (1-\alpha\lambda_{\min})\|x_k-x^*\|_{\text{along}} = 0.9992\|x_k-x^*\|_{\text{along}}$, converging at rate $(0.9992)^k$ --- consistent with the $\sim$11500-step theoretical estimate in Table~\ref{tab:complexity}. Near $x_1^* = 1$ (close to the optimum), the along-valley gradient $2(t-1) \to 0$, so gradient information in the ``useful'' direction vanishes and GD stagnates.
\end{remark}

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations. The key motivation is that a fixed step size $\alpha$ for gradient descent must satisfy $\alpha < 2/L$ (where $L = \lambda_{\max}(H)$ is the Lipschitz constant of $\nabla f$) for convergence, and the optimal rate requires $\alpha = 2/(\mu + L)$. For Benchmark C, $L$ is large and $\mu$ is tiny ($\kappa \approx 2508$), forcing an impractically small fixed step. Adaptive methods address this by \emph{rescaling} the step per coordinate or per iteration.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$  \Comment{Oracle step exploiting $f^{\star}$}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step is theoretically motivated by minimising the upper bound on $\|x_{k+1} - x^{\star}\|^2$ when $f$ satisfies the Polyak--{\L}ojasiewicz (PL) condition $\|\nabla f(x)\|^2 \geq 2\mu(f(x) - f^{\star})$. Under this condition, the step size $\alpha_k = (f(x_k)-f^{\star})/\|g_k\|^2$ satisfies $f(x_{k+1}) - f^{\star} \leq (1 - \mu/L)(f(x_k) - f^{\star})$ when $f$ is $L$-smooth.

\begin{remark}[Derivation of the Polyak Step Formula]
The Polyak step arises by minimising $\|x_{k+1} - x^{\star}\|^2$ over all gradient steps $x_{k+1} = x_k - \alpha_k g_k$:
\[
\|x_{k+1} - x^{\star}\|^2 = \|x_k - x^{\star}\|^2 - 2\alpha_k\, g_k^T(x_k - x^{\star}) + \alpha_k^2\|g_k\|^2.
\]
Setting the derivative w.r.t.\ $\alpha_k$ to zero yields the \emph{oracle step} $\alpha_k^{\star} = g_k^T(x_k - x^{\star})/\|g_k\|^2$. Since $x^{\star}$ is unknown, we lower-bound the numerator via convexity: $g_k^T(x_k - x^{\star}) \geq f(x_k) - f^{\star}$. Substituting $\alpha_k = (f(x_k) - f^{\star})/\|g_k\|^2$ (a computable lower bound on $\alpha_k^{\star}$) gives the \emph{Polyak descent inequality}:
\[
\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - \frac{(f(x_k) - f^{\star})^2}{\|g_k\|^2}.
\]
Under PL ($\|g_k\|^2 \geq 2\mu(f(x_k) - f^{\star})$), this becomes $\|x_{k+1} - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - (f(x_k) - f^{\star})/(2\mu)$: guaranteed distance decrease proportional to current suboptimality. The divergence under misspecification follows immediately: replacing $f^{\star}$ with $\tilde{f}^{\star} < f^{\star}$ gives $\alpha_k > \alpha_k^{\star}$, overshooting $x^{\star}$ and potentially increasing $\|x_{k+1} - x^{\star}\|^2$. Concretely, for Benchmark A with $f^{\star} = 0.4826$ but specified $\tilde{f}^{\star} = 0$: once $x_k$ is close to $x^{\star}$, $f(x_k) - \tilde{f}^{\star} \approx 0.4826 \gg f(x_k) - f^{\star} \approx 0$, causing $\alpha_k \approx 0.4826/\|g_k\|^2 \to \infty$ as $g_k \to 0$.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Coordinate-wise cumulative squared gradient}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad adapts the step size \emph{per coordinate}: coordinates with historically large gradients receive smaller updates, and sparse coordinates receive larger updates. The effective learning rate for coordinate $i$ is $\alpha_0/\sqrt{G_{k,i}+\varepsilon}$, which decays monotonically as $G_{k,i}$ accumulates. This ensures convergence but causes the learning rate to become vanishingly small for long runs. The theoretical justification is an online learning regret bound: for a convex sequence of functions, Adagrad achieves regret $O(\sqrt{T \sum_i \|g_{1:T,i}\|^2})$ which is never worse than the standard $O(\sqrt{T})$ regret of constant-step GD and can be $O(s\sqrt{T})$ for sparse problems where only $s$ coordinates are active. This adaptivity is key for natural language processing problems where word-frequency gradients are highly sparse.

Geometrically, Adagrad applies a \emph{diagonal preconditioning}: the update $x_{k+1} = x_k - \text{diag}(G_{k+1})^{-1/2} g_k$ approximates the preconditioned gradient descent $x_{k+1} = x_k - D^{-1} g_k$ where $D \approx \frac{1}{k}\sum_{i=1}^k g_i g_i^T$ (a rank-1 approximation of the average outer product of gradients). For a quadratic $f(x) = \frac{1}{2}x^T H x$, the ideal preconditioning is $D = H$, which would give one-step convergence. Adagrad adapts toward $H$ along the gradient directions visited, reducing the effective condition number for those coordinates.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop fixes Adagrad's freezing problem by replacing the cumulative sum with an \emph{exponential moving average} of squared gradients (decay parameter $\beta = 0.9$). The effective step for coordinate $i$ is $\alpha_0 / \sqrt{v_{k,i} + \varepsilon}$ where $v_{k,i} \approx (1-\beta)\sum_{j=0}^{k}\beta^{k-j} g_{j,i}^2$. With $\beta = 0.9$, the effective memory window is $\approx 1/(1-0.9) = 10$ iterations: gradients older than $\sim 10$ steps receive negligible weight. This allows the effective step to stabilise rather than decay to zero, making it suitable for non-stationary objectives and long training runs. However, near the minimum of Benchmark A, the gradients become small and variable, and $v_{k,i}$ can also become small (past small gradients are forgotten), causing step-size oscillation --- explaining why RMSprop reaches $f \approx 0.503$ rather than $f^* = 0.4826$ on Benchmark A.

\begin{remark}[Adagrad Regret Guarantees vs RMSprop]
Adagrad was specifically designed for online convex optimisation and carries a formal \emph{regret bound}: after $T$ rounds, $\sum_{t=1}^T f_t(x_t) - \min_x \sum_{t=1}^T f_t(x) \leq (\|x_0 - x^*\| / \alpha_0) \sum_i \sqrt{\sum_{t=1}^T g_{t,i}^2} = O(\sqrt{T} \cdot \|G_T\|_F)$. This guarantee holds even for adversarially chosen sequences $\{f_t\}$ and is information-theoretically optimal for online learning. The key insight is that $G_{T,i} = \sum_t g_{t,i}^2$ grows slowly for sparse gradients (rare features receive larger effective steps), making Adagrad well-suited for NLP problems.

RMSprop, by contrast, was introduced as a heuristic for mini-batch neural network training (Hinton, 2012 lecture notes) and lacks formal regret guarantees in the adversarial online learning setting. Reddi et al.\ (2018) showed that similar exponential-forgetting methods (including Adam) can fail to converge on simple convex problems due to the forgetting of past gradients allowing spuriously large steps. The benefit of RMSprop's forgetting is \emph{non-stationarity adaptation}: when gradients shift (e.g., different layers activate at different training stages), forgetting old gradient scales allows the method to adapt. For the stationary, deterministic objectives in this report, this adaptation is unnecessary, and Adagrad's accumulation provides a stronger convergence guarantee.
\end{remark}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$  \Comment{Accumulated velocity}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method (Polyak, 1964) adds a momentum term that accumulates past gradient directions. For a quadratic with eigenvalues $\mu \leq \lambda \leq L$, the optimal parameters are $\alpha^* = 4/(\sqrt{L}+\sqrt{\mu})^2$ and $\beta^* = ((\sqrt{L}-\sqrt{\mu})/(\sqrt{L}+\sqrt{\mu}))^2$, achieving a convergence rate $(({\sqrt{\kappa}-1})/({\sqrt{\kappa}+1}})^k$ --- an improvement over GD's $(({\kappa-1})/({\kappa+1}})^k$ rate by effectively taking a square root of the condition number.

\begin{remark}[Heavy Ball: Practical vs Theoretically Optimal Parameters]
For Benchmark C ($L \approx 1001.6$, $\mu \approx 0.399$): the theoretically optimal parameters are $\alpha^*_C = 4/(\sqrt{1001.6}+\sqrt{0.399})^2 \approx 4/(31.65+0.632)^2 \approx 0.00384$ and $\beta^*_C \approx ((31.65-0.632)/(31.65+0.632))^2 \approx 0.923$. However, the code uses $\alpha_C = 0.0008$ and $\beta_C = 0.86$. The theoretical $\alpha^*$ would require exact knowledge of $L$ and $\mu$ at the minimum; the used value is $4.8\times$ smaller for safety. For Benchmark A ($\kappa \approx 1.12$): $\beta^*_A \approx ((1.016-0.962)/(1.016+0.962))^2 \approx 0.00075$ (near-zero), but the code uses $\beta_A = 0.88$. The discrepancy arises because the theoretical formula assumes a quadratic objective with globally constant $L$ and $\mu$; on non-quadratic problems, a high $\beta$ (near 1) empirically smooths gradient oscillations even when the theoretical optimal is near zero.
\end{remark}

\begin{remark}[Heavy Ball Convergence via Characteristic Polynomial]
The Heavy Ball convergence rate can be derived via spectral analysis. On a 1D quadratic $f(x) = \frac{1}{2}\lambda x^2$, the iteration $x_{k+1} = (1+\beta - \alpha\lambda)x_k - \beta x_{k-1}$ is a linear second-order recurrence. Its characteristic polynomial is $r^2 - (1+\beta-\alpha\lambda)r + \beta = 0$, with roots
\[
r_{1,2} = \tfrac{1}{2}\!\left[(1+\beta-\alpha\lambda) \pm \sqrt{(1+\beta-\alpha\lambda)^2 - 4\beta}\right].
\]
Convergence requires $|r_{1,2}| < 1$; the optimal strategy is \emph{critical damping}: choose $\alpha,\beta$ so both roots lie on the circle $|r| = \sqrt{\beta}$ (complex conjugate pair). This requires the discriminant to vanish, $(1+\beta-\alpha\lambda)^2 = 4\beta$, giving $\alpha\lambda = (1-\sqrt{\beta})^2$. For $\lambda \in [\mu, L]$, one cannot satisfy this for all $\lambda$ simultaneously. The globally optimal $(\alpha^*, \beta^*)$ balance the worst-case roots at $\lambda = \mu$ and $\lambda = L$, yielding $|r_{1,2}| = \sqrt{\beta^*} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$ as the spectral radius for both extremes. This is the formal derivation of the ``square root of condition number'' speedup: $\rho_{\text{HB}} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$ versus GD's $\rho_{\text{GD}} = (\kappa-1)/(\kappa+1)$. Nesterov's adaptive $\beta_k = (k-1)/(k+2)$ is \emph{not} equivalent to Heavy Ball with time-varying $\beta$; it achieves the same asymptotic rate on strongly convex quadratics but through a different Lyapunov function argument (Polyak \& Juditsky, 1992).
\end{remark}

\begin{remark}
The Polyak step size satisfies a minimax optimality property: for any $L$-smooth function satisfying the PL condition with parameter $\mu$, using the exact $f^{\star}$ yields convergence $f(x_k) - f^{\star} \leq (1-\mu/L)^k (f(x_0)-f^{\star})$. When $f^{\star}$ is misspecified (too low), the step size $\alpha_k \to \infty$ near the true minimum, causing the instability observed on Benchmarks A and B.
\end{remark}

\begin{proposition}[Polyak Divergence Under Misspecification]\label{prop:polyak-div}
Let $f$ be $\mu$-strongly convex with unique minimiser $x^{\star}$ and true minimum $c = f(x^{\star}) > 0$. If the Polyak step uses $\tilde{f}^{\star} = 0 < c$, then near the true minimum the step size grows without bound:
\begin{equation}
\alpha_k = \frac{f(x_k)}{\|\nabla f(x_k)\|^2 + \varepsilon} \;\xrightarrow[x_k \to x^{\star}]{}\; \frac{c}{0 + \varepsilon} = \frac{c}{\varepsilon}.
\end{equation}
More precisely, using $\mu\|x-x^{\star}\|^2/2 \leq f(x)-c \leq L\|x-x^{\star}\|^2/2$ and $\mu\|x-x^{\star}\| \leq \|\nabla f(x)\| \leq L\|x-x^{\star}\|$:
\begin{equation}
\alpha_k \geq \frac{c}{\|\nabla f(x_k)\|^2} \geq \frac{c}{L^2\|x_k - x^{\star}\|^2}.
\end{equation}
As the iterate $x_k$ approaches $x^{\star}$, the denominator $\|x_k - x^{\star}\|^2 \to 0$ while the numerator $c > 0$ remains constant, so $\alpha_k \to +\infty$. A step $x_{k+1} = x_k - \alpha_k \nabla f(x_k)$ with $\alpha_k \gg 1/L$ overshoots, launching the iterate far from $x^{\star}$.
\end{proposition}

\begin{remark}[PL Condition vs Strong Convexity]
Strong convexity ($\mu$-SC: $f(x) \geq f(x^{\star}) + \frac{\mu}{2}\|x-x^{\star}\|^2$ for all $x$) implies the Polyak--{\L}ojasiewicz (PL) condition $\|\nabla f(x)\|^2 \geq 2\mu(f(x)-f^{\star})$. However, PL is strictly weaker: it can hold for non-convex functions (e.g., over-parameterised neural networks where every local minimum is global). This distinction matters because PL guarantees linear convergence for GD without the need for strong convexity, making it relevant for modern deep learning theory.
\end{remark}

\subsection{Comparative Analysis and Theoretical Connections}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} Polyak achieves the fastest convergence when $f^{\star}$ is known exactly (Benchmark C: $f = 0.009$ in 120 steps, vs GD's $f = 3.51$, a $374\times$ improvement). Adagrad and Heavy Ball match GD on well-conditioned problems (A, B: all reach $f^*$ within float tolerance) and substantially improve over GD on ill-conditioned Rosenbrock (Heavy Ball $f = 1.40$ vs GD $f = 3.51$, $2.5\times$ improvement; Adagrad $f = 2.20$, $1.6\times$ improvement).
\item \textbf{Adaptivity mechanism:} Adagrad and RMSprop perform \emph{coordinate-wise} scaling, effectively computing a diagonal preconditioning matrix $D_k^{-1/2}$ that partially compensates for the Hessian's anisotropy ($\kappa \to \kappa_{\text{eff}} \leq \kappa$). Polyak performs \emph{scalar} step adaptation using global suboptimality information. Heavy Ball adds directional momentum (no explicit coordinate scaling) and achieves an improved rate $\propto ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^k$ by replacing $\kappa$ with $\sqrt{\kappa}$. Note: coordinate scaling helps most for high-dimensional sparse problems; for $d=2$ it is less impactful, explaining the moderate improvement on Rosenbrock.
\item \textbf{Stability:} GD is most stable (step bounded by $1/L$). Adagrad is uniformly stable due to its monotonically non-increasing effective step ($\alpha_k \propto 1/\sqrt{k}$), guaranteeing $O(\sqrt{T})$ regret for convex problems. RMSprop may oscillate if $\beta$ is too large, but the exponential forgetting ($\beta = 0.9$, half-life $\approx 6$ iterations) provides robustness. Polyak is unstable when $f^{\star}$ is misspecified (see Proposition~\ref{prop:polyak-div}).
\item \textbf{Effect of condition number:} On Benchmark A ($\kappa \approx 1.12$), the GD convergence factor $((\kappa-1)/(\kappa+1)) \approx 0.057$ gives near-geometric convergence requiring only $\sim$10 steps to $10^{-8}$ accuracy. On Rosenbrock ($\kappa \approx 2508$), the GD factor is $\approx 0.9992$, requiring $\sim$11500 steps for $10^{-8}$ accuracy. The $\kappa$-fold difference between benchmarks A and C explains why adaptive methods are essential for ill-conditioned problems.
\item \textbf{Practical implication:} Since exact $f^{\star}$ is unavailable in practice (e.g., deep learning), RMSprop and Adagrad are preferred. The Polyak step is valuable in theoretical analysis and certain structured problems where $f^{\star} = 0$ (e.g., feasibility problems, exact interpolation). For the specific class of over-parameterised models satisfying the PL condition, Polyak-style steps can be estimated without knowing $f^*$ by using line search.
\end{itemize}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum (Accelerated Gradient Descent)}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive schedule: $\beta_k \to 1$ as $k \to \infty$}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead / extrapolation point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This ``look-before-you-leap'' correction prevents overshoot: Nesterov anticipates where the momentum is taking the iterate and corrects the trajectory. The adaptive schedule $\beta_k = (k-1)/(k+2)$ (capped at $\beta_{\max}$) increases momentum gradually from 0, achieving the optimal $O(1/k^2)$ convergence rate for smooth convex functions (Nesterov, 1983):
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} \quad \text{(smooth convex)}.
\end{equation}
This compares favourably to GD's $O(1/k)$ rate and is information-theoretically optimal in the first-order oracle model. For strongly convex $\mu$-SC objectives, the rate improves further to $(1 - \sqrt{\mu/L})^k$, versus GD's $(1-\mu/L)^k$; for Rosenbrock ($\kappa \approx 2508$), the theoretical speedup factor is $\sqrt{\kappa} \approx 50$: Nesterov needs $\sim$220 iterations for $10^{-4}$ accuracy vs GD's $\sim$11500 iterations. The empirical $7\times$ speedup on Benchmark C (150 iterations) is consistent with this, noting that both methods are far from convergence after 150 steps on Rosenbrock.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First-moment (momentum) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second-moment (variance) estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$  \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam (Kingma \& Ba, 2015) combines momentum (via $m_t$, tracking the mean gradient) with per-coordinate adaptive scaling (via $v_t$, tracking gradient variance), inheriting advantages of both momentum and RMSprop. The bias correction terms $1/(1-\beta_1^t)$ and $1/(1-\beta_2^t)$ prevent early iterates from being biased towards zero. To see why, note that $m_t = (1-\beta_1)\sum_{i=1}^{t}\beta_1^{t-i} g_i$, so assuming all gradients equal $g$:
\begin{equation}
\mathbb{E}[m_t] = (1-\beta_1)\sum_{i=1}^{t}\beta_1^{t-i} g = g \cdot (1 - \beta_1^t).
\end{equation}
Thus $\hat{m}_t = m_t/(1-\beta_1^t)$ is an unbiased estimator of $g$ for all $t$. Without bias correction, $\mathbb{E}[m_1] = (1-\beta_1)g$ --- a $\beta_1$-fold underestimate of the true gradient. For $\beta_1 = 0.9$, the first-step estimate is only $10\%$ of the true gradient, causing negligibly small initial steps. The correction is particularly important in the first $\sim 50$ iterations where $\beta_1^t$ is non-negligible.

\textbf{Note on hyperparameters:} The experiments use non-default $\beta_1$ values: $0.82$ (Bench.\ A) and $0.80$ (Bench.\ B, C), compared to the commonly-recommended $\beta_1 = 0.9$. Reduced $\beta_1$ decreases momentum retention, which can improve convergence for deterministic (full-batch) gradients where oscillation near the minimum is undesirable. The default $\beta_1 = 0.9$ is designed for stochastic settings where high momentum helps average out gradient noise; in a full-batch (deterministic) setting, excessive momentum causes over-shooting. The combination $\beta_1 = 0.9, \beta_2 = 0.999$ in the original Adam paper was tuned for stochastic deep learning problems, not deterministic quadratics.

\begin{remark}[Adam Non-Convergence on Non-Convex Problems]
Reddi et al.\ (2018) showed that Adam with the standard fixed $(\beta_1, \beta_2)$ can diverge on simple non-convex problems. The intuition: Adam's exponential moving average of $v_t$ ``forgets'' past gradient information at rate $(1-\beta_2)$. On Rosenbrock, the gradient magnitude varies rapidly along the trajectory (large when far from the valley, near-zero along the valley floor). If a large gradient is quickly forgotten, the step size can increase inappropriately just as the iterate enters a steep region. Nesterov, having no per-coordinate scaling, maintains a more conservative step bounded by $1/L$, and its lookahead correction prevents overshooting the valley walls.
\end{remark}

\begin{remark}[Adam Bias Correction: Quantitative Early-Step Effect]
At step $t=1$ with $\beta_1 = 0.9$, $\beta_2 = 0.999$ and a constant gradient $g$:
$m_1 = 0.1\,g$, $v_1 = 0.001\,g^2$.
Without bias correction: $\|m_1\|/\sqrt{v_1} = 0.1|g|/\sqrt{0.001}\,|g| \approx 3.16$, so the step size is $\approx 3.16\alpha$ --- \emph{too large}. With bias correction: $\hat{m}_1 = g$, $\hat{v}_1 = g^2$, so the step is exactly $\alpha \cdot \text{sign}(g)$. Bias correction thus reduces the first step by $\approx 3.16\times$, preventing an overly aggressive initial move. This normalisation arises from the two correction factors having different convergence speeds: the first-moment correction $(1-\beta_1^t)^{-1}$ reaches $\approx 1$ by iteration $\sim 44$ (for $\beta_1 = 0.9$), but the second-moment correction $(1-\beta_2^t)^{-1}$ remains significant until iteration $\sim 4600$ (for $\beta_2 = 0.999$). Consequently, even after $m_t$ bias has vanished, the $v_t$ correction continues to suppress $\sqrt{\hat{v}_t}$ below $\sqrt{v_t}$, effectively keeping the adaptive step bounded by $\alpha$ throughout most of training. With the code's non-default $\beta_1 = 0.82$, the first-moment correction factor is $(1-0.82)^{-1} \approx 5.6$ at $t=1$, and it decays to $\approx 1.01$ by $t\approx 25$, reaching its steady state much earlier than the $v_t$ correction.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch (dataset size $m = 1000$).

\textbf{Gradient variance analysis:} For a mini-batch $\mathcal{B}$ of size $b$ drawn uniformly without replacement, the variance of the stochastic gradient $\hat{g} = \frac{1}{b}X_\mathcal{B}^T(X_\mathcal{B}\theta - y_\mathcal{B})$ is:
\begin{equation}
\text{Var}(\hat{g}) = \frac{m - b}{b(m-1)} \cdot \text{Var}(g_{\text{full}}),
\end{equation}
where $g_{\text{full}} = \frac{1}{m}X^T(X\theta - y)$ is the full-batch gradient. For $b = 5$: $\text{Var}(\hat{g}) \approx \frac{995}{5 \cdot 999} \approx 0.199 \cdot \text{Var}(g_{\text{full}})$. For $b = 40$: $\text{Var}(\hat{g}) \approx \frac{960}{40 \cdot 999} \approx 0.024 \cdot \text{Var}(g_{\text{full}})$. Thus $b=5$ has approximately $0.199/0.024 \approx 8 \times$ higher gradient variance than $b=40$, explaining the greater oscillation near the minimum in Figure~\ref{fig:q2_sgd_batch}.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Unbiased stochastic gradient: $\mathbb{E}[g] = \nabla J(\theta)$}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q2_sgd_batch.pdf}
\caption{Q2: Mini-batch SGD with batch sizes $b=5$ and $b=40$ on Benchmark A. Per epoch, $b=5$ performs $m/b = 200$ gradient updates while $b=40$ performs only 25 updates. The larger update count per epoch explains why $b=5$ converges faster per epoch: more gradient descent steps are taken. However, $b=5$ exhibits greater oscillation near the minimum due to higher stochastic gradient variance: $\text{Var}(\hat{g}) \propto \sigma^2/b$, so smaller batches produce noisier estimates. \label{fig:q2_sgd_batch}
\end{figure}

\begin{remark}[SGD Noise Floor]
Mini-batch SGD with constant step size $\alpha$ does not converge to $\theta^*$; instead it oscillates in a ball around $\theta^*$ of radius $O(\alpha \sigma^2 / b)$ where $\sigma^2$ is the gradient noise variance. This is the fundamental trade-off in SGD: to decrease the noise floor, one must either (a) decrease $\alpha$ (which slows convergence), or (b) increase batch size $b$ (which increases per-step cost). For a fixed computational budget of $T$ gradient evaluations, the optimal strategy is to use $b = O(\sqrt{T})$ to balance convergence rate with noise reduction. In our experiment, the ``noise floor'' at $J^* \approx 0.4826$ is the irreducible observation-noise loss, not the SGD noise --- for the linear regression objective with Gaussian noise, GD itself cannot go below $J^*$.
\end{remark}

\subsection{Local Approximations of $g(x) = x^4$}

Given $g(x) = x^4$, $g'(x) = 4x^3$, $g''(x) = 12x^2$, the local approximations at $x_0 = 0.25$ are:

\textbf{First-order (Taylor) approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton step minimises $\hat{g}_2$ by setting its derivative to zero:
\begin{equation}
\hat{g}_2'(x) = g'(x_0) + g''(x_0)(x - x_0) = 0 \implies x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)}.
\end{equation}
For $g(x) = x^4$: $x_{k+1} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3} = \frac{2}{3}x_k$. Starting from $x_0 = 0.25$: $x_1 = 0.167$, $x_2 = 0.111$, \ldots, converging to 0 at rate $\|x_k\| = (2/3)^k \|x_0\|$ --- \emph{linear} convergence with constant $c = 2/3$. This is a key observation: Newton's method achieves quadratic convergence $\|x_{k+1}-x^*\| \leq \frac{M}{2\mu}\|x_k-x^*\|^2$ only when the minimum is \emph{non-degenerate} ($\nabla^2 f(x^*) \succ 0$). Here $g''(0) = 12 \cdot 0 = 0$, so the Hessian vanishes at the minimum, and the standard quadratic convergence theory does not apply. Instead, the iteration satisfies the exact recurrence $x_{k+1} = \frac{2}{3}x_k$ (a linear map), giving $x_k = (2/3)^k \cdot 0.25$. The Newton step overshoots toward the origin at each step but by a bounded factor $1/3$ of the current distance, yielding stable but only linear convergence.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov regularisation for numerical stability}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: solve $H_k p_k = g_k$}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{$\alpha = 1.0$ is the pure Newton step}
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: GD uses 80 iterations; Newton uses 20 iterations with Tikhonov damping $\lambda = 10^{-8}$ (ensures $H_k$ is positive definite even when $\nabla^2 f$ is nearly singular). The damped step ($\alpha < 1$) is used for Benchmarks B and C to prevent overshooting due to rapid Hessian variation.

\begin{remark}[Tikhonov Damping Choice $\lambda = 10^{-8}$]
The Tikhonov-damped Newton direction $p_k = (H_k + \lambda I)^{-1}g_k$ interpolates between pure Newton ($\lambda \ll \lambda_{\min}(H_k)$) and gradient descent ($\lambda \gg \lambda_{\max}(H_k)$). Concretely: for $\lambda \ll \lambda_{\min}$, the step is approximately $H_k^{-1}g_k$ (Newton direction); for $\lambda \gg \lambda_{\max}$, the step becomes $g_k/\lambda$ (gradient descent with step $1/\lambda$). The choice $\lambda = 10^{-8}$ is calibrated to be below machine-precision noise ($\varepsilon_{\text{mach}} \approx 2.2 \times 10^{-16}$) amplified by the Hessian inversion: if $\lambda_{\min}(H_k) \geq \lambda$, the damping shifts the minimum eigenvalue by less than 0.003\% for Benchmarks B--C ($\lambda_{\min} \approx 1.45$), preserving the Newton direction. The specific choice $\lambda = \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ is standard: it prevents the normalised floating-point error in the Hessian ($\varepsilon_{\text{mach}}\|H_k\|$) from dominating the smallest eigenvalue while remaining orders of magnitude below any genuine near-singularity. For Rosenbrock at the starting point $(-1,1)$: $\lambda_{\min}(H) = \lambda_{\min}\bigl([[2+400-400, 400],[400, 200]]\bigr) \approx 5.4$, so $\lambda = 10^{-8}$ contributes less than $2\times10^{-9}$ to the minimum eigenvalue --- effectively zero.
\end{remark}

\begin{remark}[Quadratic vs Linear Convergence]
Newton's method achieves \emph{quadratic} convergence near a non-degenerate minimum $x^{\star}$ (where $\nabla^2 f(x^{\star}) \succ 0$):
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2,
\end{equation}
where $M$ is the Lipschitz constant of $\nabla^2 f$ and $\mu = \lambda_{\min}(\nabla^2 f(x^{\star}))$. This means the number of correct decimal digits \emph{doubles} each iteration. Gradient descent achieves at best linear convergence:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \left(1 - \frac{\mu}{L}\right)\|x_k - x^{\star}\|.
\end{equation}
For Benchmark A: $\mu \approx 0.925$, $L \approx 1.033$, so $\mu/L \approx 0.89$ and GD converges in $\sim$50 steps. For Rosenbrock: $\mu \approx 0.399$, $L \approx 1001.6$, so $\mu/L \approx 0.0004$ and GD would require $\sim$57000 steps for $10^{-8}$ accuracy. Newton avoids this issue entirely by rescaling each step by the local curvature $\nabla^2 f(x_k)^{-1}$, which accounts for the extreme anisotropy.
\end{remark}

\begin{remark}[Why $\alpha < 1$ for Rosenbrock Newton]
The Rosenbrock function $f(x) = (1-x_1)^2 + 100(x_2-x_1^2)^2$ has a highly curved Hessian: the third-order tensor $\nabla^3 f$ has large entries (e.g., $\partial^3 f / \partial x_1^3 = 1200 x_1$ up to $\pm 1200$ for $|x_1| \leq 1$). This means the quadratic model $\hat{f}(x + p) \approx f(x) + \nabla f^T p + \frac{1}{2}p^T H p$ becomes inaccurate even for small $\|p\|$. Setting $\alpha = 0.22$ effectively limits the step to a trust region where the quadratic model is still reliable. In practice, this is addressed by line search methods (Armijo backtracking) or trust-region methods that adaptively limit step size without requiring manual tuning.
\end{remark}

\begin{remark}[Newton Convergence Phases on Rosenbrock]
Newton's method has two phases on non-quadratic problems: (1) a \emph{global phase} (large distance from $x^*$) where convergence is linear, analogous to damped gradient descent in the Newton direction; (2) a \emph{local quadratic phase} (sufficiently close to $x^*$) where the quadratic model is accurate and convergence rate becomes $\|x_{k+1}-x^*\| \leq (M/2\mu)\|x_k-x^*\|^2$. The transition occurs roughly when $\|x_k - x^*\| \leq 2\mu/M$ where $M = \sup_x \|\nabla^3 f(x)\|$ (Lipschitz constant of the Hessian). For Rosenbrock with $M \approx 1200$ and $\mu \approx 0.399$, the quadratic-convergence ball has radius $\approx 2(0.399)/1200 \approx 0.00066$. With $\alpha = 0.22$, the damped Newton iterates remain in the global phase throughout 20 iterations from $x_0 = (-1, 1)$, explaining why we observe linear rather than quadratic convergence on Benchmark C.
\end{remark}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta} + O(\delta) + O\!\left(\frac{\varepsilon_{\text{mach}}}{\delta}\right),
\end{equation}
where $e_i$ is the $i$-th standard basis vector. The total error has two components: \emph{truncation error} $O(\delta)$ from the Taylor expansion, and \emph{rounding error} $O(\varepsilon_{\text{mach}}/\delta)$ from floating-point arithmetic. Minimising the sum gives the optimal step $\delta^{\star} = \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ for double precision ($\varepsilon_{\text{mach}} \approx 2.2 \times 10^{-16}$). The implementation tests $\delta = 0.05$ (moderate, below the function's variation scale) and $\delta = 0.8$ (large, spanning regions of significant curvature change). Cost: $d+1$ function evaluations per gradient.

\begin{remark}[Central Finite Differences]
The \emph{central} finite-difference approximation $\frac{f(x+\delta e_i) - f(x-\delta e_i)}{2\delta}$ achieves $O(\delta^2)$ truncation accuracy (versus $O(\delta)$ for forward FD) because the $O(\delta)$ terms cancel symmetrically. The total error is minimised at $\delta^* = (3\varepsilon_{\text{mach}}/|f'''|)^{1/3} \approx (3 \times 2.2 \times 10^{-16})^{1/3} \approx 10^{-5}$ for double precision, giving a minimum error of $\approx 10^{-10}$ versus $\approx 10^{-8}$ for forward FD. The cost doubles to $2d$ evaluations (since both $+\delta$ and $-\delta$ perturbations are needed), but the improved accuracy often justifies the overhead.
\end{remark}

\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Random unit vector}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates the gradient via a random one-dimensional probe, requiring only \textbf{2 function evaluations per iteration} regardless of dimension $d$ (compared to $d+1$ for finite differences). The key property: for $u_k \sim \text{Uniform}(\mathbb{S}^{d-1})$ (uniform on the sphere), $\mathbb{E}[\hat{g}_k] = \mathbb{E}\!\left[\frac{f(x+\delta u) - f(x)}{\delta} u\right] \approx \mathbb{E}[u\, u^T] \nabla f(x) = \frac{1}{d}\nabla f(x)$, since $\mathbb{E}[uu^T] = \frac{1}{d}I_d$ for the uniform spherical distribution. Thus $d \cdot \hat{g}_k$ is an approximately unbiased estimator of $\nabla f(x_k)$, and the code's step is $x_{k+1} = x_k - \alpha \hat{g}_k = x_k - \frac{\alpha}{d}(d\hat{g}_k)$, effectively using step size $\alpha/d$ for the unbiased gradient. The high variance ($\text{Var}[\hat{g}_k] = O(1/d)\|\nabla f\|^2$) limits the convergence rate to $O(1/\sqrt{k})$ even for strongly convex functions, versus $O((1-\mu/L)^k)$ for exact GD.

\begin{remark}[NRS Dimension Penalty on Strongly Convex Problems]
For $\mu$-strongly convex and $L$-smooth $f$, each NRS gradient estimate $\hat{g}_k$ is an approximately unbiased estimator of $\frac{1}{d}\nabla f(x_k)$, so NRS behaves like GD with an \emph{effective step size} $\alpha_{\mathrm{eff}} = \alpha/d$ applied to the true gradient. The effective convergence rate is therefore approximately $(1 - \mu/L)^k$ per true-gradient step, but each such step costs $d$ NRS iterations (since each NRS step extracts $1/d$ of the gradient information). Equivalently, the per-NRS-iteration rate is $(1-\mu/(dL))^k$ --- a condition number effectively inflated to $\kappa_{\mathrm{eff}} = dL/\mu$ versus $\kappa = L/\mu$ for exact GD. For Benchmark B ($d=2$, $\mu \approx 0.4$, $L \approx 10$, $\kappa = 25$): $\kappa_{\mathrm{eff}} \approx 50$, and the per-iteration rate is approximately $1 - 0.4/(2 \times 10) = 0.98$. After 120 iterations: $(0.98)^{120} \approx 0.091$ (suboptimality reduced to $\sim$9\% of initial), while exact GD gives $(0.96)^{120} \approx 0.0076$ ($\sim$0.76\%). The $\sim$$12\times$ gap in iteration efficiency explains why NRS converges slowest on Benchmark B (Figure~\ref{fig:q4_fval_B}) despite the same number of gradient-oracle \emph{calls}: each NRS call yields a noisy, dimensionally-diluted gradient estimate rather than the true gradient.
\end{remark}

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm (Nelder \& Mead, 1965) maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively reshapes it through four operations:
\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r = 1.0$): move worst vertex $x_{n+1}$ through the centroid $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{n+1})$.
  \item \textbf{Expansion} ($\gamma = 2.0$): if reflection improved, try $x_e = \bar{x} + \gamma(\bar{x} - x_{n+1})$ to accelerate progress.
  \item \textbf{Contraction} ($\rho = 0.5$): if reflection worsened, contract toward the centroid.
  \item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all vertices toward the best.
\end{enumerate}
Settings: initial simplex step $= 0.35$, iterations $= 160$. No gradient information is used.

\begin{remark}[Nelder--Mead as Implicit Curvature Estimator]
Although Nelder--Mead uses no explicit gradient or Hessian, the simplex implicitly encodes directional curvature information: after several reflection/expansion/contraction steps, the simplex tends to elongate along directions of shallow curvature (progress directions) and compress perpendicular to them (steep directions). For Rosenbrock, the simplex aligns itself with the valley axis $\nabla^2 f$'s minimum eigenvector. This geometric adaptation is the reason Nelder--Mead outperforms random search ($O(d)$-evaluations per effective step) on low-dimensional structured problems, despite lacking analytical curvature information.
\end{remark}

\subsection{Problem Setup}

Minimise Benchmark B subject to the inequality constraint $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$. The feasible set is $\mathcal{X} = \{x : x_1 \geq 0.5\}$ (a closed convex half-space).

\textbf{KKT Conditions:} The constrained problem is $\min_{x} f(x)$ s.t.\ $g(x) = 0.5 - x_1 \leq 0$. Under the KKT conditions, a point $x^*$ is a local minimum if there exists a multiplier $\mu^* \geq 0$ such that:
\begin{align}
\nabla f(x^*) + \mu^* \nabla g(x^*) &= 0 \quad \text{(stationarity)}, \label{eq:kkt_stat}\\
g(x^*) &\leq 0 \quad \text{(primal feasibility)}, \label{eq:kkt_pf}\\
\mu^* &\geq 0 \quad \text{(dual feasibility)}, \label{eq:kkt_df}\\
\mu^* g(x^*) &= 0 \quad \text{(complementary slackness)}. \label{eq:kkt_cs}
\end{align}
Since the unconstrained minimiser $x^* = (0.5824, 2.0)$ satisfies $g(x^*) = 0.5 - 0.5824 < 0$ (strictly feasible), complementary slackness forces $\mu^* = 0$, and the KKT conditions reduce to $\nabla f(x^*) = 0$ --- the unconstrained optimality condition. This confirms that the constraint is \emph{inactive} at the optimum: the constrained and unconstrained solutions coincide.

\textbf{Practical implication:} Since $x^* \approx (0.5824, 2.0)$ lies in the interior of $\mathcal{X}$, once any algorithm (projected GD or penalty) has driven its iterate into the feasible region $x_1 \geq 0.5$, it should converge to the same point as unconstrained GD. The penalty with $\lambda = 0.15$ fails precisely because it never fully drives the iterate into the feasible region --- the penalty gradient is too weak to overcome the objective's pull toward the infeasible region.

\textbf{Shadow price interpretation:} The Lagrange multiplier $\mu^*$ has an economic interpretation as the \emph{shadow price} of the constraint. By the envelope theorem, if the constraint is $x_1 \geq c$ (parameterised by $c$), then:
\begin{equation}
\frac{d f^*(c)}{d c} = -\mu^*(c),
\end{equation}
where $f^*(c) = \min_{x_1 \geq c} f(x)$. Here $\mu^* = 0$ because the constraint is inactive: tightening the constraint slightly (e.g., to $x_1 \geq 0.5824 - \epsilon$ for small $\epsilon > 0$) does not change the optimal value. However, if the constraint were $x_1 \geq c$ for $c > x_1^* = 0.5824$, the constraint would become active, $\mu^* > 0$, and increasing $c$ would increase $f^*(c)$ at rate $\mu^*$ per unit. The transition point is $c = 0.5824$: for $c \leq 0.5824$, $f^*(c) = 0.7244$ (constant); for $c > 0.5824$, $f^*(c)$ increases due to constraint binding.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = \begin{bmatrix} \max(0.5, z_1) \\ z_2 \end{bmatrix}$ \Comment{Euclidean projection onto $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method (Gradient Descent on Penalised Objective)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\State Define penalised objective: $F_{\lambda}(x) \gets f(x) + \lambda\, \max\bigl(0,\; 0.5 - x_1\bigr)$
\For{$k = 0, 1, \ldots, N-1$}
    \State $\nabla F_{\lambda}(x) \gets \nabla f(x) + \lambda \begin{bmatrix} -\mathbf{1}[x_1 < 0.5] \\ 0 \end{bmatrix}$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$), penalty method with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$, reduced to control the steeper penalty gradient), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected gradient descent} guarantees hard feasibility at every iteration via the projection oracle. For the constraint $x_1 \geq 0.5$, the projection $\Pi_{\mathcal{X}}(x) = (\max(0.5, x_1), x_2)^T$ has $O(1)$ cost. Under standard assumptions (the objective is $L$-smooth, $\mu$-strongly convex on $\mathcal{X}$), projected GD converges at rate $(1-\mu/L)^k$, the same as unconstrained GD. For more complex constraints (e.g., a convex set defined by a semidefinite constraint), the projection oracle can be expensive ($O(d^3)$ for an SDP), and the penalty method becomes more attractive.

\textbf{Penalty methods} trade exact feasibility for algorithmic simplicity: only a modified gradient $\nabla f + \lambda \partial \max(0, 0.5-x_1)$ is required. The weight $\lambda$ controls the trade-off: $\lambda \to \infty$ recovers the hard-constrained solution (as $F_\lambda \to f$ on the feasible set and $F_\lambda \to \infty$ on the infeasible set), but the ill-conditioning of $F_\lambda$ worsens as $\lambda$ increases, requiring smaller step sizes. This ``penalty barrier'' is why $\alpha$ must decrease from 0.05 ($\lambda \leq 1.8$) to 0.03 ($\lambda = 4.5$): the Lipschitz constant of $\nabla F_\lambda$ is $\approx L + \lambda$, and stability requires $\alpha < 2/(L+\lambda)$.

Concretely, for Benchmark B near the constraint boundary: $L = \lambda_{\max}(\nabla^2 f) \approx 10$. With $\lambda = 4.5$, $L + \lambda = 14.5$, so $\alpha_{\max} = 2/14.5 \approx 0.138$. The used step $\alpha = 0.03$ is well within this bound. With $\lambda = 0.15$: $L + \lambda = 10.15$, $\alpha_{\max} \approx 0.197$; the used $\alpha = 0.05$ is safe but the small $\lambda$ means the penalty is too weak to enforce feasibility. To achieve $\epsilon$-feasibility with pure penalty, one needs $\lambda \geq \|{\nabla f}\|/\epsilon$ at the infeasible point, which for the Q5 problem requires $\lambda \geq 0.836/0.1 = 8.36$ for a $0.1$-violation tolerance --- explaining why $\lambda = 0.15$ fails.

\textbf{Augmented Lagrangian comparison:} The augmented Lagrangian method (Hestenes--Powell--Rockafellar) defines the augmented Lagrangian as:
\begin{equation}
\mathcal{L}_\rho(x, \nu) = f(x) + \nu\, g(x) + \frac{\rho}{2}[g(x)]^2,
\end{equation}
where $g(x) = \max(0, 0.5-x_1)$ is the constraint violation and $\nu$ is the Lagrange multiplier. The multiplier update rule is $\nu_{k+1} = \nu_k + \rho\, g(x_k^*)$ where $x_k^* = \arg\min_x \mathcal{L}_\rho(x, \nu_k)$. Crucially, this achieves $g(x^*) = 0$ (exact feasibility) at \emph{finite} $\rho$, because the multiplier $\nu_k$ is adjusted to account for residual constraint violation. This contrasts with the pure penalty method which requires $\lambda \to \infty$ and worsens the Lipschitz constant $L + \lambda \to \infty$. The augmented Lagrangian maintains the same Lipschitz constant $L + \rho$ regardless of convergence progress, since $\rho$ is fixed throughout.

\begin{remark}[Why the AL Achieves Feasibility at Finite $\rho$]
For the pure penalty method, the stationarity of $F_\lambda(x)$ at $x^*(\lambda)$ reads:
$\nabla f(x^*(\lambda)) + \lambda\max(0, g(x^*(\lambda)))\nabla g(x^*(\lambda)) = 0$.
For $g(x^*(\lambda)) = 0$ to hold, either $\lambda \to \infty$ or $\nabla g = 0$ at the boundary (not true here). Hence $g(x^*(\lambda)) = O(1/\lambda)$ --- residual infeasibility that vanishes only asymptotically.

For the AL at fixed $\rho$ and current multiplier $\nu_k$, the inner-problem stationarity at $x_k^*$ is:
\[
\nabla f(x_k^*) + \bigl[\underbrace{\nu_k + \rho\, g(x_k^*)}_{\nu_k^{\text{eff}}}\bigr]\nabla g(x_k^*) = 0.
\]
This is precisely the KKT stationarity condition with \emph{effective multiplier} $\nu_k^{\text{eff}} = \nu_k + \rho\, g(x_k^*)$. The update $\nu_{k+1} = \nu_k + \rho\, g(x_k^*)$ drives $\nu_k$ toward the true KKT multiplier $\lambda^*$: when $g(x_k^*) > 0$ (violated), $\nu_{k+1} > \nu_k$, increasing the effective penalty on constraint violation in the next inner problem. At convergence $\nu^* = \lambda^*$, and the stationarity condition becomes the exact KKT condition, forcing $g(x^*) = 0$ regardless of how large or small $\rho$ is. The pure penalty, by contrast, has no dual variable to iterate toward $\lambda^*$, so it relies entirely on $\lambda \to \infty$ to enforce the constraint.
\end{remark}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a convex box polytope with 4 vertices: $(0.5,-5)$, $(5,-5)$, $(5,10)$, $(0.5,10)$).

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is obtained by coordinate-wise minimisation (since the objective is separable and the feasible set is a box):
\begin{equation}
x_i^{\star} = \begin{cases} \ell_i & \text{if } a_i > 0 \\ u_i & \text{if } a_i < 0 \end{cases}
\implies x^{\star} = (0.5,\; -5), \quad f^{\star} = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}
This illustrates the \textbf{fundamental theorem of linear programming}: the optimum of a linear objective over a non-empty bounded polytope $\mathcal{X}$ is attained at a vertex.

\begin{proposition}[LP Optimal at Vertex]\label{prop:lp}
Let $f(x) = a^T x$ (linear) and $\mathcal{X}$ a non-empty bounded polytope. Then $\min_{x \in \mathcal{X}} f(x)$ is attained at a vertex of $\mathcal{X}$.
\end{proposition}
\begin{proof}[Proof sketch]
Suppose $x^* \in \mathcal{X}$ achieves the minimum. If $x^*$ is not a vertex, it lies in the interior of some edge or face, i.e.\ $x^* = \lambda y + (1-\lambda)z$ for some $y, z \in \mathcal{X}$, $y \neq z$, $\lambda \in (0,1)$. By linearity: $f(x^*) = \lambda f(y) + (1-\lambda)f(z)$. Since $x^*$ is optimal, $f(y) \geq f(x^*)$ and $f(z) \geq f(x^*)$; thus $f(y) = f(z) = f(x^*)$. Both $y$ and $z$ are also minimisers. Applying this argument recursively (the polytope has finitely many vertices), we eventually reach a vertex with the same objective value.
\end{proof}

Since $f$ has no curvature, no interior point can be optimal unless it coincides with a boundary optimum.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear Minimisation Oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Fixed-step convex combination, $\gamma = 1-\beta$}
\EndFor
\end{algorithmic}
\end{algorithm}

The Frank--Wolfe method (Frank \& Wolfe, 1956) avoids explicit projection by replacing the projection step with a \emph{linear subproblem}: at each iteration, find the vertex $z_k$ of $\mathcal{X}$ that minimises the linearised objective $\nabla f(x_k)^T z$. For a box constraint, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} \ell_i & \text{if } (\nabla f(x_k))_i > 0, \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0. \end{cases}
\end{equation}
The iterate $x_{k+1} = (1-\gamma)x_k + \gamma z_k$ (where $\gamma = 1-\beta$ is the step size toward $z_k$) is a convex combination of $x_k$ and $z_k$, both in $\mathcal{X}$, so \textbf{feasibility is maintained by construction} without any projection oracle.

\textbf{Fixed-step vs decreasing step:} The implementation uses a fixed mixing parameter $\beta$ (i.e., fixed step $\gamma = 1-\beta$), whereas the standard Frank--Wolfe uses the decreasing schedule $\gamma_k = 2/(k+2)$. For smooth convex $f$ with Lipschitz gradient constant $L$, the standard schedule achieves:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2},
\end{equation}
where $D = \max_{x,y \in \mathcal{X}}\|x-y\|$ is the \emph{diameter} of $\mathcal{X}$. For our box $\mathcal{X} = [0.5, 5] \times [-5, 10]$, the diameter is $D = \sqrt{(5-0.5)^2 + (10-(-5))^2} = \sqrt{4.5^2 + 15^2} = \sqrt{20.25 + 225} = \sqrt{245.25} \approx 15.66$.

For the interior optimum $f_{\text{int}}(x) = (x_1-1)^2 + (x_2-5)^2$, the Hessian is $H = 2I$, so $L = 2$. The convergence bound becomes:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2 \cdot 2 \cdot 245.25}{k+2} = \frac{981}{k+2}.
\end{equation}
At $k = 180$: the bound gives $f - f^* \leq 5.37$, which is loose by $\sim 1200\times$ (empirical: $f = 0.0045$). The bound is worst-case: it assumes the starting point is at the antipodal vertex $D = 15.66$ away, whereas the actual starting distance is $\|x_0 - x^*\| = \|(1,1) - (1,5)\| = 4 \ll D$. Tighter instance-specific bounds replace $D^2$ with $\|x_0 - x^*\|^2$:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{k+2} = \frac{2 \cdot 2 \cdot 16}{182} \approx 0.352 \quad (k=180).
\end{equation}
This tighter bound ($0.352$) is still $78\times$ looser than the empirical value ($0.0045$), indicating that even the instance-specific bound is not tight for this problem.

\begin{proposition}[Frank--Wolfe $O(1/k)$ Rate via Descent Lemma]\label{prop:fw-rate}
Let $f$ be $L$-smooth and convex on $\mathcal{X}$ with diameter $D$. Using step schedule $\gamma_k = 2/(k+2)$, the Frank--Wolfe iterates satisfy $f(x_k) - f^{\star} \leq 2LD^2/(k+2)$ for all $k \geq 0$.
\end{proposition}
\begin{proof}[Proof sketch]
By $L$-smoothness applied to the convex combination $x_{k+1} = x_k + \gamma_k(z_k - x_k)$:
\begin{equation}
f(x_{k+1}) \leq f(x_k) + \gamma_k\langle\nabla f(x_k),\, z_k - x_k\rangle + \tfrac{L\gamma_k^2}{2}\|z_k - x_k\|^2
\leq f(x_k) - \gamma_k\mathcal{G}(x_k) + \tfrac{L\gamma_k^2 D^2}{2},
\end{equation}
where $\mathcal{G}(x_k) = \langle\nabla f(x_k), x_k - z_k\rangle$ is the FW gap and $\|z_k - x_k\| \leq D$. Convexity of $f$ gives $\mathcal{G}(x_k) \geq f(x_k) - f^{\star}$ (since $z_k \in \mathcal{X}$ and $f(z_k) \geq f^{\star} + \langle\nabla f(x_k), z_k - x^{\star}\rangle$ ... simplifying via $\min_{z\in\mathcal{X}} f^{\star}$). Setting $h_k = f(x_k) - f^{\star}$ and $\gamma_k = 2/(k+2)$:
\begin{equation}
h_{k+1} \leq h_k\!\left(1 - \tfrac{2}{k+2}\right) + \tfrac{2LD^2}{(k+2)^2}.
\end{equation}
Claim: $h_k \leq 2LD^2/(k+2)$. The base case $k=0$ holds whenever $f(x_0) - f^{\star} \leq LD^2$ (satisfied by $L$-smoothness since $\|x_0 - x^{\star}\| \leq D$). For the inductive step, substituting $h_k \leq 2LD^2/(k+2)$:
\[
h_{k+1} \leq \frac{2LD^2}{k+2}\cdot\frac{k}{k+2} + \frac{2LD^2}{(k+2)^2}
= \frac{2LD^2(k+1)}{(k+2)^2} \leq \frac{2LD^2}{k+3},
\]
where the last inequality uses $(k+1)(k+3) \leq (k+2)^2$, i.e., $k^2+4k+3 \leq k^2+4k+4$. \qed
\end{proof}

Fixed-step variants achieve similar rates when $\gamma = 1-\beta$ is appropriately chosen, but may oscillate near the optimum since the step does not decay, and in the boundary case create a limit cycle as analyzed above.

\subsection{Interior vs Boundary Behaviour: Theoretical Analysis}

\textbf{Interior optimum:} Frank--Wolfe exhibits characteristic ``zig-zagging'' in the approach to interior optima. Each LMO call returns a vertex $z_k \in \{(0.5,-5), (5,-5), (5,10), (0.5,10)\}$; near an interior optimum the gradient $\nabla f(x_k) \approx 0$ is small, so $z_k$ is determined by tiny gradient fluctuations, causing it to oscillate between opposite vertices.

The \textbf{Frank--Wolfe gap} (also known as the duality gap or primal gap) is defined as:
\begin{equation}
\mathcal{G}(x_k) = \max_{z \in \mathcal{X}} \langle -\nabla f(x_k), z - x_k \rangle = \langle \nabla f(x_k), x_k - z_k \rangle \geq 0.
\end{equation}
This quantity satisfies $f(x_k) - f^{\star} \leq \mathcal{G}(x_k)$ for convex $f$ (by convexity: $f(z) \geq f(x) + \langle \nabla f(x), z-x \rangle$ for all $z, x$), providing a computable convergence certificate without knowing $f^{\star}$. For the standard decreasing-step Frank--Wolfe with $\gamma_k = 2/(k+2)$, the gap satisfies $\mathcal{G}(x_k) \leq 2LD^2/(k+2)$ where $D = \text{diam}(\mathcal{X})$.

\textbf{Boundary optimum:} Frank--Wolfe is naturally suited to boundary optima because the gradient $\nabla f(x^*)$ at the constrained optimum is nonzero (it points toward the infeasible region), so the LMO consistently selects the vertex nearest the constrained optimum, driving the iterate toward the constraint boundary without explicit projection. The Frank--Wolfe gap at the constrained optimum satisfies the KKT condition: $\langle \nabla f(x^*), x - x^* \rangle \geq 0$ for all $x \in \mathcal{X}$, which is equivalent to $\mathcal{G}(x^*) = 0$.

\textbf{Comparison with Projected GD:} Projected GD achieves the constrained minimum in far fewer iterations on well-conditioned problems (linear convergence at rate $1-\mu/L$), but requires a projection oracle. Frank--Wolfe replaces the projection with a linear subproblem (the LMO), which is advantageous when $\mathcal{X}$ is a complex polytope or spectrahedron (e.g., a nuclear-norm ball $\|W\|_* \leq r$ in matrix optimisation, where the LMO has an $O(d^2)$ SVD solution vs $O(d^3)$ for projection). The $O(1/k)$ convergence of Frank--Wolfe is generally slower than projected GD's linear convergence, but each iteration is cheaper when the LMO is efficient.

\begin{remark}[LMO as a Linear Programme]
The Frank--Wolfe linear minimisation oracle $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ is precisely a linear programme (LP) with objective $a = \nabla f(x_k)$. By Proposition~\ref{prop:lp} (LP optimal at vertex), $z_k$ is always a vertex of $\mathcal{X}$. For a box constraint $\ell \leq x \leq u$, this reduces to coordinate-wise minimisation: $(z_k)_i = \ell_i$ if $(\nabla f(x_k))_i > 0$ else $u_i$. This connection explains the ``zig-zagging'' phenomenon near interior optima: as $\nabla f(x_k) \to 0$, the LMO becomes sensitive to the sign of each gradient component, and tiny gradient fluctuations cause $z_k$ to jump between opposite vertices.
\end{remark}

\begin{remark}[Frank--Wolfe Gap as Optimality Certificate]
The FW gap $\mathcal{G}(x_k) = \langle \nabla f(x_k), x_k - z_k \rangle$ is a tight optimality certificate: $\mathcal{G}(x_k) = 0$ if and only if $x_k$ satisfies the first-order KKT condition $\langle \nabla f(x_k), y - x_k \rangle \geq 0$ for all $y \in \mathcal{X}$, i.e., $x_k$ is a constrained stationary point. For convex $f$, this is equivalent to global optimality. The gap is computable from within the algorithm (since $z_k$ is already computed), providing a free convergence certificate without needing to know $f^{\star}$.
\end{remark}

The following key conclusions emerge:

\paragraph{Adaptive step-size methods (Q1).} Polyak's step size achieves superior convergence when $f^{\star}$ is exactly known, but is critically sensitive to misspecification: a 47\% underestimate of $f^{\star}$ (specifying 0 instead of 0.4826) causes 100$\times$ degradation on Benchmark A. Adagrad and Heavy Ball are the most reliable alternatives, achieving GD-level performance on well-conditioned problems and substantially outperforming GD on Rosenbrock. RMSprop's exponential forgetting makes it the most stable near the minimum, but it requires careful tuning of $\alpha_0$.

\paragraph{Momentum and Adam (Q2).} Nesterov acceleration provides the most consistent improvement across all three benchmarks ($7\times$ over GD on Rosenbrock in 150 steps), confirming the theoretical $O(1/k^2)$ acceleration for convex problems. Adam combines momentum and adaptive scaling effectively for anisotropic problems (Benchmark B) but underperforms Nesterov on the non-convex Rosenbrock, where the per-coordinate scaling is less helpful than directional momentum. Mini-batch SGD demonstrates a fundamental noise-efficiency trade-off: smaller batches update more frequently per epoch but with higher gradient variance.

\paragraph{Newton's method (Q3).} The most powerful method when Hessian information is available and the function is well-approximated by its quadratic model. On the quadratic Benchmark A, convergence is exact in one step. On the moderately nonlinear Benchmark B, convergence occurs in 5 steps with a near-unit step ($\alpha = 0.85$). Rosenbrock requires heavy damping ($\alpha = 0.22$), which partially negates Newton's advantage but still achieves 5$\times$ better convergence than GD in 20 vs 80 iterations. The computational cost per iteration ($O(d^3)$ for exact Hessian solve) is offset by the dramatically reduced iteration count.

\paragraph{Derivative-free methods (Q4).} Finite differences with an appropriate step size ($\delta = 0.05$) effectively recovers full-gradient performance at only $d+1$ function evaluations per step. Large $\delta$ ($= 0.8$) causes significant truncation error that degrades convergence. Nesterov random search converges slowest but requires only 2 function evaluations per step regardless of dimension, making it the only method here that scales to large $d$. Nelder--Mead achieves competitive results on the 2D Rosenbrock without any gradient information, adapting its simplex geometry to the valley shape. Grid search provides global guarantees but is intractable beyond $d \approx 3$.

\paragraph{Constrained optimisation (Q5).} Projected gradient descent enforces the constraint $x_1 \geq 0.5$ as a hard requirement from the first iteration, achieving zero constraint violation throughout. The projection for a half-space is trivially $O(1)$. Penalty methods provide a softer alternative: $\lambda \geq 4.5$ achieves constraint satisfaction within 10 iterations, while $\lambda = 0.15$ fails to enforce feasibility. The required reduction in step size for high $\lambda$ demonstrates the stability-feasibility trade-off inherent in penalty approaches.

\paragraph{Frank--Wolfe and linear programming (Q6).} The Frank--Wolfe algorithm maintains feasibility implicitly via convex combinations, eliminating the projection oracle. For interior optima, the $O(1/k)$ convergence rate is confirmed empirically, with smaller step sizes (larger $\beta$) producing smoother convergence near the optimum despite the vertex-oscillation phenomenon. For boundary optima, Frank--Wolfe naturally approaches the constrained minimum as the LMO consistently selects nearby vertices. The linear programming result confirms the fundamental theorem: a linear objective over a bounded polytope is minimised at a vertex.

\paragraph{Overall ranking by difficulty and method suitability.}
\begin{itemize}[nosep]
\item \textbf{Well-conditioned quadratic} (Benchmark A, $\kappa \approx 1.12$): any first-order method suffices; Newton converges in 1 step (exact). GD converges in $\sim$50 steps.
\item \textbf{Mildly nonlinear, moderate $\kappa$} (Benchmark B, $\kappa \in [3.5, 6.9]$): all momentum methods (Nesterov, Adam, Heavy Ball) and Newton all converge to $f^* = 0.7244$ within 20--30 steps. Method choice makes little difference here.
\item \textbf{Ill-conditioned nonlinear} (Benchmark C, $\kappa \approx 2508$): this is the discriminating benchmark. Theoretical GD requires $\sim$57000 steps for $10^{-8}$ accuracy; Nesterov requires only $\sim$220. In 150 steps, Nesterov ($f = 0.479$) achieves $7\times$ better convergence than GD ($f = 3.39$). Polyak (with known $f^*=0$) achieves the best result ($f = 0.009$ in 120 steps). Newton with damping reaches $f = 0.686$ in 20 steps. Derivative-free Nelder--Mead is competitive ($f < 0.01$ in 160 steps) without any gradient information.
\end{itemize}

\paragraph{Lessons for algorithm selection in practice.} The key factors driving method selection are: (1) \emph{gradient availability} --- if gradients are cheap, momentum methods dominate over derivative-free approaches; (2) \emph{Hessian availability} --- Newton's method is worth its $O(d^3)$ cost for small $d$ with nearly-quadratic objectives; (3) \emph{condition number} --- for $\kappa \gg 1$, adaptive methods (AdaGrad, RMSprop, Adam) or Nesterov acceleration are essential; (4) \emph{knowledge of $f^*$} --- Polyak step is a powerful oracle-based method but inapplicable when $f^*$ is unknown; (5) \emph{constraint complexity} --- projected GD is simple for half-spaces; Frank--Wolfe excels for polytopes with efficient LMOs; penalty methods require no special structure but may converge slowly for large $\lambda$.

\paragraph{Caveats and scope of conclusions.} These conclusions are specific to 2D problems. In high dimensions ($d \gg 2$), several rankings would change: Nelder--Mead scales as $O(d^2)$ per iteration and fails in $d > 20$; Adagrad and RMSprop become more advantageous due to coordinate-wise scaling (especially for sparse problems); Newton's $O(d^3)$ per-iteration cost becomes prohibitive. The hyperparameters were chosen by trial and error for these specific benchmarks; in practice, adaptive tuning (learning rate schedules, warm restarts, or Bayesian hyperparameter optimisation) is needed. The Rosenbrock benchmark, while a standard test case, is deterministic and low-dimensional --- stochastic problems (e.g., deep learning training) favour Adam and SGD-based methods where the noise averaging properties of momentum are essential.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm (Fixed Step)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
