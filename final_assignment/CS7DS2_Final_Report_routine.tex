\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark A), a toy neural network loss with sinusoidal perturbation (Benchmark B), and the ill-conditioned Rosenbrock function (Benchmark C). Six categories of methods are investigated: (1)~adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball); (2)~momentum-based and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD); (3)~Newton's method with second-order local approximations; (4)~derivative-free approaches (forward finite differences, Nesterov random search, Nelder--Mead simplex, grid search); (5)~constrained optimisation via projected gradient descent and penalty methods; and (6)~the Frank--Wolfe conditional gradient algorithm for linear-constrained problems. Key findings include: the Polyak step diverges catastrophically when $f^{\star}$ is misspecified, achieving the best convergence only on Rosenbrock where $f^{\star}=0$ is exact; Nesterov acceleration achieves $f=0.479$ on Rosenbrock versus $f=3.388$ for gradient descent baseline; Newton's method solves the quadratic Benchmark A in one iteration but requires damping ($\alpha=0.22$) on Rosenbrock; finite-difference gradients with $\delta=0.05$ closely track exact gradients while $\delta=0.8$ converges poorly; projected gradient descent enforces hard feasibility from iteration one unlike penalty methods; and the Frank--Wolfe algorithm naturally maintains feasibility without projection. Each method is evaluated in terms of convergence speed, trajectory behaviour, stability, and computational trade-offs.
\end{abstract}

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for full reproducibility.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a \emph{strongly convex quadratic} with constant Hessian $H = \frac{1}{m}X^T X \succ 0$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Since $X$ has i.i.d.\ $\mathcal{N}(0,1)$ entries and $m = 1000 \gg d = 2$, by the law of large numbers $H \approx I_2$, giving $\lambda_{\min}(H) \approx 0.925$, $\lambda_{\max}(H) \approx 1.033$, and condition number $\kappa \approx 1.117$. This near-isotropic curvature ($\kappa \approx 1$) explains why all first-order methods converge rapidly on Benchmark A. The OLS minimiser $\theta^{\star}_{\mathrm{OLS}} = (X^TX)^{-1}X^Ty \approx (3.028, 4.017)$ approximates the true parameter $(3, 4)$. Due to observation noise ($\sigma = 1$), the minimum loss is $J^{\star} \approx 0.4826 \neq 0$. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \mathrm{diag}(2 - \sin(x_1),\ 10)$. Since $\sin(x_1) \in [-1, 1]$, the diagonal element $2 - \sin(x_1) \in [1, 3]$, giving a condition number that varies between approximately $10/3 \approx 3.3$ and $10/1 = 10$ across the domain. At the minimum ($x_1^{\star} \approx 0.5823$), $\kappa \approx 10/(2 - \sin(0.5823)) \approx 10/1.452 \approx 6.9$, representing moderate anisotropy. The minimum lies at approximately $(0.5824,\, 2.0)$ with $f^{\star} \approx 0.7244$, obtained by solving $2(x_1-1)+\cos(x_1)=0$ numerically (the exact root $x_1^{\star} \approx 0.5824$ satisfies the stationarity condition to five decimal places). Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian is:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1,1)$: $\nabla^2 f(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}$, with eigenvalues satisfying $\lambda^2 - 1002\lambda + 400 = 0$, yielding $\lambda_1 \approx 1001.6$ and $\lambda_2 \approx 0.4$, so the condition number $\kappa = \lambda_1/\lambda_2 \approx 2504$. This extreme ill-conditioning, combined with the narrow curved valley along $x_2 = x_1^2$, makes the Rosenbrock one of the most challenging standard test functions. Away from the optimum the Hessian is not positive definite, which has critical implications for Newton's method. Starting point: $x_0 = [-1, 1]^T$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations. The fundamental limitation of standard gradient descent with fixed step $\alpha$---that $\alpha$ must satisfy $\alpha < 2/L$ globally yet a small $\alpha$ implies slow convergence---motivates methods that adapt the step to local curvature information.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step is motivated by the Cauchy--Schwarz inequality applied to the subgradient inequality:
\begin{equation}
f(x_k) - f^{\star} \leq \langle \nabla f(x_k),\, x_k - x^{\star} \rangle \leq \|\nabla f(x_k)\|\,\|x_k - x^{\star}\|,
\end{equation}
which suggests choosing $\alpha_k = (f(x_k)-f^{\star})/\|\nabla f(x_k)\|^2$ to maximally reduce the distance to the optimum in one step. Specifically, one can show:
\begin{equation}
\|x_{k+1} - x^{\star}\|^2 = \|x_k - \alpha_k \nabla f(x_k) - x^{\star}\|^2 \leq \|x_k - x^{\star}\|^2 - \frac{(f(x_k)-f^{\star})^2}{\|\nabla f(x_k)\|^2}.
\end{equation}
For strongly convex functions ($\mu > 0$) with known $f^{\star}$, this yields $\|x_{k+1} - x^{\star}\|^2 \leq (1-\mu/L)\|x_k - x^{\star}\|^2 = (1-1/\kappa)\|x_k - x^{\star}\|^2$---a linear convergence rate independent of the absolute Lipschitz constant $L$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate cumulative sum of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad adapts the learning rate per coordinate based on the cumulative history of squared gradients. Coordinates with consistently large gradients receive smaller effective steps, while rare features receive larger steps. The effective step size $\alpha_k^{(i)} = \alpha_0/\sqrt{\sum_{j=0}^{k} g_j^{(i)2}}$ is \emph{monotonically decreasing}, which ensures stability but can cause premature stagnation.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's vanishing step size by replacing the cumulative sum with an exponential moving average (decay $\beta = 0.9$). The effective memory horizon is approximately $1/(1-\beta) = 10$ recent gradient observations, allowing the step size to recover when gradients diminish.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method accumulates a velocity term $z_k$ that damps oscillations across the valley and accelerates along persistent gradient directions. For a strongly convex quadratic with condition number $\kappa$, the optimal Heavy Ball parameters yield a convergence rate of $\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$ per iteration, significantly faster than gradient descent's rate of $\left(\frac{\kappa-1}{\kappa+1}\right)^2$.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Sensitivity to $f^{\star}$:} Polyak is highly sensitive. Even small misspecification $\Delta f^{\star} = f^{\star}_{\mathrm{true}} - f^{\star}_{\mathrm{specified}} > 0$ causes divergence because $\alpha_k \approx \Delta f^{\star}/\|\nabla f\|^2 \to \infty$ as $\|\nabla f\| \to 0$. In practice, $f^{\star}$ is rarely known exactly, limiting Polyak's applicability to problems where a valid lower bound is known (e.g., losses with non-negativity constraints, or problems where the function's theoretical lower bound is known from domain knowledge). A robustified variant uses an estimated $f^{\star}$ that adapts over iterations, but this adds complexity.
\item \textbf{Effect of condition number:} Benchmark A ($\kappa \approx 1.117$) is the easiest for all methods; the near-circular contours mean that any descent direction is nearly equally good. Benchmark B ($\kappa \approx 3.5$--$6.9$) presents moderate challenge. Benchmark C ($\kappa \approx 2504$ at the optimum, much larger near the start) is where method differences matter most. On the nearly-isotropic Benchmark A, even simple GD with $\alpha = 0.08 < 2/L \approx 1.93$ converges quickly, and the advantage of adaptive methods is negligible.
\item \textbf{Convergence speed:} When $f^{\star}$ is known (Rosenbrock), Polyak is fastest. Otherwise, Heavy Ball and Adagrad match or exceed GD. The Heavy Ball method's momentum acceleration is most pronounced on Benchmark C: its momentum parameter $\beta = 0.86$ effectively reduces the condition number from $\kappa$ to $\sqrt{\kappa}$ in ideal settings, leading to faster escape from the narrow valley.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging of the second moment. Heavy Ball exhibits the characteristic overshoot-oscillation-damping pattern of underdamped systems, analogous to a physical ball rolling in a bowl with friction coefficient $1-\beta$.
\item \textbf{Stability:} GD is most stable but slowest. Adagrad is unconditionally stable (step size monotonically decreasing) at the cost of eventual stagnation near the minimum; this can be remedied with restarts. RMSprop and Heavy Ball require careful tuning of $\alpha$ and $\beta$; the stability region in $(\alpha, \beta)$ space for Heavy Ball on a quadratic is $\alpha(1+\beta) < 2/L$.
\end{itemize}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Accelerated Gradient Descent}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead step}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than the current iterate $x_k$. This ``correction before overshoot'' strategy allows the momentum to be used more aggressively without instability.

\begin{theorem}[Nesterov's Optimal Rate]
For an $L$-smooth convex function $f$, Nesterov's accelerated gradient descent with $\alpha = 1/L$ and the schedule $\beta_k = (k-1)/(k+2)$ achieves:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} = O\!\left(\frac{1}{k^2}\right),
\end{equation}
which is optimal among all first-order methods for smooth convex optimisation (matching the $\Omega(1/k^2)$ lower bound).
\end{theorem}

This $O(1/k^2)$ rate is a quadratic improvement over gradient descent's $O(1/k)$ rate for smooth convex problems, explaining Nesterov's empirically superior performance on the benchmarks.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (RMSprop) estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction for first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias correction for second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines gradient momentum (via $m_t$, equivalent to RMSprop with additional momentum) with per-coordinate adaptive scaling (via $v_t$). The bias correction terms $(1-\beta_1^t)$ and $(1-\beta_2^t)$ compensate for the initialisation at zero, which would otherwise bias the early iterates. In the deterministic setting, Adam reduces to a combination of momentum and normalised gradient descent. The update rule can be rewritten as $x_{t+1} = x_t - \alpha \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$, where the division normalises each gradient component by its recent root-mean-square magnitude.

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch. \caption{Mini-Batch SGD}

    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[SGD Variance and Convergence Floor]
For the linear regression problem with noise variance $\sigma^2$, the optimal parameter $\theta^{\star}_{\mathrm{OLS}}$ yields $J(\theta^{\star}_{\mathrm{OLS}}) = \frac{\sigma^2}{2}\,(1 - \frac{2}{m}) \approx \frac{\sigma^2}{2}$ for large $m$. Thus $J^{\star} \propto \sigma^2$, confirming the factor-36 increase in the loss floor when $\sigma$ increases by factor 6. In the high-noise setting, the irreducible variance $\mathrm{Var}(\hat{g}_{\mathcal{B}}) \approx \frac{\sigma^2}{b} + \frac{\sigma^4}{b^2}$ of the mini-batch gradient becomes comparable across batch sizes when $\sigma^2/b$ is dominated by the observation noise $\sigma^2$, explaining why both batch sizes converge to the same floor.
\end{remark}

\subsection{Comparative Discussion}

Nesterov's superiority over Adam on Rosenbrock reveals a fundamental structural difference between these two methods:

\textbf{Nesterov on Rosenbrock:} Along the valley floor $x_2 \approx x_1^2$, the gradient consistently points in a direction tangent to the valley. Nesterov accumulates momentum in this direction over successive iterations, amplifying progress. The adaptive momentum coefficient $\beta_k = (k-1)/(k+2)$ grows towards $\beta_{\max} = 0.90$, allowing increasingly large effective steps in the valley direction. This behaviour is aligned with the Rosenbrock's geometric structure.

\textbf{Adam on Rosenbrock:} Adam normalises each gradient component by $\sqrt{\hat{v}_t + \varepsilon}$. On Rosenbrock, the gradient component $g_2 = 200(x_2 - x_1^2)$ is large when far from the valley floor but small along it, while $g_1$ varies more irregularly. Adam's per-coordinate normalisation reduces the effective gradient signal by $1/\sqrt{\hat{v}_t}$, which tends to equalise the two components and disrupts the directional momentum needed to traverse the curved valley. This normalisation is a feature (not a bug) in deep learning, where gradient scales vary wildly across layers, but it is a disadvantage on geometrically coherent problems.

\textbf{General principle:} Nesterov/Heavy Ball methods that preserve gradient direction are preferred when the objective has a well-defined ``ridge'' or ``valley'' structure. Adaptive per-coordinate methods (Adam, RMSprop) are preferred when the natural coordinate system aligns with the problem's curvature axes and scales vary dramatically per coordinate.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (tangent) approximation} around $x_0 = 0.25$:
\begin{equation}
g_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.0039 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + \tfrac{1}{2}\cdot 0.375\,(x - 0.25)^2.
\end{equation}

Newton's method minimises the second-order approximation $g_2(x)$ at each step: setting $g_2'(x) = 0$ gives the Newton update $x_{k+1} = x_k - g'(x_k)/g''(x_k)$. This corresponds to the exact minimum of the local quadratic model, contrasting with the gradient step which merely moves in the descent direction without using curvature information.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, Hessian regularisation $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solution of $H_k p = g_k$)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The regularisation $\lambda I$ (with $\lambda = 10^{-8}$) ensures $H_k + \lambda I \succ 0$, making the Newton direction $p_k = (H_k + \lambda I)^{-1} g_k$ well-defined even when $H_k$ is indefinite. The damping factor $\alpha < 1$ implements a \emph{damped Newton step}, trading off the full Newton step's quadratic convergence for improved global behaviour when the Hessian model is inaccurate far from the optimum.

\begin{remark}[Hessian Indefiniteness on Rosenbrock]
At the starting point $x_0 = (-1, 1)$, the Rosenbrock Hessian is:
\begin{equation}
\nabla^2 f(-1, 1) = \begin{bmatrix} 2 + 1200\cdot 1 - 400\cdot 1 & -400\cdot(-1) \\ 400 & 200 \end{bmatrix} = \begin{bmatrix} 802 & 400 \\ 400 & 200 \end{bmatrix}.
\end{equation}
$\det = 802\cdot 200 - 400^2 = 160400 - 160000 = 400 > 0$ and trace $= 1002 > 0$, so $H$ is positive definite here. However, along the valley floor ($x_2 \approx x_1^2$) for $x_1$ close to zero, $h_{11} = 2 + 1200x_1^2 - 400x_2 \approx 2 - 400x_1^2$ can become very small, causing near-singularity. Full Newton steps ($\alpha = 1$) overshoot the valley in these regions; the damping $\alpha = 0.22$ limits step length and maintains stability throughout the trajectory.
\end{remark}

\begin{theorem}[Quadratic Convergence of Newton's Method]
Let $f$ be twice continuously differentiable with $\nabla^2 f$ Lipschitz continuous (constant $M$). If $x_0$ is sufficiently close to $x^{\star}$ with $\nabla^2 f(x^{\star}) \succ 0$, then the full Newton step ($\alpha = 1$) satisfies:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\,\|x_k - x^{\star}\|^2,
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^{\star}))$. This super-linear (quadratic) convergence means that once the iterates enter the neighbourhood of $x^{\star}$, the number of accurate digits doubles each iteration.
\end{theorem}

The critical requirement is that $x_0$ must lie within the Newton basin of attraction. For the Rosenbrock function, the highly variable Hessian ($\nabla^2 f$ changes dramatically as $x_1$ varies) means the basin is small near the starting point $(-1, 1)$, necessitating a damped step ($\alpha = 0.22$).

Settings: GD uses 80 iterations; Newton uses 20 iterations with $\lambda = 10^{-8}$.

\subsection{Computational Cost and Practical Considerations}

Each Newton iteration requires: (i) evaluating $\nabla^2 f(x_k)$ at cost $O(d^2)$; (ii) solving the linear system $H_k p_k = g_k$ at cost $O(d^3)$ via Cholesky factorisation. For $d = 2$ the cost is negligible, but for large $d$ (e.g., deep learning with $d \sim 10^7$), the $O(d^3)$ cost is prohibitive. In contrast, GD costs $O(d)$ per iteration. Despite requiring far fewer iterations, Newton's per-iteration cost limits its practical use to low-dimensional problems or Quasi-Newton approximations (e.g., L-BFGS) that avoid the full Hessian.

\subsection{Part A: Finite Difference Gradient Approximation (Benchmark B)}

\subsubsection{Forward Finite Difference}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad i = 1, \ldots, d.
\end{equation}
By Taylor expansion, the approximation error is:
\begin{equation}
\left|\frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| \leq \frac{\delta}{2}\,\left\|\frac{\partial^2 f}{\partial x_i^2}\right\|_\infty + \frac{2\varepsilon_{\mathrm{mach}}}{\delta},
\end{equation}
where the first term is truncation error $O(\delta)$ and the second is roundoff error $O(\varepsilon_{\mathrm{mach}}/\delta)$. The optimal balance gives $\delta^{\star} = \sqrt{2\varepsilon_{\mathrm{mach}}} \approx 1.5 \times 10^{-8}$ for double precision ($\varepsilon_{\mathrm{mach}} \approx 2.2 \times 10^{-16}$). The experiments test $\delta = 0.05$ (orders of magnitude above optimal, dominated by truncation error) and $\delta = 0.8$ (severely truncated).

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Uniform random direction}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The directional derivative estimate $\hat{g}_k$ is a \emph{noisy gradient estimator}: in expectation, $\mathbb{E}[\hat{g}_k] \approx \frac{1}{d}\nabla f(x_k)$ (up to $O(\delta^2)$ bias). Crucially, only \emph{two} function evaluations are required per iteration regardless of dimension $d$, compared to $d+1$ for the finite-difference gradient. This dimension-independence in cost makes random search attractive for large $d$ (e.g., hyperparameter search in machine learning), at the cost of increased variance in the gradient estimate.

\subsubsection{Finite Difference Error Analysis}

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively refines it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): move the worst vertex through the centroid of the remaining vertices: $x_r = c + \alpha_r(c - x_{\text{worst}})$.
\item \textbf{Expansion} ($\gamma = 2$): if reflection improved, try a further step: $x_e = c + \gamma(c - x_{\text{worst}})$.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection worsened, contract towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all non-best vertices towards the best.
\end{enumerate}
No gradient or Hessian information is required. Settings: initial simplex step $= 0.35$, 160 iterations.

\textbf{Dimensionality analysis:} Grid search with resolution $\Delta x$ requires $(L/\Delta x)^d$ evaluations in a domain of side length $L$. For $d = 100$ and $\Delta x = 0.1$ in $[0,1]^{100}$, this would require $10^{100}$ evaluations---astronomically beyond any computational budget. This \emph{curse of dimensionality} makes grid search inapplicable beyond $d \approx 4$--$5$ in practice.

Finite-difference gradient descent scales as $O(d)$ evaluations per iteration (one base evaluation plus one per coordinate): for $d = 2$ this is 3 evaluations/iteration, for $d = 10^4$ this becomes $10^4$ evaluations/iteration---still feasible, but expensive. The approximation error is $O(\delta)$ uniformly over all components when the Hessian is bounded.

Nesterov random search's $O(1)$ cost per iteration---only 2 function evaluations regardless of $d$---is the critical advantage for very high dimensions. However, the variance of the gradient estimator scales as $O(d)$: $\mathrm{Var}[\hat{g}_k] = \frac{1}{\delta^2}\mathrm{Var}[f(x+\delta u) - f(x)] \cdot \|u\|^2 \propto d \cdot \|\nabla f(x)\|^2$. Thus, while the cost is $O(1)$, the effective signal-to-noise ratio decreases as $O(1/\sqrt{d})$, requiring $O(d/\varepsilon^2)$ total iterations to achieve precision $\varepsilon$---the same total cost as finite-difference gradient descent but with much smaller memory requirements.

\subsection{Problem Formulation and KKT Conditions}

The constrained problem is:
\begin{equation}
\min_{x \in \mathbb{R}^2}\; f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1), \quad \text{subject to}\quad x_1 \geq 0.5.
\label{eq:q5_prob}
\end{equation}
Starting point: $x_0 = (0.2, 4.0)$, which is \emph{infeasible} since $x_1^0 = 0.2 < 0.5$.

The Karush--Kuhn--Tucker (KKT) optimality conditions for~\eqref{eq:q5_prob} require at the constrained optimum $x^{\star}$:
\begin{align}
\nabla f(x^{\star}) - \mu^{\star}\, e_1 &= 0 \quad (\text{stationarity, where } e_1 = [1, 0]^T) \\
x_1^{\star} &\geq 0.5 \quad (\text{primal feasibility}) \\
\mu^{\star} &\geq 0 \quad (\text{dual feasibility}) \\
\mu^{\star}(x_1^{\star} - 0.5) &= 0 \quad (\text{complementary slackness}).
\end{align}

Since the unconstrained minimum of $f$ lies at $x^{\star}_{\mathrm{unc}} \approx (0.5823, 2.0)$ with $x_1^{\star} \approx 0.5823 > 0.5$, the constraint is \emph{inactive} at the unconstrained minimiser. By complementary slackness, $\mu^{\star} = 0$, and the KKT conditions reduce to $\nabla f(x^{\star}) = 0$. Therefore, the constrained and unconstrained minima coincide: $x^{\star}_{\mathrm{con}} = x^{\star}_{\mathrm{unc}} \approx (0.5823, 2.0)$, $f^{\star} \approx 0.7244$.

This observation is important: the constraint $x_1 \geq 0.5$ is \emph{not binding} at the optimum, so the challenge lies entirely in handling the infeasible starting point and ensuring the iterates reach the feasible region.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = (\max(0.5,\, z_1),\, z_2)$ \Comment{Hard feasibility enforcement}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method (Linear Penalty)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $\nabla F_{\lambda}(x) \gets \nabla f(x) + \lambda \cdot \mathbf{1}_{[x_1 < 0.5]}(-e_1)$ \Comment{Subgradient}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$, as larger $\lambda$ increases the effective Lipschitz constant of $F_\lambda$ and requires a smaller step); unconstrained GD ($\alpha = 0.07$). All methods run for 100 iterations.

\textbf{Projected GD} enforces hard feasibility at every iteration: $x_k \in \mathcal{X}$ for all $k \geq 1$. The projection $\Pi_\mathcal{X}(z) = (\max(0.5, z_1), z_2)$ is trivial for this halfspace constraint.

\begin{proposition}[Projected GD Convergence]
For an $L$-smooth convex function $f$ and convex closed $\mathcal{X}$, projected GD with $\alpha = 1/L$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{L\|x_0 - x^{\star}\|^2}{2k}.
\end{equation}
\end{proposition}

The starting point $x_0 = (0.2, 4.0)$ has $\|x_0 - x^{\star}\|^2 = (0.2-0.582)^2 + (4.0-2.0)^2 \approx 0.146 + 4.0 = 4.146$. After projection at iteration 1, $x_1 = (0.5, \cdot)$ is feasible, and subsequent convergence is dictated by the gradient step within $\mathcal{X}$.

\textbf{Penalty methods} replace exact feasibility with a soft constraint term. As $\lambda \to \infty$, the penalised minimiser converges to the constrained minimiser, but large $\lambda$ stiffens the problem (increasing its effective Lipschitz constant to $L + \lambda$), requiring smaller step sizes. The penalty $\lambda \max(0, 0.5 - x_1)$ is a \emph{linear penalty}, which is exact (recovers the constrained solution at finite $\lambda$) when the KKT conditions hold. Here, since the constraint is inactive at the optimum, any $\lambda > 0$ suffices in theory, but a very small $\lambda$ converges slowly.

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm \hfill [30 marks]}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a bounded box).

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is:
\begin{equation}
x^{\star} = \arg\min_{x \in \mathcal{X}}\, (x_1 + 2x_2) = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
Since both components of $a$ are positive, each coordinate is independently minimised at its lower bound. This illustrates the \textbf{fundamental theorem of linear programming}: the optimum of a linear objective over a bounded polytope is always attained at a vertex (extreme point) of the feasible set. The optimum never lies in the interior of a facet unless the objective is parallel to that facet, in which case the entire facet is optimal.

\label{fig:q6_lp}
\end{figure}

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial feasible point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (remains in $\mathcal{X}$)}
\EndFor
\end{algorithmic}
\end{algorithm}

The Linear Minimisation Oracle (LMO) for the box $\mathcal{X}$ decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}, \quad l_i \leq x_i \leq u_i.
\end{equation}
This requires $O(d)$ operations per iteration. Since $x_{k+1} = \beta x_k + (1-\beta) z_k$ is a convex combination of $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$, the iterates remain feasible \emph{without projection}.

\textbf{The Frank--Wolfe gap} (duality gap) at iteration $k$ is:
\begin{equation}
\delta_k = \nabla f(x_k)^T (x_k - z_k) \geq 0.
\end{equation}
For convex $f$, $\delta_k \geq f(x_k) - f^{\star}$ provides a computable upper bound on the primal gap and serves as a natural stopping criterion: if $\delta_k < \varepsilon$, then $f(x_k) - f^{\star} < \varepsilon$.

\begin{theorem}[Frank--Wolfe Convergence]
Let $f$ be $L$-smooth and convex, and $\mathcal{X}$ bounded with diameter $D = \max_{x,y \in \mathcal{X}} \|x - y\|$. With step size $\gamma_k = 2/(k+2)$ (i.e., $\beta_k = k/(k+2)$):
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2} = O\!\left(\frac{1}{k}\right).
\end{equation}
The fixed-$\beta$ implementation used here does not achieve this optimal schedule but captures the qualitative convergence behaviour.
\end{theorem}

\subsection{Interior vs.\ Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits characteristic ``zig-zag'' behaviour: the LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ at each step, and the convex combination gradually averages these jumps towards the interior optimum. The theoretical $O(1/k)$ rate is significantly slower than projected gradient descent's $O(1/k)$ rate with the same constant, and far slower than Newton's $O(1/k^2)$ or better. However, Frank--Wolfe's advantage is the \emph{avoidance of any projection}: all that is needed is the LMO, which for polytopes is often more efficient than computing a general projection.

\textbf{Boundary optimum:} Frank--Wolfe is naturally suited to boundary optima because the LMO $z_k$ frequently selects the vertex closest to the constrained solution, driving iterates towards the boundary without explicit projection. The fixed $\beta = 0.93$ schedule provides a step size of $1 - \beta = 0.07$ towards the vertex per iteration. Under the optimal schedule $\gamma_k = 2/(k+2)$, convergence would be faster, especially in early iterations.

\begin{remark}[Affine Invariance of Frank--Wolfe]
Unlike gradient descent, the Frank--Wolfe algorithm is \emph{affine invariant}: its trajectory is unchanged under any invertible affine transformation of the domain. This is because the LMO $z_k = \arg\min_{x \in \mathcal{X}} \nabla f(x_k)^T x$ and the gradient $\nabla f(x_k)$ transform consistently. In contrast, gradient descent's convergence rate depends on the condition number $\kappa = L/\mu$ of the Hessian, which changes under affine transformations. This affine invariance makes Frank--Wolfe naturally suited to problems where the geometry of $\mathcal{X}$ (rather than the function's Hessian) determines the convergence behaviour.
\end{remark}

\textbf{Comparison with projected GD:} The Frank--Wolfe algorithm offers:
\begin{itemize}[nosep]
\item \textbf{Advantage}: No projection oracle needed; iterates are always in $\mathcal{X}$; sparse iterates (convex combinations of vertices) with potential memory efficiency.
\item \textbf{Disadvantage}: Slower $O(1/k)$ convergence (no acceleration possible without losing projection-free property); zig-zag behaviour near the optimum.
\end{itemize}

\subsection{Key Conclusions}

\textbf{1. Problem structure determines method suitability.} The Rosenbrock function ($\kappa \approx 2504$) exposes the limitations of first-order methods most starkly: GD requires a tiny step size ($\alpha = 0.0012$) mandated by stability, making progress negligible. Nesterov's accelerated method, by building momentum along the valley floor, achieves a 7$\times$ improvement over GD. Newton's method, armed with curvature information, navigates the ill-conditioned valley efficiently, but requires damping because the Hessian is non-positive-definite away from the optimum.

\textbf{2. Knowledge of $f^{\star}$ is critical for Polyak.} The Polyak step achieves the best convergence when $f^{\star}$ is known (Rosenbrock), but diverges catastrophically under any misspecification. This sensitivity limits Polyak's applicability to problems with known optimal values or valid lower bounds.

\textbf{3. Adaptive methods (Adagrad, RMSprop, Adam) provide robustness.} These methods automatically scale steps by gradient history, making them relatively insensitive to the step-size choice. Adam, combining momentum and per-coordinate scaling, is the most robust and is the de facto standard in deep learning---though Nesterov outperforms it on strongly anisotropic problems like Rosenbrock.

\textbf{4. Newton's method is unmatched in iteration count but costly per iteration.} One Newton step solves the quadratic Benchmark A exactly. For non-quadratic problems, damping is required, but even 20 damped Newton steps dramatically outperform 80 GD steps on Rosenbrock. The $O(d^3)$ per-iteration cost limits use to $d \lesssim 10^4$; for large-scale problems, Quasi-Newton methods (L-BFGS) provide a practical alternative.

\textbf{5. Derivative-free methods trade accuracy for generality.} Finite-difference gradients closely approximate exact gradients when $\delta$ is well-chosen, but require $O(d)$ evaluations per step. Nesterov random search requires only 2 evaluations per step at the cost of high gradient variance. Grid search is the only method guaranteeing global optimality (to grid resolution), but is exponentially intractable beyond $d \approx 4$--$5$.

\textbf{6. Constraint handling requires architectural choices.} Projected gradient descent enforces hard feasibility every iteration at the cost of a projection oracle. Penalty methods are simpler but require careful tuning of $\lambda$ and cannot guarantee strict feasibility. Frank--Wolfe avoids projection entirely by solving a linear subproblem at each step, inheriting the structure of the feasible set.

\textbf{7. Algorithm selection in practice.}

\begin{table}[H]
\centering
\caption{Practical algorithm selection guide based on problem characteristics.}
\label{tab:selection}
\begin{tabular}{lll}
\toprule
\textbf{Problem type} & \textbf{Recommended method} & \textbf{Reason} \\
\midrule

The overarching principle is to exploit all available structure: second-order information (Newton), problem geometry (Frank--Wolfe), or constraint simplicity (projection) can each yield order-of-magnitude improvements over naive gradient descent. The benchmarks in this report illustrate how dramatically algorithm choice affects convergence, particularly under high condition numbers ($\kappa \approx 2504$ for Rosenbrock) where first-order methods with fixed step sizes face fundamental speed limits.

\begin{lstlisting}[caption={Gradient Descent}]
def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex}]
def nelder_mead(loss_fn, x0, step, n_iters, alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1])
        f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1])
            f_e = loss_fn(x_e)
            if f_e < f_r: simplex[-1] = x_e; f_vals[-1] = f_e
            else: simplex[-1] = x_r; f_vals[-1] = f_r
        else:
            if f_r < f_vals[-1]:
                x_c = centroid + rho * (x_r - centroid)
                f_c = loss_fn(x_c)
                if f_c <= f_r: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
            else:
                x_c = centroid + rho * (simplex[-1] - centroid)
                f_c = loss_fn(x_c)
                if f_c < f_vals[-1]: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
