\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark functions: a linear regression quadratic loss (Benchmark A), a toy neural network loss with sinusoidal perturbation (Benchmark B), and the Rosenbrock function (Benchmark C). Six categories of methods are systematically investigated across six questions: adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball); momentum-based accelerated methods (Nesterov, Adam, mini-batch SGD); Newton's method with second-order local approximations; derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead simplex, grid search); constrained optimisation via projected gradient descent and penalty methods; and the Frank--Wolfe conditional gradient algorithm for linearly constrained problems. Each method is evaluated with respect to convergence speed, trajectory behaviour, numerical stability, and theoretical convergence guarantees. Key findings include: (i) the Polyak step size diverges when $f^{\star}$ is misspecified but achieves the best Rosenbrock result ($f = 0.0014$ after 200 iterations) when $f^{\star} = 0$ is exact; (ii) Nesterov acceleration outperforms Adam on the ill-conditioned Rosenbrock function due to its $O(1/k^2)$ convergence guarantee; (iii) Newton's method solves the quadratic Benchmark A in one iteration; (iv) finite differences with $\delta = 0.05$ closely track exact gradients while $\delta = 0.8$ introduces substantial truncation error; (v) projected gradient descent enforces hard feasibility from the first iteration, in contrast to penalty methods which require tuning of the penalty weight $\lambda$.
\end{abstract}

\section{Introduction}
\label{sec:intro}
% ============================================================

Optimisation is the mathematical backbone of modern machine learning and data analysis. The central task is to find a parameter vector $x^{\star}$ minimising an objective function $f: \mathbb{R}^d \to \mathbb{R}$, possibly subject to constraints. The choice of algorithm significantly affects convergence speed, accuracy, and computational cost. Different problem structures --- convexity, smoothness, conditioning, availability of gradients, and presence of constraints --- favour different methods.

This report investigates six families of algorithms over three carefully chosen benchmark functions designed to expose complementary algorithmic weaknesses. \textbf{Benchmark A} (linear regression) is a convex quadratic with exact closed-form solution, making it ideal for demonstrating single-step Newton convergence and the noise floor imposed by observation error. \textbf{Benchmark B} (toy neural network loss) is a smooth non-quadratic function with moderate anisotropy (Hessian eigenvalue ratio approximately 3:1 to 10:1 depending on position), suitable for comparing adaptive scaling methods. \textbf{Benchmark C} (Rosenbrock) is a non-convex function with condition number $\kappa \approx 2000$ at the minimum, specifically designed to be challenging for gradient-based methods.

All experiments are implemented in Python with \texttt{NumPy}, using a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The complete code is provided in \texttt{CS7DS2\_Final\_Code\_25336453.py} and reproduced in Appendix~\ref{app:code}.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strongly convex quadratic with Hessian $H = \frac{1}{m}X^T X$ (constant, positive definite). The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Because of observation noise $\varepsilon$, the minimum loss satisfies $J^{\star} \approx 0.4826$ rather than zero; the empirical minimiser $\hat{\theta}_{\text{OLS}} = (X^TX)^{-1}X^Ty$ converges to $\theta^{\star}$ as $m \to \infty$. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
\label{eq:bench_B}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient and Hessian are:
\begin{equation}
\nabla f = \begin{bmatrix} 2(x_1 - 1) + \cos(x_1) \\ 10(x_2 - 2) \end{bmatrix}, \qquad
H(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The diagonal Hessian has eigenvalues $\lambda_1 = 2 - \sin(x_1) \in [1, 3]$ and $\lambda_2 = 10$, giving a local condition number ranging from $\kappa \approx 10/3$ to $\kappa \approx 10$. The sinusoidal term introduces mild non-quadratic curvature. The unique minimum is at $x^{\star} \approx (0.582, 2.0)$ (solving $2(x_1 - 1) + \cos(x_1) = 0$) with $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
\label{eq:rosenbrock}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the minimum is:
\begin{equation}
H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}, \qquad \lambda_{\min} \approx 0.5, \quad \lambda_{\max} \approx 1001,
\end{equation}
giving condition number $\kappa \approx 2002$. This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock the definitive stress-test for optimisation algorithms. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems (Questions 5--6): $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$, a box constraint set.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is theoretically motivated: for any convex $f$, setting $\alpha_k = (f(x_k) - f^{\star}) / \|\nabla f(x_k)\|^2$ is the largest step that can be taken without increasing the distance to any optimal point $x^{\star}$. The $\varepsilon$ term prevents division by zero.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad applies a larger effective rate to coordinates with consistently small gradients and a smaller rate to those with large gradients. The cumulative sum $G_k = \sum_{t=0}^{k-1} g_t \odot g_t$ grows monotonically, causing the effective step $\alpha_0 / \sqrt{G_k}$ to decay and eventually vanish --- a key limitation on non-stationary problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay factor $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop (Hinton, 2012) addresses Adagrad's vanishing step issue by replacing the cumulative sum with an exponential moving average. The decay factor $\beta = 0.9$ (used throughout) means only the most recent gradient magnitudes dominate the scaling, maintaining a non-vanishing effective step size.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method adds a momentum term $\beta z_k$ to the gradient update. For strongly convex quadratic objectives with condition number $\kappa = L/\mu$, the optimal hyperparameter choices are:
\begin{equation}
\alpha^{\star} = \frac{4}{(\sqrt{L} + \sqrt{\mu})^2}, \qquad \beta^{\star} = \left(\frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1}\right)^2,
\end{equation}
yielding linear convergence with rate $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1) < (\kappa-1)/(\kappa+1)$ (GD's rate). For $\kappa \gg 1$, the Heavy Ball rate is $\approx 1 - 2/\sqrt{\kappa}$ vs.\ GD's $\approx 1 - 2/\kappa$, a factor of $\sqrt{\kappa}$ improvement. However, Heavy Ball's optimal convergence guarantee applies only to quadratics; for general non-quadratic functions, convergence guarantees are weaker and oscillation can occur.

\begin{theorem}[Polyak Step-Size Convergence]
For a convex function $f$ with exact $f^{\star}$, the Polyak step-size iteration satisfies
\[
\min_{0 \le k \le K} \bigl(f(x_k) - f^{\star}\bigr) \;=\; O(1/K).
\]
Moreover, if $f$ is $\mu$-strongly convex and $f^{\star}$ is known exactly, geometric convergence holds: $f(x_k) - f^{\star} \le (1 - \mu/L)^k (f(x_0) - f^{\star})$.
\end{theorem}

\begin{remark}[Polyak Sensitivity to $f^{\star}$]
The Polyak step size is exquisitely sensitive to the accuracy of $f^{\star}$. The step formula $\alpha_k = (f(x_k) - f^{\star}) / (\|\nabla f(x_k)\|^2 + \varepsilon)$ requires $f^{\star} \leq f(x_k)$ for all $k$ (otherwise a negative step is taken towards ascent). When $f^{\star} = 0 < J^{\star}_{\text{true}} \approx 0.4826$ (Benchmarks A and B), the numerator remains bounded away from zero while the denominator vanishes as the gradient shrinks, causing $\alpha_k \to \infty$. On Benchmark C where $f^{\star} = 0$ is exact, the numerator $f(x_k) - 0$ and denominator $\|\nabla f\|^2$ both shrink, yielding a well-behaved decaying step.
\end{remark}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly (Benchmark C), Polyak converges fastest. On Benchmarks A and B (where $f^{\star}$ is misspecified), Adagrad and Heavy Ball match GD baseline performance without the instability risk.

\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging of squared gradients, which filters out gradient noise. Heavy Ball shows characteristic overshoot followed by damped oscillation, consistent with a second-order dynamical system.

\item \textbf{Stability:} GD baseline is unconditionally stable for $\alpha < 2/L$ (Lipschitz smoothness constant). Adagrad is stable by design since its effective step decreases monotonically. Polyak is \emph{conditionally stable} only when $f^{\star}$ is accurately known; misspecification is catastrophic.

\item \textbf{Practical recommendation:} In practice, exact $f^{\star}$ is rarely known, limiting Polyak's applicability to problems where a tight lower bound on the optimal value can be established (e.g., non-negative losses where a data-dependent lower bound is available). Adagrad and RMSprop are more robust choices for general problems.

\item \textbf{Effect of condition number:} Table~\ref{tab:smoothness} shows that Benchmark A's near-unit conditioning ($\kappa \approx 1.12$) makes it trivially easy for all methods. Benchmark B's $\kappa \approx 10$ is moderate. Benchmark C's $\kappa \approx 1604$ at the starting point is the key driver of difficulty: GD needs a tiny step $\alpha < 0.0025$, requiring $\sim 1600$ steps to achieve the same relative error as one Newton step.

\item \textbf{Memory cost:} GD and Polyak: $O(d)$. Adagrad and RMSprop: $O(d)$ for the accumulated squared gradients. Heavy Ball: $O(d)$ for the momentum buffer. All methods are memory-efficient for large $d$.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, maximum momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Acceleration]
For a convex, $L$-smooth function $f$ and the Nesterov scheme with $\alpha = 1/L$ and the schedule $\beta_k = (k-1)/(k+2)$,
\[
f(x_k) - f^{\star} \;\leq\; \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} \;=\; O(1/k^2).
\]
This is \emph{optimal} for first-order methods on convex smooth problems (Nesterov, 1983).
\end{theorem}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This allows the algorithm to ``correct'' its trajectory before overshooting. Conceptually, Nesterov evaluates the gradient one momentum step ahead, combining the momentum direction with the local gradient information at that look-ahead point.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (mean) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (variance) estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam (Kingma \& Ba, 2015) combines momentum (via $m_t$) with per-coordinate adaptive scaling (via $v_t$), inheriting advantages of both RMSprop and momentum methods. The bias correction terms $(1 - \beta_1^t)^{-1}$ and $(1 - \beta_2^t)^{-1}$ are essential in early iterations: without them, the exponential averages are initialised at zero and systematically underestimate the true gradient moments.

\begin{remark}[FISTA: Nesterov for Composite Problems]
Nesterov acceleration extends naturally to composite objectives $\min_x\{f(x) + g(x)\}$ where $f$ is smooth convex and $g$ is convex but possibly non-smooth (e.g.\ $g = \lambda\|x\|_1$ for LASSO). FISTA (Beck \& Teboulle, 2009) replaces the gradient step with a proximal step and retains Nesterov's momentum schedule to achieve $O(1/k^2)$ convergence. This connection demonstrates that the acceleration principle is not limited to smooth objectives.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD is applied to Benchmark A using learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at each epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y) \in \mathbb{R}^{m \times d} \times \mathbb{R}^m$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute indices $\{1, \ldots, m\}$ into $\pi$
    \For{each mini-batch $\mathcal{B} \subset \pi$ of size $b$}
        \State $g_{\mathcal{B}} \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, g_{\mathcal{B}}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

With $m = 1000$ samples, batch size $b = 5$ gives $1000/5 = 200$ updates per epoch, while $b = 40$ gives $1000/40 = 25$ updates per epoch. The stochastic gradient $g_{\mathcal{B}}$ is an unbiased estimate of the full gradient $\nabla J(\theta)$, with variance $\mathbb{V}[g_{\mathcal{B}}] = O(1/b)$.

\includegraphics[width=0.85\textwidth]{figures/q2_sgd_batch.pdf}

\begin{remark}[SGD Convergence Rates]
For a $\mu$-strongly convex, $L$-smooth objective and constant step size $\alpha \leq 1/(2L)$:
\begin{itemize}[nosep]
\item \textbf{Constant $\alpha$}: $\mathbb{E}[f(\theta_k)] - f^{\star} \leq (1-\mu\alpha)^k (f(\theta_0) - f^{\star}) + \alpha \sigma^2 / (2\mu)$, where $\sigma^2$ is the gradient variance. Geometric decay to a noise-floor of $O(\alpha \sigma^2 / \mu)$, not zero.
\item \textbf{Decaying $\alpha_k = a/(b+k)$}: Achieves $\mathbb{E}[f(\theta_k)] - f^{\star} = O(1/k)$ convergence to the true optimum. The standard Polyak--Ruppert averaging scheme further achieves $O(1/\sqrt{k})$ for non-convex objectives.
\end{itemize}
In the experiments, a fixed $\alpha = 0.06$ is used; the observed noise floor at $J^{\star} \approx 0.4826$ is the irreducible observation noise, not the SGD stationarity error.
\end{remark}

\begin{remark}[Bias-Variance Trade-off in SGD]
The batch size controls a bias-variance trade-off in the gradient estimate. For a fixed noise model with variance $\sigma^2$, the variance of the mini-batch gradient estimate scales as $\sigma^2 / b$. Smaller $b$ reduces computation per update but increases gradient noise, slowing convergence near the minimum. Optimal batch size in practice balances hardware parallelism, memory, and this trade-off; in deep learning, batch sizes of 32--256 are common.
\end{remark}

\subsection{Local Approximations of $g(x) = x^4$}

Consider $g(x) = x^4$, expanded around $x_0 = 0.25$.

\textbf{First-order (linear/tangent) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.0039 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton step minimises the second-order model: setting $\partial/\partial x [g(x_0) + g'(x_0)(x-x_0) + \tfrac{1}{2}g''(x_0)(x-x_0)^2] = 0$ gives
\[
x_1 = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\]
Iterating: $g'(0.1667) = 4 \times 0.1667^3 = 0.0185$, $g''(0.1667) = 12 \times 0.1667^2 = 0.333$, so $x_2 = 0.1667 - 0.0185/0.333 = 0.1111$. Newton's method converges to $x^{\star} = 0$ quadratically for $g(x) = x^4$ (though $g''(0) = 0$ makes the rate super-linear rather than strictly quadratic at the limit).

The second-order approximation (parabola) captures local curvature and fits well in a neighbourhood of $x_0$. The Newton step steps to the minimum of the parabola, providing a far better estimate than a gradient step alone.}

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0, 1]$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$.

\begin{theorem}[Newton's Method for Quadratics]
\label{thm:newton_quadratic}
For a strongly convex quadratic $f(\theta) = \tfrac{1}{2}\theta^T H \theta - b^T\theta$, Newton's method with $\alpha = 1$ converges in exactly \emph{one iteration}: $\theta_1 = \theta_0 - H^{-1}(\nabla f(\theta_0)) = H^{-1}b = \theta^{\star}$.
\end{theorem}

\begin{theorem}[Quadratic Convergence of Newton's Method]
Let $f$ be twice continuously differentiable with $L$-Lipschitz Hessian, $\mu$-strongly convex. Then, in a sufficiently small neighbourhood of $x^{\star}$, Newton's method with $\alpha = 1$ satisfies:
\[
\|x_{k+1} - x^{\star}\| \;\leq\; \frac{L}{2\mu}\|x_k - x^{\star}\|^2.
\]
This is \emph{quadratic} (super-linear) convergence: the number of correct digits roughly doubles each iteration.
\end{theorem}

\begin{remark}[Quasi-Newton Methods as a Practical Alternative]
Computing the exact Hessian costs $O(d^2)$ per iteration and requires solving a $d \times d$ linear system ($O(d^3)$), making Newton infeasible for large $d$. \textbf{Quasi-Newton methods} (e.g., BFGS) maintain an approximate inverse Hessian $B_k^{-1}$ updated via rank-1 or rank-2 updates from gradient differences:
\[
B_{k+1} \gets B_k + \frac{y_k y_k^T}{y_k^T s_k} - \frac{B_k s_k s_k^T B_k}{s_k^T B_k s_k}, \quad s_k = x_{k+1} - x_k,\ y_k = \nabla f(x_{k+1}) - \nabla f(x_k).
\]
BFGS achieves super-linear convergence (between linear and quadratic) at $O(d^2)$ cost per iteration, making it the method of choice for medium-scale unconstrained optimisation ($d \lesssim 10^4$).
\end{remark}

\begin{remark}[Why Damping is Required on Rosenbrock]
The Rosenbrock Hessian at $(x_1, x_2)$ is
\[
H = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\]
At $x_0 = (-1, 1)$: $H_{11} = 2 + 1200 - 400 = 802$, $H_{12} = 400$, $H_{22} = 200$. The full Newton step $p = H^{-1}\nabla f$ may be large and point far from the quadratic approximation's validity region. A damping factor $\alpha = 0.22$ trades convergence speed for stability, ensuring each step stays within the region where the quadratic model is accurate.
\end{remark}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
The total approximation error has two competing components:
\begin{itemize}[nosep]
\item \textbf{Truncation error}: $O(\delta)$ from the Taylor remainder $\tfrac{1}{2}\delta f''(\xi)$ for some $\xi \in [x, x+\delta e_i]$.
\item \textbf{Round-off error}: $O(\varepsilon_{\text{mach}}/\delta)$ from floating-point cancellation when computing $f(x + \delta e_i) - f(x)$.
\end{itemize}
Balancing these gives the optimal step size $\delta^{\star} = \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$. For central differences, $[\partial f / \partial x_i](x) \approx [f(x + \delta e_i) - f(x - \delta e_i)]/(2\delta)$, the truncation error improves to $O(\delta^2)$ at the cost of one additional function evaluation per coordinate.

\begin{figure}[H]
\centering
\includegraphics[width=0.88\textwidth]{figures/q4_fd_error.pdf}
\caption{Q4: Finite difference gradient approximation error vs.\ step size $\delta$ for Benchmark B evaluated at $x = (0.5, 3.0)$. Forward differences (blue) achieve minimum error $\approx 10^{-8}$ near $\delta \sim 10^{-8}$, consistent with the optimal $\delta^{\star} = \sqrt{\varepsilon_{\text{mach}}}$. For large $\delta$, truncation error dominates (slope +1 on log--log scale). Central differences (red) achieve smaller truncation error (slope +2) and a lower error floor. The assignment's choice $\delta = 0.05$ is clearly in the truncation-dominated regime with manageable error, while $\delta = 0.8$ incurs large truncation error.}
\label{fig:q4_fd_error}
\end{figure}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov (Gaussian Smoothing) Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, smoothing radius $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta\, u_k) - f(x_k)}{\delta}\, u_k$
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The estimator $\hat{g}_k$ approximates the directional derivative along $u_k$, yielding an unbiased estimate of a smoothed gradient: $\mathbb{E}[\hat{g}_k] = \nabla f_{\delta}(x_k)$ where $f_{\delta}(x) = \mathbb{E}_{u}[f(x + \delta u)]$. Only \emph{two} function evaluations are required per iteration, regardless of dimension $d$ --- a key advantage in high-dimensional settings where finite differences cost $O(d)$ evaluations.

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ (since $d = 2$) and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): move the worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2$): if the reflected point improves on the best, try extending further.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection fails, contract the worst vertex towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all vertices towards the best.
\end{enumerate}
No gradient information is required; the simplex geometry adapts to the local landscape by expanding where progress is made and contracting where it is not.

Settings: initial simplex step $= 0.35$, $\alpha_r = 1.0$ (reflection), $\gamma = 2.0$ (expansion), $\rho = 0.5$ (contraction), $\sigma = 0.5$ (shrink), iterations $= 160$.

\begin{remark}[Convergence Properties of Nelder--Mead]
Nelder--Mead lacks a general convergence guarantee: McKinnon (1998) constructed a 2D example where the method converges to a non-optimal point. In practice, it often performs well for $d \leq 5$ but its per-iteration cost grows as $O(d)$ function evaluations in the worst case (shrink step), and the simplex can degenerate (become flat). For Rosenbrock in $d = 2$, the method adapts well to the banana-shaped valley, producing convergence in 160 iterations to a competitive objective value compared to gradient-based methods with only 120 iterations.
\end{remark}

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}

\subsection{Problem Formulation}

Minimise $f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)$ (Benchmark B) subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$.

In standard form: $\min_{x \in \mathbb{R}^2} f(x)$ s.t.\ $h(x) := -x_1 + 0.5 \leq 0$.

\subsection{KKT Optimality Conditions}

\begin{definition}[KKT Conditions]
For the problem $\min f(x)$ s.t.\ $h(x) \leq 0$, a point $x^{\star}$ is KKT-stationary if there exists a multiplier $\mu^{\star} \geq 0$ such that:
\begin{align}
\nabla f(x^{\star}) + \mu^{\star} \nabla h(x^{\star}) &= 0 \quad (\text{stationarity}), \label{eq:kkt_stat}\\
h(x^{\star}) &\leq 0 \quad (\text{primal feasibility}), \label{eq:kkt_pf}\\
\mu^{\star} &\geq 0 \quad (\text{dual feasibility}), \label{eq:kkt_df}\\
\mu^{\star} h(x^{\star}) &= 0 \quad (\text{complementary slackness}). \label{eq:kkt_cs}
\end{align}
Under a constraint qualification (e.g., LICQ), KKT conditions are \emph{necessary} for local optimality; for convex $f$ and convex $h$, they are also \emph{sufficient}.
\end{definition}

\begin{proposition}[Constrained Optimum Coincides with Unconstrained Optimum]
\label{prop:inactive_constraint}
The unconstrained minimum of Benchmark B is at $x^{\star} = (0.582, 2.0)$, satisfying $x_1^{\star} = 0.582 > 0.5$. Therefore, the constraint $x_1 \geq 0.5$ is \emph{inactive} at $x^{\star}$.
\end{proposition}

\begin{proof}
At $x^{\star}$, complementary slackness \eqref{eq:kkt_cs} gives $\mu^{\star}(-x_1^{\star} + 0.5) = \mu^{\star}(0.5 - 0.582) = -0.082\mu^{\star} = 0$, forcing $\mu^{\star} = 0$. The stationarity condition \eqref{eq:kkt_stat} then reduces to $\nabla f(x^{\star}) = 0$, the unconstrained first-order condition, which is satisfied by definition of $x^{\star}$.
\end{proof}

Proposition~\ref{prop:inactive_constraint} implies that both projected GD and penalty methods should converge to the same point $(0.582, 2.0)$ given sufficient iterations, since the constraint does not restrict the unconstrained optimum.

\subsection{Algorithms}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection operator $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = \bigl(\max(0.5, z_{k+1,1}),\ z_{k+1,2}\bigr)$ \Comment{Project onto $\{x_1 \geq 0.5\}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Exact Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, -x_1 + 0.5)$ \Comment{Penalised objective}
    \State Compute $\nabla F_{\lambda}(x_k)$: gradient of $f$ plus $-\lambda e_1 \cdot \mathbf{1}[x_{1,k} < 0.5]$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: projected GD ($\alpha = 0.08$); penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$); unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\begin{remark}[Stability Constraint for Penalty Methods]
The penalised objective $F_{\lambda}(x) = f(x) + \lambda \max(0, -x_1 + 0.5)$ has Lipschitz gradient constant $L_{\lambda} = L_f + \lambda$ in the infeasible region (where the penalty is linear in $x_1$). Gradient descent stability requires $\alpha < 2/L_{\lambda}$. Increasing $\lambda$ from 1.8 to 4.5 (a factor of 2.5$\times$) requires reducing the step size by a comparable factor: $\alpha = 0.05 \to 0.03$. This inverse relationship between $\lambda$ and stable step size is a fundamental limitation of pure penalty methods, motivating augmented Lagrangian approaches where a smaller $\lambda$ suffices.
\end{remark}

\textbf{Projected GD} enforces feasibility as a hard constraint at every iteration by applying the projection operator after each gradient step. \subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box / bounded polytope).

\textbf{Linear Programme (LP):} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$.

\begin{proposition}[LP Solution]
Since both components of $a = [1, 2]^T$ are positive, the objective $f(x) = x_1 + 2x_2$ is minimised by setting each $x_i$ to its lower bound:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
\end{proposition}

This exemplifies the \textbf{Fundamental Theorem of Linear Programming}: the optimum of a linear objective over a bounded polytope is always attained at a vertex. The vertices of $\mathcal{X}$ are its four corners: $(0.5, -5)$, $(5, -5)$, $(0.5, 10)$, $(5, 10)$.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (projection-free)}
\EndFor
\end{algorithmic}
\end{algorithm}

The \textbf{linear minimisation oracle} (LMO) solves a linear programme at each iteration. For box constraints, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations. Crucially, Frank--Wolfe \emph{never projects} onto $\mathcal{X}$; iterates are always convex combinations of feasible points and are therefore automatically feasible.

\begin{definition}[Frank--Wolfe Gap]
The \emph{Frank--Wolfe duality gap} at $x_k$ is:
\begin{equation}
g_{\text{FW}}(x_k) = \max_{y \in \mathcal{X}}\, \nabla f(x_k)^T(x_k - y) = \nabla f(x_k)^T(x_k - z_k).
\end{equation}
It satisfies $g_{\text{FW}}(x_k) \geq f(x_k) - f^{\star} \geq 0$, providing a computable upper bound on the sub-optimality gap. $g_{\text{FW}}(x^{\star}) = 0$ iff $x^{\star}$ is a constrained optimum.
\end{definition}

\begin{theorem}[Frank--Wolfe Convergence Rate]
For a convex, $L$-smooth objective $f$ and the standard step-size $\gamma_k = 2/(k+2)$, Frank--Wolfe satisfies:
\[
f(x_k) - f^{\star} \leq \frac{2LC_{\mathcal{X}}^2}{k+2} = O(1/k),
\]
where $C_{\mathcal{X}} = \max_{x, y \in \mathcal{X}} \|x - y\|$ is the diameter of the feasible set. The fixed-$\beta$ variant used in the experiments is a special case; its convergence rate depends on the choice of $\beta$.
\end{theorem}

The $O(1/k)$ rate of Frank--Wolfe is identical to subgradient methods for convex objectives but inferior to projected gradient descent's $O(e^{-ck})$ for strongly convex problems. Frank--Wolfe's distinctive advantages are:
\begin{itemize}[nosep]
\item \textbf{Projection-free:} Projecting onto nuclear norm balls ($\|X\|_* \leq r$) requires a full SVD ($O(n^3)$), while the LMO only needs a rank-1 SVD ($O(n^2)$). In trace-norm regularisation (matrix completion), this provides a factor-$n$ speedup.
\item \textbf{Sparse iterates:} Each $z_k$ is a vertex of $\mathcal{X}$ (an extreme point); the iterate $x_k = \sum_{t} \gamma_t z_t$ is a convex combination of at most $k$ vertices. This yields \emph{sparse solutions} automatically, valued in compressed sensing and structured prediction.
\item \textbf{Natural for polytopes:} Many combinatorial constraint sets (flow polytopes, spanning tree polytopes, permutation matrices) have efficient LMOs but no closed-form projection.
\end{itemize}

\textbf{Boundary optimum:} Frank--Wolfe handles boundary optima naturally. This is a significant practical advantage over methods that require knowing the optimal value (like Polyak).

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has systematically investigated six families of optimisation algorithms across three benchmark problems of increasing difficulty. The key conclusions are:

\begin{enumerate}[nosep]
\item \textbf{Adaptive step-size methods (Q1):} The Polyak step size achieves optimal convergence when $f^{\star}$ is known exactly (Benchmark C: $f \to 0.001$ at 200 iterations) but diverges catastrophically when $f^{\star}$ is misspecified (Benchmarks A, B). Adagrad and Heavy Ball provide robust alternatives without requiring knowledge of $f^{\star}$.

\item \textbf{Momentum and stochastic methods (Q2):} Nesterov acceleration achieves $O(1/k^2)$ convergence, substantially outperforming both GD ($O(1/k)$ for convex) and Adam on the ill-conditioned Rosenbrock function. Mini-batch SGD exhibits a batch-size trade-off: smaller batches provide more frequent updates but higher gradient variance, while larger batches provide smoother convergence at lower per-epoch progress.

\item \textbf{Newton's method (Q3):} For quadratic objectives (Benchmark A), Newton converges in exactly one step. For near-quadratic objectives (Benchmark B), convergence is achieved in $\sim$5 steps. On the ill-conditioned Rosenbrock, damping ($\alpha = 0.22$) is necessary for stability, but even damped Newton outperforms GD by a factor of $\sim$5 in objective value for the same iteration count. The quadratic convergence rate $\|x_{k+1} - x^{\star}\| = O(\|x_k - x^{\star}\|^2)$ near the optimum is confirmed empirically.

\item \textbf{Derivative-free methods (Q4):} Finite differences with $\delta = 0.05$ closely approximate the exact gradient (error $\approx 0.22$), confirming that $\delta$ is in the truncation-dominated regime. The larger $\delta = 0.8$ incurs substantial truncation error (error $\approx 3.88$). Nesterov random search, while dimension-efficient (2 evaluations per step), converges slowly due to its stochastic directional gradient. The $O(\delta)$ vs.\ $O(\delta^2)$ trade-off between forward and central differences is confirmed in Figure~\ref{fig:q4_fd_error}.

\item \textbf{Constrained optimisation (Q5):} Since the unconstrained minimum $(0.582, 2.0)$ is interior to the feasible region $\{x_1 \geq 0.5\}$, KKT conditions confirm that $\mu^{\star} = 0$ and the constrained and unconstrained optima coincide. Projected GD enforces feasibility exactly from iteration 1 via the $\max$ projection. Penalty methods achieve approximate feasibility, with convergence quality improving with $\lambda$.

\item \textbf{Frank--Wolfe and linear programmes (Q6):} The LP $\min x_1 + 2x_2$ over a box has its optimum at the vertex $(0.5, -5)$, confirming the Fundamental Theorem of LP. Frank--Wolfe converges at $O(1/k)$ to both interior and boundary constrained optima without requiring a projection oracle, making it attractive for complex constraint sets where the LMO is tractable.
\end{enumerate}

\textbf{Overall recommendation:} For well-conditioned problems ($\kappa$ small) with known $f^{\star}$, Polyak is optimal. For general smooth convex problems without known $f^{\star}$, Nesterov acceleration provides the theoretically optimal $O(1/k^2)$ rate with no additional hyperparameter knowledge required beyond the Lipschitz constant $L$. When second derivatives are available and the dimension is small ($d \lesssim 100$), Newton provides quadratic convergence; for $d \gtrsim 1000$, quasi-Newton (BFGS) is preferable. Mini-batch SGD with a carefully tuned batch size and decaying learning rate is the method of choice for large-scale machine learning. Derivative-free methods (Nelder--Mead for $d \leq 5$; Nesterov random search for $d \gg 1$) should be reserved for black-box or simulation-based objectives where gradient computation is infeasible.

\paragraph{Connection to Course Theory.} The experiments directly validate the key theoretical results from the course lectures:
\begin{itemize}[nosep]
\item Gradient descent convergence rate $O((1-\mu/L)^k)$ confirmed on Benchmark A ($\kappa \approx 1.12$, rapid convergence).
\item Nesterov's $O(1/k^2)$ advantage over GD's $O(1/k)$ confirmed on Benchmarks B and C.
\item Newton's exact one-step solution for quadratics (Theorem~\ref{thm:newton_quadratic}) confirmed on Benchmark A.
\item The Polyak step size's dependence on $f^{\star}$ accuracy confirmed by the divergence on Benchmarks A/B and rapid convergence on C.
\item KKT conditions confirming inactive constraint at Benchmark B's optimum (Proposition~\ref{prop:inactive_constraint}).
\item Frank--Wolfe's $O(1/k)$ rate and projection-free feasibility confirmed in Q6.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex}]
def nelder_mead(loss_fn, x0, step, n_iters, alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1])
        f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1])
            f_e = loss_fn(x_e)
            if f_e < f_r: simplex[-1] = x_e; f_vals[-1] = f_e
            else: simplex[-1] = x_r; f_vals[-1] = f_r
        else:
            if f_r < f_vals[-1]:
                x_c = centroid + rho * (x_r - centroid)
                f_c = loss_fn(x_c)
                if f_c <= f_r: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
            else:
                x_c = centroid + rho * (simplex[-1] - centroid)
                f_c = loss_fn(x_c)
                if f_c < f_vals[-1]: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
