\begin{abstract}
This report presents a comprehensive comparative study of six categories of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C). The methods investigated are: (Q1)~adaptive step-size methods---Polyak step, Adagrad, RMSprop, and Heavy Ball; (Q2)~momentum and stochastic methods---Nesterov acceleration, Adam, and mini-batch SGD; (Q3)~Newton's method with local second-order approximations; (Q4)~derivative-free optimisation---finite differences, Nesterov random search, Nelder--Mead, and grid search; (Q5)~constrained optimisation via projected gradient descent and penalty methods; and (Q6)~the Frank--Wolfe conditional gradient algorithm for box-constrained problems. Key findings include: the Polyak step diverges when $f^\star$ is misspecified but converges fastest when $f^\star = 0$ is exact (Rosenbrock); Nesterov acceleration achieves its theoretical $O(1/k^2)$ rate advantage on smooth problems; Newton's method solves the quadratic problem exactly in one iteration and achieves near-quadratic convergence on nonlinear benchmarks; derivative-free methods trade accuracy for gradient-free evaluation at $O(N^d)$ cost in grid search; projected gradient descent enforces feasibility as a hard constraint while penalty methods offer a soft approximation; and Frank--Wolfe replaces the projection oracle with a cheaper linear minimisation oracle at the cost of slower $O(1/k)$ convergence.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2}\theta^T H \theta - b^T \theta + c,
\end{equation}
where $H = \frac{1}{m}X^T X \in \mathbb{R}^{2\times 2}$, $b = \frac{1}{m}X^T y$, and $c = \frac{1}{2m}\|y\|^2$. The gradient is $\nabla J(\theta) = H\theta - b = \frac{1}{m}X^T(X\theta - y)$. This function is $\mu$-strongly convex and $L$-smooth with constants $\mu = \lambda_{\min}(H)$ and $L = \lambda_{\max}(H)$.

Numerically, $\mu \approx 0.474$, $L \approx 1.524$, and condition number $\kappa = L/\mu \approx 3.2$. Due to observation noise, the minimiser is the OLS solution $\theta_{\text{OLS}} = H^{-1}b \approx (3.028, 4.017)^T$ with $J^\star \approx 0.4826$. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \text{diag}(2 - \sin(x_1),\ 10)$. The Hessian is \emph{state-dependent}: the $(1,1)$ entry varies in $[1, 3]$ as $x_1$ changes, making this a mildly nonlinear problem. The condition number varies from approximately $10/3 \approx 3.3$ to $10/1 = 10$.

The unique minimiser is found by solving $2(x_1 - 1) + \cos(x_1) = 0$ and $x_2 = 2$, giving $x^\star \approx (0.5824, 2.0000)$ with $f^\star \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $x^\star = (1, 1)$ with $f^\star = 0$. The Hessian at the optimum is:
\begin{equation}
\nabla^2 f(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues approximately $\lambda_1 \approx 0.5$ and $\lambda_2 \approx 1001.5$, giving condition number $\kappa \approx 2000$. This extreme ill-conditioning, combined with the narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions for gradient-based methods. Starting point: $x_0 = [-1, 1]^T$.

For constrained problems: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\subsection{Notation and Assumptions}
Throughout this report, $\|\cdot\|$ denotes the Euclidean norm. A function $f$ is said to be $L$-\emph{smooth} if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x - y\|$ for all $x, y$, and $\mu$-\emph{strongly convex} if $f(y) \geq f(x) + \nabla f(x)^T(y-x) + \frac{\mu}{2}\|y-x\|^2$ for all $x, y$. The theoretical convergence results cited in each section assume the relevant smoothness and/or convexity conditions hold.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations and Theoretical Motivation}

Four adaptive methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for 120 iterations.

For gradient descent with constant step size $\alpha \leq 1/L$, the following convergence rate holds for $\mu$-strongly convex, $L$-smooth functions:
\begin{theorem}[Gradient Descent Convergence]
\label{thm:gd}
For $f$ that is $\mu$-strongly convex and $L$-smooth, GD with step size $\alpha = 1/L$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \left(1 - \frac{\mu}{L}\right)^k \bigl(f(x_0) - f^\star\bigr).
\end{equation}
The convergence factor $\rho = 1 - \mu/L = 1 - 1/\kappa$ approaches 1 for ill-conditioned problems ($\kappa \gg 1$), explaining the slow convergence of GD on Rosenbrock.
\end{theorem}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive, problem-dependent step}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step is motivated by the observation that for a quadratic $f$, the step $\alpha_k = (f(x_k) - f^\star)/\|\nabla f(x_k)\|^2$ gives the exact optimal step along the gradient direction. For convex functions with known $f^\star$, the Polyak step achieves the convergence guarantee:
\begin{equation}
f(x_k) - f^\star \leq \frac{\|x_0 - x^\star\|^2}{2\sum_{j=0}^{k-1}\alpha_j},
\end{equation}
resulting in an $O(1/k)$ rate. Crucially, this guarantee requires that $f^\star$ be known exactly.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Coordinate-wise accumulated squared gradient}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's per-coordinate scaling $\alpha_0/\sqrt{G_k^{(i)}}$ automatically assigns larger steps to infrequently updated coordinates and smaller steps to frequently updated ones. This is particularly beneficial for anisotropic problems where different coordinates have vastly different curvatures. The denominator $\sqrt{G_k}$ is monotonically increasing, however, so the effective step size decays to zero over time---a limitation on non-stationary problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's vanishing step issue by replacing the full history with an exponential moving average of squared gradients. The ``forgetting factor'' $\beta = 0.9$ gives recent gradients higher weight, preventing the effective step from vanishing on stationary problems.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method adds a ``momentum'' term: the update accumulates a weighted history of past gradients. For a quadratic $f$ with condition number $\kappa$, the optimal Heavy Ball parameters achieve a convergence factor of $\rho_{\text{HB}}^{\star} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$, compared to $\rho_{\text{GD}}^{\star} = (\kappa-1)/(\kappa+1)$ for optimal GD. This is a significant improvement for large $\kappa$.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$). Otherwise, Adagrad and Heavy Ball match or exceed GD.
\item \textbf{Rate:} GD exhibits linear convergence at rate $\rho = 1 - 1/\kappa$ (Theorem~\ref{thm:gd}). Adagrad achieves sublinear $O(1/\sqrt{k})$ regret in the online learning setting but can exhibit faster rates on well-conditioned smooth problems. Heavy Ball with optimal $(\alpha, \beta)$ achieves the accelerated rate $\rho_{\mathrm{HB}} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1) \ll \rho_{\mathrm{GD}}$ for quadratics.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves. Heavy Ball shows characteristic damped oscillation due to momentum.
\item \textbf{Stability:} GD is most stable but slowest. Polyak is \emph{unstable} when $f^\star$ is misspecified. Adagrad is uniformly stable but may converge too slowly due to the monotonically shrinking step size.
\item \textbf{Practical implication:} Exact $f^\star$ is rarely known in practice, limiting Polyak's applicability. Heavy Ball and Adagrad are strong defaults for smooth and sparse-gradient problems respectively.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations and Theoretical Motivation}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This ``look before you leap'' strategy allows Nesterov to correct its trajectory using gradient information from where the iterate is heading, preventing overshoot. The theoretical consequence is:

\begin{theorem}[Nesterov Optimal Rate]
\label{thm:nesterov}
For $f$ that is $L$-smooth and convex, the Nesterov momentum scheme with $\alpha = 1/L$ and the schedule $\beta_k = (k-1)/(k+2)$ achieves:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2},
\end{equation}
i.e., an $O(1/k^2)$ convergence rate, compared to GD's $O(1/k)$ rate.  This is optimal among all first-order methods for smooth convex optimisation.
\end{theorem}

This represents a quadratic speedup in the iteration count required to achieve a given precision. For Benchmark~A with $L \approx 1.524$ and $\|x_0 - x^\star\| \approx 5.06$, the bound predicts $f(x_k) - f^\star \leq 2 \times 1.524 \times 5.06^2 / (k+1)^2 \approx 78 / (k+1)^2$.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum (via $m_t$, equivalent to RMSprop on the gradient) with per-coordinate adaptive scaling (via $v_t$), inheriting advantages of both. The bias-correction factors $(1 - \beta_1^t)^{-1}$ and $(1 - \beta_2^t)^{-1}$ are critical in early iterations: without them, the moments are biased towards zero because $m_0 = v_0 = \mathbf{0}$. Note that unlike Nesterov, Adam does not have a proven $O(1/k^2)$ convergence rate for general smooth convex functions; its convergence guarantees are weaker and problem-dependent.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data randomly shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $\hat{g} \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The stochastic gradient $\hat{g}$ is an unbiased estimator of the full gradient $\nabla J(\theta) = H\theta - b$, with variance $\text{Var}(\hat{g}) \propto \sigma^2/b$ where $\sigma^2$ is the per-sample gradient variance. Smaller batches $b$ provide more updates per epoch (higher frequency) but with higher gradient variance, creating a bias--variance tradeoff.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.0039 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton update minimises the second-order approximation. Setting $\partial/\partial x [g'(x_0)(x - x_0) + \frac{1}{2}g''(x_0)(x-x_0)^2] = 0$ yields:
\begin{equation}
x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3}.
\end{equation}
This contracts towards 0 with ratio 2/3 per step, confirming linear convergence for $g(x) = x^4$ from this initialisation. For a general function near its minimiser $x^\star$, Newton's method achieves quadratic convergence (see Theorem~\ref{thm:newton}).

\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$. The first-order approximation captures only the local slope and deviates rapidly. The second-order approximation fits well in a neighbourhood, and its minimum provides a superior estimate of the function's minimum compared to a gradient step. Newton's method exploits this by stepping to the minimum of the local quadratic model.}

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, Hessian damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised Hessian (ensures PD)}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Newton's Quadratic Convergence]
\label{thm:newton}
Let $f$ be twice continuously differentiable with $\nabla^2 f$ being $M$-Lipschitz in a neighbourhood of $x^\star$, and let $\nabla^2 f(x^\star)$ be positive definite. For full Newton steps ($\alpha = 1$), if $\|x_0 - x^\star\|$ is sufficiently small, then:
\begin{equation}
\|x_{k+1} - x^\star\| \leq \frac{M}{2\mu}\|x_k - x^\star\|^2,
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^\star)) > 0$. This \emph{quadratic convergence} means the number of correct decimal digits roughly doubles at each iteration, once the iterate enters a sufficiently small neighbourhood of $x^\star$.
\end{theorem}

\begin{remark}
For a strictly quadratic function, $\nabla^2 f$ is constant (the $M$-Lipschitz constant is $M = 0$), so the Newton step with $\alpha = 1$ finds $x^\star$ exactly in a single iteration, regardless of the starting point.
\end{remark}

Settings: GD runs for 80 iterations; Newton runs for only 20 iterations with Hessian regularisation $\lambda = 10^{-8}$.

\subsection{Computational Cost Analysis}

Newton's method requires computing and factorising the Hessian at each iteration. For the 2-dimensional problems studied here, this overhead is negligible. For large-scale problems ($n \gg 1$), \emph{quasi-Newton} methods such as L-BFGS approximate the Hessian using rank-1 or rank-2 updates from gradient differences, achieving superlinear convergence at $O(n)$--$O(n^2)$ cost per iteration.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
The approximation error for twice-differentiable $f$ decomposes as:
\begin{equation}
\left\|\hat{g} - \nabla f(x)\right\| \leq \underbrace{\frac{\delta}{2}\max_i\left|\frac{\partial^2 f}{\partial x_i^2}(x)\right|}_{\text{Truncation error: }O(\delta)} + \underbrace{\frac{n^{1/2}\,\varepsilon_{\mathrm{mach}}\,|f(x)|}{\delta}}_{\text{Rounding error: }O(\varepsilon_{\mathrm{mach}}/\delta)}.
\end{equation}
The total error is minimised when the two terms balance, giving the optimal step $\delta^\star \sim \sqrt{\varepsilon_{\mathrm{mach}} \cdot |f(x)|/\max_i|\partial^2 f/\partial x_i^2|} \approx 10^{-5}$ to $10^{-8}$ for double precision. The assignment tests $\delta = 0.05$ (moderately good) and $\delta = 0.8$ (poor, large truncation error).

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q4_fd_error_analysis.pdf}
\caption{Q4: Finite difference gradient error $\|\hat{g} - \nabla f\|$ as a function of $\delta$ for Benchmark~B at $x = (0.5, 2.5)$. The characteristic U-shape reflects the competition between truncation ($O(\delta)$, right slope) and rounding ($O(\varepsilon_{\mathrm{mach}}/\delta)$, left slope). The minimum error occurs near $\delta \approx 10^{-5}$. The values $\delta = 0.05$ (green) and $\delta = 0.8$ (red) used in the experiments are both to the right of the optimal, where truncation error dominates. Their error ratio ($\approx 16\times$) matches the ratio of their $\delta$ values.}
\label{fig:q4_fd_error}
\end{figure}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_n)$, normalise: $u_k \gets u_k / \|u_k\|$ (uniform on sphere)
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The random search gradient estimator satisfies $\mathbb{E}[\hat{g}_k \mid x_k] = \frac{1}{n}\nabla f(x_k) + O(\delta^2)$, i.e., it estimates a scaled version of the gradient up to a bias of $O(\delta^2)$. This method requires only \emph{two} function evaluations per iteration regardless of dimension $n$, making its per-iteration cost dimension-independent---a key advantage over finite-difference GD which requires $n+1$ evaluations per step.

However, the effective step towards the true gradient direction is only a $1/n$ fraction on average, making convergence $n$ times slower than finite-difference GD with the same number of iterations. The parameters used are $\alpha = 0.025$, $\delta = 0.08$, run for 220 iterations.

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively refines it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): move the worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2.0$): if reflection improves, try a further extension.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection is worse than all but the worst, contract towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all vertices towards the best vertex.
\end{enumerate}

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}

\subsection{Problem Setup}

Minimise Benchmark~B:
\begin{equation}
\min_{x} f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad \text{subject to } x_1 \geq 0.5,
\end{equation}
starting from the infeasible point $x_0 = (0.2, 4.0)$.

The unconstrained minimum is at $x^\star_{\text{unc}} \approx (0.582, 2.000)$ with $f^\star_{\text{unc}} \approx 0.724$. Since $0.582 > 0.5$, the unconstrained minimiser is \emph{feasible}, so the constrained minimum coincides with the unconstrained one: $x^\star = (0.582, 2.000)$, $f^\star = 0.7244$.

\subsection{Projected Gradient Descent}

For a halfspace constraint $\{x : x_1 \geq 0.5\}$, the orthogonal projection is the element-wise operation:
\begin{equation}
\Pi_{\mathcal{X}}(y) = \bigl(\max(0.5,\, y_1),\; y_2\bigr)^T,
\end{equation}
which is a trivial $O(1)$ computation. For general convex sets $\mathcal{X}$, projection can be expensive (e.g., semidefinite cone: $O(n^3)$; nuclear norm ball: requires full SVD).

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = (\max(0.5,\, z_1),\; z_2)^T$ \Comment{Project onto feasible set}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[PGD Convergence]
For $f$ that is $L$-smooth and convex, and compact convex $\mathcal{X}$, PGD with $\alpha = 1/L$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{L\|x_0 - x^\star\|^2}{2k}.
\end{equation}
If additionally $f$ is $\mu$-strongly convex on $\mathcal{X}$, the rate improves to linear: $f(x_k) - f^\star \leq (1 - \mu/L)^k (f(x_0) - f^\star)$.
\end{theorem}

PGD enforces the constraint as a \emph{hard} requirement at every iteration: the iterates $\{x_k\}$ are guaranteed feasible for all $k \geq 0$.

\subsection{Penalty Method}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda\,\max(0,\, 0.5 - x_1)$ \Comment{Penalised objective (soft constraint)}
    \State $\nabla F_\lambda(x) \gets \nabla f(x) - \lambda\,\mathbf{1}[x_1 < 0.5]\,e_1$ \Comment{Gradient with penalty}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

As $\lambda \to \infty$, the penalty method converges to the solution of the constrained problem. However, large $\lambda$ increases the condition number of the penalised problem, requiring smaller step sizes for stability. The penalty gradient contains a kink at $x_1 = 0.5$, making exact gradient computation straightforward but introducing a non-smooth component.

Settings: PGD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All methods run for 100 iterations.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (an axis-aligned box / polytope).

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is obtained by minimising each coordinate independently. Since $a_1 = 1 > 0$ and $a_2 = 2 > 0$, both components are minimised at their lower bounds:
\begin{equation}
x^\star = (0.5,\, -5), \qquad f^\star = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates a fundamental result: the minimum of a linear objective over a bounded polytope is attained at a \emph{vertex} (extreme point). For an $\ell$-dimensional box with $2n$ vertices, the LP can be solved in $O(n)$ time by component-wise minimisation.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: $x_{k+1} \in \mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Frank--Wolfe Convergence]
\label{thm:fw}
Let $f$ be $L$-smooth and convex on the compact convex set $\mathcal{X}$ with diameter $D = \max_{x,y \in \mathcal{X}}\|x-y\|$. The standard Frank--Wolfe algorithm with step size $\gamma_k = 2/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{2LD^2}{k+2},
\end{equation}
i.e., an $O(1/k)$ convergence rate. In the implementation used here, the fixed mixing parameter $\beta$ replaces the decaying schedule $\gamma_k = 1 - \beta$ and does not achieve the same theoretical guarantee.
\end{theorem}

\textbf{Key advantage of Frank--Wolfe over PGD:} The LMO step for box constraints requires only $O(n)$ operations (component-wise sign decisions), while projection onto general convex sets can require $O(n^3)$ computation. For problems where projection is expensive (e.g., nuclear norm ball, spectrahedra), Frank--Wolfe is substantially cheaper per iteration. Furthermore, the iterates are always convex combinations of feasible points, guaranteeing feasibility without any explicit projection.

For box constraints, the LMO solution is:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
Thus $z_k$ is always a vertex of $\mathcal{X}$.

\subsection{Interior vs.\ Boundary Behaviour and Method Comparison}

\textbf{Interior optimum:} Frank--Wolfe exhibits the characteristic zig-zag pattern near the optimum. The LP vertex $z_k$ alternates between extreme points of $\mathcal{X}$ as the gradient direction fluctuates, and the iterates are pulled towards these vertices. The $O(1/k)$ convergence rate means that to achieve precision $\varepsilon$, approximately $O(1/\varepsilon)$ iterations are required---10 times more than Nesterov's $O(1/\sqrt{\varepsilon})$ requirement. For interior optima, projected gradient descent converges faster.

\textbf{Boundary optimum:} Frank--Wolfe is better suited to boundary optima because $z_k$ frequently identifies the correct constraint-boundary vertex. The iterate converges to the constraint boundary naturally via the convex combination structure. However, the fixed-$\beta$ schedule does not achieve the theoretical $O(1/k)$ rate; using the standard decaying $\gamma_k = 2/(k+2)$ schedule would improve convergence.

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This study demonstrates that the choice of optimisation algorithm is problem-dependent and governed by three key factors: the smoothness and conditioning of the objective, the availability of gradient information, and the presence of constraints.

\textbf{On smooth, well-conditioned problems} (Benchmark~A, $\kappa \approx 3$), most first-order methods converge within a comparable number of iterations, and Newton's method finds the exact solution in a single step. Adaptive step-size methods (Adagrad, Heavy Ball) match GD without tuning the step size for each benchmark.

\textbf{On ill-conditioned problems} (Benchmark~C, $\kappa \approx 2000$), the advantages of acceleration become decisive. Nesterov momentum reduces the required iteration count from the GD rate $(1-1/\kappa)^k$ to the accelerated rate $O(1/k^2)$. Damped Newton's method achieves further improvement by using second-order curvature information directly.

\textbf{For derivative-free settings}, the accuracy--cost tradeoff is steep. Finite-difference GD with a well-chosen $\delta$ closely tracks exact GD at $d+1$ function evaluations per step. Nelder--Mead and random search trade gradient fidelity for freedom from any derivative computation, at the cost of slower and noisier convergence.

\textbf{For constrained optimisation}, projected gradient descent offers the simplest hard-constraint enforcement when projection is cheap. Penalty methods provide an alternative when projection is expensive, at the cost of approximate feasibility for finite $\lambda$. Frank--Wolfe is preferred when only a cheap linear oracle is available, particularly for problems with sparse optimal solutions.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum with Adaptive $\beta_k$ Schedule}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z      # Evaluate gradient at lookahead point
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser with Bias Correction}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # Bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Hessian Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))  # Tikhonov regularisation
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g  # Fallback to gradient direction
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # Linear minimisation oracle for box constraints
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z  # Convex combination
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
