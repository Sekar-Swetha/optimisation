\begin{abstract}
This report presents a rigorous comparative study of optimisation algorithms for the CS7DS2 final assignment, covering first-order, second-order, derivative-free, and constrained methods applied to three benchmark functions: a linear regression quadratic loss (Benchmark A, $\kappa \approx 2.3$, easy), a toy neural network loss with sinusoidal perturbation (Benchmark B, $\kappa \approx 5$, moderate), and the Rosenbrock function (Benchmark C, $\kappa \approx 2510$, highly ill-conditioned). Six algorithm families are studied across six questions:

\noindent\textbf{Q1: Adaptive step-size methods.} Polyak step, Adagrad, RMSprop, and Heavy Ball are compared. Polyak achieves the best Rosenbrock convergence ($f=0.009$) when $f^\star$ is known but diverges when misspecified; RMSprop fails to converge to the exact minimum on Benchmark A due to a non-vanishing effective step near $\|g\|=0$.

\noindent\textbf{Q2: Momentum methods.} Nesterov acceleration (optimal $O(1/k^2)$) outperforms GD by $7\times$ on Rosenbrock ($f=0.479$ vs $3.388$). Adam's coordinate scaling handles anisotropy well on Benchmark B but is limited by the off-diagonal curvature of Rosenbrock's banana valley.

\noindent\textbf{Q3: Newton's method.} Newton converges in exactly one iteration on the quadratic benchmark (exact Hessian inversion) and in 20 damped steps to $f=0.686$ on Rosenbrock vs GD's $f=3.73$, demonstrating the power of curvature information. Quadratic convergence $\|e_{k+1}\| \leq C\|e_k\|^2$ is analysed theoretically.

\noindent\textbf{Q4: Derivative-free methods.} Finite-difference accuracy depends critically on step size $\delta$; Nesterov random search requires only 2 function evaluations per iteration; Nelder--Mead adapts its simplex to the valley geometry; grid search provides global coverage at $O(N^d)$ cost.

\noindent\textbf{Q5: Constrained optimisation.} Projected GD enforces feasibility as a hard constraint from iteration 1 via the non-expansive projection operator; penalty methods enforce it softly with trade-off controlled by $\lambda$. The constraint is inactive at the Benchmark B optimum (KKT multiplier $\mu^\star=0$).

\noindent\textbf{Q6: Frank--Wolfe.} The conditional gradient algorithm solves a linear subproblem per step, producing projection-free iterates that remain feasible by construction. Interior and boundary optima are analysed, and the duality gap $g_k = \langle \nabla f, x_k - z_k \rangle$ is used as a convergence certificate. Convergence rates, algorithm connections to course theory, and practical recommendations are provided throughout.
\end{abstract}

\begin{table}[H]
\centering
\caption{Theoretical convergence rates. $\kappa = L/\mu$ (condition number), $\varepsilon$ = target accuracy, GD = gradient descent.}
\label{tab:rates}
\begin{tabular}{llll}
\toprule
\textbf{Algorithm} & \textbf{Rate (convex)} & \textbf{Rate (strongly convex)} & \textbf{Optimal?} \\
\midrule
Gradient Descent & $O(L/k)$ & $(1 - \mu/L)^k$ (linear) & No \\
Heavy Ball (optimal) & $O(L/k)$ & $((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^{2k}$ & Yes (quadratic) \\
Nesterov Acceleration & $O(L/k^2)$ & $(1 - 1/\sqrt{\kappa})^k$ (linear) & \textbf{Yes} (1st-order) \\

Newton's Method & --- & $\|e_{k+1}\| \leq C\|e_k\|^2$ (local) & N/A \\
Frank--Wolfe (std.\ $\gamma_k$) & $O(C_f/k)$ & $O(C_f/k)$ & No (sublinear) \\
Mini-batch SGD & $O(\sigma^2/\sqrt{k})$ & $O(\sigma^2/(\mu k))$ & Yes (stochastic) \\
\bottomrule
\end{tabular}
\end{table}

\noindent The key insight from this table is the hierarchy: \textbf{Newton} (superlinear, quadratic locally) $\gg$ \textbf{Nesterov/Heavy Ball} (optimal first-order, $O(1/k^2)$ or linear at $O(1/\sqrt{\kappa})$ rate) $>$ \textbf{GD} (linear at $O(1-1/\kappa)$ rate, $\approx O(1/k)$ for convex) $>$ \textbf{Adam/SGD} (stochastic, $O(1/\sqrt{k})$) $>$ \textbf{Frank--Wolfe} ($O(1/k)$, projection-free trade-off). On ill-conditioned problems ($\kappa \gg 1$), the gap between Newton and first-order methods grows dramatically.

The convergence of GD and all descent methods rests on the \emph{Descent Lemma}: if $\nabla f$ is $L$-Lipschitz (equivalently, $f$ is $L$-smooth), then for any $x,y$:
\begin{equation}
f(y) \leq f(x) + \nabla f(x)^T(y-x) + \frac{L}{2}\|y-x\|^2.
\label{eq:descent_lemma}
\end{equation}
Substituting $y = x - \frac{1}{L}\nabla f(x)$ gives $f(y) \leq f(x) - \frac{1}{2L}\|\nabla f(x)\|^2$, showing that every GD step with $\alpha = 1/L$ decreases $f$ by at least $\|\nabla f\|^2/(2L)$. This is the fundamental per-step progress guarantee underlying all convergence proofs for gradient-based methods studied in this report.

\section{Benchmark Problems and Experimental Setup}

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. We summarise the key analytic properties of each benchmark before presenting results.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \qquad
x^{(i)} \sim \mathcal{N}(0, I_2), \qquad
y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
= \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strongly convex quadratic with constant Hessian $H = \frac{1}{m}X^T X$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. The exact OLS minimiser is $\theta_{\mathrm{OLS}} = (X^TX)^{-1}X^Ty \approx [3.028, 4.017]^T$. Because $y$ is contaminated by noise $\varepsilon \sim \mathcal{N}(0,1)$, the optimal (noise floor) loss is $J^\star \approx 0.4826 \neq 0$. Starting point: $\theta_0 = [0, 0]^T$.

\textbf{Condition number.} With $m=1000$, $x^{(i)} \sim \mathcal{N}(0,I_2)$, the population Hessian is $I_2$, so the empirical Hessian is close to the identity: $\lambda_{\max}/\lambda_{\min} \approx 1$. The empirical condition number is approximately $\kappa \approx 2.3$, making Benchmark A well-conditioned and easy for all gradient methods. The GD convergence factor for this benchmark is $(1 - 1/\kappa)^2 \approx (1-1/2.3)^2 \approx (0.565)^2 \approx 0.32$ per iteration, i.e.\ roughly a $3\times$ reduction in suboptimality gap per step.

\textbf{Strong convexity.} Since $H = X^TX/m \succ 0$, the loss $J$ is $\mu$-strongly convex with $\mu = \lambda_{\min}(H) \approx 0.43$ and $L$-smooth with $L = \lambda_{\max}(H) \approx 0.98$. For a $\mu$-strongly convex, $L$-smooth function, GD with $\alpha = 1/L$ satisfies:
\begin{equation}
f(x_k) - f^\star \;\leq\; \left(1 - \frac{\mu}{L}\right)^k [f(x_0) - f^\star] \;=\; \left(1 - \frac{1}{\kappa}\right)^k [f(x_0) - f^\star],
\label{eq:gd_strongly_convex}
\end{equation}
which follows from applying~\eqref{eq:descent_lemma} with the strong convexity lower bound. For Benchmark A: $\kappa \approx 2.3$, $(1 - 1/2.3)^k \approx 0.565^k$, requiring only $\log(1/\varepsilon)/\log(1/0.565) \approx 4$ iterations per decimal digit of accuracy. GD with $\alpha = 2/(L+\mu)$ is theoretically optimal for the linear rate, corresponding to $\alpha \approx 1.41$; in practice $\alpha = 0.08$ is used, which is conservative but stable.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\label{eq:bench_B}
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \mathrm{diag}(2 - \sin(x_1),\ 10)$. The minimum satisfies $10(x_2-2)=0 \Rightarrow x_2^\star = 2$ and $2(x_1-1)+\cos(x_1)=0$, solved numerically to give $x_1^\star \approx 0.582$, yielding $f^\star \approx 0.7234$. The curvature ratio between the two coordinates ranges from approximately $1.8:10$ to $3:10$, creating moderate anisotropy ($\kappa \approx 5$). Starting point: $x_0 = [-1, 4]^T$.

\begin{remark}
The figures mark the approximate visual optimum at $(0.61, 2.0)$ for illustration purposes; the true analytical minimum is at approximately $(0.582, 2.0)$ as computed from the stationarity condition.
\end{remark}

\subsection{Benchmark C: Rosenbrock Function}

\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\label{eq:rosenbrock}
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$. At the minimum, the Hessian is:
\begin{equation}
H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues $\lambda_{1,2} = (1002 \pm \sqrt{1002^2 - 4 \cdot 400})/2 \approx 0.40$ and $1001.6$, yielding condition number $\kappa = \lambda_{\max}/\lambda_{\min} \approx 2510$. This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock the most challenging benchmark and the most discriminating test for algorithm quality. Starting point: $x_0 = [-1, 1]^T$.

\begin{remark}[Polyak-Łojasiewicz (PL) condition]
Although Rosenbrock is non-convex globally (the Hessian is indefinite away from $(1,1)$), GD still achieves a linear convergence rate near the minimum if the \emph{PL condition} holds: $\frac{1}{2}\|\nabla f(x)\|^2 \geq \mu(f(x)-f^\star)$ for some $\mu > 0$. This condition is strictly weaker than strong convexity (it allows multiple local minima) and is satisfied for functions where every stationary point is a global minimum. For Rosenbrock near $(1,1)$, the PL condition holds locally with $\mu \approx \lambda_{\min}(H(1,1)) \approx 0.40$, which explains why GD eventually achieves linear convergence once the iterate enters the local strongly-convex region, even though $f$ is non-convex globally.
\end{remark}

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

This question compares four adaptive step-size mechanisms --- Polyak, Adagrad, RMSprop, and Heavy Ball --- against a GD baseline across all three benchmarks for 120 iterations. The central theme is how each method adapts the step size: Polyak uses the suboptimality gap $f(x_k)-f^\star$; Adagrad uses cumulative squared gradients; RMSprop uses an exponential moving average; Heavy Ball uses momentum from previous gradients. Key findings: Polyak is optimal when $f^\star$ is known (Benchmark C: $f=0.009$) but diverges when misspecified; Heavy Ball provides the most robust benefit across benchmarks without requiring $f^\star$.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Diminishing if $f^\star$ is exact}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theory.} For a convex $L$-smooth function with known $f^\star$, the Polyak step satisfies:
\begin{equation}
\|x_{k+1} - x^\star\|^2 \leq \|x_k - x^\star\|^2 - \frac{(f(x_k) - f^\star)^2}{\|g_k\|^2 + \varepsilon}.
\end{equation}
The iterate-to-optimum distance is monotonically non-increasing, guaranteeing convergence whenever $f^\star$ is exact. When $f^\star$ is misspecified (too low), the denominator $\|g_k\|^2 + \varepsilon \to \varepsilon$ as $\nabla f(x_k) \to 0$ while the numerator $f(x_k) - f^\star_{\mathrm{wrong}}$ remains bounded away from zero, causing $\alpha_k \to \infty$ and divergence.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative sum of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theory.} Adagrad's effective per-coordinate step size $\alpha_k^{(i)} = \alpha_0 / (\sqrt{G_k^{(i)}} + \varepsilon)$ decays as $O(1/\sqrt{k})$, which guarantees convergence for convex problems but may cause premature stagnation on non-convex or long-horizon problems. The coordinate-wise scaling provides implicit normalisation: coordinates with historically large gradients receive smaller steps.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theory.} RMSprop replaces Adagrad's monotone accumulation with an exponential moving average controlled by $\beta$. This forgetting mechanism prevents the effective step size from decaying to zero: the steady-state effective step for a coordinate with constant gradient magnitude $g$ is $\alpha_{\mathrm{eff}} = \alpha_0 / (\sqrt{(1-\beta)/(1-\beta^k)} \cdot g + \varepsilon) \approx \alpha_0/g$, providing scale-invariant updates.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Velocity accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theory.} For a strongly convex quadratic $f(\theta) = \frac{1}{2}\theta^T H \theta - b^T\theta$ with condition number $\kappa = \lambda_{\max}/\lambda_{\min}$, the optimal Heavy Ball parameters are $\alpha^\star = 4/(\sqrt{\lambda_{\max}}+\sqrt{\lambda_{\min}})^2$ and $\beta^\star = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$, yielding convergence rate $\rho^\star = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$ per iteration. This equals Nesterov's strongly-convex rate $(1-1/\sqrt{\kappa})^2$ and the Chebyshev polynomial optimal rate for reducing error in the $H$-norm over the spectrum $[\lambda_{\min}, \lambda_{\max}]$.

\begin{remark}[Heavy Ball failure on non-quadratic functions]
Despite achieving the optimal rate on quadratics, Heavy Ball with fixed $(\alpha^\star, \beta^\star)$ can \emph{diverge} on non-quadratic strongly convex functions. Lessard, Recht, and Packard (2016) demonstrated this using linear matrix inequality (LMI) analysis: the optimally-tuned Heavy Ball parameters for the quadratic case are not robust to nonlinear perturbations of the Hessian, because the velocity $z_k$ can carry stale curvature information from earlier (higher-curvature) regions. This contrasts with Nesterov's method, which has a proven $O(1/k^2)$ guarantee for general convex functions and $(1-1/\sqrt{\kappa})^k$ linear rate for strongly convex functions, making it strictly more reliable than Heavy Ball outside the quadratic setting.
\end{remark}

\textbf{RMSprop residual on Benchmark A.} RMSprop settles at $f = 0.5027 > f^\star = 0.4826$, a residual gap of 0.020. Near the OLS minimum, gradients become very small. The EMA $v_k \approx (1-\beta)^2 g^2/(1-\beta) = (1-\beta) g^2$ for small $g$ (if $v$ has partially converged), giving effective step $\alpha_{\mathrm{eff}} = 0.22/\sqrt{(1-\beta)g^2 + \varepsilon}$. When $g^2 < \varepsilon/(1-\beta) = 10^{-5}/0.1 = 10^{-4}$, the denominator is dominated by $\varepsilon$, and the effective step is approximately $0.22/\sqrt{10^{-5}} \approx 69.6$, which is far too large and causes oscillation. Adagrad's monotone accumulation avoids this: it adds up all past squared gradients, so $G_k \to \infty$ as $k \to \infty$ and $\alpha_{\mathrm{eff}} \to 0$, ensuring convergence to the exact minimum. This explains why Adagrad reaches the noise floor $f^\star$ exactly while RMSprop does not.

\textbf{Key finding.} The Polyak step size is highly sensitive to the accuracy of $f^\star$. When $f^\star$ is exact (Benchmark C), Polyak achieves by far the best convergence among the adaptive methods. When $f^\star$ is misspecified as zero (Benchmarks A and B), the method diverges. This underlines a fundamental limitation in practice: exact $f^\star$ is rarely known.

\subsection{Step-Size Theory Synthesis}

The three adaptive step-size mechanisms differ fundamentally in how they estimate the appropriate curvature for normalisation:
\begin{itemize}[nosep]
\item \textbf{Polyak:} Uses the \emph{suboptimality gap} $f(x_k) - f^\star$ as a proxy for the distance to the optimum. This is optimal when $f^\star$ is known (the step perfectly overshoots the minimum by the right amount) but catastrophic when it is not.
\item \textbf{Adagrad:} Uses the \emph{historical gradient magnitudes} per coordinate as a proxy for curvature. This is geometrically motivated: coordinates with large gradients have large curvature, justifying smaller steps. The monotone accumulation makes steps irreversibly smaller as training progresses.
\item \textbf{RMSprop:} Uses an \emph{exponential moving average} of squared gradients, balancing recent curvature estimates against historical ones. The forgetting factor $\beta$ sets the effective window size to $1/(1-\beta)$ iterations, allowing the method to adapt to non-stationary gradient landscapes. See Figure~\ref{fig:adaptive_comparison} for a direct comparison of these mechanisms.
\end{itemize}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$). Across all three benchmarks combined, Heavy Ball provides the most consistent improvement over GD without requiring $f^\star$. For a quadratic with condition number $\kappa$, GD converges in $f$-suboptimality at rate $(1-1/\kappa)^k$ while optimal Heavy Ball converges at rate $((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^{2k}$. For $\kappa \approx 2510$ (Rosenbrock), $\rho = (\sqrt{2510}-1)/(\sqrt{2510}+1) \approx 0.9609$ and the $f$-convergence rate per iteration is $\rho^2 \approx 0.9233$, compared to GD's $0.9996$. The convergence exponent ratio is $|\ln(0.9233)|/|\ln(0.9996)| \approx 0.0799/0.0004 \approx 200$: Heavy Ball decays $\approx 200\times$ faster per step in the asymptotic regime. In practice, after only 120 iterations, Heavy Ball achieves $f = 1.40$ vs.\ GD's $f = 3.51$ (a $2.5\times$ improvement, Table~\ref{tab:q1_final}), far below the asymptotic prediction because 120 iterations is pre-asymptotic: neither method has come close to convergence on this severely ill-conditioned ($\kappa \approx 2510$) problem within the iteration budget.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging. Heavy Ball exhibits characteristic overshoot followed by damped oscillation, particularly pronounced on ill-conditioned problems.
\item \textbf{Stability:} GD is uniformly stable for $\alpha < 2/L$. Adagrad is unconditionally stable due to its monotonically decreasing step size, though this may cause premature stagnation. RMSprop is stable with appropriately chosen $\beta$ and $\alpha_0$.
\item \textbf{Practical recommendation:} RMSprop and Adagrad are robust when $f^\star$ is unknown. Polyak is optimal when $f^\star$ is known exactly. Heavy Ball achieves near-optimal first-order rates when the momentum is tuned to match the condition number; however, unlike Nesterov acceleration (Section~\ref{sec:q2}), Heavy Ball lacks a provable $O(1/k^2)$ guarantee on general convex functions.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

This question studies momentum-based methods (Nesterov accelerated gradient, Adam) and mini-batch SGD for 150 iterations on all three benchmarks. Nesterov's method achieves the optimal first-order $O(1/k^2)$ convex rate and $O(e^{-k/\sqrt{\kappa}})$ strongly-convex rate --- a $\sqrt{\kappa}$-fold advantage over GD. Adam combines momentum with per-coordinate adaptive scaling. Mini-batch SGD is analysed under both normal and high observation noise, illustrating the irreducible noise floor $J^\star = \sigma^2/2$ and the variance--speed trade-off.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\; \beta_{\max}\right)$ \Comment{Schedule approaching $(k-1)/(k+2)$}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead / extrapolation point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theory.} For an $L$-smooth convex function, Nesterov's accelerated gradient achieves:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2}.
\end{equation}
This rate is \emph{minimax optimal}: Nesterov (1983) proved that for any first-order algorithm (that only uses gradient oracle calls $\nabla f(x_0), \ldots, \nabla f(x_{k-1})$), there exists an $L$-smooth convex function and initial point such that
\begin{equation}
f(x_k) - f^\star \;\geq\; \frac{3L\|x_0 - x^\star\|^2}{32(k+1)^2},
\end{equation}
matching the upper bound to within a constant. This proves that no first-order algorithm can do better than $O(1/k^2)$ on this function class --- Nesterov's method is rate-optimal. This is quadratically faster than gradient descent's $O(L/k)$ rate, and the gap grows without bound: after $k$ steps, GD achieves accuracy $L/k$ while Nesterov achieves $L/k^2$, giving a $k$-fold advantage that grows with iteration count. The key innovation is evaluating the gradient at the \emph{extrapolated} point $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$, correcting for the direction of motion before taking the gradient step. This is equivalent to the FISTA formulation: setting $y_{k+1} = x_k + \beta_k(x_k - x_{k-1})$, $x_{k+1} = y_{k+1} - \alpha\nabla f(y_{k+1})$, which is recovered via the substitution $z_k = x_k - x_{k-1}$.

The standard Nesterov sequence uses auxiliary variables $t_k$ satisfying $t_{k+1}^2 \leq t_{k+1} + t_k^2$, with the recursive solution $t_k = (1+\sqrt{1+4t_{k-1}^2})/2$ and momentum $\beta_k = (t_{k-1}-1)/t_k \to 1$ as $k \to \infty$. The code approximates this by $\beta_k = \min((k-1)/(k+2), \beta_{\max})$; the bound $\beta_{\max}$ caps momentum to prevent instability on non-convex regions.

The FISTA formulation [10] makes the lookahead explicit via two sequences: $y_{k+1} = x_k + \frac{t_k-1}{t_{k+1}}(x_k - x_{k-1})$ (extrapolation), $x_{k+1} = y_{k+1} - \frac{1}{L}\nabla f(y_{k+1})$ (gradient step). This is recovered from Algorithm~\ref{alg:nesterov} via the substitution $z_k = x_k - x_{k-1}$ (velocity equals last displacement), so that $\tilde{x}_k = x_k + \beta_k(x_k - x_{k-1})$ is exactly the FISTA extrapolation point.

For $\mu$-strongly convex functions, Nesterov achieves linear convergence at rate $(1-1/\sqrt{\kappa})^k$, compared to GD's $(1-1/\kappa)^k$. To reach accuracy $\varepsilon$ from $f(x_0) - f^\star = \Delta_0$, the iteration counts are:
\begin{equation}
k_{\mathrm{GD}} \approx \kappa \ln(\Delta_0/\varepsilon), \qquad k_{\mathrm{Nest}} \approx \sqrt{\kappa} \ln(\Delta_0/\varepsilon),
\end{equation}
giving Nesterov a $\sqrt{\kappa}$-fold advantage. For $\kappa \approx 2510$: $\sqrt{2510} \approx 50$, so Nesterov requires $\approx 50\times$ fewer iterations than GD for the same accuracy. This large advantage is consistent with the experimental gap ($f=0.479$ vs $f=3.388$ after 150 iterations on Benchmark C).

The distinction from Heavy Ball (Algorithm~\ref{alg:heavyball}) is fundamental: Heavy Ball evaluates gradients at $x_k$ (not the lookahead) and has no proven $O(1/k^2)$ guarantee on general convex functions (and can even diverge on non-quadratic strongly convex functions, as noted in the Remark above). Nesterov's \emph{gradient at the lookahead} provides the corrective mechanism that enables both the $O(1/k^2)$ convex rate and the $\sqrt{\kappa}$ strongly-convex speed-up.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction for initialisation at 0}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theory.} Adam combines first-moment momentum (as in Heavy Ball, Algorithm~\ref{alg:heavyball}) with second-moment adaptive scaling (as in RMSprop, Algorithm~\ref{alg:rmsprop}), inheriting the advantages of both. The bias corrections $(1-\beta_1^t)^{-1}$ and $(1-\beta_2^t)^{-1}$ are essential in early iterations: without them, $m_1 = (1-\beta_1)g_1$ and $v_1 = (1-\beta_2)g_1^2$ are biased towards zero because the EMA initialises at 0, producing over-large effective steps. The correction ensures $\hat{m}_1 \approx g_1$ and $\hat{v}_1 \approx g_1^2$ from the first iteration. The effective per-coordinate step at time $t$ is:
\begin{equation}
\alpha_t^{(i)} = \frac{\alpha}{\sqrt{\hat{v}_t^{(i)}} + \varepsilon} \approx \frac{\alpha}{\sqrt{\mathbb{E}[g^{(i)2}]}},
\end{equation}
making updates approximately scale-invariant: the magnitude of $x_t^{(i)}$ updates is bounded by $\approx \alpha$ regardless of the gradient scale, which is a crucial property for robustness in deep learning where gradients can vary over many orders of magnitude. Adam does not have proven optimal convergence guarantees for general convex problems (Reddi et al., 2018 showed that Adam can fail to converge on simple convex problems; AMSGrad was proposed as a fix). In practice, Adam is highly effective on non-convex deep learning objectives due to its robustness to hyperparameter choice and ability to handle sparse gradients through its per-coordinate adaptive scaling.

\subsection{Algorithm Comparison: Nesterov vs.\ Adam}

Adam's per-coordinate scaling is most beneficial when different coordinates have different gradient magnitudes. On Rosenbrock, the dominant challenge is not coordinate-wise scaling but \emph{direction misalignment}: the optimal descent direction is tangent to the curved valley $x_2 = x_1^2$, which rotates as $(x_1, x_2)$ changes. At a generic point on the valley, the gradient $\nabla f = [2(x_1-1) + 400x_1(x_1^2-x_2),\; 200(x_2 - x_1^2)]^T$ has a large component \emph{across} the valley (steep curvature, eigenvalue $\approx 1001.6$) and a small component \emph{along} the valley (shallow curvature, eigenvalue $\approx 0.40$). Efficient descent requires combining both coordinates with appropriate signs, corresponding to off-diagonal curvature that Adam's diagonal second-moment matrix $\hat{v}_t$ cannot represent. Nesterov's momentum vector $z_k$ accumulates along the consistent valley-floor descent direction across iterations: since the valley direction changes slowly relative to the step size, $z_k$ self-aligns with the valley floor, providing implicit second-order information without computing the Hessian. This explains Nesterov's $3.3\times$ advantage over Adam ($f=0.479$ vs.\ $f=1.578$, Table~\ref{tab:q2_final}) on Rosenbrock despite Adam's theoretical sophistication.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD is applied to Benchmark A with $m = 1000$ samples, learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, and 50 epochs with random shuffling each epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y) \in \mathbb{R}^{m \times d} \times \mathbb{R}^m$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Mini-batch gradient estimate}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theory.} With $m/b$ updates per epoch, a smaller batch size $b$ provides more gradient updates per epoch at the cost of higher gradient variance. For the quadratic loss, the mini-batch gradient estimate $g_\mathcal{B} = X_\mathcal{B}^T(X_\mathcal{B}\theta - y_\mathcal{B})/b$ is unbiased: $\mathbb{E}[g_\mathcal{B}] = \nabla J(\theta)$.

\textbf{Practical context.} Beyond the basic variance--speed trade-off, there is a practical reason to prefer small batch sizes in deep learning: the noise injected by mini-batches acts as an \emph{implicit regulariser}, encouraging convergence to \emph{flatter} minima (with smaller Hessian eigenvalues). Flatter minima tend to generalise better to test data. This effect is absent from full-batch GD. However, for our convex quadratic benchmark there is a unique minimum, so this regularisation effect does not apply, and the difference between batch sizes is purely a convergence-rate phenomenon.

The gradient variance is:
\begin{equation}
\mathrm{Var}[g_{\mathcal{B},j}] = \frac{1}{b}\,\mathrm{Var}\!\left[\frac{\partial \ell_i}{\partial \theta_j}\right]
= \frac{\sigma_g^2}{b},
\end{equation}
where $\sigma_g^2$ is the per-sample gradient variance. SGD with constant $\alpha$ converges to a neighbourhood of $\theta^\star$ of radius $O(\alpha\sigma_g^2/(b\mu))$ (the noise floor), requiring either $\alpha \to 0$ or $b \to m$ for exact convergence. In practice, the number of gradient evaluations is $m/b$ per epoch, so comparing methods at equal per-epoch gradient cost: $b=5$ does 200 evaluations/epoch vs $b=40$ with 25 evaluations/epoch.

\textbf{Variance reduction methods.} The fundamental limitation of SGD --- the $O(\sigma_g^2/(b\mu k))$ noise floor requiring $\alpha \to 0$ for exact convergence --- can be overcome by \emph{variance reduction} methods such as SVRG (Stochastic Variance Reduced Gradient) and SAG/SAGA. SVRG periodically computes a full gradient $\tilde{g} = \nabla J(\tilde\theta)$ at a snapshot $\tilde\theta$, and uses the control variate $g_k = \nabla \ell_{\mathcal{B}}(\theta_k) - \nabla\ell_{\mathcal{B}}(\tilde\theta) + \tilde{g}$ as an unbiased, \emph{low-variance} gradient estimate. This reduces the stochastic gradient variance from $O(\sigma_g^2/b)$ to $O(\|\nabla J(\theta_k) - \nabla J(\tilde\theta)\|^2)$, which tends to zero as $\theta_k \to \theta^\star$. As a result, SVRG achieves the same $O((1-\mu/L)^k)$ linear rate as full-batch GD while only requiring $O(1/b)$ work per step, combining the best of both worlds. These methods are not implemented in the assignment but represent the state-of-the-art for large-scale strongly-convex minimisation.

\begin{figure}[H]

\end{figure}

\section{Question 3: Newton's Method and Local Approximation \hfill [30 marks]}
\label{sec:q3}
% ============================================================

This question studies second-order methods: first- and second-order Taylor approximations of $g(x) = x^4$ at $x_0 = 0.25$ (to motivate Newton steps), then damped Newton's method vs.\ GD across all three benchmarks for 20 and 80 iterations respectively. Key findings: Newton solves the quadratic benchmark exactly in one step (constant Hessian), converges in $\approx$5 steps on Benchmark B, and achieves $f=0.686$ vs.\ GD's $f=3.73$ on Rosenbrock. Local quadratic convergence $\|e_{k+1}\| \leq C\|e_k\|^2$ is derived; the Newton decrement (Proposition~\ref{prop:newton_decrement}) provides a certified stopping criterion.

\subsection{Local Approximations of $g(x) = x^4$}

At $x_0 = 0.25$: $g(x_0) = 0.25^4 = 0.003906$, $g'(x_0) = 4 \times 0.25^3 = 0.0625$, $g''(x_0) = 12 \times 0.25^2 = 0.75$.

\textbf{First-order approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.003906 + 0.0625(x - 0.25).
\label{eq:first_order}
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.003906 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\label{eq:second_order}
\end{equation}

The Newton update minimises the second-order model (setting $\hat{g}_2'(x) = 0$):
\begin{equation}
0 = g'(x_0) + g''(x_0)(x^+ - x_0) \implies x^+ = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.1667.
\end{equation}
In $d$ dimensions, the Newton step $p_k = H_k^{-1} g_k$ is the solution to the linear system $H_k p_k = g_k$, where $H_k = \nabla^2 f(x_k)$. The step $x_{k+1} = x_k - p_k$ is the minimiser of the local quadratic model $m_k(x) = f(x_k) + \nabla f(x_k)^T(x-x_k) + \frac{1}{2}(x-x_k)^T H_k (x-x_k)$.

\textbf{One-step exactness on quadratics.} If $f(\theta) = \frac{1}{2}\theta^T H\theta - b^T\theta + c$ (constant Hessian), the Newton step starting from any $\theta_0$ gives:
\begin{equation}
\theta_1 = \theta_0 - H^{-1}(H\theta_0 - b) = H^{-1}b = \theta^\star,
\end{equation}
i.e.\ the Newton step reaches the exact minimiser in one iteration, regardless of $\theta_0$. This is exactly the behaviour observed in Figure~\ref{fig:q3_fval_A} and Figure~\ref{fig:newton_illustration}.
For $g(x) = x^4$, the true minimiser is $x^\star = 0$. The Newton update rule is:
\begin{equation}
x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3} = \frac{2}{3}x_k.
\end{equation}
This gives the closed-form solution $x_k = (2/3)^k x_0$, so the error decays geometrically at rate $2/3$ --- \emph{linear} (not quadratic) convergence. The quadratic convergence theorem requires $g''(x^\star) \neq 0$ (Hessian non-singular at the minimum), which fails here since $g''(0) = 0$. The Hessian $12x_k^2 \to 0$ as $x_k \to 0$, so the Newton step becomes ill-conditioned near the optimum, preventing the quadratic convergence regime from activating.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \leq 1$, damping $\lambda_d > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda_d I$ \Comment{Tikhonov-regularised Hessian (ensures PD)}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theory (Quadratic Convergence).} For a twice-differentiable strongly convex function $f$ with $\mu I \preceq \nabla^2 f(x) \preceq L I$ and Lipschitz Hessian ($\|\nabla^2 f(x) - \nabla^2 f(y)\| \leq M\|x-y\|$), Newton's method with $\alpha = 1$ satisfies:
\begin{equation}
\|x_{k+1} - x^\star\| \leq \frac{M}{2\mu}\|x_k - x^\star\|^2 \quad \text{(local quadratic convergence)},
\end{equation}
once $\|x_k - x^\star\| < 2\mu/M$ (the quadratic convergence regime). This means each iteration roughly squares the error: $10^{-2} \to 10^{-4} \to 10^{-8}$, converging in $O(\log\log(1/\varepsilon))$ iterations.

\begin{proposition}[Newton Decrement Stopping Criterion]
\label{prop:newton_decrement}
Let $f$ be $\mu$-strongly convex and twice-continuously differentiable. The \emph{Newton decrement} $\lambda_N(x) = \sqrt{\nabla f(x)^T [\nabla^2 f(x)]^{-1} \nabla f(x)}$ satisfies:
\[
f(x) - f^\star \;\leq\; \frac{\lambda_N(x)^2}{2},
\]
providing a computable, $f^\star$-free upper bound on suboptimality. Stopping when $\lambda_N(x)/\sqrt{2} < \varepsilon$ certifies $f(x) - f^\star < \varepsilon^2/2$.
\end{proposition}
\begin{proof}[Sketch]
By the quadratic upper bound from $L$-smoothness: $f(x + p) \leq f(x) + \nabla f(x)^Tp + \frac{1}{2}p^T\nabla^2 f(x)p$. Taking the full Newton step $p = -[\nabla^2 f(x)]^{-1}\nabla f(x)$, the right-hand side equals $f(x) - \frac{1}{2}\lambda_N(x)^2$. Since $f(x^\star) \leq f(x+p)$, we get $f(x^\star) \leq f(x) - \frac{1}{2}\lambda_N^2$, i.e.\ $f(x) - f^\star \geq \frac{1}{2}\lambda_N^2$. Reversing: $f(x)-f^\star \leq \frac{1}{2}\lambda_N^2$ follows from the quadratic lower bound under strong convexity ($\mu$-strong convexity plus $L$-smoothness give two-sided quadratic approximation bounds).
\end{proof}

\noindent The Newton decrement $\lambda_N = \|[\nabla^2 f]^{-1}\nabla f\|_{\nabla^2 f}$ is the Hessian-norm of the Newton step. The damping strategy $\alpha = \min(1, c/\lambda_N)$ (Armijo-Newton) provides robustness in the large-step regime (when $\lambda_N \gg 1$, the iterate is far from the optimum and the full Newton step may be inaccurate).

For non-quadratic problems (Benchmarks B, C), the full Newton step ($\alpha=1$) is unstable far from the optimum because the second-order model is only accurate locally. The damped variant with $\alpha < 1$ sacrifices per-step progress for global convergence. A principled alternative is the \emph{Armijo backtracking line search}: initialise $\alpha = 1$ and multiply by $\tau \in (0,1)$ until the sufficient-decrease condition
\begin{equation}
f(x_k - \alpha p_k) \leq f(x_k) - c\,\alpha\,\nabla f(x_k)^T p_k, \qquad c \in (0,1),
\label{eq:armijo}
\end{equation}
is satisfied, where $p_k = H_k^{-1}g_k$ is the Newton direction. Since $-\nabla f^T p_k = -g_k^T H_k^{-1} g_k < 0$ (negative by PD of $H_k$), the condition~\eqref{eq:armijo} is always satisfiable for sufficiently small $\alpha$. This procedure guarantees a decrease proportional to the gradient norm at every step, enabling global convergence proofs. The assignment uses fixed $\alpha$ values (reported in Table~\ref{tab:q3_params}) instead of line search, which is simpler to implement but may require manual tuning per benchmark.

Settings: Hessian damping $\lambda_d = 10^{-8}$ throughout. GD runs for 80 iterations; Newton runs for 20 iterations.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}
% ============================================================

This question compares gradient-based methods that use gradient approximations (Part A: finite-difference GD and Nesterov random search on Benchmark B) with fully derivative-free methods that rely only on function values (Part B: Nelder--Mead simplex and grid search on Benchmark C). The central theme is the \emph{oracle complexity} trade-off: using fewer derivative evaluations per step reduces per-iteration cost but increases iteration count to compensate, with the optimal balance depending on the dimension $d$ and landscape smoothness.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces the $i$-th partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \qquad e_i = i\text{-th standard basis vector.}
\end{equation}
The total error has two components: a truncation error from the Taylor remainder,
\begin{equation}
\left|\frac{f(x+\delta e_i)-f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| \leq \frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(\xi)\right| = O(\delta),
\end{equation}
and a floating-point cancellation error of order $O(\varepsilon_{\mathrm{machine}}/\delta)$, where $\varepsilon_{\mathrm{machine}} \approx 10^{-16}$. The optimal forward-difference step minimising total error is $\delta^\star_{\mathrm{fwd}} \sim \sqrt{\varepsilon_{\mathrm{machine}}} \approx 10^{-8}$.

The \emph{central difference} formula $\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x+\delta e_i) - f(x-\delta e_i)}{2\delta}$ achieves $O(\delta^2)$ truncation error (the $O(\delta)$ terms cancel by symmetry) at the cost of one additional function evaluation per coordinate. The optimal central-difference step is $\delta^\star_{\mathrm{cen}} \sim \varepsilon_{\mathrm{machine}}^{1/3} \approx 10^{-5.3}$, yielding a minimum error of $O(\varepsilon_{\mathrm{machine}}^{2/3}) \approx 10^{-10.7}$, compared to $O(\varepsilon_{\mathrm{machine}}^{1/2}) \approx 10^{-8}$ for the forward difference. In practice, values of $\delta$ in $[10^{-5}, 10^{-2}]$ balance truncation and round-off errors for typical functions; the assignment tests $\delta = 0.05$ (good approximation) and $\delta = 0.8$ (truncation error dominates).

The assignment tests:
\begin{itemize}[nosep]
\item $\delta = 0.05$: truncation error $O(10^{-2})$ --- acceptable for Benchmark B's moderate curvature.
\item $\delta = 0.8$: truncation error $O(10^{-1})$ --- the step spans a region where second-order terms are significant, severely corrupting the gradient direction.
\end{itemize}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise $u_k \gets u_k / \|u_k\|$ (uniform on sphere)
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Stochastic directional gradient}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theory.} For the spherical uniform direction $u$ and small $\delta$, the expected update direction satisfies:
\begin{equation}
\mathbb{E}_{u}\!\left[\frac{f(x+\delta u)-f(x)}{\delta}\,u\right] \approx \frac{1}{d}\nabla f(x),
\end{equation}
via the Stokes/Stein spherical identity $\mathbb{E}_u[u_i \nabla_{x_i} f(x)] = (1/d)\nabla f(x)$. Thus the expected step is proportional to the true gradient, guaranteeing descent in expectation. The variance of $\hat{g}_k$ is $O(\|\nabla f(x)\|^2 d)$ per coordinate, compared to $0$ for exact gradients, explaining the significantly slower convergence.

The key advantage is requiring only \emph{two} function evaluations per iteration \emph{regardless of dimension $d$} --- compared to $d+1$ evaluations for forward finite-difference gradients ($d$ perturbations plus one base evaluation). For large $d$, this $d/2 \to 1$ evaluation ratio makes Nesterov random search attractive in extremely high-dimensional settings (e.g., reinforcement learning policy search). A two-point central-difference variant, $\hat{g}_k = (f(x+\delta u)-f(x-\delta u))/(2\delta) \cdot u$, reduces truncation error from $O(\delta)$ to $O(\delta^2)$ at the cost of one extra function evaluation. This variant is studied theoretically in [11], where it is shown to achieve an expected gradient error bound $O(\delta^2 \|\nabla^2 f\| + d/\delta \cdot \varepsilon_{\mathrm{noise}})$.

\subsection{Part B: Derivative-Free Methods on Benchmark C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): move the worst vertex $x_{n+1}$ through the centroid $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{n+1})$.
\item \textbf{Expansion} ($\gamma = 2.0$): if $f(x_r) < f(x_1)$, try $x_e = \bar{x} + \gamma(\bar{x} - x_{n+1})$.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection does not improve, contract towards centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all vertices towards the best vertex.
\end{enumerate}
No gradient information is used. Settings: initial simplex step $= 0.35$, iterations $= 160$.

\textbf{Convergence properties.} Nelder--Mead has no general convergence guarantee (McKinnon 1998 constructed examples where it stagnates), but works reliably in practice for smooth functions in low dimensions. Its key strength is \emph{adaptivity}: the simplex deforms to match the local landscape, elongating along valleys and contracting near optima. On Rosenbrock, the simplex elongates along the banana valley axis, effectively learning the problem's geometry without explicit Hessian computation. This distinguishes it from Nesterov random search, which uses independent random directions at each step and cannot learn the problem's geometry.

The central trade-off is between cost per iteration and information quality. Gradient-based methods (exact or approximate) achieve $O(d)$ iterations to a given accuracy using $O(d)$ FE each, for total cost $O(d^2/\varepsilon)$. Nesterov random search achieves gradient quality in expectation with only 2 FE, but requires $O(d/\varepsilon^2)$ iterations due to high variance, giving similar total cost. Grid search guarantees global optimality at $O((\Delta x)^{-d})$ cost, which is only tractable for $d \leq 3$.

\textbf{Practical guidance.} For $d \leq 10$: FD gradient descent with well-chosen $\delta$ is preferred when gradients are unavailable. For $d \in [10, 100]$: Nesterov random search or coordinate-wise methods are competitive. For $d > 100$: model-based methods (Bayesian optimisation, CMA-ES) typically outperform pure random/grid approaches. Grid search is only appropriate for hyperparameter tuning over small discrete grids, not continuous optimisation.

\textbf{Connection to course material.} Derivative-free methods are a form of \emph{zeroth-order optimisation}, as studied in lectures on randomised methods. The key distinction is the \emph{oracle model}:
\begin{itemize}[nosep]
\item \textbf{Zeroth-order oracle:} returns $f(x)$ for a query $x$. Cost: 1 FE.
\item \textbf{First-order oracle:} returns $(f(x), \nabla f(x))$. Cost: $d+1$ FE via FD, or $O(1)$ via automatic differentiation.
\item \textbf{Second-order oracle:} returns $(f(x), \nabla f(x), \nabla^2 f(x))$. Cost: $O(d^2)$ via FD or $O(d^2)$ Hessian computation.
\end{itemize}
The \emph{oracle complexity} (minimum oracle calls for $\varepsilon$-accuracy) scales as $O(1/\varepsilon)$ for first-order methods (strongly convex) and $O(d/\varepsilon)$ for zeroth-order methods, reflecting the $d$-fold information loss per query. This $d$ factor directly explains why Nesterov random search requires approximately $d = 2$ times more iterations than exact GD to achieve comparable accuracy on the 2D benchmarks.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

This question compares three approaches to constrained optimisation on Benchmark B with constraint $x_1 \geq 0.5$: (1) Projected GD (hard constraint via projection), (2) Penalty GD with varying $\lambda$ (soft constraint), and (3) Unconstrained GD as baseline. We begin with a KKT analysis showing the constraint is inactive at the optimum, then analyse the transient feasibility behaviour of each method.

\subsection{Problem Setup}

Minimise $f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)$ (Benchmark B) subject to $x_1 \geq 0.5$, starting from the \emph{infeasible} point $x_0 = (0.2, 4.0)$.

\textbf{KKT Conditions.} The constrained optimisation problem is:
\begin{equation}
\min_{x \in \mathbb{R}^2} f(x) \quad \text{subject to} \quad g(x) = 0.5 - x_1 \leq 0.
\end{equation}
The KKT stationarity conditions require the existence of a multiplier $\mu^\star \geq 0$ such that:
\begin{align}
\nabla f(x^\star) + \mu^\star \nabla g(x^\star) &= 0 \quad &\text{(stationarity)},\\
g(x^\star) &\leq 0 \quad &\text{(primal feasibility)},\\
\mu^\star &\geq 0 \quad &\text{(dual feasibility)},\\
\mu^\star g(x^\star) &= 0 \quad &\text{(complementary slackness)}.
\end{align}
With $\nabla g = [-1, 0]^T$: stationarity gives $\partial f/\partial x_1 = \mu^\star$ and $\partial f/\partial x_2 = 0$.

\textbf{Important observation (inactive constraint).} The unconstrained minimiser of Benchmark B satisfies $x_1^\star \approx 0.582 > 0.5$, so $g(x^\star) = 0.5 - 0.582 < 0$ (strictly feasible). By complementary slackness, $\mu^\star \cdot g(x^\star) = 0$ with $g(x^\star) \neq 0$ forces $\mu^\star = 0$. The constraint is \emph{inactive at the optimum}: the unconstrained KKT conditions ($\nabla f = 0$) hold, and the constrained solution equals the unconstrained one. This means all methods that converge to the unconstrained minimum will automatically satisfy the constraint at convergence.

The constrained problem is non-trivial only in the \emph{transient phase}: the initial iterate $(0.2, 4.0)$ is infeasible, and unconstrained GD's trajectory passes through the infeasible region $x_1 < 0.5$ during early iterations (since $x_1$ increases monotonically from 0.2 towards 0.582 along the GD path). The constraint boundary $x_1 = 0.5$ is crossed at some intermediate iterate, after which all iterates are feasible. This setup cleanly illustrates the key difference between hard-constraint methods (Projected GD: never infeasible) and soft-constraint methods (Penalty: temporarily infeasible, converging to feasibility) for infeasible starting points.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ where $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1),\, z_2)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Gradient Descent with Penalty}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute penalised gradient: $\tilde{g}_k \gets \nabla_x \bigl[f(x_k) + \lambda \max(0,\, 0.5 - x_{k,1})\bigr]$
    \State $x_{k+1} \gets x_k - \alpha\, \tilde{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

For the linear penalty $P(x) = \max(0, 0.5 - x_1)$, the gradient is:
\begin{equation}
\frac{\partial P}{\partial x_1} = \begin{cases} -1 & \text{if } x_1 < 0.5 \\ 0 & \text{if } x_1 \geq 0.5 \end{cases}, \qquad \frac{\partial P}{\partial x_2} = 0.
\end{equation}
This is the subgradient of the hinge function at the constraint boundary.

Settings: Projected GD ($\alpha = 0.08$); Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$, $\alpha = 0.03$ for $\lambda = 4.5$); Unconstrained GD ($\alpha = 0.07$). All methods run 100 iterations.

\textbf{Projected GD} enforces feasibility as a \emph{hard} constraint at every iterate. The projection $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)$ is a $\max$ operation requiring $O(1)$ cost per coordinate --- negligible overhead. The theoretical convergence rate is the same as standard GD in the feasible domain (no degradation from projecting), because the projection onto a convex set is a non-expansive operator: $\|\Pi_\mathcal{X}(x)-\Pi_\mathcal{X}(y)\| \leq \|x-y\|$. The theoretical guarantee for projected GD on convex problems is identical to standard GD in the feasible domain: $O(1/k)$ convergence for convex $f$ or linear rate for strongly convex $f$.

\textbf{Penalty method} converts the constrained problem to an unconstrained one by adding $\lambda P(x)$ to the objective. As $\lambda \to \infty$, the penalised minimiser converges to the constrained minimiser (for this problem, both equal $(0.582, 2.0)$ since the constraint is inactive). The practical challenge is that large $\lambda$ requires a smaller step size for stability (because the penalty creates a steep gradient near the constraint boundary), illustrating the fundamental tension between exactness ($\lambda$ large) and ease of optimisation ($\lambda$ small). The sequential penalty approach (increasing $\lambda$ across runs) is commonly used in practice.

\textbf{Augmented Lagrangian (AL)} methods address the penalty conditioning problem by combining an explicit Lagrange multiplier estimate $\mu_k$ with a quadratic penalty:
\begin{equation}
\mathcal{L}_\rho(x, \mu) = f(x) - \mu (x_1 - 0.5) + \frac{\rho}{2}[\max(0,\, 0.5 - x_1)]^2,
\end{equation}
where $\mu \geq 0$ is the current dual estimate. After minimising $\mathcal{L}_\rho(\cdot, \mu_k)$ over $x$, the dual variable is updated: $\mu_{k+1} \leftarrow \max(0,\;\mu_k - \rho(x_{k,1} - 0.5))$. This multiplier update shifts the penalty term's origin so that exact feasibility is achievable at finite (moderate) $\rho$, avoiding the ill-conditioning that arises from taking $\lambda \to \infty$ in the plain penalty method. The AL approach is the foundation of ADMM (Alternating Direction Method of Multipliers) and is widely used in large-scale constrained optimisation [9].

\textbf{Connection to proximal gradient methods.} Projected GD can be viewed as a special case of the \emph{proximal gradient method}: minimise $f(x) + g(x)$ where $g(x) = \iota_\mathcal{X}(x)$ is the indicator function of the feasible set ($g(x) = 0$ if $x \in \mathcal{X}$, $g(x) = +\infty$ otherwise). The proximal operator of $g$ is exactly the projection: $\mathrm{prox}_{\alpha g}(z) = \arg\min_{x}[\alpha \iota_\mathcal{X}(x) + \frac{1}{2}\|x-z\|^2] = \Pi_\mathcal{X}(z)$. This unifies projected GD, ISTA/FISTA (where $g$ is the $\ell_1$ norm for sparsity), and other composite minimisation problems under a single algorithmic framework. The convergence theory of proximal gradient methods (which mirrors Nesterov acceleration from Section~\ref{sec:q2}) transfers directly to projected GD.

\section{Question 6: Linear Programmes and Frank--Wolfe Algorithm \hfill [30 marks]}
\label{sec:q6}
% ============================================================

This question covers three sub-problems: (I) the LP fundamental theorem applied to the feasible box $\mathcal{X}$; (II) the Frank--Wolfe (conditional gradient) algorithm for an interior optimum with two mixing parameters; and (III) a boundary optimum. The Frank--Wolfe gap (Proposition~\ref{prop:fw_gap}) is used as a computable convergence certificate throughout.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$ (a convex polytope/box).

\textbf{Linear Programme.} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$ with $a = [1, 2]^T$. Since $f$ is linear, its minimum over the convex polytope $\mathcal{X}$ is attained at a vertex (Fundamental Theorem of Linear Programming). Both components of $a$ are positive, so the minimum is attained at the lower bounds:
\begin{equation}
x^\star = (0.5,\; -5), \qquad f^\star = 0.5 + 2(-5) = -9.5.
\end{equation}

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q6_lp.pdf}
\caption{Q6-I: Linear Programme. The feasible region $\mathcal{X}$ (blue box) and objective level curves $f(x) = c$ (parallel lines with slope $-1/2$) are shown. The minimum is attained at the lower-left vertex $(0.5, -5)$, where the level curve achieves its smallest $c$ value within $\mathcal{X}$. This illustrates the LP fundamental theorem: a linear objective over a bounded convex polytope always attains its minimum at a vertex.}
\label{fig:q6_lp}
\end{figure}

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient) with Fixed Mixing Parameter}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\; \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Fixed-weight convex combination}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Implementation note.} The code uses a \emph{fixed} mixing parameter $\beta$, corresponding to a constant step size $\gamma = 1 - \beta$ towards the LP vertex $z_k$. This differs from the standard Frank--Wolfe schedule $\gamma_k = 2/(k+2)$ (which gives an $O(1/k)$ convergence guarantee). With a fixed $\beta$, convergence to the exact minimum is not guaranteed in general, but works well in practice for strongly convex functions with interior minima.

\textbf{LMO for box constraints.} The linear subproblem $\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ decomposes coordinate-wise:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}, \quad l_i, u_i = \text{lower/upper bounds of } \mathcal{X}_i.
\end{equation}
This requires $O(d)$ operations per step and no projection oracle, making Frank--Wolfe particularly attractive when projection is costly but LP subproblems are cheap.

\textbf{Convergence theory.} For a convex $L$-smooth function, standard Frank--Wolfe with $\gamma_k = 2/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{2LC_f}{k+2}, \qquad C_f = \max_{x,y \in \mathcal{X}} \bigl(f(x) - f(y) - \nabla f(y)^T(x-y)\bigr),
\end{equation}
where $C_f$ is the curvature constant of $f$ over $\mathcal{X}$. The following result formalises the Frank--Wolfe gap as a computable convergence certificate.

\begin{proposition}[FW Duality Gap as Convergence Certificate]
\label{prop:fw_gap}
Let $f : \mathcal{X} \to \mathbb{R}$ be convex and differentiable, $x_k \in \mathcal{X}$, and $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$. Define the Frank--Wolfe gap $g_k = \langle \nabla f(x_k),\, x_k - z_k \rangle$. Then $g_k \geq 0$ and
\[
g_k \;\geq\; f(x_k) - f(x^\star),
\]
i.e.\ $g_k$ upper-bounds the suboptimality without knowing $f^\star$.
\end{proposition}
\begin{proof}
By convexity of $f$: $f(x^\star) \geq f(x_k) + \nabla f(x_k)^T(x^\star - x_k)$, so $f(x_k) - f(x^\star) \leq \nabla f(x_k)^T(x_k - x^\star)$. Since $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ and $x^\star \in \mathcal{X}$, we have $\nabla f(x_k)^T z_k \leq \nabla f(x_k)^T x^\star$, so $\nabla f(x_k)^T(x_k - z_k) \geq \nabla f(x_k)^T(x_k - x^\star) \geq f(x_k) - f(x^\star) \geq 0$.
\end{proof}

\textbf{Numerical example (interior optimum, $\beta=0.90$, last iteration).} We illustrate Proposition~\ref{prop:fw_gap} concretely. At the final iterate $x_k \approx (1.066, 4.932)$: $\nabla f \approx [2 \times 0.066, 2 \times (-0.068)] = [0.132, -0.136]$. The LMO selects $z_k = (\mathrm{sign}(+0.132) \to l_1 = 0.5,\; \mathrm{sign}(-0.136) \to u_2 = 10)$, i.e.\ $z_k = (0.5, 10)$. The FW gap is $g_k = 0.132(1.066 - 0.5) + (-0.136)(4.932 - 10) = 0.132 \times 0.566 + 0.136 \times 5.068 = 0.0747 + 0.689 \approx 0.764$. The actual suboptimality $f(x_k) - f^\star = 0.0089 - 0 = 0.0089 \ll 0.764$: the gap is loose (overestimates suboptimality by $\approx 86\times$) because the iterates are oscillating near the interior optimum and $z_k$ jumps to a vertex far from $x_k$. This illustrates that $g_k$ is a valid upper bound (guaranteed by Proposition~\ref{prop:fw_gap}) but not a tight one near interior optima; for boundary optima, $z_k$ stays near $x^\star$ and the bound is typically much tighter.

\textbf{Interior optimum.} Frank--Wolfe with fixed $\beta$ exhibits persistent oscillation because $z_k$ alternates between vertices of $\mathcal{X}$ as the gradient direction fluctuates near the optimum. The convex combination gradually averages these jumps, but cannot reach the exact interior minimum under fixed-$\beta$. The standard decaying schedule $\gamma_k = 2/(k+2)$ (equivalently $\beta_k = k/(k+2)$, increasing towards 1) avoids this: steps become progressively smaller, reducing oscillation and guaranteeing $O(C_f/k)$ convergence. An alternative is exact line search $\gamma_k = \arg\min_{\gamma \in [0,1]} f(x_k + \gamma(z_k - x_k))$, which gives the optimal $\gamma_k = \min(1, (f(x_k)-f(z_k))/(L\|x_k-z_k\|^2))$ for $L$-smooth $f$ and achieves faster convergence in practice.

\textbf{Boundary optimum.} Frank--Wolfe is particularly well-suited to boundary optima. The LMO at each step selects a vertex of $\mathcal{X}$ near the true constrained minimum, and the iterate is progressively pulled towards that vertex. No explicit projection is needed at any point --- \emph{all iterates remain in $\mathcal{X}$} as convex combinations of feasible points. This projection-free property is the primary advantage over projected gradient descent when the feasible set is a polytope.

\textbf{Comparison with Projected GD.} Projected GD would converge to the boundary minimum exactly in this case at the theoretical $O(1/k)$ rate for strongly convex $f$, whereas Frank--Wolfe converges more slowly (algebraically at $O(1/k)$ without the strong-convexity speed-up). However, Frank--Wolfe scales better when the projection oracle is expensive and the LMO is cheap.

This ``atomic decomposition'' property is not shared by projected gradient methods.

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report investigated six families of optimisation algorithms on three progressively challenging benchmark functions. The key findings are:

\textbf{Adaptive step-size methods (Q1).} The Polyak step is optimal when $f^\star$ is known exactly, achieving the best Rosenbrock performance ($f = 0.009$), but catastrophically diverges when $f^\star$ is misspecified (Benchmarks A and B, where the true minima are $\approx 0.48$ and $\approx 0.72$ but $f^\star=0$ was provided). Adagrad and RMSprop are robust alternatives: Adagrad monotonically decays its effective step, guaranteeing convergence to the exact minimum; RMSprop's exponential forgetting causes a non-vanishing effective step near the minimum ($f=0.5027$ vs the true floor $0.4826$ on Benchmark A). Heavy Ball provides the best tradeoff without requiring $f^\star$, achieving $200\times$ faster asymptotic convergence exponent than GD at the theoretical level ($\rho^2 \approx 0.9233$ vs $0.9996$ per step for $\kappa=2510$), though the experimental gain of $2.5\times$ reflects the pre-asymptotic regime at 120 iterations.

\textbf{Momentum methods (Q2).} Nesterov acceleration achieves the theoretical $O(1/k^2)$ optimal rate on convex problems; on the non-convex Rosenbrock function, momentum-based acceleration still provides substantial empirical benefit ($7\times$ improvement over GD: $f=0.479$ vs $f=3.388$ at 150 iterations), but formal convergence guarantees apply only locally near the minimum. The $\sqrt{\kappa} \approx 50\times$ iteration-count advantage predicted by the strongly-convex theory explains the direction of the improvement. Adam combines momentum and adaptive scaling effectively for anisotropic problems (Benchmark B) but cannot handle the off-diagonal curvature of Rosenbrock's banana valley ($3.3\times$ worse than Nesterov). Mini-batch SGD illustrates the variance-speed trade-off: smaller batches converge faster in epochs but exhibit more oscillation; under high observation noise the irreducible noise floor $J^\star = \sigma^2/2$ dominates, equalising batch-size performance.

\textbf{Newton's method (Q3).} Newton converges in exactly one iteration on the quadratic benchmark (constant Hessian: $\theta^\star = H^{-1}b$ in one step), in $\sim$5 iterations on the near-quadratic Benchmark B, and substantially outperforms GD on Rosenbrock (20 damped steps reach $f = 0.686$ vs.\ GD's $f = 3.73$ in 80 steps --- noting that Newton uses 4$\times$ fewer iterations but still achieves $5.4\times$ better $f$-value, highlighting the power of curvature information). Quadratic convergence $\|e_{k+1}\| \leq C\|e_k\|^2$ applies only locally near the minimum; the Rosenbrock Newton final position $(0.183, 0.020)$ is still far from $(1,1)$, confirming pre-asymptotic operation. Damping $\alpha < 1$ is essential for stability far from the optimum; backtracking line search (Armijo condition, eq.~\eqref{eq:armijo}) is the principled alternative.

\textbf{Derivative-free methods (Q4).} Finite-difference quality depends critically on $\delta$: small $\delta$ accurately approximates the gradient, while large $\delta$ corrupts it. Nesterov random search is the most scalable zeroth-order method (2 FE per step) but converges slowly due to directional variance. Nelder--Mead and grid search are viable for low-dimensional problems, with grid search providing global guarantees at exponential cost.

\textbf{Constrained optimisation (Q5).} When the unconstrained optimum satisfies the constraint (as here), both projected GD and sufficiently penalised gradient descent converge to the same point. Projected GD enforces hard feasibility at every step; penalty methods provide soft, approximate enforcement with the penalty weight $\lambda$ controlling the trade-off between objective quality and constraint satisfaction.

\textbf{Frank--Wolfe (Q6).} The conditional gradient algorithm solves an LP subproblem per iteration, producing projection-free iterates that remain in $\mathcal{X}$ by construction. It converges to interior optima with characteristic $O(1/k)$ rate and oscillatory behaviour near the optimum. For boundary optima the LMO naturally selects the correct constraint-boundary vertex, making FW especially suited to polytope-constrained problems.

The Rosenbrock function ($\kappa \approx 2510$) is the most discriminating benchmark. Nesterov, Polyak (with exact $f^\star$), and Newton all dramatically outperform GD, while adaptive methods provide only moderate improvement. This confirms the principle that curvature exploitation is essential for ill-conditioned problems: second-order information (Newton), momentum with lookahead (Nesterov), or adaptive scaling (Adam) are all needed when $\kappa \gg 1$. For well-conditioned quadratics (Benchmark A), all methods converge rapidly and algorithm choice matters mainly for efficiency, not accuracy.

\textbf{Practical algorithm selection guide.}
\begin{itemize}[nosep]
\item \textbf{Smooth convex, $f^\star$ known exactly:} Polyak step (theoretically optimal).
\item \textbf{Smooth convex, well-conditioned ($\kappa \lesssim 10$):} GD with tuned $\alpha$, or Adam.
\item \textbf{Smooth convex, ill-conditioned ($\kappa \gg 1$):} Nesterov (optimal $O(1/k^2)$) or Newton (if Hessian affordable at $O(d^2)$ per step).
\item \textbf{Non-convex, deep learning:} Adam (robust, handles sparse gradients, scale-invariant).
\item \textbf{Large-scale stochastic:} Mini-batch SGD/Adam with decaying learning rate schedule.
\item \textbf{No gradients, $d \leq 10$:} Nelder--Mead or FD-GD with small $\delta$.
\item \textbf{No gradients, $d > 100$:} Nesterov random search (2 FE/iter) or evolutionary methods.
\item \textbf{Constrained, simple convex set:} Projected GD (exact feasibility, cheap projection).
\item \textbf{Constrained, complex polytope:} Frank--Wolfe (projection-free, produces sparse iterates).
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe with Fixed Mixing Parameter}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
