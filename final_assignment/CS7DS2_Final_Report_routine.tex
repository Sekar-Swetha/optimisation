\begin{abstract}
% ============================================================
This report presents a systematic comparative study of nine first-order and second-order
optimisation algorithms applied across three benchmark problems of increasing difficulty: a
quadratic linear regression loss (Benchmark A), a mildly nonlinear toy neural network loss
(Benchmark B), and the highly ill-conditioned Rosenbrock function (Benchmark C).

Question~1 investigates adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball),
demonstrating that the Polyak step size achieves optimal performance when the true minimum
$f^\star$ is known exactly but diverges otherwise, while Adagrad and Heavy Ball reliably
outperform constant-rate gradient descent.  Question~2 examines momentum and stochastic
methods (Nesterov acceleration, Adam, mini-batch SGD), finding that Nesterov achieves the
largest gain on Rosenbrock by exploiting its $O(1/k^2)$ convergence guarantee, and that
mini-batch noise reduces as batch size increases.  Question~3 analyses Newton's method and
second-order local approximations: Newton converges in a single step on the quadratic
benchmark and dramatically outperforms gradient descent on all benchmarks within 20
iterations.  Question~4 covers derivative-free and derivative-approximation methods; a
carefully chosen finite-difference step $\delta = 0.05$ closely matches exact gradient
descent, while Nelder--Mead successfully navigates Rosenbrock without any gradient
information.  Question~5 studies constrained optimisation via projected gradient descent and
penalty methods, establishing that since the unconstrained minimiser of Benchmark~B lies in
the interior of the feasible region, all methods converge to the same constrained optimum.
Finally, Question~6 analyses the Frank--Wolfe algorithm for linearly constrained problems,
comparing convergence rates and iterate behaviour for interior and boundary optima, and
identifying the characteristic zig-zag pattern arising from the LP subproblem.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for full
reproducibility. This section defines the three benchmark objective functions used
throughout the assignment, characterises their curvature properties, and specifies the
starting points.

% ------------------------------------------
\subsection{Benchmark A: Linear Regression Quadratic Loss}
% ------------------------------------------

A synthetic dataset with $m = 1000$ samples is generated as follows:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \sim \mathcal{N}(0, I_2), \quad
y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\label{eq:bench_A_data}
\end{equation}
The mean squared error loss is defined by
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
          = \frac{1}{2m}\|X\theta - y\|^2,
\label{eq:bench_A_loss}
\end{equation}
where $X \in \mathbb{R}^{m \times 2}$ has rows $x^{(i)T}$.  The gradient and Hessian are
\begin{equation}
\nabla J(\theta) = \frac{1}{m} X^T(X\theta - y), \qquad
H_A = \nabla^2 J(\theta) = \frac{1}{m} X^T X \quad (\text{constant}).
\end{equation}
By the law of large numbers, $H_A \to I_2$ as $m \to \infty$ for standard normal data, so the
condition number $\kappa(H_A) = \lambda_{\max}/\lambda_{\min} \approx 1.07$ (near-isotropic).
Because $\varepsilon^{(i)} \neq 0$, the minimum loss is $J^\star \approx 0.4826$, not zero.
Starting point: $\theta_0 = [0,\, 0]^T$.

% ------------------------------------------
\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
% ------------------------------------------

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\label{eq:bench_B_loss}
\end{equation}
The gradient and Hessian are
\begin{align}
\nabla f(x) &= \bigl[2(x_1 - 1) + \cos(x_1),\quad 10(x_2 - 2)\bigr]^T, \\
H_B(x)      &= \operatorname{diag}\bigl(2 - \sin(x_1),\; 10\bigr).
\end{align}
Since $\sin(x_1) \in [-1, 1]$, the $(1,1)$ eigenvalue satisfies $\lambda_1(x) = 2 - \sin(x_1)
\in [1, 3]$, while $\lambda_2 = 10$ everywhere.  The condition number therefore satisfies
$\kappa(H_B) \in [10/3,\, 10]$, creating \emph{moderate anisotropy}.

The unconstrained minimum satisfies $\nabla f = 0$, giving $x_2^\star = 2$ and $x_1^\star$
from $2(x_1 - 1) + \cos(x_1) = 0$.  Numerical root-finding yields $x_1^\star \approx 0.582$,
with $f^\star = f(0.582, 2.0) \approx 0.7244$.  Note: the code marks the optimum at
$(0.61, 2.0)$ in contour figures for visualisation purposes; the precise minimiser is
$(0.582, 2.0)$.  Starting point: $x_0 = [-1,\, 4]^T$.

% ------------------------------------------
\subsection{Benchmark C: Rosenbrock Function}
% ------------------------------------------

\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\label{eq:bench_C_loss}
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$.  The Hessian at the optimum is
\begin{equation}
H_C(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues approximately $1002$ and $0.5$, giving condition number $\kappa \approx 2000$.
This extreme ill-conditioning, combined with the narrow curved valley along $x_2 \approx x_1^2$,
makes Rosenbrock the most challenging of the three benchmarks.  Starting point:
$x_0 = [-1,\, 1]^T$.

% ------------------------------------------
\subsection{Gradient Descent Baseline}
% ------------------------------------------

The constant step-size gradient descent update is $x_{k+1} = x_k - \alpha \nabla f(x_k)$.
For an $L$-smooth $\mu$-strongly convex function, the optimal fixed step size is
$\alpha^* = 1/L$, giving convergence rate $1 - \mu/L = 1 - 1/\kappa$.
Larger steps risk divergence (stability requires $\alpha < 2/L$); smaller steps are
unnecessarily slow.  The per-benchmark GD step sizes are set to be stable and near-optimal
given the spectrum of each Hessian.

% ------------------------------------------
\subsection{Algorithm Formulations}
% ------------------------------------------

Four adaptive methods are compared against the GD baseline across all three benchmarks
for $N = 120$ iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require $x_0$, $f^{\star}$, $\varepsilon > 0$, $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
When $f^\star$ is known exactly, the Polyak step size can be shown to satisfy
$f(x_{k+1}) - f^\star \le C \bigl(f(x_k) - f^\star\bigr)^2$ in a neighbourhood of the
minimum, achieving superlinear convergence.  However, when $f^\star$ is underestimated (set
to $0$ while the true minimum is $f^\star_{\text{true}} > 0$), the numerator
$f(x_k) - 0 \to f^\star_{\text{true}} > 0$ while the denominator
$\|g_k\|^2 \to 0$ near the minimum, causing $\alpha_k \to \infty$ and inducing divergence.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha_0 > 0$, $\varepsilon > 0$, $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate sum of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
Adagrad's effective per-coordinate learning rate $\alpha_0/(\sqrt{G_k^{(i)}} + \varepsilon)$
is \emph{monotonically decreasing} in $k$ because $G_k^{(i)}$ accumulates all past squared
gradients.  Coordinates with historically large gradients receive smaller effective steps,
implementing automatic feature-frequency scaling.  The disadvantage is that the effective step
eventually becomes negligibly small, preventing continued adaptation.
\end{remark}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha_0 > 0$, $\beta \in (0,1)$, $\varepsilon > 0$, $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Adagrad vs RMSprop: memory length]
Adagrad is a special case of RMSprop with $\beta = 1$ (perfect memory: all past gradients
equally weighted).  By setting $\beta = 0.9$, RMSprop forgets old gradients with half-life
$\tau = -1/\log(0.9) \approx 9.5$ iterations: only the most recent $\sim$10 gradients
carry appreciable weight.  This limits the denominator's growth, preventing the effective
step from vanishing.  The stationary EMA value satisfies $v_\infty = \mathbb{E}[g^2]$,
so the effective step $\alpha_0/\sqrt{\mathbb{E}[g^2]}$ normalises each coordinate by its
RMS gradient magnitude --- a principled scale invariance.
\end{remark}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\beta \in (0,1)$, $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
Heavy Ball accumulates a velocity $z_k$ — a running average of past gradients weighted
exponentially by $\beta$.  For a quadratic loss with Hessian eigenvalues in $[\mu, L]$, the
optimal settings $\alpha = 4/((\sqrt{L} + \sqrt{\mu})^2)$ and
$\beta = ((\sqrt{L} - \sqrt{\mu})/(\sqrt{L} + \sqrt{\mu}))^2$ reduce the effective condition
number from $\kappa$ to $\sqrt{\kappa}$, doubling the asymptotic convergence rate relative to GD.
\end{remark}

\subsection{Comparative Analysis}

\paragraph{Theoretical convergence rates.}
For a $\mu$-strongly convex, $L$-smooth function, constant-step GD with $\alpha = 2/(\mu + L)$
satisfies
\begin{equation}
f(x_k) - f^\star \le \left(\frac{\kappa - 1}{\kappa + 1}\right)^{2k} \bigl(f(x_0) - f^\star\bigr),
\label{eq:gd_rate}
\end{equation}
where $\kappa = L/\mu$ is the condition number.  For near-isotropic Benchmark A
($\kappa \approx 1.07$), GD converges almost immediately: the ratio $((\kappa-1)/(\kappa+1))^2
\approx 0.001$, giving geometric decay with a very small base.  For Rosenbrock near the
minimum ($\kappa \approx 2000$), GD's ratio $\approx (1999/2001)^2 \approx 0.998$, requiring
roughly $\log(1/\varepsilon)/\log(1/0.998) \approx 1150$ iterations per decimal digit of
accuracy.  This explains why GD makes minimal progress on Rosenbrock in 120 iterations.

\begin{table}[H]
\centering
\caption{Q1: Theoretical GD convergence rate $((\kappa-1)/(\kappa+1))^2$ and iterations
needed to reduce the gap by $10^{-3}$, as a function of condition number $\kappa$.}
\label{tab:gd_convergence_rate}
\begin{tabular}{lccc}
\toprule
\textbf{Benchmark} & \textbf{$\kappa$ (approx.)} & \textbf{Contraction factor} & \textbf{Iters for $10^{-3}$ gap} \\
\midrule
A (Linear Reg.)  & 1.07  & 0.001   & $\approx 3$   \\
B (Toy NN)       & 3--10 & 0.11--0.67 & 6--20      \\
C (Rosenbrock)   & 2000  & 0.998   & $\approx 3450$ \\
\bottomrule
\end{tabular}
\end{table}

Heavy Ball and Nesterov momentum address this precisely: on quadratics with condition
number $\kappa$, optimal momentum achieves an effective condition number of $\sqrt{\kappa}$,
reducing the required iterations by a factor of $\sqrt{\kappa}$.  For Rosenbrock
($\kappa \approx 2000$), this theoretical speedup is $\sqrt{2000} \approx 45\times$,
reducing 3450 GD iterations to $\sim$75 iterations for the same accuracy --- consistent
with Heavy Ball achieving $f = 1.40$ versus GD's $f = 3.51$ in 120 iterations.

\begin{itemize}[nosep]
\item \textbf{Convergence speed:}
  When $f^\star$ is known exactly (Benchmark C), Polyak converges fastest by far.
  Otherwise, Heavy Ball provides the greatest benefit over GD through its momentum
  accumulation, particularly on the ill-conditioned Rosenbrock function.
  Adagrad and GD reach comparable final values on Benchmarks A and B.

\item \textbf{Smoothness:}
  RMSprop produces the smoothest convergence curves due to its exponential moving average,
  which filters gradient noise.  Heavy Ball exhibits characteristic overshoot oscillations
  before damping.

\item \textbf{Stability:}
  Constant-rate GD is the most stable across all settings; Polyak is \emph{unstable} when
  $f^\star$ is misspecified.  Adagrad is uniformly stable (monotone step decay prevents
  instability) but may stagnate.  RMSprop requires careful $\alpha_0$ tuning; too large a
  rate can cause oscillations near the minimum.

\item \textbf{Practical implication:}
  In practice, exact $f^\star$ is unknown, ruling out Polyak for general use.  RMSprop and
  Adagrad are robust default choices.  Heavy Ball is effective when momentum can be tuned
  to the curvature ratio of the problem.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods}
\label{sec:q2}
\noindent\textbf{[30 marks]}
% ============================================================

% ------------------------------------------
\subsection{Algorithm Formulations}
% ------------------------------------------

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\beta_{\max} \in (0,1)$, $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\bigl((k-1)/(k+2),\; \beta_{\max}\bigr)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Nesterov vs Heavy Ball]
The critical difference between Nesterov and Heavy Ball is the \emph{lookahead evaluation}
$\tilde{x}_k = x_k + \beta_k z_k$: the gradient is computed at the \emph{predicted} next
position rather than the current one.  This correction allows Nesterov to ``look before it
leaps,'' reducing overshoot compared to Heavy Ball.  On convex functions,
Nesterov provably achieves $O(1/k^2)$ convergence (Theorem~\ref{thm:nesterov}) versus GD's
$O(1/k)$, halving the exponent.  The adaptive schedule
$\beta_k = (k-1)/(k+2) \nearrow \beta_{\max}$ increases momentum gradually, ensuring
stability in early iterations when the iterate may still be far from the optimum.
\end{remark}

\begin{theorem}[Nesterov $O(1/k^2)$ rate]
\label{thm:nesterov}
For an $L$-smooth convex function $f$ and step size $\alpha = 1/L$, Nesterov acceleration
satisfies
\[
  f(x_k) - f^\star \leq \frac{2 L \|x_0 - x^\star\|^2}{(k+1)^2}.
\]
This matches the optimal convergence rate for first-order methods on this function class
(by the lower bound of Nemirovsky and Yudin, 1983).  Contrast with GD's $O(1/k)$ rate:
to achieve accuracy $\varepsilon$, GD requires $O(1/\varepsilon)$ iterations, while
Nesterov requires only $O(1/\sqrt{\varepsilon})$ --- a quadratic improvement.
\end{theorem}

\begin{remark}[Role of the adaptive $\beta_k$ schedule]
The schedule $\beta_k = (k-1)/(k+2)$ is chosen to match the momentum required by the
$O(1/k^2)$ rate: it increases from $\beta_1 = 0$ to its asymptote $\beta_k \to 1$,
building momentum gradually.  Starting with zero momentum prevents overshoot early when
the iterate may be far from the minimum; the growing momentum then accelerates convergence
as the iterate enters the ``fast phase'' near the optimum.  The cap $\beta_{\max}$ prevents
instability on nonconvex landscapes like Rosenbrock, where applying full momentum
$\beta_k \to 1$ could cause the iterate to escape the valley.
\end{remark}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\beta_1, \beta_2 \in (0,1)$, $\varepsilon > 0$, $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$
        \Comment{Biased first moment estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$
        \Comment{Biased second moment estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$
        \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Bias correction in Adam]
At iteration $t$, the exponential moving average satisfies $\mathbb{E}[m_t] = (1 - \beta_1^t)
\mathbb{E}[g]$ --- a biased underestimate of the true mean gradient.  Without correction,
early iterates would use a step artificially scaled by $(1 - \beta_1^t)$: at $t = 1$ with
$\beta_1 = 0.82$, the uncorrected effective step would be $(1 - 0.82)\alpha = 0.18\alpha$
instead of $\alpha$.  The bias correction $\hat{m}_t = m_t/(1 - \beta_1^t)$ eliminates this
distortion.  Adam thus combines (i) RMSprop-style per-coordinate scaling with (ii)
heavy-ball-style momentum, while ensuring effective step sizes are consistent across
all iterations.  As $t \to \infty$, both correction factors approach 1 and Adam reduces
to a momentum-augmented version of RMSprop, with the bias correction becoming negligible.
\end{remark}

\begin{remark}[Why Nesterov outperforms Adam on Rosenbrock]

dampening the effective step precisely in directions of large curvature --- counteracting
the momentum effect.  Nesterov, by contrast, builds consistent directional velocity without
any coordinate-wise normalisation, making it far more effective at traversing the narrow,
curved Rosenbrock valley.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent (Benchmark A)}
% ------------------------------------------

Mini-batch SGD replaces the full gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$
with a stochastic estimate over a mini-batch $\mathcal{B}$ of size $b$:
\begin{equation}
g_k^{(\mathcal{B})} = \frac{1}{b} X_\mathcal{B}^T (X_\mathcal{B}\theta - y_\mathcal{B}).
\end{equation}
This estimate is \emph{unbiased}: $\mathbb{E}_{\mathcal{B}}[g_k^{(\mathcal{B})}] = \nabla J(\theta)$,
but has variance
\begin{equation}
\mathrm{Var}\bigl[g_k^{(\mathcal{B})}\bigr] = \frac{1}{b}\,\mathrm{Var}\bigl[g^{(\text{single})}\bigr]
\propto \frac{\sigma^2_{\varepsilon}}{b},
\label{eq:sgd_variance}
\end{equation}
where $\sigma^2_\varepsilon$ is the noise variance of the data.  This variance is the dominant
source of stochastic error: a fixed step-size $\alpha$ cannot eliminate the gradient noise,
so SGD exhibits a \emph{noise floor} $J_{\text{floor}} \approx J^\star + O(\alpha \sigma^2/b)$
below which it cannot converge.  Reducing $\alpha$ lowers the floor but slows progress.
Settings: $\alpha = 0.06$, $b \in \{5, 40\}$, 50 epochs, with data shuffled each epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require $(X, y)$, $\theta_0$, $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Permute data indices randomly
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets b^{-1} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\centering
\includegraphics[width=0.88\textwidth]{figures/q2_sgd_batch.pdf}

% ------------------------------------------
\subsection{Local Approximations of $g(x) = x^4$}
% ------------------------------------------

Consider the 1-D test function $g(x) = x^4$ expanded around $x_0 = 0.25$.
The derivatives are $g'(x) = 4x^3$, $g''(x) = 12x^2$, giving $g(0.25) = 0.0039$,
$g'(0.25) = 0.0625$, $g''(0.25) = 0.75$.

\textbf{First-order (linear) approximation:}
\begin{equation}
\ell(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625\,(x - 0.25).
\label{eq:taylor1}
\end{equation}
This tangent line approximation captures the local slope but deviates
rapidly from $g(x)$ as $|x - x_0|$ grows.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
q(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
      = 0.0039 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\label{eq:taylor2}
\end{equation}
Minimising $q(x)$ over $x$ by setting $q'(x) = 0$ yields the Newton update:
\begin{equation}
x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)}.
\end{equation}
This is the Newton step: it steps to the minimiser of the local quadratic model, which is
a much better approximation to the true minimiser than the gradient step alone.

\begin{figure}[H]
\centering
\includegraphics[width=0.88\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$.  The tangent line
(Eq.~\ref{eq:taylor1}) has no minimum and diverges from the true function rapidly.  The
quadratic approximation (Eq.~\ref{eq:taylor2}) fits the function well near $x_0$ and
has a minimum at $x^\star \approx 0.25 - 0.0625/0.75 \approx 0.167$, which serves as
the Newton update from $x_0 = 0.25$.  The curvature captured by the second-order term
is essential for guiding the optimisation.}
\label{fig:q3_approx}
\end{figure}

% ------------------------------------------
\subsection{Newton's Method in $\mathbb{R}^n$}
% ------------------------------------------

The 1-D derivation generalises directly to $n$ dimensions.  The second-order Taylor
expansion of $f: \mathbb{R}^n \to \mathbb{R}$ around $x_k$ is
\begin{equation}
f(x) \approx q_k(x) = f(x_k) + \nabla f(x_k)^T(x - x_k) + \tfrac{1}{2}(x - x_k)^T
\nabla^2 f(x_k)(x - x_k).
\end{equation}
Setting $\nabla q_k(x) = 0$ gives the Newton step:
\begin{equation}
x_{k+1} = x_k - \bigl[\nabla^2 f(x_k)\bigr]^{-1} \nabla f(x_k).
\end{equation}
Compared to GD which takes a step of length $\alpha$ in the negative gradient direction,
Newton premultiplies the gradient by the \emph{inverse Hessian}: this rescales each
dimension by the inverse curvature, accounting for the anisotropy of the landscape.
For an isotropic landscape ($\nabla^2 f = cI$), Newton reduces to GD with $\alpha = 1/c$.

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require $x_0$, step size $\alpha$, Hessian damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1} g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Quadratic convergence]
For a twice-continuously-differentiable function $f$ whose Hessian is $M$-Lipschitz
continuous (i.e.\ $\|\nabla^2 f(x) - \nabla^2 f(y)\| \le M\|x - y\|$), Newton's
method with full step ($\alpha = 1$) satisfies
\[
  \|x_{k+1} - x^\star\| \le \frac{M}{2\mu}\|x_k - x^\star\|^2,
\]
where $\mu > 0$ is the strong-convexity constant of $f$.  This \emph{quadratic convergence}
means the number of correct decimal digits doubles at each iteration once
$\|x_k - x^\star\|$ is sufficiently small.  The damping ($\alpha < 1$) sacrifices the
locally quadratic rate in exchange for global convergence when the iterate is far from
$x^\star$ or the Hessian is ill-conditioned.
\end{remark}

\paragraph{One-step convergence on quadratic functions.}
For the quadratic loss $J(\theta) = \tfrac{1}{2m}\|X\theta - y\|^2$, the gradient is
$\nabla J(\theta) = H_A(\theta - \theta^{\star}_{\text{OLS}})$ where $H_A = X^TX/m$ and
$\theta^{\star}_{\text{OLS}} = (X^TX)^{-1}X^Ty$ is the least-squares solution.
The Newton update with $\alpha = 1$ gives:
\begin{equation}
\theta_{k+1} = \theta_k - H_A^{-1}\nabla J(\theta_k) = \theta_k - H_A^{-1} H_A(\theta_k - \theta^{\star}_{\text{OLS}}) = \theta^{\star}_{\text{OLS}}.
\end{equation}
Thus Newton's method converges \emph{exactly} in one step for any quadratic, independent of
the starting point $\theta_0$ and condition number $\kappa(H_A)$.  This is a fundamental
property: quadratics are ``zeroth-degree residuals'' for Newton's method.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation}
\label{sec:q4}
\noindent\textbf{[30 marks]}
% ============================================================

% ------------------------------------------
\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}
% ------------------------------------------

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation for each partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\label{eq:fd}
\end{equation}
where $e_i$ is the $i$-th standard basis vector.

\textbf{Error analysis:} By Taylor's theorem,
$f(x + \delta e_i) = f(x) + \delta \partial_i f(x) + \tfrac{\delta^2}{2}\partial_i^2 f(x) + O(\delta^3)$.
Substituting into Eq.~\eqref{eq:fd} gives
\[
  \frac{f(x + \delta e_i) - f(x)}{\delta} = \partial_i f(x) + \underbrace{\tfrac{\delta}{2}\partial_i^2 f(x)}_{\text{truncation error }O(\delta)}.
\]
Additionally, finite-precision arithmetic introduces rounding error $O(\varepsilon_{\text{mach}}/\delta)$
(where $\varepsilon_{\text{mach}} \approx 10^{-16}$ for double precision).  The total error is
\[
  E(\delta) = C_1 \delta + C_2 \varepsilon_{\text{mach}} / \delta,
\]
minimised at $\delta_{\text{opt}} = \sqrt{C_2/C_1 \cdot \varepsilon_{\text{mach}}} \approx 10^{-8}$.
The assignment tests $\delta = 0.05$ (good: truncation-dominated but small) and
$\delta = 0.8$ (poor: large truncation error).

Each gradient evaluation in $\mathbb{R}^2$ requires $d + 1 = 3$ function evaluations.
Note that the \emph{central difference} approximation
$\partial_i f(x) \approx (f(x + \delta e_i) - f(x - \delta e_i))/(2\delta)$ achieves
$O(\delta^2)$ truncation error (instead of $O(\delta)$) at the cost of $2d + 1$
evaluations; the optimal $\delta_{\text{central}} \approx \varepsilon_{\text{mach}}^{1/3}
\approx 10^{-5}$, giving much higher accuracy for the same $\delta$.  The assignment
uses forward differences for simplicity.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\delta > 0$, $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$; normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
        \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The estimate $\hat{g}_k$ satisfies $\mathbb{E}[\hat{g}_k] = \nabla f_\delta(x_k)$, where
$f_\delta$ is a Gaussian-smoothed approximation to $f$.  Only \textbf{two} function
evaluations are needed per iteration regardless of dimension, making this method
dimension-independent in cost-per-step.  The trade-off is slow convergence: only
$O(1/\sqrt{k})$ progress is achievable in the stochastic sense.

\subsubsection{Nelder--Mead Simplex Method}

Nelder--Mead maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$, iteratively
replacing the worst vertex through four geometric operations applied in order:

\begin{enumerate}[nosep]
\item \textbf{Reflection:} Reflect the worst vertex $x_w$ through the centroid $\bar{x}$ of the remaining vertices:
  $x_r = \bar{x} + \alpha_r(\bar{x} - x_w)$, $\alpha_r = 1$.
\item \textbf{Expansion:} If $x_r$ is the new best, expand further:
  $x_e = \bar{x} + \gamma(x_r - \bar{x})$, $\gamma = 2$.
\item \textbf{Contraction (outside/inside):} If $x_r$ does not improve sufficiently,
  contract toward the centroid or the worst vertex: $x_c = \bar{x} + \rho(x_r - \bar{x})$,
  $\rho = 0.5$.
\item \textbf{Shrink:} If contraction fails, shrink all vertices toward the best:
  $x_i \gets x_1 + \sigma(x_i - x_1)$, $\sigma = 0.5$.
\end{enumerate}

No gradient information is required; each iteration requires at most $d + 1 = 3$ function
evaluations.  Settings: initial simplex step $= 0.35$, iterations $= 160$.

Nelder--Mead achieves $f \approx 10^{-3}$ on Rosenbrock in 160 simplex iterations

effectively implementing an implicit curvature adaptation that gradient descent cannot.
However, Nelder--Mead lacks theoretical convergence guarantees for $d > 1$; its success
here relies on the smooth, unimodal landscape of Rosenbrock.
\end{remark}

\item \textbf{When to use finite differences:} If gradients are costly or unavailable but

% ------------------------------------------
\subsection{Problem Setup and Key Observation}
% ------------------------------------------

Minimise Benchmark B subject to the inequality constraint $x_1 \geq 0.5$, starting from
the infeasible point $x_0 = (0.2,\, 4.0)$.

\begin{remark}[Constraint activity]
\label{rem:inactive}
The unconstrained minimiser of Benchmark B is at $x^\star \approx (0.582,\, 2.0)$.  Since
$0.582 > 0.5$, this point is \emph{feasible}, and the constraint $x_1 \geq 0.5$ is
\textbf{inactive} at the optimum.  Consequently, the constrained and unconstrained minima
coincide, and any feasibility-enforcing method should ultimately converge to the same
point $(0.582, 2.0)$.  The key differences between methods lie in:
(a) how quickly they achieve feasibility from the infeasible start $(0.2, 4.0)$, and
(b) whether the path respects the feasible region throughout optimisation.
\end{remark}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}\!\bigl(x_k - \alpha\, \nabla f(x_k)\bigr)$
        \Comment{$\Pi_\mathcal{X}$: project to $x_1 \ge 0.5$, i.e.\ $x_1 \leftarrow \max(0.5, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require $x_0$, $\alpha > 0$, $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_\lambda(x_k) \gets f(x_k) + \lambda\,\max(0,\, 0.5 - x_1^{(k)})$
    \State $\nabla F_\lambda(x_k) \gets \nabla f(x_k) + \lambda [-\mathbf{1}\{x_1 < 0.5\}, 0]^T$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\paragraph{KKT optimality conditions.}
The constrained problem $\min_{x_1 \geq 0.5} f_B(x)$ has the Lagrangian
\[
  \mathcal{L}(x, \mu) = f_B(x) - \mu(x_1 - 0.5), \quad \mu \geq 0.
\]
The KKT conditions for a local minimum $x^\star$ are:
\begin{align*}
\text{(Stationarity):}\quad & \nabla f_B(x^\star) = \mu e_1, \\
\text{(Primal feasibility):}\quad & x_1^\star \geq 0.5, \\
\text{(Dual feasibility):}\quad & \mu \geq 0, \\
\text{(Complementary slackness):}\quad & \mu(x_1^\star - 0.5) = 0.
\end{align*}
Since $x_1^\star \approx 0.582 > 0.5$, complementary slackness forces $\mu = 0$, reducing
the stationarity condition to $\nabla f_B(x^\star) = 0$ --- the unconstrained optimality
condition.  This confirms Remark~\ref{rem:inactive}: the constraint is inactive and the
constrained solution equals the unconstrained one.

Settings: Projected GD ($\alpha = 0.08$); Penalty with $\lambda \in \{0.15,\, 1.8,\, 4.5\}$
($\alpha = 0.05$ for $\lambda \le 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$);
unconstrained GD ($\alpha = 0.07$).  All run for 100 iterations.

\paragraph{Projected GD analysis.}
Projected GD enforces feasibility as a hard constraint at every iteration via the
projection oracle $\Pi_\mathcal{X}(x): x_1 \leftarrow \max(0.5, x_1)$.  For simple box
constraints, the projection is a coordinate-wise clamp, requiring $O(d)$ operations.
For $\mu$-strongly convex $f$ with $L$-smooth gradient, projected GD with $\alpha = 1/L$
satisfies the same linear convergence rate as unconstrained GD when the optimal point is
in the interior:
\[
  \|x_{k+1} - x^\star\| \le (1 - \mu/L)\|x_k - x^\star\|.
\]
Since the constraint is inactive here, projected GD effectively reduces to standard GD
once the iterate reaches the feasible region (after one step from $(0.2, 4.0)$).

\paragraph{Penalty method analysis.}
The penalised objective $F_\lambda(x) = f(x) + \lambda \max(0, 0.5 - x_1)$ can be
viewed as an exact $\ell^1$ penalty.  The function $F_\lambda$ is not differentiable at
$x_1 = 0.5$; the subgradient $\partial F_\lambda / \partial x_1 = \partial f/\partial x_1
- \lambda \cdot \mathbf{1}\{x_1 < 0.5\}$ is used in the code.  As $\lambda$ increases:
(a) the penalty gradient $-\lambda$ at $x_1 < 0.5$ drives the iterate toward feasibility
more aggressively, and (b) the penalised Hessian near $x_1 = 0.5$ becomes effectively
larger, requiring a smaller step size to maintain convergence.  In the limit $\lambda \to
\infty$, the solution to $\min_x F_\lambda(x)$ converges to the constrained solution
--- here $(0.582, 2.0)$ since the constraint is inactive.  For active constraints, larger
$\lambda$ would be needed to achieve near-feasibility.

\subsection{Feasible Set and Linear Programme}
% ------------------------------------------

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$
(a bounded box, a polytope with four vertices: $(0.5,-5)$, $(5,-5)$, $(5,10)$, $(0.5,10)$).

\textbf{Linear Programme (Q6-I):} For $f(x) = a^T x$ with $a = [1,\, 2]^T$, since $a_i > 0$
for both coordinates, $f$ is minimised at the lower bound of each coordinate independently:
\begin{equation}
x^\star_{\text{LP}} = (0.5,\; -5), \qquad f^\star = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates the \emph{fundamental theorem of linear programming}: for a linear objective
over a bounded polytope, the minimum is attained at a vertex.

% ------------------------------------------
\subsection{Frank--Wolfe Algorithm}
% ------------------------------------------

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$
        \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
        \Comment{Convex combination step}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[LMO for box constraints]
For the box $\mathcal{X}$, the LP subproblem decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
This requires $O(d)$ operations per iteration.  Crucially, Frank--Wolfe
\textbf{requires no projection oracle}: iterates are always convex combinations of
elements of $\mathcal{X}$, hence automatically feasible.
\end{remark}

\begin{remark}[Frank--Wolfe gap and convergence rate]
Define the \emph{Frank--Wolfe gap} at $x_k$ as
\[
  g(x_k) = \max_{z \in \mathcal{X}} \nabla f(x_k)^T(x_k - z) = \nabla f(x_k)^T(x_k - z_k).
\]
This is an upper bound on the suboptimality: $f(x_k) - f^\star \le g(x_k)$.  For a smooth
$L$-Lipschitz-gradient convex function and the standard $\gamma_k = 2/(k+2)$ step schedule,
the Frank--Wolfe algorithm satisfies
\[
  f(x_k) - f^\star \le \frac{2LC^2}{k+2},
\]
where $C = \max_{x,z \in \mathcal{X}}\|x - z\|$ is the diameter of $\mathcal{X}$.  This
is $O(1/k)$ --- slower than GD's $O(1/k)$ dependence on problem parameters, but the
method avoids the potentially expensive projection step.  The fixed-$\beta$ schedule used
here does not provide this theoretical guarantee but simplifies the implementation.
\end{remark}

\subsection{Interior vs Boundary Comparison}
% ------------------------------------------

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits the characteristic zig-zag
because the LMO selects extreme vertices of $\mathcal{X}$ that may lie far from the current
iterate.  As $x_k$ approaches the interior optimum, the gradient $\nabla f(x_k) \to 0$
and the LMO solution $z_k$ becomes sensitive to rounding errors in the gradient sign,
producing oscillation.  The convex combination with large $\beta$ damps this, but
at the cost of smaller effective steps early on.  The $O(1/k)$ convergence rate is
fundamentally slower than projected-gradient methods ($O(1/k)$ to $O(1/k^2)$ with
momentum) but avoids any projection computation.

\textbf{Boundary optimum:} Frank--Wolfe is naturally suited to boundary optima because
the LMO consistently returns the boundary vertex in the descent direction.  The iterates
$x_k$ approach the boundary monotonically (in the $x_1$ component) without requiring an
explicit projection.  The residual gap $f(x_{140}) - f^\star \approx 0.11$ reflects
that the fixed-$\beta$ schedule does not decay, so the algorithm never takes the final
fine-grained steps needed to converge exactly to $(0.5, 0)$.  A diminishing step
schedule $\gamma_k = 2/(k+2)$ (standard Frank--Wolfe) would asymptotically close this gap.

\paragraph{Frank--Wolfe gap as a stopping criterion.}
A practically important feature of Frank--Wolfe is that the FW gap $g(x_k) = \nabla f(x_k)^T
(x_k - z_k)$ provides a computable \emph{upper bound} on the suboptimality:
$f(x_k) - f^\star \le g(x_k)$.  This bound requires no knowledge of $f^\star$ and can be
computed from existing quantities at each step.  For the interior case with $\beta = 0.90$
and 180 iterations, we compute the FW gap from the final gradient and LP vertex to certify
suboptimality.  This is a key advantage of Frank--Wolfe over projected gradient methods,
which do not naturally yield such a certificate.

\paragraph{Comparison summary.}
\begin{itemize}[nosep]
\item \textbf{When the optimum is interior:} Frank--Wolfe converges at $O(1/k)$ but may
  oscillate near the minimum due to LP vertex jumping.  Projected gradient methods with
  momentum (Nesterov) achieve $O(1/k^2)$ and are preferred if projection is cheap.
\item \textbf{When the optimum is on the boundary:} Frank--Wolfe naturally approaches the
  boundary without projection, making it competitive.  The fixed-$\beta$ schedule used
  here introduces residual oscillation; the standard $\gamma_k = 2/(k+2)$ decay eliminates
  this at the cost of smaller early steps.
\item \textbf{Constraint structure:} For general convex constraints (e.g.\ nuclear norm balls,
  positive semidefinite cones), the FW LMO may be computationally \emph{cheaper} than
  projection, making Frank--Wolfe the preferred choice.
\end{itemize}

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has systematically applied nine categories of optimisation algorithms to three
benchmark problems of increasing difficulty, drawing the following principal conclusions:

\begin{enumerate}[nosep]
\item \textbf{Benchmark sensitivity of adaptive methods (Q1):} Polyak's step size is
  optimal when $f^\star$ is exactly known but catastrophically fails when it is not.
  Adagrad and Heavy Ball are robust alternatives that outperform constant-rate GD on all
  benchmarks.  RMSprop's exponential forgetting mechanism prevents the dying step-size
  problem of Adagrad, making it preferable for long training runs.

\item \textbf{Momentum and the Rosenbrock challenge (Q2):} Nesterov acceleration provides
  the most significant improvement over GD on the ill-conditioned Rosenbrock function,
  directly reflecting its $O(1/k^2)$ convergence guarantee.  Adam's per-coordinate scaling
  is less effective on this problem because the dominant challenge is global momentum
  along the curved valley, not anisotropy.

\item \textbf{Newton's superiority on near-quadratic problems (Q3):} Newton's method
  converges in a single step on the quadratic Benchmark A and in just a few steps on
  Benchmark B.  Even with aggressive damping ($\alpha = 0.22$) on Rosenbrock, Newton
  outperforms GD by $5\times$ within a fraction of the iterations, at the cost of
  requiring Hessian computation.

\item \textbf{Gradient approximation quality matters (Q4):} Finite differences with a
  well-chosen $\delta$ can closely approximate exact GD at the cost of $d$ additional
  function evaluations per step.  Nelder--Mead demonstrates that gradient-free methods
  can navigate complex landscapes, but are limited to low dimensions.

\item \textbf{Constraint inactivity simplifies the constrained problem (Q5):} When the
  unconstrained optimum is feasible (as here for Benchmark B), all feasibility-enforcing
  methods converge to the same constrained minimum.  Projected GD enforces hard feasibility
  at every step; penalty methods trade exactness for implementation simplicity.

\item \textbf{Frank--Wolfe is projection-free but converges slowly (Q6):} By solving
  a linear subproblem at each step, Frank--Wolfe maintains feasibility without projection.
  The characteristic zig-zag of LP vertex jumps limits the convergence rate to $O(1/k)$,
  making it competitive primarily in settings where projections are expensive.
\end{enumerate}

Across all experiments, the Rosenbrock function consistently proved most challenging,
illustrating how ill-conditioning ($\kappa \approx 2000$) fundamentally limits first-order
methods and motivates second-order or momentum-based approaches.

noise or unknown curvature structure.

\paragraph{Practical guidelines.}
\begin{itemize}[nosep]
\item Use \textbf{Newton's method} when $d$ is small, Hessians are computable, and the
  function is smooth (e.g.\ linear regression, logistic regression).
\item Use \textbf{Nesterov or Adam} for large-scale first-order settings, preferring Nesterov
  when the objective has consistent curvature direction, and Adam when gradients are noisy
  or sparse.
\item Use \textbf{projected gradient descent} when constraints are simple (box, simplex) and
  the projection is cheap.  Penalty methods are a viable alternative when only approximate
  feasibility is acceptable.
\item Use \textbf{Frank--Wolfe} when the feasible set has a complex projection but an
  efficient LP oracle (e.g.\ nuclear norm balls, polytopes with special structure).
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak step size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum (lookahead)}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z          # extrapolation point
        g = grad_fn(lookahead)              # gradient at lookahead
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)     # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's method with Hessian damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))   # Tikhonov regularisation
        try:
            p = np.linalg.solve(H_reg, g)       # Newton direction
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (conditional gradient)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LMO: box linear minimisation (per-coordinate)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z      # convex combination step
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
