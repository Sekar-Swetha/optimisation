\begin{abstract}
This report presents a comprehensive comparative study of six families of optimisation algorithms applied to three canonical benchmark problems: a linear regression quadratic loss (Benchmark~A, condition number $\kappa \approx 1.12$), a toy neural network loss with sinusoidal perturbation (Benchmark~B, $\kappa \in [3.5, 6.9]$), and the Rosenbrock function (Benchmark~C, $\kappa \approx 2508$ at the optimum). The investigations encompass: (1)~adaptive step-size methods --- Polyak, Adagrad, RMSprop, and Heavy Ball; (2)~momentum and stochastic methods --- Nesterov acceleration, Adam, and mini-batch SGD; (3)~Newton's method with local second-order approximation; (4)~derivative-free approaches including finite differences, Nesterov random search, Nelder--Mead, and grid search; (5)~constrained optimisation via projected gradient descent and penalty methods; and (6)~the Frank--Wolfe conditional gradient algorithm for linear-constrained problems.

Key findings are: the Polyak step size achieves the best first-order convergence when $f^\star$ is known ($f = 0.0094$ on Rosenbrock) but diverges when $f^\star$ is misspecified; Nesterov acceleration provides a $7\times$ improvement over GD on Rosenbrock, consistent with the theoretical $O(k)$-factor advantage of $O(1/k^2)$ over $O(1/k)$; Newton's method converges in one step on the quadratic loss and provides a $5.4\times$ objective improvement over GD in $4\times$ fewer iterations on Rosenbrock, with L-BFGS the scalable quasi-Newton alternative; the outstanding result is Nelder--Mead achieving $f \approx 10^{-29}$ (machine precision) on Rosenbrock in 160 iterations, outperforming every gradient-based method; finite difference accuracy is governed by competing truncation ($O(\delta)$ for forward, $O(\delta^2)$ for centered) and floating-point errors, with optimal $\delta \approx 10^{-8}$ and $10^{-5.3}$ respectively; projected gradient descent enforces feasibility as a hard constraint from iteration~1 via an $O(1)$ clip, while the augmented Lagrangian achieves exact constraint satisfaction at finite penalty via multiplier updates; and Frank--Wolfe converges at $O(1/k)$ with a computable FW gap certificate, and is preferred over projected GD when the LP oracle is cheaper than projection.

Each method is evaluated in terms of convergence speed, trajectory behaviour, stability, computational cost, and alignment with the theoretical guarantees from the CS7DS2 course.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmarks span a spectrum of problem difficulty, from a well-conditioned convex quadratic to a highly ill-conditioned non-convex landscape.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
x^{(i)} \sim \mathcal{N}(0, I_2), \quad
y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad
\varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The objective is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2
= \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strongly convex quadratic with Hessian $H = \frac{1}{m}X^T X$ (constant throughout optimisation). Numerical computation gives eigenvalues $\lambda_{\min} \approx 0.9247$ and $\lambda_{\max} \approx 1.0325$, yielding condition number $\kappa(H) = \lambda_{\max}/\lambda_{\min} \approx 1.12$. This near-isotropic curvature means almost any step size within the stability range $\alpha < 2/\lambda_{\max} \approx 1.94$ will converge rapidly. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Due to observation noise, the irreducible loss floor is $J^{\star} = \frac{1}{2}\sigma^2 \approx 0.4826$ (not zero). Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is diagonal:
\begin{equation}
H(x_1) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
Since $\sin(x_1) \in [-1, 1]$, the first diagonal entry ranges from $1$ to $3$, giving a condition number $\kappa \in [10/3, 10/1] \approx [3.3, 10]$. Along the optimisation trajectory from $x_0 = [-1, 4]^T$ to $x^\star$, $\kappa$ ranges from approximately $3.5$ to $6.9$. The minimum satisfies $df/dx_2 = 0 \Rightarrow x_2^\star = 2$ and $df/dx_1 = 2(x_1-1) + \cos(x_1) = 0$, which gives $x_1^\star \approx 0.5824$ (solved numerically via Newton's method applied to this scalar equation). Thus $x^\star \approx (0.5824, 2.000)$ and $f^\star = (x_1^\star - 1)^2 + \sin(x_1^\star) \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\textit{Note:} the visualisation code marks the optimum at $(0.61, 2.0)$, which is a slight overestimate; the true value $x_1^\star \approx 0.582$ is used in this report.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$. The Hessian is:
\begin{equation}
H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1,1)$: eigenvalues are approximately $0.399$ and $1001.6$, giving $\kappa \approx 2508$. Near the starting point $x_0 = (-1, 1)$: the largest eigenvalue is approximately $1001$, so the stability bound for gradient descent is $\alpha < 2/1001 \approx 0.002$. The code uses $\alpha = 0.0012$, which is safely conservative. The characteristic narrow, curved valley along $x_2 = x_1^2$ and the extreme ill-conditioning make Rosenbrock one of the most widely used stress tests for optimisation algorithms.

\textbf{Rosenbrock valley geometry:} The function can be decomposed as $f(x) = (1-x_1)^2 + 100(x_2 - x_1^2)^2$. Along the valley floor $x_2 = x_1^2$: $f = (1-x_1)^2$, a univariate quadratic in $x_1$. Perpendicular to the valley: curvature $\approx 200$ (from the $100 \times 2$ coefficient). Along the valley: curvature $\approx 2$ (the $(1-x_1)^2$ term). The ratio $200/2 = 100$ is the local condition number along the valley; the additional curvature variation along the trajectory explains the global $\kappa \approx 2508$ estimate at the optimum. This means:
\begin{itemize}[nosep]
  \item \textbf{Phase 1} (fast): descent from $x_0 = (-1,1)$ towards the valley floor — the large $x_2$ gradient drives fast convergence perpendicular to the valley.
  \item \textbf{Phase 2} (slow): traversal along the valley from $x_1 \approx -1$ to $x_1 = 1$ --- the small valley curvature $\approx 2$ means GD's step $\alpha \approx 0.001$ (constrained by the perpendicular curvature) barely advances, requiring $\sim$1/(2 × 0.001) = 500 iterations to traverse unit distance along the valley.
\end{itemize}
This two-phase structure explains why all first-order methods plateau before reaching $(1,1)$ in 120--150 iterations.

\subsection{Benchmark D: Constrained Feasible Set}
For Questions~5 and 6, the feasible set is:
\begin{equation}
\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a box (polytope), for which the projection and the linear minimisation oracle are both $O(d)$ operations.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for $N = 120$ iterations (see Table~\ref{tab:algo_props} in Section~\ref{sec:summary} for a complete properties reference of all methods). The baseline GD update is:
\begin{equation}
x_{k+1} = x_k - \alpha\,\nabla f(x_k).
\end{equation}
For a $\mu$-strongly convex function with $L$-Lipschitz gradient, GD with $\alpha = 1/L$ converges at:
\begin{equation}
f(x_k) - f^\star \leq \left(1 - \frac{\mu}{L}\right)^k [f(x_0) - f^\star].
\end{equation}
On Benchmark A ($\kappa = L/\mu \approx 1.12$, $\alpha = 0.08 < 1/L \approx 0.97$), the effective rate is $(1 - 1/1.12)^k \approx 0.107^k$ --- extremely fast. On Benchmark C ($\kappa \approx 2508$ near the starting point, $\alpha = 0.0012 \approx 1/L_{\text{local}}$), the rate is $(1 - 1/2508)^k \approx 0.9996^k$ --- extremely slow. Note: Rosenbrock is not globally strongly convex (it is not even convex globally), so the rate bound holds only locally near the optimum. The used step size is guided by the local Hessian eigenvalue bound rather than global strong convexity.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive, requires knowing $f^\star$}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is theoretically optimal for convex functions when $f^\star$ is known exactly: it guarantees convergence in $O(1/k)$ without requiring knowledge of the Lipschitz constant $L$. However, it is highly sensitive to errors in $f^\star$. A related but distinct method is the \emph{Barzilai--Borwein} (BB) step, which estimates the local curvature from consecutive iterates without requiring $f^\star$. Using $s_k = x_k - x_{k-1}$ and $y_k = \nabla f(x_k) - \nabla f(x_{k-1})$, the two BB variants are:
\begin{equation}
\alpha_k^{\text{BB1}} = \frac{s_k^T s_k}{s_k^T y_k}, \qquad \alpha_k^{\text{BB2}} = \frac{s_k^T y_k}{y_k^T y_k}.
\end{equation}
Both satisfy the secant equation $\alpha_k^{-1} I \cdot s_k \approx y_k$ (a scaled-identity approximation to the Hessian): BB1 is the Rayleigh quotient $s_k^T H s_k / s_k^T s_k$ along the most recent step direction; BB2 is its harmonic counterpart. BB does not reduce the objective monotonically but paired with a non-monotone line search (Grippo et al.\ 1986 condition $f(x_{k+1}) \leq \max_{j \leq M}f(x_{k-j}) - \sigma\alpha_k\|g_k\|^2$), it achieves superlinear practical convergence. For Benchmark~A, BB1 would estimate $\alpha \approx s_k^T s_k / (s_k^T y_k) \approx 1/\lambda_{\text{mid}} \approx 0.978^{-1} \approx 1.02$ after a few steps --- very close to the optimal $\alpha = 1/L \approx 0.97$, without any knowledge of $L$ or $f^\star$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative squared gradient (per coordinate)}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad adapts the learning rate per coordinate: directions with large historical gradients receive smaller effective steps. The cumulative sum $G_k$ is monotonically non-decreasing, so the effective learning rate $\alpha_0/\sqrt{G_k}$ is \emph{monotonically decreasing}. This guarantees stability but can cause premature convergence on non-stationary problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's monotone decay by replacing the cumulative sum with an exponential moving average (decay $\beta = 0.9$). This means $v_k$ tracks the \emph{recent} gradient magnitude rather than the full history, allowing the effective learning rate to increase again if gradients become smaller. RMSprop was proposed by Hinton (unpublished, Coursera lecture) as a heuristic; it lacks the convergence guarantees of Adagrad but is more robust in practice.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, g_k$ \Comment{Accumulated velocity}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method (Polyak, 1964) accumulates a momentum term $z_k$ that damps oscillations in high-curvature directions. For a quadratic with condition number $\kappa$, the optimal Heavy Ball parameters are $\alpha^\star = 4/(1+\sqrt{1/\kappa})^2$ and $\beta^\star = ((1-1/\sqrt{\kappa})/(1+1/\sqrt{\kappa}))^2$, achieving convergence rate:
\begin{equation}
O\!\left(\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^k\right),
\end{equation}
dramatically better than GD's $O\!\left(\left(\frac{\kappa-1}{\kappa+1}\right)^k\right)$. For Benchmark A ($\kappa \approx 1.117$): optimal HB rate $\approx 0.028^k$ vs GD's optimal rate $0.055^k$ (about $2\times$ fewer iterations for the same accuracy; both are negligible since both converge in $\sim$2 iterations to noise floor). For Benchmark C ($\kappa \approx 2508$): optimal HB rate $\approx 0.961^k$ vs GD's $0.9992^k$ --- HB needs $\sim$115 vs GD's $\sim$5750 iterations to reach $\varepsilon = 0.01$ with optimal parameters, a $50\times$ speedup. The used parameters ($\alpha = 0.045, \beta = 0.88$ for Benchmark A) are \emph{not} the theoretically optimal values ($\alpha^\star \approx 1.02, \beta^\star \approx 0.0008$) due to the non-convexity of Rosenbrock requiring conservative settings. The momentum vector $z_k$ accumulates gradient \emph{at the current point}, unlike the Nesterov variant which evaluates the gradient \emph{at the lookahead point} (see Section~\ref{sec:q2}).

For Benchmarks A and B, the Polyak step is set with $f^\star = 0$, which \emph{misspecifies} the true minimum: $J^\star_A \approx 0.4826$ and $f^\star_B \approx 0.7244$. On Benchmark C, $f^\star = 0$ is correct (the Rosenbrock minimum is exactly zero). This deliberate misspecification allows the experiment to study the method's sensitivity to the $f^\star$ estimate.

\textbf{Polyak divergence mechanism (quantitative):} The Polyak step satisfies $\alpha_k = (J(x_k) - 0)/(\|g_k\|^2 + \varepsilon)$. On Benchmark~A, as the iterate $x_k$ approaches $\hat{\theta}_{\text{OLS}}$:
\begin{itemize}[nosep]
  \item The gradient magnitude satisfies $\|g_k\|^2 = \|\frac{1}{m}X^T(X x_k - y)\|^2 \to 0$ at rate $O\bigl((1-\mu/L)^k\bigr)$.
  \item The objective satisfies $J(x_k) - 0 \to J^\star \approx 0.4826 > 0$ (the irreducible noise floor).
\end{itemize}
Thus $\alpha_k \to J^\star / \varepsilon = 0.4826 / 10^{-4} = 4826$ as $k \to \infty$. Such a large step size causes a huge overshoot past the optimum, and the process oscillates divergently. This analysis also implies that any method that uses $f^\star = 0$ as the target will fail when the true minimum is nonzero, and the failure magnitude scales with $J^\star / \varepsilon$.

\textbf{Key finding 1 (Polyak):} The Polyak step size is highly sensitive to $f^\star$ accuracy. When $f^\star = 0$ deviates from the true minimum (Benchmarks A and B), $\alpha_k \to \infty$ as $\|\nabla f\| \to 0$ while $f(x_k) - f^\star \to f^\star_{\text{true}} > 0$. On Benchmark~C, $f^\star = 0$ is exact, giving the best first-order convergence ($f = 0.0094$).

\textbf{Why Polyak still oscillates near the Rosenbrock minimum:} Even with correct $f^\star = 0$, the Polyak iterate oscillates around $(1,1)$ (visible in Figure~\ref{fig:comprehensive_C}) rather than converging to it. Near $(1,1)$ on the valley floor (where $x_2 - x_1^2 \approx 0$, i.e., $\delta_2 \approx 2\delta_1$):
\begin{align}
f &\approx \delta_1^2, \qquad \|\nabla f\|^2 \approx 4\delta_1^2 + O(\delta^2),
\quad\Rightarrow\quad \alpha_k \approx \frac{\delta_1^2}{4\delta_1^2 + \varepsilon} \approx \frac{1}{4}.
\end{align}
But the Rosenbrock Hessian at $(1,1)$ has maximum eigenvalue $L \approx 1001.6$, so the stability bound is $\alpha < 2/L \approx 0.002$. The Polyak step $\alpha_k \approx 0.25 \gg 0.002$ is thus $\sim$125$\times$ larger than the stability bound, causing persistent oscillations. The Polyak method has no mechanism to respect local Lipschitz constants --- it scales only to the remaining suboptimality gap, not to the local curvature. This explains why $f$ oscillates between $0.009$ and $0.02$ rather than converging monotonically to zero.

\textbf{Key finding 2 (RMSprop vs Adagrad):} On Benchmark A, Adagrad achieves the noise floor ($f = 0.4826$) while RMSprop settles slightly above it ($f = 0.5027$). This is because Adagrad's \emph{monotonically decreasing} step size $\alpha_0/\sqrt{G_k}$ ultimately shrinks to zero, causing the iterates to stop moving (i.e., the method converges to a fixed point). RMSprop's step size $\alpha_0/\sqrt{v_k + \varepsilon}$ does not necessarily shrink to zero --- in steady state, $v_k \approx (1-\beta)^{-1}\mathbb{E}[g_k^2]$ which remains bounded away from zero as long as the gradient is nonzero. This means RMSprop may continue taking small steps near the minimum, oscillating rather than converging to an exact fixed point.

\textbf{Adaptive methods as diagonal preconditioning:} Adagrad, RMSprop, and Adam can all be interpreted as applying a diagonal preconditioner $P_k$ to the gradient before the update: $x_{k+1} = x_k - P_k g_k$, where $P_k = \text{diag}\!\bigl(\alpha_0/(\sqrt{G_{k+1,i}} + \varepsilon)\bigr)$ for Adagrad. The ideal preconditioner is $H^{-1}$ (Newton direction); diagonal adaptive methods approximate this cheaply. For a quadratic $f(x) = \frac{1}{2}x^T H x$ with diagonal $H = \text{diag}(h_1, \ldots, h_d)$, the Adagrad accumulator satisfies $G_{k,i} = \sum_j g_{j,i}^2 \approx k \cdot h_i^2 \|x_0 - x^\star\|_i^2$ asymptotically (since $g_{j,i} \approx h_i (x_{j,i} - x^\star_i)$ for a quadratic). Hence $P_{k,ii} \approx \alpha_0 / (|h_i|\sqrt{k} \cdot c)$: the effective step per coordinate is inversely proportional to the Hessian diagonal magnitude, providing equal convergence speed in each coordinate even under extreme anisotropy. This is why Adagrad outperforms GD on problems with disparate curvatures (e.g., NLP embeddings with $h_i \in [10^{-3}, 10^3]$) and why it was originally motivated for \emph{sparse} high-dimensional settings. However, all diagonal preconditioners miss off-diagonal Hessian structure: for Rosenbrock, whose Hessian has a large off-diagonal entry $-400x_1$ coupling $x_1$ and $x_2$, even the ideal diagonal preconditioner cannot reduce the condition number below $\kappa_{\text{diag}} = \max(h_{ii})/\min(h_{ii}) \approx 1002/2 = 501$ (from the diagonal alone at $(1,1)$), versus Newton's $H^{-1}$ which reduces $\kappa$ to $1$ exactly. This explains why Adam and Adagrad achieve $f \approx 1.6$--$2.2$ on Rosenbrock, while damped Newton (which captures off-diagonal coupling) reaches $f = 0.017$ in only 40 iterations.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$, a $374\times$ improvement over GD). Otherwise, Adagrad and Heavy Ball match or exceed GD baseline performance. Heavy Ball's $f = 1.40$ vs.\ GD's $f = 3.51$ on Rosenbrock demonstrates the momentum benefit even for non-convex problems, where the formal quadratic-convergence guarantee does not apply but the practical benefit is still realised.
\item \textbf{Step-size behaviour:} Adagrad's step decays at $O(1/\sqrt{k})$ asymptotically (since $G_k = \sum_{i=1}^k g_i^2 \sim O(k)$ for bounded gradients), guaranteeing stability but slowing late-stage convergence. RMSprop's exponential moving average with $\beta = 0.9$ maintains $v_k \approx (1-\beta)^{-1} \cdot \mathbb{E}[g_k^2]$ in steady state, so the effective step stabilises rather than decaying to zero. The Polyak step is adaptive in a fundamentally different sense: it scales to the \emph{remaining suboptimality} rather than to accumulated gradients.
\item \textbf{Stability:} GD baseline is the most stable but slowest. Polyak is unstable when $f^\star$ is misspecified. Heavy Ball can oscillate on ill-conditioned non-quadratic problems (no theoretical guarantee beyond quadratics), but is empirically stable here. Formal counter-examples to Heavy Ball convergence on smooth non-convex functions have been constructed (Lessard, Recht \& Packard 2016): the coupled linear system $(x_k, x_{k-1})$ can have unstable complex eigenvalues when the Hessian changes along the trajectory, causing divergence even for mild non-quadratic perturbations. For Rosenbrock, the Hessian changes dramatically from starting point $(-1,1)$ (where $H_{11} \approx 602$, $H_{22} = 200$, $H_{12} = 400$) to the optimum $(1,1)$ (where $H_{11} \approx 1202$, $H_{22} = 200$, $H_{12} = -400$); the conservative parameters ($\alpha = 0.0008, \beta = 0.86$) keep the coupled dynamics in the stable regime empirically, but this is not guaranteed for all trajectories.
\item \textbf{Adagrad vs.\ RMSprop --- when to prefer each:} Adagrad was designed for \emph{sparse gradient} settings (e.g., text embeddings in NLP, where only a few coordinates have non-zero gradient per update). In sparse settings, $G_{k,i}$ accumulates slowly for rarely-updated coordinates, giving those coordinates large effective steps; frequently-updated coordinates receive shrinking steps. This yields per-feature adaptation that is beneficial for NLP tasks. RMSprop is preferred for \emph{dense gradient} settings (e.g., image classification, continuous optimisation) where gradient sparsity is absent and Adagrad's premature decay is a liability. The benchmark problems here have dense gradients, explaining why Adagrad's advantage over RMSprop is marginal.
\item \textbf{Practical implication:} In practice, exact $f^\star$ is rarely known, limiting Polyak to problems where a lower bound on the optimal value is available (e.g., $f^\star = 0$ for a non-negative loss like cross-entropy with label smoothing). RMSprop and Adam are more robust adaptive choices for general deep learning applications.
\item \textbf{Polyak--Łojasiewicz (PL) condition:} All the GD convergence rates derived above assumed strong convexity ($\mu > 0$). The \emph{PL condition} is a weaker alternative: $\|\nabla f(x)\|^2 \geq 2\mu(f(x) - f^\star)$ for all $x$. Under PL, GD with $\alpha = 1/L$ achieves the same linear rate $(1-\mu/L)^k$ as for strongly convex functions, \emph{without} requiring convexity. The PL condition holds when the loss landscape has no spurious local minima (all stationary points are global minima), which is empirically observed for deep neural networks in the overparameterised regime. Rosenbrock satisfies PL locally near $(1,1)$ (since the Hessian is positive definite there), explaining why GD eventually converges linearly along the valley floor. However, Rosenbrock violates PL globally: there exist saddle points and regions where $\|\nabla f\|$ is large but $f - f^\star$ is also large (e.g., along the ridge $x_2 = x_1^2$ far from $(1,1)$), confirming non-convexity.
\end{itemize}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead / extrapolation point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Velocity update at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The critical distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This ``look before you step'' correction allows Nesterov to anticipate the momentum's direction and correct trajectory before overshooting. The schedule $\beta_k = (k-1)/(k+2)$ gradually increases momentum from 0, avoiding early-iteration instability.

\textbf{Alternative interpretation:} Substituting $\tilde{x}_k = x_k + \beta_k z_k$ into the update $x_{k+1} = x_k + z_{k+1} = x_k + \beta_k z_k - \alpha \nabla f(\tilde{x}_k)$ gives:
\begin{equation}
x_{k+1} = \tilde{x}_k - \alpha \nabla f(\tilde{x}_k).
\end{equation}
This is exactly one step of gradient descent \emph{applied at the lookahead point} $\tilde{x}_k$! The ``Nesterov magic'' is that the lookahead point is a better approximation of where the iterate will end up than $x_k$ itself, so gradient information from $\tilde{x}_k$ is more predictive of the right descent direction.

Nesterov's method achieves an \textbf{optimal} $O(1/k^2)$ convergence rate for smooth convex functions, compared to $O(1/k)$ for gradient descent. For a convex function with $L$-Lipschitz gradient:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{k^2}.
\end{equation}
This $O(1/k^2)$ rate matches the \emph{lower bound} for first-order methods on smooth convex problems: Nesterov (1983) showed that for any first-order method using only $\{x_0, \nabla f(x_0), \ldots, \nabla f(x_{k-1})\}$ information, there exist smooth convex functions with $L$-Lipschitz gradient for which $f(x_k) - f^\star \geq \frac{3L\|x_0-x^\star\|^2}{32(k+1)^2}$. Since Nesterov's own method achieves $O(1/k^2)$, it is complexity-optimal (no first-order method can do asymptotically better on worst-case smooth convex problems). For $\mu$-strongly convex functions, the optimal first-order rate is $O((1-\sqrt{\mu/L})^k)$, achieved by FISTA and related methods. Note that Newton's method, while second-order (using Hessians), is not first-order and is not constrained by this lower bound.

\textbf{Strongly convex case --- FISTA:} When $f$ is $\mu$-strongly convex and $L$-smooth ($\kappa = L/\mu$), the optimal first-order rate $O\!\left((1-\sqrt{\mu/L})^k\right)$ is achieved by variants such as FISTA (Fast Iterative Shrinkage-Thresholding Algorithm). The critical modification is that the momentum coefficient becomes \emph{constant} rather than increasing with $k$: $\beta^\star = (\sqrt{L} - \sqrt{\mu})/(\sqrt{L} + \sqrt{\mu}) = (\sqrt{\kappa} - 1)/(\sqrt{\kappa} + 1)$, matching the Heavy Ball optimal momentum parameter (Section~\ref{sec:q1}). The relationship between the three key rates is summarised as:
\begin{equation}
\underbrace{\left(1 - \frac{1}{\kappa}\right)^k}_{\text{GD, rate} \approx 1-1/\kappa}
\;\gg\;
\underbrace{\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^k}_{\text{FISTA/HB optimal, rate} \approx 1-2/\sqrt{\kappa}}
\;\gg\;
\underbrace{\frac{2LR^2}{k^2}}_{\text{Nesterov (non-SC), } O(1/k^2)}
\end{equation}
for large $\kappa$. On Benchmark~A ($\kappa \approx 1.12$): FISTA rate $= (\sqrt{1.12}-1)/(\sqrt{1.12}+1) \approx 0.058/2.058 \approx 0.028$, similar to GD's optimal rate $\approx 0.055$ since $\kappa \approx 1$. On Benchmark~C ($\kappa \approx 2508$): FISTA rate $\approx (50.08-1)/(50.08+1) \approx 0.961^k$, compared to GD's $0.9992^k$ --- FISTA requires $\sim$115 vs.\ GD's $\sim$5750 iterations to reduce error by $10^2$, a $50\times$ speedup. Rosenbrock is not globally strongly convex (it is non-convex away from the valley), so the full guarantee holds only locally near $(1,1)$; but the FISTA rate $\approx 0.961^k$ confirms why Nesterov momentum is so valuable for ill-conditioned problems. The increasing schedule $\beta_k = (k-1)/(k+2)$ used here is the non-strongly-convex version; replacing it with the fixed $\beta^\star$ would give the FISTA strongly-convex rate if local strong convexity is exploited.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (EMA of gradient)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (EMA of sq.\ gradient)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam (Kingma \& Ba, 2015) combines first-moment momentum (analogous to Heavy Ball) with per-coordinate adaptive scaling (analogous to RMSprop). The bias-correction terms $(1-\beta_1^t)^{-1}$ and $(1-\beta_2^t)^{-1}$ compensate for the zero-initialisation of $m_0 = v_0 = 0$: without correction, early estimates are biased towards zero, artificially reducing the effective step size in the first few iterations. After bias correction, the magnitude of the Adam update per coordinate is:
\begin{equation}
\left|\Delta x_{t,i}\right| = \alpha\,\frac{|\hat{m}_{t,i}|}{\sqrt{\hat{v}_{t,i}} + \varepsilon}
\leq \frac{\alpha}{\varepsilon},
\end{equation}
since $|\hat{m}_{t,i}| \leq \sqrt{\hat{v}_{t,i}}$ (by the Cauchy--Schwarz inequality applied to the EMA weights). This provides an inherent step-size upper bound of $\alpha/\varepsilon$ per coordinate, regardless of gradient magnitude.

\textbf{Adam on Rosenbrock --- per-coordinate analysis:} The Rosenbrock gradient at $x$ is:
\begin{equation}
g_1 = -2(1-x_1) - 400x_1(x_2 - x_1^2), \qquad g_2 = 200(x_2 - x_1^2).
\end{equation}
Near the starting point $(-1,1)$: $|g_1| \approx 2 \times 2 + 400 \times 1 \times (1 - 1) = 4$ (large, driving fast $x_1$ progress) and $|g_2| = 200(1-1) = 0$. However, deep in the valley near $(0.5, 0.25)$: $|g_1| \approx 2 \times 0.5 + 400 \times 0.5 \times 0 = 1$ (small: near-optimal in $x_1$) and $|g_2| \approx 200 \times 0.25 \approx 50$ (large: $x_2$ not yet at $x_1^2$). The $x_2$ direction therefore experiences much larger gradient magnitudes during the valley traversal phase, so Adam assigns a \emph{smaller} effective step to $x_2$ (via larger $\hat{v}_{t,2}$) --- compensating for the 200× curvature. This per-coordinate scaling explains why Adam ($f = 1.578$) substantially outperforms GD ($f = 3.388$) but is outperformed by Nesterov ($f = 0.479$): Adam's scaling addresses the \emph{diagonal} curvature mismatch, but cannot capture the \emph{off-diagonal} coupling in the Rosenbrock Hessian (the $-400x_1$ off-diagonal term), whereas Nesterov's lookahead accumulates momentum along the consistent valley direction regardless of the coordinate system.

\textbf{Note on Adam convergence guarantees:} The original Adam paper (Kingma \& Ba, 2015) claimed convergence under stochastic settings; however, Reddi et al.\ (2018) showed that Adam can \emph{diverge} on simple convex problems due to the exponential moving average failing to track rapidly changing gradients. The \textsc{AMSGrad} fix (using $\max(\hat{v}_{t-1}, \hat{v}_t)$ instead of $\hat{v}_t$ in the denominator) restores convergence, though the practical performance difference is often negligible. The implementations here run in a deterministic setting, so convergence is not an issue.

\begin{itemize}[nosep]

  \item Nesterov with optimal schedule: $f(x_k) - f^\star \leq \frac{2L R^2}{k^2}$
\end{itemize}
Nesterov's bound is tighter by a factor of $k/4$: at $k = 100$, Nesterov provides a $25\times$ improvement in the theoretical guarantee. On Rosenbrock at $k = 150$, the observed ratio is $3.388/0.479 \approx 7$, consistent with the $O(k)$ factor advantage. The gap between theory ($25\times$) and practice ($7\times$) arises because Rosenbrock is non-convex and the initialised momentum schedule $\beta_k = (k-1)/(k+2)$ conservatively starts at 0, delaying the full benefit of Nesterov's acceleration.

\textbf{Adaptive restarts for Nesterov:} On non-convex problems, Nesterov's accumulated momentum can point in the wrong direction when the iterate crosses a ridge or enters a new local region. The \emph{adaptive restart} heuristic (O'Donoghue \& Candès, 2015) resets the momentum to zero (sets $\beta_k = 0$ and $z_k = 0$) when either the \emph{gradient restart condition} $\nabla f(x_k)^T(x_k - x_{k-1}) > 0$ (the step direction opposes the gradient) or the \emph{function restart condition} $f(x_{k+1}) > f(x_k)$ (objective increased) is detected. Intuitively, a restart is triggered when the momentum is unhelpful, discarding the accumulated history and restarting from a clean gradient step. For strongly convex functions, the restarted Nesterov achieves the linear convergence rate $O\!\left((1-\sqrt{\mu/L})^k\right)$ without requiring knowledge of $\mu$ (unlike FISTA). On Rosenbrock, the valley's curvature change between the $x_1$ and $x_2$ directions can cause momentum accumulation in the $x_2$ direction after reaching the valley floor --- a restart would clear this and allow fresh acceleration along the $x_1$ direction of the valley. The implementation cost is one extra condition check per iteration, with no computational overhead if restarts are rare.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data permuted at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute the $m$ sample indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The stochastic gradient $g$ is an unbiased estimate of $\nabla J(\theta)$: $\mathbb{E}[g] = \nabla J(\theta)$. Its variance per coordinate is:
\begin{equation}
\mathrm{Var}(g_i) = \frac{1}{b^2}\sum_{j \in \mathcal{B}}\mathrm{Var}\!\left([X^T(X\theta - y)]_j / 1\right) = \frac{1}{b}\,\sigma_{\text{grad}}^2,
\end{equation}
where $\sigma_{\text{grad}}^2$ is the per-sample gradient variance. The variance decays as $1/b$: doubling the batch size halves the noise. SGD with constant learning rate $\alpha$ does not converge to the exact minimiser; the steady-state oscillation radius is $O(\alpha \sigma_{\text{grad}} / \sqrt{b})$ around $\theta^\star$. To achieve $\varepsilon$-accuracy in expectation, a decaying schedule $\alpha_k \to 0$ (e.g., $\alpha_k = \alpha_0 / (k+1)$, giving the Robbins--Monro condition $\sum \alpha_k = \infty$, $\sum \alpha_k^2 < \infty$) is needed, achieving the $O(1/k)$ rate. With constant $\alpha = 0.06$, the plots show convergence to a \emph{noise floor} rather than the exact minimum:
\begin{equation}
\liminf_{k \to \infty} \mathbb{E}\!\left[J(\theta_k) - J^\star\right] \approx \frac{\alpha\,\sigma_{\text{grad}}^2}{b\,\mu},
\end{equation}
where $\mu$ is the strong convexity constant. The epoch-level update count is $\lfloor m/b \rfloor$ per epoch: $200$ updates for $b = 5$ and $25$ updates for $b = 40$ per epoch of $m = 1000$ samples.

\textbf{Learning rate schedules:} The choice of step size schedule significantly impacts convergence of SGD. Common practical schedules are:
\begin{itemize}[nosep]
  \item \textbf{Step decay:} $\alpha_k = \alpha_0 \cdot \rho^{\lfloor k/T \rfloor}$ (e.g., $\rho = 0.1$ every $T$ epochs). Simple but requires tuning $T$ and $\rho$.
  \item \textbf{Exponential decay:} $\alpha_k = \alpha_0\, e^{-\lambda k}$. Satisfies Robbins--Monro since $\sum \alpha_k < \infty$ for $\lambda > 0$.
  \item \textbf{Polynomial decay:} $\alpha_k = \alpha_0/(1 + k)^\gamma$ for $\gamma \in (0.5, 1]$. Satisfies $\sum \alpha_k = \infty$, $\sum \alpha_k^2 < \infty$ for $\gamma > 0.5$, guaranteeing convergence to $\theta^\star$.
  \item \textbf{Cosine annealing} (standard in deep learning): $\alpha_k = \alpha_{\min} + \tfrac{1}{2}(\alpha_0 - \alpha_{\min})(1 + \cos(\pi k/T))$. Decays smoothly from $\alpha_0$ to $\alpha_{\min}$ over $T$ iterations, then optionally restarts (SGDR). Avoids the sharp step of step decay.
  \item \textbf{Linear warmup + cosine decay:} Start with small $\alpha$ for $W$ warmup steps (prevents early instability when the model is far-from-minimum), then apply cosine schedule. The standard in large-scale transformer training (BERT, GPT).
\end{itemize}
For the experiments here (constant $\alpha = 0.06$, 50 epochs), the plots show the noise-floor plateau characteristic of constant learning rate SGD; applying a cosine schedule that decays to $\alpha_{\min} \approx 0$ by epoch 50 would eliminate the plateau and drive the iterate to $\theta^\star$.

\textbf{Variance reduction (SVRG):} The fundamental limitation of SGD is the irreducible noise floor $\approx \alpha \sigma^2 / (b\mu)$ imposed by gradient variance. The \emph{stochastic variance reduced gradient} (SVRG, Johnson \& Zhang 2013) removes this limitation by maintaining a periodically-updated full-gradient snapshot $\tilde{g} = \nabla f(\tilde{x})$. The per-iteration estimator is:
\begin{equation}
\tilde{v}_k = \nabla f_i(x_k) - \nabla f_i(\tilde{x}) + \tilde{g},
\end{equation}
where $i$ is a randomly selected sample and $\tilde{x}$ is the anchor (snapshot) updated every $s$ steps. This estimator is unbiased: $\mathbb{E}[\tilde{v}_k] = \nabla f(x_k)$. Its variance satisfies:
\begin{equation}
\mathbb{E}[\|\tilde{v}_k - \nabla f(x_k)\|^2] \leq 4L\,[f(x_k) - f^\star + f(\tilde{x}) - f^\star],
\end{equation}
which tends to \emph{zero} as $x_k \to x^\star$ (unlike the fixed $\sigma^2$ of plain SGD). With step size $\alpha \leq 1/(10L)$ and update frequency $s = 2m$, SVRG achieves the same linear convergence rate as full GD:
\begin{equation}
\mathbb{E}[f(x_k) - f^\star] \leq \left(1 - \frac{1}{10\kappa}\right)^k [f(x_0) - f^\star],
\end{equation}
but at cost $O(m + s/b) = O(m)$ per full pass rather than GD's $O(m)$ per step for the same $n$ gradient evaluations. The total complexity to reach $\varepsilon$-accuracy is $O((m + \kappa)\log(1/\varepsilon))$: when $\kappa \ll m$ (as in large data / weakly regularized problems), SVRG costs essentially $O(m\log(1/\varepsilon))$ --- the same as running GD once but with linear convergence. For the Benchmark~A setting ($m = 1000$, $\kappa \approx 1.12$, $\mu \approx 0.92$), vanilla SGD with constant $\alpha$ already converges to the noise floor quickly; SVRG's advantage is most pronounced when $\kappa \gg 1$ and the data set is large ($m \gg \kappa$), e.g., $m = 10^6$, $\kappa = 10^3$.

\subsection{Local Approximations of $g(x) = x^4$}

Around a point $x_0$, a smooth function $g(x)$ can be approximated by its first- and second-order Taylor expansions.

\textbf{First-order (linear) approximation} around $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3(x - 0.25) = 0.003906 + 0.0625(x - 0.25).
\end{equation}
This tangent line correctly models the slope at $x_0$ but does not capture curvature, leading to poor approximation away from $x_0$.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.003906 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
The Newton step $x_{k+1} = x_k - g'(x_k)/g''(x_k)$ minimises this local quadratic model, corresponding to a full step to the model's stationary point. For $g(x) = x^4$:
\begin{equation}
x_{k+1} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3} = \frac{2}{3}x_k.
\end{equation}
The Newton iterates satisfy $x_k = (2/3)^k x_0$, converging \emph{linearly} (not quadratically) to $x^\star = 0$. The lack of quadratic convergence arises because $g''(x^\star) = g''(0) = 0$ --- the Hessian is singular at the optimum, violating the strong convexity condition $\nabla^2 g(x^\star) \succ 0$ required for quadratic convergence.

\textbf{Quantitative impact:} To reach $|x_k| \leq 10^{-10}$ starting from $x_0 = 0.25$:
\begin{itemize}[nosep]
  \item \textbf{Linear rate $(2/3)^k$:} Need $(2/3)^k \times 0.25 \leq 10^{-10}$, so $k \geq \ln(4 \times 10^{-10})/\ln(2/3) \approx 55$ iterations.
  \item \textbf{Hypothetical quadratic rate} (if $g''(0) > 0$): $|x_{k+1}| \leq C |x_k|^2$, so $|x_k| \sim (C|x_0|)^{2^k}/C$. Starting with $C|x_0| \approx 0.5$: $|x_5| \approx 0.5^{32} \approx 10^{-10}$ --- 5 iterations for the same accuracy as 55 linear steps!
\end{itemize}
This dramatic difference (5 vs.\ 55 iterations for the same accuracy) illustrates why quadratic convergence is so desirable and why the Hessian singularity ($g''(x^\star) = 0$) is catastrophic for Newton's method.

\subsection{Newton's Method}

For a smooth function, Newton's method solves a linear system at each iteration to find the Newton direction:

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0, 1]$, damping $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise Hessian for numerical stability}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$ \Comment{$\alpha = 1$: pure Newton; $\alpha < 1$: damped}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Quadratic convergence:} For a strongly convex, twice-differentiable function with Lipschitz Hessian, the undamped Newton method ($\alpha = 1$) with exact Hessian satisfies:
\begin{equation}
\|x_{k+1} - x^\star\| \leq C\,\|x_k - x^\star\|^2
\end{equation}
for some constant $C > 0$ once the iterate enters a neighbourhood of $x^\star$ (the quadratic convergence region). This \emph{quadratic} rate means the number of correct digits doubles each iteration, compared to GD's linear rate of $O((1-\mu/L)^k)$.

\textbf{Quadratic convergence basin:} The standard sufficient condition for quadratic convergence is $\|x_0 - x^\star\| < 2\mu / (L_H)$, where $\mu$ is the strong convexity constant and $L_H$ is the Lipschitz constant of $\nabla^2 f$. For Rosenbrock, $L_H$ is very large near $x_0 = (-1,1)$ due to rapid Hessian variation, so the undamped Newton step immediately overshoots the basin. The heavy damping $\alpha = 0.22$ used here is a pragmatic response: it guarantees descent at the cost of slowing convergence to the linear regime $O(\alpha^k)$ until the iterate gets close enough to $(1,1)$ for the quadratic regime to activate. Specifically, 20 iterations of Newton with $\alpha = 0.22$ bring the iterate from $(-1,1)$ to $(0.183, 0.020)$ --- progress along the valley, but still far from the local quadratic convergence basin.

For non-convex problems (e.g., Rosenbrock away from the optimum) or problems with rapidly changing Hessians, a damped step ($\alpha < 1$) prevents overshooting into regions where the quadratic model is inaccurate. The quadratic convergence region (``basin of attraction'' for Newton) is entered only when $x_k$ is sufficiently close to $x^\star$; outside this region, a damped or globalised Newton step is needed.

\textbf{Line search as a principled alternative to fixed damping:} The fixed step $\alpha = 0.22$ is conservative; a principled alternative is \emph{backtracking line search} with the Armijo sufficient-decrease condition. Starting from $\alpha = 1$, repeatedly halve $\alpha$ until:
\begin{equation}
f\!\left(x_k - \alpha\, p_k\right) \leq f(x_k) - c_1\, \alpha\, \nabla f(x_k)^T p_k, \qquad c_1 \in (0, 1)\ (\text{typically } 10^{-4}).
\end{equation}
Since the Newton direction $p_k = H_k^{-1} g_k$ is a descent direction whenever $H_k \succ 0$ (i.e.\ $g_k^T p_k = g_k^T H_k^{-1} g_k > 0$), the Armijo condition is always satisfiable by shrinking $\alpha$. The \emph{strong Wolfe curvature condition} $|\nabla f(x_k - \alpha_k p_k)^T p_k| \leq c_2|\nabla f(x_k)^T p_k|$ (with $c_2 \in (c_1, 1)$, typically $0.9$ for Newton) additionally prevents excessively short steps. The key advantage of line search over fixed $\alpha$: far from $x^\star$, the step shrinks automatically to prevent overshooting; near $x^\star$, the unit step $\alpha_k = 1$ is accepted (since the quadratic model is accurate there), automatically restoring quadratic convergence. For Rosenbrock, a line-search Newton would take increasingly large accepted steps as the iterate approaches $(1,1)$, whereas the fixed $\alpha = 0.22$ permanently limits per-iteration progress. The inner loop adds only $O(\log_2(1/\alpha_{\min}))$ function evaluations per outer step and is the standard practice in production Newton solvers.

\textbf{Trust region methods:} A complementary globalisation strategy restricts the Newton step to a \emph{trust region} ball $\|p_k\| \leq \Delta_k$ rather than performing a line search. The local subproblem is:
\begin{equation}
p_k^\star = \arg\min_{\|p\| \leq \Delta_k}\; m_k(p) = f(x_k) + g_k^T p + \tfrac{1}{2}\, p^T H_k\, p.
\end{equation}
The radius $\Delta_k$ is adapted by comparing actual decrease $\rho_k^{\text{act}} = f(x_k) - f(x_k + p_k)$ to predicted decrease $\rho_k^{\text{pred}} = m_k(0) - m_k(p_k^\star)$. If $\rho_k = \rho_k^{\text{act}}/\rho_k^{\text{pred}} \approx 1$, the model is accurate and $\Delta_{k+1}$ is expanded; if $\rho_k < 0.1$, the step is rejected and $\Delta_{k+1}$ is shrunk. Key advantages over line search: (i)~trust region methods are globally convergent to a first-order stationary point for any smooth $f$ (including non-convex) even when $H_k$ is indefinite; (ii)~when the constraint $\|p\| \leq \Delta_k$ is active, the solution moves to the boundary and can escape saddle points, whereas pure Newton can be attracted to saddle points if $H_k$ has negative eigenvalues. For $d = 2$, the subproblem is solved exactly in $O(1)$; for large $d$, the Steihaug truncated conjugate gradient solves it in $O(d)$ Hessian-vector products (the dominant cost for large-scale Newton). Trust region and line search methods are largely equivalent in theory; trust region is more robust when Hessians are frequently indefinite (non-convex deep learning training) or poorly scaled.

\textbf{Comparison with gradient descent:} GD's linear convergence rate $O\!\left((1-\mu/L)^k\right)$ depends on the condition number $\kappa = L/\mu$. For Benchmark~C with $\kappa \approx 2508$, GD converges at rate $\approx (1 - 1/2508)^k \approx 0.9996^k$ --- reaching $\varepsilon$-accuracy requires $O(\kappa \log(1/\varepsilon))$ iterations. Newton's quadratic convergence requires only $O(\log\log(1/\varepsilon))$ iterations in the local regime, independent of $\kappa$.

\textbf{Settings:} GD uses 80 iterations; Newton uses only 20 iterations, reflecting the superior per-iteration progress. Tikhonov damping $\lambda = 10^{-8}$ ensures $H_k + \lambda I \succ 0$ (positive definite) without materially affecting the Newton direction when $H_k$ is itself well-conditioned.

\textbf{Scalability and per-evaluation cost:} Newton's per-iteration cost is $O(d^2)$ for computing the Hessian and $O(d^3)$ for solving the linear system $H_k p_k = g_k$ (Cholesky factorisation). To compare methods fairly, one should consider \emph{total flops to reach $\varepsilon$-accuracy}, not just iteration count:
\begin{itemize}[nosep]
  \item \textbf{GD} (Benchmark~C, $\kappa \approx 2508$): rate $(1-\mu/L)^k \approx 0.9992^k$, cost per iter $O(d)$. Iterations needed: $\ln(1/\varepsilon)/\ln(1/0.9992) \approx 1250 \ln(1/\varepsilon)$. Total: $O(d \cdot \kappa \ln(1/\varepsilon))$.
  \item \textbf{Newton} (local quadratic regime): iterations $O(\log\log(1/\varepsilon))$, cost $O(d^3)$. Total: $O(d^3)$ (essentially independent of $\varepsilon$). For $d = 2$: $O(8)$ flops; for $d = 1000$: $O(10^9)$ flops --- impractical.
  \item \textbf{L-BFGS} ($m = 10$ vectors): superlinear rate, cost $O(md)$ per iter $\approx O(10d)$. Total: $O(md \cdot \text{few iterations})$ --- the practical sweet spot for $d \in [10^3, 10^7]$.
\end{itemize}
For the 2-dimensional benchmarks here ($d = 2$), Newton's $O(d^3) = O(8)$ per-iteration overhead is completely negligible --- the bottleneck is the number of iterations and gradient evaluations. This is why Newton's 20 iterations with $\alpha = 0.22$ easily outperforms GD's 80 iterations. L-BFGS is motivated only by large $d$.

For a neural network with $d = 10^8$ parameters, Newton would require $10^{24}$ flops to form and factor the Hessian --- completely infeasible. L-BFGS at $m=20$, $d=10^8$ requires $20 \times 10^8 = 2 \times 10^9$ flops per iteration, feasible on a GPU.

\textbf{Hessian-vector products for large-scale Newton:} Even forming the full Hessian $H \in \mathbb{R}^{d \times d}$ is impractical for large $d$. However, solving the Newton system $H_k p_k = g_k$ by \emph{conjugate gradient} (CG) requires only \emph{Hessian-vector products} $H_k v$ (not the full Hessian). A Hessian-vector product can be computed via double-pass automatic differentiation in the cost of 2 backward passes:
\begin{equation}
H(x)\,v = \frac{d}{d\varepsilon}\!\bigl[\nabla f(x + \varepsilon v)\bigr]\!\Bigr|_{\varepsilon=0} \approx \frac{\nabla f(x + \varepsilon v) - \nabla f(x)}{\varepsilon}
\end{equation}
(forward-mode AD on the gradient). CG solves $Hp = g$ in at most $d$ iterations (exact arithmetic), where each CG step requires one Hv product. The \emph{truncated Newton} (TN) method stops CG early at $k_{\text{CG}} \ll d$ steps, giving an \emph{inexact} Newton step with per-outer-iteration cost $O(k_{\text{CG}} \cdot \text{cost}(Hv)) = O(k_{\text{CG}} \cdot \text{cost}(\nabla^2 f))$. This avoids the $O(d^3)$ Cholesky solve: for $d = 10^5$, full Newton costs $10^{15}$ flops while TN with $k_{\text{CG}} = 100$ costs $\sim 200$ gradient evaluations --- feasible on a GPU. TN with CG is the basis of modern large-scale second-order methods in deep learning (e.g., Hessian-free optimisation, Khatri--Rao for structured Hessians).

\textbf{Gauss--Newton for least squares (Benchmark A):} For residual-based objectives $J(\theta) = \frac{1}{2m}\|r(\theta)\|^2$ where $r(\theta) = X\theta - y$ is the residual vector, Newton's Hessian decomposes as:
\begin{equation}
\nabla^2 J(\theta) = \frac{1}{m}J_r^T J_r + \frac{1}{m}\sum_{i=1}^m r_i(\theta)\, \nabla^2 r_i(\theta),
\end{equation}
where $J_r$ is the Jacobian of $r$. The \emph{Gauss--Newton} (GN) approximation drops the second term (weighted by residuals $r_i$, which are small near the optimum) to obtain:
\begin{equation}
H_{\text{GN}} = \frac{1}{m}J_r^T J_r \approx \nabla^2 J(\theta).
\end{equation}
The GN step is $p_k = H_{\text{GN}}^{-1}\nabla J(\theta_k)$. For Benchmark~A, the model is \emph{linear} in $\theta$ (i.e., $r(\theta) = X\theta - y$), so $J_r = X$ (constant) and $\nabla^2 r_i = 0$ for all $i$: \textbf{Gauss--Newton is exactly Newton} for linear regression, and the GN step gives $\theta_{\text{OLS}} = (X^T X)^{-1}X^T y$ in one iteration. For nonlinear models (e.g., a neural network with sigmoid activations), GN drops the second-derivative term at the cost of a biased Hessian approximation, but remains positive semi-definite $H_{\text{GN}} = J_r^T J_r / m \succeq 0$ even when the true Hessian is indefinite, making it more numerically stable. GN is also the basis of the \emph{Levenberg--Marquardt} algorithm for nonlinear least squares: $p_k = (H_{\text{GN}} + \lambda I)^{-1}\nabla J(\theta_k)$, which interpolates between GN ($\lambda \to 0$) and steepest descent ($\lambda \to \infty$), adjusting $\lambda$ via a trust-region-like update.

\subsection{Part A: Finite Differences and Nesterov Random Search on Benchmark B}

\subsubsection{Finite Difference Gradient Approximation}

When analytical gradients are unavailable, they can be approximated using function evaluations.
\textbf{Forward finite difference} (one-sided):
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\label{eq:fd}
\end{equation}
\textbf{Centered finite difference} (two-sided, higher order):
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}.
\label{eq:cfd}
\end{equation}
The total approximation error has two competing sources:
\begin{itemize}[nosep]
\item \textbf{Truncation error:} $O(\delta)$ for forward FD, $O(\delta^2)$ for centered FD (the antisymmetric first-order terms cancel in the centered case, leaving only third-order remainders).
\item \textbf{Floating-point rounding error:} $O(\varepsilon_{\text{mach}}/\delta)$ for both, from subtractive cancellation in $f(x + \delta e_i) - f(x \pm \delta e_i)$.
\end{itemize}
Balancing truncation and rounding gives: forward FD optimal $\delta_{\text{opt}} \approx \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$; centered FD optimal $\delta_{\text{opt}} \approx \varepsilon_{\text{mach}}^{1/3} \approx 10^{-5.3} \approx 5 \times 10^{-6}$ (a larger safe range for $\delta$). The assignment tests $\delta = 0.05$ (accurate for forward FD) and $\delta = 0.8$ (truncation error $\approx O(\delta L_2 / 2) \approx O(4)$, dominated by the second-order Taylor remainder for forward FD). For a $d$-dimensional problem, forward FD requires $d + 1$ function evaluations per step; centered FD requires $2d$ evaluations but achieves the $O(\delta^2)$ truncation advantage. Figure~\ref{fig:q4_fd_error} confirms this analysis numerically on Benchmark~B.

\textbf{Complex step method --- avoiding cancellation entirely:} A superior alternative to finite differences for gradient computation is the \emph{complex step} method. Perturbing $x_i$ by a purely imaginary step $ih$:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{\mathrm{Im}\bigl[f(x + ih\, e_i)\bigr]}{h}.
\end{equation}
By the Taylor expansion $f(x + ihe_i) = f(x) + ih f'(x) - \frac{h^2}{2}f''(x) + O(h^3)$, taking the imaginary part gives $\mathrm{Im}[f(x+ihe_i)] = h f'(x) + O(h^3)$, so the error is $O(h^2)$ --- \emph{without any subtractive cancellation}, since the imaginary and real parts of $f$ live in separate channels and never cancel. This means $h$ can be set as small as $10^{-50}$ (or even machine epsilon $\varepsilon_{\text{mach}} \approx 10^{-16}$) without any precision loss, yielding near-exact gradients. The forward FD, centered FD, complex step, and automatic differentiation methods are compared in Table~\ref{tab:grad_methods}. The complex step requires the function to accept complex-valued inputs (satisfied automatically by NumPy operations but not by compiled C/Fortran code or external simulation solvers); when applicable, it is the preferred method for derivative validation and gradient checking, having replaced finite differences in many scientific computing workflows.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$, seed
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta\, u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Nesterov random search is a zeroth-order method: it requires only \emph{two} function evaluations per iteration regardless of dimension $d$, making it \emph{dimension-independent} in cost per step. The gradient estimate $\hat{g}_k$ is a one-sided directional derivative along a random unit direction. Its expected value satisfies:
\begin{equation}
\mathbb{E}_u[\hat{g}_k] = \frac{f(x+\delta u) - f(x)}{\delta}\, u \approx \mathbb{E}_u[(\nabla f(x)^T u)\, u] = \frac{1}{d}\nabla f(x),
\end{equation}
where the last equality uses the identity $\mathbb{E}_{u \sim \mathrm{Unif}(\mathbb{S}^{d-1})}[u u^T] = \frac{1}{d}I_d$. Thus the method is unbiased up to a factor of $d$ and the $O(\delta)$ truncation bias from the finite difference approximation. Effectively, the algorithm (without a $d$ rescaling factor) makes expected steps of size $\alpha/d$ relative to the gradient magnitude, compared to $\alpha$ for exact GD. To match GD's per-iteration expected progress, one would need $\alpha_{\text{NRS}} = d \cdot \alpha_{\text{GD}}$. In practice, however, the high variance of the single-direction estimate (which grows as $\Theta(\|\nabla f\|^2)$ per step) requires a \emph{smaller} step size for stability --- NRS uses $\alpha = 0.025$ versus GD's $\alpha = 0.06$ on Benchmark~B, reflecting this variance-induced stability constraint. The variance of $\hat{g}_k$ is $O(\|\nabla f(x_k)\|^2)$, which does not decrease in high dimensions relative to the bias (unlike the full-gradient case), and is the principal bottleneck for NRS convergence speed.

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$. At each iteration, the worst vertex (highest function value) is replaced by a new point determined by one of four geometric operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): move worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2$): if reflection improved, attempt a further step in the same direction.
\item \textbf{Outside contraction} ($\rho = 0.5$): if reflection is moderate, contract between centroid and reflected point.
\item \textbf{Inside contraction} and \textbf{shrink} ($\sigma = 0.5$): if all contractions fail, shrink the entire simplex towards the best vertex.
\end{enumerate}
No gradient information is required. The method exploits the simplex geometry to probe the landscape and adapts its shape to the local topology. Convergence is generally sub-linear and there are no global convergence guarantees, but empirically it works well for smooth low-dimensional problems.

\textbf{Simplex geometry and Rosenbrock:} The initial simplex around $x_0 = (-1, 1)$ has vertices at $(-1, 1)$, $(-1+0.35, 1)$, $(-1, 1+0.35)$. As the simplex descends, it deforms to probe the narrow banana valley: the \emph{reflection} step moves the worst vertex to the opposite side of the centroid (probing the function in the descent direction), while \emph{contraction} shrinks the simplex to align with the valley floor. The result is that by iteration $\sim$50, the simplex has elongated to lie \emph{along} the valley $x_2 \approx x_1^2$, effectively performing line searches along the valley floor without any explicit gradient computation. This geometric adaptation is what allows Nelder--Mead to achieve $f \approx 10^{-29}$ (machine precision) --- the simplex contracts until its diameter is below double-precision resolution.

\textbf{Limitations:} Nelder--Mead is unreliable for $d \geq 10$ because the $d+1 = 11$ simplex vertices cannot efficiently probe a $d$-dimensional landscape (the volume of the $d$-simplex relative to the containing hypercube decreases as $d!/d^d \to 0$). It is also sensitive to the initial simplex size: too large causes missed features; too small gives slow progress. The $\text{step}=0.35$ used here was chosen empirically.

Settings: initial simplex step $= 0.35$, iterations $= 160$.

\textbf{Key observations:}
\begin{itemize}[nosep]
\item For high-dimensional problems ($d \gg 1$), FD requires $d + 1$ evaluations per step versus 2 for Nesterov random search, which is dimension-independent. However, FD provides a biased gradient estimate with $O(\delta)$ error per component, whereas Nesterov random search's estimate has zero bias in expectation (it estimates a directional derivative, not the full gradient) but high variance.
\item Nelder--Mead and grid search do not scale: Nelder--Mead is typically reliable only for $d \leq 10$--$20$, and grid search is intractable beyond $d \approx 3$ due to the curse of dimensionality ($55^{10} \approx 2.5 \times 10^{17}$ evaluations for $d=10$).
\item When analytical gradients are available, the cost of gradient computation is typically $O(nd)$ for loss functions summing over $n$ data points --- the same order as one function evaluation --- so the per-iteration cost of exact GD is comparable to FD GD but without the approximation error.
\item \textbf{Finite difference vs.\ automatic differentiation:} In modern machine learning, automatic differentiation (AD/backpropagation) computes \emph{exact} gradients at cost $O(d)$ forward passes by applying the chain rule symbolically to the computation graph. This completely supersedes FD for differentiable code: for a neural network with $d = 10^7$ parameters, FD would require $10^7 + 1$ forward passes per gradient versus AD's 1 backward pass. FD remains relevant when: (i) code is black-box (legacy Fortran/C code, simulation software); (ii) the function is not programmatically differentiable (discontinuities, non-differentiable operations like sorting); or (iii) quick prototyping without implementing a backward pass. Nesterov random search is the FD analog for very high dimensions where even $d+1$ evaluations are too expensive.
\item \textbf{Bayesian optimisation} is the modern standard for expensive black-box functions (where each evaluation costs minutes or hours: hyperparameter tuning, molecular design, simulation code). It maintains a Gaussian process (GP) surrogate model $\hat{f}(x) \sim \mathcal{GP}(\mu(x), k(x, x'))$ over all past evaluations, and selects the next evaluation point by maximising an \emph{acquisition function} that balances exploration (high GP uncertainty $\sigma(x)$) and exploitation (low GP mean $\mu(x)$). Common acquisition functions include Upper Confidence Bound (UCB): $a_{\text{UCB}}(x) = \mu(x) + \sqrt{\beta_t}\,\sigma(x)$; Expected Improvement (EI): $a_{\text{EI}}(x) = \mathbb{E}[\max(f^\star - \hat{f}(x), 0)]$; and Thompson sampling. With a squared-exponential GP kernel, Bayesian optimisation achieves sub-linear \emph{cumulative regret} bounds in $d$ dimensions. The limitation is GP inference: adding each new point costs $O(n^3)$ (matrix inversion of the $n \times n$ kernel matrix for $n$ past evaluations), limiting scalability to $n \leq 10^3$--$10^4$ evaluations. For Rosenbrock with cheap function evaluations, Bayesian optimisation is not preferred over Nelder--Mead or random search; but for functions costing $\geq 10^2$ seconds per evaluation, it is typically the most sample-efficient method.
\end{itemize}

\subsection{Problem Formulation}

Minimise Benchmark~B subject to a linear inequality constraint:
\begin{equation}
\min_{x \in \mathbb{R}^2}\ f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad\text{subject to}\quad x_1 \geq 0.5.
\end{equation}
The starting point $x_0 = (0.2, 4.0)$ is \emph{infeasible} ($x_1 = 0.2 < 0.5$). Note that the unconstrained minimum of $f$ is at $x^\star \approx (0.582, 2.0)$ with $x_1^\star \approx 0.582 > 0.5$: the unconstrained optimum is already \emph{interior} to the feasible set. Thus, the true constrained and unconstrained optima coincide, and the constraint is \emph{inactive} at the optimum. However, the trajectory from $x_0 = (0.2, 4.0)$ may pass through the infeasible region, and the different methods handle this differently.

\textbf{KKT conditions:} For the problem $\min_x f(x)$ s.t.\ $h(x) \leq 0$ (where $h(x) = 0.5 - x_1$), the KKT conditions require:
\begin{equation}
\nabla f(x^\star) + \mu\, \nabla h(x^\star) = 0, \quad \mu \geq 0, \quad \mu h(x^\star) = 0.
\end{equation}
With $\nabla h = -e_1$: $\nabla f_1(x^\star) = \mu$ and $\nabla f_2(x^\star) = 0$. Since the unconstrained optimum satisfies $x_1^\star \approx 0.582 > 0.5$ (strictly interior to $\mathcal{F}$), the complementary slackness condition gives $\mu = 0$ (constraint is inactive), and the KKT conditions reduce to $\nabla f(x^\star) = 0$ --- the unconstrained stationary point is automatically KKT-optimal. The Lagrangian for this problem is:
\begin{equation}
\mathcal{L}(x, \mu) = f(x) + \mu(0.5 - x_1), \qquad \mu \geq 0.
\end{equation}
\textbf{Strong duality:} Since $f$ is convex and the constraint $h(x) = 0.5 - x_1 \leq 0$ is linear (hence convex), the Slater constraint qualification holds (take any $x$ with $x_1 > 0.5$), guaranteeing $p^\star = d^\star$ where $p^\star$ and $d^\star$ are the primal and dual optimal values. At the constrained optimum $x^\star$ with $\mu^\star = 0$: $f(x^\star) = \mathcal{L}(x^\star, 0) = p^\star = d^\star$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection operator $\Pi_{\mathcal{F}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{F}}(z_{k+1})$ where $\Pi_{\mathcal{F}}(z) = (\max(0.5, z_1),\, z_2)$ \Comment{Project onto $\{x : x_1 \geq 0.5\}$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection $\Pi_{\mathcal{F}}(z) = \arg\min_{x \in \mathcal{F}} \|x - z\|^2$ for the half-space $\{x : x_1 \geq 0.5\}$ reduces to a simple clipping operation: $\Pi_\mathcal{F}(z) = (\max(0.5, z_1),\, z_2)^T$. This $O(1)$ operation makes PGD very efficient for halfspace constraints. For more complex feasible sets, the projection cost increases:
\begin{itemize}[nosep]
  \item \textbf{Box constraints} $\{x : l \leq x \leq u\}$: $O(d)$ per coordinate, same as halfspace.
  \item \textbf{Simplex} $\{x \geq 0 : \mathbf{1}^T x = 1\}$: $O(d \log d)$ via sorting.
  \item \textbf{$\ell^2$ ball} $\{x : \|x\| \leq r\}$: $O(d)$ via $x \mapsto r \cdot x/\|x\|$.
  \item \textbf{Positive semidefinite cone} $\{X \succeq 0\}$: $O(d^3)$ via eigendecomposition.
\end{itemize}
For large-scale problems with expensive projections (e.g., nuclear norm ball, flow polytopes), Frank--Wolfe (Section~\ref{sec:q6}) is preferred. PGD guarantees feasibility at every iterate.

\textbf{PGD as proximal gradient descent:} Projected GD is a special case of the more general \emph{proximal gradient method}, which handles objectives of the form $F(x) = f(x) + h(x)$ where $f$ is smooth and $h$ is non-smooth but has a tractable \emph{proximal operator}:
\begin{equation}
\mathrm{prox}_{\alpha h}(z) = \arg\min_{x}\left\{ h(x) + \tfrac{1}{2\alpha}\|x - z\|^2 \right\}.
\end{equation}
The proximal gradient update is $x_{k+1} = \mathrm{prox}_{\alpha h}(x_k - \alpha \nabla f(x_k))$. For the indicator function $h(x) = \iota_{\mathcal{F}}(x)$ (which equals $0$ if $x \in \mathcal{F}$ and $+\infty$ otherwise), the proximal operator becomes exactly the Euclidean projection $\Pi_\mathcal{F}$, recovering projected GD. For $\ell^1$ regularization $h(x) = \lambda\|x\|_1$ (used in LASSO for sparse regression), the proximal operator is the soft-thresholding function $S_{\alpha\lambda}(z)_i = \text{sign}(z_i)\max(|z_i| - \alpha\lambda, 0)$, giving the ISTA algorithm. Adding Nesterov momentum to the proximal step gives FISTA (Fast ISTA), achieving the optimal $O(1/k^2)$ rate for non-smooth convex $f + h$. This framework unifies PGD (Q5), Nesterov (Q2), and FISTA under the single proximal gradient paradigm.

\begin{algorithm}[H]
\caption{Penalty Method (Linear Penalty)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_\lambda(x) \gets f(x) + \lambda\max(0,\; 0.5 - x_1)$ \Comment{Penalised objective}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method adds a soft violation cost proportional to $\lambda$ and does not guarantee feasibility. The effective gradient includes an extra $-\lambda$ in the $x_1$ direction when $x_1 < 0.5$ (penalty gradient: $\nabla_x[\lambda\max(0, 0.5-x_1)] = -\lambda e_1$ if $x_1 < 0.5$, else $0$). As $\lambda \to \infty$, the penalised solution converges to the constrained solution, but large $\lambda$ requires smaller step sizes to maintain stability.

Settings: PGD uses $\alpha = 0.08$. Penalty uses $\alpha = 0.05$ for $\lambda \in \{0.15, 1.8\}$ and $\alpha = 0.03$ for $\lambda = 4.5$ (to compensate for the larger effective Lipschitz constant). Unconstrained GD uses $\alpha = 0.07$. All run for 100 iterations.

In this experiment, all methods eventually converge to the same feasible minimum because the unconstrained and constrained optima coincide. The key differences lie in \emph{how} feasibility is enforced and the convergence rate during the initial infeasible phase:

\textbf{Projected GD} enforces feasibility as a hard constraint immediately (from iteration~1), making it the only method that never visits an infeasible point after initialisation. It is exact, computationally cheap for simple constraints ($O(1)$ projection), and does not require tuning $\lambda$. The projection onto $\{x : x_1 \geq 0.5\}$ is trivial here, but for complex convex constraint sets the projection can be computationally expensive.

\textbf{Penalty method} requires only a modified gradient and no projection oracle. The trade-off is that feasibility is enforced softly: $\lambda$ must be chosen large enough to drive the iterate to the feasible set, but too large a $\lambda$ amplifies the effective Lipschitz constant to $L(F_\lambda) = L(f) + \lambda$, requiring stable step size $\alpha \leq 2/(L(f) + \lambda)$ --- consistent with the smaller $\alpha = 0.03$ used for $\lambda = 4.5$.

\textbf{Penalty parameter continuation:} Rather than choosing a fixed $\lambda$, a practical strategy is the \emph{continuation} (homotopy) approach: solve a sequence of penalised subproblems with increasing $\lambda_0 < \lambda_1 < \cdots < \lambda_K$ (e.g., $\lambda_{j+1} = 2\lambda_j$), warm-starting each subproblem from the previous solution. This is beneficial for three reasons: (i)~a small initial $\lambda$ keeps the effective Lipschitz constant $L + \lambda$ small, allowing larger step sizes and faster early progress; (ii)~warm starting reduces the number of inner iterations (since the solution changes only slightly between consecutive $\lambda$-levels); (iii)~the final iterate at $\lambda_K \gg L(f)$ is nearly feasible. The Q5 experiment with $\lambda \in \{0.15, 1.8, 4.5\}$ corresponds to two doublings ($\times 12$ and $\times 2.5$), illustrating the first three levels of a continuation. This approach is the basis of \emph{interior-point} (barrier) methods. The log-barrier function $B_t(x) = f(x) - (1/t)\log(x_1 - 0.5)$ adds an infinite penalty for approaching the constraint boundary and is differentiable everywhere in the interior $\{x : x_1 > 0.5\}$. As $t \to \infty$, the minimiser $x^\star(t)$ of $B_t$ traces the \emph{central path} towards the true constrained optimum $x^\star$: the duality gap satisfies $f(x^\star(t)) - f^\star = (m/t)$ for $m$ inequality constraints. Each central-path point is found by Newton's method (since $B_t$ is smooth and strongly convex near the path), and the outer loop increments $t$ (typically $t \leftarrow \mu t$ with $\mu \approx 10$). The resulting \emph{path-following} algorithm solves the constrained problem in $O(\sqrt{m}\log(m/\varepsilon))$ Newton steps (Nesterov--Nemirovskii polynomial complexity), far fewer than the $O(m/\varepsilon)$ iterations of the penalty method. Interior-point methods are the gold standard for LP and QP in practice (MOSEK, Gurobi, IPOPT).

\textbf{Quantitative constraint violation bound:} For a problem where the unconstrained minimum is infeasible (i.e., the constraint is \emph{active} at the true minimum), the penalty minimiser $x^\star_\lambda$ satisfies $g(x^\star_\lambda) = \max(0, 0.5 - x_1^\star_\lambda) = O(1/\lambda)$ as $\lambda \to \infty$. This can be seen from the stationarity condition: at $x^\star_\lambda$, the penalty gradient $-\lambda e_1$ (active when $x_1 < 0.5$) must balance the objective gradient $\nabla f_1 = 2(x_1-1) + \cos(x_1) \approx -1 + \cos(0.5) \approx -0.12$ at the boundary. In the present experiment, the unconstrained minimum is \emph{inside} the feasible set ($x_1^\star \approx 0.582 > 0.5$), so the penalty minimiser satisfies $g(x^\star_\lambda) = 0$ for any $\lambda > 0$, explaining why all three $\lambda$ values achieve zero constraint violation at convergence (Table~\ref{tab:q5_final}).

An improvement over the pure penalty approach is the \emph{augmented Lagrangian} (multiplier) method, which combines a quadratic penalty with a Lagrange multiplier estimate:
\begin{equation}
L_\rho(x, \mu) = f(x) + \mu\,\max(0,\, 0.5 - x_1) + \frac{\rho}{2}\bigl[\max(0,\, 0.5 - x_1)\bigr]^2.
\end{equation}

\begin{algorithm}[H]
\caption{Augmented Lagrangian / Method of Multipliers}
\label{alg:aug_lag}
\begin{algorithmic}[1]
\Require Initial point $x_0$, penalty $\rho > 0$, multiplier $\mu_0 = 0$, iterations $K$
\For{$k = 0, 1, \ldots, K-1$}
    \State $x_{k+1} \gets \arg\min_x\; L_\rho(x,\, \mu_k)$ \Comment{Unconstrained inner minimisation}
    \State $\mu_{k+1} \gets \mu_k + \rho\,\max(0,\, 0.5 - (x_{k+1})_1)$ \Comment{Multiplier ascent step}
\EndFor
\end{algorithmic}
\end{algorithm}

The key advantage over the pure penalty method is that the augmented Lagrangian provides \emph{exact constraint satisfaction at a finite $\rho$}: as the multiplier $\mu_k$ converges to the true KKT multiplier $\mu^\star$, the inner minimisation at fixed $\rho$ achieves the constrained minimum exactly (rather than requiring $\rho \to \infty$). The inner problem (Algorithm~\ref{alg:aug_lag}, line 2) can be solved by a few steps of gradient descent since it is an unconstrained smooth problem for fixed $\mu_k$.

\textbf{ADMM (Alternating Direction Method of Multipliers):} The augmented Lagrangian is the foundation of ADMM, which splits the primal variable into $(x, z)$ linked by a consensus constraint $Ax + Bz = c$. The ADMM updates are:
\begin{align}
x^{(k+1)} &\gets \arg\min_x\; L_\rho(x,\, z^{(k)},\, \mu^{(k)}), \label{eq:admm_x}\\
z^{(k+1)} &\gets \arg\min_z\; L_\rho(x^{(k+1)},\, z,\, \mu^{(k)}), \label{eq:admm_z}\\
\mu^{(k+1)} &\gets \mu^{(k)} + \rho\,(A x^{(k+1)} + B z^{(k+1)} - c). \label{eq:admm_mu}
\end{align}
By alternating between the $x$- and $z$-subproblems (which can often be solved in closed form or independently), ADMM decouples objectives that would otherwise be hard to minimise jointly. The canonical application is \emph{distributed learning} over $N$ machines: assign $x_i$ = local model on machine $i$, $z$ = global consensus variable, and constraint $x_i = z$ for all $i$. Each machine solves its local $x_i$-subproblem using its own data, then sends the result to a parameter server that computes the $z$-update (global average). The multiplier $\mu_i$ update enforces consensus across machines. This structure enables parallelism: all $N$ machines update $x_i$ simultaneously, with $O(1)$ communication rounds per ADMM iteration regardless of dataset size. ADMM converges at $O(1/k)$ for convex $f$ (Boyd et al., 2011) and achieves linear convergence when both $f$ and $g$ are strongly convex. In the present Q5 setting (single constraint $x_1 \geq 0.5$), the $x$- and $z$-splits reduce to the augmented Lagrangian described above, making ADMM equivalent to Algorithm~\ref{alg:aug_lag}.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box polytope).

\textbf{Linear Programme:} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$.

Since $a = [1, 2]^T$ has all positive components, the minimum is attained at the lower bound of every coordinate:
\begin{equation}
x^\star = (0.5,\; -5), \qquad f^\star = 0.5 + 2 \times (-5) = -9.5.
\end{equation}
This illustrates the \textbf{fundamental theorem of linear programming}: the optimum of a linear objective over a bounded polytope is always attained at a vertex (\emph{basic feasible solution}). The proof is by contradiction: if $x^\star$ is interior, then for any nonzero direction $d$, one of $x^\star \pm d$ has lower objective, so we can improve by moving along $a$'s descent direction until hitting a face, then an edge, then a vertex. For the box $\mathcal{X}$ with vertices at the $2^d$ corners $\{l_i, u_i\}^d$, there are only $2^2 = 4$ vertices to check in 2D; the LP is solved in $O(d)$ by coordinate-wise argmin. The LP subproblem solved by Frank--Wolfe for box constraints is:
\begin{equation}
\arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z \quad\Leftrightarrow\quad
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
where $[l_i, u_i]$ are the box bounds. This is an $O(d)$ operation per coordinate, making Frank--Wolfe very efficient for box constraints.

\subsection{Frank--Wolfe Algorithm (Conditional Gradient)}

\begin{algorithm}[H]
\caption{Frank--Wolfe / Conditional Gradient Method}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Feasible $x_0 \in \mathcal{X}$, fixed mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LP subproblem)}
    \State $x_{k+1} \gets \beta\, x_k + (1-\beta)\, z_k$ \Comment{Convex combination; equivalently $x_k + \gamma_k(z_k - x_k)$ with $\gamma_k = 1-\beta$}
\EndFor
\end{algorithmic}
\end{algorithm}

The Frank--Wolfe algorithm (1956) maintains feasibility automatically: since $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$, any convex combination $\beta x_k + (1-\beta)z_k \in \mathcal{X}$ (by convexity of $\mathcal{X}$). No projection is needed. The step $\gamma_k = 1 - \beta$ is fixed here; the standard decaying schedule $\gamma_k = 2/(k+2)$ achieves the optimal $O(1/k)$ convergence rate for smooth convex $f$ over a compact convex set:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\,\mathrm{diam}(\mathcal{X})^2}{k+2},
\end{equation}
where $L$ is the Lipschitz constant of $\nabla f$ and $\mathrm{diam}(\mathcal{X}) = \|u - l\|$ for box bounds. With fixed $\gamma = 1 - \beta$, convergence is guaranteed as long as $z_k$ converges to a fixed point, but the oscillatory behaviour of $z_k$ (which jumps between vertices) can prevent exact convergence at a fixed step.

\textbf{Proof sketch of $O(1/k)$ rate for the decaying schedule $\gamma_k = 2/(k+2)$:} Since $f$ is $L$-smooth and $x_{k+1} = x_k + \gamma_k(z_k - x_k)$, applying the descent lemma gives:
\begin{align}
f(x_{k+1}) &\leq f(x_k) + \gamma_k\, \nabla f(x_k)^T(z_k - x_k) + \frac{L\gamma_k^2}{2}\|z_k - x_k\|^2 \\
&\leq f(x_k) - \gamma_k G_k + \frac{L\gamma_k^2}{2}\,\mathrm{diam}(\mathcal{X})^2,
\end{align}
where $G_k = \nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - f^\star$ (FW gap lower-bounds suboptimality). Subtracting $f^\star$ and denoting $h_k = f(x_k) - f^\star$:
\begin{equation}
h_{k+1} \leq (1-\gamma_k)\, h_k + \frac{L\gamma_k^2}{2}\,\mathrm{diam}(\mathcal{X})^2.
\end{equation}
One verifies by induction with $\gamma_k = 2/(k+2)$ and $C = 2L\,\mathrm{diam}(\mathcal{X})^2$ that $h_k \leq C/(k+2)$: the initial case $h_0 \leq C/2$ follows from the bound on $f(x_0) - f^\star$; the inductive step uses $(1-2/(k+2))\cdot C/(k+2) + (L\cdot 4/(k+2)^2 \cdot C/(2L)) = C/(k+3)$. This confirms the $O(1/k)$ rate and shows that the choice $\gamma_k = 2/(k+2)$ is \emph{optimal} for the decaying-step schedule.

\textbf{Sparsity-inducing property:} After $k$ iterations, the Frank--Wolfe iterate is a convex combination of \emph{at most $k+1$ vertices} of $\mathcal{X}$ (since each step adds at most one vertex $z_k$ to the representation). For the $\ell^1$ ball $\mathcal{X} = \{x : \|x\|_1 \leq \tau\}$, the vertices are $\pm \tau e_i$ (coordinate vectors), so FW naturally produces \emph{sparse} solutions: $x_k = \sum_{i \leq k} \gamma_i \cdot (\pm \tau e_{j_i})$ has at most $k+1$ nonzero components. This structural property makes FW a natural choice for compressed sensing, sparse signal recovery, and matrix completion problems where sparse or low-rank solutions are desired.

\textbf{Frank--Wolfe gap} as a convergence certificate:
\begin{equation}
G_k = \nabla f(x_k)^T (x_k - z_k) \geq f(x_k) - f^\star \geq 0.
\end{equation}
The gap $G_k$ is computable without knowing $f^\star$ and provides an upper bound on the current suboptimality. When $G_k \approx 0$, convergence is certified.

\subsection{Interior vs.\ Boundary Behaviour --- Analysis}

\textbf{Interior optimum:} Frank--Wolfe converges to the interior minimum, but via a zig-zag path: at each step, the LP oracle selects a vertex of $\mathcal{X}$ based on the current gradient, and the convex combination slowly moves the iterate towards that vertex. Near the interior minimum $(1, 5)$, the gradient $\nabla f(x_k) \approx 2(x_k - (1, 5))$ is small and may alternate sign in individual components, causing the LP oracle to select alternating vertices. The fixed mixing parameter $\gamma = 1-\beta$ prevents these oscillations from being averaged away. The $O(1/k)$ rate is confirmed by the approximate power-law decrease visible in the convergence plots.

The \textbf{Frank--Wolfe gap} at the final iterate can be computed as:
\begin{equation}
G_k = \nabla f(x_k)^T(x_k - z_k) = \sum_i (\nabla f(x_k))_i \cdot [(x_k)_i - (z_k)_i] \geq f(x_k) - f^\star,
\end{equation}
providing a certificate of near-optimality without knowing $f^\star$.

\textbf{Boundary optimum:} Frank--Wolfe is particularly suited to boundary optima. When the unconstrained minimum $(0, 0)$ lies outside $\mathcal{X}$ (violating $x_1 \geq 0.5$), the gradient at $x_k$ near the $x_1 = 0.5$ boundary has $(\nabla f(x_k))_1 = 2(x_k)_1 \approx 1 > 0$, so the LP oracle \emph{consistently} selects $z_1 = 0.5$ (lower bound). This drives $x_k$ towards the boundary naturally, without projection. However, the $x_2$ component oscillates: the LP oracle selects $z_2 = 10$ when $x_2 < 0$ (since $(\nabla f)_2 = 2x_2 < 0$) and $z_2 = -5$ when $x_2 > 0$, causing zigzagging in the $x_2$ direction. The final $x_2 = -0.337$ (below $x_2^\star = 0$) reflects this oscillation.

\textbf{FW gap at the boundary optimum (quantitative):} At the final iterate $x_{140} \approx (0.5, -0.337)$, the gradient is $\nabla f(x_{140}) = [2 \times 0.5,\; 2 \times (-0.337)] = [1.0,\; -0.674]$. The LP oracle selects $z_{140} = (0.5,\; 10)$ (since $g_1 > 0 \Rightarrow z_1 = 0.5$ and $g_2 < 0 \Rightarrow z_2 = 10$). The Frank--Wolfe gap is:
\begin{equation}
G_{140} = \nabla f(x_{140})^T(x_{140} - z_{140}) = 1.0 \times (0.5 - 0.5) + (-0.674) \times (-0.337 - 10) = 6.97.
\end{equation}
Since $G_{140} \geq f(x_{140}) - f^\star = 0.3637 - 0.25 = 0.1137$, the gap certifies at most $6.97$ suboptimality, which is loose (the actual suboptimality is only $0.11$). The gap is large because the LP vertex $z_{140}$ is a distant corner of the feasible box, which is far from the optimum. The decaying schedule $\gamma_k = 2/(k+2)$ would give the proven bound:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\,\mathrm{diam}(\mathcal{X})^2}{k+2} = \frac{2 \times 2 \times (4.5^2 + 15^2)^{1/2\,2}}{142} \approx \frac{2 \times 2 \times 247.25}{142} \approx 6.96,
\end{equation}
which is indeed loose but provides a computable certificate. The fixed-$\beta$ schedule used here does not satisfy this bound in general.

\textbf{Comparison with Projected GD:} Frank--Wolfe avoids the projection operation at the cost of a slower $O(1/k)$ convergence rate compared to Projected GD's linear rate $O\!\left((1-\mu/L)^k\right)$. Projected GD with $\mu = 1$ (strong convexity of $f(x) = x_1^2 + x_2^2$) and $L = 2$ would converge at rate $0.5^k$, reaching $\varepsilon = 0.01$ in $\log_{0.5}(0.01) \approx 7$ iterations --- far faster than Frank--Wolfe's $O(1/k)$ rate.

\textbf{Frank--Wolfe application domains:} The method is preferred when the LP oracle is cheaper than projection. Canonical examples include:
\begin{itemize}[nosep]
  \item \textbf{Nuclear norm ball} $\{X : \|X\|_* \leq \tau\}$: projection requires a truncated SVD ($O(d^3)$ per step), whereas the LP oracle requires only one full SVD to find the top singular vectors ($O(d^2)$) --- a $d$-fold savings.
  \item \textbf{Spectrahedra} (semidefinite programming): projection onto $\{X \succeq 0, \mathrm{tr}(X) \leq \tau\}$ is $O(d^3)$, but the LP oracle is a rank-1 update $d \cdot v v^T$ where $v$ is the leading eigenvector of $-\nabla f$ --- found via power iteration in $O(d^2)$.
  \item \textbf{Flow polytopes and matching polytopes} in combinatorial optimisation: LP oracle reduces to a shortest-path or maximum-matching computation over a graph, which is polynomial-time efficient, while projection onto these polytopes can be NP-hard.
  \item \textbf{Sparse recovery} with $\ell^1$ ball constraint: LP oracle is a simple coordinate-argmin ($O(d)$); projection is a sort-based operation ($O(d \log d)$) --- both are cheap, but FW naturally generates sparse solutions since $z_k$ is always a vertex (a single coordinate vector $\pm \tau e_i$).
\end{itemize}

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has investigated six families of optimisation algorithms across three canonical benchmarks. The key findings are summarised below, ordered by the problem insight each method illuminates.

\textbf{Adaptive step-size methods (Q1):} The Polyak step is theoretically optimal ($O(1/k)$ convergence) when the optimal value $f^\star$ is known exactly (Rosenbrock: $f = 0.0094$, a $374\times$ improvement over GD in the same 120 iterations), but diverges catastrophically when $f^\star$ is misspecified (Benchmark A: $f = 2.72$ vs.\ $f^\star = 0.4826$). The failure mode is analytically clear: $\alpha_k = (f(x_k) - f^\star_{\text{wrong}})/\|\nabla f(x_k)\|^2 \to \infty$ as $\|\nabla f\| \to 0$ with $f(x_k) \not\to f^\star_{\text{wrong}}$. Adagrad's monotone decaying step size provides stability at the cost of slowing late convergence ($O(1/\sqrt{k})$ decay); RMSprop's exponential forgetting factor ($\beta = 0.9$) prevents this premature decay but lacks Adagrad's rigorous convergence guarantee for sparse gradient settings. Heavy Ball with optimal parameters achieves a rate of $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1) \approx 0.961^k$ on quadratics with $\kappa \approx 2508$, versus GD's $(\kappa-1)/(\kappa+1) \approx 0.9992^k$ --- a $50\times$ speedup in number of iterations. On the non-convex Rosenbrock with conservative parameters ($\alpha = 0.0008, \beta = 0.86$), the \emph{actual} Heavy Ball rate is much slower, but still outperforms GD ($f = 1.40$ vs.\ $3.51$). Heavy Ball's theoretical convergence guarantee holds \emph{only for quadratics}; for non-convex problems, there is no convergence guarantee beyond empirical stability.

\textbf{Momentum and SGD (Q2):} Nesterov's $O(1/k^2)$ optimal rate (vs.\ GD's $O(1/k)$) provides a $7\times$ improvement over GD on Rosenbrock in 150 iterations. The key mechanism is the lookahead: by evaluating $\nabla f(\tilde{x}_k)$ at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$, Nesterov ``corrects'' the momentum direction before overshooting --- this is fundamentally different from Heavy Ball which evaluates the gradient at $x_k$ and cannot self-correct. Adam's per-coordinate scaling handles anisotropy but is less effective on the Rosenbrock valley ($f = 1.579$ vs.\ Nesterov's $f = 0.479$) because the valley's curvature is highly nonlinear: per-coordinate scaling cannot capture the off-diagonal structure of the Rosenbrock Hessian. Mini-batch SGD shows the fundamental batch-size trade-off: smaller batches $b = 5$ give $200$ gradient steps per epoch (faster progress) but gradient variance $\propto 1/b$; larger batches $b = 40$ give $25$ steps per epoch with lower variance. When observation noise dominates sampling noise ($\sigma = 6$), both batch sizes converge to the same elevated floor $J^\star \approx 18$.

\textbf{Newton's method (Q3):} Newton achieves \emph{one-step} convergence on the quadratic loss (Benchmark A) --- this is exact for quadratics since Newton's direction is $H^{-1}\nabla J$ and $H^{-1}\nabla J(\theta_0) = \theta_0 - \theta^{\text{OLS}}$ gives the exact solution in a single step. For near-quadratic problems (Benchmark B: $\sim$5 iterations), quadratic local convergence begins immediately. On Rosenbrock (Benchmark C), Newton achieves a $5.4\times$ objective improvement over GD using only 20 vs.\ 80 iterations; the heavy damping $\alpha = 0.22$ is necessary because the Rosenbrock Hessian changes rapidly along the curved valley (20 steps with $\alpha = 0.22$ have not yet entered the quadratic convergence basin). Newton's $O(d^3)$ per-iteration cost (Hessian inversion) limits scalability to large-scale problems, motivating quasi-Newton methods (L-BFGS) that approximate $H^{-1}$ from gradient differences.

\textbf{Derivative-free methods (Q4):} Finite differences with $\delta = 0.05$ successfully approximate the gradient with negligible accuracy loss (error $\approx 10^{-3}$, in the flat region of Figure~\ref{fig:q4_fd_error}). For double-precision arithmetic, the forward FD optimal perturbation is $\delta \approx 10^{-8}$ (minimising $O(\delta) + O(\varepsilon_{\text{mach}}/\delta)$), while centered FD achieves $O(\delta^2)$ truncation with optimal $\delta \approx 10^{-5.3}$. In practice, centered FD is preferred when two extra evaluations per dimension are affordable. The coarse $\delta = 0.8$ introduces gradient error $\approx 0.3$ (truncation-dominated), causing slow convergence. Nesterov random search requires only 2 evaluations per step (dimension-independent) but its single-direction estimate has high variance and expected step $\alpha/d$, requiring more iterations. The \textbf{outstanding result} is Nelder--Mead achieving machine precision ($f \approx 10^{-29}$) on Rosenbrock in 160 iterations --- better than every gradient-based method --- by adapting its 2D simplex geometry to the curved valley. Grid search provides a global optimality guarantee (to within grid resolution $\Delta x \approx 0.073$) but scales as $O(N^d)$, making it intractable for $d > 3$.

\textbf{Constrained optimisation (Q5):} Projected GD enforces the constraint as a hard condition from iteration~1 via a simple $\max(0.5, \cdot)$ clip, achieving exact feasibility at zero additional cost for this simple constraint set. Penalty methods approximate the constrained problem and require $\lambda \geq 1.8$ for reliable convergence in 100 iterations; when the unconstrained minimum is infeasible, the penalty minimiser satisfies constraint violation $O(1/\lambda)$. The augmented Lagrangian (Algorithm~\ref{alg:aug_lag}) achieves exact constraint satisfaction at finite $\rho$ by jointly updating the Lagrange multiplier $\mu_k$, forming the basis for ADMM.

\textbf{Frank--Wolfe (Q6):} The LP oracle maintains feasibility automatically via convexity; the FW gap $G_k$ provides a computable near-optimality certificate without knowing $f^\star$. The fixed-$\beta$ schedule causes vertex oscillation near the interior optimum; the decaying schedule $\gamma_k = 2/(k+2)$ provides the certified $O(1/k)$ rate. Frank--Wolfe is preferred over projected GD when the LP oracle is cheaper than projection (nuclear norm ball, spectrahedra, flow polytopes, sparse recovery with $\ell^1$ constraints).

\textbf{Overall recommendation:} Algorithm selection depends critically on the condition number $\kappa$, problem dimension $d$, and gradient availability. Based on the experiments in this report:

\begin{itemize}[nosep]
  \item \textbf{$\kappa < 10$, any $d$:} Plain GD converges in $O(\kappa \log(1/\varepsilon))$ iterations, which is fast. Nesterov or Adam offer modest gains. This covers Benchmarks A and B.
  \item \textbf{$\kappa \in [10, 10^3]$, $d \leq 10^4$:} Use Nesterov (theoretical $\sqrt{\kappa}$ speedup) or L-BFGS (superlinear rate, $O(md)$ per iter). Heavy Ball with optimal parameters is $50\times$ faster than GD at $\kappa = 2508$ for quadratics. Avoid plain SGD unless the dataset is very large.
  \item \textbf{$\kappa > 10^3$, $d$ small:} Use Newton with line search or trust region (Benchmark C: Newton reaches $f = 0.017$ in 40 iters vs GD's $f = 3.39$ in 150 iters). Alternatively, for derivative-free access, Nelder--Mead achieves machine precision for $d \leq 10$.
  \item \textbf{$d > 10^6$ (large-scale):} Use mini-batch SGD, Adam, or SVRG. Newton and L-BFGS become impractical ($O(d^3)$ and $O(md)$ respectively). SVRG achieves GD-quality convergence at $O(m + \kappa)$ per epoch for strongly convex losses.
  \item \textbf{Constrained problems:} Use projected GD when projection is $O(d)$ (box, halfspace, $\ell^2$ ball). Use Frank--Wolfe when the LP oracle is cheaper than projection (nuclear norm ball, spectrahedra, sparse recovery). Use augmented Lagrangian / ADMM for distributed or equality-constrained settings. Avoid pure penalty method when the unconstrained minimum is infeasible (violation $O(1/\lambda)$, requires $\lambda \to \infty$).
  \item \textbf{No gradients available:} Nelder--Mead for $d \leq 10$ (machine-precision accuracy possible, as demonstrated on Rosenbrock). Nesterov random search for $d \gg 1$ (2 FE per step, dimension-independent). Grid search only for $d \leq 3$ (exponential cost $O(N^d)$).
\end{itemize}

The most surprising finding in this report is that \textbf{Nelder--Mead outperforms every gradient-based method on Rosenbrock} ($f \approx 10^{-29}$ in 160 iterations, using only $\leq 800$ function evaluations with no gradient information). This highlights that problem-specific geometry can dominate theoretical rate guarantees: Rosenbrock's 2D curved valley is well-matched to the simplex geometry, while gradient methods are handicapped by the extreme condition number $\kappa \approx 2508$.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g   # velocity accumulates gradient
        x = x - z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum (lookahead)}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)  # increasing schedule
        lookahead = x + beta_k * z                  # extrapolation point
        g = grad_fn(lookahead)                       # gradient at lookahead
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser (with bias correction)}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # bias-corrected
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Tikhonov Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))       # Tikhonov regularisation
        try:
            p = np.linalg.solve(H_reg, g)           # solve H*p = g
        except np.linalg.LinAlgError:
            p = g                                    # fallback to gradient
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (Fixed Step)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LP oracle for box constraints: O(d) per coordinate
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # gamma = 1-beta towards z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
