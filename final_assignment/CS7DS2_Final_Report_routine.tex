\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss, a toy neural network loss with sinusoidal perturbation, and the Rosenbrock function. Six categories of methods are investigated: (1) adaptive step-size methods --- Polyak, Adagrad, RMSprop, and Heavy Ball; (2) momentum-based and stochastic methods --- Nesterov acceleration, Adam, and mini-batch SGD; (3) Newton's method with second-order local approximations; (4) derivative-free approaches --- finite differences, Nesterov random search, Nelder--Mead, and grid search; (5) constrained optimisation via projected gradient descent and penalty methods; and (6) the Frank--Wolfe conditional gradient algorithm for linear-constrained problems. Each method is evaluated for convergence speed, trajectory behaviour, stability, and computational trade-offs. Results demonstrate that second-order methods achieve the fastest convergence per iteration on smooth functions, momentum-based methods can nearly match this with lower per-iteration cost, and derivative-free methods remain viable when gradients are unavailable despite their slower convergence rates. Constraint handling analysis reveals projected gradient descent enforces feasibility exactly from the first iteration, while penalty methods offer implementation simplicity at the cost of approximate constraint satisfaction.
\end{abstract}

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = (\theta^{\star})^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
The Hessian $H = \frac{1}{m}X^T X$ is constant and positive semi-definite. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. The smoothness constant equals the largest eigenvalue $L = \lambda_{\max}(H)$, and the strong convexity constant $\mu = \lambda_{\min}(H) > 0$ (almost surely for $m > 2$). Due to observation noise, the irreducible minimum is $J^{\star} \approx \sigma^2/2 \approx 0.4826$, not zero. Starting point: $\theta_0 = [0,\, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \operatorname{diag}(2 - \sin(x_1),\ 10)$. Since $|\sin(x_1)| \leq 1$, the $x_1$-curvature lies in $[1, 3]$, giving a local condition number $\kappa \in [10/3, 10]$. The sinusoidal term models a non-convex perturbation typical in neural network losses. The minimum lies at approximately $(x_1^\star,\, x_2^\star) = (0.582,\, 2.000)$ with $f^\star \approx 0.7244$ (solving $2(x_1-1)+\cos(x_1)=0$ numerically). Starting point: $x_0 = [-1,\, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is $(1, 1)$ with $f^\star = 0$. The Hessian at the optimum has eigenvalues $\lambda_1 \approx 0.5$ and $\lambda_2 \approx 1000$, giving condition number $\kappa \approx 2000$. This extreme ill-conditioning, combined with the narrow curved valley along $x_2 = x_1^2$, makes the Rosenbrock function one of the most challenging standard test functions for first-order methods. Starting point: $x_0 = [-1,\, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems (Questions 5 and 6):
\begin{equation}
\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a compact convex polytope (axis-aligned box) with diameter $D = \|(5 - 0.5,\, 10 - (-5))\| = \sqrt{4.5^2 + 15^2} \approx 15.66$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

For an $L$-smooth convex function (i.e.\ $\|\nabla f(x) - \nabla f(y)\| \leq L\|x - y\|$ for all $x,y$), gradient descent with constant step $\alpha \leq 1/L$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{\|x_0 - x^\star\|^2}{2\alpha k}.
\end{equation}
For $\mu$-strongly convex functions, the rate improves to:
\begin{equation}
\|x_k - x^\star\|^2 \leq \left(1 - \frac{\mu}{L}\right)^k \|x_0 - x^\star\|^2,
\end{equation}
i.e.\ linear (geometric) convergence with rate $\rho = 1 - \mu/L = 1 - 1/\kappa$. The condition number $\kappa$ thus directly governs convergence speed --- the larger $\kappa$, the slower GD converges. Adaptive step-size methods aim to overcome this dependence by automatically scaling the learning rate to the local curvature.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Proportional to suboptimality}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size minimises the distance to $x^\star$ at each iteration (under exact $f^\star$): for a subgradient $g_k \in \partial f(x_k)$, it can be shown that $\|x_{k+1} - x^\star\|^2 \leq \|x_k - x^\star\|^2 - \alpha_k^2\|g_k\|^2$, which is minimised by $\alpha_k = (f(x_k) - f^\star)/\|g_k\|^2$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Cumulative sum of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad scales each coordinate's learning rate inversely by the root of its gradient history, effectively giving larger steps to rarely-updated parameters. For convex functions, Adagrad achieves a regret bound of $O(\sqrt{T})$ over $T$ rounds, implying convergence at $O(\log T/\sqrt{T})$. Its principal limitation is that the accumulated sum $G_k$ grows without bound, causing the effective step size to decay to zero and stalling convergence in later iterations.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's vanishing step problem by replacing the cumulative sum with an exponential moving average (decay $\beta = 0.9$). The effective second moment estimate $v_k \approx \mathbb{E}[g^2]$ remains bounded, preventing the step from collapsing. Formally, $v_k$ is a geometric series: $v_k = (1-\beta)\sum_{j=0}^{k-1}\beta^{k-1-j} g_j^2$, where recent gradients receive exponentially higher weight.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Momentum accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method adds a momentum term $\beta z_k$ to the gradient update. For a quadratic $f$ with condition number $\kappa$, the optimal Heavy Ball parameters yield the convergence rate $\rho = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$, improving upon GD's rate of $(1 - 1/\kappa)$. Physically, the iterate behaves like a ball rolling in the objective landscape, accumulating speed in consistent descent directions and decelerating across oscillating directions.

\begin{remark}
\label{rem:hb_nonquadratic}
The optimal $O(\sqrt{\kappa})$ rate of Heavy Ball applies only to strongly convex \emph{quadratic} functions. For general non-quadratic objectives, Heavy Ball does not enjoy the same guarantees and can even diverge for poorly chosen $(\alpha, \beta)$. In contrast, Nesterov's acceleration (Section~\ref{sec:q2}) achieves $O(1/k^2)$ for \emph{any} smooth convex function, making it theoretically superior. Empirically, this is reflected in Table~\ref{tab:q1_final}: Heavy Ball achieves $f = 1.40$ on Rosenbrock vs Nesterov's $f = 0.479$ (Table~\ref{tab:q2_final}).
\end{remark}

\subsection{Analysis of Polyak Divergence}

The Polyak step size is derived under the assumption that $f^\star$ is known exactly. The divergence on Benchmarks A and B can be characterised precisely.

\begin{proposition}
\label{prop:polyak_diverge}
Let $f^\star_{\rm true}$ be the true minimum. If the Polyak step is run with specified $\tilde{f}^\star < f^\star_{\rm true}$, then as $\|g_k\| \to 0$ the step size satisfies
\[
\alpha_k = \frac{f(x_k) - \tilde{f}^\star}{\|g_k\|^2 + \varepsilon} \;\to\; \frac{f^\star_{\rm true} - \tilde{f}^\star}{\varepsilon} \quad \text{(a large constant)},
\]
causing the iterate to take a large step away from the minimum and re-enter a region of non-negligible gradient, resulting in sustained oscillations.
\end{proposition}

On Benchmark A, $f^\star_{\rm true} \approx 0.4826$ and $\tilde{f}^\star = 0$, giving $\alpha_k \approx 0.4826/\varepsilon = 4826$ near the minimum. On Benchmark C, $f^\star_{\rm true} = \tilde{f}^\star = 0$, so $\alpha_k = f(x_k)/(\|g_k\|^2 + \varepsilon)$ shrinks properly as the iterate converges.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest (Benchmark C: $f = 0.0094$). For misspecified $f^\star$, Adagrad and Heavy Ball match or exceed the GD baseline.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential gradient averaging. Heavy Ball exhibits characteristic overshoot--oscillation typical of under-damped momentum.
\item \textbf{Stability:} GD baseline is most stable but slowest. Polyak is \emph{unstable} when $f^\star$ is misspecified. Adagrad is uniformly stable (monotonically decreasing step), though this causes premature stagnation on long-horizon problems.
\item \textbf{Ill-conditioned landscapes:} Benchmarks A and B are well-conditioned ($\kappa \approx 2$--$5$), so all converging methods reach the minimum equally well. Benchmark C ($\kappa \approx 2000$) reveals significant performance differences: Polyak $\gg$ Heavy Ball $>$ Adagrad $>$ GD $\approx$ RMSprop.
\item \textbf{Practical implication:} In practice, exact $f^\star$ is rarely known, limiting Polyak's applicability. RMSprop and Adam (Q2) dominate practical deep learning because they combine RMSprop's per-coordinate adaptation with momentum.
\end{itemize}

\subsection{Theoretical Background: Accelerated Convergence}

Gradient descent achieves $O(1/k)$ convergence for smooth convex functions. Nesterov (1983) showed this is \emph{not} optimal --- an improved $O(1/k^2)$ rate is achievable by first-order methods, now known as Nesterov acceleration.

\begin{theorem}[Nesterov, 1983]
\label{thm:nesterov}
Let $f$ be convex and $L$-smooth. Nesterov's accelerated gradient method with step $\alpha = 1/L$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2}.
\end{equation}
This $O(1/k^2)$ rate is \emph{optimal} among all first-order methods (cannot be improved by any algorithm that uses only gradient queries).
\end{theorem}

The key insight of Nesterov's method is the \emph{lookahead}: the gradient is evaluated at a point $\tilde{x}_k$ that looks ahead in the direction of the current momentum, allowing the method to ``correct'' its trajectory before it overshoots.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Gradient at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The adaptive schedule $\beta_k = (k-1)/(k+2) \to 1$ gradually increases momentum to achieve the optimal rate. The update is equivalent to: $x_{k+1} = \tilde{x}_k - \alpha\nabla f(\tilde{x}_k) + \beta_k^2 z_{k-1}$, which differs from Heavy Ball only in evaluating the gradient at the \emph{lookahead} point rather than $x_k$.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{EMA of gradient (1st moment)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{EMA of squared gradient (2nd moment)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction: compensates for zero initialisation}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum (via $m_t \approx \mathbb{E}[g]$) with per-coordinate adaptive scaling (via $v_t \approx \mathbb{E}[g^2]$), combining the advantages of RMSprop and Nesterov momentum. The bias correction terms $1/(1-\beta^t)$ are critical in early iterations: without them, $m_1 = (1-\beta_1)g_1 \approx 0$ would give nearly zero effective gradient, stalling convergence.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices into batches $\mathcal{B}_1, \ldots, \mathcal{B}_{m/b}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Bias-variance trade-off in mini-batch size:} The stochastic gradient satisfies $\mathbb{E}[g] = \nabla J(\theta)$ (unbiased) with variance $\operatorname{Var}[g] = \sigma^2_g/b$, where $\sigma^2_g$ is the per-sample gradient variance. Smaller $b$ gives more gradient updates per epoch ($m/b$ updates vs a single update for full-batch) but higher variance; larger $b$ reduces variance at the cost of fewer updates.

\subsection{SGD with Increased Observation Noise}

\textbf{Convergence theory for SGD:} For a $\mu$-strongly convex, $L$-smooth function, mini-batch SGD with learning rate $\alpha_k = c/(\mu k)$ achieves:
\begin{equation}
\mathbb{E}[f(\theta_k)] - f^\star \leq \frac{L c^2 \sigma^2_g}{2b\mu k},
\end{equation}
where $\sigma^2_g$ is the per-sample gradient variance and $b$ is the batch size. This $O(1/k)$ rate is optimal for stochastic first-order methods (vs $O(1/k^2)$ for deterministic Nesterov). The expected suboptimality scales as $\sigma^2_g/(b \cdot k)$, confirming that larger batches reduce variance (and thus the optimisation noise floor) proportional to $1/b$.

The experiment is repeated with output noise scaled by factor 6 ($\varepsilon^{(i)} \sim \mathcal{N}(0,36)$ vs $\mathcal{N}(0,1)$).

\subsection{Local Approximations of $g(x) = x^4$}

At $x_0 = 0.25$, we have $g(x_0) = 0.25^4 = 0.00390625$, $g'(x_0) = 4(0.25)^3 = 0.0625$, $g''(x_0) = 12(0.25)^2 = 0.75$.

\textbf{First-order (tangent) approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.00391 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.00391 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

By Taylor's theorem, the approximation errors are: $|g(x) - \hat{g}_1(x)| = O(|x-x_0|^2)$ and $|g(x) - \hat{g}_2(x)| = O(|x-x_0|^3)$. The Newton step minimises $\hat{g}_2$: setting $\hat{g}_2'(x) = 0$ gives $x_{\text{Newton}} = x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 = 0.1\overline{6}$, which is closer to the true minimum $x^\star = 0$ than the gradient step $x_0 - \alpha g'(x_0)$ for any fixed $\alpha$.

\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$. The tangent line deviates rapidly from the true function. The parabola fits well in a neighbourhood of $x_0$, and its minimum (Newton step) provides a superior estimate of the function's minimum than any fixed-step gradient descent step. \subsection{Newton's Method and Quadratic Convergence}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Quadratic Convergence of Newton's Method]
\label{thm:newton_conv}
Suppose $f$ is twice continuously differentiable, $\nabla^2 f(x^\star) \succ 0$ with smallest eigenvalue $\mu > 0$, and $\nabla^2 f$ is $M$-Lipschitz. Then Newton's method with $\alpha = 1$ (undamped) satisfies, in a neighbourhood of $x^\star$:
\begin{equation}
\|x_{k+1} - x^\star\| \leq \frac{M}{2\mu}\|x_k - x^\star\|^2.
\end{equation}
This is \emph{quadratic convergence}: the number of correct digits roughly doubles at each iteration.
\end{theorem}

In contrast, gradient descent achieves only \emph{linear convergence}: $\|x_{k+1} - x^\star\| \leq (1 - \mu/L)\|x_k - x^\star\|$. The cost per Newton iteration is higher ($O(d^3)$ for Hessian inversion vs $O(d)$ for a gradient step), but the dramatic reduction in iteration count typically makes Newton superior when $d$ is small and the function is smooth.

The damping parameter $\lambda = 10^{-8}$ in \texttt{newtons\_method} regularises the Hessian $H_k + \lambda I$, ensuring positive definiteness when $H_k$ may be singular or nearly singular (as occurs on Rosenbrock where $H_{\rm Rosenbrock}$ can be very ill-conditioned). The step size $\alpha < 1$ provides a \emph{global convergence} safeguard: the pure Newton step ($\alpha = 1$) only has local convergence guarantees, while damping ($\alpha < 1$) trades convergence speed for a larger basin of attraction.

\subsection{Computational Trade-offs}

Newton's method requires computing and inverting the $d \times d$ Hessian at each iteration: $O(d^2)$ to evaluate $\nabla^2 f$ and $O(d^3)$ (or $O(d^2)$ using conjugate gradient) to solve the linear system. For $d = 2$ (as in these experiments), this is negligible --- Newton's entire 20-iteration run completes in microseconds. However, in high-dimensional settings ($d \sim 10^6$--$10^9$), full Newton is computationally prohibitive, motivating quasi-Newton methods (L-BFGS approximates $H^{-1}$ using $O(md)$ operations with $m \ll d$ history vectors) and second-order-free methods (Adam, Nesterov).

The Rosenbrock results demonstrate Newton's strength precisely where gradient methods struggle most: extreme ill-conditioning ($\kappa \approx 2000$). While GD achieves $f = 3.73$ in 80 iterations (barely leaving the starting valley), Newton reaches $f = 0.69$ in 20 iterations --- a $5.4\times$ improvement with $4\times$ fewer iterations. This remarkable efficiency gain is a direct consequence of Newton's iteration complexity being \emph{independent} of the condition number near the minimum: the quadratic convergence rate $\|x_{k+1} - x^\star\|^2 \leq C\|x_k - x^\star\|^4$ means the method does not ``feel'' ill-conditioning once it enters the quadratic convergence regime.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta} = \frac{\partial f}{\partial x_i}(x) + \frac{\delta}{2}\frac{\partial^2 f}{\partial x_i^2}(x) + O(\delta^2).
\label{eq:fd}
\end{equation}
The bias (truncation error) is $O(\delta)$; rounding errors add $O(\varepsilon_{\rm mach}/\delta)$ where $\varepsilon_{\rm mach} \approx 10^{-16}$. The optimal $\delta$ balances these: $\delta_{\rm opt} \sim \sqrt{\varepsilon_{\rm mach}} \approx 10^{-8}$. In practice, $\delta = 0.05$ gives small truncation error while remaining numerically well-conditioned, whereas $\delta = 0.8$ introduces large truncation error because the function curvature on Benchmark B is $O(1)$--$O(10)$, so the $O(\delta^2 \cdot f'')\approx O(0.64\cdot 10) = O(6.4)$ bias term is large relative to the gradient.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$ \Comment{Uniform on unit sphere}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Theoretical properties:} The estimator $\hat{g}_k$ is an unbiased estimate of $\frac{1}{d}\nabla f(x_k)$ in expectation (after normalisation): $\mathbb{E}[u(u^T\nabla f)] = \frac{1}{d}\nabla f$ for $u \sim \text{Uniform}(\mathbb{S}^{d-1})$. The noise variance scales as $O(\|\nabla f\|^2/d)$, so convergence rate degrades with dimension $d$ --- this is the fundamental curse of dimensionality for zeroth-order methods. For fixed $d=2$, the method requires $\sim 3\times$ more iterations than exact GD (empirically), but uses only 2 function evaluations per step.

Figure~\ref{fig:fd_error} plots the gradient approximation error $\|\hat{g}_\delta - \nabla f\|$ as a function of $\delta$ for Benchmark B at a fixed test point, demonstrating the classical U-shaped error curve. For very small $\delta$, rounding errors dominate; for large $\delta$, truncation error dominates; and the optimal $\delta \approx 10^{-8}$ balances both. The tested values $\delta = 0.05$ and $\delta = 0.8$ are both in the truncation-dominated regime, but $\delta = 0.05$ incurs a truncation error $\approx 6\times$ smaller than $\delta = 0.8$.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations.
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1.0$): Move worst vertex $x_w$ through centroid $\bar{x}$ of remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_w)$.
\item \textbf{Expansion} ($\gamma = 2.0$): If reflection improved, try $x_e = \bar{x} + \gamma(\bar{x} - x_w)$.
\item \textbf{Contraction} ($\rho = 0.5$): If reflection worsened, try $x_c = \bar{x} + \rho(x_r - \bar{x})$ (outside) or $\bar{x} + \rho(x_w - \bar{x})$ (inside).
\item \textbf{Shrink} ($\sigma = 0.5$): If contraction fails, contract all vertices towards best: $x_i \gets x_1 + \sigma(x_i - x_1)$.
\end{enumerate}
No gradient information is required. Settings: initial simplex step $= 0.35$, 160 iterations.

\subsection{Problem Setup and KKT Conditions}

Minimise $f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)$ subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$.

Rewriting as a standard minimisation problem: minimise $f(x)$ subject to $g(x) = 0.5 - x_1 \leq 0$. The KKT first-order optimality conditions are:
\begin{align}
\nabla f(x^\star) + \lambda^\star \nabla g(x^\star) &= 0 && \text{(stationarity)} \\
g(x^\star) &\leq 0 && \text{(primal feasibility)} \\
\lambda^\star &\geq 0 && \text{(dual feasibility)} \\
\lambda^\star g(x^\star) &= 0 && \text{(complementary slackness)}
\end{align}
where $\lambda^\star \geq 0$ is the Lagrange multiplier for the constraint.

\textbf{Case analysis:} If the unconstrained minimum $(0.582, 2.0)$ satisfies $x_1 \geq 0.5$, then $\lambda^\star = 0$ and the constraint is inactive. Indeed, $0.582 > 0.5$, so the constrained minimum \emph{coincides with the unconstrained minimum}: $x^\star = (0.582, 2.0)$, $f^\star \approx 0.7244$, $\lambda^\star = 0$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project back: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space constraint $x_1 \geq 0.5$, the projection is $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)^T$, a simple component-wise clipping. Projected GD satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{L\|x_0 - x^\star\|^2}{2k},
\end{equation}
achieving the same $O(1/k)$ rate as unconstrained GD on smooth convex functions.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $\nabla F_{\lambda}(x) \gets \nabla f(x) - \lambda \mathbf{1}[x_1 < 0.5]\, e_1$ \Comment{Subdifferential}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method solves the unconstrained problem $\min_x F_\lambda(x) = f(x) + \lambda\, \phi(g(x))$ where $\phi(t) = \max(0, t)$ is the hinge penalty. As $\lambda \to \infty$, the penalty solution converges to the constrained solution: $x^\star(\lambda) \to x^\star$ as $\lambda \to \infty$. In practice, large $\lambda$ creates a steep, nearly discontinuous gradient near the constraint boundary, requiring smaller step sizes for stability.

\textbf{Settings:} Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$, $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$ (i.e.\ $a = [1, 2]^T$). Since $f$ is linear and $\mathcal{X}$ is a bounded convex polytope, the \emph{Fundamental Theorem of Linear Programming} guarantees that the minimum is attained at a vertex of $\mathcal{X}$. Since $a_1 = 1 > 0$ and $a_2 = 2 > 0$, each component is minimised at its lower bound:
\begin{equation}
x^\star = (0.5,\; -5), \qquad f^\star = 1\times 0.5 + 2\times(-5) = -9.5.
\end{equation}

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination, stays in $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Frank--Wolfe Convergence]
\label{thm:fw_conv}
Let $f$ be $L$-smooth and convex on a compact convex set $\mathcal{X}$ with diameter $D = \max_{x,y\in\mathcal{X}}\|x-y\|$. Frank--Wolfe with $\gamma_k = 2/(k+2)$ (standard decaying step) satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{2LD^2}{k+2}.
\end{equation}
\end{theorem}

The implementation here uses a fixed $\beta$ (rather than the decaying $\gamma_k = 1 - \beta = 2/(k+2)$ of the standard analysis), which trades the theoretical $O(1/k)$ guarantee for empirical stability. The key advantages over projected gradient methods are: (1) no projection is required --- iterates remain in $\mathcal{X}$ by convexity; (2) the LMO for box constraints decomposes coordinate-wise, requiring only $O(d)$ operations.

\textbf{LMO for box constraints:} The LP subproblem $\min_{z \in \mathcal{X}} \nabla f(x_k)^T z$ decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
where $[l_i, u_i]$ are the coordinate bounds. This always returns a vertex of the box $\mathcal{X}$.

\textbf{Frank--Wolfe gap} as a convergence certificate: $h_k = \nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - f^\star$ (by convexity). When $h_k \leq \varepsilon$, we have verified $\varepsilon$-optimality without knowing $f^\star$.

\subsection{Interior vs Boundary Behaviour: Theoretical Analysis}

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits characteristic ``zig-zag.'' The LMO returns vertices of $\mathcal{X}$, which are far from the interior optimum, causing the convex combination to overshoot. The convergence rate is $O(1/k)$ (Theorem~\ref{thm:fw_conv}). Smaller step size ($\beta = 0.985$) reduces oscillation at the cost of slower initial descent, but achieves better final accuracy on the 180-iteration run.

\textbf{Boundary optimum:} Frank--Wolfe is naturally suited because the LMO consistently selects a vertex near the constrained optimum. The boundary case converges slower than the interior case in terms of absolute $f$ reduction because the residual gap $0.364 - 0.25 = 0.114$ requires the iterate to travel close to a vertex of $\mathcal{X}$, but the mixing parameter $\beta = 0.93$ limits the step size. Implementing the standard decaying schedule $\gamma_k = 2/(k+2)$ (equivalent to $\beta = 1 - 2/(k+2) = k/(k+2) \to 1$) would accelerate convergence in this case.

\textbf{Standard decaying step schedule:} The theoretical $O(1/k)$ guarantee of Theorem~\ref{thm:fw_conv} requires $\gamma_k = 2/(k+2)$, equivalent to $\beta_k = k/(k+2) \to 1$. The fixed $\beta$ used in the implementation converges faster initially (when $\beta < k/(k+2)$, i.e.\ the step is larger than the decaying schedule) but may oscillate at later stages when $\beta > k/(k+2)$ (the step is now larger than optimal). The boundary case ($\beta = 0.93$) has not converged to $f^\star = 0.25$ after 140 iterations, whereas the decaying schedule would guarantee convergence by Theorem~\ref{thm:fw_conv}. This illustrates the practical importance of using the theoretically motivated step-size schedule.

\textbf{Frank--Wolfe vs Projected GD:} Frank--Wolfe avoids projections, making it preferable when projections are expensive (e.g.\ nuclear norm ball, spectrahedron). Projected GD converges faster per iteration when projections are cheap (e.g.\ box constraints), because it can take gradient steps of size $1/L$ vs Frank--Wolfe's steps that are constrained to convex combinations with potentially distant vertices. Additionally, Frank--Wolfe provides a natural convergence certificate via the Frank--Wolfe gap $h_k = \nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - f^\star$ --- this duality gap certifies optimality without knowing $f^\star$, an important practical advantage.

\subsection{Key Findings and Theoretical Connections}

\begin{enumerate}
\item \textbf{Condition number dominates convergence.} The Rosenbrock function ($\kappa \approx 2000$) is the most discriminating benchmark: methods that exploit curvature (Newton, Nesterov, Polyak with correct $f^\star$) achieve dramatically lower final values than vanilla GD ($\kappa$-rate $\approx 1 - 1/2000 = 0.9995$ per step, requiring thousands of iterations for convergence). On Benchmarks A and B ($\kappa \approx 2$--$5$), all stable methods achieve essentially the same final accuracy.

\item \textbf{Second-order methods are most efficient per iteration.} Newton's method solves Benchmark A exactly in one iteration and achieves $5.4\times$ better Rosenbrock accuracy in $4\times$ fewer iterations than GD. The quadratic convergence rate (Theorem~\ref{thm:newton_conv}) is exponentially faster than GD's linear rate. This advantage is decisive in low-dimensional problems where the $O(d^3)$ Hessian cost is acceptable.

\item \textbf{Nesterov acceleration bridges the gap.} Nesterov achieves the optimal $O(1/k^2)$ convergence rate (Theorem~\ref{thm:nesterov}) without requiring Hessian evaluations, reducing the Rosenbrock objective to $0.479$ vs GD's $3.388$ in the same 150 iterations. This demonstrates that the lookahead mechanism effectively compensates for ill-conditioning without second-order information.

\item \textbf{Adaptive methods are robust but not always fastest.} Adagrad, RMSprop, and Adam adapt to per-coordinate curvature, achieving reliable convergence across diverse problem types without careful step-size tuning. Adam combines momentum and adaptivity and is the dominant method in deep learning, but Nesterov's structured acceleration outperforms it on the highly non-quadratic Rosenbrock. The practical trade-off is: Adam for black-box robustness, Nesterov for well-understood convex problems.

\item \textbf{Derivative-free methods are viable but slower.} Finite differences with small $\delta$ closely approximates exact GD at the cost of $d+1$ function evaluations per step (vs 1 gradient evaluation). Nesterov random search trades gradient quality for 2-evaluation simplicity, at the cost of $O(d)$ slower convergence. Grid search uniquely guarantees global optimum discovery but is only feasible for $d \leq 3$.

\item \textbf{Constraint handling: hard vs soft.} Projected GD enforces feasibility exactly from the first iteration, making it preferable when constraint satisfaction is critical. Penalty methods are simpler to implement but require careful tuning of $\lambda$ --- too small leaves constraints violated, too large destabilises the optimisation. Frank--Wolfe offers a natural constraint-satisfaction guarantee without projection by construction, at the cost of a slower $O(1/k)$ convergence rate.

\item \textbf{Polyak sensitivity.} The Polyak step size achieves the fastest convergence when $f^\star$ is precisely known (Rosenbrock: $f = 0.009$ in 120 iterations), but diverges catastrophically when $f^\star$ is misspecified, as characterised in Proposition~\ref{prop:polyak_diverge}. This fundamental sensitivity limits its practical applicability to problems with known optimal values (e.g.\ certain convex programs where $f^\star$ can be computed by duality).
\end{enumerate}

\subsection{Limitations and Future Work}

All experiments are limited to $d = 2$ dimensions. Several phenomena would change at scale: Newton's $O(d^3)$ cost becomes prohibitive for $d > 10^3$; gradient noise in SGD dominates at large $m$; Nesterov's acceleration advantage over GD persists but the absolute convergence rate depends on $L$, which grows with $d$. Future work could extend to $d \geq 100$ using quasi-Newton (L-BFGS), stochastic Nesterov, or second-order-free adaptive methods (Adam, AdaFactor). The penalty method experiments could be extended to augmented Lagrangian and ADMM, which overcome the ill-conditioning introduced by large $\lambda$.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
