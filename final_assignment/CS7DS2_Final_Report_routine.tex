\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark A), a toy neural network loss with sinusoidal perturbation (Benchmark B), and the Rosenbrock function (Benchmark C).
Six categories of methods are investigated across six questions:
(Q1) adaptive step-size methods---Polyak, Adagrad, RMSprop, and Heavy Ball---where the Polyak step diverges when $f^\star$ is misspecified but achieves best-in-class convergence on Rosenbrock when $f^\star = 0$ is exact;
(Q2) momentum-based and stochastic methods---Nesterov acceleration achieves the theoretically optimal $O(1/k^2)$ rate and outperforms Adam on the ill-conditioned Rosenbrock function, while mini-batch SGD shows the classical variance-versus-frequency trade-off;
(Q3) Newton's method---single-step convergence on quadratic losses and an order-of-magnitude speedup over gradient descent on Rosenbrock in far fewer iterations;
(Q4) derivative-free approaches---finite differences match exact gradients with small $\delta$, while Nelder--Mead and grid search trade scalability for gradient-free operation;
(Q5) constrained optimisation---projected gradient descent enforces hard feasibility at every iterate, whereas penalty methods trade exact feasibility for simplicity, with convergence dependent on the penalty weight $\lambda$;
(Q6) the Frank--Wolfe algorithm---achieves $O(1/k)$ convergence without a projection oracle, with qualitatively different behaviour at interior versus boundary optima.
Each method is evaluated in terms of convergence speed, trajectory behaviour, stability, and computational trade-offs, with theoretical convergence guarantees connecting each algorithm to first-principles analysis.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.
The three benchmark functions span a spectrum of difficulty: Benchmark A is a well-conditioned convex quadratic; Benchmark B is a mildly anisotropic near-quadratic; Benchmark C is the highly ill-conditioned Rosenbrock function.

To systematically understand algorithm behaviour we characterise each benchmark through its \emph{$L$-smoothness} constant and, where applicable, its \emph{strong convexity} modulus.

\begin{definition}[$L$-Smooth Function]
A differentiable function $f$ is $L$-smooth if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x - y\|$ for all $x, y$.
Equivalently, $\nabla^2 f(x) \preceq L I$ everywhere.
For smooth gradient descent with step size $\alpha \leq 1/L$ to guarantee descent, $L$ is the key parameter.
\end{definition}

\begin{definition}[$\mu$-Strongly Convex Function]
A differentiable function $f$ is $\mu$-strongly convex ($\mu > 0$) if $\nabla^2 f(x) \succeq \mu I$ everywhere.
The \emph{condition number} $\kappa = L/\mu$ determines how many gradient descent steps are required.
\end{definition}

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \in \mathbb{R}^2, \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2}\theta^T H \theta - b^T \theta + c,
\end{equation}
where $H = \frac{1}{m}X^T X$, $b = \frac{1}{m}X^T y$, $c = \frac{1}{2m}\|y\|^2$.
The gradient is $\nabla J(\theta) = H\theta - b = \frac{1}{m}X^T(X\theta - y)$.
Since $X \sim \mathcal{N}(0, I_{m\times 2})$, by the law of large numbers $H \approx I_2$, giving $L \approx \lambda_{\max}(H) \approx 1.06$ and $\mu \approx \lambda_{\min}(H) \approx 0.94$, yielding a condition number $\kappa \approx 1.13$.
Due to observation noise, the minimum loss is $J^{\star} \approx 0.4826$, not zero. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \operatorname{diag}(2 - \sin(x_1),\ 10)$.
The unconstrained minimum satisfies $2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$, yielding $x_1^{\star} \approx 0.582$, and $x_2^{\star} = 2$ exactly.
At the optimum: $H^{\star} = \operatorname{diag}(2 - \sin(0.582),\ 10) \approx \operatorname{diag}(1.45,\ 10)$, giving $L \approx 10$, $\mu \approx 1.45$, and $\kappa \approx 6.9$.
The sinusoidal term $\sin(x_1)$ introduces mild non-quadratic curvature but the Hessian remains positive definite throughout the relevant domain, so the function is locally strongly convex near its minimum.
The minimum value is $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is:
\begin{equation}
H^{\star} = \begin{bmatrix} 2 + 1200(1)^2 - 400(1) & -400(1) \\ -400(1) & 200 \end{bmatrix} = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}.
\end{equation}
The eigenvalues of $H^{\star}$ satisfy $\lambda_{1,2} = 501 \pm \sqrt{501^2 - (802 \cdot 200 - 400^2)} \approx 1001.6$ and $0.4$, giving a condition number $\kappa \approx 2504$.
This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions.
The function is \emph{not} globally strongly convex (the Hessian is only positive definite near the minimum), which fundamentally limits gradient-based methods that rely on global curvature bounds.
Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained problems: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$, which is a bounded box polytope.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

For a $\mu$-strongly convex, $L$-smooth function $f$, gradient descent with step size $\alpha = 1/L$ converges geometrically:
\begin{equation}
\label{eq:gd_rate}
f(x_k) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)^k \bigl(f(x_0) - f^{\star}\bigr) = \left(1 - \frac{1}{\kappa}\right)^k \bigl(f(x_0) - f^{\star}\bigr).
\end{equation}
This motivates adaptive step-size methods: by locally estimating curvature or gap information, they achieve faster practical convergence without requiring global $L$ and $\mu$.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Minimises upper bound of subgradient model}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size minimises the exact upper bound $f(x_{k+1}) \leq f(x_k) + \nabla f(x_k)^T(x_{k+1} - x_k)$ under the assumption that $\nabla f$ is locally linear. When $f^\star$ is known exactly, this achieves the optimal convergence rate $1 - 1/\kappa$ for strongly convex functions.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's effective per-coordinate step size $\alpha_k^{(i)} = \alpha_0 / (\sqrt{\sum_{j=0}^{k} g_j^{(i)2}} + \varepsilon)$ decays monotonically: coordinates with historically large gradients receive smaller updates, providing implicit regularisation. In the online convex optimisation setting, Adagrad achieves an $O(\sqrt{T})$ regret bound, which is optimal among subgradient methods without strong convexity assumptions.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop (Hinton, 2012) addresses Adagrad's diminishing step size by replacing cumulative accumulation with an exponential moving average with decay $\beta \in (0,1)$.
The effective step in coordinate $i$ is $\alpha_0 / \sqrt{\mathbb{E}[g^{(i)2}] + \varepsilon}$, where the expectation is approximate and adapts to recent curvature.
Unlike Adagrad, RMSprop does not have formal convergence guarantees for non-convex functions, but empirically stabilises training.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method (Polyak, 1964) can be derived as a discretisation of the second-order ODE $\ddot{x} + \gamma \dot{x} + \nabla f(x) = 0$, which represents a damped harmonic oscillator.
For a quadratic $f$ with eigenvalues in $[\mu, L]$, the optimal parameters are $\alpha^{\star} = 4/(\sqrt{L} + \sqrt{\mu})^2$ and $\beta^{\star} = ((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^2$, yielding convergence rate $(\sqrt{\kappa}-1)^2/(\sqrt{\kappa}+1)^2 \approx 1 - 2/\sqrt{\kappa}$, which is asymptotically superior to GD's rate $1 - 2/(\kappa+1) \approx 1 - 2/\kappa$.

\textbf{Key finding on Polyak divergence:} The Polyak step size is highly sensitive to the accuracy of $f^{\star}$.
When the specified $f^{\star} = 0$ deviates from the true minimum $f^{\star}_{\text{true}} > 0$ (Benchmarks A and B), the method diverges because:
\begin{equation}
\alpha_k = \frac{f(x_k) - 0}{\|\nabla f(x_k)\|^2 + \varepsilon} \xrightarrow[x_k \to x^{\star}]{}  \frac{f^{\star}_{\text{true}}}{\varepsilon},
\end{equation}
which for $f^{\star}_{\text{true}} = 0.4826$ and $\varepsilon = 10^{-4}$ gives $\alpha_k \approx 4826$---a catastrophically large step that causes oscillation and divergence.
On Benchmark C where $f^{\star} = 0$ is exact, the step size $\alpha_k = f(x_k)/(\|\nabla f(x_k)\|^2 + \varepsilon) \to 0$ gracefully as $x_k \to (1,1)$, producing well-behaved convergence.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$) because it uses global suboptimality information. For unknown $f^{\star}$, Heavy Ball and Adagrad match or exceed GD due to momentum and per-coordinate scaling respectively.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging of squared gradients. Heavy Ball shows characteristic overshoot followed by damped oscillation, a signature of the second-order ODE dynamics.
\item \textbf{Stability:} GD baseline is most stable but slowest. Polyak is \emph{unstable} when $f^{\star}$ is misspecified, with divergence rate $\sim \alpha_k \sim f^{\star}_{\text{true}}/\varepsilon$ per step. Adagrad is uniformly stable: its step size is non-increasing by construction, guaranteeing eventual convergence but potentially at a very small rate.
\item \textbf{Practical implication:} In practice, exact $f^{\star}$ is rarely known (and in stochastic settings the irreducible loss floor is not available in closed form), limiting Polyak's applicability. Adagrad suits sparse-gradient settings; RMSprop suits non-stationary objectives; Heavy Ball suits well-conditioned strongly convex problems.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Theoretical Background}

Standard gradient descent on an $L$-smooth convex function achieves $f(x_k) - f^{\star} \leq O(1/k)$.
A fundamental question is whether this rate is optimal.
Nesterov (1983) showed that for first-order methods, $O(1/k^2)$ is optimal for smooth convex functions, and constructed a momentum scheme that achieves this rate.

\begin{theorem}[Nesterov Optimal Rate]
\label{thm:nesterov}
For an $L$-smooth convex function $f$, Nesterov's accelerated gradient method with $\alpha = 1/L$ and $\beta_k = (k-1)/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\end{equation}
This $O(1/k^2)$ rate is optimal among all first-order methods on the class of smooth convex functions.
\end{theorem}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Momentum update with lookahead gradient}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ (an extrapolated point ahead of the current iterate) rather than at $x_k$ itself.
This anticipatory gradient evaluation allows Nesterov to correct its trajectory before overshooting, which is the mechanism behind its superior convergence guarantee.
The adaptive $\beta_k = (k-1)/(k+2)$ schedule increases momentum gradually from 0, achieving the $O(1/k^2)$ rate of Theorem~\ref{thm:nesterov} in the convex setting.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (curvature) estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction: $\mathbb{E}[\hat{m}_t] \approx \mathbb{E}[g_t]$}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias correction: $\mathbb{E}[\hat{v}_t] \approx \mathbb{E}[g_t^2]$}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam (Kingma \& Ba, 2015) combines the directional bias of momentum (via $m_t$, analogous to Nesterov/Heavy Ball) with per-coordinate adaptive scaling (via $v_t$, analogous to RMSprop), inheriting advantages of both.
The bias correction terms $(1 - \beta_j^t)$ counteract the initialisation bias from $m_0 = v_0 = 0$: without correction, early iterates would take steps biased towards zero because the exponential averages have not yet converged.
The effective step size per coordinate is $\alpha_t^{(i)} \approx \alpha / \sqrt{\mathbb{E}[g^{(i)2}]}$, which is roughly $\alpha$ times the signal-to-noise ratio in the gradient.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch.
For $m = 1000$ samples, batch $b = 5$ performs $m/b = 200$ parameter updates per epoch, while $b = 40$ performs $25$ updates per epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The stochastic gradient $g_{\mathcal{B}}$ is an unbiased estimator of the true gradient: $\mathbb{E}_{\mathcal{B}}[g_{\mathcal{B}}] = \nabla J(\theta)$. Its variance is $\operatorname{Var}[g_{\mathcal{B}}] = O(\sigma^2/b)$, where $\sigma^2$ is the gradient variance under a single sample. Smaller batch sizes decrease variance proportionally to $b$ but increase the number of updates per epoch.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.00391 + 0.0625(x - 0.25).
\end{equation}
This is a tangent line that captures only the local slope; it diverges from the true function rapidly.

\textbf{Second-order approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.00391 + 0.0625(x - 0.25) + \tfrac{1}{2} \cdot 0.75(x - 0.25)^2,
\end{equation}
using $g''(x) = 12x^2$, so $g''(0.25) = 12(0.0625) = 0.75$.
Minimising this quadratic model: setting $\partial/\partial x = 0$ gives $x_{\text{Newton}} = x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 = 0.25 - 0.0833 = 0.1667$.
The Newton update $x_{k+1} = x_k - g'(x_k)/g''(x_k)$ can be iterated, and for $g(x) = x^4$ will converge to the true minimiser $x^{\star} = 0$ since $g$ has a unique minimum there.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{figures/q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$. The first-order approximation (tangent line) captures only the local slope and diverges rapidly from the true function. The second-order approximation (parabola) captures curvature and fits the function well in a neighbourhood of $x_0$. The parabola's minimum at $x = 0.167$ is a much better estimate of the true minimum $x^{\star} = 0$ than a gradient step $x_0 - \alpha g'(x_0)$ would be for any fixed $\alpha$, motivating the use of Newton's method.}

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping (Hessian Regularisation)}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Levenberg--Marquardt regularisation}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton direction $p_k = H_k^{-1} g_k$ solves the local quadratic model:
\begin{equation}
\min_p \left\{ f(x_k) + g_k^T p + \tfrac{1}{2} p^T H_k\, p \right\} \quad \Longrightarrow \quad p_k = H_k^{-1} g_k.
\end{equation}
The key theoretical property is \textbf{quadratic local convergence}: near a non-degenerate local minimum $x^{\star}$ where $\nabla^2 f(x^{\star}) \succ 0$, the error satisfies:
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2,
\end{equation}
where $M$ is the Lipschitz constant of $\nabla^2 f$ and $\mu = \lambda_{\min}(\nabla^2 f(x^{\star}))$.
This means the number of correct decimal digits roughly doubles each iteration once $x_k$ is sufficiently close to $x^{\star}$---a dramatic improvement over gradient descent's linear convergence.

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}
% ============================================================

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad e_i = \text{$i$th standard basis vector.}
\end{equation}
This requires $n+1 = 3$ function evaluations in $\mathbb{R}^2$ (one for $f(x)$, one for each coordinate perturbation).
The approximation error has two sources that create an optimal $\delta$:
\begin{itemize}[nosep]
\item \textbf{Truncation error} from the Taylor expansion: $O(\delta)$, favouring small $\delta$.
\item \textbf{Floating-point cancellation}: for small $\delta$, $f(x + \delta e_i) - f(x) \approx O(\varepsilon_{\text{mach}})$, giving error $O(\varepsilon_{\text{mach}}/\delta)$, favouring large $\delta$.
\end{itemize}
The optimal $\delta$ balances these: $\delta^{\star} \sim \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$.
The assignment tests $\delta = 0.05$ (near-optimal for Benchmark B given its gradient scale) and $\delta = 0.8$ (large, causing significant truncation error).

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates a directional derivative along a uniform random unit vector $u_k$, requiring only \emph{two} function evaluations per iteration regardless of dimension $n$.
The estimator $\hat{g}_k$ satisfies $\mathbb{E}[u_k \hat{g}_k^T] = (1/n)\nabla f(x_k)$ (up to normalisation), providing an unbiased but high-variance gradient estimate.
The variance grows as $O(1/\delta^2)$ for the finite-difference component and $O(1)$ from random direction sampling, giving expected convergence $O(1/\sqrt{k})$ for convex functions---significantly slower than GD's $O(1/k)$.

\subsection{Part B: Derivative-Free Methods on Benchmark C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): Move the worst vertex through the centroid of the remaining $n$ vertices.
\item \textbf{Expansion} ($\gamma = 2$): If reflection improved, try a further step in the same direction.
\item \textbf{Outer/inner contraction} ($\rho = 0.5$): If reflection worsened, contract towards the centroid from outside or inside.
\item \textbf{Shrink} ($\sigma = 0.5$): If contraction fails, contract all vertices towards the best vertex.
\end{enumerate}
No gradient information is required; all operations use only function values at the simplex vertices.
The simplex is initialised with step 0.35 from $x_0 = (-1, 1)$.

\subsection{Problem Setup and KKT Conditions}

Minimise Benchmark B subject to the constraint $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$:
\begin{equation}
\min_{x \in \mathbb{R}^2} f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1) \quad \text{subject to} \quad x_1 \geq 0.5.
\end{equation}
This is a constrained optimisation problem with a single inequality constraint $g(x) = 0.5 - x_1 \leq 0$.
The first-order necessary optimality conditions (KKT conditions) are:
\begin{align}
\nabla f(x^{\star}) + \mu \nabla g(x^{\star}) &= 0, \quad \mu \geq 0, \\
g(x^{\star}) &\leq 0, \\
\mu\, g(x^{\star}) &= 0. \quad \text{(complementary slackness)}
\end{align}
Since the unconstrained minimiser is $x^{\star}_{\text{unc}} \approx (0.582, 2.0)$ with $x_1^{\star} = 0.582 > 0.5$, the constraint is \emph{inactive} at the optimum ($g(x^{\star}) = 0.5 - 0.582 < 0$), so by complementary slackness $\mu = 0$, and the KKT conditions reduce to $\nabla f(x^{\star}) = 0$.
This means the constrained and unconstrained optima coincide: $x^{\star}_{\text{con}} = x^{\star}_{\text{unc}} \approx (0.582, 2.0)$ with $f^{\star} = 0.7244$.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Standard gradient step (possibly infeasible)}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project back: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

The projection onto the half-space $\{x : x_1 \geq 0.5\}$ is $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1),\, z_2)^T$---a simple component-wise clipping operation.
For a convex set $\mathcal{X}$ and $L$-smooth $f$, projected GD with $\alpha \leq 1/L$ converges at the same rate as unconstrained GD: $f(x_k) - f^{\star} \leq O(1/k)$ for convex $f$.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Non-smooth penalised objective}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$ \Comment{Gradient of penalised objective}
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty gradient is:
\begin{equation}
\nabla F_{\lambda}(x) = \nabla f(x) + \lambda \nabla[\max(0, 0.5 - x_1)] = \nabla f(x) - \lambda\, \mathbf{1}[x_1 < 0.5]\, e_1,
\end{equation}
where $e_1 = (1, 0)^T$.
As $\lambda \to \infty$, the penalty method solution converges to the true constrained optimum.
However, larger $\lambda$ increases the effective Lipschitz constant of $F_\lambda$ (adding a discontinuous gradient), necessitating a smaller step size for stability.

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected GD} is the method of choice when:
\begin{enumerate}[nosep]
\item Feasibility must be maintained at every iterate (e.g., safety-critical constraints).
\item The projection oracle is computationally cheap (box constraints, half-spaces, $\ell_2$ balls).
\end{enumerate}
Its limitation is that for complex constraint sets (e.g., semidefinite cones, combinatorial constraints), the projection can be as hard as the original problem.

\textbf{Penalty methods} are preferred when:
\begin{enumerate}[nosep]
\item The projection is expensive but gradient computation is cheap.
\item Only approximate feasibility at convergence is required.
\end{enumerate}
The penalty parameter $\lambda$ must be chosen carefully: too small leads to residual violation, too large makes the penalised objective ill-conditioned ($\kappa \propto \lambda$) and requires smaller step sizes.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$, a bounded box polytope with 4 vertices: $(0.5, -5)$, $(5, -5)$, $(5, 10)$, $(0.5, 10)$.

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is:
\begin{equation}
\min_{x \in \mathcal{X}} x_1 + 2x_2 \quad \Longrightarrow \quad x^{\star} = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
Since both components of $a$ are positive, the minimum is attained at the lower bounds of each coordinate independently.
This illustrates the \textbf{fundamental theorem of linear programming}: the optimum of a linear objective over a bounded polytope is attained at a vertex.
Every linear programme over a bounded polytope can be solved in polynomial time (Ellipsoid method, Interior-point method) or by simplex traversal (exponential worst case, but fast in practice).

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: stays in $\mathcal{X}$}
\EndFor
\end{algorithmic}
\end{algorithm}

The key insight is that if $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$ (both feasible), then any convex combination $\beta x_k + (1-\beta) z_k \in \mathcal{X}$ (by convexity of $\mathcal{X}$).
This guarantees feasibility without a projection oracle, at the cost of only linear convergence.

For box constraints, the LMO subproblem decouples per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0, \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0, \end{cases}
\end{equation}
requiring $O(d)$ operations per iteration---far cheaper than a projection onto a general convex set.

\begin{theorem}[Frank--Wolfe Convergence Rate]
For an $L$-smooth convex function $f$ over a convex compact set $\mathcal{X}$ with diameter $D = \max_{x,y \in \mathcal{X}} \|x - y\|$, the Frank--Wolfe method with step sizes $\gamma_k = 2/(k+2)$ achieves:
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2} = O(1/k).
\end{equation}
This $O(1/k)$ rate is slower than projected gradient descent's $O(1/k)$ with $O(\sqrt{k})$ steps to $\varepsilon$-accuracy but with the advantage of no projection oracle.
\end{theorem}

\begin{remark}
The Frank--Wolfe gap $h_k = \max_{y \in \mathcal{X}} \nabla f(x_k)^T (x_k - y) = \nabla f(x_k)^T (x_k - z_k)$ provides a computable upper bound on the suboptimality $f(x_k) - f^{\star}$, making it a useful stopping criterion.
\end{remark}

In our implementation, a fixed mixing parameter $\beta$ is used (replacing the standard $\gamma_k = 1 - \beta$ schedule), trading off convergence guarantee for empirical stability.

\subsection{Interior vs.\ Boundary Behaviour Comparison}

\textbf{Interior optimum:} Frank--Wolfe exhibits a characteristic ``zig-zag'' pattern because the LP vertex $z_k$ is always a corner of $\mathcal{X}$, far from the interior minimum. The iterate $x_{k+1} = \beta x_k + (1-\beta) z_k$ is pulled towards this corner each step and then back as the gradient changes. With the optimal step-size schedule $\gamma_k = 2/(k+2)$, the theoretical rate would be $O(1/k)$; the fixed $\beta$ schedule gives approximately the same rate empirically but with more oscillation near the minimum.

\textbf{Boundary optimum:} Frank--Wolfe is particularly well-suited to boundary optima because the LP oracle naturally selects $z_k = (0.5, y)$ (the constraint-active corner) when the gradient pushes $x_1$ below 0.5. The iterate converges towards the boundary without requiring an explicit projection, demonstrating the core advantage of Frank--Wolfe for linearly constrained problems. The residual gap ($f = 0.364$ vs.\ $f^{\star} = 0.25$) after 140 iterations with $\beta = 0.93$ is consistent with the $O(1/k)$ rate: $f(x_{140}) - f^{\star} \approx 2LD^2/140$, where $D = \max_{x,y \in \mathcal{X}} \|x-y\| \approx \sqrt{4.5^2 + 15^2} \approx 15.7$ and $L = 2$ for $f = x_1^2 + x_2^2$.

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has investigated six families of optimisation algorithms across benchmark problems spanning well-conditioned, anisotropic, and highly ill-conditioned objective functions.
The key findings, organised by algorithm family, are as follows.

\textbf{Adaptive step-size methods (Q1):}
The Polyak step size achieves superior convergence when the optimal value $f^{\star}$ is known exactly (Benchmark C: $f = 0.009$ in 120 iterations), but fails catastrophically when $f^{\star}$ is misspecified (Benchmarks A and B: divergence with $\alpha_k \to f^{\star}_{\text{true}}/\varepsilon$).
Adagrad and Heavy Ball provide robust, parameter-friendly alternatives; RMSprop's exponential averaging offers smoothness at the cost of a harder-to-tune base learning rate.

\textbf{Momentum and stochastic methods (Q2):}
Nesterov acceleration achieves the theoretically optimal $O(1/k^2)$ convergence rate and outperforms all first-order methods on the ill-conditioned Rosenbrock function ($f = 0.479$ vs.\ GD's $f = 3.388$ in 150 iterations), due to momentum accumulation along the consistent valley-floor direction.
Adam's per-coordinate scaling is effective for anisotropic problems but less competitive on strongly nonlinear ill-conditioned landscapes.
Mini-batch SGD exhibits the classical variance--frequency trade-off: smaller batches converge faster per epoch but with higher oscillation near the minimum; increased observation noise raises the irreducible loss floor and diminishes the relative advantage of smaller batches.

\textbf{Newton's method (Q3):}
Newton's method exploits the full curvature of the Hessian to achieve quadratic local convergence $\|x_{k+1} - x^{\star}\| \leq C\|x_k - x^{\star}\|^2$, compared to gradient descent's linear convergence $(1 - 1/\kappa)^k$.
For the quadratic Benchmark A, this reduces to a \emph{single-step} convergence.
On Rosenbrock, 20 damped Newton steps achieve $f = 0.686$ vs.\ GD's $f = 3.73$ in 80 steps---demonstrating the per-iteration superiority of second-order information, at the cost of computing and inverting the $n\times n$ Hessian.

\textbf{Derivative-free methods (Q4):}
Finite-difference gradient approximation with $\delta = 0.05$ closely tracks exact GD, confirming that small $\delta$ provides accurate gradient information.
Large $\delta = 0.8$ introduces truncation errors comparable in magnitude to the gradient, causing trajectory deviation and poor convergence.
Nelder--Mead and grid search trade gradient information for wider applicability to non-differentiable or black-box objectives; grid search's exponential scaling in dimension makes it suitable only for $d \leq 3$.

\textbf{Constrained optimisation (Q5):}
Projected gradient descent enforces feasibility as a hard constraint at every iteration, achieving zero constraint violation from iteration 1 at the cost of a projection oracle.
Since the constraint is inactive at the true optimum (KKT multiplier $\mu = 0$), projected GD converges to the same point as unconstrained GD.
Penalty methods with large $\lambda$ converge to the constrained minimum with residual violation decreasing approximately $\propto 1/\lambda$, but require smaller step sizes due to increased effective Lipschitz constants.

\textbf{Frank--Wolfe (Q6):}
Frank--Wolfe eliminates the projection oracle requirement through a linear minimisation oracle, maintaining feasibility via convex combinations.
Its $O(1/k)$ convergence rate matches the theoretical lower bound for first-order methods without projection, but oscillates at interior optima due to LP vertex jumping.
At boundary optima, the algorithm naturally selects constraint-active vertices, making it particularly effective for such problems.

\textbf{Overall conclusions:}
The choice of algorithm should be guided by:
(1) \emph{problem structure} (quadratic, convex, non-convex, constrained);
(2) \emph{available information} (exact gradient, approximate gradient, function values only);
(3) \emph{computational budget} (per-iteration cost vs.\ convergence rate);
(4) \emph{problem conditioning} (condition number $\kappa$ determines how much momentum/Newton help).
For well-conditioned convex problems, any first-order method works well.
For ill-conditioned problems, Nesterov acceleration and Newton's method provide significant practical advantages.
For constrained problems, projected gradient descent is preferable when projection is cheap, and Frank--Wolfe when only a linear oracle is available.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):

    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex}]
def nelder_mead(loss_fn, x0, step, n_iters, alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1])
        f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1])
            f_e = loss_fn(x_e)
            if f_e < f_r: simplex[-1] = x_e; f_vals[-1] = f_e
            else: simplex[-1] = x_r; f_vals[-1] = f_r
        else:
            if f_r < f_vals[-1]:
                x_c = centroid + rho * (x_r - centroid)
                f_c = loss_fn(x_c)
                if f_c <= f_r: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
            else:
                x_c = centroid + rho * (simplex[-1] - centroid)
                f_c = loss_fn(x_c)
                if f_c < f_vals[-1]: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
