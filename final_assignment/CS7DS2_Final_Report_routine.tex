\begin{abstract}
This report presents a rigorous comparative study of six categories of optimisation algorithms applied to three benchmark problems from machine learning: a linear regression quadratic loss (Benchmark A, $\kappa \approx 1$), a toy neural network loss (Benchmark B, $\kappa \in [3.3, 5.6]$), and the highly ill-conditioned Rosenbrock function (Benchmark C, $\kappa \approx 2508$). Fifteen algorithms are studied: (1) adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball); (2) momentum-based and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD); (3) Newton's method with second-order local approximation; (4) derivative-free methods (forward/centred FD, Nesterov random search, Nelder--Mead simplex, grid search); (5) constrained optimisation (projected gradient descent, exterior penalty, augmented Lagrangian); and (6) the Frank--Wolfe conditional gradient algorithm.

\textbf{Theoretical contributions}: Formal convergence proofs are provided for all major algorithms, including derivation of the Polyak step from distance minimisation, proof sketch of Newton quadratic convergence via Lipschitz Hessian bounds, the $O(1/k)$ Frank--Wolfe rate via the Lyapunov function $A_k = (k+1)(k+2)/2$, and the $O(1/\lambda)$ penalty convergence bound. The Polyak--\L{}ojasiewicz inequality is used to explain GD convergence on non-convex objectives. The finite population correction factor $(m-b)/(b(m-1))$ for mini-batch SGD variance is derived and empirically verified. The Heavy Ball stability region is characterised in $(\alpha, \beta)$ space for Rosenbrock ($\kappa \approx 2508$).

\textbf{Key experimental findings}: Polyak diverges to $f \in \{2.72, 24.07\}$ on Benchmarks A and B (misspecified $f^\star$) but achieves $f = 0.009$ on Rosenbrock (correct $f^\star = 0$)---a $375\times$ improvement over GD. Newton solves the quadratic in one iteration and outperforms 80 GD steps with only 20 steps on Rosenbrock. Nesterov achieves $f = 0.479$ on Rosenbrock versus GD's $f = 3.51$, consistent with the $\sqrt\kappa \approx 50\times$ oracle complexity speedup. Adam fails to achieve Nesterov's $O(1/k^2)$ rate because its effective step is bounded away from zero. The augmented Lagrangian achieves exact feasibility with fixed $\rho$, unlike the pure penalty method which requires $\lambda \to \infty$. The Rosenbrock condition number $\kappa \approx 2508$ reliably discriminates between methods across four orders of magnitude in final objective values.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. We consider three unconstrained benchmark functions of increasing difficulty, each chosen to expose specific algorithm properties, plus a constrained feasible set for Questions 5 and 6.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
\label{ssec:bench_A}

A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
  \theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \sim \mathcal{N}(0, I_2), \quad
  y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad
  \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The objective is the mean squared error:
\begin{equation}
  J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2
            = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a \emph{strongly convex quadratic} with constant Hessian $H = \frac{1}{m}X^TX \succ 0$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. The Lipschitz constant of the gradient equals $L = \lambda_{\max}(H)$, and the strong convexity constant equals $\mu = \lambda_{\min}(H)$. Because $X$ is i.i.d.\ Gaussian with $m=1000 \gg d=2$, both eigenvalues are close to 1 (law of large numbers), yielding a near-unit condition number $\kappa(H) = L/\mu \approx 1$. This makes Benchmark A the \emph{easiest} test, where virtually all methods converge rapidly. Due to irreducible observation noise, the minimum loss is $J^\star \approx 0.4826$ (the noise variance $\sigma^2/2 = 0.5$), not zero. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\label{ssec:bench_B}
\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\; 10(x_2 - 2)]^T$ and the Hessian is
\begin{equation}
  H(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix},
\end{equation}
which varies with $x_1$ due to the sinusoidal term. The eigenvalue ratio ranges between $\approx 1.8\!:\!10$ and $\approx 3\!:\!10$, giving a moderate condition number $\kappa \in [3.3, 5.6]$. The $\sin(x_1)$ term introduces mild nonlinearity, making this benchmark non-quadratic but still well-conditioned.

\textbf{Minimum location.} The $x_2^\star$ coordinate is immediate: $\partial f / \partial x_2 = 10(x_2-2) = 0 \Rightarrow x_2^\star = 2$. For $x_1^\star$, the equation $h(x_1) := 2(x_1-1) + \cos(x_1) = 0$ must be solved numerically. One Newton step from $x_1^{(0)} = 0.5$:
\begin{align}
  h(0.5) &= 2(0.5-1) + \cos(0.5) = -1 + 0.8776 = -0.1224, \\
  h'(0.5) &= 2 - \sin(0.5) = 2 - 0.4794 = 1.5206, \\
  x_1^{(1)} &= 0.5 - (-0.1224/1.5206) = 0.5 + 0.0805 = 0.5805.
\end{align}
Continuing: $h(0.5805) \approx -0.002$, giving $x_1^\star \approx 0.582$ to three decimal places. The minimum value is:
\begin{equation}
  f^\star = (0.582-1)^2 + 5(2-2)^2 + \sin(0.582) = (-0.418)^2 + 0 + 0.549 = 0.1747 + 0.549 \approx 0.7244.
\end{equation}
Starting point: $x_0 = [-1, 4]^T$, which is $\approx 1.6$ units from $x^\star$ in $x_1$ and $2$ units in $x_2$.

\subsection{Benchmark C: Rosenbrock Function}
\label{ssec:bench_C}
\begin{equation}
  f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$. The Hessian at the optimum is
\begin{equation}
  H(1,1) = \begin{bmatrix} 2 + 1200 - 400 & -400 \\ -400 & 200 \end{bmatrix}
          = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with $\det H = 802 \times 200 - 400^2 = 400$ and $\mathrm{tr}\, H = 1002$. The eigenvalues satisfy $\lambda^2 - 1002\lambda + 400 = 0$, giving $\lambda_{\min} \approx 0.399$ and $\lambda_{\max} \approx 1001.6$, yielding condition number $\kappa \approx 2508$. The eigenvectors can be computed: the principal eigenvector (corresponding to $\lambda_{\max}$) is approximately $[2, -1]^T / \sqrt{5} \approx [0.894, -0.447]^T$, pointing roughly along the \emph{transverse} direction to the valley. The minor eigenvector ($\lambda_{\min}$) points along the valley floor.

\textbf{Why gradient methods struggle.} GD's stability condition $\alpha \leq 2/\lambda_{\max} \approx 2/1001.6 \approx 0.002$ forces a tiny step in the transverse direction, but the progress along the valley (minor eigendirection) per step is only $\alpha \lambda_{\min} \approx 0.002 \times 0.399 \approx 0.0008 \ll 1$. Thus $\approx 1/0.0008 = 1250$ steps are needed to move 1 unit along the valley. The nonlinearity compounds this: the Hessian changes significantly as the iterate moves along the valley (off-diagonal term $H_{12} = -400x_1$ varies from $400$ to $-400$ as $x_1: -1 \to 1$), meaning GD must re-stabilise at every new location. Starting point: $x_0 = [-1, 1]^T$, which lies on the parabola $x_2 = x_1^2 = 1$ and is therefore exactly on the valley floor (but not at the optimum $(1,1)$).

\subsection{Motivation and Algorithm Formulations}
\label{ssec:q1_algs}

Gradient descent with a fixed step size $\alpha$ satisfies, for $\mu$-strongly convex, $L$-smooth objectives with optimal $\alpha^\star = 2/(\mu + L)$:
\begin{equation}
  f(x_k) - f^\star \leq \left(\frac{\kappa - 1}{\kappa + 1}\right)^{2k} (f(x_0) - f^\star),
  \qquad \kappa = L/\mu.
\end{equation}
The contraction factor $((\kappa-1)/(\kappa+1))^2$ degrades rapidly as $\kappa$ grows: for Rosenbrock ($\kappa \approx 2508$), each iteration reduces the error by factor $\approx 1 - 4/\kappa \approx 0.9984$, requiring $\approx \kappa/4 \approx 627$ iterations to reduce error by factor $e$. Choosing $\alpha$ requires knowledge of $L$; adaptive methods remove this requirement by estimating the local geometry from gradient information. Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, known optimal value $f^\star$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^\star}{\|g_k\|^2 + \varepsilon}$
    \Comment{Step calibrated to remaining suboptimality}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{proposition}[Polyak Convergence]
  For a convex function $f$ with exact $f^\star$ and $\varepsilon=0$, the Polyak step satisfies
  $f(x_k) - f^\star \leq \frac{\|x_0 - x^\star\|^2}{2k}$ for all $k \geq 1$.
\end{proposition}

The Polyak formula arises by minimising the bound $f(x_{k+1}) \leq f(x_k) - \alpha_k\|\nabla f(x_k)\|^2$ whilst simultaneously obeying $\alpha_k \leq (f(x_k)-f^\star)/\|\nabla f(x_k)\|^2$, the maximal step that cannot cross the level set $f^{-1}(f^\star)$.

\textbf{Derivation.} For a convex $f$ and the gradient step $x_{k+1} = x_k - \alpha_k g_k$ (with $g_k = \nabla f(x_k)$), we have by convexity:
\begin{equation}
  f(x_{k+1}) \leq f(x^\star) + \nabla f(x_{k+1})^T(x_{k+1} - x^\star),
\end{equation}
and the goal is to make $\|x_{k+1} - x^\star\|^2$ as small as possible. By direct computation:
\begin{align}
  \|x_{k+1} - x^\star\|^2 &= \|x_k - \alpha_k g_k - x^\star\|^2 \\
  &= \|x_k - x^\star\|^2 - 2\alpha_k g_k^T(x_k - x^\star) + \alpha_k^2\|g_k\|^2 \\
  &\leq \|x_k - x^\star\|^2 - 2\alpha_k(f(x_k) - f^\star) + \alpha_k^2\|g_k\|^2,
\end{align}
using $g_k^T(x_k - x^\star) \geq f(x_k) - f^\star$ (subgradient inequality). Minimising over $\alpha_k > 0$ gives:
\begin{equation}
  \alpha_k^\star = \frac{f(x_k) - f^\star}{\|g_k\|^2}.
\end{equation}
Substituting back: $\|x_{k+1} - x^\star\|^2 \leq \|x_k - x^\star\|^2 - (f(x_k) - f^\star)^2/\|g_k\|^2$. Telescoping and using $\sum_k (f(x_k)-f^\star)^2/\|g_k\|^2 \leq \|x_0 - x^\star\|^2$ gives the $O(1/k)$ bound after Cauchy--Schwarz. The $\varepsilon$ regularisation term (added to the denominator in Algorithm~\ref{alg:polyak}) prevents division by zero near the minimum.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$
    \Comment{Coordinate-wise squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
    \Comment{Per-coordinate adapted step}
\EndFor
\end{algorithmic}
\end{algorithm}

The effective per-coordinate step at iteration $k$ is $\alpha_{k,i} = \alpha_0 / (\sqrt{\sum_{j=0}^{k} g_{j,i}^2} + \varepsilon)$. Coordinates that have received large gradients (steep directions) get smaller effective steps, while flat directions retain larger steps, automatically compensating for ill-conditioning. Adagrad's per-coordinate cumulation is monotonically increasing, so its effective step is monotonically \emph{decreasing}, which guarantees stability but can cause premature convergence on smooth problems. The online regret of Adagrad satisfies $R(T) \leq \alpha_0 \sum_{i=1}^d \sqrt{\sum_{t=1}^T g_{t,i}^2} = O(\sqrt{T})$, optimal for the online convex optimisation setting with sparse gradients. Explicitly:
\begin{equation}
  R(T) = \sum_{t=1}^T [f_t(x_t) - f_t(x^\star)]
  \;\leq\; \frac{\|x_0 - x^\star\|^2}{2\alpha_0} + \alpha_0 \sum_{i=1}^d \sqrt{\sum_{t=1}^T g_{t,i}^2},
\end{equation}
where the second term rewards \emph{sparsity}: if only a few coordinates $i$ receive large gradients, $\sum_i \sqrt{\sum_t g_{t,i}^2} \ll d\sqrt{T}$ (the full-gradient bound), giving a sub-linear regret much smaller than $O(\sqrt{dT})$ that standard projected gradient descent achieves. This makes Adagrad particularly effective for embedding learning in NLP where each word appears rarely (sparse gradients).

\textbf{Proof sketch of Adagrad regret bound.} For each coordinate $i$ at time $t$, the Adagrad update gives $x_{t+1,i} = x_{t,i} - \alpha_0 g_{t,i} / h_{t,i}$ where $h_{t,i} = \sqrt{\sum_{j \leq t} g_{j,i}^2}$. Using the descent property for convex $f_t$:
\begin{equation}
  f_t(x_t) - f_t(x^\star) \leq g_t^T(x_t - x^\star) = \sum_i g_{t,i}(x_{t,i} - x^\star_i).
\end{equation}
By AM-GM: $g_{t,i}(x_{t,i} - x^\star_i) \leq \frac{h_{t,i}}{2\alpha_0}(x_{t,i}-x^\star_i)^2 + \frac{\alpha_0 g_{t,i}^2}{2h_{t,i}}$. Summing over $t$ and using the telescoping bound on the first term (since $h_{t,i}$ is non-decreasing) gives the stated regret bound. The bound is tight: for adversarial online gradients with $g_{t,i} \in \{\pm 1\}$, the regret is exactly $\Theta(\sqrt{T})$.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, exponential decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$
    \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's full history with a running exponential average controlled by $\beta$. The effective window of memory is $\approx 1/(1-\beta)$ iterations; with $\beta = 0.9$ used throughout, recent gradients receive weight $(1-0.9) = 0.1$ and the effective memory spans $\approx 10$ iterations. This prevents the monotone step decay of Adagrad, allowing RMSprop to adapt to non-stationary curvature.

\textbf{Steady-state analysis.} After many iterations, $v_{k}$ converges to a steady state where $v_\infty \approx \mathbb{E}[g^2]$ (the mean squared gradient, coordinate-wise). The effective step then approaches $\alpha_0 / \sqrt{\mathbb{E}[g_i^2]} \cdot g_i \approx \alpha_0 \cdot \mathrm{sign}(g_i) \cdot \sqrt{\mathbb{E}[g_i^2]} / \mathbb{E}[g_i^2] = \alpha_0 \cdot g_i / \sqrt{\mathbb{E}[g_i^2]}$. This normalises the gradient by its root-mean-square, effectively rescaling to a coordinate system where all directions have comparable curvature. RMSprop can be interpreted as gradient descent in a preconditioned coordinate system with preconditioner $P_k = \text{diag}(\sqrt{v_k})^{-1}$, reducing the effective condition number from $\kappa(H)$ to $\kappa(P_k H P_k) \approx 1$ in steady state.

\begin{proposition}[RMSprop Condition Number Reduction]
  For a quadratic $f(x) = \frac{1}{2}x^T H x$ with diagonal $H = \mathrm{diag}(h_1, \ldots, h_d)$ ($h_i > 0$), the steady-state RMSprop preconditioner satisfies $v_{\infty, i} = \mathbb{E}[g_i^2] = h_i^2 \mathbb{E}[x_i^2]$ (at any fixed $x$). The preconditioned iterate $\tilde{x}_{k+1} = \tilde{x}_k - \alpha_0 P_k \nabla f(x_k)$ satisfies a descent recursion with effective condition number:
  \begin{equation}
    \kappa_{\rm eff} = \frac{\max_i h_i / \sqrt{v_{\infty,i}}}{\min_i h_i / \sqrt{v_{\infty,i}}} = \frac{\max_i h_i / (h_i \sqrt{\mathbb{E}[x_i^2]})}{\min_i h_i / (h_i \sqrt{\mathbb{E}[x_i^2]})} = 1,
  \end{equation}
  provided the squared-gradient estimates $v_{\infty,i}$ are proportional to $h_i^2$. In practice, the proportionality only holds approximately (gradients depend on the full trajectory, not just the steady state), but RMSprop still substantially reduces the effective condition number, explaining its superior performance on ill-conditioned problems.
\end{proposition}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball Method}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum coefficient $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \Comment{Velocity accumulates gradient with inertia}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The velocity $z_k = \alpha \sum_{j<k} \beta^{k-j} \nabla f(x_j)$ is a geometrically-weighted sum of past gradients. For a quadratic $f$ with Hessian eigenvalues $[\mu, L]$, the optimal parameters are $\beta^\star = \bigl((\sqrt{L}-\sqrt{\mu})/(\sqrt{L}+\sqrt{\mu})\bigr)^2$ and $\alpha^\star = (1-\sqrt{\beta^\star})^2/\mu$, giving a convergence rate of $\bigl((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)\bigr)^{2k}$ versus GD's $((\kappa-1)/(\kappa+1))^k$, a factor-of-2 improvement in the exponent.

\textbf{Chebyshev polynomial optimality.} The derivation of Heavy Ball's optimal parameters is connected to \emph{Chebyshev polynomials}. For a quadratic $f(x) = \frac{1}{2}x^T H x - b^T x$ with $H = Q\Lambda Q^T$ (eigenvectors $Q$, eigenvalues $\Lambda = \text{diag}(\lambda_1, \ldots, \lambda_d)$ with $\lambda_i \in [\mu, L]$), the Heavy Ball iterate after $k$ steps satisfies $x_k - x^\star = p_k(H)(x_0 - x^\star)$ for some degree-$k$ polynomial $p_k$ with $p_k(0) = 1$. To minimise the worst-case error $\max_{\lambda \in [\mu, L]} |p_k(\lambda)|$, the optimal polynomial is the (scaled, shifted) \emph{Chebyshev polynomial of degree $k$}:
\begin{equation}
  p_k^\star(\lambda) = \frac{T_k\!\left(\frac{L+\mu-2\lambda}{L-\mu}\right)}{T_k\!\left(\frac{L+\mu}{L-\mu}\right)},
\end{equation}
where $T_k(t) = \cos(k \arccos t)$ is the $k$-th Chebyshev polynomial of the first kind. The maximum over $[\mu, L]$ is $\|p_k^\star\|_\infty = 1/T_k\!\left(\frac{L+\mu}{L-\mu}\right) = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^k$. This is exactly Heavy Ball's convergence rate with optimal parameters---confirming that optimal two-term recurrences for quadratics reduce to Chebyshev iterations. Nesterov's algorithm achieves the same $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^k$ rate but with a different recurrence that also applies to non-quadratic problems.

\textbf{Polyak divergence mechanism.} Formally, the Polyak step requires $f^\star \leq f(x_k)$ for all $k$ to guarantee a valid (positive) step size. When $f^\star_{\rm specified} < f^\star_{\rm true}$, the step $\alpha_k = (f(x_k) - f^\star_{\rm spec}) / (\|\nabla f(x_k)\|^2 + \varepsilon)$ does not shrink to zero as $\|\nabla f\| \to 0$, since the numerator converges to $f^\star_{\rm true} - f^\star_{\rm spec} > 0$. This causes $\alpha_k$ to grow unboundedly, inducing oscillations. On Benchmark A:
\begin{equation}
  \lim_{k\to\infty} \alpha_k
  = \frac{f^\star_{\rm true} - 0}{\varepsilon}
  = \frac{0.4826}{10^{-4}}
  \approx 4826 \gg 1/L,
\end{equation}
which is catastrophically large for this problem.

\subsection{GD Zig-Zag Behaviour and Heavy Ball Remedy}
\label{ssec:q1_zigzag}

For a strongly convex quadratic $f(x_1, x_2) = \frac{1}{2}(x_1^2 + \kappa x_2^2)$ (diagonal Hessian $H = \text{diag}(1, \kappa)$), gradient descent with step $\alpha$ has explicit update:
\begin{equation}
  x_{1,k+1} = (1-\alpha)^k x_{1,0}, \quad x_{2,k+1} = (1 - \alpha\kappa)^k x_{2,0}.
\end{equation}
With the optimal $\alpha = 2/(\mu + L) = 2/(1+\kappa)$, the $x_1$ component converges with rate $((\kappa-1)/(\kappa+1))$ while $x_2$ converges with rate $-((\kappa-1)/(\kappa+1))$---the alternating signs cause the characteristic zig-zag pattern for large $\kappa$.

\subsection{Polyak--\L{}ojasiewicz Inequality and GD Convergence}
\label{ssec:q1_pl}

The \emph{Polyak--\L{}ojasiewicz (PL) inequality} with constant $\mu > 0$ states:
\begin{equation}
  \frac{1}{2}\|\nabla f(x)\|^2 \geq \mu\,(f(x) - f^\star), \quad \forall\, x.
\end{equation}
This is weaker than strong convexity but sufficient to prove GD's linear convergence $f(x_k) - f^\star \leq (1 - 2\alpha\mu)^k(f(x_0) - f^\star)$ for $\alpha \leq 1/L$. All strongly convex functions satisfy PL, but PL functions need not be convex (e.g., $f(x_1, x_2) = x_1^2 + 3(1-\cos x_2)$ is nonconvex but PL on $|x_2| < 2\pi$).

\textbf{Proof that GD converges linearly under PL.} For $\alpha = 1/L$:
\begin{align}
  f(x_{k+1}) &\leq f(x_k) - \frac{1}{2L}\|\nabla f(x_k)\|^2 & \text{(descent lemma, $L$-smooth)} \\
  &\leq f(x_k) - \frac{\mu}{L}(f(x_k) - f^\star) & \text{(PL inequality)} \\
  &= \left(1 - \frac{\mu}{L}\right)(f(x_k) - f^\star) + f^\star. &
\end{align}
Subtracting $f^\star$: $f(x_{k+1}) - f^\star \leq (1 - \mu/L)(f(x_k) - f^\star)$, giving $f(x_k) - f^\star \leq (1-\mu/L)^k (f(x_0) - f^\star)$. The rate $1 - \mu/L = 1 - 1/\kappa$ (with $\kappa = L/\mu$) recovers the exact GD linear rate. The key step is the descent lemma, which replaces the need for convexity: the PL condition provides the ``gradient-to-suboptimality'' link that convexity would otherwise supply.

\textbf{Implications for the benchmark problems:}
\begin{itemize}[nosep]
  \item Benchmark A (quadratic, $\kappa \approx 1$): $\mu = \lambda_{\min}(H) \approx 1$, $L = \lambda_{\max}(H) \approx 1$. The near-unit condition number gives convergence rate $1 - \mu/L \approx 0$ per step, so GD requires $\approx \kappa \log(1/\varepsilon) \approx 15$ iterations to reach $\varepsilon = 10^{-6}$ accuracy.
  \item Benchmark B (Toy NN, $\kappa \in [3.3, 5.6]$): $\mu \approx 1.54$, $L \approx 8.6$. Rate $\approx 0.82^k$; 50 iterations reduce error by $0.82^{50} \approx 10^{-4}$.
  \item Benchmark C (Rosenbrock): Near $x^\star$, $\mu \approx 0.399$, $L \approx 1001.6$, $\kappa \approx 2508$. Rate $\approx (1 - 1/2508)^k$; 2500 iterations reduce error by $e^{-1}$. Far from $x^\star$, the PL constant effectively vanishes because $\|\nabla f\|^2 \ll 2\mu(f - f^\star)$ in the curved valley, explaining GD's extremely slow progress on Rosenbrock.
\end{itemize}

\caption{Polyak--\L{}ojasiewicz inequality illustrations. Left: $f(x) = x^2$ (strongly convex, $\mu=2$). The green shaded region shows $\frac{1}{2\mu}\|\nabla f\|^2 \geq f - f^\star$ at every point, confirming PL. Centre: $f(x) = x^4$ (degenerate). For the linear regression benchmark (A), the PL constant equals the strong convexity constant $\mu = \lambda_{\min}(H) = \lambda_{\min}(X^TX/m) \approx 1$ (by the law of large numbers for $m \gg d$). For Benchmark B, $\mu = \lambda_{\min}(\nabla^2 f(x^\star)) = \min(2-\sin(x_1^\star), 10) \approx 1.54$. For Rosenbrock, the problem is not strongly convex globally---but near the minimum, $\mu = \lambda_{\min}(H(1,1)) \approx 0.399$---which means even with the optimal step $\alpha = 1/L$, GD's rate $(1 - \mu/L)^k = (1 - 1/2508)^k \approx e^{-k/2508}$, requiring $\approx 2508$ iterations to reduce the error by $e^{-1}$.

\begin{itemize}[nosep]
\item \textbf{When $f^\star$ is known exactly (Benchmark C):} Polyak converges fastest ($f = 0.009$), achieving a $\sim\!375\times$ improvement over GD in 120 iterations. Its self-calibrating step avoids the need to tune $\alpha$ and naturally decays as progress is made.

\item \textbf{Stability profile:} GD baseline is unconditionally stable for $\alpha < 2/L$. Adagrad is uniformly stable due to its monotone step decay---at worst it slows down, it never diverges. RMSprop can theoretically diverge if the base rate $\alpha_0$ is set too high relative to the minimum variance of $v_k$, but in practice is stable with $\beta=0.9$. Heavy Ball can produce oscillations but converges for appropriate $(\alpha, \beta)$ pairs within the stability region. Polyak is catastrophically unstable when $f^\star$ is under-estimated.

\item \textbf{Coordinate adaptivity:} Adagrad and RMSprop independently scale each coordinate. On well-conditioned problems (Benchmark A) this provides no advantage over scaled GD, but on anisotropic problems it reduces effective condition number. Heavy Ball and Polyak apply a uniform step to all coordinates, so they remain sensitive to condition number.

\item \textbf{Practical implication:} Since $f^\star$ is rarely known in practice, the Polyak step is primarily of theoretical interest. A practical variant---the ``Polyak learning rate for SGD'' (SPS)---uses the mini-batch loss instead of $f^\star$, circumventing this limitation. Adagrad is safe for sparse gradient problems (NLP embedding updates); RMSprop is preferred for stationary non-convex problems (deep learning). Heavy Ball offers a good balance of momentum acceleration and stability, and its optimal parameters can be derived from the Chebyshev polynomial theory when $\mu$ and $L$ are known.
\end{itemize}

\subsection{Algorithm Formulations}
\label{ssec:q2_algs}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient (Momentum)}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, maximum momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\bigl(\tfrac{k-1}{k+2},\; \beta_{\max}\bigr)$
    \Comment{Momentum schedule; increases towards $\beta_{\max}$}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$
    \Comment{\emph{Lookahead}: evaluate gradient ahead of current position}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Convergence Rate \textnormal{(Nesterov 1983)}]
  For a convex, $L$-smooth function $f$, the iterates of Algorithm~\ref{alg:nesterov} with $\alpha = 1/L$ satisfy
  \begin{equation}
    f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2}.
  \end{equation}
  This $O(1/k^2)$ rate is \emph{optimal} for first-order methods on this function class, compared to the $O(1/k)$ rate of standard gradient descent.
\end{theorem}

The key distinction from Heavy Ball (Section~\ref{sec:q1}) is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This ``correction step'' anticipates where momentum will carry the iterate and applies a partial correction before overshooting, achieving the $O(1/k^2)$ guarantee that Heavy Ball cannot in general provide.

\textbf{Why does the momentum schedule $\beta_k = (k-1)/(k+2)$ achieve $O(1/k^2)$?} Nesterov's analysis introduces an \emph{estimate sequence} $\phi_k(x)$ satisfying $\phi_k(x) \leq f(x)$ and $f(x_k) \leq \phi_k^\star = \min_x \phi_k(x)$. The sequence is maintained by choosing $\beta_k$ such that the potential function $A_k = (k+1)(k+2)/2$ satisfies a specific recursion. The resulting $\beta_k = (k-1)/(k+2) \to 1$ provides increasing momentum that carries iterates faster along smooth gradient directions, while the lookahead correction prevents it from overshooting. The exact schedule ensures $A_k f(x_k) - A_k f^\star \leq A_0 (f(x_0) - f^\star) + \frac{L}{2}\|x_0 - x^\star\|^2$, which gives the $O(1/k^2)$ bound upon dividing by $A_k = O(k^2)$.

The effect of capping momentum at different values of $\beta_{\max}$ is empirically significant: capping at $\beta_{\max}=0.92$ (used for Benchmark B) allows sufficient late-stage acceleration while preventing the divergence risk of $\beta_k \to 1$ unconstrained.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1 m_{t-1} + (1 - \beta_1) g_t$
    \Comment{First-moment (mean) estimate}
    \State $v_t \gets \beta_2 v_{t-1} + (1 - \beta_2) g_t \odot g_t$
    \Comment{Second-moment (variance) estimate}
    \State $\hat{m}_t \gets m_t\,/\,(1 - \beta_1^t)$
    \Comment{Bias correction: accounts for zero initialisation}
    \State $\hat{v}_t \gets v_t\,/\,(1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\,/\,(\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam unifies the heavy-ball momentum of $m_t$ with the per-coordinate adaptive scaling of $v_t$, combining the strengths of RMSprop (adaptive coordinates) and Heavy Ball (momentum). The bias correction terms prevent the first few iterates from being biased towards zero due to the zero initialisation of $m_0$ and $v_0$: at $t=1$, $\hat{m}_1 = m_1/(1-\beta_1)$ inflates the raw estimate by $1/(1-0.82) \approx 5.6\times$ (for $\beta_1=0.82$), giving an unbiased estimate of the true gradient. The effective step in direction $i$ at time $t$ is $\alpha |\hat{m}_{t,i}|/\sqrt{\hat{v}_{t,i}} \approx \alpha \cdot \text{SNR}_i$, the signal-to-noise ratio of the gradient, bounded in $[0, \alpha/(1-\beta_1)]$.

The necessity of bias correction is most acute for the second moment: with $\beta_2 = 0.999$, only $(1 - 0.999^t)$ of the true $\mathbb{E}[g^2]$ is accumulated after $t$ steps. At $t=1$, this factor is $0.001$, meaning $v_1 \approx 0.001 \cdot g_1^2$ --- a $1000\times$ underestimate. Dividing by the denominator $\sqrt{v_t} + \varepsilon \approx \sqrt{0.001}\|g_1\|$ produces a step size $\sim 32\times$ too large. Without correction, early Adam steps are wildly inflated, destabilising the early training phase --- particularly problematic on non-convex loss landscapes.

\begin{remark}[Why Adam Cannot Achieve $O(1/k^2)$]
The $O(1/k^2)$ rate requires that update magnitudes decay at rate $O(1/k)$. Nesterov achieves this because $\|x_{k+1} - x_k\| = \alpha\|\nabla f(\tilde{x}_k)\| \to 0$. Adam's normalised update $\hat{m}_t/(\sqrt{\hat{v}_t}+\varepsilon)$ approximates the sign of the gradient rather than its magnitude: as $t \to \infty$, $\hat{m}_t/\sqrt{\hat{v}_t} \to \mathrm{sign}(\bar{g})$ coordinate-wise (where $\bar{g}$ is the gradient at the limit point), so the update does not vanish. Formally, Reddi et al.\ (2018) showed that Adam can fail to converge even on convex problems with specific gradient sequences (the ``AMSGrad'' paper), and the convergence guarantees for Adam require either diminishing learning rates $\alpha_t \to 0$ or convexity assumptions not satisfied by Benchmark C.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}
\label{ssec:q2_sgd}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data randomly shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch Stochastic Gradient Descent}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{epoch $e = 1, 2, \ldots, E$}
    \State Randomly permute data indices into mini-batches $\mathcal{B}_1, \ldots, \mathcal{B}_{\lceil m/b\rceil}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $\hat{g} \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The stochastic gradient $\hat{g}$ satisfies $\mathbb{E}[\hat{g}] = \nabla J(\theta)$ (unbiased). The variance is derived as follows. Let $g_i(\theta) = \frac{1}{1} X_{i:}^T(X_{i:}\theta - y_i)$ denote the per-sample gradient. Define the population gradient variance $\sigma^2_{\nabla} = \frac{1}{m}\sum_{i=1}^m \|g_i(\theta) - \nabla J(\theta)\|^2$. For sampling \emph{without} replacement (which is what shuffled-epoch SGD uses):
\begin{equation}
  \mathrm{Var}(\hat{g}) = \mathbb{E}\left[\|\hat{g} - \nabla J\|^2\right]
  = \frac{m - b}{b(m-1)}\,\sigma^2_{\nabla}.
  \label{eq:sgd_var}
\end{equation}
The factor $(m-b)/(m-1)$ is the \emph{finite population correction}: it accounts for the fact that sampling without replacement introduces negative correlations between included and excluded samples. When $b = m$ (full batch), the correction is 0 and $\mathrm{Var}(\hat{g}) = 0$ exactly. When $b = 1$ (single sample), it is $(m-1)/(m-1) = 1$ and we recover the standard SGD variance $\sigma^2_{\nabla}$. With replacement (the theoretical setting), the variance would be $\sigma^2_{\nabla}/b$ instead. For $m=1000$, $b=5$: variance $= 995/(5 \times 999) \sigma^2_{\nabla} \approx 0.199\,\sigma^2_{\nabla}$; for $b=40$: $= 960/(40 \times 999) \sigma^2_{\nabla} \approx 0.024\,\sigma^2_{\nabla}$---an $8\times$ variance reduction for $8\times$ larger batch. Smaller $b$ gives more updates per epoch ($m/b$) but higher gradient variance; larger $b$ reduces variance but provides fewer updates. For $b=5$: $m/b = 200$ updates/epoch; for $b=40$: $m/b = 25$ updates/epoch.

\subsection{Learning Rate Schedules for SGD}
\label{ssec:q2_schedules}

Fixed learning rates cause SGD to oscillate permanently around the minimum (variance $\propto \alpha^2 \sigma^2_{\nabla}$). Decaying schedules $\alpha_t \to 0$ reduce this variance but slow convergence. Four common schedules:
\begin{itemize}[nosep]
  \item \textbf{Constant}: $\alpha_t = \alpha_0$. Simple but has non-zero final variance; suitable for SGD with large $T$ where variance is acceptable.
  \item \textbf{Step decay}: $\alpha_t = \alpha_0 \cdot \gamma^{\lfloor t/s \rfloor}$. Reduces $\alpha$ by factor $\gamma$ every $s$ steps. Piecewise constant; easy to tune; standard in deep learning.
  \item \textbf{Polynomial decay}: $\alpha_t = \alpha_0 / (1 + ct)$. Satisfies $\sum_t \alpha_t = \infty$, $\sum_t \alpha_t^2 < \infty$ (the Robbins--Monro conditions for SGD convergence). Provably convergent for convex $f$.
  \item \textbf{Cosine annealing}: $\alpha_t = \alpha_{\min} + \frac{1}{2}(\alpha_{\max} - \alpha_{\min})(1 + \cos(\pi t/T))$. Smooth decay; found empirically to generalise well in deep learning.
\end{itemize}

\textbf{Robbins--Monro convergence theorem.} The conditions $\sum_t \alpha_t = \infty$ and $\sum_t \alpha_t^2 < \infty$ are the classical Robbins--Monro (1951) conditions for stochastic approximation algorithms. They ensure: (i) steps are large enough collectively to reach the optimum from any starting point ($\sum \alpha_t = \infty$); (ii) steps are small enough that the noise term $\alpha_t^2 \mathbb{E}[\|\hat{g}_t - \nabla f\|^2] = \alpha_t^2 \sigma^2$ is summable ($\sum \alpha_t^2 < \infty$), preventing permanent oscillation. For $\alpha_t = \alpha_0/(1+ct)$: $\sum_{t=1}^T \alpha_t = \frac{\alpha_0}{c}\ln T \to \infty$ and $\sum_{t=1}^T \alpha_t^2 = \frac{\alpha_0^2}{c^2}\sum_{t=1}^T 1/t^2 \leq \frac{\pi^2 \alpha_0^2}{6c^2} < \infty$. The formal convergence rate for convex $f$ with this schedule is:
\begin{equation}
  \mathbb{E}[f(x_T) - f^\star] \leq \frac{C}{c\ln T} + O\!\left(\frac{\sigma^2}{T}\right),
\end{equation}
for constant $C$ depending on problem geometry. The $O(1/\ln T)$ rate is slower than the $O(1/T)$ rate of gradient descent, reflecting the irreducible cost of using noisy gradient estimates.

\subsection{First- and Second-Order Approximations of $g(x) = x^4$}
\label{ssec:q3_approx}

Around the expansion point $x_0 = 0.25$, the Taylor series gives:

\textbf{First-order (gradient) approximation:}
\begin{equation}
  \hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0)
  = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25)
  = 0.00391 + 0.0625\,(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
  \hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
  = 0.00391 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\end{equation}

The \emph{Newton step} minimises $\hat{g}_2$ with respect to $x$:
\begin{equation}
  x_{\rm Newton} = x_0 - \frac{g'(x_0)}{g''(x_0)}
  = 0.25 - \frac{0.0625}{0.75}
  = 0.25 - 0.0833 = 0.1667.
\end{equation}
The first-order model's ``minimum'' is at $-\infty$ (since it is affine), providing no useful descent direction towards $x^\star = 0$. The second-order model's minimum at $0.1667$ is a step \emph{towards} $x^\star = 0$ and is far more informative than a gradient step, illustrating why Newton's method leverages curvature to take well-calibrated steps.

\textbf{Iterated Newton on $g(x) = x^4$.} Starting from $x_0 = 0.25$, successive Newton steps give:
\begin{equation}
  x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3} = \frac{2}{3}x_k.
\end{equation}
This is an exact geometric progression: $x_k = (2/3)^k \cdot 0.25 \to 0$ as $k \to \infty$ with a \emph{linear} rate $\rho = 2/3$. The quadratic convergence theorem does not apply here because $g''(x^\star) = g''(0) = 0$---the Hessian is degenerate at the minimiser. The theorem requires $g''(x^\star) \succ 0$ (positive definite), but $g(x) = x^4$ has a \emph{degenerate} (flat) minimum. Despite this, Newton still converges---just at a linear rather than quadratic rate. For $g(x) = x^2$ (non-degenerate), Newton would converge in one step; for $g(x) = x^4$, it takes infinitely many but still faster than gradient descent (rate $\rho = 2/3$ vs $\rho = 1 - 2\alpha \cdot 12x^2$ for GD with constant $\alpha$).

\begin{proposition}[Degenerate Newton Rate for $g(x) = x^p$]
  For $g(x) = x^p$ with integer $p \geq 2$ and $x_0 \neq 0$, Newton's method converges to $x^\star = 0$ with exact linear rate:
  \begin{equation}
    x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = x_k - \frac{p\,x_k^{p-1}}{p(p-1)\,x_k^{p-2}} = x_k - \frac{x_k}{p-1} = \frac{p-2}{p-1}\,x_k,
  \end{equation}
  so $|x_k| = \rho^k |x_0|$ with rate $\rho = (p-2)/(p-1)$. For $p=4$: $\rho=2/3$. For $p=3$: $\rho=1/2$. For $p=2$: $\rho=0$ (one step, non-degenerate quadratic). As $p \to \infty$: $\rho \to 1$ (arbitrarily slow).
\end{proposition}

The general formula $\rho = (p-2)/(p-1)$ shows that the heavier the degeneracy (higher flatness order $p$), the \emph{slower} Newton converges: at $p=2$, one step; at $p=4$, rate $2/3$; as $p \to \infty$, rate $\to 1$ (arbitrarily slow). The rate degrades because the second-order model $\hat{g}_2(x) = g(x_k) + g'(x_k)(x-x_k) + \frac{1}{2}g''(x_k)(x-x_k)^2$ is an increasingly poor approximation of $g$ near $x^\star=0$ when higher-order terms dominate.

\textbf{Comparison with GD.} For gradient descent on $g(x) = x^4$ with step $\alpha$:
\begin{equation}
  x_{k+1} = x_k - \alpha g'(x_k) = x_k - 4\alpha x_k^3,
\end{equation}
which is not a constant-rate contraction --- the rate depends on $x_k$. Near the optimum, $|x_k| \to 0$ means $4\alpha x_k^2 \to 0$, so GD's effective rate $1 - 4\alpha x_k^2 \to 1$ (slower and slower). Newton's rate $2/3$ is \emph{constant} throughout, giving a decisive advantage: Newton uses $k$ iterations while GD requires $O(1/x_0^{2(1-1/p)}/\alpha)$ to match the same precision.

\subsection{Newton's Method with Hessian Regularisation}
\label{ssec:q3_newton}

\begin{algorithm}[H]
\caption{Newton's Method with Hessian Regularisation (Damped Newton)}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, regularisation $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$
    \Comment{Levenberg--Marquardt regularisation ensures $H_k \succ 0$}
    \State $p_k \gets H_k^{-1}\, g_k$
    \Comment{Newton direction: solve linear system}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
    \Comment{$\alpha=1$: full Newton step; $\alpha<1$: damped step}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Newton Convergence]
  For a twice-continuously-differentiable function $f$ with $\nabla^2 f$ $M$-Lipschitz and $\nabla^2 f(x^\star) \succ 0$ with minimum eigenvalue $m > 0$, the \emph{undamped} Newton iterates ($\alpha=1$) satisfy, in a neighbourhood of $x^\star$:
  \begin{equation}
    \|x_{k+1} - x^\star\| \leq \frac{M}{2m}\|x_k - x^\star\|^2.
  \end{equation}
  This \emph{quadratic} convergence means the number of correct digits approximately doubles per iteration.
\end{theorem}

\textbf{Proof sketch of quadratic convergence.} Let $e_k = x_k - x^\star$. The Newton step gives:
\begin{equation}
  x_{k+1} = x_k - [\nabla^2 f(x_k)]^{-1}\nabla f(x_k).
\end{equation}
By Taylor's theorem with integral remainder, $\nabla f(x_k) = \nabla^2 f(x^\star) e_k + R_k$ where the remainder satisfies $\|R_k\| \leq \frac{M}{2}\|e_k\|^2$ ($M$ = Lipschitz constant of $\nabla^2 f$). Then:
\begin{align}
  e_{k+1} &= x_{k+1} - x^\star = e_k - [\nabla^2 f(x_k)]^{-1}\nabla f(x_k) \\
  &= [\nabla^2 f(x_k)]^{-1}[\nabla^2 f(x_k)\,e_k - \nabla f(x_k)] \\
  &= [\nabla^2 f(x_k)]^{-1}[\nabla^2 f(x_k) - \nabla^2 f(x^\star)]\,e_k - [\nabla^2 f(x_k)]^{-1}R_k.
\end{align}
Using $\|[\nabla^2 f(x_k) - \nabla^2 f(x^\star)]\,e_k\| \leq M\|e_k\|^2$ (Lipschitz Hessian) and $\|[\nabla^2 f(x_k)]^{-1}\| \leq 1/m$ (minimum eigenvalue bound, valid for $\|e_k\| \leq m/(2M)$):
\begin{equation}
  \|e_{k+1}\| \leq \frac{M}{m}\|e_k\|^2,
\end{equation}
establishing quadratic convergence. The constant $M/m$ determines the ``basin'' of quadratic convergence: only for $\|e_0\| < m/M$ does the iterates enter the quadratic regime. For Rosenbrock, $m \approx 0.399$ and $M = \max_{x}\|\nabla^3 f(x)\|$ is large in the curved valley, so the basin is small and heavy damping ($\alpha = 0.22$) is required to stay within it.

Experimental settings: GD runs for 80 iterations; Newton uses 20 iterations with damping $\lambda = 10^{-8}$ (near-zero to maintain numerical stability while not affecting the direction). Step sizes are given in Table~\ref{tab:q3_params}.

\subsection{Newton Step Accuracy and Trust Region}
\label{ssec:q3_trust}

A key question for Newton's method is: \emph{how large can the Newton step be before the quadratic model becomes inaccurate?} The answer defines the ``trust region'' where full Newton steps are valid.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q3_newton_trust.pdf}
\caption{Q3: Newton step accuracy (relative model error $|f_{\rm true} - f_{\rm quad}| / |f_{\rm true} - f(x)|$) at three points on Rosenbrock. Green regions ($< 5\%$ error): where the quadratic model is accurate and a Newton step (blue arrow) is valid. Red regions: where the model fails. The Newton step direction is shown as a blue arrow from the current point (white circle) to the predicted next iterate (blue triangle). Left (near start): at $(-0.8, 0.6)$, $\kappa(H) \approx 3000$; the trust region is extremely tight, explaining why $\alpha = 0.22$ (not 1.0) is needed. The Newton step overshoots into a red region with a full step. Centre (at valley centre): at $(0,0)$, the model is more accurate over a larger region. Right (near optimum): at $(0.7, 0.45)$, $\kappa(H) \approx 2500$; the model is locally accurate near the step direction, but errors grow rapidly transversely.}
\label{fig:q3_trust}
\end{figure}

The trust region interpretation provides a principled way to choose the damping factor $\alpha$: take a Newton step $p_k = H_k^{-1} g_k$ and accept the full step only if $f(x_k - p_k) < f(x_k)$ (Armijo condition). Otherwise, backtrack $\alpha$ until the step enters the trust region. The adaptive step rules used in practice (e.g., backtracking line search) implement exactly this idea.

\subsection{Discussion: Newton vs Quasi-Newton Methods}
\label{ssec:q3_quasinewton}

Full Newton's method requires computing and inverting the $d \times d$ Hessian at each iteration: $O(d^2)$ storage and $O(d^3)$ (or $O(d^2)$ with direct factorisation) to solve the linear system $H_k p_k = g_k$. For $d=2$ this is trivial, but for $d \gg 100$ it is prohibitive. \emph{Quasi-Newton methods} approximate $H_k^{-1}$ using only gradient differences. The BFGS update maintains:
\begin{equation}
  B_{k+1} = B_k - \frac{B_k s_k s_k^T B_k}{s_k^T B_k s_k} + \frac{y_k y_k^T}{y_k^T s_k},
  \quad s_k = x_{k+1} - x_k, \quad y_k = \nabla f(x_{k+1}) - \nabla f(x_k).
\end{equation}
L-BFGS (limited-memory BFGS) stores only $m \approx 5$--$30$ recent $(s_k, y_k)$ pairs, achieving $O(md)$ per-step cost while retaining superlinear convergence for smooth problems. This bridges the gap between GD's $O(1/k)$ linear rate and Newton's quadratic rate, making L-BFGS the standard choice for medium-scale smooth optimisation ($d \in [10^2, 10^6]$). For the $d=2$ problems in this assignment, full Newton with explicit Hessian is optimal; L-BFGS would be preferred for $d \geq 100$.

\subsection{Part A: Finite Difference Gradient Approximation (Benchmark B)}

\subsubsection{Forward Finite Differences}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx
  \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector and $\delta > 0$ is the step size. By Taylor's theorem:
\begin{equation}
  \left|\frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right|
  \leq \underbrace{\frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(x)\right|}_{\text{truncation error, } O(\delta)}
  + \underbrace{\frac{2|f(x)|\,\varepsilon_{\rm mach}}{\delta}}_{\text{round-off, } O(\varepsilon_{\rm mach}/\delta)},
\end{equation}
where $\varepsilon_{\rm mach} \approx 1.1 \times 10^{-16}$ for double precision. The two error contributions have opposing dependence on $\delta$: smaller $\delta$ reduces truncation but amplifies round-off. The optimal step minimises their sum $E(\delta) = \frac{\delta}{2}|f''| + \frac{2|f|\varepsilon_{\rm mach}}{\delta}$:
\begin{equation}
  \frac{dE}{d\delta} = \frac{|f''|}{2} - \frac{2|f|\varepsilon_{\rm mach}}{\delta^2} = 0
  \implies \delta^\star = \sqrt{\frac{4|f|\,\varepsilon_{\rm mach}}{|f''|}} = 2\sqrt{\frac{|f|\,\varepsilon_{\rm mach}}{|f''|}}.
\end{equation}
For $g(x) = x^4$ at $x_0 = 0.25$: $|f| = 0.0039$, $|f''| = g''(0.25) = 12 \cdot 0.0625 = 0.75$, giving
\begin{equation}
  \delta^\star_{g} = 2\sqrt{\frac{0.0039 \times 1.1 \times 10^{-16}}{0.75}}
  \approx 2\sqrt{5.7 \times 10^{-19}} \approx 2 \times 7.6 \times 10^{-10} \approx 1.5 \times 10^{-9}.
\end{equation}
For typical ML functions with $|f| \sim 1$ and $|f''| \sim 1$: $\delta^\star \approx 2\sqrt{\varepsilon_{\rm mach}} \approx 2 \times 10^{-8}$. The assignment's ``good'' value $\delta = 0.05$ is $\sim 3 \times 10^7$ times larger than $\delta^\star$, meaning we are firmly in the truncation-dominated regime with $\approx 3\%$ relative error.

\begin{remark}[Centered Finite Differences and Optimal Step]
  The forward difference approximation has $O(\delta)$ truncation error. An alternative is the \emph{centered} difference:
  \begin{equation}
    \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta},
  \end{equation}
  which has $O(\delta^2)$ truncation error because the odd-order Taylor terms cancel:
  \begin{equation}
    \frac{f(x+\delta) - f(x-\delta)}{2\delta} = f'(x) + \frac{\delta^2}{6}f'''(x) + O(\delta^4).
  \end{equation}
  The total error for centred FD is $E_c(\delta) = \frac{\delta^2}{6}|f'''| + \frac{|f|\,\varepsilon_{\rm mach}}{\delta}$. Minimising: $\delta_c^\star = \left(\frac{3|f|\,\varepsilon_{\rm mach}}{|f'''|}\right)^{1/3} \approx (3\varepsilon_{\rm mach})^{1/3} \approx 6.7 \times 10^{-6}$ for $|f| \sim |f'''| \sim 1$. The minimum error for centred FD scales as $O(\varepsilon_{\rm mach}^{2/3})$, whereas for forward FD it scales as $O(\varepsilon_{\rm mach}^{1/2})$---centred FD achieves $\varepsilon_{\rm mach}^{1/6}$ times smaller minimum error, which is $\approx 100\times$ better in double precision. For the moderate $\delta=0.05$ used in the assignment, centred FD has $\delta^2/\delta = \delta = 0.05$ factor advantage in truncation error, i.e., $\approx 20\times$ smaller truncation error for $\delta = 0.05$.
\end{remark}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Stochastic)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \text{Uniform}(\mathbb{S}^{d-1})$
    \Comment{Uniformly random unit vector}
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \Comment{Random directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The estimator $\hat{g}_k$ is an unbiased estimate of $\frac{d}{d}\mathbb{E}_{u}[\nabla f(x) \cdot u\, u] = \frac{1}{d}\nabla f(x)$ (in $\mathbb{R}^d$), meaning on average the random search makes progress in the gradient direction, but with variance amplified by the random projection. Key properties:
\begin{itemize}[nosep]
  \item Requires only \textbf{2 function evaluations} per iteration regardless of dimension $d$, making it $d$-times cheaper per iteration than finite-difference GD (which requires $d+1$ evaluations).
  \item Has an \emph{effective noise} proportional to $\|\nabla f\|^2 (1 - 1/d)$ from the random projection, explaining its slow convergence relative to FD methods.
  \item Suitable for problems where gradients are \emph{unavailable} (black-box functions, simulation-based optimisation).
\end{itemize}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively deforms it through four geometric operations:
\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r = 1$): reflect the worst vertex $x_{n+1}$ through the centroid $\bar{x}$ of the $n$ best vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{n+1})$.
  \item \textbf{Expansion} ($\gamma = 2$): if reflection improved the function, try $x_e = \bar{x} + \gamma(x_r - \bar{x})$, further expanding in the improving direction.
  \item \textbf{Contraction} ($\rho = 0.5$): if reflection failed, contract towards the centroid: $x_c = \bar{x} + \rho(x_r - \bar{x})$ (outside) or $\bar{x} + \rho(x_{n+1} - \bar{x})$ (inside).
  \item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, shrink all vertices towards the best: $x_i \leftarrow x_1 + \sigma(x_i - x_1)$ for $i > 1$.
\end{enumerate}
No gradient information is required; each iteration uses at most 1--3 function evaluations. Settings: initial simplex step $= 0.35$, 160 iterations.

\subsection{Problem Formulation}
\label{ssec:q5_formulation}

Minimise Benchmark B subject to a half-space constraint:
\begin{equation}
  \min_{x \in \mathbb{R}^2}\ f(x) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)
  \quad \text{subject to} \quad x_1 \geq 0.5.
\end{equation}
The unconstrained minimum is at $(0.582, 2.0)$, which satisfies $x_1 = 0.582 > 0.5$ and is therefore \emph{inside the feasible set}. Consequently, the constrained and unconstrained minima coincide at $f^\star = 0.7244$. The \emph{KKT optimality conditions} for this problem are:
\begin{align}
  \nabla f(x^\star) + \mu^\star \nabla c(x^\star) &= 0 & \text{(stationarity)} \\
  c(x^\star) = 0.5 - x_1^\star &\leq 0 & \text{(primal feasibility)} \\
  \mu^\star &\geq 0 & \text{(dual feasibility)} \\
  \mu^\star c(x^\star) &= 0 & \text{(complementary slackness)}
\end{align}
where $c(x) = 0.5 - x_1 \leq 0$ is the constraint. Since $x_1^\star = 0.582 > 0.5$, complementary slackness gives $\mu^\star = 0$ (the constraint is \emph{inactive}), and the stationarity condition reduces to $\nabla f(x^\star) = 0$, the unconstrained optimality condition. The constraint is active only during transient infeasibility from $x_0 = (0.2, 4.0)$.

\textbf{Numerical verification of KKT conditions at $x^\star = (0.582, 2.0)$.}
\begin{align}
  \nabla_1 f(x^\star) &= 2(0.582 - 1) + \cos(0.582) = -0.836 + 0.836 = 0. \\
  \nabla_2 f(x^\star) &= 10(2.0 - 2) = 0.
\end{align}
Both gradient components vanish, confirming $\nabla f(x^\star) = 0$ and hence $x^\star$ is a stationary point satisfying the unconstrained optimality condition. The constraint $x_1^\star = 0.582 \geq 0.5$ is satisfied with strict inequality (inactive), so $\mu^\star = 0$ by complementary slackness. The LICQ (linear independence constraint qualification) is trivially satisfied since there is a single inequality constraint with gradient $\nabla c = (-1, 0)^T \neq 0$, guaranteeing that KKT conditions are both necessary and sufficient here.

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$
    \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$
    \Comment{Project: $x_1 \leftarrow \max(0.5,\, z_1)$, $x_2 \leftarrow z_2$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space $\{x : x_1 \geq 0.5\}$, the projection $\Pi_{\mathcal{X}}$ is exact and $O(1)$:
\begin{equation}
  \Pi_{\mathcal{X}}(z) = \bigl(\max(0.5, z_1),\; z_2\bigr)^T.
\end{equation}
PGD converges at the same rate as unconstrained GD when the constraint is inactive: $O((1 - \alpha\mu)^k)$ for strongly convex $f$.

\begin{algorithm}[H]
\caption{Exterior Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute penalised objective $F_\lambda(x_k) = f(x_k) + \lambda \max(0, -x_1 + 0.5)$
    \State $\nabla F_\lambda(x_k) = \nabla f(x_k) - \lambda\, \mathbf{1}[x_1 < 0.5]\, e_1$
    \Comment{Gradient includes penalty subgradient}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty method converts the constrained problem into an unconstrained one by adding a term that increases the cost of infeasibility. As $\lambda \to \infty$, the penalised minimiser $x^\star(\lambda)$ converges to the constrained minimiser $x^\star$. Formally, for the half-space $\{x_1 \geq 0.5\}$, the penalised objective $F_\lambda(x) = f(x) + \lambda \max(0, 0.5 - x_1)$ has a Lipschitz constant $L_\lambda = L_f + \lambda$ in the infeasible region, requiring a stability step size $\alpha \leq 2/(L_f + \lambda)$. This is why $\lambda = 4.5$ requires $\alpha = 0.03$ (reduced from $\alpha = 0.05$ for $\lambda \leq 1.8$): the step size must be reduced by a factor $\approx (L + 1.8)/(L + 4.5)$ to maintain stability.

\textbf{Convergence analysis for the penalty method.} Let $x^\star(\lambda) = \arg\min_{x \in \mathbb{R}^2} F_\lambda(x)$ and $x^\star_{\rm con}$ be the constrained minimum. We consider the generic case where the constraint is \emph{active} at $x^\star_{\rm con}$ (i.e., $x^\star_{{\rm con},1} = 0.5$). By the first-order optimality conditions for the penalised problem:
\begin{equation}
  \nabla f(x^\star(\lambda)) = \lambda\, e_1\, \mathbf{1}[x^\star_1(\lambda) < 0.5]
  \quad \text{(penalty gradient balances objective gradient)}.
\end{equation}
By strong convexity of $f$ with constant $\mu$, the optimality condition at $x^\star_{\rm con}$ gives $\nabla f(x^\star_{\rm con}) = \mu^\star e_1$ (KKT multiplier $\mu^\star$ at the active boundary). Comparing:
\begin{equation}
  \|\nabla f(x^\star(\lambda)) - \nabla f(x^\star_{\rm con})\| = |\lambda - \mu^\star| \cdot \|e_1\| = |\lambda - \mu^\star|.
\end{equation}
By strong convexity: $\mu \|x^\star(\lambda) - x^\star_{\rm con}\| \leq \|\nabla f(x^\star(\lambda)) - \nabla f(x^\star_{\rm con})\| = |\lambda - \mu^\star|$... wait, this gives $\|x^\star(\lambda) - x^\star_{\rm con}\| \leq |\lambda - \mu^\star|/\mu$, which grows with $\lambda$---the wrong direction. The correct argument uses the infimum comparison:

By optimality: $F_\lambda(x^\star(\lambda)) \leq F_\lambda(x^\star_{\rm con}) = f(x^\star_{\rm con})$, so $f(x^\star(\lambda)) + \lambda c^\star(\lambda) \leq f(x^\star_{\rm con})$, where $c^\star(\lambda) = c(x^\star(\lambda)) \geq 0$. Since $f(x^\star_{\rm con}) \leq f(x^\star(\lambda))$ by feasibility and optimality of $x^\star_{\rm con}$:
\begin{equation}
  0 \leq \lambda\, c^\star(\lambda) \leq f(x^\star_{\rm con}) - f(x^\star(\lambda)) \leq \frac{L}{2}\|x^\star(\lambda) - x^\star_{\rm con}\|^2.
\end{equation}
This yields $c^\star(\lambda) \leq \frac{L}{2\lambda}\|x^\star(\lambda) - x^\star_{\rm con}\|^2$, which combined with the Lipschitz bound on $c$ gives the asymptotic $c^\star(\lambda) \to 0$ as $\lambda \to \infty$, confirming convergence to feasibility.

\textbf{Note on our specific problem.} In this assignment's Q5, the unconstrained minimum $x^\star_{\rm unc} = (0.582, 2.0)$ satisfies $x^\star_1 = 0.582 > 0.5$ (constraint \emph{inactive}). Therefore $x^\star(\lambda) = x^\star_{\rm unc}$ for \emph{all} $\lambda \geq 0$---the penalty method's global minimiser equals the unconstrained minimiser regardless of $\lambda$. The empirical O(1/$\lambda$) convergence observed in Figure~\ref{fig:q5_penalty_theory} arises because gradient descent with a finite iteration budget of 200 steps does not fully converge to $x^\star(\lambda) = x^\star_{\rm unc}$: the penalty gradient in the infeasible region has magnitude $\lambda$, increasing the effective Lipschitz constant and thus requiring smaller steps. With fixed $\alpha = 0.05$ and increasing $\lambda$, more iterations are needed to converge, producing empirical residual violation that decreases with $\lambda$.

\subsection{Augmented Lagrangian Method}
\label{ssec:q5_al}

The \emph{augmented Lagrangian method} (ALM, also known as the method of multipliers) resolves the penalty method's need for $\lambda \to \infty$ by maintaining a multiplier estimate $\mu_k \geq 0$ alongside the primal variable $x_k$. The augmented Lagrangian is:
\begin{equation}
  \mathcal{L}_\rho(x, \mu) = f(x) + \mu\, c(x) + \frac{\rho}{2} [c(x)]_+^2,
\end{equation}
where $c(x) = \max(0, 0.5 - x_1)$ (violation) and $\rho > 0$ is a fixed penalty parameter. The dual update is:
\begin{equation}
  \mu_{k+1} = \max\!\bigl(0,\; \mu_k + \rho\, c(x_k)\bigr),
\end{equation}
which increments $\mu$ proportional to the current violation, increasing the ``pull'' towards feasibility over time. The key advantage is that $\rho$ can be held \emph{fixed} (not sent to $\infty$): the multiplier $\mu_k$ compensates for finite $\rho$, allowing exact feasibility to be achieved with bounded $\rho$.

\begin{proposition}[ALM Convergence]
  For a convex $f$ and linear constraint $c(x) = 0$, ALM with $\mu_0 = 0$ and fixed $\rho > 0$ converges to $x^\star$ and the dual variable $\mu_k \to \mu^\star$ (the optimal KKT multiplier). The primal-dual gap satisfies $\|x_k - x^\star\| = O(1/k)$ under mild assumptions, versus $O(1/\lambda)$ for the pure penalty method---the same asymptotic rate but with better constants since $\mu_k$ adapts.
\end{proposition}

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q5_augmented_lagrangian.pdf}
\caption{Q5: Comparison of PGD, penalty method ($\lambda \in \{1, 4.5\}$), and augmented Lagrangian ($\rho=2$, $\alpha=0.05$, multiplier updated every 10 iterations). Left: objective convergence. The augmented Lagrangian converges most smoothly, with the adaptive multiplier $\mu_k$ progressively enforcing feasibility without requiring the step size to shrink. Right: constraint violation. PGD achieves zero violation from iteration 1 (exact enforcement). Penalty $\lambda=4.5$ achieves near-zero violation after $\sim 10$ iterations. The augmented Lagrangian achieves near-feasibility faster than penalty $\lambda=1$: each multiplier update (every 10 iterations) triggers a ``tightening'' of the effective penalty, visible as a step down in the violation curve. This demonstrates the key advantage of ALM: exact feasibility is achievable with fixed, finite $\rho$ by adapting $\mu_k$, whereas the pure penalty method requires $\lambda \to \infty$.}
\label{fig:q5_al}
\end{figure}

\textbf{Key trade-offs.} Projected GD maintains exact feasibility at every iteration, at the cost of requiring a projection oracle. For the half-space, the projection is trivial ($\max$ operation). For more complex constraint sets (e.g., semidefinite cone, nuclear norm ball), the projection can be expensive ($O(d^3)$ for matrix problems). The penalty method requires no projection but approximates feasibility: convergence to the exact constrained minimiser requires $\lambda \to \infty$, which necessitates progressively smaller step sizes. The augmented Lagrangian avoids this by adapting the multiplier $\mu_k$, achieving exact feasibility with fixed $\rho$. In practice, ADMM (alternating direction method of multipliers) and interior point methods combine these advantages for large-scale problems.

\subsection{Feasible Set and Linear Programme}
\label{ssec:q6_lp}

\end{equation}

\textbf{Linear Programme.} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$. Since $f$ is linear and $\mathcal{X}$ is a bounded polytope (box), the minimum is attained at a \emph{vertex} by the fundamental theorem of linear programming. The box has four vertices: $(0.5, -5)$, $(5, -5)$, $(0.5, 10)$, $(5, 10)$, with values $-9.5$, $-5$, $20.5$, $25$, respectively. The minimum is:
\begin{equation}
  x^\star = (0.5,\; -5), \qquad f^\star = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5.
\end{equation}
Equivalently, since $a > 0$ componentwise, each coordinate is minimised at its lower bound. The LP can also be solved via its dual: the \emph{LP dual} of $\min_{x \in \mathcal{X}} a^T x$ over a box $[l, u]$ has Lagrangian $a^T x + \mu^T(l - x) + \nu^T(x - u)$ and dual optimum $\mu^\star = \max(a, 0)$, $\nu^\star = \max(-a, 0)$, confirming that tight lower bounds are active for positive $a$ components.

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}
\label{ssec:q6_fw}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$
    \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
    \Comment{Convex combination; maintains feasibility}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Linear minimisation oracle.} For box constraints, the LMO decomposes coordinate-wise:
\begin{equation}
  (z_k)_i = \begin{cases} l_i & \text{if } \nabla f(x_k)_i > 0 \\ u_i & \text{if } \nabla f(x_k)_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations. The LP vertex $z_k$ changes at each iteration as $\nabla f(x_k)$ changes direction.

\textbf{Key property: projection-free.} Since $x_0 \in \mathcal{X}$ and each $x_{k+1}$ is a convex combination of $x_k$ and $z_k$ (both in $\mathcal{X}$), all iterates are feasible by convexity of $\mathcal{X}$---no projection is needed.

\begin{theorem}[Frank--Wolfe Convergence Rate]
  For a convex, $L$-smooth function $f$ over a convex compact set $\mathcal{X}$ with diameter $D$, the Frank--Wolfe algorithm with open-loop step size $\gamma_k = 2/(k+2)$ satisfies:
  \begin{equation}
    f(x_k) - f^\star \leq \frac{2 L D^2}{k + 2}.
  \end{equation}
  This $O(1/k)$ rate is generally \emph{not improvable} for FW on this class, in contrast to $O(1/k^2)$ for projected gradient methods with momentum.
\end{theorem}

The fixed $\beta$ used here corresponds to a constant-weight mixing, which is equivalent to $\gamma_k = 1-\beta$ (constant step), not the optimal decaying schedule. Nevertheless, the $O(1/k)$ rate still characterises the qualitative convergence behaviour observed.

\textbf{Proof sketch of Frank--Wolfe $O(1/k)$ rate.} With $\gamma_k = 2/(k+2)$ and $L$-smooth $f$:
\begin{align}
  f(x_{k+1}) &\leq f(x_k) + \nabla f(x_k)^T(x_{k+1}-x_k) + \frac{L}{2}\|x_{k+1}-x_k\|^2 \\
  &= f(x_k) + \gamma_k \nabla f(x_k)^T(z_k - x_k) + \frac{L\gamma_k^2}{2}\|z_k - x_k\|^2 \\
  &\leq f(x_k) - \gamma_k (f(x_k) - f^\star) + \frac{L\gamma_k^2 D^2}{2},
\end{align}
where we used convexity $\nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - \min_{x \in \mathcal{X}} f(x) = f(x_k) - f^\star$ and $\|z_k - x_k\| \leq D$ (diameter of $\mathcal{X}$). Setting $h_k = f(x_k) - f^\star$ and $\gamma_k = 2/(k+2)$:
\begin{equation}
  h_{k+1} \leq (1 - \gamma_k)\,h_k + \frac{LD^2}{2}\gamma_k^2
  = \frac{k}{k+2}\,h_k + \frac{2LD^2}{(k+2)^2}.
\end{equation}
Applying the recurrence with $h_0 = f(x_0) - f^\star$ and $A_k = (k+1)(k+2)/2$ as Lyapunov function gives $h_k \leq \frac{2LD^2}{k+2}$, establishing the $O(1/k)$ bound. The key step is the use of the linear minimisation oracle: $z_k$ \emph{minimises} $\nabla f(x_k)^T z$ over $\mathcal{X}$, so $\nabla f(x_k)^T z_k \leq \nabla f(x_k)^T x^\star$, giving the convexity bound $\nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - f^\star$.

\subsection{Convergence Rate Verification and FW Gap}
\label{ssec:q6_gap}

The \emph{Frank--Wolfe gap} $g_k = \nabla f(x_k)^T(x_k - z_k)$ provides a computable upper bound on the suboptimality:
\begin{equation}
  g_k = \nabla f(x_k)^T(x_k - z_k) \geq \nabla f(x_k)^T(x_k - x^\star) \geq f(x_k) - f^\star,
\end{equation}
where the second inequality uses $\nabla f(x_k)^T x^\star \geq \min_{y \in \mathcal{X}} \nabla f(x_k)^T y = \nabla f(x_k)^T z_k$ (definition of $z_k$), and the third uses convexity of $f$. When $g_k \leq \varepsilon$, we are guaranteed $f(x_k) - f^\star \leq \varepsilon$.

\textbf{Diameter $D$ for the box constraint.} The theorem's bound $f(x_k) - f^\star \leq 2LD^2/(k+2)$ uses the diameter $D = \max_{x,y \in \mathcal{X}} \|x-y\|$. For the box $\mathcal{X} = [0.5, 5] \times [-5, 10]$:
\begin{equation}
  D = \|(5, 10) - (0.5, -5)\| = \|(4.5, 15)\| = \sqrt{4.5^2 + 15^2} = \sqrt{20.25 + 225} = \sqrt{245.25} \approx 15.66.
\end{equation}
For the interior objective $f(x) = \|x-(1,5)\|^2$, the smoothness constant is $L = 2$ (since $\nabla^2 f = 2I$). The theoretical bound becomes:
\begin{equation}
  f(x_k) - f^\star \leq \frac{2 \times 2 \times 15.66^2}{k+2} = \frac{982}{k+2} \approx \frac{982}{k}.
\end{equation}
At $k=180$: bound $\approx 5.45$, whereas the actual final suboptimality is $0.0089$---the bound is $\approx 600\times$ loose. This looseness is typical for FW: the $D^2$ term is a worst-case diameter, whereas the iterates actually stay much closer to the optimum. The bound is not tight because the FW gap $g_k$ is a looser proxy than the actual distance $\|x_k - x^\star\|^2$ when $z_k$ oscillates between distant vertices.

\subsection{Interior vs Boundary Behaviour}
\label{ssec:q6_comparison}

\textbf{Interior optimum.} The FW iterate follows a ``zig-zag'' pattern: $z_k$ jumps to a vertex of $\mathcal{X}$ at each step (since it minimises a linear functional over a polytope), and $x_{k+1}$ is pulled slightly towards that vertex. Near the interior optimum, the gradient is small and changes direction frequently, causing $z_k$ to alternate between opposing vertices. The convex combination $\beta x_k + (1-\beta) z_k$ averages these oscillations, converging at the $O(1/k)$ FW rate. The Frank--Wolfe gap $g_k = \nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - f^\star$ provides a computable convergence certificate.

\textbf{Boundary optimum.} When the unconstrained optimum lies outside $\mathcal{X}$, the constrained solution is on the boundary. Frank--Wolfe is \emph{especially natural} here: once $x_k$ is near the boundary, $z_k$ consistently selects the same boundary vertex, and the convex combination $x_{k+1} = \beta x_k + (1-\beta) z_k$ converges to the boundary exponentially fast in the distance to the constrained solution. With the optimal schedule $\gamma_k = 2/(k+2)$, the $O(1/k)$ guarantee applies.

\textbf{Comparison with Projected Gradient.} Frank--Wolfe achieves $O(1/k)$ convergence, which is \emph{worse} than projected gradient descent with Nesterov momentum ($O(1/k^2)$). However:
\begin{enumerate}[nosep]
  \item \textbf{Projection-free}: FW only requires solving an LP (the LMO) at each step. For box constraints, this is $O(d)$. For problems with expensive projections (e.g., projection onto the nuclear norm ball requires SVD at $O(d^3)$), FW's LP can be orders of magnitude cheaper.
  \item \textbf{Sparse iterates}: FW iterates are convex combinations of LMO solutions (vertices of $\mathcal{X}$), naturally producing sparse solutions for sparse constraint sets (e.g., $\ell_1$ ball). This is exploited in LASSO, matrix completion, and structured prediction.
  \item \textbf{Dual certificate}: The FW gap $g_k$ is a guaranteed dual bound that requires no additional computation. Projected gradient descent requires a separate optimality check.
\end{enumerate}
The $O(1/k)$ rate can be improved to $O(1/k^2)$ by using the ``away-step'' or ``pairwise'' Frank--Wolfe variants, which also move iterates away from non-optimal vertices, reducing the zig-zag inefficiency near the optimum.

\begin{remark}[Why Standard FW Cannot Achieve $O(1/k^2)$]
  The FW ``zig-zag'' inefficiency is not merely a practical issue---it reflects a fundamental lower bound. Lacoste-Julien and Jaggi (2015) show that for any polytope $\mathcal{X}$ and smooth $f$ whose minimiser is in the interior of $\mathcal{X}$, standard FW with the canonical step schedule $\gamma_k = 2/(k+2)$ satisfies a lower bound $f(x_k) - f^\star \geq C/k$ for some problem-dependent constant $C > 0$. The zig-zag arises because FW iterates are convex combinations of \emph{at most $k$ vertices}: $x_k \in \mathrm{conv}\{z_0, z_1, \ldots, z_{k-1}\}$. When the optimal point $x^\star$ lies in the relative interior, FW must ``cycle'' through all supporting vertices, and the combination can only approximate $x^\star$ at rate $O(1/k)$ in terms of the LP vertex weights. The away-step variant (AFW) augments each step with an ``away'' direction $d_k^{\rm away} = x_k - v_k$ (where $v_k$ is the \emph{worst} active vertex for the current gradient), which removes weight from non-productive vertices and achieves $O(\rho^k)$ linear convergence for strongly convex $f$ over a polytope. The practical consequence for this assignment: if $x^\star$ is interior (as in the interior optimum case), the zig-zag of standard FW is unavoidable at the $O(1/k)$ rate; for boundary optima (where the minimiser is a vertex), FW can converge in finitely many steps.
\end{remark}

\textbf{Lower bound on FW convergence.} To see why $O(1/k)$ is tight, consider $f(x) = \|x - x^\star\|^2$ on the simplex $\Delta^d = \{x \geq 0, \mathbf{1}^T x = 1\}$ with $x^\star = \mathbf{1}/d$ (centre of simplex). The FW iterate selects $z_k = e_j$ (standard basis vector) where $j = \arg\min_j \nabla f(x_k)_j$. The distance $\|x_k - x^\star\|^2$ satisfies a recurrence that yields $\|x_k - x^\star\|^2 = O(1/k)$, confirming the bound is tight up to constants.

Adam requires more FEs than Nesterov because its effective convergence rate is slower ($O(1/k^{1.4})$ vs $O(1/k^2)$). Nesterov random search (NRS) requires the most FEs due to its $O(d/k)$ convergence rate. This chart empirically confirms the theoretical oracle hierarchy: Newton $\ll$ Nesterov $\approx$ HB $<$ Adam $<$ GD $\ll$ NRS.}
\label{fig:all_methods_oracle}
\end{figure}

\subsection{Categorised Conclusions}
\label{ssec:conclusions}

\begin{enumerate}
\item \textbf{Adaptive step-size methods (Q1).} Adagrad is the most reliable when $f^\star$ is unknown: it adapts to coordinate-wise curvature and guarantees non-increasing loss. Heavy Ball provides momentum-based acceleration at the cost of requiring tuned $(\alpha,\beta)$ parameters. The Polyak step is theoretically elegant and converges fastest when $f^\star$ is known, but is practically limited by the requirement of exact optimal value knowledge. RMSprop provides a balance between adaptivity and stability, with its exponential forgetting preventing step-size collapse. The Polyak--Lojasiewicz inequality explains why GD still converges on strongly convex problems even when the precise strong convexity constant $\mu$ is not used.

\item \textbf{Momentum and stochastic methods (Q2).} Nesterov's $O(1/k^2)$ rate is theoretically optimal (minimax) for first-order smooth convex optimisation and is empirically demonstrated on Rosenbrock, outperforming all first-order methods without curvature information. The momentum schedule $\beta_k = (k-1)/(k+2)$ is derived from the estimate sequence framework and is not ad hoc. Adam's combination of momentum and per-coordinate scaling makes it robust across problem types and the dominant choice in deep learning, though it does not achieve Nesterov's theoretical rate because its effective step is bounded away from zero (preventing $O(1/k^2)$ decay). Mini-batch SGD with the Robbins--Monro decaying schedule is provably convergent; cosine annealing schedules provide smooth variance reduction.

\item \textbf{Newton's method (Q3).} Second-order information is transformative: Newton converges in one step for quadratic objectives (Benchmark A) and in $\sim5$ steps for near-quadratic objectives (Benchmark B). The quadratic convergence rate $\|e_{k+1}\| \leq (M/m)\|e_k\|^2$ requires the Hessian to be $M$-Lipschitz and positive definite at $x^\star$. On Rosenbrock, the rapidly changing Hessian requires heavy damping ($\alpha=0.22$), reducing Newton's advantage, but even damped Newton outperforms 80 iterations of GD with only 20 steps. The degenerate case $g''(x^\star) = 0$ (e.g., $g(x) = x^4$) gives only linear convergence at rate $2/3$, demonstrating the importance of the positive-definiteness assumption.

\item \textbf{Derivative-free methods (Q4).} The optimal finite-difference step $\delta^\star \approx 1.5 \times 10^{-9}$ for $g(x) = x^4$ at $x_0 = 0.25$ minimises the trade-off between $O(\delta)$ truncation and $O(\varepsilon_{\rm mach}/\delta)$ round-off errors. The centred difference reduces truncation to $O(\delta^2)$ at cost of $2d$ evaluations. Nesterov random search is $d$-times cheaper per iteration but converges $d$-times slower, making the per-evaluation cost comparable. Nelder--Mead is effective for low-dimensional ($d \leq 10$) black-box problems. Grid search is guaranteed to find the global optimum to within grid resolution but is intractable in high dimensions.

\item \textbf{Constrained optimisation (Q5).} Projected GD is the gold standard when a cheap projection is available: it enforces feasibility exactly at every iteration, achieving the same convergence rate as unconstrained GD. Penalty methods provide a simpler implementation but require $\lambda$ tuning and only asymptotically approach feasibility (violation $= O(1/\sqrt\lambda)$ from the strong convexity argument). The augmented Lagrangian method achieves exact feasibility with fixed $\rho$ by adapting the multiplier $\mu_k$, converging at the same $O(1/k)$ dual convergence rate without the degraded step-size problem of pure penalty methods.

\item \textbf{Frank--Wolfe (Q6).} The conditional gradient algorithm avoids projections by replacing them with the computationally cheap linear minimisation oracle, exploiting the polytope structure of $\mathcal{X}$. The $O(1/k)$ convergence rate follows from the Lyapunov recursion with $A_k = O(k^2)$. It naturally handles sparsity-inducing constraint sets (e.g., $\ell_1$ or nuclear norm balls) where the LP vertex is a sparse/low-rank solution. Its $O(1/k)$ convergence rate is slower than projected gradient methods ($O(1/k^2)$ with Nesterov), but the projection-free property makes it preferable for high-dimensional structured problems (e.g., matrix completion, LASSO). The FW gap $g_k = \nabla f(x_k)^T(x_k - z_k) \geq f(x_k) - f^\star$ provides a valid, computable convergence certificate that is at most $O(T)$-tight at a given iterate but may be loose instantaneously.
\end{enumerate}

\subsection{Benchmark as a Discriminating Test}
\label{ssec:benchmark_analysis}

The Rosenbrock function effectively separates methods by their exploitation of geometric structure:
\begin{itemize}[nosep]
  \item \emph{No curvature, no $f^\star$ knowledge}: GD, $f \approx 3.5$ (120 iters)
  \item \emph{Coordinate-wise adaptivity}: Adagrad $f \approx 2.2$, Heavy Ball $f \approx 1.4$ (120 iters)
  \item \emph{Momentum/acceleration}: Nesterov $f \approx 0.48$, Newton $f \approx 0.69$ (150/20 iters)
  \item \emph{Oracle $f^\star$ knowledge}: Polyak $f \approx 0.009$ (120 iters)
\end{itemize}
This progression from $f \approx 3.5$ to $f \approx 0.009$ across six orders of magnitude illustrates how algorithmic knowledge of the problem structure (curvature, $f^\star$) directly translates to faster convergence, at the cost of additional assumptions or computation per iteration.

\subsection{Computational Cost vs.\ Accuracy}
\label{ssec:cost_analysis}

A fair comparison must account for cost per iteration:
\begin{itemize}[nosep]
  \item \textbf{Gradient descent, Polyak, Heavy Ball}: 1 gradient evaluation per iteration ($O(md)$ for linear regression, $O(d)$ for explicit-gradient problems).
  \item \textbf{Adagrad, RMSprop, Nesterov, Adam}: 1--2 gradient evaluations per iteration ($2\times$ for Nesterov due to lookahead), plus $O(d)$ arithmetic for accumulated statistics.
  \item \textbf{Newton's method}: 1 gradient + 1 Hessian + 1 linear system solve per iteration ($O(d^2)$ for Hessian, $O(d^3)$ or $O(d^2)$ for solve). For $d=2$ this is trivial, but for $d \gg 100$ the cost is prohibitive unless quasi-Newton approximations (L-BFGS) are used.
  \item \textbf{Derivative-free}: Finite differences require $d+1$ evaluations; random search requires 2. Nelder--Mead requires at most $d+1$ per iteration.
\end{itemize}
When normalised by gradient evaluation count, Newton's one-step convergence on Benchmark A (1 effective Newton step $\approx d^2$ gradient steps) represents a dramatic speedup for small $d$, but this advantage reverses for large $d$ where Hessian computation dominates.

\subsection{Limitations and Extensions}
\label{ssec:limitations}

Several limitations of this study merit acknowledgement:
\begin{enumerate}[nosep]
  \item \textbf{Low dimensionality}: All experiments use $d=2$ parameters. In high dimensions ($d \sim 10^6$, as in neural networks), second-order methods become impractical, and methods like Adam and AdaGrad become essential. The NRS oracle complexity of $O(d/\varepsilon)$ function evaluations scales poorly with $d$ (see Figure~\ref{fig:q4_oracle}), while FD GD's $d+1$ evaluations per step also become expensive.
  \item \textbf{Fixed hyperparameters}: Step sizes and momentum coefficients were tuned manually. Automatic learning rate schedules (cosine annealing, warm restarts, Section~\ref{ssec:q2_schedules}) or backtracking line search would change the relative performance. The Heavy Ball stability region (Figure~\ref{fig:q1_hb_stab}) shows how sensitive HB is to parameter choice for large $\kappa$.
  \item \textbf{Limited iterations}: 120--180 iterations may not reveal asymptotic convergence behaviour for all methods. At 500 iterations (Figure~\ref{fig:q1_rates}), Nesterov has converged further from Rosenbrock's neighbourhood on the valley than at 150 iterations, suggesting the relative ranking of Nesterov vs Newton would further improve with more iterations.
  \item \textbf{Convex vs.\ nonconvex}: Convergence guarantees cited (Nesterov $O(1/k^2)$, FW $O(1/k)$, penalty $O(1/\lambda)$) apply to convex objectives. Benchmark B is nonconvex (Hessian $\nabla^2 f = \text{diag}(2-\sin x_1, 10)$ can be near-singular for $x_1 \approx \pi/2 \approx 1.57$) and Rosenbrock is strongly nonconvex globally. All results represent local convergence behaviour starting from specific initialisations.
  \item \textbf{Stochastic vs.\ deterministic}: All gradient computations use the full gradient except in the mini-batch SGD experiments. In practice, neural network training uses stochastic gradients throughout. The SGD convergence results (Section~\ref{ssec:q2_sgd}) are limited to the well-conditioned Benchmark A; stochastic methods on Rosenbrock would show much higher variance.
  \item \textbf{No line search}: All methods use fixed step sizes. Backtracking line search (Armijo condition) or exact line search would automatically find valid step sizes and would remove the need for the heavy damping $\alpha = 0.22$ in Newton (Section~\ref{ssec:q3_trust}).
  \item \textbf{Single run}: Only a single run (fixed seed 42) is reported for deterministic methods, and 10 seeds for NRS. Bootstrap confidence intervals or multiple random seeds across all methods would give a more reliable picture of method variance.
\end{enumerate}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
