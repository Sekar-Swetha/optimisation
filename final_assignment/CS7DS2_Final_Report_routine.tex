\begin{abstract}
This report presents a systematic comparative study of optimisation algorithms applied to three
benchmark functions: a convex quadratic linear regression loss (Benchmark~A), a mildly non-convex
toy neural-network loss with sinusoidal perturbation (Benchmark~B), and the highly non-convex,
ill-conditioned Rosenbrock function (Benchmark~C).  Six families of methods are investigated over
six questions: (i)~adaptive step-size rules --- Polyak, Adagrad, RMSprop, and Heavy Ball --- each
exhibiting distinct sensitivity to curvature and knowledge of the optimal value; (ii)~momentum-based
methods --- Nesterov acceleration and Adam --- achieving the first-order optimal $O(1/k^2)$ rate
and combining momentum with adaptive scaling, respectively; (iii)~Newton's method with local
second-order approximations, achieving quadratic convergence in practice; (iv)~derivative-free
optimisation --- forward finite differences, Nesterov random search, Nelder--Mead, and grid search
--- suitable when analytic gradients are unavailable; (v)~constrained optimisation via projected
gradient descent and quadratic penalty methods; and (vi)~the Frank--Wolfe conditional-gradient
algorithm for linearly-constrained problems.  Experimental results confirm theoretical predictions:
methods with curvature information (Newton, Adam) substantially outperform constant step-size
gradient descent on ill-conditioned problems; Nesterov acceleration achieves $f = 0.479$ on
Rosenbrock compared to $f = 3.388$ for gradient descent at 150 iterations; and the Polyak step-size
rule is optimal only when the exact minimum value $f^\star$ is known.  All code and figures are
fully reproducible from the supplied Python file.
\end{abstract}

\subsection{Benchmark A: Convex Quadratic Loss (Linear Regression)}

A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
  \theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \sim \mathcal{N}(0, I_2), \quad
  y^{(i)} = {\theta^{\star}}^{\!\top} x^{(i)} + \varepsilon^{(i)},
  \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The empirical mean-squared-error loss is:
\begin{equation}
  J(\theta) = \frac{1}{2m}\sum_{i=1}^{m}\!\bigl(\theta^{\top}x^{(i)} - y^{(i)}\bigr)^2
            = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strongly convex quadratic with \textbf{constant} Hessian $H = \frac{1}{m}X^{\top}X \succ 0$.
The gradient is $\nabla J(\theta) = \frac{1}{m}X^{\top}(X\theta - y)$.  Due to observation noise the
minimum loss is $J^{\star} \approx 0.4826 > 0$; the exact minimiser (OLS solution) is
$\theta^{\star}_{\mathrm{OLS}} = (X^{\top}X)^{-1}X^{\top}y \approx (3.028, 4.017)$.  Starting
point: $\theta_0 = [0, 0]^{\top}$.

\subsection{Benchmark B: Non-Convex Toy Loss with Sinusoidal Perturbation}
\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
  \label{eq:bench_B}
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\; 10(x_2 - 2)]^{\top}$ and the Hessian is
$H(x) = \operatorname{diag}(2 - \sin(x_1),\, 10)$.  The non-convex term $\sin(x_1)$ perturbs the
otherwise strongly convex quadratic.  The curvature ratio in $x_2$ to $x_1$ is approximately $10$
to $2$, creating moderate anisotropy (condition number $\kappa \approx 5$).  The unique global
minimum satisfies $2(x_1 - 1) + \cos(x_1) = 0$ and $x_2 = 2$; solving numerically gives
$x_1^{\star} \approx 0.5824$, so $x^{\star} \approx (0.5824, 2.0)$ with
$f^{\star} \approx 0.7244$.  Starting point: $x_0 = [-1, 4]^{\top}$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
  f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
  \label{eq:rosenbrock}
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$.  The Hessian at the optimum is
\begin{equation}
  H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},
\end{equation}
with eigenvalues approximately $2$ and $1000$, giving condition number $\kappa \approx 500$.  This
severe ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes the
Rosenbrock function the most challenging of the three benchmarks.  Starting point:
$x_0 = [-1, 1]^{\top}$.

\subsection{Benchmark D: Box-Constrained Feasible Set}
For Questions~5--6: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive methods are compared against constant step-size gradient descent (GD) as a baseline.
All methods run for $N = 120$ iterations.

\medskip
\noindent\textbf{Gradient Descent (Baseline).}  The update is $x_{k+1} = x_k - \alpha \nabla f(x_k)$
with a fixed learning rate $\alpha > 0$.  For a $\mu$-strongly convex, $L$-smooth objective this
achieves linear convergence:
$f(x_k) - f^\star \leq \left(\frac{\kappa - 1}{\kappa + 1}\right)^{2k}(f(x_0) - f^\star)$,
where $\kappa = L/\mu$.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \Comment{Oracle step requires exact $f^\star$}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent When $f^\star$ is known exactly and $f$ is convex, the Polyak step satisfies
$f(x_{k+1}) - f^\star \leq \tfrac{1}{2}\|x_k - x^\star\|^2 \cdot (1 - 1/(2\|x_k - x^\star\|^2))$
--- it is optimal for the class of convex functions satisfying the \emph{Polyak--\L ojasiewicz}
(PL) condition.  When $f^\star$ is misspecified, the numerator $f(x_k) - f^\star$ is permanently
offset, causing $\alpha_k \to \infty$ as $\nabla f(x_k) \to 0$, inducing instability.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Coordinate-wise sum of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent Adagrad scales the learning rate for coordinate $i$ by $1/\sqrt{\sum_{t=0}^{k} g_{t,i}^2}$.
Coordinates that receive large gradients are penalised with a small step, while infrequently-updated
coordinates receive a large step.  However, since $G_k$ accumulates and never decreases,
the effective learning rate decays monotonically to zero, which can prematurely stall convergence
on non-stationary objectives.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent RMSprop addresses Adagrad's monotonic decay by replacing the full history sum with an
exponential moving average controlled by $\beta$.  This gives the effective step size a ``memory''
of roughly $1/(1-\beta)$ recent iterations, adapting to local curvature rather than global history.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \Comment{Accumulated gradient with momentum}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent For a quadratic $f$ with Hessian eigenvalues $\mu \leq \lambda_i \leq L$, the optimal
Heavy Ball parameters are $\alpha^* = 4/(\sqrt{L}+\sqrt{\mu})^2$ and
$\beta^* = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2$, achieving the linear rate
$((\sqrt{\kappa}-1)/(\sqrt{\kappa}+1))^{2k}$, which is \emph{asymptotically optimal} for
first-order methods on strongly convex quadratics.

\textbf{Key findings.}
\begin{itemize}[nosep]
\item \emph{Polyak sensitivity:}  The Polyak step is highly sensitive to $f^{\star}$.  When
  $f^{\star}$ is specified as~0 but the true minimum is positive (Benchmarks~A and~B), the step
  size grows unboundedly as $\|\nabla f\| \to 0$, preventing convergence to the true minimiser.
  On Benchmark~C where $f^{\star} = 0$ is correct, Polyak delivers the best result among all five
  methods.
\item \emph{Adagrad vs RMSprop:}  Adagrad's cumulative sum of squared gradients causes monotonic
  step-size decay, which is beneficial for ill-conditioned problems (Benchmark~C: $f = 2.203$) but
  slightly suboptimal where a constant effective rate is preferable.  RMSprop's exponential
  forgetting ($\beta = 0.9$) stabilises the step but can introduce oscillation near the minimum.
\item \emph{Heavy Ball advantage:}  The momentum term $\beta z_k$ accumulates gradient information
  along persistent descent directions, effectively increasing the step along eigenspaces with
  consistently aligned gradients.  This provides the most benefit on ill-conditioned problems
  (Benchmark~C) where GD must take tiny steps for stability.
\end{itemize}

\subsection{Comparative Summary}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is exact, Polyak converges fastest
  (Rosenbrock: $f = 0.024$).  Without exact $f^{\star}$, Heavy Ball and Adagrad are the
  best-performing adaptive alternatives.
\item \textbf{Robustness:} GD baseline is most predictable.  Adagrad is uniformly stable due to
  its non-increasing step, though this can cause premature stalling on non-stationary or
  large-dimension problems.  RMSprop is more robust than Adagrad on long horizons.
\item \textbf{Practical recommendation:} RMSprop or Heavy Ball for problems where $f^{\star}$
  is unknown.  Polyak is ideal for problems with known minimum value (e.g.\ feasibility problems,
  or when lower-bound estimates are available).
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\; \beta_{\max}\right)$
    \Comment{Momentum schedule increases towards $\beta_{\max}$}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$
    \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Acceleration]\label{thm:nesterov}
For a convex $L$-smooth objective with step size $\alpha \leq 1/L$, the Nesterov accelerated
gradient method achieves:
\begin{equation}
  f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2},
\end{equation}
which is optimal ($O(1/k^2)$) for the class of convex Lipschitz-smooth functions, compared to
$O(1/k)$ for gradient descent.
\end{theorem}

The critical distinction from Heavy Ball is the \emph{lookahead evaluation}: the gradient is
computed at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$.  This
allows Nesterov to \emph{anticipate} overshoot and correct its trajectory, delivering the optimal
convergence rate.  The schedule $\beta_k = (k-1)/(k+2)$ is the original choice from Nesterov
(1983), capped at $\beta_{\max}$ for non-convex problems.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, $\beta_1 \in [0,1)$, $\beta_2 \in [0,1)$,
         $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\; v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$
    \Comment{First moment (EMA of gradient)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$
    \Comment{Second moment (EMA of squared gradient)}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$
    \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t / (1 - \beta_2^t)$
    \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent Adam (Kingma \& Ba, 2015) unifies momentum ($m_t \approx$ running mean of $g_t$) with
per-coordinate adaptive scaling ($1/\sqrt{\hat{v}_t}$).  The bias-correction terms
$1 - \beta_j^t$ prevent the initial estimates from being biased towards zero during
warm-up.  Note that Adam is a \emph{non-monotone} method --- $f(x_{t+1}) > f(x_t)$ is possible
--- and convergence guarantees require additional assumptions (e.g.\ diminishing learning rate or
bounded gradient variance).

\subsection{Nesterov vs Heavy Ball: Theoretical Comparison}

Comparing Q1's Heavy Ball (Benchmark~C: $f = 1.400$) with Q2's Nesterov ($f = 0.479$) reveals a
qualitative difference.  Both methods use a momentum term, but:
\begin{itemize}[nosep]
\item \textbf{Heavy Ball} evaluates the gradient at the \emph{current} point $x_k$ and adds
  momentum: $x_{k+1} = x_k - \alpha g_k + \beta(x_k - x_{k-1})$.  On strongly convex quadratics
  it achieves the optimal linear rate but may oscillate on non-convex problems.
\item \textbf{Nesterov} evaluates the gradient at the \emph{extrapolated} point
  $\tilde{x}_k = x_k + \beta_k z_k$, obtaining a corrective signal that prevents overshooting.
  This modification upgrades the convergence rate from $O(1/k)$ to $O(1/k^2)$ on convex problems.
\end{itemize}
The practical advantage on Rosenbrock ($1.400 \to 0.479$) confirms the theoretical superiority
of the lookahead gradient evaluation on non-quadratic landscapes.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$

        \Comment{Unbiased gradient estimate: $\mathbb{E}[g_j] = \nabla J(\theta)$}
        \State $\theta \gets \theta - \alpha\, g_j$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{First- and Second-Order Local Approximations of $g(x) = x^4$}

Consider $g(x) = x^4$ with $g'(x) = 4x^3$ and $g''(x) = 12x^2$.  At $x_0 = 0.25$:
$g(x_0) = 0.25^4 = 0.003906$, $g'(x_0) = 4 \cdot 0.25^3 = 0.0625$,
$g''(x_0) = 12 \cdot 0.25^2 = 0.75$.

\textbf{First-order (tangent-line) approximation:}
\begin{equation}
  \hat{g}_1(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.003906 + 0.0625\,(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
  \hat{g}_2(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
             = 0.003906 + 0.0625\,(x - 0.25) + 0.375\,(x - 0.25)^2.
\end{equation}

The Newton update steps to the minimum of $\hat{g}_2$:
\begin{equation}
  x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)}.
\end{equation}
This is equivalent to minimising the local quadratic model.  Since $g(x) = x^4$ has $g''(0) = 0$
the Newton direction is undefined at the minimum, which motivates the Hessian regularisation
$H \leftarrow H + \lambda I$ used in practice.

\subsection{Newton's Method with Hessian Regularisation}

\begin{algorithm}[H]
\caption{Newton's Method with Tikhonov Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$
    \Comment{Regularise to ensure positive definiteness}
    \State $p_k \gets H_k^{-1}\, g_k$
    \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Quadratic Convergence of Newton's Method]\label{thm:newton}
Near a non-degenerate minimiser $x^\star$ (where $\nabla^2 f(x^\star) \succ 0$ and $\alpha = 1$),
Newton's method converges \emph{quadratically}:
\begin{equation}
  \|x_{k+1} - x^\star\| \leq C \|x_k - x^\star\|^2
\end{equation}
for some constant $C > 0$ depending on the local Hessian Lipschitz constant.  This implies that
once the iterate is within a ball of radius $1/C$ around $x^\star$, the number of correct decimal
digits \emph{doubles} at every iteration.
\end{theorem}

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}
% ============================================================

\subsection{Part A: Finite Differences and Nesterov Random Search (Benchmark B)}

\subsubsection{Forward Finite-Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector.  The approximation error has two competing
components:
\begin{itemize}[nosep]
  \item \emph{Truncation error} (from Taylor expansion): $O(\delta)$ --- larger $\delta$ means
    larger bias.
  \item \emph{Floating-point cancellation error}: $O(\varepsilon_{\mathrm{mach}}/\delta)$ ---
    smaller $\delta$ amplifies cancellation in $f(x+\delta e_i) - f(x)$.
\end{itemize}
The optimal $\delta$ balancing these is $\delta^* \approx \sqrt{\varepsilon_{\mathrm{mach}}}
\approx 10^{-8}$ for double precision.  The experiment contrasts $\delta = 0.05$ (near-optimal
for this problem) and $\delta = 0.8$ (severely truncation-dominated).

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \Comment{Random-direction directional derivative}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent This zeroth-order method estimates a directional derivative along a uniformly random
unit vector, requiring only \emph{two} function evaluations per iteration regardless of dimension
$d$.  By contrast, FD gradient estimation costs $d+1$ evaluations.  The estimator satisfies
$\mathbb{E}[\hat{g}_k] = \nabla f(x_k)$ (in expectation over $u_k$), so the method is an unbiased
stochastic gradient method, but with significantly higher variance than exact gradient descent.

\subsection{Part B: Derivative-Free Methods on Benchmark C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and
iteratively transforms it through four operations:
\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r = 1$): reflect the worst vertex through the centroid of
    the remaining vertices.
  \item \textbf{Expansion} ($\gamma = 2$): if reflection improves the worst, try extending
    further in the same direction.
  \item \textbf{Contraction} ($\rho = 0.5$): if reflection fails to improve, contract towards
    the centroid.
  \item \textbf{Shrink} ($\sigma = 0.5$): if contraction also fails, shrink all vertices towards
    the best.
\end{enumerate}
No gradient or function-derivative information is required --- only function evaluations at
vertices ($\leq 3$ per iteration).  Settings: initial step $= 0.35$, iterations $= 160$.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Setup and Optimality}

Minimise Benchmark~B subject to the constraint $x_1 \geq 0.5$, starting from the
\emph{infeasible} point $x_0 = (0.2, 4.0)$.

\begin{remark}[Constraint Activity]\label{rem:inactive}
  The unconstrained minimum of Benchmark~B is at $x^{\star} \approx (0.582, 2.0)$ with
  $x_1^{\star} \approx 0.582 > 0.5$.  Therefore the constraint $x_1 \geq 0.5$ is \emph{inactive}
  at the global minimiser: $x^{\star}$ is already feasible.  This means the constrained and
  unconstrained solutions coincide, and all correctly-converging methods should reach
  $f \approx 0.7244$.  The interest of this experiment lies not in finding a different optimal
  point, but in observing how quickly and reliably each method achieves feasibility and
  convergence from the infeasible start.
\end{remark}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$
    \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1})$
    \Comment{Project onto $\mathcal{C} = \{x_1 \geq 0.5\}$: $x_1 \leftarrow \max(0.5, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent For the constraint $x_1 \geq 0.5$, the projection is the simple clipping
$\Pi_{\mathcal{C}}(z)_1 = \max(0.5, z_1)$, $\Pi_{\mathcal{C}}(z)_2 = z_2$.  The key
theoretical property is that projection is \emph{non-expansive}:
$\|\Pi_{\mathcal{C}}(z) - x^\star\| \leq \|z - x^\star\|$ for any $x^\star \in \mathcal{C}$,
which is essential for convergence proofs.

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_\lambda(x) \gets f(x) + \lambda \max(0,\; -x_1 + 0.5)$
    \Comment{$\ell_1$ penalty for constraint $x_1 \geq 0.5$}
    \State $\nabla F_\lambda(x) \gets \nabla f(x) - \lambda\,\mathbf{1}[x_1 < 0.5]\, e_1$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$, 100 iterations), Penalty with
$\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for
$\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$).

\subsection{Theoretical Discussion}

\textbf{Projected GD} enforces feasibility as a hard constraint at every iteration.  The
projection oracle is trivial ($O(1)$) for simple constraints like $x_1 \geq 0.5$, but can be
expensive ($O(n)$ to $O(n^3)$) for complex constraint sets such as semidefinite cones or general
polytopes.

\textbf{Penalty methods} trade exact feasibility for simplicity: only a modified gradient is
needed.  The weight $\lambda$ controls the trade-off.  As $\lambda \to \infty$, the penalised
minimiser converges to the constrained minimiser, but large $\lambda$ increases the condition
number of $\nabla^2 F_\lambda$, requiring smaller step sizes for stability.  This is reflected
in the step-size reduction from $\alpha = 0.05$ ($\lambda = 1.8$) to $\alpha = 0.03$ ($\lambda = 4.5$).

A practical remedy is the \emph{augmented Lagrangian} (method of multipliers), which adds a
dual-variable update to drive exact feasibility without requiring $\lambda \to \infty$.

\subsection{Part I: Linear Programme over the Box $\mathcal{X}$}

\textbf{Problem:} Minimise $f(x) = a^{\top}x$ with $a = [1, 2]^{\top}$ over
$\mathcal{X} = \{x : 0.5 \leq x_1 \leq 5,\, -5 \leq x_2 \leq 10\}$.

\begin{proposition}[LP Optimality over a Box]
For a linear objective $f(x) = a^{\top}x$ over a box $\{l \leq x \leq u\}$, the minimiser is:
\begin{equation}
  x_i^{\star} = \begin{cases} l_i & \text{if } a_i > 0 \\ u_i & \text{if } a_i < 0 \end{cases}
\end{equation}
This follows from the \emph{Fundamental Theorem of Linear Programming}: the optimum of a linear
objective over a bounded polytope is attained at a vertex.
\end{proposition}

Since $a_1 = 1 > 0$ and $a_2 = 2 > 0$, both coordinates are minimised at their lower bounds:
\begin{equation}
  x^{\star} = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}

\subsection{Frank--Wolfe Algorithm (Conditional Gradient Method)}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^{\top} x$
    \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$
    \Comment{Convex combination: $x_{k+1} \in \mathcal{X}$ by convexity}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent\textbf{Key properties:}
\begin{itemize}[nosep]
\item \emph{Feasibility maintenance:} Since $x_k, z_k \in \mathcal{X}$ (convex set) and
  $x_{k+1}$ is a convex combination, $x_{k+1} \in \mathcal{X}$ automatically.  No projection
  is needed.
\item \emph{LMO for box constraints:} The linear subproblem decouples per coordinate:
  $(z_k)_i = l_i$ if $(\nabla f(x_k))_i > 0$, else $u_i$.  Cost: $O(d)$.
\item \emph{Convergence rate:} For a convex $L$-smooth objective with the standard
  $\gamma_k = 2/(k+2)$ step-size schedule, Frank--Wolfe converges at $O(1/k)$:
  $f(x_k) - f^\star \leq 4LD^2/(k+2)$, where $D = \max_{x,y \in \mathcal{X}}\|x-y\|$.
\item \emph{Limitation:} With fixed $\beta$, the standard $O(1/k)$ convergence rate is not
  guaranteed; an adaptive $\gamma_k = 2/(k+2)$ schedule is preferable in practice.
\end{itemize}

\subsection{Interior vs Boundary Behaviour: Theoretical Discussion}

\textbf{Interior optimum.}  When $x^\star$ is in the interior of $\mathcal{X}$, the LP vertex
$z_k = \arg\min_{z \in \mathcal{X}} g^{\top}z$ always lies at an extreme point of $\mathcal{X}$,
far from $x^\star$.  The Frank--Wolfe iterate is a weighted average of $x_k$ (near $x^\star$)
and $z_k$ (far from $x^\star$), causing a characteristic zig-zag: each step pulls $x_{k+1}$
slightly away from $x^\star$ in the direction of some extreme point.  This produces an $O(1/k)$
convergence rate, which cannot be improved in general without additional problem structure.

\textbf{Boundary optimum.}  When $x^\star$ lies on the boundary, the LP vertex $z_k$ frequently
coincides with (or is near) $x^\star$.  This is the setting where Frank--Wolfe excels over
projected gradient descent: the LP subproblem implicitly performs the projection onto the active
constraint, without requiring an explicit projection oracle.  This makes Frank--Wolfe attractive
for structured feasible sets (e.g., nuclear-norm balls, spectrahedra) where projection is
expensive but LP is tractable.

\subsection{Key Findings and Theoretical Connections}

\textbf{1.~Step-size adaptivity is crucial for ill-conditioned problems.}
The Rosenbrock condition number $\kappa \approx 500$ means that the GD convergence factor
$(\kappa-1)/(\kappa+1) \approx 0.996$ per iteration, requiring $\sim\!1700$ iterations to reduce
the error by a factor of $e^{-3}$.  Adaptive methods (Heavy Ball, Nesterov, Polyak) break
this barrier by exploiting problem geometry.

\textbf{2.~Knowledge of $f^\star$ versus curvature.}
The Polyak step uses knowledge of $f^\star$ to set the step size; Adagrad/RMSprop use gradient
history to estimate curvature; Heavy Ball/Nesterov use momentum to exploit gradient consistency;
Newton uses the exact Hessian.  The empirical ordering on Rosenbrock
(Newton $\sim$ Polyak $\gg$ Nesterov $>$ Heavy Ball $>$ Adagrad $>$ GD) reflects the hierarchy:
exact second-order or oracle information $\gg$ accelerated first-order $\gg$ adaptive first-order
$\gg$ standard gradient descent.

\textbf{3.~Derivative-free methods are competitive for smooth, low-dimensional problems.}
On Benchmark~B, FD gradient descent closely tracks exact GD given a good perturbation
($\delta = 0.05$), demonstrating that gradient approximation is often viable.  However, on
Benchmark~C (ill-conditioned), even small approximation errors degrade the gradient direction
significantly, making derivative-free methods much less competitive.

\textbf{4.~Frank--Wolfe is projection-free but slower per iteration.}
Compared to projected gradient descent (Q5), Frank--Wolfe (Q6) requires no projection but
achieves an $O(1/k)$ rate rather than the faster $O(((\kappa-1)/(\kappa+1))^k)$ linear rate of
projected GD on strongly convex problems.  Frank--Wolfe is most competitive when projection is
expensive (e.g., nuclear-norm constraints in matrix completion).

\textbf{5.~Mini-batch size controls the variance--bias trade-off in SGD.}
Smaller batches ($b = 5$) provide more updates per epoch but higher gradient variance.  Larger
batches ($b = 40$) provide smoother estimates but fewer updates.  Under high observation noise
($\sigma = 6$), the variance from the data dominates the mini-batch variance, equalising the
relative performance of the two batch sizes.

\subsection{Practical Recommendations}

\begin{itemize}[nosep]
\item \textbf{Smooth convex or mildly non-convex, unknown $f^\star$:} Use Adam or Nesterov with
  tuned hyperparameters.  Nesterov is theoretically optimal ($O(1/k^2)$) for convex problems.
\item \textbf{Strongly convex quadratics:} Newton's method in one step (exact) or Heavy Ball
  with optimal tuning.
\item \textbf{No gradient available:} Finite differences (small $\delta$) for low-to-moderate
  $d$; Nesterov random search for high $d$.
\item \textbf{Constrained, projection is cheap:} Projected gradient descent for hard constraint
  satisfaction.
\item \textbf{Constrained, LP is cheap:} Frank--Wolfe for automatic feasibility without
  projection.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball (Polyak Momentum)}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z      # extrapolated point
        g = grad_fn(lookahead)          # gradient at lookahead
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)     # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Hessian Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (Conditional Gradient)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
