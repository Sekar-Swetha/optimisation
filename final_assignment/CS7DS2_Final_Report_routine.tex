\begin{abstract}
This report presents a systematic comparative study of ten optimisation algorithms across six method families, applied to three benchmark functions of increasing difficulty: a convex quadratic (linear regression), a perturbed quadratic (toy neural network), and the highly ill-conditioned Rosenbrock function.
The families investigated are: (1) adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball); (2) momentum and stochastic methods (Nesterov accelerated gradient, Adam, mini-batch SGD); (3) Newton's method with Hessian regularisation; (4) derivative-free methods (finite differences, Nesterov random search, Nelder--Mead, grid search); (5) constrained optimisation via projected gradient descent and penalty methods; and (6) the Frank--Wolfe algorithm for box-constrained problems.

Key findings: Polyak step sizes diverge on benchmarks where $f^{\star}$ is misspecified but achieve the fastest convergence on Rosenbrock where the true minimum is zero. Newton's method converges in a single step on the quadratic benchmark, demonstrating exact quadratic convergence. Nesterov acceleration consistently outperforms standard gradient descent and Adam on Rosenbrock ($f = 0.479$ vs $f = 3.388$ for GD, 150 iterations). Projected gradient descent enforces hard constraints exactly from the first iteration, while penalty methods require large penalty weights ($\lambda \geq 4.5$) to approach feasibility within 100 iterations. Frank--Wolfe converges to both interior and boundary optima without explicit projection, exhibiting the characteristic $O(1/k)$ convergence slowdown near the minimum.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmark functions are chosen to span a range of structural difficulties relevant to practical optimisation.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
  \theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \in \mathbb{R}^2,\quad
  y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad
  \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
  J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2
            = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strongly convex quadratic with constant Hessian $H = \frac{1}{m}X^T X \succ 0$ and gradient $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$.  The smoothness constant (largest eigenvalue of $H$) and strong convexity modulus (smallest eigenvalue) govern the convergence of all gradient methods.  Due to observation noise, the noise floor is $J^{\star} \approx 0.4826$ (the minimum is attained at the OLS solution $\hat{\theta} = (X^T X)^{-1} X^T y$, not at $\theta^{\star}$).  Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Quadratic Loss}
\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
  \label{eq:benchB}
\end{equation}
The gradient and Hessian are:
\begin{equation}
  \nabla f = \begin{bmatrix} 2(x_1 - 1) + \cos(x_1) \\ 10(x_2 - 2) \end{bmatrix}, \qquad
  H(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The diagonal Hessian implies that the two coordinates are decoupled in curvature.  The eigenvalue ratio $H_{11}/H_{22}$ varies between approximately $(2-1)/10 = 0.1$ and $(2+1)/10 = 0.3$, giving a condition number of roughly $3$--$10$ and thus moderate anisotropy.  The minimum satisfies $2(x_1^{\star} - 1) + \cos(x_1^{\star}) = 0$, yielding $x_1^{\star} \approx 0.582$; the second coordinate satisfies $x_2^{\star} = 2.0$ exactly.  The minimum value is $f^{\star} \approx 0.7244$.  Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
  f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
  \label{eq:rosenbrock}
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$.  The Hessian at the optimum $(1,1)$ is:
\begin{equation}
  H(1,1) = \begin{bmatrix} 2 + 1200 - 400 & -400 \\ -400 & 200 \end{bmatrix}
           = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}.
\end{equation}
Its eigenvalues are $\lambda_{\max} \approx 1001.6$ and $\lambda_{\min} \approx 0.40$, giving a condition number $\kappa \approx 2504$.  This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the canonical challenge functions for benchmarking optimisation algorithms.  Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}
For the constrained problems in Questions 5 and 6, the feasible set is:
\begin{equation}
  \mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a box (axis-aligned polytope), which admits closed-form projection and enables an efficient Frank--Wolfe linear minimisation oracle.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

Gradient descent with a fixed step size $\alpha$ satisfies:
\begin{equation}
  x_{k+1} = x_k - \alpha\, \nabla f(x_k).
\end{equation}
For an $L$-smooth, $\mu$-strongly convex function, the optimal constant step size is $\alpha^{\star} = 2/(L + \mu)$ and the convergence rate is geometric:
\begin{equation}
  f(x_k) - f^{\star} \leq \left(\frac{\kappa - 1}{\kappa + 1}\right)^{2k} \bigl(f(x_0) - f^{\star}\bigr), \quad \kappa = L/\mu.
\end{equation}
Adaptive methods aim to overcome the dependence on knowing $L$ and $\mu$ by computing step sizes that respond to the local geometry.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Polyak step size}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
For a convex function with known $f^{\star}$, the Polyak step size minimises the distance to the solution in one step under a quadratic upper bound.  It achieves $O(1/k)$ convergence for convex functions and linear convergence for strongly convex functions — without requiring knowledge of the Lipschitz constant $L$.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
The effective step size in coordinate $i$ is $\alpha_0/\sqrt{\sum_{j=0}^{k} g_{j,i}^2}$, which decays monotonically.  Coordinates with historically large gradients receive small steps, providing automatic adaptation to anisotropic loss landscapes.  The monotonic decay is a double-edged feature: it prevents oscillation but may cause premature convergence on non-stationary or long-horizon problems.
\end{remark}

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay factor $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
RMSprop fixes Adagrad's vanishing-step-size problem by using an exponential moving average (EMA) with decay factor $\beta = 0.9$.  The effective memory horizon is approximately $1/(1-\beta) = 10$ iterations, so the effective step size adapts to recent curvature rather than the full history.
\end{remark}

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum coefficient $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Accumulate momentum}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
The Heavy Ball method is a discrete-time analogue of a damped harmonic oscillator.  For a quadratic $f$ with condition number $\kappa$, the optimal parameters $\alpha^{\star} = \bigl(\frac{2}{{\sqrt{L}+\sqrt{\mu}}}\bigr)^2$ and $\beta^{\star} = \bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\bigr)^2$ yield convergence in $O(\sqrt{\kappa})$ iterations, compared to $O(\kappa)$ for GD.  However, these theoretical guarantees do not extend to general non-quadratic functions, where Heavy Ball can exhibit oscillatory or even divergent behaviour.
\end{remark}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} On Rosenbrock where $f^{\star}$ is known exactly, Polyak converges fastest ($f = 0.009$ in 120 iterations). On well-conditioned benchmarks, Adagrad, RMSprop, and Heavy Ball all match GD.
\item \textbf{Stability on ill-conditioned problems:} Heavy Ball ($\beta = 0.86$) outperforms GD on Rosenbrock by a factor of $\sim 2.5\times$ in final objective value, demonstrating the benefit of momentum accumulation along the principal curvature direction.
\item \textbf{Adagrad vs RMSprop:} Adagrad's monotonically decaying step is beneficial early in training (when the trajectory is far from the minimum and large steps would overshoot) but limits late-stage convergence. RMSprop's EMA preserves adaptivity but does not decay to zero, explaining its slightly worse final value on Benchmark A.
\item \textbf{Practical recommendation:} When $f^{\star}$ is unknown (the typical case), Heavy Ball with well-chosen $(\alpha, \beta)$ or RMSprop with a small base rate are preferable to Polyak. Adagrad is a safe default for sparse or varying-gradient problems.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Theoretical Background}

Standard gradient descent on an $L$-smooth convex function achieves an $O(1/k)$ convergence rate for the suboptimality gap. Nesterov's accelerated gradient method improves this to $O(1/k^2)$, which is optimal for first-order methods in the sense of Nesterov's lower bound complexity theorem. Adam combines first- and second-moment estimates to construct a gradient direction that is adaptively normalised per coordinate.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, maximum momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Momentum update}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Acceleration, informal]
For a convex, $L$-smooth function $f$, the standard Nesterov scheme with $\alpha = 1/L$ and momentum weights $t_k = (k+1)/2$ satisfies:
\begin{equation}
  f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2},
\end{equation}
improving the $O(1/k)$ rate of gradient descent to $O(1/k^2)$.
\end{theorem}

The critical difference from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This allows the method to anticipate the effect of momentum and correct its trajectory before overshooting, which is the mechanism that enables the acceleration guarantee.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moment parameters $\beta_1, \beta_2 \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{Exponential first moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Exponential second moment (adaptive scaling)}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$ \Comment{Bias correction for first moment}
    \State $\hat{v}_t \gets v_t / (1 - \beta_2^t)$ \Comment{Bias correction for second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
The bias correction terms in Adam ensure that at $t = 1$, the effective step is approximately $\alpha$, not $\alpha(1-\beta_1)$ (which would be $0.2\alpha$ for $\beta_1 = 0.8$). Without bias correction, the first few iterates would be dominated by the zero initialisation of the moment vectors, leading to extremely small early steps and slow initial convergence.
\end{remark}

Adam combines the trajectory smoothing of momentum ($m_t$ approximates a low-pass filtered version of $g_t$) with the per-coordinate scaling of RMSprop ($v_t$ tracks recent squared gradient magnitude). The effect is a method that is both direction-stabilised and curvature-adapted, inheriting advantages from both families.

\subsection{Mini-Batch Stochastic Gradient Descent}
\label{sec:sgd}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch. Each epoch consists of $m/b$ parameter updates.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$ into $\pi$
    \For{each mini-batch $\mathcal{B} = \{\pi_{jb+1}, \ldots, \pi_{(j+1)b}\}$}
        \State $\hat{g} \gets \dfrac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The stochastic gradient $\hat{g}$ is an unbiased estimator of the full gradient $\nabla J(\theta)$, but with variance $\text{Var}[\hat{g}] \propto \sigma^2/b$ (where $\sigma^2$ is the noise variance) that decreases with batch size.

\subsection{Local Approximations of $g(x) = x^4$}

At a point $x_0$, the Taylor expansion of $g$ to second order is:
\begin{equation}
  g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2.
\end{equation}
For $g(x) = x^4$ at $x_0 = 0.25$: $g(0.25) = 0.00391$, $g'(0.25) = 4(0.25)^3 = 0.0625$, $g''(0.25) = 12(0.25)^2 = 0.75$.

\textbf{First-order approximation:}
\begin{equation}
  g(x) \approx 0.00391 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order approximation:}
\begin{equation}
  g(x) \approx 0.00391 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton update for $g$ minimises the second-order approximation:
\begin{equation}
  x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3}.
\end{equation}
Starting from $x_0 = 0.25$: $x_1 = 0.25 - 0.0833 = 0.1667$, $x_2 = 0.1111$, $x_3 = 0.0741,\ldots$ converging to $x^{\star} = 0$ at a quadratic rate.

\begin{figure}[H]
\centering
\includegraphics[width=0.85\textwidth]{q3_approximation.pdf}
\caption{Q3: Local approximations of $g(x) = x^4$ at $x_0 = 0.25$. The first-order approximation (tangent line) captures only the local slope and deviates rapidly. The second-order approximation fits the curvature accurately within a neighbourhood of $x_0$. The minimum of the quadratic model at $x \approx 0.167$ provides a better estimate of the true minimum ($x^{\star} = 0$) than the gradient step, motivating Newton's method as a second-order refinement over gradient descent.}
\label{fig:q3_approx}
\end{figure}

\subsection{Newton's Method with Hessian Regularisation}

\begin{algorithm}[H]
\caption{Newton's Method with Damping and Tikhonov Regularisation}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised Hessian (ensures positive definiteness)}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Local Quadratic Convergence of Newton's Method]
Let $f$ be twice continuously differentiable and let $x^{\star}$ be a local minimiser with $\nabla^2 f(x^{\star}) \succ 0$.  Assume the Hessian is Lipschitz continuous: $\|\nabla^2 f(x) - \nabla^2 f(y)\| \leq M\|x - y\|$.  Then for the undamped Newton method ($\alpha = 1$) initialised sufficiently close to $x^{\star}$:
\begin{equation}
  \|x_{k+1} - x^{\star}\| \leq \frac{M}{2\mu}\|x_k - x^{\star}\|^2,
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^{\star}))$.  This is the \emph{quadratic convergence} property: the error is squared at each step.
\end{theorem}

Damping ($\alpha < 1$) is necessary when the current iterate is far from $x^{\star}$, where the Hessian approximation may be inaccurate and undamped Newton steps can overshoot.  On Benchmark C, the Rosenbrock Hessian changes rapidly along the valley, necessitating $\alpha = 0.22$ to avoid divergence.

\textbf{Settings:} GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$.  The step sizes are listed in Table~\ref{tab:q3_params}.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}
% ============================================================

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of each partial derivative is:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta},
\end{equation}
where $e_i$ is the $i$-th standard basis vector.  This approximation has two competing error sources: the truncation error $O(\delta)$ from the first-order Taylor expansion, and the round-off amplification error $O(\varepsilon_{\text{mach}}/\delta)$ from floating-point subtraction of nearly equal quantities.  The optimal $\delta$ balances these:
\begin{equation}
  \delta^{\star} \sim \sqrt{\varepsilon_{\text{mach}}} \approx \sqrt{10^{-16}} = 10^{-8} \quad (\text{double precision}).
\end{equation}
The assignment tests $\delta = 0.05$ (small, good approximation) and $\delta = 0.8$ (large, poor approximation because the Taylor truncation error dominates and the step spans a region of significant curvature change).  Note that FD GD uses $\alpha = 0.08$ in this experiment.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order Gradient Estimation)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \text{Uniform}(\mathcal{S}^{d-1})$ (uniformly on the unit sphere)
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Stochastic directional gradient}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{proposition}
The random search gradient estimate satisfies:
\begin{equation}
  \mathbb{E}_{u}[\hat{g}_k] \approx \frac{d}{\delta^2} \int \nabla f(x_k)^T u\, u \, \mathrm{d}u = \frac{1}{d}\sum_i \frac{\partial f}{\partial x_i} \approx \frac{1}{d}\nabla f(x_k),
\end{equation}
i.e., it is a biased (by factor $1/d$ in 2D) estimator of the gradient.  More precisely, $\mathbb{E}[\hat{g}_k] = \frac{1}{d}\nabla f(x_k)$ when $d > 1$, which is compensated in practice by the choice of $\alpha$.
\end{proposition}

The key practical advantage of Nesterov random search is that it requires only \textbf{two function evaluations per iteration regardless of dimension} $d$ --- compare with the $d+1$ evaluations required by forward FD to compute the full gradient. This makes it particularly attractive in high dimensions where the function is expensive to evaluate.

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n + 1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it via four operations:
\begin{enumerate}[nosep]
  \item \textbf{Reflection} ($\alpha_r = 1$): Move the worst vertex through the centroid of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{\text{worst}})$.
  \item \textbf{Expansion} ($\gamma = 2$): If reflection improved, try further: $x_e = \bar{x} + \gamma(\bar{x} - x_{\text{worst}})$.
  \item \textbf{Contraction} ($\rho = 0.5$): If reflection worsened, contract towards centroid.
  \item \textbf{Shrink} ($\sigma = 0.5$): Contract all vertices towards the best point.
\end{enumerate}
No gradient or Hessian information is required; the simplex geometry adapts purely from function evaluations.  Initial simplex step $= 0.35$, 160 iterations.

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Setup}

Minimise $f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)$ subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$.  The unconstrained minimiser lies at $(0.582, 2.0)$, which satisfies $x_1^{\star} = 0.582 > 0.5$, so the constraint is \emph{inactive} at the true optimum.  This is an important distinction: the constrained and unconstrained minima coincide in this case, but the starting point is infeasible and the methods differ in how they restore feasibility.

\subsection{Methods}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Gradient step (possibly infeasible)}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space constraint $x_1 \geq 0.5$, the projection reduces to $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1),\, z_2)$, which is a trivial $O(1)$ operation.

\begin{algorithm}[H]
\caption{Penalty Method (Exact Penalty)}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $\nabla F_{\lambda}(x) \gets \nabla f(x) - \lambda \mathbf{1}[x_1 < 0.5]\, e_1$ \Comment{Gradient of penalised objective}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalised objective is $F_{\lambda}(x) = f(x) + \lambda \max(0, -x_1 + 0.5)$, adding a linear penalty for constraint violation.  For the $\ell_1$ penalty, $F_{\lambda}$ is exact in the sense that for large enough $\lambda$, the minimiser of $F_{\lambda}$ coincides with the constrained minimiser.

\textbf{Settings:} Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ (step size $\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$ to maintain stability), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\begin{itemize}[nosep]
\item \textbf{Projected GD} enforces feasibility as a hard constraint at every iteration: the projection step guarantees $x_k \in \mathcal{X}$ for all $k \geq 1$. The cost is requiring a projection oracle; for simple box/half-space constraints this is trivial, but for complex constraint sets (e.g., semidefinite matrices, $\ell_1$ balls) projection can be computationally expensive.

\item \textbf{Penalty methods} avoid projection at the cost of exact feasibility: they solve an unconstrained problem and rely on the penalty weight $\lambda$ to drive feasibility. The $\ell_1$ penalty is an exact penalty (the minimiser is feasible for any $\lambda$ exceeding the optimal Lagrange multiplier), but the penalised gradient is discontinuous at the constraint boundary, which can cause instability with large $\lambda$.

\item \textbf{In this specific problem}, the unconstrained and constrained minima coincide ($x_1^{\star} = 0.582 > 0.5$), so all correctly parameterised methods converge to the same point. The difference lies solely in the path taken and whether infeasible iterates are visited.
\end{itemize}

\subsection{Feasible Set and Linear Programme}

The feasible set for all Q6 experiments is:
\begin{equation}
  \mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\; -5 \leq x_2 \leq 10\}.
\end{equation}

\textbf{Linear Programme (Q6-I):} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$.  Since $a_1 = 1 > 0$ and $a_2 = 2 > 0$, the minimum is achieved by setting both coordinates to their lower bounds:
\begin{equation}
  x^{\star} = (0.5,\; {-5}), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}

\begin{theorem}[Fundamental Theorem of Linear Programming]
The minimum of a linear objective over a bounded polytope $\mathcal{P}$ is attained at a vertex (extreme point) of $\mathcal{P}$.
\end{theorem}

This theorem explains why $x^{\star}$ is at a corner of $\mathcal{X}$: any interior or edge point can be improved by moving along the $-a$ direction until a vertex is reached.

The gradient $\nabla f = [1, 2]^T$ points towards the upper-right, so the descent direction $-\nabla f$ points towards the lower-left corner.}

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{s \in \mathcal{X}}\, \nabla f(x_k)^T s$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination step}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
Key properties of Frank--Wolfe:
\begin{enumerate}[nosep]
  \item \textbf{Feasibility preservation:} Since $x_0 \in \mathcal{X}$, $z_k \in \mathcal{X}$ (LMO), and $\mathcal{X}$ is convex, the convex combination $x_{k+1} = \beta x_k + (1-\beta) z_k \in \mathcal{X}$ for all $k$.  No projection oracle is needed.
  \item \textbf{LMO for box constraints:} For $\mathcal{X} = \prod_i [l_i, u_i]$, the LMO decomposes per coordinate: $(z_k)_i = l_i$ if $(\nabla f(x_k))_i > 0$, else $(z_k)_i = u_i$.  This requires $O(d)$ operations.
  \item \textbf{Convergence rate:} For smooth convex $f$, the standard Frank--Wolfe with step size $\gamma_k = 2/(k+2)$ achieves $f(x_k) - f^{\star} \leq \frac{2LC_f}{k+2}$ where $C_f = \max_{x,y \in \mathcal{X}} (f(y) - f(x) - \nabla f(x)^T(y-x))$ is the curvature constant.  This $O(1/k)$ rate is slower than gradient descent's $O(1/k^2)$ (under strong convexity) but avoids any projection computation.
\end{enumerate}
\end{remark}

For box constraints, the LMO is:
\begin{equation}
  (z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0, \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0. \end{cases}
\end{equation}

\subsection{Interior Optimum: $f(x) = (x_1 - 1)^2 + (x_2 - 5)^2$}

The unconstrained minimum $(1, 5)$ satisfies $0.5 \leq 1 \leq 5$ and $-5 \leq 5 \leq 10$, so it lies strictly inside $\mathcal{X}$.  Two mixing parameters are tested: $\beta = 0.90$ (10\% step towards LP vertex per iteration) and $\beta = 0.985$ (1.5\% step), starting from $x_0 = (1, 1)$.

This seemingly counterintuitive result arises because larger steps ($\beta = 0.90$) introduce greater oscillation: at each iteration, $z_k$ selects a vertex of $\mathcal{X}$ (e.g., $(0.5, -5)$ or $(0.5, 10)$) far from the current iterate, so larger steps cause the iterate to move too far. \subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits persistent oscillation because the LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ at each iteration.  The convex combination step gradually averages these jumps.  With the fixed $\beta$ schedule used here, the method does not achieve faster than $O(1/k)$ convergence.  The standard Frank--Wolfe with adaptive step sizes $\gamma_k = 2/(k+2)$ would give convergence rate $O(1/k)$ with the duality gap $g_k = \nabla f(x_k)^T(x_k - z_k)$ as a certificate.

\textbf{Boundary optimum:} Frank--Wolfe is well-suited to boundary optima because the LMO frequently selects the vertex nearest to the constrained minimiser (here, the lower bound $x_1 = 0.5$).  The iterates naturally approach the boundary without explicit projection.  The residual gap (0.364 vs $f^{\star} = 0.25$) indicates that 140 iterations are insufficient for full convergence to the boundary point; the fixed $\beta = 0.93$ schedule does not provide sufficient step decay near the optimum.  Using the decreasing schedule $\gamma_k = 2/(k+2)$ instead of fixed $\beta$ would eliminate this residual gap asymptotically.

\begin{remark}
The Frank--Wolfe algorithm is particularly valuable when the feasible set $\mathcal{X}$ is a complex polytope (e.g., the simplex, the nuclear norm ball, the spectrahedron) for which projection is expensive but the LP oracle is cheap.  Box constraints, as used here, represent the simplest case where both projection and LMO are $O(d)$.  In higher-dimensional problems (e.g., $d \sim 10^4$) over more complex feasible sets, Frank--Wolfe's projection-free nature becomes a decisive advantage.
\end{remark}

\subsection{Conclusions}

\textbf{(1) Problem structure dominates method selection.}  On the strongly convex quadratic (Benchmark A), virtually all methods converge to the noise floor $J^{\star} \approx 0.4826$; the choice of method matters little.  On the ill-conditioned Rosenbrock function, method choice is decisive: Newton's method, Nesterov acceleration, and correctly-specified Polyak all achieve dramatically better final values than standard GD.

\textbf{(2) Newton's method is unmatched on (near-)quadratic objectives.}  On Benchmark A, Newton converges in a single iteration regardless of the condition number, because the Hessian is global and constant.  On Benchmark B, it converges in $\sim 5$ steps.  The cost --- computing and inverting the $d \times d$ Hessian --- limits its applicability to small or medium dimensions, but within those constraints it offers the fastest convergence per iteration.

\textbf{(3) Polyak step requires exact $f^{\star}$.}  Polyak achieves the best Rosenbrock result ($f = 0.0094$ in 120 iterations) but diverges catastrophically when $f^{\star}$ is misspecified. This is the fundamental limitation of the method in practice: the true optimal value is almost never known in advance.

\textbf{(4) Nesterov acceleration is the best practical first-order method for ill-conditioned problems.}  Without second-order information, Nesterov achieves $f = 0.479$ on Rosenbrock in 150 iterations --- $7\times$ better than GD ($f = 3.39$) and $3\times$ better than Adam ($f = 1.58$). The theoretical $O(1/k^2)$ guarantee (vs $O(1/k)$ for GD) translates into a significant practical advantage on the banana-shaped valley.

\textbf{(5) Projected gradient descent provides exact constraint enforcement from iteration 1.}  For the constrained problem in Question 5, projected GD maintains feasibility at every step at negligible cost (a single $\max$ operation per coordinate). Penalty methods can achieve comparable solutions but require careful tuning of $\lambda$ and typically use smaller step sizes for stability.

\textbf{(6) Frank--Wolfe is a viable projection-free alternative for linearly constrained problems.}  By replacing the projection step with a much cheaper LP solve, Frank--Wolfe trades a slower $O(1/k)$ convergence rate for reduced per-iteration cost.  This trade-off favours Frank--Wolfe when the feasible set has cheap LP oracles but expensive projections (e.g., trace-norm constrained matrices, flow polytopes).

\textbf{(7) Derivative-free methods scale poorly but provide useful baselines.}  Nelder--Mead demonstrates that progress can be made on the Rosenbrock landscape without any gradient information, but it is substantially slower than gradient-based methods.  Grid search guarantees global optimality within its resolution but is fundamentally limited to $d \leq 3$ by its $O(N^d)$ evaluation cost.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Regularisation}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
