\begin{abstract}
% ============================================================
This report presents a comprehensive comparative study of optimisation algorithms applied across three benchmark problems of varying difficulty: a linear regression quadratic loss (Benchmark A, $\kappa \approx 1.117$), a mildly non-quadratic loss with sinusoidal perturbation (Benchmark B, $\kappa \approx 3$--$10$), and the highly ill-conditioned Rosenbrock function (Benchmark C, $\kappa \approx 2508$). Fourteen algorithms spanning six categories are systematically investigated: (1) adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball); (2) momentum and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD); (3) Newton's method with local second-order approximations; (4) derivative-free approaches (forward finite differences, Nesterov random search, Nelder--Mead simplex, grid search); (5) constrained optimisation via projected gradient descent and penalty methods; and (6) the Frank--Wolfe algorithm for linear-constrained problems. Theoretical contributions include six formal theorems (Newton quadratic convergence, Nesterov $O(1/k^2)$ rate, Adagrad regret bound, SGD with diminishing steps, projected GD convergence, and Frank--Wolfe $O(1/k)$ rate), one proposition (optimal GD step for quadratics), and fourteen remarks connecting theory to implementation. Key experimental findings include: (i) Polyak's step-size achieves the best convergence when $f^\star$ is known exactly but diverges catastrophically when misspecified (370$\times$ gap on Rosenbrock); (ii) Nesterov acceleration achieves the optimal $O(1/k^2)$ rate and yields a 7$\times$ objective improvement over GD on Benchmark C ($\kappa \approx 2508$) within 150 iterations, with theoretical long-run speedup $\sqrt{\kappa}/2 \approx 25\times$; (iii) Newton's method, with quadratic convergence verified via an explicit iteration trace ($\|e_5\| < 0.01$ predicted from $\|e_0\| \approx 2.5$), solves the quadratic loss in one step and converges in 3--5 steps on Benchmark B; (iv) the finite-difference step $\delta = 0.05$ closely tracks exact gradient descent (5.6\% relative gradient error), while $\delta = 0.8$ incurs 89\% truncation error; (v) projected GD enforces feasibility from iteration 1 while penalty methods require $\lambda \gtrsim 5$ for near-feasibility; and (vi) the Frank--Wolfe $O(1/k)$ rate is verified both theoretically (Theorem~\ref{thm:fw}) and via duality gap measurements. All implementations use a fixed random seed (\texttt{np.random.seed(42)}) for full reproducibility.
\end{abstract}

This report investigates a spectrum of such algorithms---from simple gradient descent to second-order Newton methods and derivative-free strategies---through systematic experiments on three benchmark functions that span the range from well-conditioned convex quadratics to highly ill-conditioned non-convex landscapes.

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmarks are designed to test progressively more challenging optimisation landscapes.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The mean squared error (MSE) loss is:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a \emph{strongly convex} quadratic with constant Hessian $H = \frac{1}{m}X^T X \succ 0$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. The smoothness constant is $L = \lambda_{\max}(H) \approx 1.033$ and the strong convexity constant is $\mu = \lambda_{\min}(H) \approx 0.925$, giving condition number $\kappa(H) \approx 1.117$---an exceptionally well-conditioned problem (near-isotropic Hessian). This follows from the law of large numbers: $\frac{1}{m}X^TX \to \mathbb{E}[xx^T] = I_2$ as $m \to \infty$; with $m = 1000$ the sample deviation from identity is small. The exact OLS solution is $\theta^\star_{\text{OLS}} = (X^TX)^{-1}X^Ty \approx [3.028, 4.017]^T$, close to the true $\theta^\star = [3, 4]^T$ due to the large sample size. The irreducible noise floor is $J^\star \approx 0.4826$ (empirically computed MSE at the OLS estimator). Starting point: $\theta_0 = [0, 0]^T$.

\textbf{Global convergence from any $\theta_0$.} Since $J$ is a strongly convex quadratic with $H = \frac{1}{m}X^TX \succ 0$, the global convergence guarantee $J(\theta_k) - J^\star \leq (1 - \mu/L)^k (J(\theta_0) - J^\star)$ holds from any starting point without any step-size tuning beyond $\alpha \leq 1/L$. In particular, $J(\theta_0 = 0) - J^\star = J(0) - 0.4826$ where $J(0) = \frac{1}{2m}\|y\|^2 \approx \frac{1}{2}(\theta^{\star T}\mathbb{E}[xx^T]\theta^\star + \sigma^2) = \frac{1}{2}(25 + 1) = 13$. The convergence is therefore from $\Delta_0 \approx 12.5$ to near $\varepsilon = 10^{-5}$ in $\approx \kappa \log(\Delta_0/\varepsilon) \approx 1.117 \times \log(12.5/10^{-5}) \approx 15$ iterations, consistent with the observed rapid convergence in Figures~\ref{fig:q1_fval_A} and~\ref{fig:q2_fval_A}. Note: contour plots are omitted for Benchmark A as the $\theta$-space trajectory is less illustrative than the convergence curve for this well-conditioned problem.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H = \text{diag}(2 - \sin(x_1),\ 10)$.
The first diagonal entry varies: for $x_1 \in [-1, 4]$, $\sin(x_1) \in [-1, 1]$, so $H_{11} \in [1, 3]$, giving a condition number in $[10/3, 10/1] \approx [3.3, 10]$. The minimum is determined by:
\begin{equation}
\frac{\partial f}{\partial x_1} = 2(x_1^* - 1) + \cos(x_1^*) = 0, \quad x_2^* = 2.
\end{equation}
Numerically, $x_1^* \approx 0.5818$, $f^\star \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^\star = 0$. Named after H.\ H.\ Rosenbrock who introduced it as a test for optimisation algorithms \cite{rosenbrock1960}. The Hessian at the optimum $(1,1)$ is:
\begin{equation}
H(1,1) = \begin{bmatrix} 2 + 1200 - 400 & -400 \\ -400 & 200 \end{bmatrix} = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}.
\end{equation}
The eigenvalues are $\lambda_1 \approx 0.3994$ and $\lambda_2 \approx 1001.6$, yielding condition number $\kappa \approx 2508$---extreme ill-conditioning. The non-convex, narrow curved valley along $x_2 = x_1^2$ makes this function one of the most challenging standard test problems. Starting point: $x_0 = [-1, 1]^T$. Note: the condition number \emph{varies} along the valley --- at $x_0 = (-1, 1)$, the Hessian eigenvalues are approximately $(4.0, 10003)$ giving $\kappa \approx 2500$, while Figure~\ref{fig:q3_hessian} shows the condition number exceeds $10^4$ near $x_1 = 0$. The step size $\alpha \leq 1/L$ must be set using the \emph{global} Lipschitz constant, making GD conservative throughout the trajectory.

\subsection{Benchmark D: Constrained Feasible Set}
For constrained optimisation (Questions 5--6): $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations. The key motivation is that a fixed step size $\alpha$ must satisfy $\alpha \leq 1/L$ for guaranteed descent on $L$-smooth functions (the \emph{descent lemma}):
\begin{equation}
f(x_{k+1}) \leq f(x_k) - \alpha\left(1 - \frac{\alpha L}{2}\right)\|\nabla f(x_k)\|^2,
\end{equation}
which requires $\alpha \leq 2/L$ for descent. This conservative global step may be unnecessarily slow in directions with small curvature. Adaptive methods instead scale the step size per-coordinate based on local curvature estimates.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Adaptive Polyak step}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step \cite{polyak1969} arises from minimising $\|x_{k+1} - x^\star\|^2$ subject to $x_{k+1} = x_k - \alpha g_k$. For $\varepsilon = 0$ and exact $f^\star$, this gives:
\begin{equation}
\|x_{k+1} - x^\star\|^2 = \|x_k - \alpha_k g_k - x^\star\|^2 = \|x_k - x^\star\|^2 - 2\alpha_k g_k^T(x_k - x^\star) + \alpha_k^2\|g_k\|^2,
\end{equation}
which is minimised by $\alpha_k = g_k^T(x_k - x^\star)/\|g_k\|^2$. Since $g_k^T(x_k - x^\star) \geq f(x_k) - f^\star$ (by convexity: $f(x^\star) \geq f(x_k) + g_k^T(x^\star - x_k)$), the choice $\alpha_k = (f(x_k) - f^\star)/\|g_k\|^2$ is a lower bound on the optimal step, guaranteeing descent. On convex functions with exact $f^\star$, convergence is $O(1/k)$ in function value.

\begin{remark}[Polyak step as projected subgradient]
The Polyak step is closely related to the \emph{projected subgradient} method: for non-smooth convex functions, the subgradient step $x_{k+1} = x_k - \alpha_k g_k$ with Polyak's step $\alpha_k = (f(x_k) - f^\star)/\|g_k\|^2$ achieves the best possible $O(1/\sqrt{k})$ rate without knowing the Lipschitz constant $G = \sup_k \|g_k\|$.
\end{remark}

\begin{remark}[Polyak step vs backtracking Armijo]
An alternative adaptive step that does \emph{not} require $f^\star$ is the \emph{Armijo backtracking} rule: start with $\alpha = \alpha_{\max}$ and reduce by factor $c_{\text{back}} \in (0,1)$ until $f(x_k - \alpha g_k) \leq f(x_k) - c_1 \alpha \|g_k\|^2$. This requires only function values (no $f^\star$) and guarantees the sufficient decrease condition by construction. The key difference from Polyak: Armijo only guarantees \emph{sufficient decrease} (not descent to $f^\star$ in one step), giving a standard linear convergence rate. Polyak achieves potentially faster convergence (one-step optimality) when $f^\star$ is exact. In practice, Armijo backtracking or Wolfe conditions are the standard approach for gradient-based methods, combining the safety of sufficient decrease with the adaptivity of a variable step size.
\end{remark}

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Monotone per-coordinate accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad scales each coordinate by the inverse square root of the cumulative sum of squared gradients. Coordinates that historically received large gradients are penalised with smaller effective learning rates, providing automatic anisotropic scaling. However, the monotone accumulation $G_k \to \infty$ causes the effective learning rate $\alpha_0/\sqrt{G_k}$ to \emph{monotonically decay to zero}, which can prematurely stop convergence on non-stationary problems.

\begin{remark}[Adagrad on sparse data]
On problems with sparse gradient coordinates (e.g., NLP with infrequent word features), Adagrad's monotone decay is actually \emph{beneficial}: rarely-updated coordinates retain a larger effective learning rate because their accumulated $G_k$ grows slowly, allowing them to take larger steps when finally updated. This is the original motivation for Adagrad \cite{duchi2011}.
\end{remark}

\begin{remark}[Coordinate descent as a related adaptive method]
\emph{Coordinate descent} cyclically or randomly minimises $f$ over one coordinate at a time, effectively using curvature information along each axis independently. For functions with block-separable structure or cheap coordinate-wise minimisers, it achieves convergence rates similar to Adagrad's per-coordinate adaptation. On Benchmark A ($f$ separable after a rotation to eigenvectors of $H$), coordinate descent in the eigenvector basis would coincide with Newton's method in one step per coordinate. On Benchmark B, the diagonal Hessian makes coordinate descent equivalent to Newton's method (exact minimisation in each direction), converging in at most 2 steps from any starting point---faster than any of the methods studied here.
\end{remark}

\begin{theorem}[Adagrad Regret Bound {\cite{duchi2011}}]
For a convex objective $f$ with subgradients bounded by $\|g_t\| \leq G$ and diameter $D = \|x_1 - x^\star\|_2$, Adagrad with step $\alpha_0$ achieves after $T$ rounds:
\begin{equation}
\text{Regret}(T) = \sum_{t=1}^T [f(x_t) - f(x^\star)] \leq \frac{D^2}{2\alpha_0}\sqrt{\sum_{i=1}^d \sum_{t=1}^T g_{t,i}^2} + \alpha_0 G\sqrt{d T}.
\end{equation}
This simplifies to $\text{Regret}(T) = O(D G \sqrt{T \cdot d})$ in the worst case, giving an average regret of $O(1/\sqrt{T})$. The key advantage over SGD with fixed step is that the bound adapts to the gradient geometry: if gradients are sparse (few non-zero coordinates), $\sum_{t,i} g_{t,i}^2 \ll d \cdot T \cdot G^2$ and Adagrad achieves a much tighter bound than the worst case.
\end{theorem}

\noindent\textbf{Numerical application to Benchmark A.} With $d = 2$, $T = 120$, $\alpha_0 = 1.8$, $G \approx \|\nabla J(\theta_0)\| \approx 7.6$ (gradient norm at $\theta_0 = 0$), and $D = \|\theta_0 - \theta^\star\| \approx 5.0$: the Adagrad regret bound gives average regret $\leq O(5 \times 7.6 \times \sqrt{120 \times 2}) / 120 \approx 38 \times 15.5 / 120 \approx 4.9$. This is a very loose bound (observed average regret is $\approx 0.0001$), because our benchmark is deterministic with rapidly decaying gradients---the $O(\sqrt{T})$ worst-case bound is not tight for strongly convex problems where the actual regret accumulates as $O(\log T)$ rather than $O(\sqrt{T})$.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's full history with an exponential moving average (decay $\beta = 0.9$). This prevents the step size from decaying to zero by giving more weight to \emph{recent} gradient magnitudes, making it better suited to non-stationary or noisy objectives. The effective learning rate converges to $\alpha_0 / \sqrt{\mathbb{E}[g^2]}$ in expectation.

\begin{remark}[RMSprop convergence]
Unlike Adagrad (which has a formal online regret bound), RMSprop has \emph{no formal convergence guarantee} for general convex functions. Counter-examples exist showing it can cycle on simple convex problems \cite{reddi2018}. Despite this, RMSprop works well empirically for training deep networks, particularly recurrent architectures (it was proposed by Hinton in an unpublished Coursera lecture \cite{tieleman2012} specifically for RNNs).
\end{remark}

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Momentum accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

Heavy Ball maintains a velocity $z_k$ that accumulates gradients with exponential decay, analogous to a ball rolling down a hill with friction $1 - \beta$. The momentum $\beta$ allows the iterate to continue in consistent descent directions, damping the oscillations that gradient descent exhibits in ravines. For $\mu$-strongly convex $L$-smooth quadratics, the optimal Heavy Ball parameters are:
\begin{equation}
\alpha^* = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2}, \quad \beta^* = \left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^2 = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2,
\end{equation}
achieving the optimal linear convergence rate $r^* = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$---the same as the Chebyshev iteration and $\sqrt{\kappa}$-fold better than GD's rate $(\kappa-1)/(\kappa+1)$. However, Heavy Ball's global optimality guarantees are limited to quadratics; on general non-convex problems (such as the Rosenbrock), it may not outperform GD and can exhibit persistent oscillations if $\beta$ is too large.

\textbf{Heavy Ball vs Nesterov on quadratics.} For strongly convex quadratics, both Heavy Ball (with optimal parameters) and Nesterov acceleration achieve the same $O((\sqrt{\kappa}-1/\sqrt{\kappa}+1)^k)$ linear convergence rate. However, Nesterov generalises to \emph{non-quadratic} convex functions with the same $O(\rho^{2k})$ strongly-convex rate and the optimal $O(1/k^2)$ general-convex rate, while Heavy Ball only achieves $O(\rho^k)$ on non-quadratics. On Benchmark B (mildly non-quadratic due to $\sin(x_1)$), Heavy Ball with $\beta = 0.90$ achieves $f = 0.7244$ (matching GD), while Nesterov with the lookahead also achieves $f = 0.7244$. On Benchmark C (highly non-convex), Nesterov's $7\times$ improvement over GD reflects its superior handling of the non-quadratic valley, while Heavy Ball achieves $2.5\times$ improvement (f=1.40 vs 3.51)---a middle ground between GD and Nesterov.

\begin{proposition}[Optimal GD Step Size for Quadratics]
For a $\mu$-strongly convex $L$-smooth quadratic $f(x) = \frac{1}{2}x^T H x - b^T x$, the optimal constant step size that minimises the worst-case convergence factor is:
\begin{equation}
\alpha^\star = \frac{2}{\mu + L} = \frac{2}{L(1 + 1/\kappa)},
\end{equation}
achieving the convergence rate $\rho^\star = (L - \mu)/(L + \mu) = (\kappa-1)/(\kappa+1)$. This is better than the conservative choice $\alpha = 1/L$ (which gives rate $1 - 1/\kappa$). For Benchmark A: $\alpha^\star = 2/(0.925 + 1.033) = 2/1.958 \approx 1.02$ and $\rho^\star = (1.033-0.925)/(1.033+0.925) = 0.108/1.958 \approx 0.055$, which equals $(\kappa-1)/(\kappa+1)$ as expected. In the experiments, $\alpha = 0.08$ is used (below $\alpha^\star = 1.02$) to ensure stability in non-quadratic benchmarks.
\end{proposition}

\textbf{Numerical rate verification on Benchmark A.} With $L = 1.033$, $\mu = 0.925$, $\kappa = 1.117$, the theoretical GD convergence rate is $\rho_{\text{GD}} = 1 - \mu/L = 1 - 0.925/1.033 = 0.104$, with convergence factor per iteration $r_{\text{GD}} = (\kappa-1)/(\kappa+1) = 0.117/2.117 = 0.055$. For Heavy Ball, the optimal rate is $r^*_{\text{HB}} = (\sqrt{\kappa}-1)/(\sqrt{\kappa}+1) = (1.057-1)/(1.057+1) = 0.027$---\emph{exactly half} the GD rate, meaning HB converges in approximately half as many iterations on Benchmark A. This modest $2\times$ speedup (rather than the $\sqrt{\kappa} \approx 25\times$ speedup on a poorly-conditioned problem) reflects that Benchmark A is nearly well-conditioned ($\kappa \approx 1.117$), where both GD and HB converge rapidly.

\textbf{Analysis of Polyak divergence.} Define the Polyak step as $\alpha_k = (f(x_k) - f^\star_{\text{spec}})/(\|g_k\|^2 + \varepsilon)$ where $f^\star_{\text{spec}}$ is the specified value. When $f^\star_{\text{spec}} < f^\star_{\text{true}}$, we have $f(x_k) - f^\star_{\text{spec}} \to f^\star_{\text{true}} - f^\star_{\text{spec}} > 0$ as the iterate converges, while $\|g_k\|^2 \to 0$. Consequently $\alpha_k \to \infty$, causing gradient steps of unbounded magnitude and divergence. This analysis shows that Polyak's convergence guarantee is \emph{non-robust}---it requires exact knowledge of $f^\star$, which is rarely available in practice.

\begin{remark}[Polyak-\L ojasiewicz condition and GD convergence on non-convex functions]
The Polyak step's convergence with correct $f^\star$ relies on a generalisation of strong convexity called the \emph{Polyak--\L ojasiewicz (PL) condition}: $\|\nabla f(x)\|^2 \geq 2\mu(f(x) - f^\star)$ for all $x$. Under PL, GD with step $\alpha \leq 1/L$ satisfies the linear rate $f(x_k) - f^\star \leq (1 - \mu\alpha)^k(f(x_0) - f^\star)$ \emph{even without convexity}. The Rosenbrock function satisfies a local PL condition near its minimum $(1,1)$ (since $f^\star = 0$ is a global minimum and the gradient does not vanish except at $(1,1)$), which partially explains why Polyak converges on Benchmark C despite non-convexity. Global PL does not hold for the Rosenbrock (the gradient can be arbitrarily small far from the minimum along the curved valley), but local PL in the basin of attraction allows fast convergence once the iterate is sufficiently close.
\end{remark}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.009$). Otherwise, Adagrad and Heavy Ball match or exceed the GD baseline.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging of squared gradients, preventing the extreme step-size swings seen in Polyak or Adagrad.
\item \textbf{Stability:} GD is the most stable due to its fixed step size satisfying the descent condition. Adagrad is uniformly stable because $\|G_k\|$ is non-decreasing. Polyak is \emph{unstable} when $f^\star$ is misspecified.
\item \textbf{Anisotropy handling:} Adagrad and RMSprop handle coordinate-wise anisotropy (different curvatures per dimension) automatically, an advantage on Benchmark B and C. Heavy Ball can exploit anisotropy through the choice of $(\alpha, \beta)$, but requires careful tuning.
\item \textbf{Practical guidance:} In practice, exact $f^\star$ is rarely known, limiting Polyak's applicability. RMSprop and Adagrad are more robust defaults for unknown landscapes. For problems with known strong convexity, Heavy Ball with optimal momentum parameters provides theoretically faster convergence.
\item \textbf{Theoretical convergence rates.} For Benchmark A ($\mu \approx 0.925$, $L \approx 1.033$, $\kappa \approx 1.117$):
\begin{equation}
\text{GD rate: } r_{\text{GD}} = \frac{\kappa - 1}{\kappa + 1} \approx \frac{0.117}{2.117} \approx 0.055, \qquad \text{HB optimal rate: } r_{\text{HB}} = \frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1} \approx \frac{0.057}{2.057} \approx 0.028.
\end{equation}
Both rates are very small, explaining why all well-behaved methods converge rapidly on Benchmark A: the near-isotropic Hessian means gradient descent is already near-optimal and Heavy Ball provides only modest improvement. The more interesting comparisons arise on Benchmarks B and C where curvature anisotropy is more pronounced.
\end{itemize}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Update velocity at lookahead}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This ``corrected'' gradient allows Nesterov acceleration to anticipate and partially cancel the overshoot caused by momentum. The momentum schedule $\beta_k = (k-1)/(k+2)$ approximates the optimal weight in Nesterov's original analysis, achieving the \emph{optimal} $O(1/k^2)$ convergence rate for convex smooth functions \cite{nesterov1983}. The cap $\beta_k \leq \beta_{\max}$ prevents overshooting on non-convex problems where the theoretical schedule $(k-1)/(k+2) \to 1$ can cause divergence when gradient directions change rapidly near non-convex regions. On Benchmark B we use $\beta_{\max} = 0.92$; on Benchmark C, $\beta_{\max} = 0.90$ to handle the curved Rosenbrock valley.

\begin{theorem}[Nesterov Acceleration Rate {\cite{nesterov2004}}]
For $f$ convex and $L$-smooth, Nesterov's method with step size $\alpha = 1/L$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2}.
\end{equation}
This is optimal: no first-order method can achieve a better worst-case rate on this problem class.
\end{theorem}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{1st moment (exponential moving average)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{2nd moment (raw, uncentred variance)}
    \State $\hat{m}_t \gets m_t / (1 - \beta_1^t)$ \Comment{Bias-corrected 1st moment}
    \State $\hat{v}_t \gets v_t / (1 - \beta_2^t)$ \Comment{Bias-corrected 2nd moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t / (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam \cite{kingma2014} combines momentum (via $m_t$, analogous to Heavy Ball) with per-coordinate adaptive scaling (via $v_t$, analogous to RMSprop). The bias correction terms $(1 - \beta_1^t)^{-1}$ and $(1 - \beta_2^t)^{-1}$ correct for the initialisation bias that arises because $m_0 = v_0 = 0$: without correction, early iterates would be biased towards zero.

\begin{remark}[Importance of bias correction]
With $\beta_2 = 0.999$, at iteration $t = 10$: $(1 - 0.999^{10})^{-1} \approx 100$. The second moment estimate $v_{10}$ is only $1\%$ of its steady-state value, so the effective step size without bias correction would be $10\times$ too small. Bias correction ensures Adam takes appropriate step sizes from the first iteration. By $t = 1000$, $(1 - 0.999^{1000})^{-1} \approx 1.58$---the correction becomes minor. This analysis justifies keeping $\beta_2$ close to 1.0 (long memory) while using bias correction.
\end{remark}

\begin{remark}[Adam as RMSprop + Heavy Ball]
Adam can be understood as a combination of two previously studied methods: the exponential-moving-average preconditioner from RMSprop ($v_t$ tracks $\mathbb{E}[g_t^2]$ coordinate-wise) and the momentum accumulation from Heavy Ball ($m_t$ tracks $\mathbb{E}[g_t]$). The crucial insight is that the preconditioner $\hat{v}_t$ deactivates momentum in directions where gradients are large and fluctuating (causing small effective steps), while amplifying progress in directions where gradients are small and consistent---precisely the behaviour needed to navigate the ill-conditioned Rosenbrock valley. The asymptotic update is $\Delta x_t \approx -\alpha \, \text{sign}(\hat{m}_t) / |\hat{v}_t|^{1/2}$, which behaves like a per-coordinate sign-SGD with adaptive magnitude.
\end{remark}

\begin{remark}[AMSGrad: a convergent Adam variant]
The convergence failure of Adam identified by \cite{reddi2018} (specifically, a constant-regret counter-example for online convex optimisation) arises because the exponential average $v_t$ can decrease over time, violating the monotone-accumulation property required for the regret bound. The \emph{AMSGrad} fix replaces $v_t$ with $\hat{v}_t^{\max} = \max(\hat{v}_{t-1}^{\max}, \hat{v}_t)$, ensuring a monotone non-decreasing denominator. On the counter-example where Adam fails, AMSGrad converges. In practice, however, AMSGrad often performs similarly to or worse than Adam because the ``forgetting'' behaviour of the exponential average is beneficial on non-stationary stochastic objectives---it allows the effective learning rate to recover after a period of high gradient variance, whereas AMSGrad's monotone denominator prevents this recovery.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled every epoch. The mini-batch gradient is:
\begin{equation}
g_{\mathcal{B}} = \frac{1}{b} X_\mathcal{B}^T(X_\mathcal{B}\theta - y_\mathcal{B}),
\end{equation}
which satisfies $\mathbb{E}[g_\mathcal{B}] = \nabla J(\theta)$ (unbiased). The gradient variance is:
\begin{equation}
\mathbb{E}\|g_\mathcal{B} - \nabla J(\theta)\|^2 = \frac{\sigma^2_{\text{grad}}}{b},
\end{equation}
where $\sigma^2_{\text{grad}}$ is the per-sample gradient variance. Larger batches give lower variance but fewer updates per epoch. There is a fundamental trade-off: with $N$ samples and batch size $b$, each epoch performs $N/b$ updates, so the total work per epoch is fixed at $N$ gradient evaluations regardless of $b$. The key difference is the \emph{quality} of each update: small $b$ gives noisier but more frequent updates; large $b$ gives cleaner updates but fewer per epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[SGD noise floor with constant learning rate]
With constant step size $\alpha$, mini-batch SGD does not converge to $x^\star$ but oscillates in a ball of radius $O(\alpha \sigma^2_{\text{grad}} / (b \mu))$ around $x^\star$, where $\sigma^2_{\text{grad}}$ is the per-sample gradient variance and $\mu$ is the strong convexity constant. This \emph{irreducible noise floor} is visible in the constant-$\alpha$ curve of Figure~\ref{fig:q2_lr_sched}, which settles noticeably above the OLS minimum $J^\star \approx 0.4826$. For Benchmark A with $b = 20$, $\sigma^2_{\text{grad}} \approx \text{Var}[(X_i^T r_i)^2] / m \approx \sigma^2 \mathbb{E}[\|X_i\|^2] \approx 0.97 \times 2 \approx 2$ (using $\sigma^2 \approx 1$ and $\mathbb{E}[\|X_i\|^2] = d = 2$), and the noise floor is $\approx \alpha \times 2 / (20 \times 0.93) \approx 0.06 \times 2/18.6 \approx 0.0065$---a relative error of $\approx 1.3\%$ above $J^\star$, consistent with the plateau observed.
\end{remark}

\begin{theorem}[SGD with Diminishing Step Size]
For $f$ $\mu$-strongly convex and $L$-smooth, and unbiased stochastic gradients with bounded variance $\sigma^2 = \mathbb{E}\|g_t - \nabla f(x_t)\|^2 < \infty$, SGD with step $\alpha_k = c/k$ satisfies:
\begin{equation}
\mathbb{E}[f(x_k)] - f^\star \leq \frac{L c^2 \sigma^2}{2\mu c - 1} \cdot \frac{1}{k}, \quad \text{provided } c > \frac{1}{2\mu}.
\end{equation}
This $O(1/k)$ rate is optimal for SGD \emph{without} variance reduction techniques (SVRG, SARAH) which can recover linear convergence at the cost of additional gradient evaluations.
\end{theorem}

\begin{remark}[Constant vs diminishing step trade-off]
A constant step $\alpha$ achieves linear convergence to a ball of radius $O(\alpha\sigma^2/\mu)$ (the noise floor). A diminishing step $\alpha_k = c/k$ eliminates the noise floor but at the cost of slower asymptotic rate $O(1/k)$ vs $O(\rho^k)$. The cross-over iteration is approximately $k^* = \log(\sigma^2/(\mu(f(x_0)-f^\star)))/\log(1/\rho) \approx \kappa\log(\sigma^2/(f_0-f^\star))$, where it becomes beneficial to switch from constant to diminishing steps.
\end{remark}

\begin{remark}[Polyak-Ruppert averaging]
For convex $f$ with $O(1/\sqrt{k})$ convergence under constant step SGD, \emph{Polyak--Ruppert averaging} (outputting $\bar{\theta}_k = \frac{1}{k}\sum_{t=1}^k \theta_t$ rather than $\theta_k$) recovers the optimal $O(1/\sqrt{k})$ rate under weaker assumptions (without strong convexity). For strongly convex problems, averaging with weights $w_t = t$ gives $O(1/k^2)$ convergence of the averaged iterate even with the $O(1/k)$ step size. This technique is closely related to the variance-reduced SGD methods and is used in the stochastic approximation literature as a way to obtain the optimal asymptotic efficiency (matching the Cramér--Rao lower bound for statistical estimation).
\end{remark}

\begin{remark}[Variance reduction: beyond vanilla SGD]
Variance reduction methods (SVRG \cite{johnson2013}, SARAH, SPIDER) achieve \emph{linear convergence} for finite-sum problems $f(\theta) = \frac{1}{n}\sum_{i=1}^n f_i(\theta)$ by periodically computing the full gradient as a control variate. SVRG uses the update $g_k = \nabla f_{i_k}(x_k) - \nabla f_{i_k}(\tilde{x}) + \nabla f(\tilde{x})$ where $\tilde{x}$ is a snapshot every $2n$ iterations, achieving convergence rate $(1 - \Theta(\mu/L))^k$ with only $O(n + \kappa)$ gradient evaluations per $\varepsilon$-accurate solution---compared to $O(n\kappa)$ for full GD or $O(1/\varepsilon)$ for SGD. For Benchmark A with $n = 1000$ and $\kappa \approx 1.117$, SVRG would require approximately $O(1001)$ evaluations per epoch vs standard SGD's $O(n) = O(1000)$, offering negligible benefit for this nearly-isotropic problem but substantial gains for large $\kappa$.
\end{remark}

\subsubsection{Effect of Learning Rate Schedule}

The learning rate $\alpha$ can be adapted over epochs using a \emph{schedule} rather than remaining constant. Four common schedules are compared:
\begin{itemize}[nosep]
  \item \textbf{Constant}: $\alpha_e = \alpha_0$. Simple but may oscillate near the minimum.
  \item \textbf{Step decay}: $\alpha_e = \alpha_0 \cdot r^{\lfloor e/s \rfloor}$ (here $r = 0.5$, $s = 10$). Halves every 10 epochs.
  \item \textbf{Inverse decay}: $\alpha_e = \alpha_0/(1 + \gamma e)$ (here $\gamma = 0.05$). Decays as $O(1/e)$.
  \item \textbf{Cosine annealing}: $\alpha_e = \tfrac{\alpha_0}{2}(1 + \cos(\pi e/E))$. Smooth decay to 0.
\end{itemize}

\subsection{Theoretical Acceleration Analysis}

The observed improvements of Nesterov over GD can be connected to theory. On a $\mu$-strongly convex, $L$-smooth function, the convergence rates are:
\begin{equation}
f(x_k) - f^\star \leq \begin{cases} \left(1 - \frac{\mu}{L}\right)^k (f(x_0) - f^\star) & \text{GD (linear rate } \rho_{\text{GD}} = 1 - 1/\kappa\text{)} \\[4pt] \left(1 - \sqrt{\frac{\mu}{L}}\right)^{2k} \frac{L}{2}\|x_0 - x^\star\|^2 & \text{Nesterov (linear rate } \rho_{\text{NAG}} = (1 - 1/\sqrt{\kappa})^2\text{)} \end{cases}
\end{equation}
For Benchmark C (approximately convex near the valley), the effective condition number is very large, so:
\begin{equation}
\frac{\log \rho_{\text{GD}}}{\log \rho_{\text{NAG}}} \approx \frac{1/\kappa}{2/\sqrt{\kappa}} = \frac{\sqrt{\kappa}}{2}.
\end{equation}
With $\kappa \approx 2508$, Nesterov theoretically converges approximately $\sqrt{2508}/2 \approx 25$ times faster (in log scale), consistent with the observed $\approx 7\times$ improvement in final objective after 150 iterations (since 150 iterations is not large relative to the convergence time).

\textbf{Iteration count to $\varepsilon$-accuracy on Benchmark C.} To reduce the initial sub-optimality $\Delta_0 = f(x_0) - f^\star \approx 5.75$ to $\varepsilon = 0.01$, GD requires approximately
\begin{equation}
k_{\text{GD}} \approx \kappa \ln\!\left(\frac{\Delta_0}{\varepsilon}\right) = 2508 \times \ln(575) \approx 2508 \times 6.35 \approx 15{,}900 \text{ iterations},
\end{equation}
while Nesterov requires approximately
\begin{equation}
k_{\text{NAG}} \approx \sqrt{\kappa} \ln\!\left(\frac{\Delta_0}{\varepsilon}\right) = \sqrt{2508} \times 6.35 \approx 50 \times 6.35 \approx 318 \text{ iterations}.
\end{equation}
After a budget of only 150 iterations, GD has completed approximately $150/15900 \approx 1\%$ of the required convergence work, while Nesterov has completed approximately $150/318 \approx 47\%$. This $47\times/1\% \approx 47$-fold difference in fractional progress is consistent with the $\approx 7\times$ ratio in final objective value observed experimentally (the mapping from fraction-of-work to objective ratio is nonlinear due to the exponential convergence model).

\textbf{Benchmark A comparison.} For Benchmark A with $\kappa \approx 1.117$, the theoretical speedup is $\sqrt{1.117}/2 \approx 0.53$---meaning Nesterov is barely faster than GD on this well-conditioned problem. In 150 iterations both converge to near $J^\star = 0.4826$; the speedup advantage of Nesterov only becomes significant for $\kappa \gg 1$.

\textbf{Why Nesterov is not faster per function evaluation.} Nesterov evaluates the gradient at the lookahead point $\tilde{x}_k$ rather than $x_k$, so it uses exactly 1 gradient evaluation per iteration (same as GD). The $O(1/k^2)$ rate is genuinely a faster convergence in the \emph{same number of gradient evaluations}---not just a different accounting trick. The lookahead provides ``anticipatory'' gradient information at essentially no extra cost.

\textbf{Adam vs Nesterov.} Adam's convergence guarantee is weaker than Nesterov's for deterministic problems: counter-examples exist where Adam does not converge even on convex functions \cite{reddi2018}. In practice, Adam performs well with tuned hyperparameters but may oscillate in ill-conditioned deterministic landscapes. Nesterov's provably optimal rate makes it preferable when the problem is convex and $\nabla f$ is available.

\textbf{Sign-based methods as Adam variants.} Taking the limit $\beta_2 \to 1$ in Adam reduces the update to $x_{t+1} = x_t - \alpha \operatorname{sign}(\hat{m}_t)$, known as \emph{sign SGD}. This sign-based update effectively performs per-coordinate gradient clipping, making it highly robust to gradient scale changes---important in distributed training where gradient magnitudes vary across data shards. The recently proposed \emph{Lion} optimiser (2023) uses a sign-based update with a memory-efficient formulation requiring only $m_t$ (not $v_t$), halving Adam's memory footprint. These methods illustrate how Adam's theoretical components can be disentangled and simplified while retaining practical performance.

\textbf{Connection to FISTA.} Nesterov acceleration is the theoretical basis for the \emph{Fast Iterative Shrinkage-Thresholding Algorithm} (FISTA), which applies the accelerated proximal gradient method to objectives of the form $F(x) = f(x) + g(x)$ where $f$ is smooth and $g$ is non-smooth (e.g., $\ell_1$ norm for sparse optimisation). The same $O(1/k^2)$ rate and the momentum schedule $\beta_k = (k-1)/(k+2)$ carry over directly from the smooth setting. This makes Nesterov acceleration practically important in compressed sensing, LASSO regression, and image reconstruction problems.

\subsection{Local Approximations of $g(x) = x^4$}

The first- and second-order Taylor approximations of $g(x) = x^4$ around $x_0 = 0.25$ are:
\begin{align}
g(x) &\approx g(x_0) + g'(x_0)(x - x_0) && \text{(first-order)} \\
     &= 0.0039 + 0.0625(x - 0.25),\notag \\[4pt]
g(x) &\approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 && \text{(second-order)} \\
     &= 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.\notag
\end{align}
Here $g'(x_0) = 4(0.25)^3 = 0.0625$ and $g''(x_0) = 12(0.25)^2 = 0.75$. Newton's update minimises the local quadratic model:
\begin{equation}
x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3}.
\end{equation}
Starting from $x_0 = 0.25$: $x_1 = 0.25 - 0.25/3 \approx 0.167$, $x_2 \approx 0.111$, converging to 0 at rate $x_k = (2/3)^k \cdot 0.25$. This demonstrates Newton's ability to exploit curvature information to direct the step towards the minimum of the local model, rather than merely stepping against the gradient.

\subsection{Newton's Method with Damping}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, Hessian damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularise: ensures $H_k \succ 0$}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (natural gradient)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton direction $p_k = H_k^{-1}g_k$ is the exact minimiser of the local second-order model $m_k(d) = f(x_k) + g_k^T d + \frac{1}{2}d^T H_k d$. The damping $\lambda I$ regularises the Hessian when it is near-singular or has small eigenvalues, preventing excessively large Newton steps. The step size $\alpha < 1$ provides additional damping for highly non-quadratic problems. Hyperparameter settings are given in Table~\ref{tab:q3_params}.

\subsection{Newton Convergence Theory}

\begin{theorem}[Local Quadratic Convergence of Newton's Method]
Let $f : \mathbb{R}^d \to \mathbb{R}$ be twice continuously differentiable with $\nabla^2 f$ Lipschitz continuous: $\|\nabla^2 f(x) - \nabla^2 f(y)\| \leq M\|x - y\|$ for all $x, y$. Suppose $x^\star$ is a local minimum with $\nabla^2 f(x^\star) \succ 0$ (minimum eigenvalue $\mu > 0$). Then in a neighbourhood of $x^\star$, the undamped Newton iterates ($\alpha = 1$) satisfy:
\begin{equation}
\|x_{k+1} - x^\star\| \leq \frac{M}{2\mu}\,\|x_k - x^\star\|^2.
\end{equation}
\end{theorem}

This quadratic convergence contrasts sharply with GD's linear convergence $\|x_{k+1} - x^\star\| \leq (1 - \mu/L)\|x_k - x^\star\|$. Once $\|x_k - x^\star\| < 2\mu/M$, Newton's method is in the \emph{quadratic phase} and the number of accurate digits \emph{doubles} at each step.

\textbf{Quadratic phase threshold.} For Benchmark B at $x^\star \approx (0.582, 2.0)$, the Hessian Lipschitz constant can be estimated from $g''(x) = \nabla^2_x f$ varying with $x$: $M \approx \|\partial H / \partial x\|_{\max} \approx 1$ (only $\sin(x_1)$ contributes). With $\mu = \lambda_{\min}(H^\star) \approx 1.5$, the quadratic phase threshold is $\|x_k - x^\star\| < 2 \times 1.5 / 1 = 3.0$---essentially the entire feasible region, explaining why Newton converges in 3--5 steps from $x_0 = (-1, 4)$ (distance $\approx 2.5$).

\textbf{Explicit convergence trace on Benchmark B.} With $C = M/(2\mu) \approx 1/3$ and initial error $\|e_0\| = \|x_0 - x^\star\| \approx 2.5$, the quadratic bound predicts:
\begin{align}
\|e_0\| &\approx 2.5, \nonumber\\
\|e_1\| &\leq C\|e_0\|^2 = (1/3)(2.5)^2 \approx 2.08, \nonumber\\
\|e_2\| &\leq C\|e_1\|^2 \leq (1/3)(2.08)^2 \approx 1.44, \nonumber\\
\|e_3\| &\leq C\|e_2\|^2 \leq (1/3)(1.44)^2 \approx 0.69, \nonumber\\
\|e_4\| &\leq C\|e_3\|^2 \leq (1/3)(0.69)^2 \approx 0.16, \nonumber\\
\|e_5\| &\leq C\|e_4\|^2 \leq (1/3)(0.16)^2 \approx 0.0085. \nonumber
\end{align}
This theoretical trace predicts convergence to $\|e_5\| < 0.01$ in 5 iterations, consistent with the experimental observation of convergence in 3--5 steps (Figure~\ref{fig:q3_newton_quad}). The digit-doubling is clearly visible: from $\|e_3\| \approx 0.69$ to $\|e_5\| \approx 0.009$, approximately 2 decimal digits of accuracy are gained per 2 iterations in the quadratic phase.

\textbf{Global vs local convergence.} Newton's method is only \emph{locally} convergent: outside the quadratic phase, it may diverge, overshoot to a saddle point, or converge to a different local minimum. For the Rosenbrock (non-convex), the undamped Newton step can escape the Rosenbrock valley entirely for bad initialisation. The damping $\alpha = 0.22$ acts as a globalisation strategy by reducing step size until a descent condition is satisfied, analogous to backtracking line search (Armijo condition: $f(x_k - \alpha p_k) \leq f(x_k) - c_1 \alpha g_k^T p_k$ for some $c_1 \in (0, 1)$). A full line search implementation would automatically determine $\alpha$ at each iteration, typically achieving convergence from a much wider range of starting points.

\subsection{Practical Limitations and Quasi-Newton Methods}

While Newton's method is theoretically superior, it has practical limitations:
\begin{enumerate}[nosep]
\item \textbf{Hessian cost:} Computing $\nabla^2 f(x_k)$ requires $O(d^2)$ gradient evaluations or $O(d^2)$ second derivatives; for $d = 10^6$ parameters, this is $10^{12}$ operations---infeasible.
\item \textbf{Linear solve:} The system $H_k p_k = g_k$ requires $O(d^3)$ operations for dense $H_k$ (or $O(d^2)$ with conjugate gradients). This limits Newton to problems with $d \lesssim 10^4$.
\item \textbf{Non-convexity:} For non-convex $f$, $H_k$ may be indefinite, making $p_k = H_k^{-1} g_k$ an ascent direction. The Hessian regularisation $H_k + \lambda I$ (with $\lambda = 10^{-8}$ in this implementation) addresses this.
\end{enumerate}

\textbf{Quasi-Newton methods} (e.g., BFGS, L-BFGS) resolve the Hessian cost by maintaining a low-rank \emph{approximation} $B_k \approx H_k^{-1}$ updated with each gradient change:
\begin{equation}
B_{k+1} = B_k + \frac{s_k s_k^T}{s_k^T y_k} - \frac{B_k y_k y_k^T B_k}{y_k^T B_k y_k}, \quad s_k = x_{k+1} - x_k, \quad y_k = g_{k+1} - g_k.
\end{equation}
BFGS achieves superlinear (but not quadratic) convergence using only gradient information, making it far more practical for larger $d$. L-BFGS further reduces memory to $O(md)$ by keeping only the last $m$ curvature pairs.

\textbf{Newton-CG.} For very large $d$ where even $O(d^2)$ Hessian storage is too expensive, the Newton system $H_k p_k = g_k$ can be solved \emph{approximately} using conjugate gradients (CG) with a termination condition $\|H_k p_k - g_k\| \leq \eta_k \|g_k\|$ (``forcing sequence'' $\eta_k \to 0$). This \emph{inexact Newton} or Newton-CG method retains superlinear convergence as long as $\eta_k \to 0$, while the per-iteration cost is $O(d \cdot \text{CG iters})$ times the Hessian-vector product cost, which can be $O(d)$ via automatic differentiation (Pearlmutter's trick).

\textbf{Warm-starting Newton from a first-order method.} A practical strategy is to run GD or Nesterov acceleration for $k_0$ iterations to enter the basin of attraction of $x^\star$, then switch to Newton's method to exploit its quadratic convergence. For Benchmark C, the ``Nesterov phase'' provides a good initialisation: after 30 Nesterov iterations the iterate is closer to the valley floor, and a few Newton steps can then refine to high accuracy. This hybrid approach is standard in interior-point methods and trust-region solvers.

\textbf{Rosenbrock performance context.} The damped Newton step $\alpha = 0.22$ means that only 22\% of the full Newton direction is taken. This is necessary because the Hessian changes rapidly along the Rosenbrock valley. A more principled approach would use a line search (backtracking Armijo or Wolfe conditions) to select $\alpha$ automatically, which would adapt to the local curvature and likely converge faster than the fixed $\alpha = 0.22$.

\textbf{Trust region methods.} An alternative globalisation strategy is the \emph{trust region} approach: instead of computing $p = H_k^{-1}g_k$ and then scaling by $\alpha$, trust region methods restrict the step to a ball of radius $\Delta_k$:
\begin{equation}
p_k = \arg\min_{p:\|p\| \leq \Delta_k} g_k^T p + \tfrac{1}{2}p^T H_k p.
\end{equation}
The radius $\Delta_k$ is updated based on how well the quadratic model predicts the actual function decrease. Trust region methods are more robust than line search for ill-conditioned problems like the Rosenbrock, because the constraint $\|p_k\| \leq \Delta_k$ automatically prevents excessively large steps even when $H_k$ is nearly singular.

\textbf{Newton for constrained problems: interior-point methods.} When constraints are present (as in Q5), Newton's method can be applied to the KKT system directly. \emph{Interior-point} (barrier) methods add a logarithmic barrier $-\mu_t \sum_i \log(-g_i(x))$ to the objective, shrinking $\mu_t \to 0$ over iterations, and apply Newton's method to the barrier-augmented problem. This achieves $O(\sqrt{n}\log(1/\varepsilon))$ Newton iterations for linear constraints, making it the gold standard for convex programmes. For the halfspace constraint $x_1 \geq 0.5$ in Q5, the barrier would be $-\mu \log(x_1 - 0.5)$, adding a gradient term $-\mu/(x_1 - 0.5)$ that prevents the iterate from crossing the constraint boundary without any explicit projection step.

\subsection{Part A: Finite Differences and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \hat{g}_i(x;\delta) = \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
The approximation error decomposes into two competing terms:
\begin{equation}
\|\hat{g} - \nabla f\| \leq \underbrace{\frac{\delta}{2}\|H\|}_{\text{truncation error}} + \underbrace{\frac{\varepsilon_{\text{mach}} \|f(x)\|}{\delta}}_{\text{round-off error}}.
\end{equation}
The optimal step balances these terms: $\delta^\star = \sqrt{2\varepsilon_{\text{mach}}\|f\| / \|H\|}$. At $x_0 = (-1, 4)$ with $\|H\| \approx 10$ and $f(x_0) \approx 9$, this gives $\delta^\star \approx 6.3 \times 10^{-9}$.

\textbf{Central differences.} An alternative is the \emph{central} finite-difference approximation:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta},
\end{equation}
which has truncation error $O(\delta^2)$ rather than $O(\delta)$ (higher-order accuracy), at the cost of $2d$ function evaluations per gradient call (vs $d+1$ for forward FD). The optimal central-difference step is $\delta^\star_{\text{central}} = (\varepsilon_{\text{mach}}/|f|)^{1/3}$, which is larger than the forward-FD optimum. At Benchmark B's starting point: $\delta^\star_{\text{central}} = (10^{-16}/9)^{1/3} \approx 2.3 \times 10^{-6}$---four orders of magnitude larger than the forward-FD optimum $\delta^\star \approx 6\times 10^{-9}$. This means central differences are less sensitive to numerical round-off and maintain accuracy for a wider range of $\delta$. This implementation uses the more efficient forward FD, which is typically sufficient for gradient-based optimisation when $\delta$ is chosen carefully.

\begin{remark}[Automatic Differentiation vs Finite Differences]
Modern deep learning frameworks (PyTorch, JAX, TensorFlow) implement \emph{reverse-mode automatic differentiation} (backpropagation), which computes the exact gradient $\nabla f(x) \in \mathbb{R}^d$ in a single forward--backward pass costing approximately $4\times$ the cost of evaluating $f$---\emph{independent of $d$}. By contrast, forward FD requires $d+1$ function evaluations, scaling linearly in dimension. For $d = 2$ as in Benchmarks B and C, the cost ratio is $3:4$ (FD:autodiff), making them comparable. At $d = 100$, the ratio is $101:4$, and at $d = 10^6$, it is $10^6:4$---making autodiff $250{,}000\times$ cheaper than FD per gradient. This cost discrepancy is the fundamental reason automatic differentiation, not FD, is used to train neural networks. The \emph{Pearlmutter trick} (forward-mode autodiff) further enables computing Hessian--vector products $\nabla^2 f(x)\,v$ at only $2\times$ the cost of a gradient evaluation, enabling Newton-CG without explicit Hessian storage.
\end{remark}

\subsubsection{Nesterov Random Search (Zeroth-Order)}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Random directional derivative}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method uses only two function evaluations per iteration regardless of dimension $d$, making its per-iteration cost dimension-\emph{independent}. The random direction estimate relates to the Gaussian-smoothed function:
\begin{equation}
\mathbb{E}_u\!\left[\frac{f(x + \delta u) - f(x)}{\delta} u\right] = \nabla f_\delta(x), \quad f_\delta(x) = \mathbb{E}_{u \sim \mathcal{N}(0, I)}[f(x + \delta u)],
\end{equation}
where $f_\delta$ is a ``smoothed'' version of $f$ with $\|\nabla f_\delta(x) - \nabla f(x)\| = O(\delta)$. For smooth $f$, the estimate is thus approximately unbiased. The trade-off is the high variance of directional estimates: each step effectively uses a rank-1 gradient approximation in a random direction, compared to the full rank-$d$ gradient used by exact GD. For a $d$-dimensional problem, roughly $d$ NRS steps are needed to accumulate the same gradient information as one exact GD step, explaining the $O(d/k)$ effective rate per function evaluation (vs $O(1/k)$ for GD). In $d = 2$ dimensions, this is only a factor of 2 slower per evaluation, which is consistent with the observed performance.

\subsubsection{Results on Benchmark B}

\textbf{Why $\delta = 0.8$ fails on Benchmark B.} At the starting point $x_0 = (-1, 4)$, the Hessian spectral norm is $\|H(x_0)\| \approx 10$ (largest eigenvalue). The truncation error for forward FD is:
\begin{equation}
\|\hat{g}(\delta = 0.8) - \nabla f\| \approx \frac{\delta}{2}\|H\| = \frac{0.8}{2} \times 10 = 4.0, \quad \text{vs} \quad \|\hat{g}(\delta = 0.05) - \nabla f\| \approx \frac{0.05}{2} \times 10 = 0.25.
\end{equation}
With $\|\nabla f(x_0)\| \approx 4.5$, the relative error for $\delta = 0.8$ is $4.0/4.5 \approx 89\%$---the FD gradient is almost entirely dominated by truncation error and is useless for optimisation. For $\delta = 0.05$, the relative error is $0.25/4.5 \approx 5.6\%$---acceptable for optimisation purposes. This quantitative analysis confirms the figure observations.

\subsection{Part B: Derivative-Free Methods on Rosenbrock (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm \cite{nelder1965} maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$. It iteratively transforms the simplex through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): reflect the worst vertex $x_{n+1}$ through the centroid $\bar{x}$ of the remaining vertices: $x_r = \bar{x} + \alpha_r(\bar{x} - x_{n+1})$.
\item \textbf{Expansion} ($\gamma = 2$): if reflection improved beyond the best vertex, try an extended step: $x_e = \bar{x} + \gamma(\bar{x} - x_{n+1})$.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection failed to improve, contract: $x_c = \bar{x} + \rho(x_r - \bar{x})$ (outside) or $x_c = \bar{x} + \rho(x_{n+1} - \bar{x})$ (inside).
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction also failed, shrink all vertices towards the best: $x_i \gets x_1 + \sigma(x_i - x_1)$.
\end{enumerate}
No gradient information is required; only function values at simplex vertices. Initial simplex step: 0.35; iterations: 160 (total function evaluations $\leq 160 \times (n+1) = 160 \times 3 = 480$ in the worst case---comparable to GD's 160 gradient evaluations each costing $d+1 = 3$ FE, i.e.\ $160 \times 3 = 480$ FE total). In practice, Nelder--Mead uses fewer evaluations per iteration than the worst case because many steps are single-vertex reflections.

\begin{algorithm}[H]
\caption{Nelder--Mead Simplex Method}
\label{alg:nelderMead}
\begin{algorithmic}[1]
\Require Initial simplex $\{x_1, \ldots, x_{n+1}\}$, parameters $\alpha_r{=}1$, $\gamma{=}2$, $\rho{=}0.5$, $\sigma{=}0.5$, iterations $N$
\For{$k = 1, 2, \ldots, N$}
    \State Sort so $f(x_1) \leq \cdots \leq f(x_{n+1})$ \Comment{$x_{n+1}$ is worst, $x_1$ is best}
    \State $\bar{x} \gets \frac{1}{n}\sum_{i=1}^{n} x_i$ \Comment{Centroid of non-worst vertices}
    \State $x_r \gets \bar{x} + \alpha_r(\bar{x} - x_{n+1})$ \Comment{Reflect worst through centroid}
    \If{$f(x_1) \leq f(x_r) < f(x_n)$}
        \State $x_{n+1} \gets x_r$ \Comment{Accept reflection}
    \ElsIf{$f(x_r) < f(x_1)$}
        \State $x_e \gets \bar{x} + \gamma(\bar{x} - x_{n+1})$; $x_{n+1} \gets \arg\min(f(x_r), f(x_e))$ \Comment{Expand}
    \ElsIf{$f(x_r) < f(x_{n+1})$}
        \State $x_c \gets \bar{x} + \rho(x_r - \bar{x})$; \textbf{if} $f(x_c) \leq f(x_r)$: $x_{n+1} \gets x_c$ \textbf{else}: shrink \Comment{Outside contraction}
    \Else
        \State $x_c \gets \bar{x} + \rho(x_{n+1} - \bar{x})$; \textbf{if} $f(x_c) \leq f(x_{n+1})$: $x_{n+1} \gets x_c$ \textbf{else}: shrink \Comment{Inside contraction}
    \EndIf
    \State \textbf{Shrink (if needed)}: $x_i \gets x_1 + \sigma(x_i - x_1)$ for $i = 2, \ldots, n+1$
\EndFor
\end{algorithmic}
\end{algorithm}

When comparing derivative-based and derivative-free methods, a fair cost metric is the total number of \emph{function evaluations} rather than iterations, since different methods have different evaluation costs:
\begin{itemize}[nosep]
\item Exact GD: 1 gradient evaluation $\approx d+1$ function evaluations (by finite differences, or 1 backward pass).
\item FD GD: exactly $d+1 = 3$ function evaluations per iteration (for $d=2$).
\item NRS: exactly 2 function evaluations per iteration.
\end{itemize}

\begin{remark}[Nelder--Mead convergence]
Nelder--Mead has \emph{no formal convergence guarantee} for $d \geq 2$; McKinnon \cite{mckinnon1998} constructed examples in $\mathbb{R}^2$ where it converges to a non-stationary point. It often performs well in practice for low-dimensional smooth problems but can fail on non-smooth or high-dimensional functions. The convergence observed here on Rosenbrock is typical of low-dimensional smooth problems where the simplex can adapt to the local geometry. For $d > 5$, more principled methods (gradient-based or model-based surrogate optimisation) are generally preferred.
\end{remark}

Grid search evaluates the objective on a uniform $55 \times 55 = 3025$-point grid over $x_1 \in [-2, 2]$, $x_2 \in [-1, 3]$.

\caption{Grid Search (2D)}
\label{alg:gridSearch}
\begin{algorithmic}[1]
\Require Domain $[l_1, u_1] \times [l_2, u_2]$, grid resolution $N$ per axis

\textbf{Oracle complexity analysis.} To reach $\varepsilon$-optimality, the number of function evaluations scales as: $O(L/\varepsilon)$ for gradient descent (exact gradients, $d+1$ FE each), $O(dL/\varepsilon)$ for zeroth-order methods (2 FE each), and $O(N^d)$ fixed for grid search (grid-resolution-limited). The $d$-fold penalty of zeroth-order methods arises because random directional derivatives estimate $\nabla f$ with variance $\Theta(d)$ times larger than the zero-variance exact gradient. Formally, to achieve $\mathbb{E}[f(x_k)] - f^\star \leq \varepsilon$ from any initialisation, NRS requires $O(dL\|x_0-x^\star\|^2/\varepsilon)$ function evaluations. For $d = 2$, this is only $2\times$ worse per evaluation than GD, consistent with the observed results in Figure~\ref{fig:q4_per_eval}. For $d = 10^4$ (e.g., a small neural network), NRS would require $10^4\times$ more evaluations than gradient descent---making gradient-based methods essential.

\subsection{Problem Setup and Optimality Conditions}

Minimise $f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)$ subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$. The starting point is infeasible: $g(x_0) = 0.5 - 0.2 = 0.3 > 0$. The initial constraint violation is $v_0 = 0.3$, and $f(x_0) = (0.2-1)^2 + 5(4-2)^2 + \sin(0.2) = 0.64 + 20 + 0.199 = 20.84$, far from the constrained minimum $f^\star = 0.7244$ at $x^\star = (0.582, 2.0)$.

\textbf{Lagrangian and Dual Problem.} The Lagrangian of the constrained problem is:
\begin{equation}
\mathcal{L}(x, \mu) = f(x) + \mu\,(0.5 - x_1), \quad \mu \geq 0.
\end{equation}
The dual function is $d(\mu) = \min_{x \in \mathbb{R}^2} \mathcal{L}(x, \mu)$ and the Lagrange dual problem is $\max_{\mu \geq 0} d(\mu)$. Strong duality holds here (Slater's condition is satisfied since the unconstrained minimum $x^\star = (0.582, 2.0)$ satisfies $x_1 > 0.5$, giving a strictly feasible point).

\textbf{KKT Conditions.} The constrained problem can be written as $\min f(x)\ \text{s.t.}\ g(x) \leq 0$ where $g(x) = 0.5 - x_1$. The Karush--Kuhn--Tucker (KKT) conditions \cite{boyd2004} require, at a constrained minimum $x^\star$:
\begin{align}
\nabla f(x^\star) + \mu \nabla g(x^\star) &= 0 \quad \Rightarrow \quad \nabla_{x_1} f(x^\star) - \mu = 0, \label{eq:kkt1}\\
\mu\, g(x^\star) &= 0 \quad \text{(complementary slackness)}, \label{eq:kkt2}\\
g(x^\star) &\leq 0, \quad \mu \geq 0. \label{eq:kkt3}
\end{align}
For the unconstrained minimum at $x_1^\star \approx 0.582 > 0.5$, the constraint is inactive ($g < 0$), so $\mu = 0$ and the constrained minimum equals the unconstrained minimum: $x^\star = (0.582, 2.0)$, $f^\star = 0.7244$. This is confirmed by the experimental results.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

Projected GD alternates between a gradient step in $\mathbb{R}^d$ and a projection back onto the feasible set $\mathcal{X}$. For the half-space constraint $x_1 \geq 0.5$, the projection is $\Pi_\mathcal{X}(z) = (\max(0.5, z_1), z_2)^T$---a trivial $O(1)$ operation. For more complex feasible sets (e.g., the $\ell_1$ ball), projection may require solving a non-trivial sub-problem.

\begin{theorem}[Projected GD Convergence]
For $f$ that is $\mu$-strongly convex and $L$-smooth, and $\mathcal{X}$ convex and closed, projected GD with step size $\alpha = 1/L$ satisfies:
\begin{equation}
\|x_{k+1} - x^\star\|^2 \leq \left(1 - \frac{\mu}{L}\right)^k \|x_0 - x^\star\|^2, \quad x^\star = \arg\min_{x \in \mathcal{X}} f(x).
\end{equation}
The rate is identical to unconstrained GD; projection does not slow convergence when $x^\star \in \mathcal{X}$.
\end{theorem}

For Benchmark B with $\mu \approx 1.5$ (minimum Hessian eigenvalue at $x^\star$) and $L \approx 10$ (maximum), projected GD converges at rate $1 - 1.5/10 = 0.85$ per iteration. From $x_0 = (0.2, 4.0)$ with $\|x_0 - x^\star\|^2 \approx (0.58-0.2)^2 + (2-4)^2 = 0.14 + 4 \approx 4.14$, we need $k \geq \log(4.14/\varepsilon)/\log(1/0.85) \approx 42$ iterations to reach $\varepsilon = 10^{-3}$, consistent with the observed convergence in Figure~\ref{fig:q5_fval}.

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Penalty method convergence analysis.} The penalised objective $F_\lambda(x) = f(x) + \lambda \max(0, g(x))$ has smoothness constant $L_\lambda = L + \lambda / 2$ in the infeasible region (where $\max(0,g(x)) = g(x)$ contributes linear penalty gradient). The GD step size must satisfy $\alpha \leq 1/L_\lambda$: for $\lambda = 4.5$, $L_\lambda \approx 10 + 2.25 = 12.25$, requiring $\alpha \leq 0.082$---hence the $\alpha = 0.03$ choice provides a safety margin. The equilibrium constraint violation of the penalty method is approximately $v^\star \approx \|\nabla f(x^\star)\| / \lambda = 0 / \lambda = 0$ at the true minimiser where $\nabla f(x^\star) = 0$. For the under-penalised case ($\lambda = 0.15$), the gradient of $f$ at the infeasible equilibrium $x^\text{eq}$ must balance the penalty gradient $-\lambda e_1 = -0.15 e_1$: the residual violation is $v = 0.5 - x_1^\text{eq} \approx \|\partial f / \partial x_1\| / \lambda \approx 0.7/0.15 \approx 4.7$ in the worst case, which would place $x_1 \approx 0.5 - 4.7$---far below the constraint boundary, consistent with the large violation visible in Figure~\ref{fig:q5_violation}.

\textbf{Projected GD} enforces feasibility as a \emph{hard} constraint at every iteration, regardless of step size or penalty weight. This is the fundamental advantage: the iterates $x_k \in \mathcal{X}$ always. The cost is requiring a projection oracle, which is trivial for simple sets (box, halfspace) but can be expensive or intractable for complex sets (e.g., spectrahedra, $\ell_1$ balls in high dimensions require a sort-based algorithm).

\textbf{Penalty methods} trade exact feasibility for implementation simplicity: only gradient evaluations of $F_\lambda$ are needed. The weight $\lambda$ controls the feasibility-convergence trade-off. A practical approach is to use an increasing sequence $\lambda_k \to \infty$ (the \emph{exact penalty} method) or augmented Lagrangian updates to recover primal feasibility without the numerical instability of large $\lambda$.

\textbf{Augmented Lagrangian (ADMM-type).} A more principled approach replaces the pure penalty with an augmented Lagrangian:
\begin{equation}
\mathcal{L}_{\lambda}(x, \mu) = f(x) + \mu\, g(x) + \frac{\lambda}{2}\max(0, g(x))^2, \quad g(x) = 0.5 - x_1.
\end{equation}
The dual variable $\mu$ is updated as $\mu_{k+1} = \mu_k + \lambda \max(0, g(x_k))$ after each minimisation step (the \emph{dual ascent} update). Under the KKT conditions, the dual update drives $\mu_k \to \mu^\star$, the optimal Lagrange multiplier. This allows the penalty $\lambda$ to remain \emph{fixed} at a moderate value while still achieving asymptotic feasibility, avoiding the ill-conditioning that arises from large $\lambda$ in the pure penalty method. For our problem: at the constrained minimum $x^\star = (0.582, 2.0)$ where the constraint is inactive ($g(x^\star) < 0$), the dual update yields $\mu_{k+1} = \mu_k + \lambda \times 0 = \mu_k$---no dual ascent occurs, confirming that $\mu^\star = 0$ (consistent with the KKT complementary slackness condition $\mu g(x^\star) = 0$). Augmented Lagrangian methods are widely used in practice (e.g., ADMM for distributed optimisation) precisely because they combine the simplicity of penalty methods with the stability of dual variable tracking.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box/polytope with 4 vertices).

\textbf{Linear Programme.} For $f(x) = a^T x$ with $a = [1, 2]^T$, the minimum over $\mathcal{X}$ is:
\begin{equation}
\min_{x \in \mathcal{X}} a^T x = \sum_{i=1}^{2} \min_{x_i \in [l_i, u_i]} a_i x_i = a_1 l_1 + a_2 l_2 = 1 \cdot 0.5 + 2 \cdot (-5) = -9.5,
\end{equation}
attained at $x^\star = (0.5, -5)$. This exemplifies the \emph{fundamental theorem of linear programming}: the optimum of a linear objective over a bounded polytope is always attained at a vertex.

\subsection{Frank--Wolfe Algorithm}

The Frank--Wolfe algorithm \cite{frank1956}, also known as the conditional gradient method, replaces the projection step of projected gradient descent with a linear minimisation oracle (LMO) over the feasible set.

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (stays in $\mathcal{X}$)}
\EndFor
\end{algorithmic}
\end{algorithm}

For box constraints, the LP sub-problem decomposes coordinate-wise in $O(d)$:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0, \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0. \end{cases}
\end{equation}
Crucially, \emph{no projection oracle is needed}---the iterate $x_{k+1} = \beta x_k + (1-\beta) z_k$ is always feasible by convexity of $\mathcal{X}$ (both $x_k$ and $z_k$ are in $\mathcal{X}$). This makes FW appealing for problems where projection is expensive.

\textbf{Standard step schedule.} The canonical FW analysis uses the \emph{decreasing} step $\gamma_k = 2/(k+2)$ (equivalently $\beta_k = k/(k+2)$), which gives the optimal $O(1/k)$ rate. The implementation here uses a \emph{fixed} mixing parameter $\beta \in \{0.90, 0.985\}$ (equivalently constant step $\gamma = 1-\beta$), which gives the same asymptotic $O(1/k)$ rate but with larger oscillation for early iterations when the optimal $\gamma_k = 2/(k+2)$ would be larger than the fixed $\gamma$.

\textbf{Frank--Wolfe gap.} The quantity $h_k = \nabla f(x_k)^T(x_k - z_k) \geq 0$ is the \emph{Frank--Wolfe duality gap} and serves as a certificate of (near-)optimality: $h_k = 0$ iff $x_k$ is optimal.

\begin{theorem}[Frank--Wolfe Convergence Rate {\cite{jaggi2013}}]
\label{thm:fw}
For $f$ convex and $L$-smooth over compact convex $\mathcal{X}$ with $\text{diam}(\mathcal{X}) = \max_{x,y \in \mathcal{X}}\|x - y\|$, Frank--Wolfe with the decreasing step $\gamma_k = 2/(k+2)$ satisfies:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\,\text{diam}(\mathcal{X})^2}{k+2}, \quad \text{and} \quad \min_{0 \leq j \leq k} h_j \leq \frac{2L \text{diam}(\mathcal{X})^2}{k+2} = O\!\left(\frac{1}{k}\right).
\end{equation}
\end{theorem}

\noindent\textbf{Numerical verification.} For our box $\mathcal{X} = [0.5,5] \times [-5,10]$: $\text{diam}(\mathcal{X}) = \sqrt{(5-0.5)^2 + (10+5)^2} \approx \sqrt{20.25 + 225} = \sqrt{245.25} \approx 15.66$. The interior optimum problem has $f(x) = (x_1-1)^2 + (x_2-5)^2$ with $L = 2$ (Hessian $= 2I$). After $k = 140$ iterations: theoretical bound $= 2 \times 2 \times 15.66^2/(140+2) = 4 \times 245.4 / 142 \approx 6.91$. The observed gap is much smaller ($< 0.01$, Figure~\ref{fig:q6_gap}), indicating the $O(1/k)$ bound is a loose worst-case estimate for this specific problem.

For the fixed-$\beta$ implementation, the same $O(1/k)$ rate holds with a different constant depending on $\gamma = 1-\beta$.

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum.} Frank--Wolfe converges but exhibits a characteristic oscillatory pattern. The LP oracle $z_k$ selects a vertex of $\mathcal{X}$ at each iteration, jumping between extreme points as the gradient direction fluctuates near the interior optimum. The convex combination averaging smooths these jumps over time. The $O(1/k)$ rate is confirmed by the duality gap analysis (Figure~\ref{fig:q6_gap}).

\textbf{Boundary optimum.} FW is better suited to boundary optima: when $x^\star \in \partial\mathcal{X}$, the LP oracle $z_k$ consistently selects the nearest vertex, and the iterates are pulled directly towards it without oscillation. The residual gap ($f = 0.364$ vs $f^\star = 0.25$) after 140 iterations is due to the fixed $\beta = 0.93$ step size not decaying as needed. Using the standard $\gamma_k = 2/(k+2)$ schedule (shrinking steps) would improve convergence to $O(1/k)$ with a better constant.

\textbf{Why Frank--Wolfe cannot achieve $O(1/k^2)$.} Unlike gradient methods that can use Nesterov acceleration to achieve $O(1/k^2)$, Frank--Wolfe is fundamentally limited to $O(1/k)$ by a lower bound argument. The reason is that FW moves along a direction $z_k - x_k$ that points towards a vertex of $\mathcal{X}$, which is not necessarily aligned with the optimal direction. Specifically, the ``zig-zag'' behaviour observed when the optimum is in the interior (LP vertices on opposite sides of the optimum) creates irreducible oscillation. The optimal $\gamma_k = 2/(k+2)$ step-size schedule (rather than fixed $\beta$) achieves the exact $O(1/k)$ rate with constant:
\begin{equation}
f(x_k) - f^\star \leq \frac{2L\,\text{diam}(\mathcal{X})^2}{k + 2},
\end{equation}
where $\text{diam}(\mathcal{X}) = \sqrt{(5-0.5)^2 + (10-(-5))^2} \approx 15.7$ for our box constraint.

\textbf{Comparison with Projected GD.} Projected GD with Nesterov acceleration achieves $O(1/k^2)$ convergence and exact feasibility at each step. The advantage of FW is that it replaces projection with an LP oracle, which is computationally cheaper for complex polytopes where projection requires solving a non-trivial optimisation sub-problem. For the box constraint studied here, projection is trivial ($O(d)$), so projected GD is preferable. FW becomes advantageous for atomic norm constraints (e.g., nuclear norm for matrix completion), totally unimodular polytopes (e.g., flow networks), or $\ell_1$-norm balls, where LP oracles have efficient closed-form solutions.

\textbf{FW on non-convex problems.} For non-convex $f$, the FW gap $h_k = \nabla f(x_k)^T(x_k - z_k) \geq 0$ remains a valid first-order stationarity certificate: $h_k = 0$ implies $x_k$ is a local minimum over $\mathcal{X}$ (the gradient has no descent direction within $\mathcal{X}$). This makes FW applicable to non-convex constrained problems (e.g., low-rank matrix optimisation over the spectahedron), where it converges to stationary points at the same $O(1/k)$ rate. The experiments here use convex functions, so the FW gap also certifies global optimality.

\textbf{Away-step Frank--Wolfe.} A known weakness of standard FW is that it can only \emph{add} vertices to the active set (the support of $x_k$), never remove them. This causes slow convergence for interior optima because the zig-zag pattern between opposite vertices must be averaged out over many iterations. The \emph{away-step FW} variant (Lacoste-Julien \& Jaggi, 2015) additionally considers a step \emph{away} from the least useful vertex in the active set:
\begin{equation}
v_k = \arg\max_{s \in \mathcal{S}_k} \nabla f(x_k)^T s, \quad p_k = x_k - v_k \quad (\text{away direction}),
\end{equation}
where $\mathcal{S}_k$ is the current active set. By allowing away-steps, the method can drop vertices from the active set and achieves \emph{linear convergence} $O(\rho^k)$ for strongly convex $f$ over polytopes---a significant improvement over the standard $O(1/k)$ rate. For the interior optimum experiment, away-steps would reduce the LP-vertex oscillation visible in Figure~\ref{fig:q6_evolution}.

\textbf{Sparsity and Frank-Wolfe.} An important property of FW is that iterates $x_k$ are \emph{sparse} combinations of extreme points (vertices) of $\mathcal{X}$ by construction. For the $\ell_1$ ball $\mathcal{X} = \{x : \|x\|_1 \leq c\}$, the LP oracle selects $z_k = c \cdot e_{i^\star}$ where $i^\star = \arg\max_i |(\nabla f)_i|$---a sparse vertex. The iterate $x_k$ is then a convex combination of at most $k$ such vertices, guaranteeing at most $k$-sparse solutions. This makes FW a natural choice for \emph{sparse learning} problems (e.g., LASSO, Group LASSO) without requiring the proximal step that projected methods need \cite{jaggi2013}.

The principal conclusions are:

\begin{enumerate}[label=\textbf{\arabic*.}]

\item \textbf{Adaptive step-size methods} (Q1): The Polyak step size is theoretically optimal when $f^\star$ is known exactly but fails catastrophically when misspecified. For practical use without exact $f^\star$ knowledge, Adagrad (for sparse gradients) and RMSprop (for non-stationary problems) are more robust. Heavy Ball provides momentum-based acceleration without requiring $f^\star$.

\item \textbf{Momentum and acceleration} (Q2): Nesterov acceleration achieves the optimal $O(1/k^2)$ convergence rate for convex smooth functions and demonstrates a $7\times$ advantage over GD on the ill-conditioned Rosenbrock. Adam combines momentum with adaptive scaling, offering robustness to anisotropic curvature at the cost of more hyperparameters. Mini-batch SGD trades gradient noise for faster per-epoch progress; smaller batches converge faster per epoch but exhibit more oscillation near the noise floor.

\item \textbf{Newton's method} (Q3): Second-order methods achieve superlinear (quadratic) convergence by utilising full Hessian information. Newton solves the quadratic regression problem in a single step and converges in $\sim$3--5 steps on Benchmark B, consistent with the explicit quadratic convergence trace ($\|e_5\| < 0.01$ predicted from $M/(2\mu) \approx 1/3$). The extreme ill-conditioning of the Rosenbrock necessitates damping ($\alpha = 0.22$), limiting Newton to linear convergence in practice; a warm-start from 30 Nesterov iterations would provide a better initialisation for Newton's local convergence phase.

\item \textbf{Derivative-free methods} (Q4): The forward finite-difference step $\delta = 0.05$ provides high-fidelity gradient approximations (within $10^3\times$ of the approximation error floor), while $\delta = 0.8$ incurs substantial truncation error. Nesterov random search is dimension-independent per step but slow overall. Nelder--Mead navigates the Rosenbrock valley effectively without gradient information. Grid search provides global coverage but scales exponentially with dimension.

\item \textbf{Constrained optimisation} (Q5): Projected GD enforces hard constraints from the first iteration. Penalty methods provide a soft alternative requiring only gradient evaluations of the penalised objective; large $\lambda$ effectively enforces feasibility but requires smaller step sizes. The KKT analysis confirms that the unconstrained minimum $(0.582, 2.0)$ is also the constrained minimum (inactive constraint), which both methods recover.

\item \textbf{Frank--Wolfe} (Q6): FW converges at the theoretical $O(1/k)$ rate (confirmed by duality gap analysis) and handles constraints without projection, using only a coordinate-wise LP oracle on box constraints. Interior optima exhibit LP-vertex oscillation; boundary optima are handled more naturally. Smaller step fractions ($\beta = 0.985$) reduce oscillation but slow initial convergence.

\end{enumerate}

\textbf{Quantitative summary.} The key comparative numbers from this study are:
\begin{itemize}[nosep]
\item GD vs Newton on Benchmark A: Newton converges in \textbf{1 iteration} vs GD's 80 (\textbf{80$\times$ speedup}).
\item GD vs Nesterov on Benchmark C ($\kappa \approx 2508$): Nesterov reaches $f = 0.479$ vs GD's $f = 3.39$ after 150 iterations ($\approx$\textbf{7$\times$} better), with theoretical speedup $\sqrt{\kappa}/2 \approx 25\times$ in the large-$k$ limit.
\item FD $\delta = 0.05$ vs $\delta = 0.8$: $\delta = 0.8$ incurs \textbf{89\% relative gradient error} vs 5.6\% for $\delta = 0.05$---a factor of $16\times$ worse.
\item Polyak with correct $f^\star$ vs wrong $f^\star$ on Rosenbrock: \textbf{0.0094 vs 3.51} (final $f$)---a 370$\times$ gap.
\item Heavy Ball vs GD on Benchmark A ($\kappa \approx 1.117$): \textbf{2$\times$} faster convergence rate ($r^*_{\text{HB}} = 0.028$ vs $r_{\text{GD}} = 0.055$).
\end{itemize}

\textbf{Overall.} The choice of algorithm must be matched to the problem structure: use Newton's method when Hessians are available and the cost of their computation and inversion is affordable; use Nesterov acceleration for large-scale convex problems; use Adam for deep learning applications with noisy stochastic gradients; use FW for structured constrained problems where LP oracles are cheap; and use derivative-free methods (Nelder--Mead, random search) only when gradients are unavailable.

\midrule

\medskip
\noindent\textbf{Limitations and future directions.} Several extensions would strengthen the analysis:
\begin{itemize}[nosep]
\item \textbf{Larger dimensions.} The two-dimensional benchmarks limit the practical relevance of the derivative-free comparison. Repeating the NRS vs GD analysis at $d = 100$ would dramatically illustrate the $d$-fold penalty of zeroth-order methods.
\item \textbf{Stochastic regimes.} The adaptive methods (Adagrad, Adam) are primarily motivated for \emph{stochastic} settings. Re-running Q1 experiments with mini-batch gradients (batch size 32) would show the relative advantage of Adam and RMSprop over deterministic GD, which enjoys no benefit from adaptive scaling in the noiseless case.
\item \textbf{Second-order methods in higher dimensions.} At $d = 50$--$500$, the quasi-Newton (L-BFGS) vs full Newton trade-off becomes practically relevant. L-BFGS with memory $m = 20$ would match full Newton convergence on smooth problems at a fraction of the Hessian storage cost.
\item \textbf{Non-smooth objectives.} Extending Q5 to include a non-smooth penalty term (e.g., $\ell_1$ regularisation) would require the proximal gradient method and clarify when the KKT conditions need to be replaced by subdifferential optimality conditions.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Damped Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Projected Gradient Descent}]
def projected_gd(grad_fn, loss_fn, x0, alpha, projection, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        z = x - alpha * grad_fn(x)   # gradient step
        x = projection(z)             # project onto feasible set
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

# Projection onto x1 >= 0.5 halfspace:
projection_halfspace = lambda z: np.array([max(0.5, z[0]), z[1]])
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe (Conditional Gradient)}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
