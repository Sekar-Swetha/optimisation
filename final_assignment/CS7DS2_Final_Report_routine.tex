\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss, a toy neural network loss with sinusoidal perturbation, and the Rosenbrock function. Six families of methods are investigated: (1) adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball), (2) momentum and stochastic methods (Nesterov acceleration, Adam, mini-batch SGD), (3) Newton's method with local quadratic approximations, (4) derivative-free approaches (finite differences, Nesterov random search, Nelder--Mead, grid search), (5) constrained optimisation via projected gradient descent and penalty methods, and (6) the Frank--Wolfe algorithm for linear-constrained problems. Each method is evaluated in terms of convergence speed, trajectory behaviour, stability, and computational trade-offs. Key theoretical results---including the $O(1/k^2)$ rate of Nesterov acceleration, quadratic convergence of Newton's method, and the $O(1/k)$ sublinear rate of Frank--Wolfe---are related to the empirical observations throughout. Results confirm that algorithm performance is strongly governed by the local geometry of the loss landscape: condition number, curvature anisotropy, and the presence of a curved valley.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility.

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and two parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \in \mathbb{R}^2, \quad y^{(i)} = \theta^{\star T} x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2}\theta^T H\theta - b^T\theta + c,
\end{equation}
where $H = \frac{1}{m}X^TX \in \mathbb{R}^{2\times 2}$ is the Hessian (constant), $b = \frac{1}{m}X^Ty$, and $c$ is a constant. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y) = H\theta - b$. Due to observation noise, the true minimiser is $\theta_{\mathrm{OLS}} = H^{-1}b \approx [3, 4]^T$ and the irreducible minimum loss is $J^{\star} \approx 0.4826$. The condition number $\kappa(H) \approx 1.5$ (near-isotropic), making Benchmark A relatively easy for gradient methods. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H = \mathrm{diag}(2 - \sin(x_1),\ 10)$. The diagonal structure means the coordinate axes are the eigenvectors; the eigenvalues are $\lambda_1 = 2 - \sin(x_1)$ and $\lambda_2 = 10$. Along the trajectory from $x_0 = (-1, 4)$ to $x^{\star} \approx (0.582, 2)$, we have $\sin(x_1) \in [-0.841, 0.549]$, so $\lambda_1 \in [1.45, 2.84]$ and the condition number $\kappa(H) = \lambda_2/\lambda_1 = 10/\lambda_1$ ranges from $\approx 3.5$ (at the starting point) to $\approx 6.9$ (at the minimum)---moderate anisotropy. The minimum lies at approximately $(0.582, 2.0)$ with $f^{\star} \approx 0.7244$. The sinusoidal term $\sin(x_1)$ introduces mild nonlinearity, shifting the minimum away from the nominal point $(1, 2)$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is
\begin{equation}
\nabla^2 f(1,1) = \begin{bmatrix} 2 + 1200 - 400 & -400 \\ -400 & 200 \end{bmatrix} = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix},

with eigenvalues $\lambda_1 \approx 0.5$ and $\lambda_2 \approx 1001.5$, giving condition number $\kappa \approx 2003$. This extreme ill-conditioning, combined with a narrow curved valley along $x_2 = x_1^2$, makes Rosenbrock one of the most challenging standard test functions. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Theoretical Foundation for Gradient Descent}
\label{subsec:gd_theory}

The following proposition underpins the convergence analysis throughout this report. It shows that $L$-smoothness (upper curvature bound) combined with strong convexity (lower curvature bound) is the natural setting for linear convergence.

\begin{proposition}[GD Linear Convergence]
Let $f$ be $\mu$-strongly convex and $L$-smooth (i.e., $\mu I \preceq \nabla^2 f(x) \preceq LI$ for all $x$). Gradient descent with step size $\alpha = 1/L$ satisfies
\begin{equation}
f(x_{k+1}) - f^{\star} \leq \left(1 - \frac{\mu}{L}\right)\!\bigl(f(x_k) - f^{\star}\bigr).
\end{equation}
Unrolling the recursion: $f(x_k) - f^{\star} \leq (1-\mu/L)^k(f(x_0)-f^{\star}) \to 0$ at a linear (geometric) rate.
\end{proposition}
\begin{proof}[Proof sketch]
\emph{Step 1 (descent lemma from $L$-smoothness):} $f(x_{k+1}) \leq f(x_k) + \nabla f(x_k)^T(x_{k+1}-x_k) + \frac{L}{2}\|x_{k+1}-x_k\|^2$. With $x_{k+1} = x_k - \frac{1}{L}\nabla f(x_k)$, this simplifies to $f(x_{k+1}) \leq f(x_k) - \frac{1}{2L}\|\nabla f(x_k)\|^2$.

\emph{Step 2 (gradient lower bound from strong convexity / PL condition):} Strong convexity implies the Polyak--\L ojasiewicz (PL) condition $\|\nabla f(x)\|^2 \geq 2\mu(f(x)-f^{\star})$ for all $x$. (Proof: optimise $f(y) \geq f(x)+\nabla f(x)^T(y-x)+\frac{\mu}{2}\|y-x\|^2$ over $y$; the optimal $y = x - \frac{1}{\mu}\nabla f(x)$ gives $f^{\star} \leq f(x) - \frac{1}{2\mu}\|\nabla f(x)\|^2$.)

\emph{Combining:} $f(x_{k+1}) - f^{\star} \leq f(x_k) - \frac{1}{2L}\|\nabla f(x_k)\|^2 - f^{\star} \leq f(x_k)-f^{\star} - \frac{\mu}{L}(f(x_k)-f^{\star}) = (1-\mu/L)(f(x_k)-f^{\star})$.
\end{proof}

This proof reveals that \emph{any} function satisfying the PL condition (a strictly weaker condition than strong convexity, but not needing quadratic structure) inherits the same linear convergence rate for GD.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is motivated by the observation that for an $L$-smooth convex function, the sub-optimality $f(x_k) - f^{\star}$ satisfies
\begin{equation}
f(x_k) - f^{\star} \geq \frac{1}{2L}\|\nabla f(x_k)\|^2,
\end{equation}
so $\alpha_k = (f(x_k)-f^{\star})/\|\nabla f(x_k)\|^2 \leq L^{-1}$, which is within the stable range. When $f^{\star}$ is known exactly, the Polyak step guarantees a convergence rate matching the optimal first-order rate for strongly convex functions.

\textbf{Geometric interpretation:} The Polyak step can be viewed as finding the step size $\alpha_k$ such that the first-order model $f(x_k) - \alpha_k \|\nabla f(x_k)\|^2$ equals $f^{\star}$. Solving for $\alpha_k$: $\alpha_k = (f(x_k) - f^{\star})/\|\nabla f(x_k)\|^2$. This is the ``ideal'' step: it uses exactly the descent rate predicted by the gradient to ``reach'' the known minimum in one step if $f$ were linear. For nonlinear $f$, this overshoots, but on average it achieves a contraction towards $x^{\star}$. The formula is also related to the Barzilai-Borwein method which estimates $\alpha_k$ from successive gradient differences rather than from $f^{\star}$, making it applicable when $f^{\star}$ is unknown.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad adapts the step size per coordinate: for coordinate $i$, the effective rate is $\alpha_k^{(i)} = \alpha_0/\sqrt{\sum_{j=0}^k g_j^{(i)2}}$. Coordinates with consistently large gradients receive smaller steps (implicitly preconditioning by the gradient geometry), while rarely-updated coordinates retain larger steps. The cumulative sum in the denominator means the step is \emph{monotonically non-increasing}, which guarantees stability but can cause premature slow-down on non-stationary problems.

\textbf{Regret bound interpretation:} Adagrad was originally motivated by online convex optimisation, where the goal is to minimise the \emph{regret} $R_T = \sum_{t=1}^T f_t(x_t) - \min_x \sum_{t=1}^T f_t(x)$ (cumulative excess loss). The Adagrad regret bound is $R_T \leq \alpha_0 \sum_{i=1}^d \sqrt{\sum_{t=1}^T g_{t,i}^2}$, which adapts to the \emph{problem geometry}: coordinates with sparse gradients (small $\sum g^2$) incur little regret because they rarely receive bad updates. This property makes Adagrad particularly effective for sparse gradients in NLP tasks (e.g., word embeddings), where most coordinates are rarely updated and benefit from retaining large step sizes.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces Adagrad's cumulative sum with an exponential moving average (EMA) with decay $\beta=0.9$. This gives each squared-gradient an effective half-life of $1/(1-\beta) = 10$ iterations, allowing the effective step size to \emph{increase} again if gradient magnitudes decrease---the key advantage over Adagrad for non-stationary objectives.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method accumulates gradient history via the momentum term $z_k$, effectively approximating the Newton direction when gradients are consistent. On strongly convex quadratics, the optimal parameters are
\begin{equation}
\alpha^* = \frac{4}{(\sqrt{L}+\sqrt{\mu})^2}, \quad \beta^* = \left(\frac{\sqrt{L}-\sqrt{\mu}}{\sqrt{L}+\sqrt{\mu}}\right)^2,
\end{equation}
yielding convergence rate $\rho^* = 1 - 2/(\sqrt{\kappa}+1)$ vs $1-1/\kappa$ for gradient descent. The speedup is dramatic for ill-conditioned problems: for Rosenbrock with $\kappa \approx 2003$, GD's rate is $1 - 1/2003 \approx 0.9995$ while the optimal Heavy Ball rate is $1 - 2/(\sqrt{2003}+1) \approx 1 - 2/45.8 \approx 0.956$. To reduce the initial error by $100\times$: GD needs $\approx \log(0.01)/\log(0.9995) \approx 9200$ iterations, while optimal Heavy Ball needs $\approx \log(0.01)/\log(0.956) \approx 102$ iterations---a 90-fold speedup on quadratics. The empirical Heavy Ball parameters used here ($\alpha = 0.0008$, $\beta = 0.86$) are not precisely optimal but achieve similar speedup in practice.

\textbf{Key finding on Polyak divergence:} The Polyak step size formula is $\alpha_k = (f(x_k) - f^{\star}) / (\|\nabla f(x_k)\|^2 + \varepsilon)$. Setting $f^{\star} = 0$ when the true minimum is $f^{\star}_{\mathrm{true}} \approx 0.48$ means the numerator remains bounded away from zero even at the optimum. As $\|\nabla f(x_k)\| \to 0$, the denominator approaches $\varepsilon = 10^{-4}$, so $\alpha_k \to 0.48/10^{-4} = 4800$. Such large steps cause the iterate to overshoot catastrophically. This illustrates that the Polyak step is \emph{theoretically optimal} only when $f^{\star}$ is known exactly; in practice, this limits its applicability.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This ``corrective'' step allows Nesterov acceleration to anticipate momentum-induced overshoot before it occurs. The adaptive schedule $\beta_k = (k-1)/(k+2)$ gradually increases momentum from 0 to $\beta_{\max}$, which is essential for the theoretical guarantee.

\textbf{Momentum vs.\ acceleration---a critical distinction:} Heavy Ball (Q1) and Nesterov (Q2) both use momentum, but they differ in a fundamental way:
\begin{itemize}[nosep]
\item \textbf{Heavy Ball} uses gradient at current iterate $x_k$ with \emph{fixed} $\beta$. It achieves $O(1/\sqrt{\kappa})$ acceleration on quadratic functions but has no convergence guarantee for non-quadratic objectives. The fixed $\beta$ can cause oscillations near the minimum on general smooth functions.
\item \textbf{Nesterov} uses gradient at lookahead $\tilde{x}_k$ with \emph{increasing} $\beta_k$. It achieves $O(1/k^2)$ convergence for all smooth convex functions, which is \emph{provably optimal} among all first-order methods (Nesterov's lower bound). No first-order algorithm---no matter how clever---can achieve better than $O(1/k^2)$ in the worst case.
\end{itemize}
This is why we classify Heavy Ball as an ``adaptive step-size'' method (Q1) and Nesterov as an ``accelerated'' method (Q2): the acceleration label carries the specific meaning of matching the information-theoretic lower bound.

\begin{theorem}[Nesterov Acceleration Rate]
For a convex $L$-smooth function $f$, the Nesterov accelerated method with $\alpha = 1/L$ and the $\beta_k$ schedule above satisfies
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2}.
\end{equation}
This $O(1/k^2)$ rate is optimal among first-order methods for smooth convex minimisation (Nesterov's lower bound), compared to $O(1/k)$ for gradient descent.
\end{theorem}

This explains why Nesterov outperforms GD dramatically on Benchmark C: with $L \approx 800$ and 150 iterations, the theoretical bound gives $f(x_{150}) - f^{\star} \lesssim 2L\|x_0-x^{\star}\|^2/151^2 \approx 0.09$, whereas GD's $O(1/k)$ bound gives $\approx 14$. The empirical results ($f = 0.479$ vs $f = 3.388$) are consistent with this large theoretical gap.

\begin{remark}[Why the polynomial schedule achieves $O(1/k^2)$]
The schedule $\beta_k = (k-1)/(k+2)$ is designed to maintain a Lyapunov function of the form $\Phi_k = t_k^2(f(x_k) - f^{\star}) + L\|v_k - x^{\star}\|^2$ where $t_k = (k+1)/2$ and $v_k$ is an auxiliary sequence. One can show that $\Phi_k \leq \Phi_{k-1}$ at each step (the Lyapunov function is non-increasing), which gives $f(x_k) - f^{\star} \leq \Phi_k/t_k^2 \leq \Phi_0/t_k^2 = 2L\|x_0-x^{\star}\|^2/(k+1)^2$. In contrast, Heavy Ball does not maintain such a Lyapunov function for non-quadratic $f$ (the momentum parameter $\beta$ is fixed rather than increasing), which is why Heavy Ball lacks a convergence guarantee beyond the quadratic setting despite having similar momentum structure.
\end{remark}

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2$, tolerance $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scale)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction for $m$}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias correction for $v$}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines first-moment momentum (as in Heavy Ball) with second-moment adaptive scaling (as in RMSprop), plus bias corrections that prevent the early iterates from being biased towards zero. Without bias correction at $t=1$ with $\beta_1 = 0.80$, $\beta_2 = 0.999$:
\begin{equation}
m_1 = (1-\beta_1)g_1 = 0.20\,g_1, \quad v_1 = (1-\beta_2)g_1^2 = 0.001\,g_1^2,
\end{equation}
giving effective step $\alpha m_1/\sqrt{v_1} = \alpha \cdot 0.20/(0.001)^{1/2} \cdot \mathrm{sign}(g_1) = \alpha \cdot 6.32 \cdot \mathrm{sign}(g_1)$---more than $6\times$ the base learning rate, potentially causing large initial overshoots. With bias correction: $\hat{m}_1 = m_1/(1-\beta_1^1) = g_1$, $\hat{v}_1 = v_1/(1-\beta_2^1) = g_1^2$, giving step $\alpha g_1/|g_1| = \alpha \,\mathrm{sign}(g_1)$---exactly $\alpha$ times the normalised gradient, well-controlled from the very first iteration. The bias-corrected $\hat{m}_t$ and $\hat{v}_t$ always restore this proper scaling regardless of the $\beta$ values chosen.

\textbf{Why does Nesterov outperform Adam on Rosenbrock?}
On Benchmarks A and B, Nesterov and Adam reach the same final value, but on Benchmark C (Rosenbrock), Nesterov achieves $f = 0.479$ vs Adam's $f = 1.579$---a threefold difference. This gap arises from two structural properties of the Rosenbrock function:

\begin{enumerate}[nosep]
\item \textbf{Gradient direction consistency along the valley:} Once on the valley floor $x_2 \approx x_1^2$, gradient descent steps all point in roughly the same direction (along the valley toward $(1,1)$). Nesterov's accumulating momentum $z_k$ builds up in this consistent direction, effectively taking increasingly large strides. Adam's momentum ($\beta_1 = 0.80$) also accumulates, but its second moment $v_t$ with $\beta_2 = 0.999$ tracks a very long EMA of squared gradients. Near the Rosenbrock valley floor, the gradient magnitude is small but highly variable (alternating between the across-valley and along-valley components), so $\sqrt{\hat{v}_t}$ overestimates the current gradient scale, causing Adam to take excessively conservative steps.
\item \textbf{Coordinate independence vs.\ joint curvature:} Adam applies per-coordinate scaling, treating $x_1$ and $x_2$ independently. The Rosenbrock valley is curved ($x_2 = x_1^2$), coupling the two coordinates. A diagonal preconditioner (Adam) cannot capture this off-diagonal coupling, while Nesterov's directional momentum implicitly aligns with the valley direction through accumulated gradient history, effectively exploiting the low-curvature direction without needing an off-diagonal preconditioner.
\end{enumerate}

This comparison illustrates a key practical insight: momentum methods (Nesterov, Heavy Ball) excel when the gradient direction is consistent over many iterations---as in the Rosenbrock valley. Adaptive methods (Adam, RMSprop) excel when the gradient magnitudes vary strongly across coordinates---as in sparse or multi-scale problems.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $\hat{g} = \frac{1}{b}\sum_{i \in \mathcal{B}} \nabla \ell_i(\theta)$ is an unbiased estimator of $\nabla J(\theta)$ with variance $\mathrm{Var}[\hat{g}] = \sigma_g^2/b$, where $\sigma_g^2 = \mathrm{Var}[\nabla\ell_i(\theta)]$ is the per-sample gradient variance. With a constant step size $\alpha$, SGD converges to a neighbourhood of the optimum with radius proportional to $\alpha \sigma_g^2/b$ rather than to the exact optimum. The convergence of SGD on a $\mu$-strongly convex $L$-smooth function satisfies:
\begin{equation}
\mathbb{E}[\|x_k - x^{\star}\|^2] \leq \left(1 - \alpha\mu\right)^k \|x_0 - x^{\star}\|^2 + \frac{\alpha \sigma_g^2}{\mu b},
\end{equation}
showing the two competing terms: exponential convergence (first term) and an irreducible noise floor $\alpha\sigma_g^2/(\mu b)$ (second term). To reach $\epsilon$-accuracy, the optimal batch size is $b = O(\sigma_g^2/(\mu \epsilon))$, trading per-iteration cost for fewer iterations.

\textbf{Cost-efficiency of mini-batching:} An important observation is that GD and mini-batch SGD use the \emph{same number of function evaluations per epoch}: GD computes one gradient over all $m$ samples ($m$ evaluations); mini-batch SGD with batch $b$ computes $m/b$ gradients ($b$ evaluations each), totalling $m$ evaluations per epoch. The difference is that mini-batch SGD makes $m/b$ parameter updates per epoch (vs 1 for GD), each with gradient variance $\sigma_g^2/b$ (vs 0 for GD). The extra updates provide more frequent parameter corrections at the cost of noise. The optimal batch size $b^* = \sigma_g^2 / (\mu \epsilon_{\text{floor}})$ balances variance (keep $\alpha\sigma_g^2/(\mu b) \leq \epsilon$) against update frequency. For Benchmark A with $\mu \approx 0.34$, $\sigma \approx 1$, and $\epsilon_{\text{floor}} = 0.01$: $b^* \approx 1/(0.34 \times 0.01) \approx 294$---much larger than the batch sizes $\{5, 40\}$ used here, suggesting these small batches are in the ``high variance'' regime where the noise floor is the binding constraint.

This fundamental bias-variance trade-off means:
\begin{itemize}[nosep]
\item Smaller $b$: more updates per epoch ($m/b$ updates), higher gradient variance, noisier trajectories, but higher total information throughput per epoch.
\item Larger $b$: fewer updates per epoch, lower gradient variance, smoother trajectories, but less parallelism in the gradient estimation.
\end{itemize}

\textbf{Eliminating the noise floor via learning rate decay:} A constant step size $\alpha$ introduces an irreducible noise floor $\alpha\sigma_g^2/(\mu b)$. To achieve $\epsilon$-accurate convergence to $x^{\star}$, the step size must decay. Two standard schedules are:
\begin{enumerate}[nosep]
\item \emph{Polynomial decay} $\alpha_k = \alpha_0/(k+1)$: satisfies the Robbins--Monro conditions $\sum_k \alpha_k = \infty$ (ensures the iterate moves far enough) and $\sum_k \alpha_k^2 < \infty$ (ensures variance vanishes). Achieves $\mathbb{E}[f(\bar x_k) - f^{\star}] = O(1/k)$ where $\bar x_k = \frac{1}{k}\sum_{j=1}^k x_j$ is the running average.
\item \emph{Diminishing exponential} (``step decay''): $\alpha_k = \alpha_0 \gamma^{\lfloor k/T\rfloor}$ for decay factor $\gamma \in (0.1, 0.5)$ every $T$ epochs. This is the standard schedule for deep learning (e.g., reduce LR by factor 10 every 30 epochs). It achieves geometric convergence up to the floor set by the current $\alpha$, then re-establishes a new (lower) floor after each decay.
\end{enumerate}
In the experiments here, constant $\alpha = 0.06$ is used, so SGD converges to the noise floor rather than to $\theta^{\star}$. The practical recommendation is to start with a large $\alpha$ for fast initial progress (the floor does not matter early), then decay to escape the floor as convergence stalls.

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = (0.25)^4 + 4(0.25)^3(x - 0.25) = 0.00391 + 0.0625(x - 0.25).
\end{equation}
This is a linear function tangent to $g$ at $x_0$. Its minimum is at $x \to -\infty$ (unbounded), so a gradient step $x_1 = x_0 - \alpha g'(x_0) = 0.25 - 0.0625\alpha$ simply moves in the descent direction without prescribing a step size.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.00391 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton step minimises this quadratic model. Setting the derivative to zero:
\begin{equation}
0 = g'(x_0) + g''(x_0)(x - x_0) \implies x^{\text{Newton}} = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\end{equation}
The general Newton update for the $n$-dimensional case is $x_{k+1} = x_k - [\nabla^2 f(x_k)]^{-1}\nabla f(x_k)$.

\textbf{Newton iteration sequence and convergence order for $g(x) = x^4$:} Since $g'(x) = 4x^3$ and $g''(x) = 12x^2$, the Newton update is:
\begin{equation}
x_{k+1} = x_k - \frac{g'(x_k)}{g''(x_k)} = x_k - \frac{4x_k^3}{12x_k^2} = x_k - \frac{x_k}{3} = \frac{2}{3}\,x_k.
\end{equation}
Starting from $x_0 = 0.25$: $x_1 = 0.1667$, $x_2 = 0.1111$, $x_3 = 0.0741$, \ldots, converging to $x^{\star} = 0$ with ratio $x_{k+1}/x_k = 2/3$ \emph{exactly}. This is \textbf{linear} convergence, not quadratic! The reason is that $g''(x^{\star}) = g''(0) = 12 \cdot 0^2 = 0$---the Hessian vanishes at the minimum. Quadratic convergence requires $\nabla^2 f(x^{\star}) \succ 0$ (the curvature at the minimum must be positive); for $g(x) = x^4$, the minimum is a ``flat'' fourth-order zero where the local quadratic model is a poor approximation even very close to $x^{\star}$. This is an important caveat: Newton's quadratic convergence guarantee does not apply to non-strongly-convex functions.

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, damping $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Hessian damping $\lambda = 10^{-8}$ ensures $H_k + \lambda I$ is positive definite even if $\nabla^2 f$ has near-zero or negative eigenvalues, preventing undefined or backward Newton steps. The damped step size $\alpha < 1$ is a safety factor for regions where the local quadratic model is a poor approximation of the true function. Settings: GD uses 80 iterations; Newton uses 20 iterations with $\lambda = 10^{-8}$.

\textbf{On quadratic convergence:} Near a non-degenerate local minimum $x^{\star}$ (where $\nabla^2 f(x^{\star}) \succ 0$), Newton's method satisfies
\begin{equation}
\|x_{k+1} - x^{\star}\| \leq C\|x_k - x^{\star}\|^2, \quad C = \frac{M}{2\mu},
\end{equation}
where $\mu = \lambda_{\min}(\nabla^2 f(x^{\star}))$ is the strong convexity constant at $x^{\star}$ and $M$ is the Lipschitz constant of $\nabla^2 f$ in a neighbourhood of $x^{\star}$ (bounding third-order Taylor terms). This \emph{quadratic convergence} means the number of correct digits roughly doubles each iteration.

\textbf{Basin of attraction:} Quadratic convergence is guaranteed only within the ball $\|x_0 - x^{\star}\| < 1/C = 2\mu/M$. For the Rosenbrock function at $x^{\star} = (1,1)$, $\mu = \lambda_{\min}(\nabla^2 f(1,1)) \approx 0.5$ and $M$ (from third derivatives) is large due to the $100(x_2 - x_1^2)^2$ term. The basin radius $2\mu/M$ is small, explaining why Newton with full steps diverges from $x_0 = (-1, 1)$: the iterate lies well outside the basin. The damped step $\alpha = 0.22$ serves as a global safeguard, keeping the iterates in a region where the function decreases even though quadratic convergence has not yet engaged.

\textbf{Why Newton on Rosenbrock stops at $(0.183, 0.020)$ after 20 iterations:} The damped Newton step ($\alpha = 0.22$) requires more iterations to cover the large initial distance from $x_0 = (-1, 1)$ to $x^{\star} = (1, 1)$. The quadratic convergence guarantee requires the iterate to first enter a \emph{basin of attraction} around $x^{\star}$, where the Hessian is approximately constant. With $\alpha = 0.22$, the effective per-step displacement is approximately $0.22 \times \|H^{-1}\nabla f\|$; from the initial point, this means many steps are spent in the nonlinear ``outer region'' before quadratic convergence kicks in near the minimum. With a full Newton step ($\alpha = 1$) and a line search (not implemented here), the method would enter the basin within 3--5 iterations and converge in approximately 10--12 total iterations. The damped fixed-step approach is more conservative, achieving $f = 0.686$ in 20 iterations---a significant improvement over GD but not yet globally convergent in this iteration budget.

\textbf{Quasi-Newton methods (L-BFGS):} The $O(d^3)$ cost of solving the linear system $H_k p = g_k$ makes Newton's method impractical for $d \gg 1000$. Limited-memory BFGS (L-BFGS) approximates the inverse Hessian $H_k^{-1}$ using the last $m$ gradient pairs $\{(x_{k-j} - x_{k-j-1},\, \nabla f(x_{k-j}) - \nabla f(x_{k-j-1}))\}_{j=0}^{m-1}$ via the BFGS rank-2 update formula. The $H_k^{-1}$ matrix is never formed explicitly; instead, the product $H_k^{-1} g_k$ is computed in $O(md)$ via the two-loop recursion. L-BFGS with $m = 10$--$20$ achieves superlinear convergence (between linear and quadratic) on smooth problems at a cost of only $O(md)$ per iteration---making it the standard quasi-Newton method for medium-to-large-scale problems in machine learning (e.g., training Gaussian processes, logistic regression). For the Rosenbrock function, L-BFGS would achieve near-Newton convergence in $\sim 50$ iterations at $O(d)$ cost per step.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
Taylor expansion gives $f(x + \delta e_i) = f(x) + \delta \frac{\partial f}{\partial x_i}(x) + \frac{\delta^2}{2}\frac{\partial^2 f}{\partial x_i^2}(x) + O(\delta^3)$, so the truncation error is
\begin{equation}
\left|\frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| = O(\delta).
\end{equation}
Including floating-point roundoff error of magnitude $\varepsilon_{\text{mach}} \approx 10^{-16}$, the total error in the gradient component is
\begin{equation}
\text{Total error} \approx \frac{L_{ii}\delta}{2} + \frac{2\varepsilon_{\text{mach}}|f|}{\delta},
\end{equation}
minimised at $\delta^* = 2\sqrt{\varepsilon_{\text{mach}}|f|/L_{ii}} \approx \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$. The assignment tests $\delta = 0.05$ (good approximation) and $\delta = 0.8$ (poor approximation).

\textbf{Comparison: centered finite differences.} The centered (symmetric) difference formula
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta}
\end{equation}
has truncation error $O(\delta^2)$ instead of $O(\delta)$, because the first-order error terms cancel by symmetry. The total error is $L_{iii}\delta^2/6 + 2\varepsilon_{\text{mach}}|f|/\delta$, minimised at $\delta^* \approx (3\varepsilon_{\text{mach}})^{1/3} \approx 10^{-5}$, achieving total error $O(\varepsilon_{\text{mach}}^{2/3}) \approx 10^{-10}$---two orders of magnitude more accurate than forward FD at the same cost per evaluation. The cost is $2d$ function evaluations per gradient (vs $d+1$ for forward FD), a factor of $\approx 2$ overhead that is negligible for small $d$. This is why centered differences are preferred in practice whenever both positive and negative perturbations are affordable.

For $\delta = 0.05$ on Benchmark B with the exact gradient at $x_0 = (-1, 4)$ being $\nabla f(x_0) = [2(-1-1)+\cos(-1), 10(4-2)]^T = [-4+0.540, 20]^T = [-3.460, 20]^T$, the FD estimate is:
\begin{align}
\hat{g}_1 &= \frac{f(-1+0.05, 4) - f(-1, 4)}{0.05} = \frac{(0.05-1-1)^2 + 5(4-2)^2 + \sin(-0.95) - [4+20+\sin(-1)]}{0.05} \approx -3.29,\\
\hat{g}_2 &= \frac{f(-1, 4+0.05) - f(-1, 4)}{0.05} \approx 20.25.
\end{align}
The errors are $|-3.460 - (-3.29)| = 0.17$ and $|20 - 20.25| = 0.25$, small relative to the gradient magnitude $\approx 20$. For $\delta = 0.8$, the forward difference spans a much larger region, where curvature effects are large, producing errors of order 1--4 that significantly distort the gradient direction. The ratio $|\text{FD error}| / \|\nabla f\| \approx 0.01$ for $\delta=0.05$ vs $\approx 0.15$--$0.25$ for $\delta=0.8$ explains the stark difference in convergence quality seen in Figure~\ref{fig:q4_fval_B}.

\textbf{FD accuracy degradation near the minimum:} As the iterate approaches the minimum, the gradient $\nabla f(x)$ becomes small ($\|\nabla f(x^*)\| = 0$), while the truncation error $O(L\delta)$ is approximately constant. The FD approximation becomes unreliable once $\|\nabla f(x)\| \lesssim L\delta$, which happens at distance $\|x - x^*\| \lesssim \delta$ from the minimum (since $\|\nabla f(x)\| \approx L\|x - x^*\|$ for smooth $f$). Thus FD GD with step $\delta$ cannot converge closer than $\approx \delta$ to $x^*$. For $\delta = 0.05$ this corresponds to a minimum objective gap of $\approx (L\delta^2)/2 = 5 \times 0.0025/2 = 0.006$---small but non-zero. For $\delta = 0.8$: minimum gap $\approx 5 \times 0.64/2 = 1.6$, explaining why FD GD with large $\delta$ cannot converge to the true minimum on Benchmark B.

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method approximates the gradient by sampling a random direction $u_k$. The expected value of $\hat{g}_k$ is $\mathbb{E}[\hat{g}_k] \approx \frac{1}{d}\nabla f(x_k)$ (in $d=2$ dimensions), so this provides an unbiased gradient estimate \emph{up to the $1/d$ scaling factor}. The key advantage is that only \textbf{two function evaluations} are required per iteration, regardless of dimension---compared to $d+1$ evaluations for the full FD gradient. The disadvantage is high variance: only one random projection of the gradient is used, so convergence requires $O(d)$ times more iterations than with the true gradient.

\subsection{Part B: Derivative-Free Methods (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): move worst vertex $x_{\text{worst}}$ through the centroid $c$ of the remaining vertices: $x_r = c + \alpha_r(c - x_{\text{worst}})$.
\item \textbf{Expansion} ($\gamma = 2$): if $x_r$ is better than the current best, try $x_e = c + \gamma(c - x_{\text{worst}})$.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection failed, contract towards centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction also failed, contract all vertices toward the best vertex.
\end{enumerate}
No gradient or function derivative is required---only objective evaluations. The algorithm uses at most 3 function evaluations per iteration ($x_r$, $x_e$ or $x_c$). The simplex adapts its shape to the local landscape, elongating along descent directions and contracting as it approaches the minimum.

Settings: initial simplex step $= 0.35$, iterations $= 160$.

\textbf{Nelder--Mead vs damped Newton on Rosenbrock:} Nelder--Mead achieves $f \approx 0.01$ in 160 iterations while Newton with $\alpha = 0.22$ achieves $f = 0.686$ in 20 iterations. At first glance this appears to show Nelder--Mead winning, but the comparison is on different budgets (160 vs 20 iterations) and different iteration costs: each Newton step requires one gradient and one $2\times2$ Hessian solve ($O(d^3) = O(8)$ operations), while each Nelder--Mead step requires 2--3 function evaluations with no linear algebra. With a fair budget comparison of total function evaluations (Newton: 20 gradient + Hessian evaluations $\approx 80$ function-equivalent evaluations; Nelder--Mead: $\approx 400$ evaluations), Newton would still converge faster if it entered the basin of attraction. The damped step $\alpha = 0.22$ is the limiting factor: a line-search Newton would converge in $\lesssim 30$ total iterations (enter basin in $\sim 10$, then quadratic convergence to 15+ correct digits), far outperforming Nelder--Mead.

\textbf{High-dimension limitations of Nelder--Mead:} While effective in $d = 2$, Nelder--Mead degrades rapidly with dimension. In $d$ dimensions, the simplex has $d+1$ vertices and the reflection, expansion, contraction, and shrink operations each move one vertex at a time. McKinnon (1998) showed that Nelder--Mead can converge to non-stationary points in $d \geq 2$---the simplex can degenerate (become nearly flat) and stop making progress. Empirically, the method is generally considered reliable only for $d \leq 10$ and is not recommended for $d > 20$. For high-dimensional hyperparameter tuning, Bayesian optimisation (Gaussian process surrogate + acquisition function) is the standard alternative, achieving $O(d)$ scaling in the dimension.

The fundamental trade-off in derivative-free optimisation is between \emph{oracle complexity} (number of function evaluations) and \emph{solution quality}. Methods that use gradient information (even approximately) achieve $O(d)$-per-iteration scaling and polynomial convergence rates. Methods that use no gradient information either require exponentially many evaluations (grid search) or achieve slower convergence per evaluation (Nesterov RS, Nelder--Mead). In the $d=2$ case of this assignment, these differences are minor, but they become critical in high-dimensional machine learning applications.

\textbf{Practical method selection for derivative-free settings:}
\begin{itemize}[nosep]
\item \textbf{Gradient available, smooth $f$:} Use exact GD or its accelerated variants. Cost: $O(d)$ gradient computation per iteration.
\item \textbf{Gradient unavailable, $d$ small ($\leq 10$):} Nelder--Mead or FD GD with small $\delta$. Nelder--Mead is more robust to noisy function evaluations. FD GD with appropriate $\delta$ matches GD closely.
\item \textbf{Gradient unavailable, $d$ large, smooth $f$:} Nesterov random search achieves $O(dL/\epsilon)$ function evaluations to reach $\epsilon$-accuracy (vs $O(\sqrt{L/\mu}/\epsilon)$ for gradient-based methods on strongly convex problems). The $d$-fold penalty reflects the information lost by projecting the full gradient onto a single random direction.
\item \textbf{Global optimum required, $d$ very small:} Grid search guarantees global optimality to within grid resolution. For $d=2$ and $N=55$, cost is $55^2 = 3025$ evaluations; for $d=10$ with the same resolution, $55^{10} \approx 2.5 \times 10^{17}$ evaluations are required---completely infeasible.
\end{itemize}

\subsection{Problem Setup and Optimality Conditions}

Minimise $f(x_1, x_2) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1)$ subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$.

The unconstrained minimum of $f$ is at $(0.582, 2.0)$, which satisfies $x_1 = 0.582 > 0.5$---so the constraint is \emph{inactive} at the solution. The constrained minimum is therefore the same as the unconstrained minimum, $x^{\star} = (0.582, 2.0)$, $f^{\star} \approx 0.7244$.

\textbf{KKT Conditions:} For the constrained problem $\min f(x)$ s.t.\ $x_1 \geq 0.5$, the KKT conditions at $x^{\star}$ are:
\begin{align}
\nabla f(x^{\star}) + \mu^{\star} \begin{bmatrix} -1 \\ 0 \end{bmatrix} &= 0 \quad \text{(stationarity)}, \\
\mu^{\star} &\geq 0 \quad \text{(dual feasibility)}, \\
\mu^{\star}(0.5 - x_1^{\star}) &= 0 \quad \text{(complementary slackness)}.
\end{align}
Since $x_1^{\star} = 0.582 > 0.5$, complementary slackness requires $\mu^{\star} = 0$. Then stationarity reduces to $\nabla f(x^{\star}) = 0$, i.e., the unconstrained optimality condition, confirming the constraint is inactive at the solution.

\textbf{Active constraint case and shadow price:} Since the constraint $x_1 \geq 0.5$ is inactive in this problem, the analysis is straightforward. For contrast, consider a modified objective $\tilde{f}(x) = x_1^2 + (x_2-5)^2$ (the boundary case from Q6), whose unconstrained minimum $(0, 5)$ violates $x_1 \geq 0.5$. The constrained minimum is at the boundary $x_1^* = 0.5$, $x_2^* = 5$. Applying KKT stationarity: $\nabla \tilde{f}(x^*) + \mu^*(-e_1) = [2(0.5), 0]^T - \mu^*[1, 0]^T = [0,0]^T$ gives $\mu^* = 1 > 0$ (the constraint is active). The KKT multiplier $\mu^* = 1$ is the \emph{shadow price}: relaxing the constraint to $x_1 \geq 0.5 - \epsilon$ reduces the optimal value by $\mu^* \epsilon = \epsilon$, i.e., allowing a 0.1 reduction in the lower bound saves 0.1 units of objective. This interpretation guides resource allocation in practice.

\textbf{Constraint qualification:} The KKT conditions are \emph{necessary} for optimality only when a constraint qualification (CQ) holds. The Linear Independence Constraint Qualification (LICQ) requires that the gradients of the active constraints at $x^{\star}$ are linearly independent. Here the constraint is $g(x) = 0.5 - x_1 \leq 0$, with $\nabla_x g = -e_1 = (-1, 0)^T$. Since the constraint is \emph{inactive} at $x^{\star}$ ($g(x^{\star}) = 0.5 - 0.582 = -0.082 < 0$), the set of active constraints is empty and LICQ is satisfied trivially. More generally, for box constraints $l_i \leq x_i \leq u_i$, LICQ always holds since the active constraint gradients are coordinate vectors $\pm e_i$, which are always linearly independent. This is one reason box constraints are particularly well-behaved for optimisation algorithms.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For the half-space constraint $\{x : x_1 \geq 0.5\}$, the Euclidean projection is simply $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)^T$. This is a $O(1)$ operation. For more complex constraint sets (e.g., the $\ell_1$ ball, the simplex, the positive semidefinite cone), the projection can be expensive to compute, which motivates penalty and Frank--Wolfe approaches.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Penalised objective}
    \State Compute $\nabla F_{\lambda}(x_k)$: adds $-\lambda$ to $(\nabla f)_1$ if $x_1 < 0.5$, else unchanged
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

The penalty function $F_\lambda(x) = f(x) + \lambda \max(0, 0.5 - x_1)$ converts the constrained problem into an unconstrained one. As $\lambda \to \infty$, the minimiser of $F_\lambda$ converges to the constrained minimiser. The penalty gradient is
\begin{equation}
\nabla F_\lambda(x) = \nabla f(x) + \lambda \cdot \partial_x[\max(0, 0.5 - x_1)] = \nabla f(x) - \lambda \mathbf{1}_{x_1 < 0.5} e_1,
\end{equation}
where $e_1 = (1, 0)^T$. This adds a repulsive force pushing the iterate away from the infeasible region when $x_1 < 0.5$.

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\textbf{Projected GD} enforces feasibility immediately and exactly at every iteration---this is the fundamental advantage for applications where constraint violation is physically inadmissible. The projection cost is trivial here ($O(1)$ per step) but can be $O(d \log d)$ for simplex projections or $O(d^3)$ for PSD cone projections.

\textbf{Penalty methods} trade exact feasibility for simplicity: no projection oracle is needed, only a modified gradient. The weight $\lambda$ must be tuned: too small fails to enforce feasibility; too large requires a smaller step size for stability (here $\alpha$ must drop from $0.05$ to $0.03$ when $\lambda$ increases from $1.8$ to $4.5$) because the penalised objective has Lipschitz constant $L_\lambda \approx L_f + \lambda$, requiring $\alpha < 2/L_\lambda$.

\textbf{Augmented Lagrangian comparison:} The augmented Lagrangian method (ALM, also called ADMM in the dual decomposition setting) adds a dual variable $\mu$ updated as:
\begin{equation}
\mathcal{L}_\lambda(x, \mu) = f(x) + \mu(0.5 - x_1) + \frac{\lambda}{2}(0.5 - x_1)_+^2,
\end{equation}
with dual update $\mu_{k+1} = \mu_k + \lambda(0.5 - x_1^k)_+$. ALM achieves exact feasibility at a finite (not necessarily large) $\lambda$ without the instability issue, since the dual variable $\mu$ converges to the KKT multiplier $\mu^{\star} = 0$ (inactive constraint). In the present problem with an inactive constraint, all methods (PGD, penalty with sufficient $\lambda$) converge to the same unconstrained minimum, making the differences subtle; the advantage of ALM is most significant for \emph{active} constraints where the dual variable is nonzero.

\textbf{Convergence rate comparison for constrained methods:} For the box constraint $\mathcal{X} = [l, u]^d$ used in this problem, the convergence rates are:
\begin{itemize}[nosep]
\item Projected GD with $\alpha = 1/L$: $f(x_k) - f^{\star} \leq L\|x_0 - x^{\star}\|^2/(2k)$ → $O(1/k)$.
\item PGD + Nesterov acceleration: $f(x_k) - f^{\star} \leq 2L\|x_0 - x^{\star}\|^2/(k+1)^2$ → $O(1/k^2)$.
\item Frank--Wolfe: $f(x_k) - f^{\star} \leq 2LD^2/(k+2)$ → $O(1/k)$, same rate as PGD but with \emph{larger constant} $D^2 \approx 245$ vs $\|x_0 - x^{\star}\|^2 \approx 4$ here.
\end{itemize}
For the specific problem in Q5/Q6, the starting point $x_0 = (0.2, 4)$ satisfies $\|x_0 - x^{\star}\|^2 \approx (0.582-0.2)^2 + (2-4)^2 \approx 4.15$, while the box diameter $D^2 \approx 245$ makes Frank--Wolfe's constant $\approx 59\times$ larger than PGD's. This confirms that FW's advantage---projection-free operation---must overcome a substantially larger sub-optimality constant for well-conditioned box constraints.

\textbf{Summary:} For simple constraint sets with cheap projections, projected GD (optionally with Nesterov acceleration for $O(1/k^2)$) is preferred. For complex constraint sets where projections are expensive (e.g., nuclear norm ball, flow polytopes), penalty or Frank--Wolfe methods avoid the projection cost at the expense of a larger convergence constant or rate.

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box/polytope).

\textbf{Linear Programme:} For $f(x) = a^T x$ with $a = [1, 2]^T$:
\begin{equation}
\min_{x \in \mathcal{X}} x_1 + 2x_2.
\end{equation}
Since both components of $a$ are positive, minimising over the box decomposes into minimising each coordinate independently: $\min_{x_1 \in [0.5, 5]} x_1 = 0.5$ and $\min_{x_2 \in [-5, 10]} 2x_2 = 2(-5) = -10$. Therefore:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This illustrates the \textbf{fundamental theorem of linear programming}: for a linear objective over a bounded polytope, the optimum is always attained at a vertex. The interior of the feasible set is never optimal for a linear objective.

\textbf{Dual LP and strong duality:} The dual of a box-constrained LP $\min c^Tx$ s.t.\ $l \leq x \leq u$ can be written as $\max \sum_i (l_i \lambda_i^+ - u_i \lambda_i^-)$ s.t.\ $\lambda_i^+ - \lambda_i^- = c_i$, $\lambda_i^+, \lambda_i^- \geq 0$. By LP strong duality, the primal and dual optima coincide. The dual optimal variables are the KKT multipliers: for coordinate $i$ at the lower bound ($x_i^* = l_i$, occurring when $c_i > 0$), the active multiplier is $\lambda_i^+ = c_i \geq 0$; for coordinate $i$ at the upper bound ($x_i^* = u_i$, occurring when $c_i < 0$), the active multiplier is $\lambda_i^- = -c_i \geq 0$. In our problem: $c = (1, 2)^T$ with both components positive, so both coordinates are at their lower bounds ($x_1^* = 0.5$, $x_2^* = -5$) with dual variables $\lambda_1^+ = 1$, $\lambda_2^+ = 2$. The dual objective value is $0.5 \times 1 + (-5) \times 2 = -9.5 = f^{\star}$, confirming strong duality.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient Method)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (step towards $z_k$)}
\EndFor
\end{algorithmic}
\end{algorithm}

The Frank--Wolfe update replaces the gradient step and projection of projected GD with a linear minimisation over $\mathcal{X}$. For a box constraint, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations. Since $x_{k+1}$ is a convex combination of $x_k \in \mathcal{X}$ and $z_k \in \mathcal{X}$, all iterates automatically remain in $\mathcal{X}$---\textbf{no explicit projection is ever needed}.

The \textbf{Frank--Wolfe gap} $g(x_k) = \nabla f(x_k)^T(x_k - z_k) \geq 0$ serves as both a measure of sub-optimality and a stopping criterion: $g(x_k) = 0$ implies $x_k$ is optimal (since the linear relaxation has no better point in $\mathcal{X}$).

\begin{proposition}[FW Gap as Sub-Optimality Certificate]
For any convex $f$ and $x_k \in \mathcal{X}$, the Frank--Wolfe gap satisfies
\begin{equation}
f(x_k) - f^{\star} \leq g(x_k) = \nabla f(x_k)^T(x_k - z_k).
\end{equation}
\end{proposition}
\begin{proof}
By convexity of $f$: $f^{\star} = f(x^{\star}) \geq f(x_k) + \nabla f(x_k)^T(x^{\star} - x_k)$. Therefore $f(x_k) - f^{\star} \leq \nabla f(x_k)^T(x_k - x^{\star})$. Since $z_k = \arg\min_{s \in \mathcal{X}} \nabla f(x_k)^T s$ and $x^{\star} \in \mathcal{X}$, we have $\nabla f(x_k)^T z_k \leq \nabla f(x_k)^T x^{\star}$, so $\nabla f(x_k)^T(x_k - z_k) \geq \nabla f(x_k)^T(x_k - x^{\star}) \geq f(x_k) - f^{\star}$.
\end{proof}
This means the FW gap is a \emph{computable upper bound on the sub-optimality} at each iteration, providing a rigorous stopping criterion without knowing $f^{\star}$. Furthermore, $g(x_k) = 0$ if and only if $x_k$ is a KKT point of the constrained problem: $g(x_k) = \max_{s \in \mathcal{X}} \nabla f(x_k)^T(x_k - s) = 0$ iff $\nabla f(x_k)^T(x_k - s) \leq 0$ for all $s \in \mathcal{X}$, which (for convex $f$ and $\mathcal{X}$) is equivalent to the first-order optimality (variational inequality) condition $\nabla f(x_k)^T(x - x_k) \geq 0$ for all $x \in \mathcal{X}$.  In our interior-optimum experiment, $g(x_{180}) = 0.764$ while $f(x_{180}) - f^{\star} = 0.0089 - 0 = 0.0089$, so the gap bound is loose by a factor of $\approx 86$---typical for Frank--Wolfe near an interior optimum where the oscillating $z_k$ prevents a tight linearisation.

\begin{theorem}[Frank--Wolfe Convergence]
For a convex $L$-smooth function $f$ with $\mathrm{diam}(\mathcal{X}) = D$, using the standard step size $\gamma_k = 2/(k+2)$ (not fixed $\beta$):
\begin{equation}
f(x_k) - f^{\star} \leq \frac{2LD^2}{k+2}.
\end{equation}
This $O(1/k)$ rate is sublinear but projection-free, and each iteration costs only an LMO ($O(d)$ for box constraints) rather than a projection which can be expensive for complex constraint sets.
\end{theorem}

\textbf{Note on the fixed-$\beta$ variant used here:} Rather than the standard $\gamma_k = 2/(k+2)$ schedule, this experiment uses a fixed $\beta$ (so the step towards $z_k$ is constant at $1 - \beta$). This departs from the theoretically optimal schedule and can cause persistent oscillation, but allows a controlled comparison of large vs small step sizes.

For $\mathcal{X} = [0.5,5] \times [-5,10]$: $D = \sqrt{(5-0.5)^2 + (10-(-5))^2} = \sqrt{20.25 + 225} = \sqrt{245.25} \approx 15.66$.

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits the characteristic ``zig-zag'' pattern because the LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$. To see why, consider the state at $k = 180$: $x_{180} \approx (1.066, 4.932)$, giving $\nabla f = [0.132, -0.136]^T$ and $z_{180} = (0.5, 10)$. The next iterate is $x_{181} = 0.90(1.066, 4.932) + 0.10(0.5, 10) \approx (1.009, 5.439)$. Now $\nabla f(x_{181}) = [0.019, 0.878]^T$ (both positive), so the LMO flips: $z_{181} = (0.5, -5)$. This gives $x_{182} \approx 0.90(1.009, 5.439) + 0.10(0.5, -5) \approx (0.959, 4.395)$. The $x_2$ component oscillates between $\approx 4.4$ and $\approx 5.4$---a persistent zig-zag caused by the LMO alternating between $z_k = (0.5, 10)$ and $z_k = (0.5, -5)$ as the gradient's $x_2$ component alternates in sign near $x_2^* = 5$. The standard decaying schedule $\gamma_k = 2/(k+2) \to 0$ suppresses these oscillations, but the fixed $\beta = 0.90$ maintains them indefinitely. The $O(1/k)$ convergence rate of the standard (non-fixed-$\beta$) variant would be $f(x_k) - f^{\star} \leq 2LD^2/(k+2)$; empirically, the fixed-$\beta$ variant performs similarly but with more variance.

\textbf{Boundary optimum:} Frank--Wolfe is well-suited to boundary optima because the LMO consistently selects the vertex nearest to the true constrained optimum. The iterate $x_k$ approaches the boundary through repeated convex combinations with $z_k = (0.5, -5)$ (the LMO vertex). The slow convergence ($f = 0.364$ vs $f^{\star} = 0.25$ after 140 iterations) reflects the limitation of fixed $\beta$: the step size $1-\beta = 0.07$ is too large to achieve fine convergence near the boundary. Using $\gamma_k = 2/(k+2)$ (decaying step) would achieve $f(x_{140}) - 0.25 \leq 2LD^2/142 \approx 2 \times 2 \times 245.25/142 \approx 6.9$---worse than empirical due to the loose bound, but the principle confirms that a decaying schedule is needed for near-optimal convergence on boundary problems.

\textbf{Frank--Wolfe gap as a diagnostic:} For the interior case ($f = (x_1-1)^2+(x_2-5)^2$) at the final iterate $x_{180} \approx (1.066, 4.932)$ with $\beta = 0.90$:
\begin{align}
\nabla f(x_{180}) &= [2(1.066-1),\; 2(4.932-5)]^T = [0.132,\; -0.136]^T, \\
z_{180} &= \arg\min_{s \in \mathcal{X}} \nabla f^T s = (0.5,\; 10) \quad \text{(lower bound for positive } g_1 \text{, upper bound for negative } g_2\text{)}, \\
g(x_{180}) &= \nabla f(x_{180})^T(x_{180} - z_{180}) = 0.132 \times (1.066 - 0.5) + (-0.136)\times(4.932 - 10) = 0.075 + 0.689 = 0.764.
\end{align}
A positive FW gap of $0.764$ confirms the algorithm has \emph{not} converged: the direction $z_{180} - x_{180}$ still provides significant descent (the FW gap bounds the sub-optimality as $f(x_k) - f^{\star} \leq g(x_k)$ for convex $f$). This confirms that 180 iterations with fixed $\beta = 0.90$ is insufficient for tight convergence; more iterations or a decaying step size are needed. For comparison, the standard $\gamma_k = 2/(k+2)$ schedule would give $\gamma_{180} = 2/182 \approx 0.011$---much smaller than $1 - \beta = 0.10$---resulting in finer updates near the optimum and less oscillation.

\textbf{Comparison:} Both interior and boundary cases benefit from Frank--Wolfe's projection-free property. For complex constraint sets where projections are expensive (e.g., nuclear norm balls, flow polytopes), Frank--Wolfe is the method of choice. For box constraints as here, projected gradient descent achieves $O(1/k^2)$ with Nesterov acceleration vs $O(1/k)$ for Frank--Wolfe, so the projection-free advantage does not outweigh the slower convergence rate for simple constraint sets. The fundamental trade-off is: Frank--Wolfe trades convergence rate ($O(1/k)$ vs $O(1/k^2)$) for oracle cost (LMO at $O(d)$ vs projection at $O(d \log d)$ or higher).

\label{sec:conclusion}

This report has studied 14 optimisation algorithms across six categories, applied to three benchmark problems of increasing difficulty (isotropic quadratic, mildly anisotropic nonlinear, and highly ill-conditioned nonlinear). The following overarching conclusions emerge from the experiments.

\subsection{Effect of Problem Structure on Algorithm Performance}

The condition number $\kappa$ is the primary determinant of difficulty for gradient-based methods. On Benchmark A ($\kappa \approx 1.5$), all stable methods converge in few iterations and the differences are negligible. On Benchmark C ($\kappa \approx 2003$), the gap between methods is enormous: Newton's method (which implicitly inverts the condition number through $H^{-1}g$) achieves $f = 0.686$ in 20 iterations, while gradient descent achieves only $f = 3.73$ in 80 iterations. Nesterov acceleration reduces the $\kappa$-dependence from $O(\kappa)$ to $O(\sqrt{\kappa})$ in the linear convergence exponent, explaining its strong performance on Benchmark C.

\subsection{Computational Cost vs Convergence Rate}

Table~\ref{tab:cost} summarises the per-iteration computational cost of each method, which must be weighed against its convergence rate. Newton's method illustrates the fundamental trade-off: it converges in $O(\log(1/\epsilon))$ iterations (quadratic convergence) but requires $O(d^3)$ work per iteration for the Hessian solve, making it impractical for deep learning where $d \sim 10^8$.

\subsection{Information Content per Iteration}

A unifying perspective on all methods studied is the amount of curvature information they use per iteration:

\begin{itemize}[nosep]
\item \textbf{GD, Nesterov, Heavy Ball} (1 gradient = $d$ directional floats): No curvature information; the gradient captures only the local slope. These methods require $O(\kappa)$ to $O(\sqrt{\kappa})$ iterations for strongly convex problems.
\item \textbf{Adagrad, RMSprop, Adam} (1 gradient + diagonal curvature estimate = $2d$ floats): Per-coordinate curvature estimated from squared gradients. This provides a diagonal preconditioner, effectively normalising the gradient by coordinate scale. Benefits most when coordinate scales differ vastly (sparse features, multi-scale problems).
\item \textbf{Newton} (1 gradient + full Hessian = $d + d^2$ floats): Full curvature information; the Hessian $\nabla^2 f$ captures both diagonal and off-diagonal coupling. Transforms the $\kappa$-dependent problem into a $\kappa$-independent one: the Newton step $H^{-1}g$ solves the optimality condition exactly at each iteration for quadratic $f$.
\item \textbf{Derivative-free methods} ($\leq 3$ function values = 3 floats): Only objective values; no directional derivative information. Must reconstruct gradient geometry from function differences, requiring $O(d)$ more evaluations.
\end{itemize}

The convergence rate hierarchy (Newton $\gg$ Nesterov $>$ GD $>$ derivative-free) directly mirrors this information hierarchy. Methods that use more curvature information converge faster but at higher per-iteration cost.

\subsection{Adaptive Methods: A Hierarchy of Curvature Awareness}

Table~\ref{tab:summary} summarises the performance hierarchy observed. Methods can be ranked by how well they approximate the ideal preconditioner (the Hessian inverse):

\begin{enumerate}[nosep]
\item \textbf{Newton} (exact Hessian inverse): best convergence, $O(d^3)$ cost per step.
\item \textbf{Adam/Adagrad/RMSprop} (diagonal curvature): adapts per-coordinate, moderate overhead.
\item \textbf{Nesterov/Heavy Ball} (momentum only, no curvature): improves GD by $O(\sqrt{\kappa})$.
\item \textbf{GD} (isotropic step): simplest, slowest on ill-conditioned problems.
\item \textbf{Derivative-free} (no gradient info): lowest information per iteration, highest function-evaluation cost.
\end{enumerate}

\subsection{Cross-Question Synthesis: Rosenbrock Benchmark}

Benchmark C (Rosenbrock, $\kappa \approx 2003$) provides the most informative comparison across all six questions, as every method struggles with the curved, highly ill-conditioned valley. Table~\ref{tab:summary} shows a clear hierarchy of performance:

\begin{enumerate}[nosep]
\item \textbf{Methods using exact $f^{\star}$ (Q1 Polyak, 120 it.)}: $f = 0.009$---best result, but requires knowledge of the global minimum.
\item \textbf{Methods using Hessian curvature (Q3 Newton, 20 it.)}: $f = 0.686$---quadratic convergence enables efficient progress even in 20 steps; L-BFGS would do better still.
\item \textbf{Derivative-free (Q4 Nelder--Mead, 160 it.)}: $f \approx 0.01$---remarkably competitive, exploiting the smooth valley structure without gradients.
\item \textbf{First-order with acceleration (Q2 Nesterov, 150 it.)}: $f = 0.479$---$O(1/k^2)$ rate overcomes the condition number through momentum.
\item \textbf{Adaptive gradient (Q1 Heavy Ball, 120 it.)}: $f = 1.40$---momentum alone (without lookahead) achieves solid acceleration.
\item \textbf{Adaptive second moment (Q2 Adam, 150 it.)}: $f = 1.58$---per-coordinate scaling cannot capture the off-diagonal coupling of the Rosenbrock valley.
\item \textbf{Standard gradient (Q1 GD, 120 it.)}: $f = 3.51$---linear convergence at rate $1 - 1/\kappa \approx 0.9995$ is far too slow for 120 iterations.
\end{enumerate}

The progression from GD to Newton to derivative-free methods on Rosenbrock illustrates a central lesson: \emph{exploiting problem structure} (whether through $f^{\star}$, Hessian curvature, or simplex geometry) is far more important than the specific algorithmic family for difficult problems. For easy problems (Benchmark A, $\kappa \approx 1.5$), all methods reach the same final value within 150 iterations and the structural differences are irrelevant.

\subsection{Key Practical Takeaways}

\begin{itemize}[nosep]
\item \textbf{When $f^{\star}$ is known:} The Polyak step size is theoretically optimal and achieves the best Rosenbrock convergence (0.009 vs 0.479 for Nesterov). In practice, $f^{\star}$ is rarely known, making Polyak impractical.
\item \textbf{For smooth strongly convex problems:} Heavy Ball or Nesterov with tuned step sizes provide the best first-order convergence. Nesterov's $O(1/k^2)$ rate is provably optimal.
\item \textbf{For highly nonlinear/ill-conditioned problems:} Newton's method (or quasi-Newton approximations like L-BFGS, not studied here) dramatically outperforms all first-order methods, at the cost of Hessian computation/inversion.
\item \textbf{For constrained problems:} Projected GD enforces feasibility exactly and efficiently when the projection is cheap. Penalty methods offer a gradient-only alternative but require tuning $\lambda$ and may need a decreasing step size. Frank--Wolfe avoids projection entirely, at the cost of a slower $O(1/k)$ rate.
\item \textbf{For derivative-free settings:} Nelder--Mead is effective in low dimensions (reaches $f \approx 0.01$ on Rosenbrock in 160 iterations), but has no convergence guarantee in general. Grid search guarantees global optimality within the grid resolution but requires exponentially many evaluations in high dimensions.
\item \textbf{Adam's practical dominance in deep learning} stems from its combination of momentum (faster initial progress) and per-coordinate adaptation (robustness to varying gradient scales across layers), even though its theoretical worst-case rates match only $O(1/\sqrt{k})$.
\end{itemize}

\subsection{Limitations and Extensions}

This study has several limitations that point towards important extensions:

\begin{enumerate}[nosep]
\item \textbf{Low dimensionality:} All benchmarks use $d = 2$ parameters. In modern deep learning, $d \sim 10^6$--$10^9$, making Newton's method completely infeasible ($O(d^3)$ cost) and changing the relative performance of Nesterov vs Adam (in high dimensions, Adam's per-coordinate adaptation is more valuable than in $d=2$ where the two-coordinate gradients have similar scales).
\item \textbf{No line search:} The step sizes used here are fixed hyperparameters tuned manually. Line search (Armijo/Wolfe conditions) would automatically choose the step size, removing the need for manual tuning and often accelerating convergence (especially for Newton's method where a line search would guarantee global convergence by avoiding the basin-of-attraction constraint).
\item \textbf{Deterministic benchmarks:} Benchmarks A, B, C are deterministic (given the fixed seed). In stochastic deep learning, the noise floor of SGD and the stability of Adam are much more critical---the constant-step-size noise floor can be large enough to prevent convergence to a good solution.
\item \textbf{No second-order stochastic methods:} Methods like K-FAC (Kronecker-Factored Approximate Curvature) and AdaHessian approximate the Hessian in a scalable way for deep networks, bridging the gap between Adam ($O(d)$ curvature) and Newton ($O(d^2)$--$O(d^3)$ curvature). These are active research areas not covered in this study.
\end{enumerate}

\subsection{Relevance to Data Science and Machine Learning}

The six categories of methods studied in this report directly underpin modern machine learning practice:

\begin{itemize}[nosep]
\item \textbf{Adaptive step-size and momentum methods (Q1, Q2):} Adam and its variants (AdamW, NAdam) are the default optimiser for training deep neural networks. The theoretical understanding developed here---bias correction, momentum accumulation, per-coordinate scaling---directly explains why Adam converges reliably on large models with heterogeneous gradient scales across layers. Nesterov acceleration forms the theoretical basis for the fast convergence of modern training schedules (cosine annealing combined with momentum).

\item \textbf{Newton's method (Q3):} In small-to-medium scale problems (logistic regression, Gaussian processes), Newton and quasi-Newton methods (L-BFGS, which approximates $H^{-1}$ using the last $m$ gradient pairs) are preferred due to their superlinear convergence. The experiment confirms that Newton's method requires only a fraction of the iterations of GD, justifying the $O(d^3)$ overhead for low-dimensional models.

\item \textbf{Derivative-free methods (Q4):} Hyperparameter optimisation in machine learning (tuning learning rates, architecture choices) often requires evaluating the objective (validation loss after full training) without access to derivatives. Bayesian optimisation (not studied here) is the practical alternative for $d \leq 20$, but Nelder--Mead and random search remain competitive baselines.

\item \textbf{Constrained optimisation (Q5):} Many machine learning problems are naturally constrained: LASSO regression adds an $\ell_1$ norm constraint on $\theta$; support vector machines optimise over the positive orthant; probability simplex constraints appear in attention mechanisms. Projected gradient descent is the standard approach when the projection is cheap; penalty methods are used when the constraint is soft.

\item \textbf{Frank--Wolfe (Q6):} The conditional gradient method is widely used in structured prediction (sequence models), matrix completion (nuclear norm minimisation), and sparse learning (Lasso), where the feasible set has a special structure enabling efficient LMO computation. The $O(1/k)$ convergence rate is the price paid for projection-free operation.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
