\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C). Six categories of methods are investigated: (1)~adaptive step-size methods --- Polyak, Adagrad, RMSprop, and Heavy~Ball --- compared over 120 iterations; (2)~momentum-based and stochastic methods --- Nesterov acceleration, Adam, and mini-batch SGD; (3)~Newton's method with Hessian-based local quadratic approximation; (4)~derivative-free approaches --- finite differences, Nesterov random search, Nelder--Mead simplex, and grid search; (5)~constrained optimisation via projected gradient descent and penalty methods; and (6)~the Frank--Wolfe algorithm for linear-constrained problems. Key findings include: the Polyak step size diverges when $f^{\star}$ is misspecified but achieves fastest convergence on Rosenbrock when $f^{\star}=0$ is exact ($f=0.009$, best among Q1 methods); Nesterov acceleration achieves $f=0.479$ on Rosenbrock versus $f=3.388$ for gradient descent, demonstrating its $O(1/k^2)$ rate advantage; Newton's method solves the quadratic regression problem in a single iteration via the OLS closed form; Nelder--Mead converges to the Rosenbrock minimum ($f\approx 0$) in 160 iterations without any gradient information; and projected gradient descent enforces feasibility from iteration~1 with zero constraint violation, whereas penalty methods achieve approximate feasibility controlled by the penalty weight. Theoretical convergence rates are stated formally for gradient descent ($O(\rho^k)$), Nesterov ($O(1/k^2)$), and Newton (quadratic). The Frank--Wolfe duality gap is presented as a computable convergence certificate. Each method is evaluated with respect to convergence speed, trajectory behaviour, theoretical guarantees, and computational trade-offs.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. This section establishes the mathematical properties of each benchmark problem that govern how different optimisation algorithms behave.

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2),\quad y^{(i)} = (\theta^{\star})^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a \emph{strongly convex quadratic} with constant Hessian $H = \frac{1}{m}X^TX \succ 0$. The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$ and the exact minimiser is the OLS solution $\theta^{\star}_{\mathrm{OLS}} = (X^TX)^{-1}X^Ty$. Due to observation noise $\varepsilon^{(i)} \sim \mathcal{N}(0,1)$, the irreducible loss floor is $J^{\star} = \frac{\sigma^2}{2} \approx 0.4826$ (not zero). The Hessian eigenvalues are approximately $\lambda_{\min} \approx 0.48$ and $\lambda_{\max} \approx 1.52$, giving a condition number $\kappa = \lambda_{\max}/\lambda_{\min} \approx 3.2$, making this a \emph{well-conditioned} problem. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^T$ and the Hessian is $H(x) = \mathrm{diag}(2 - \sin(x_1),\ 10)$. The Hessian eigenvalues range over $\lambda_1 \in [1, 3]$ (since $\sin(x_1) \in [-1,1]$) and $\lambda_2 = 10$, giving a varying condition number in the range $\kappa \in [10/3, 10/1] \approx [3.3, 10]$. The sinusoidal term $\sin(x_1)$ introduces mild nonlinearity in the first coordinate.

The unconstrained minimum satisfies $\nabla f = 0$, i.e., $10(x_2 - 2) = 0 \Rightarrow x_2^* = 2$ and $2(x_1 - 1) + \cos(x_1^*) = 0$, which gives $x_1^* \approx 0.5824$ by Newton's method applied to the scalar equation. The minimum value is $f^* \approx (0.5824-1)^2 + 0 + \sin(0.5824) = 0.1744 + 0.5499 \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^* = 0$. The Hessian is:
\begin{equation}
H(x) = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum $(1,1)$: $H = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}$. The eigenvalues are $\lambda_{1,2} = \frac{1002 \pm \sqrt{1002^2 - 4 \cdot 400}}{2} \approx 1001.6$ and $0.4$, giving condition number $\kappa \approx 2504$. For vanilla GD, this implies a convergence factor $\rho = (\kappa-1)/(\kappa+1) \approx 0.9992$ --- meaning GD loses less than $0.1\%$ of error per step, requiring $\sim 11{,}500$ iterations to reduce error by $10^{-4}$. The eigenvector corresponding to $\lambda_{\min} \approx 0.4$ points approximately along the valley floor $(x_2 = x_1^2)$, and the eigenvector for $\lambda_{\max} \approx 1001.6$ points nearly perpendicularly. This extreme anisotropy means GD is pulled transversely by large curvature and advances only marginally along the valley. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}

For constrained optimisation problems (Questions~5 and~6):
\begin{equation}
\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a convex polytope (box), which makes projection and LP subproblems tractable in closed form.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Motivation and Theoretical Background}

Vanilla gradient descent with constant step size $\alpha$ has linear convergence rate on $\mu$-strongly convex, $L$-smooth functions:
\begin{equation}
f(x_{k+1}) - f^* \leq \left(1 - \frac{2\alpha\mu L}{\mu + L}\right)\,(f(x_k) - f^*),
\end{equation}
with the optimal fixed step $\alpha^* = 2/(\mu + L)$ yielding contraction factor $\rho = (\kappa - 1)/(\kappa + 1)$ where $\kappa = L/\mu$. When $\kappa$ is large (e.g., $\kappa \approx 2504$ for Rosenbrock), $\rho \approx 1 - 2/\kappa$, and convergence is extremely slow. Adaptive step-size methods attempt to overcome this limitation by automatically adjusting the learning rate based on gradient history.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Constant Step-Size Gradient Descent (Baseline)}
\label{alg:gd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, regularisation $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step minimises the upper bound $f(x_k) - f^* \leq \frac{\|x_k - x^*\|^2}{\alpha_k}$ with respect to $\alpha_k$, producing the step that most aggressively reduces the distance to the optimal set under a linear model. For subgradient methods, the Polyak step provides convergence guarantees \emph{without} requiring knowledge of $L$ or $\mu$.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, regularisation $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad implicitly builds a diagonal preconditioner: the effective step in coordinate $i$ is $\alpha_0 / \sqrt{\sum_{j=0}^{k} g_{j,i}^2}$. This provides large steps when historical gradients are small (infrequently updated features) and small steps when gradients are large. The disadvantage is that $G_k$ grows monotonically, so the effective step eventually vanishes, stalling learning on non-stationary problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay $\beta \in (0,1)$, regularisation $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop resolves Adagrad's diminishing step problem via exponential forgetting: $v_k$ tracks a weighted average of recent squared gradients, so the effective step size remains bounded away from zero. With $\beta = 0.9$, the memory horizon is approximately $1/(1-\beta) = 10$ iterations.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball update can be rewritten as $x_{k+1} = x_k - \alpha \nabla f(x_k) + \beta(x_k - x_{k-1})$, revealing it as a second-order difference equation with a ``memory'' term. On strongly convex quadratics, the optimal Heavy Ball parameters give an accelerated convergence rate with contraction factor $\rho_{\mathrm{HB}} = (\sqrt{\kappa} - 1)/(\sqrt{\kappa} + 1) \ll (\kappa-1)/(\kappa+1) = \rho_{\mathrm{GD}}$, a substantial improvement over plain GD. However, this optimality does not extend to general nonconvex functions.

\subsection{Comparative Analysis and Discussion}

\begin{itemize}[nosep]
\item \textbf{Polyak divergence mechanism (formal):} When $f^{\star}_{\text{specified}} < f^{\star}_{\text{true}}$ (as occurs here for Benchmarks~A and~B where $f^{\star}_{\text{specified}}=0$ but $f^{\star}_{\text{true}}>0$), the Polyak step can diverge. Near the true minimum where $x_k \approx x^*$ and $\nabla f(x_k) \approx 0$:
\[
\alpha_k = \frac{f(x_k) - 0}{\|\nabla f(x_k)\|^2 + \varepsilon} \approx \frac{f^*_{\text{true}}}{\|\nabla f(x_k)\|^2} \to \infty \quad \text{as } \nabla f(x_k) \to 0.
\]
The update becomes $x_{k+1} = x_k - \alpha_k \nabla f(x_k) \approx x_k - \frac{f^*_\text{true}}{\|\nabla f\|^2}\nabla f$, taking a huge step away from $x^*$. This is precisely the divergence observed in Figures~\ref{fig:q1_fval_A} and~\ref{fig:q1_fval_B}.
\item \textbf{Adagrad vs RMSprop:} Both apply per-coordinate scaling, but Adagrad's monotone accumulation $G_k = \sum g_j^2$ eventually stalls convergence, while RMSprop's exponential window $v_k = \beta v_{k-1} + (1-\beta)g_k^2$ maintains adaptivity throughout. On well-conditioned problems (Benchmarks A and B), both perform similarly. On ill-conditioned Rosenbrock, the small base rate required for RMSprop stability removes its advantage.
\item \textbf{Heavy Ball on Rosenbrock:} Momentum $\beta=0.86$ builds velocity along the valley's longitudinal direction (along $x_2 \approx x_1^2$), achieving $f=1.40$ versus GD's $f=3.51$. This demonstrates that momentum is highly effective when the gradient direction is consistent over many iterations.
\item \textbf{Practical recommendation:} For problems where $f^{\star}$ is unavailable (most applications), Heavy~Ball or RMSprop provide the best balance of speed and robustness. Adagrad is preferred when coordinates have highly heterogeneous gradient magnitudes (e.g., sparse features in NLP). Polyak is most powerful when $f^{\star}$ is accurately known.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Theoretical Motivation}

Gradient descent achieves a convergence rate of $O(1/k)$ for convex, $L$-smooth objectives. Nesterov (1983) showed that this rate can be improved to $O(1/k^2)$ using an accelerated scheme, which is \emph{optimal} among first-order methods for smooth convex optimisation (matching the lower bound of Nemirovsky and Yudin). This section implements and compares Nesterov acceleration and Adam, along with mini-batch SGD.

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q_theory_convergence_rates.pdf}
\caption{Theoretical convergence rates illustration. Left: number of iterations required to achieve $\varepsilon = 10^{-4}$ accuracy as a function of condition number $\kappa$. Gradient descent requires $O(\kappa)$ iterations; the optimal momentum method requires only $O(\sqrt{\kappa})$, a substantial advantage for large $\kappa$. Vertical lines mark the condition numbers of the three benchmarks: Bench.~A ($\kappa \approx 3.2$, well-conditioned), Bench.~B ($\kappa \leq 10$, moderate), Bench.~C ($\kappa \approx 2504$, severely ill-conditioned). Right: the asymptotic gap between GD's $O(1/k)$ and Nesterov's $O(1/k^2)$ rates on a log-log scale, illustrating that Nesterov provides increasingly larger speedups as $k$ grows.}
\label{fig:theory_rates}
\end{figure}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, max momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Increasing momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolated) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key difference from Heavy~Ball is the \emph{lookahead}: the gradient is evaluated at the extrapolated point $\tilde{x}_k = x_k + \beta_k z_k$ rather than at $x_k$. This allows Nesterov to ``look ahead'' in the momentum direction and correct course before overshooting. The adaptive schedule $\beta_k = (k-1)/(k+2) \to 1$ gradually increases momentum, achieving the optimal $O(1/k^2)$ convergence rate for convex smooth functions:
\begin{equation}
f(x_k) - f^* \leq \frac{2L\|x_0 - x^*\|^2}{(k+1)^2},
\end{equation}
compared to $O(1/k)$ for standard gradient descent.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\quad v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scaling)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam unifies momentum (via $m_t \approx$ running mean of gradients) with per-coordinate adaptive scaling (via $v_t \approx$ running mean of squared gradients), inheriting advantages of both Nesterov and RMSprop. The bias corrections $(1-\beta_j^t)$ compensate for the zero initialisation of $m_0$ and $v_0$, ensuring that early estimates are not biased towards zero. Without bias correction, the effective step size in early iterations would be $\alpha \cdot (1-\beta_1^t)/\sqrt{(1-\beta_2^t)}$, which would be inappropriately small.

The effective per-coordinate step size in Adam is:
\begin{equation}
\tilde{\alpha}_{t,i} = \frac{\alpha}{\sqrt{\hat{v}_{t,i}} + \varepsilon} \approx \frac{\alpha}{\sqrt{\mathbb{E}[g_{t,i}^2]} + \varepsilon},
\end{equation}
which normalises by the root-mean-square of recent gradients, keeping steps in $[\varepsilon, \alpha/\varepsilon]$. This bounds the effective step regardless of gradient magnitude, providing implicit regularisation of the step size across coordinates. The parameter $\varepsilon = 10^{-8}$ prevents division by zero and also sets a maximum effective step of $\alpha/\varepsilon$ for near-zero-gradient coordinates.

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark~A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, and $E = 50$ epochs, with data shuffled at each epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\to \pi$
    \For{each mini-batch $\mathcal{B} = \{\pi(j),\ldots,\pi(j+b-1)\}$}
        \State $\hat{g} \gets \dfrac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

The mini-batch gradient $\hat{g} = \frac{1}{b}\sum_{i \in \mathcal{B}} \nabla_i f(\theta)$ is an unbiased estimator of the full gradient $\nabla J(\theta)$ when batches are sampled uniformly, with variance:
\begin{equation}
\mathbb{E}\|\hat{g} - \nabla J\|^2 = \frac{\sigma_g^2}{b},
\end{equation}
where $\sigma_g^2$ is the per-sample gradient variance. Larger batches reduce variance but perform fewer parameter updates per epoch.

\subsection{SGD Convergence Theory}

For mini-batch SGD on a $\mu$-strongly convex, $L$-smooth objective with gradient noise variance $\sigma_g^2$ and batch size $b$, the expected suboptimality satisfies:
\begin{equation}
\mathbb{E}[f(\theta_k)] - f^* \leq (1 - \mu\alpha)^k (f(\theta_0) - f^*) + \frac{\alpha L \sigma_g^2}{2\mu b}.
\end{equation}
The first term decays geometrically; the second is an irreducible \emph{noise floor} proportional to $\alpha/b$. Setting $\alpha = c/k$ (decaying step) allows both terms to vanish, recovering $O(1/k)$ convergence. With fixed step $\alpha = 0.06$ (as here), the method converges to a ball of radius $O(\alpha\sigma_g^2/b)$ around $f^*$, visible as the flat floor in Figure~\ref{fig:q2_sgd_batch}.

\subsection{Discussion: Nesterov vs Adam on Non-Quadratic Problems}

On smooth quadratics, both Nesterov and Adam are near-optimal. On the Rosenbrock function, Nesterov outperforms Adam significantly ($f=0.479$ vs $f=1.579$). This can be explained as follows:

\begin{itemize}[nosep]
\item \textbf{Nesterov's advantage:} The Rosenbrock valley has a consistent direction (along $x_2 = x_1^2$). Once an iterate reaches the valley, the gradient consistently points in approximately the same direction. Nesterov's growing momentum $\beta_k \to \beta_{\max}$ builds velocity along this direction, providing compound acceleration.
\item \textbf{Adam's limitation:} Adam normalises each coordinate independently by $\sqrt{\hat{v}_t}$. In Rosenbrock, the $x_2$ coordinate has large curvature (200) while $x_1$ has highly variable curvature. The coordinate-wise normalisation distorts the effective step direction relative to the true Newton direction, sometimes working against valley progress.
\item \textbf{Gradient noise:} Mini-batch SGD on non-convex problems can escape local minima due to gradient noise, a property neither Nesterov nor Adam exploits (since they use exact gradients here).
\end{itemize}

\subsection{Local Approximations of $g(x) = x^4$}

For a function $g: \mathbb{R} \to \mathbb{R}$ with at least two continuous derivatives, the Taylor expansion around a point $x_0$ gives local polynomial approximations. At $x_0 = 0.25$: $g(x_0) = 0.25^4 = 3.906 \times 10^{-3}$, $g'(x_0) = 4 \times 0.25^3 = 0.0625$, $g''(x_0) = 12 \times 0.25^2 = 0.75$.

\textbf{First-order (linear) approximation:}
\begin{equation}
\hat{g}_1(x) = g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625(x - 0.25).
\end{equation}
This captures only the slope at $x_0$; it equals $g$ to $O(x-x_0)$ accuracy.

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
\hat{g}_2(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
This equals $g$ to $O((x-x_0)^2)$ accuracy and captures local curvature.

The Newton step for minimising $g$ is derived by minimising $\hat{g}_2$: setting $\hat{g}_2'(x) = 0$ gives $x^* = x_0 - g'(x_0)/g''(x_0) = 0.25 - 0.0625/0.75 = 0.25 - 0.0833 = 0.1667$. The true minimiser of $g(x) = x^4$ is at $x = 0$, so Newton requires multiple steps to converge (unlike for quadratic objectives where it converges in one step).

\subsection{Newton's Method with Hessian Regularisation}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0, 1]$, Hessian damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Levenberg--Marquardt regularisation}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Newton direction $p_k = H_k^{-1} g_k$ minimises the second-order Taylor model $m_k(p) = f(x_k) + g_k^T p + \frac{1}{2} p^T H_k p$, giving $p_k^*$ such that $\nabla_p m_k = 0$. When the true Hessian is used, this results in quadratic convergence:
\begin{theorem}[Quadratic Convergence of Newton's Method]
\label{thm:newton_qconv}
If $f$ is twice continuously differentiable with $L_H$-Lipschitz Hessian and the Hessian satisfies $H(x^*) \succeq \mu I$, then for $x_0$ sufficiently close to $x^*$:
\begin{equation}
\|x_{k+1} - x^*\| \leq \frac{L_H}{2\mu}\,\|x_k - x^*\|^2.
\end{equation}
\end{theorem}
This \emph{quadratic} convergence means that the number of correct digits doubles at each iteration, in contrast to the linear convergence $\|x_{k+1} - x^*\| \leq \rho\|x_k - x^*\|$ of gradient descent. The damping $\lambda = 10^{-8}$ in Algorithm~\ref{alg:newton} ensures numerical stability when $H_k$ is near-singular, and the step $\alpha < 1$ is used in practice on non-quadratic problems to remain within the region where the quadratic model is accurate.

\subsection{Newton vs Gradient Descent: Summary and Quasi-Newton Extensions}

Newton's method achieves dramatically fewer iterations at the cost of computing and inverting the $d \times d$ Hessian per step: $O(d^2)$ to compute $\nabla^2 f$ via finite differences, plus $O(d^3)$ for the linear system solve $H_k p_k = g_k$. For $d = 2$ (as in these benchmarks), this overhead is negligible and Newton is clearly superior. For large $d$ (e.g., $d \sim 10^6$ in deep learning), Hessian storage ($O(d^2)$ memory) and inversion become prohibitive.

\textbf{Quasi-Newton methods} resolve this by building a low-rank approximation $B_k \approx H_k^{-1}$ from gradient differences, avoiding explicit Hessian computation. The BFGS update uses the secant condition $B_{k+1}(g_{k+1} - g_k) = x_{k+1} - x_k$ and maintains $B_k$ as a rank-2 update:
\begin{equation}
B_{k+1} = \left(I - \rho_k s_k y_k^T\right) B_k \left(I - \rho_k y_k s_k^T\right) + \rho_k s_k s_k^T,
\end{equation}
where $s_k = x_{k+1} - x_k$, $y_k = g_{k+1} - g_k$, $\rho_k = 1/(y_k^T s_k)$. BFGS achieves superlinear convergence (between linear and quadratic), with the L-BFGS variant using only the last $m$ vector pairs for $O(md)$ memory. These methods would outperform all gradient-based methods studied here on large-scale problems while avoiding the $O(d^3)$ cost of exact Newton.

\section{Question 4: Derivative Approximation and Derivative-Free Optimisation \hfill [30 marks]}
\label{sec:q4}
% ============================================================

\subsection{Part A: Gradient Approximation Methods (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

When analytical gradients are unavailable, the forward finite-difference approximation uses:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad i = 1, \ldots, d.
\end{equation}
By Taylor's theorem, the truncation error is $\frac{\delta}{2}|\partial^2 f/\partial x_i^2| + O(\delta^2)$. Floating-point cancellation introduces an error of order $\varepsilon_{\mathrm{mach}}/\delta \approx 10^{-16}/\delta$. The total error is minimised at:
\begin{equation}
\delta^* \approx \sqrt{2\varepsilon_{\mathrm{mach}}\,/\,L_2} \approx \sqrt{2 \times 10^{-16}\,/\,L_2},
\end{equation}
where $L_2$ is the bound on $|\partial^2 f/\partial x_i^2|$. For typical smooth functions, $\delta^* \approx 10^{-8}$. The experiment tests $\delta = 0.05$ (reasonable) and $\delta = 0.8$ (poor --- truncation error dominates). Each gradient evaluation requires $d+1 = 3$ function evaluations (1 for $f(x)$, $d$ for the perturbed values).

\subsubsection{Nesterov Random Search (Zeroth-Order)}

\begin{algorithm}[H]
\caption{Nesterov Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I)$; normalise $u_k \gets u_k / \|u_k\|$ (uniform on sphere)
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Directional derivative estimate}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This is a zeroth-order method: only function values are used, no partial derivatives. The estimator $\hat{g}_k$ approximates the directional derivative in the random direction $u_k$, and $\mathbb{E}[\hat{g}_k] = \nabla f(x_k)$ for spherically symmetric distributions (in the limit $\delta \to 0$). The key advantage is \emph{only 2 function evaluations per step} regardless of dimension $d$, making it favourable for large-scale black-box optimisation where $d \gg 2$. The disadvantage is high gradient variance: $\hat{g}_k$ estimates only a 1D projection of the full $d$-dimensional gradient.

\subsubsection{Hyperparameters and Convergence Results on Benchmark B}

\begin{table}[H]

\subsection{Part B: Derivative-Free Methods on Rosenbrock}

\subsubsection{Nelder--Mead Simplex Method}

Nelder--Mead maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively deforms it through four geometric operations applied to the worst vertex $x_w$ relative to the centroid $\bar{x}$ of the remaining vertices:

\begin{enumerate}[nosep]
\item \textbf{Reflection:} $x_r = \bar{x} + \alpha_r(\bar{x} - x_w)$ with $\alpha_r = 1.0$. If $f(x_b) \leq f(x_r) < f(x_{w-1})$, accept.
\item \textbf{Expansion:} $x_e = \bar{x} + \gamma(\bar{x} - x_w)$ with $\gamma = 2.0$. If $f(x_r) < f(x_b)$ and $f(x_e) < f(x_r)$, accept $x_e$.
\item \textbf{Contraction:} $x_c = \bar{x} + \rho(x_r - \bar{x})$ or $\bar{x} + \rho(x_w - \bar{x})$ with $\rho = 0.5$.
\item \textbf{Shrink:} All vertices contract towards the best: $x_i \gets x_1 + \sigma(x_i - x_1)$ with $\sigma = 0.5$.
\end{enumerate}

Each iteration uses at most $O(n)$ function evaluations. No gradient information is required, making Nelder--Mead suitable for non-smooth, noisy, or black-box objective functions. The initial simplex uses step size 0.35 from $x_0 = (-1, 1)$, run for 160 iterations.

\begin{equation}

\section{Question 5: Constrained Optimisation \hfill [30 marks]}
\label{sec:q5}
% ============================================================

\subsection{Problem Statement}

Minimise Benchmark~B subject to a halfspace constraint:
\begin{equation}
\min_{x \in \mathcal{C}}\; f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1), \quad \mathcal{C} = \{x \in \mathbb{R}^2 : x_1 \geq 0.5\}.
\end{equation}
Starting from the \emph{infeasible} point $x_0 = (0.2, 4.0)$ with $x_{0,1} = 0.2 < 0.5$.

\subsection{KKT Conditions}

The constrained problem can be written as $\min_x f(x)$ subject to $h(x) = 0.5 - x_1 \leq 0$. The Karush--Kuhn--Tucker (KKT) necessary conditions for a local constrained minimum $x^*$ state that there exists a multiplier $\mu^* \geq 0$ such that:
\begin{align}
&\nabla f(x^*) + \mu^* \nabla h(x^*) = 0, & \text{(stationarity)}\\
&h(x^*) \leq 0, & \text{(primal feasibility)}\\
&\mu^* \geq 0, & \text{(dual feasibility)}\\
&\mu^* h(x^*) = 0. & \text{(complementary slackness)}
\end{align}
For this problem: $\nabla h(x) = [-1, 0]^T$, so stationarity gives $\partial f/\partial x_1 = \mu^*$ and $\partial f/\partial x_2 = 0$. The unconstrained minimum satisfies $\nabla f = 0$ at $x^* \approx (0.5824, 2.0)$, which is \emph{feasible} ($x_1^* = 0.5824 > 0.5$). Therefore, the constrained minimum coincides with the unconstrained minimum and $\mu^* = 0$ (the constraint is inactive). Both projected GD and the penalty method should converge to $(0.582, 2.0)$ with $f^* \approx 0.7244$.

\subsection{Algorithms}

\begin{algorithm}[H]
\caption{Projected Gradient Descent (PGD)}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{C}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{C}}(z_{k+1})$ \Comment{Project: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

For $\mathcal{C} = \{x : x_1 \geq 0.5\}$, the projection is $\Pi_{\mathcal{C}}(z) = (\max(0.5, z_1),\, z_2)^T$ --- a single max operation. PGD inherits the convergence guarantee of gradient descent for the constrained problem: for $\alpha \leq 1/L$, the iterates satisfy $f(x_{k+1}) \leq f(x_k)$ and $x_k \in \mathcal{C}$ for all $k \geq 1$.

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\State Define $F_\lambda(x) = f(x) + \lambda\max(0, 0.5 - x_1)$ \Comment{$L^1$ penalty for $x_1 \geq 0.5$}
\For{$k = 0, 1, \ldots, N-1$}
    \State Compute subgradient: $\tilde{\nabla}F_\lambda(x_k) = \nabla f(x_k) - \lambda\cdot\mathbf{1}[x_{k,1} < 0.5]\,e_1$
    \State $x_{k+1} \gets x_k - \alpha\, \tilde{\nabla}F_\lambda(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

As $\lambda \to \infty$, the penalty method minimiser $x^*(\lambda) \to x^*$ (the true constrained minimum). For finite $\lambda$, the method solves an \emph{approximation} to the original problem, with feasibility error $O(1/\lambda)$.

\subsection{Hyperparameter Settings}

All methods run for 100 iterations from $x_0 = (0.2, 4.0)$:
\begin{itemize}[nosep]
\item Projected GD: $\alpha = 0.08$
\item Penalty ($\lambda = 0.15$): $\alpha = 0.05$
\item Penalty ($\lambda = 1.8$): $\alpha = 0.05$
\item Penalty ($\lambda = 4.5$): $\alpha = 0.03$ (smaller step required for stability with larger penalty gradient)
\item Unconstrained GD: $\alpha = 0.07$
\end{itemize}

\subsection{Comparative Discussion}

\textbf{Projected gradient descent} guarantees feasibility at every iterate --- this is its fundamental advantage. The projection $\Pi_{\mathcal{C}}$ acts as a hard constraint enforcer; it is computationally cheap ($O(d)$) for simple constraint sets (halfspaces, boxes, $\ell_2$ balls), but may be expensive for complex constraints (general polytopes, semidefinite constraints).

\textbf{Penalty methods} avoid the need for a projection oracle, requiring only gradient computation of the augmented objective $F_\lambda$. Their disadvantages are:
\begin{enumerate}[nosep]
\item Exact feasibility is never achieved for finite $\lambda$ (only approximately, with error $O(1/\lambda)$).
\item Larger $\lambda$ improves feasibility accuracy but requires smaller step sizes for stability, leading to slower convergence.
\item The penalty function $\lambda\max(0, 0.5 - x_1)$ is \emph{non-smooth} at $x_1 = 0.5$, which can cause oscillations near the constraint boundary.
\end{enumerate}

The \emph{augmented Lagrangian} method combines the advantages of both: the augmented Lagrangian objective is
\begin{equation}
\mathcal{L}_\lambda(x, \mu) = f(x) + \mu\, h(x) + \frac{\lambda}{2}\,h(x)^2,
\end{equation}
where $h(x) = \max(0, 0.5 - x_1)$ is the constraint violation, and $\mu$ is a dual multiplier updated iteratively as $\mu_{t+1} = \mu_t + \lambda h(x^*(t))$. Unlike the pure penalty method, the dual update corrects for the finite penalty, achieving exact feasibility without requiring $\lambda \to \infty$. The quadratic term $(\lambda/2) h(x)^2$ improves conditioning of the inner minimisation problem compared to pure linear penalty. The alternating direction method of multipliers (ADMM) is a further decomposition that allows distributed computation.

\subsection{Feasible Set and Linear Programme}

\textbf{Feasible set:} $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$ (a box).

\textbf{Linear programme:} Minimise $f(x) = a^Tx = x_1 + 2x_2$ over $\mathcal{X}$ (with $a = [1, 2]^T$).

\begin{proposition}
The minimum of a linear objective over a bounded convex polytope is attained at a vertex.
\end{proposition}

Since all components of $a = [1, 2]^T$ are positive, the minimum is at the lower bounds:
\begin{equation}
x^{\star}_{\mathrm{LP}} = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
This follows directly from the separability of box constraints: $\min_{x \in \mathcal{X}} (x_1 + 2x_2) = \min_{x_1 \in [0.5,5]} x_1 + \min_{x_2 \in [-5,10]} 2x_2 = 0.5 + (-10) = -9.5$.

\subsection{Frank--Wolfe (Conditional Gradient) Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe with Fixed Mixing Parameter}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (step toward LMO solution)}
\EndFor
\end{algorithmic}
\end{algorithm}

\textbf{Key properties:}
\begin{itemize}[nosep]
\item \textbf{Projection-free:} Iterates are always convex combinations of feasible points, so $x_k \in \mathcal{X}$ for all $k$ without any explicit projection.
\item \textbf{LP subproblem:} For box constraints, the LMO decomposes per coordinate: $(z_k)_i = l_i$ if $(\nabla f)_i > 0$, else $(z_k)_i = u_i$. This costs $O(d)$.
\item \textbf{Standard Frank--Wolfe} uses the decaying step $\gamma_k = 2/(k+2)$ (so $\beta_k = 1 - \gamma_k = k/(k+2)$) and achieves convergence $f(x_k) - f^* \leq \frac{2L D^2}{k+2}$ where $D = \mathrm{diam}(\mathcal{X})$ and $L$ is the gradient Lipschitz constant. Our implementation uses a \emph{fixed} $\beta$, which is equivalent to a constant step size and may not achieve the theoretical $O(1/k)$ rate.
\end{itemize}

The \textbf{Frank--Wolfe gap} $g_k = \nabla f(x_k)^T(x_k - z_k) \geq 0$ provides a computable upper bound on the suboptimality $f(x_k) - f^*$ and serves as a convergence certificate.

\subsection{Frank--Wolfe Duality Gap as Convergence Certificate}

The \textbf{Frank--Wolfe duality gap} is defined as:
\begin{equation}
g_k = \nabla f(x_k)^T(x_k - z_k) = \max_{z \in \mathcal{X}}\, \nabla f(x_k)^T(x_k - z).
\end{equation}
It satisfies $g_k \geq f(x_k) - f^*$ (a valid upper bound on suboptimality) and $g_k = 0$ if and only if $x_k$ is a constrained stationary point. The gap is computable at each iteration without knowing $f^*$, making it a practical convergence certificate.

\subsection{Interior vs Boundary Behaviour and Discussion}

\textbf{Interior optimum:} Frank--Wolfe exhibits oscillation as the LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ --- the gradient direction changes sign near the interior optimum, causing $z_k$ to alternate between opposite vertices. The iterate $x_k$, being a convex combination of past vertices, oscillates with decreasing amplitude. The smaller step ($\beta = 0.985$, effectively $\gamma = 0.015$) gives better final convergence here because the oscillation amplitude is reduced. With the \emph{standard} decaying schedule $\gamma_k = 2/(k+2)$, the step decays to zero, eventually resolving the oscillation and recovering the $O(1/k)$ guarantee.

\textbf{Boundary optimum:} Frank--Wolfe is naturally well-suited to boundary optima: when the unconstrained minimum lies outside $\mathcal{X}$, the gradient consistently points towards the LP vertex at the constrained optimum, so $z_k$ does not oscillate. The FW gap decreases monotonically (Figure~\ref{fig:q6_fw_gap}, right). The residual gap (0.114) after 140 iterations with fixed $\beta = 0.93$ reflects the fixed-step limitation --- the effective step towards $z_k$ is $(1-\beta) = 0.07$ regardless of proximity to $x^*$, causing slow final approach.

\textbf{Comparison with Projected GD:} Projected gradient descent (Section~\ref{sec:q5}) achieves zero constraint violation immediately and converges more rapidly for smooth objectives when a cheap projection is available. Frank--Wolfe's advantage emerges when the feasible set has exploitable LP structure (e.g., nuclear norm ball where LP oracle = top-1 SVD, spectrahedra where LP oracle = leading eigenvector), and the projection is expensive or intractable. For the box $\mathcal{X}$ studied here, both projection and LP are $O(d)$, making PGD preferred due to faster convergence.

% ============================================================
\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This study compared six categories of optimisation methods across three benchmark problems of increasing difficulty.

\paragraph{Adaptive step-size methods (Q1).} The Polyak step size is a powerful but fragile method: highly effective when $f^{\star}$ is known exactly (Rosenbrock, $f = 0.009$) but divergent otherwise. Adagrad and Heavy~Ball provide stable, consistent performance without requiring knowledge of $f^{\star}$. RMSprop's exponential averaging resolves Adagrad's diminishing step problem but requires careful base rate selection. \emph{Recommendation: Heavy~Ball for general use; Polyak only when the optimal value is reliably known.}

\paragraph{Momentum and stochastic methods (Q2).} Nesterov acceleration achieves a theoretically optimal $O(1/k^2)$ convergence rate and demonstrates the largest practical improvement over GD on ill-conditioned problems (Rosenbrock: $7\times$ better). Adam's per-coordinate scaling provides adaptive benefit on moderate-anisotropy problems but is less effective than pure momentum on consistently-directed problems. Mini-batch SGD trades convergence speed per epoch for stochastic exploration, with batch size controlling the variance--speed trade-off.

\paragraph{Newton's method (Q3).} Newton's method with Hessian information achieves quadratic convergence, converting a 80-iteration GD problem into a 20-iteration (or 1-iteration for quadratics) problem. The cost is $O(d^3)$ per step for exact Hessian inversion, motivating quasi-Newton approximations (BFGS, L-BFGS) for large-scale problems. Hessian regularisation ($\lambda = 10^{-8}$) ensures stability near saddle points.

\paragraph{Derivative-free methods (Q4).} Finite-difference gradient approximation with $\delta = 0.05$ nearly matches exact gradient descent on smooth functions at $d+1 = 3$ evaluations/step. Large $\delta = 0.8$ corrupts gradient direction (truncation error $\sim$4 for Benchmark~B's $x_2$ coordinate). Nesterov random search requires only 2 evaluations/step and is dimension-independent, ideal for high-dimensional black-box problems. \textbf{Nelder--Mead converges to the Rosenbrock global minimum ($f \approx 0$) in just 160 iterations} --- outperforming all gradient-based first-order methods studied in Q1--Q3 on Benchmark~C. This underscores that derivative-free simplex methods can be highly effective when the objective is smooth and the dimension is low ($d = 2$ here). Grid search offers bounded global error at exponential computational cost.

\paragraph{Constrained optimisation (Q5).} Projected gradient descent enforces feasibility as a hard constraint from iteration~1, while penalty methods achieve approximate feasibility with accuracy controlled by $\lambda$. The KKT conditions reveal that for this particular problem, the constrained optimum coincides with the unconstrained one (inactive constraint, $\mu^* = 0$), so both methods converge to the same point. In problems where the constraint is active, the methods diverge in their trajectories.

\paragraph{Frank--Wolfe (Q6).} The Frank--Wolfe algorithm maintains feasibility projection-free and exploits the LP structure of the feasible set. The fixed-$\beta$ variant used here avoids the decaying step-size schedule of standard Frank--Wolfe but introduces oscillation for interior optima. Boundary optima, where the LP vertex consistently points towards the constraint, are handled more smoothly. The algorithm is most advantageous when the LP subproblem is cheaper than the projection oracle.

\paragraph{Overall comparison.} For smooth unconstrained problems, Newton's method (when $d$ is small) or Nesterov momentum (when $d$ is large) provide the best convergence. For constrained problems with cheap projections, projected gradient descent is preferred. For black-box problems, Nesterov random search (high $d$) or Nelder--Mead (low $d$, smooth) are the most practical choices.

The central lesson from this comparative study is that \emph{no single algorithm dominates universally}. Polyak beats all first-order methods on Rosenbrock but is unusable when $f^*$ is unknown. Newton beats all first-order methods per iteration but scales poorly in $d$. Nelder--Mead finds the exact Rosenbrock minimum but would fail on a noisy or high-dimensional objective. The appropriate algorithm depends on three key factors: (1) \textbf{availability of derivatives} (analytic, finite-difference, or zero-order); (2) \textbf{problem dimensionality} (dictating whether Hessian computation and projection operations are tractable); and (3) \textbf{problem structure} (quadratic vs.\ nonlinear, smooth vs.\ noisy, unconstrained vs.\ constrained, convex vs.\ non-convex). This framework provides a principled basis for algorithm selection in practical optimisation problems arising in data analysis and machine learning.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball / Polyak Momentum}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g          # velocity accumulation
        x = x - z                         # update
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z        # extrapolated point
        g = grad_fn(lookahead)             # gradient at lookahead
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)    # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Hessian Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)  # solve H p = g
        except np.linalg.LinAlgError:
            p = g                           # fall back to GD
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        # LP subproblem: minimise g^T z over box bounds
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                      for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z    # convex combination step
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
