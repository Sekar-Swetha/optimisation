\begin{abstract}
This report presents a systematic comparative study of first-order, second-order, and derivative-free optimisation algorithms applied to three benchmark problems of increasing difficulty: a convex quadratic loss arising from noisy linear regression (Benchmark A, $\kappa \approx 10$), a mildly non-convex toy neural-network loss with sinusoidal perturbation (Benchmark B, $\kappa \approx 3$--10), and the highly ill-conditioned Rosenbrock function (Benchmark C, $\kappa \approx 2009$). Twelve algorithms across six categories are implemented and evaluated: (1) adaptive step-size methods --- Polyak, Adagrad, RMSprop, and Heavy Ball (Polyak Momentum); (2) momentum and stochastic methods --- Nesterov acceleration ($O(1/k^2)$ optimal), Adam, and mini-batch SGD with varying noise and learning rate schedules; (3) Newton's method with Hessian damping and second-order local approximations; (4) derivative-free approaches --- finite-difference gradient approximation, Nesterov Gaussian smoothing (random search), Nelder--Mead simplex, and grid search; (5) constrained optimisation via projected gradient descent and penalty methods; and (6) the Frank--Wolfe (conditional gradient) algorithm for linear-constrained box problems, alongside exact linear programming. Key quantitative findings: Polyak achieves $f = 0.009$ on Rosenbrock (best first-order result) when $f^{\star}=0$ is correctly specified, but diverges to $f = 24.1$ on Benchmark B due to misspecification; Nesterov acceleration reaches $f = 0.07$ on Rosenbrock in 300 iterations, a $20\times$ improvement over GD ($f = 1.39$); Newton's method converges in exactly one iteration on the quadratic loss and achieves $f = 0.686$ on Rosenbrock in just 20 iterations; and Frank--Wolfe with fixed-$\beta$ mixing achieves $f = 0.0046$ for the interior optimum while naturally enforcing box feasibility. The report includes 46 figures, 13 tables, and formal theorems for all key convergence guarantees.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use \texttt{np.random.seed(42)} for reproducibility. Starting points and benchmark definitions are fixed across all experiments to ensure comparability. The three benchmarks are deliberately chosen to expose complementary difficulties: Benchmark A (well-conditioned convex) isolates the effect of adaptive scaling; Benchmark B (mildly non-convex with sinusoidal perturbation) tests robustness to local non-convexity; Benchmark C (Rosenbrock, $\kappa \approx 2009$) is the canonical stress test revealing the critical role of curvature information. Together they provide a spectrum of difficulty from ``easy'' (most methods succeed) to ``hard'' (only second-order and specialised methods succeed).

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
  \theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad
  x^{(i)} \stackrel{\mathrm{i.i.d.}}{\sim} \mathcal{N}(0, I_2), \quad
  y^{(i)} = {\theta^{\star}}^{\!\top} x^{(i)} + \varepsilon^{(i)}, \quad
  \varepsilon^{(i)} \stackrel{\mathrm{i.i.d.}}{\sim} \mathcal{N}(0, 1).
  \label{eq:benchA_data}
\end{equation}
The loss function is the mean squared error:
\begin{equation}
  J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^{\top} x^{(i)} - y^{(i)}\bigr)^2
            = \frac{1}{2m}\|X\theta - y\|^2.
  \label{eq:benchA_loss}
\end{equation}
This is a strictly convex quadratic with Hessian $H = \frac{1}{m}X^{\top}X \succ 0$ (positive definite, constant). The gradient is $\nabla J(\theta) = \frac{1}{m}X^{\top}(X\theta - y)$. Due to observation noise the noise floor is $J^{\star} = \mathbb{E}[\varepsilon^2]/2 \approx 0.4826$; the optimal parameter is $\theta^{\star}_{\mathrm{OLS}} = (X^{\top}X)^{-1}X^{\top}y \approx (3.028,\, 4.017)^{\top}$. Starting point: $\theta_0 = [0, 0]^{\top}$.

\begin{remark}[Condition Number of Benchmark A]
With $m = 1000$ i.i.d.\ Gaussian features $X \sim \mathcal{N}(0, I_2)$, the sample covariance $\frac{1}{m}X^{\top}X \to I_2$ as $m \to \infty$. For finite $m = 1000$, the Marchenko--Pastur law predicts eigenvalues of $\frac{1}{m}X^{\top}X$ distributed approximately in $[(1 - \sqrt{d/m})^2, (1 + \sqrt{d/m})^2] = [(1-\sqrt{0.002})^2,\, (1+\sqrt{0.002})^2] \approx [0.91, 1.09]$, giving $\kappa \approx 1.09/0.91 \approx 1.2$. In practice (with \texttt{seed=42}), the empirical $\kappa(H) \approx 1.1$--$1.3$, making Benchmark A extremely well-conditioned: GD with any $\alpha \leq 1/1.09 \approx 0.91$ converges very quickly. The dominant challenge is not conditioning but the irreducible noise floor $J^{\star} \approx 0.4826 \neq 0$.
\end{remark}

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
  f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
  \label{eq:benchB}
\end{equation}
The gradient is $\nabla f = [2(x_1 - 1) + \cos(x_1),\ 10(x_2 - 2)]^{\top}$ and the Hessian is $H = \operatorname{diag}(2 - \sin(x_1),\ 10)$. Since $|{-}\sin(x_1)| \leq 1$, the diagonal entries of $H$ lie in $[1, 3]$ and $[10, 10]$, giving a condition number $\kappa(H)$ ranging from $\sim$3 to $\sim$10. This moderate anisotropy is representative of shallow neural-network loss surfaces.

\begin{remark}[Saddle Points and Non-Convexity in Benchmark B]
The Hessian entry $(2 - \sin(x_1))$ is always positive ($\geq 1$), so Benchmark B is \emph{strictly convex} at every point: $H \succ 0$ everywhere. Despite the sinusoidal term, there are no saddle points or local minima. This distinguishes Benchmark B from true neural network losses, where saddle points are ubiquitous (particularly in the over-parameterised regime where the number of parameters exceeds the data dimension). For non-convex functions with saddle points, standard first-order methods (GD, Nesterov, Adam) are not guaranteed to converge to a local minimum: they may converge to a saddle point where $\nabla f = 0$ but $\nabla^2 f \not\succ 0$. Adding perturbations (stochastic gradient noise, random restarts) or using negative curvature escape mechanisms (e.g., Hessian-vector products) is required. The $O(1/\sqrt{k})$ convergence rates in the non-convex column of Table~\ref{tab:convergence_theory} guarantee only $\|\nabla f(x_k)\| \leq \varepsilon$, not convergence to a local minimum.
\end{remark} The global minimum lies at approximately $x^{\star} \approx (0.582, 2.0)$ with $f^{\star} \approx 0.7244$: the $x_2$ component is exactly 2.0 (from $\partial f/\partial x_2 = 10(x_2-2) = 0$), and the $x_1$ component satisfies $2(x_1-1) + \cos(x_1) = 0$, a transcendental equation solved numerically to give $x_1^{\star} \approx 0.582$. The trajectory plots use an approximate marker at $x_1 = 0.61$, which is visually close but slightly overestimates the true optimum. Starting point: $x_0 = [-1, 4]^{\top}$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
  f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.

\end{equation}
The global minimum is at $x^{\star} = (1, 1)$ with $f^{\star} = 0$. The Hessian at the optimum is

\subsection{Theoretical Background}

Constant step-size gradient descent (GD) with step $\alpha$ satisfies $f(x_{k+1}) \leq f(x_k) - \alpha(1 - \alpha L/2)\|\nabla f(x_k)\|^2$ for $L$-smooth $f$. Setting $\alpha \leq 1/L$ ensures monotone decrease, but the optimal fixed rate $\alpha^{\star} = 2/(\mu + L)$ requires knowledge of the strong convexity constant $\mu$ and smoothness $L$. Adaptive methods remove this requirement by estimating local curvature from gradient information.

\begin{lemma}[Descent Lemma \cite{boyd2004convex}]\label{lem:descent}
If $f\colon\mathbb{R}^d \to \mathbb{R}$ is $L$-smooth (i.e., $\|\nabla f(x) - \nabla f(y)\| \leq L\|x-y\|$ for all $x,y$), then for any step $\alpha \leq 1/L$, gradient descent satisfies:
\begin{equation}
  f(x_{k+1}) \leq f(x_k) - \frac{\alpha}{2}\|\nabla f(x_k)\|^2.
  \label{eq:descent}
\end{equation}
In particular, $f$ is non-increasing along the GD trajectory, and summing \eqref{eq:descent} over $k=0,\ldots,K-1$ gives $\min_{k < K}\|\nabla f(x_k)\|^2 \leq \frac{2(f(x_0)-f^{\star})}{\alpha K} = O(1/K)$.
\end{lemma}

\begin{remark}[Polyak--\L{}ojasiewicz Condition]
Geometric convergence of GD does not require strong convexity; the weaker \emph{Polyak--\L{}ojasiewicz (PL) inequality} suffices:
\begin{equation}
  \tfrac{1}{2}\|\nabla f(x)\|^2 \geq \mu\bigl(f(x) - f^{\star}\bigr) \quad \forall\, x.
  \label{eq:pl}
\end{equation}
Under \eqref{eq:pl} with $\mu > 0$, GD with step $\alpha \leq 1/L$ achieves $f(x_k) - f^{\star} \leq (1 - \alpha\mu)^k(f(x_0)-f^{\star})$, matching the strongly convex rate. Benchmark A (quadratic) satisfies PL globally with $\mu = \lambda_{\min}(H)$; Benchmark B satisfies PL locally near the minimum. Figure~\ref{fig:pl_condition} empirically verifies the PL condition for both benchmarks by scattering $\frac{1}{2}\|\nabla f\|^2$ against $f - f^{\star}$ for random sample points; the minimum ratio gives the effective $\mu$.
\end{remark}

\begin{figure}[H]
\centering
\includegraphics[width=\textwidth]{figures/q1_pl_condition.pdf}
\caption{Q1: Empirical verification of the Polyak--\L{}ojasiewicz (PL) condition for Benchmarks A (left) and B (right). Each point is a random sample; the red line shows the tightest PL lower bound $\frac{1}{2}\|\nabla f\|^2 \geq \mu(f - f^{\star})$. \end{figure}

\begin{definition}[Polyak Step Size \cite{polyak1987introduction}]
For a differentiable function $f$ with known minimum $f^{\star}$, the Polyak step size at iteration $k$ is
\begin{equation}
  \alpha_k = \frac{f(x_k) - f^{\star}}{\|\nabla f(x_k)\|^2 + \varepsilon},
  \label{eq:polyak}
\end{equation}
where $\varepsilon > 0$ is a small regularisation constant preventing division by zero.
\end{definition}

\begin{theorem}[Polyak Convergence]\label{thm:polyak}
If $f$ is convex and $f^{\star}$ is the exact minimum value, then the Polyak step guarantees
$f(x_{k+1}) - f^{\star} \leq \frac{1}{2}(f(x_k) - f^{\star})$ per iteration when $f$ is $L$-smooth.
In particular, $f(x_k) - f^{\star} \leq (1/2)^k\,(f(x_0) - f^{\star})$, a geometric convergence rate.
\end{theorem}

When $f^{\star}$ is misspecified (set to 0 when the true minimum $f_{\mathrm{true}}^{\star} > 0$), as $x_k \to x^{\star}$ we have $\|\nabla f(x_k)\| \to 0$ while $f(x_k) - 0 \to f_{\mathrm{true}}^{\star} > 0$, causing $\alpha_k \to \infty$. The iterate then overshoots and diverges.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against constant step-size GD across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Adagrad \cite{duchi2011adaptive}}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0 > 0$, tolerance $\varepsilon > 0$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate cumulative squared gradient}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad's effective per-coordinate learning rate $\alpha_0 / (\sqrt{(G_k)_i} + \varepsilon)$ decreases monotonically as iterations proceed, since $G_k$ only increases. This eliminates the need to tune a learning rate for each coordinate but causes the step size to vanish, which can prematurely halt convergence on non-stationary problems.

\begin{algorithm}[H]
\caption{RMSprop \cite{tieleman2012lecture}}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0 > 0$, decay $\beta \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $v_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of squared gradient}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop addresses Adagrad's monotone decay by replacing the cumulative sum with an exponential moving average (EMA) with forgetting factor $\beta \in (0,1)$. The effective step size thus tracks recent gradient magnitudes rather than all historical ones, allowing the step to recover from early large gradients.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum) \cite{polyak1964methods}}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, momentum $\beta \in [0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Heavy Ball and Chebyshev Polynomials]
On the quadratic $f(\theta) = \frac{1}{2}\theta^{\top}H\theta - b^{\top}\theta$, the Heavy Ball iterates satisfy a linear recurrence. The optimal momentum parameters are chosen so that the characteristic polynomial of this recurrence equals the \emph{scaled Chebyshev polynomial} of degree $k$ shifted to the interval $[\mu, L]$, which minimises the $\ell^\infty$ norm over polynomials that vanish at 1. This connection to Chebyshev polynomials explains both the optimality of Heavy Ball on quadratics and why the optimal parameters are $\beta^{\star} = \bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\bigr)^2$: they are precisely the spectral radius of the Chebyshev recurrence. \emph{The same connection underlies Conjugate Gradient (CG), which also achieves optimal $O(1/k)$ convergence on quadratics via Krylov subspace methods, reaching machine precision in at most $d$ steps.}
\end{remark}

\begin{remark}[Heavy Ball Optimality on Quadratics]
The Heavy Ball method accumulates gradient history with exponential decay. For strongly convex quadratics with condition number $\kappa = L/\mu$, the optimal parameters
\begin{equation}
  \alpha^{\star} = \left(\frac{2}{\sqrt{\mu}+\sqrt{L}}\right)^2, \quad
  \beta^{\star} = \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^2
\end{equation}
yield a convergence rate of $\bigl(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\bigr)^k$, which equals the optimal first-order rate for this problem class. In contrast, the best GD rate is $\bigl(\frac{\kappa-1}{\kappa+1}\bigr)^k$. For large $\kappa$, the ratio of rates is approximately $\sqrt{\kappa}$, so Heavy Ball converges $\sqrt{\kappa}$ times faster per iteration. On Benchmark C, this translates to a $\sqrt{2009} \approx 45$ times faster convergence rate in the quadratic approximation near the optimum.
\end{remark}

\subsection{Comparative Analysis}

\begin{itemize}[leftmargin=1.5em]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly, Polyak converges fastest ($f = 0.0094$ on Rosenbrock). Otherwise, Heavy Ball and Adagrad match or exceed GD at negligible computational overhead.

\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential gradient averaging. Heavy Ball shows characteristic damped oscillation near the optimum.

\item \textbf{Stability:} GD with a sufficiently small $\alpha$ is the most stable method but slowest. Polyak is \emph{unstable} under $f^{\star}$ misspecification. Adagrad is uniformly stable due to monotonically decreasing step sizes, though this property is also its main limitation on long-running problems.

\item \textbf{Sensitivity to $f^{\star}$:} Polyak's utility in practice is severely limited by the requirement to know $f^{\star}$ exactly. For noisy problems (Benchmark A) or problems without a closed-form minimum (Benchmark B), this information is unavailable, and even a slight misspecification leads to divergence.

\item \textbf{Adagrad vs.\ RMSprop:} Both use per-coordinate scaling, but Adagrad's cumulative sum leads to effective step sizes $O(1/\sqrt{k})$, which vanishes too quickly. RMSprop's EMA maintains $O(1)$ effective steps, making it better suited for long training runs. This distinction is fundamental in modern deep learning practice.

\item \textbf{Heavy Ball vs.\ constant GD:} On Benchmark C, Heavy Ball reduces the final objective by $60\%$ compared to GD (1.40 vs.\ 3.51). This is consistent with theory: on strongly convex quadratics, Heavy Ball with optimal parameters achieves a convergence rate of $(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1})^k$, compared to GD's $(\frac{\kappa-1}{\kappa+1})^k$, a quadratic improvement in rate.
\end{itemize}

\noindent\textbf{Practical recommendation (Q1):} Use Adagrad for sparse or non-stationary gradient problems (natural language processing, sparse features); use RMSprop or Adam for deep learning where gradient statistics vary; use Heavy Ball for problems with known condition number and tight hyperparameter budgets. Avoid Polyak unless $f^{\star}$ is known with certainty (e.g., from problem structure or a pre-computed lower bound).

\subsection{Theoretical Background}

Gradient descent on an $L$-smooth strongly convex function with constant $\mu > 0$ converges as $O((1 - \mu/L)^k)$. For large condition numbers $\kappa = L/\mu$, this is essentially $O((1-1/\kappa)^k) \approx O(e^{-k/\kappa})$, which is extremely slow. Nesterov acceleration improves this to $O((1 - 1/\sqrt{\kappa})^k)$, an optimal rate that cannot be improved by any first-order method \cite{nesterov1983method}.

\begin{remark}[Connection to Adaptive Preconditioning and Natural Gradient]
The adaptive methods in Question~1 (Adagrad, RMSprop) can be understood as \emph{diagonal preconditioning}: the update $x_{k+1} = x_k - D_k^{-1}\nabla f(x_k)$ where $D_k = \operatorname{diag}(\sqrt{G_k} + \varepsilon I)$ effectively rescales each coordinate by its historical gradient magnitude. This reduces the effective condition number from $\kappa$ to closer to 1 for each coordinate, but does not capture off-diagonal curvature correlations. Full Newton's method (Question~3) uses the complete Hessian as preconditioner, reducing the condition number to exactly 1 for quadratics.

The \emph{natural gradient} method \cite{bottou2018optimization} generalises this further: for probabilistic models parameterised by $\theta$, the natural gradient replaces the Euclidean gradient $\nabla_\theta \mathcal{L}$ with $F(\theta)^{-1}\nabla_\theta\mathcal{L}$, where $F(\theta) = \mathbb{E}[\nabla\log p(x|\theta)\nabla\log p(x|\theta)^{\top}]$ is the Fisher information matrix. The Fisher matrix is the second-order approximation to the KL divergence, making natural gradient descent invariant to reparameterisation. Adam with $\beta_2 \to 1$ approximates $F(\theta)^{-1}$ diagonally via cumulative squared gradients $\approx \mathbb{E}[g_t^2]$. This connects all four adaptive methods (Adagrad, RMSprop, Adam, Natural Gradient) as progressively better approximations to the full second-order Newton preconditioner.
\end{remark}

\begin{theorem}[Nesterov Optimal Rate \cite{nesterov1983method}]\label{thm:nesterov}
For a convex $L$-smooth function $f$, the Nesterov accelerated gradient method with step size $\alpha = 1/L$ achieves:
\begin{equation}
  f(x_k) - f^{\star} \leq \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} = O\!\left(\frac{1}{k^2}\right),
\end{equation}
which is optimal for the class of convex $L$-smooth functions, matching the information-theoretic lower bound.
\end{theorem}

This $O(1/k^2)$ rate represents a quadratic speedup over GD's $O(1/k)$ rate. Moreover, this rate is \emph{optimal}: no first-order method can achieve $o(1/k^2)$ for general convex $L$-smooth functions in worst case \cite{nesterov1983method}, establishing Nesterov's algorithm as an optimal method for this class.

\begin{remark}[Information-Theoretic Lower Bound]
For any first-order method $\{x_k\}_{k \geq 0}$ where each $x_{k+1} \in \mathrm{span}(\nabla f(x_0), \ldots, \nabla f(x_k))$ (linear span of previous gradients), there exists an $L$-smooth convex function $f$ and starting point $x_0$ such that:
\begin{equation}
  f(x_k) - f^{\star} \geq \frac{3L\|x_0 - x^{\star}\|^2}{32(k+1)^2} = \Omega\!\left(\frac{1}{k^2}\right).
  \label{eq:lower_bound}
\end{equation}
Since Nesterov's method achieves \eqref{eq:lower_bound} up to a constant factor, it is \emph{worst-case optimal} among all first-order methods. The Heavy Ball method, despite also using gradient history, does not achieve this lower bound on general convex problems (only on quadratics).
\end{remark}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, max momentum $\beta_{\max} \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\frac{k-1}{k+2},\ \beta_{\max}\right)$
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Look-ahead point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$ \Comment{Update velocity}
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}[Look-Ahead vs.\ Heavy Ball: The Key Distinction]
Heavy Ball uses a fixed $\beta$ and evaluates the gradient at the current iterate $x_k$, accumulating a velocity $z_k$ that can overshoot on non-quadratic problems. Nesterov's critical innovation is the \emph{look-ahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$, one momentum step ahead. This allows the method to ``correct'' its trajectory using gradient information from where it will land, not where it is. The adaptive schedule $\beta_k = (k-1)/(k+2)$ (Nesterov's original sequence) satisfies $t_k^2 \leq t_{k+1}^2 - t_{k+1}$ where $t_k = (k+2)/2$, a Lyapunov condition that drives the $O(1/k^2)$ convergence. Intuitively, as $\beta_k \to 1$, the method takes increasingly large momentum steps, but the look-ahead gradient ensures these are directed \emph{toward} the optimum rather than overstepping. On Benchmark C (Figure~\ref{fig:q_heavy_vs_nesterov}), this produces a $3.5\times$ improvement in final objective value (0.07 vs.\ 0.25 after 300 iterations).
\end{remark}

\begin{algorithm}[H]
\caption{Adam Optimiser \cite{kingma2015adam}}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha > 0$, exponential decay rates $\beta_1, \beta_2 \in (0,1)$, tolerance $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (gradient EMA)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (squared gradient EMA)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias-corrected first moment}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$ \Comment{Bias-corrected second moment}
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
Adam combines momentum (via $m_t$, an EMA of gradients with $\beta_1 = 0.9$) with per-coordinate adaptive scaling (via $v_t$, an EMA of squared gradients with $\beta_2 = 0.999$). The bias correction factors $(1-\beta_1^t)^{-1}$ and $(1-\beta_2^t)^{-1}$ compensate for the fact that $m_0 = v_0 = \mathbf{0}$; without correction, early iterates would be biased toward zero. Adam can be viewed as a combination of RMSprop (via $v_t$) and momentum (via $m_t$).
\end{remark}

\begin{remark}[Armijo Line Search for Automatic Step Size]
All methods in Q1--Q2 require hand-tuned step sizes. The \emph{Armijo sufficient-decrease} condition provides a principled alternative: accept step size $\alpha_k$ if
\begin{equation}
  f(x_k - \alpha_k \nabla f(x_k)) \leq f(x_k) - c\,\alpha_k\|\nabla f(x_k)\|^2, \quad c \in (0, 1),
  \label{eq:armijo}
\end{equation}
starting from $\alpha_k = 1$ and halving until \eqref{eq:armijo} is satisfied (backtracking). This removes the need to know $L$ while guaranteeing descent at every step. Combined with the Wolfe curvature condition, it ensures the step is neither too small (sufficient decrease) nor too large (curvature condition). The Polyak step can be interpreted as a special Armijo condition where $c = 0$ and the step is set to the exact minimisation value along the gradient direction.
\end{remark}

\begin{remark}[Adam Convergence and Non-Convergence]
Despite empirical success, Adam can fail to converge on simple convex problems. Reddi et al.\ showed a counterexample where the exponential moving average of squared gradients can oscillate; their \emph{AMSGrad} variant fixes this by using $\hat{v}_t = \max(\hat{v}_{t-1}, v_t)$ (non-decreasing denominator). The theoretical $O(1/\sqrt{T})$ regret guarantee for Adam is established in the online learning setting; for offline optimisation, only $O(1/\sqrt{T})$ convergence of $\min_{k \leq T}\|\nabla f(x_k)\|^2$ has been proved \cite{kingma2015adam}. On Benchmark C, Adam achieves $f = 0.09$ (300 iterations), slightly worse than Nesterov's $f = 0.07$, consistent with Adam's adaptive scaling being less suited to the consistent ill-conditioning of Rosenbrock.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD on Benchmark A uses learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data uniformly reshuffled at every epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y) \in \mathbb{R}^{m \times d} \times \mathbb{R}^m$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1, \ldots, m\}$
    \For{each mini-batch $\mathcal{B} \subset \{1,\ldots,m\}$ with $|\mathcal{B}| = b$}
        \State $\hat{g} \gets \frac{1}{b} X_{\mathcal{B}}^{\top}(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, \hat{g}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent With batch size $b$, each epoch performs $\lceil m/b \rceil$ parameter updates. The stochastic gradient $\hat{g}$ is an unbiased estimate of the full gradient: $\mathbb{E}[\hat{g}] = \nabla J(\theta)$, with variance $\mathrm{Var}(\hat{g}) \propto 1/b$. Larger batch sizes reduce variance but increase computation per update.

\begin{proposition}[SGD Convergence on Strongly Convex Objectives \cite{bottou2018optimization}]\label{prop:sgd_conv}
For a $\mu$-strongly convex, $L$-smooth objective with mini-batch gradient variance $\sigma_b^2 = O(\sigma^2/b)$, SGD with constant step $\alpha \leq 1/L$ satisfies:
\begin{equation}
  \mathbb{E}[J(\theta_K)] - J^{\star}
  \;\leq\; (1 - \alpha\mu)^K \bigl(J(\theta_0) - J^{\star}\bigr)
  \;+\; \frac{\alpha\,\sigma_b^2}{2\mu}.
  \label{eq:sgd_variance_floor}
\end{equation}
The first term (bias) converges geometrically; the second term (variance floor) is irreducible with constant $\alpha$. Setting $\alpha = O(1/\sqrt{K})$ ensures $\mathbb{E}[J(\theta_K)] - J^{\star} = O(1/\sqrt{K})$ convergence at the cost of slower initial progress. For Benchmark A with $b=40$, $\mu \approx \lambda_{\min}(H) \approx 0.9$, $\sigma^2 \approx 1$, and $\alpha = 0.06$: the floor is $\approx 0.06/(2 \times 0.9 \times 40) \approx 8.3 \times 10^{-4}$, consistent with the convergence plots.
\end{proposition}

\begin{remark}[SGD Convergence on Quadratics]
For the quadratic loss $J(\theta) = \frac{1}{2m}\|X\theta - y\|^2$ with mini-batch SGD and constant step $\alpha$, the iterate satisfies:
\begin{equation}
  \mathbb{E}[J(\theta_k)] - J^{\star} \leq \underbrace{(1 - \alpha\mu)^k (J(\theta_0) - J^{\star})}_{\text{bias term}} + \underbrace{\frac{\alpha \sigma^2_b}{2\mu}}_{\text{variance floor}},
\end{equation}
where $\mu$ is the strong convexity constant and $\sigma^2_b = O(\sigma^2/b)$ is the gradient variance with batch size $b$. This shows that SGD with constant $\alpha$ converges to a \emph{variance floor} rather than the exact minimum; reducing $\alpha$ or increasing $b$ lowers this floor at the cost of slower convergence.
\end{remark}

\centering
\includegraphics[width=0.85\textwidth]{figures/q2_sgd_batch.pdf}
\caption{Q2: Mini-batch SGD with different batch sizes on Benchmark A. \subsection{Mini-Batch SGD: Effective Smoothness and Linear Scaling}

A subtle but important property of mini-batch SGD for linear models is the \emph{linear scaling rule}: if the batch size is multiplied by a factor $k$, the optimal learning rate should also be scaled by $k$. This arises because the effective smoothness $L_b$ of the mini-batch gradient for strongly convex problems satisfies $L_b \approx L/b$ (each mini-batch Hessian is approximately the full Hessian scaled by $b/m$), so the stability condition $\alpha \leq 1/L_b = b/L$ scales linearly. On Benchmark A with $\alpha = 0.06$ (tuned for $b = 40$), the linear scaling rule predicts $\alpha_{\mathrm{opt}}(b=5) \approx 0.06 \times 40/5 = 0.48$. In practice, the linear scaling rule breaks for very large batches ($b > m/10$ here) due to the saturation of gradient variance reduction, motivating warm-up schedules in large-batch deep learning. The variance floor Proposition~\ref{prop:sgd_conv} quantifies the trade-off: using $b=5$ vs $b=40$ with equal $\alpha$ trades an $8\times$ higher variance floor for $8\times$ more frequent parameter updates per epoch.

\subsection{Variance Reduction Methods}

A key limitation of SGD is the $O(\sigma_b^2)$ variance floor that prevents convergence to the exact minimum with constant step size (Proposition~\ref{prop:sgd_conv}). \emph{Variance reduction} methods eliminate this floor by periodically computing the full gradient as a reference point.

\begin{remark}[SVRG: Stochastic Variance Reduced Gradient]
The SVRG algorithm \cite{bottou2018optimization} alternates between: (1) computing the full gradient $\mu_s = \nabla J(\tilde\theta_s)$ at a snapshot point $\tilde\theta_s$ every $m$ iterations, and (2) taking stochastic steps with the variance-corrected gradient $\hat{g}_k = \nabla J_{\mathcal{B}}(\theta_k) - \nabla J_{\mathcal{B}}(\tilde\theta_s) + \mu_s$. The correction $-\nabla J_{\mathcal{B}}(\tilde\theta_s) + \mu_s$ has mean zero (unbiased) but \emph{reduced variance} when $\theta_k \approx \tilde\theta_s$, because the stochastic and full gradients cancel. SVRG achieves linear convergence $O(\rho^k)$ for strongly convex $J$ without requiring $\alpha \to 0$, at the cost of one full-gradient computation every $m$ stochastic steps. For Benchmark A ($m = 1000$ samples), SVRG with $m_{\mathrm{epoch}} = 1000$ would achieve the same $O((1-\mu/L)^k)$ rate as GD while processing only $\sim 2 \times$ more samples than standard SGD.
\end{remark}

\subsection{Theoretical Background}

Newton's method constructs a \emph{local quadratic model} of $f$ at each iterate $x_k$:
\begin{equation}
  q_k(p) = f(x_k) + \nabla f(x_k)^{\top} p + \tfrac{1}{2} p^{\top} \nabla^2 f(x_k)\, p,
\end{equation}
and steps to the minimiser of $q_k$: $x_{k+1} = x_k - [\nabla^2 f(x_k)]^{-1}\nabla f(x_k)$.

\begin{theorem}[Quadratic Convergence of Newton's Method]\label{thm:newton}
Suppose $f$ is twice continuously differentiable, $\nabla^2 f(x^{\star}) \succ 0$, and $\nabla^2 f$ is Lipschitz continuous with constant $M$. Then Newton's method with step size $\alpha=1$ converges \emph{locally quadratically}:
\begin{equation}
  \|x_{k+1} - x^{\star}\| \leq \frac{M}{2\lambda_{\min}(\nabla^2 f(x^{\star}))}\|x_k - x^{\star}\|^2,
\end{equation}
for $\|x_k - x^{\star}\|$ sufficiently small. This means the number of correct decimal digits doubles at each iteration.
\end{theorem}

\begin{proposition}[Exact Minimisation on Quadratics]\label{prop:newton_quad}
If $f(\theta) = \frac{1}{2}\theta^{\top} H \theta - b^{\top}\theta + c$ is a strictly convex quadratic with $H \succ 0$, then Newton's method with $\alpha = 1$ converges in \emph{exactly one iteration}: $\theta_1 = \theta_0 - H^{-1}(\nabla f(\theta_0)) = H^{-1}b = \theta^{\star}$.
\end{proposition}

\subsection{Local Approximations of $g(x) = x^4$}

\textbf{First-order (Taylor) approximation} around $x_0 = 0.25$:
\begin{equation}
  g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.0039 + 0.0625(x - 0.25).
\end{equation}
This tangent line captures the local slope but misrepresents the curvature.

\textbf{Second-order approximation:}
\begin{equation}
  g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}
The Newton update minimises this quadratic approximation, yielding:
\begin{equation}
  x_1 = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{4 \times 0.25^3}{12 \times 0.25^2} = 0.25 - \frac{0.0625}{0.75} \approx 0.167.
\end{equation}
This corresponds to stepping to the stationary point of the local quadratic model.

\subsection{Newton's Method Implementation}

\begin{algorithm}[H]
\caption{Damped Newton's Method}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0,1]$, regularisation $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Regularised (Tikhonov-damped) Hessian}
    \State Solve $H_k\, p_k = g_k$ for $p_k$ \Comment{Newton direction}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian regularisation $\lambda = 10^{-8}$ (essentially numerically zero, ensuring $H_k$ is invertible without significantly altering the Newton direction). The step size $\alpha < 1$ (damping) is used for Benchmarks B and C to handle regions where the Hessian is not positive definite or changes rapidly.

\begin{remark}[Trust-Region Interpretation of Damping]
The damped Newton update $x_{k+1} = x_k - \alpha[\nabla^2 f(x_k)]^{-1}\nabla f(x_k)$ can be understood via the \emph{trust-region} framework \cite{nocedal2006numerical}: it approximately solves
\begin{equation}
  \min_{p}\; \nabla f(x_k)^{\top} p + \tfrac{1}{2}p^{\top}\nabla^2 f(x_k)\,p \quad \text{s.t.}\ \|p\| \leq \Delta_k,
  \label{eq:trust_region}
\end{equation}
where $\Delta_k$ is implicitly set by $\alpha$. Smaller $\alpha$ restricts steps to a region where the quadratic model is trustworthy, providing \emph{global} convergence guarantees. Trust-region methods (e.g., Steihaug-CG) automatically adapt $\Delta_k$ based on the ratio of actual to predicted decrease, removing the need to hand-tune $\alpha$.
\end{remark}

\begin{remark}[Why Damping is Required on Rosenbrock]
At the starting point $x_0 = (-1, 1)$ for Rosenbrock, the Hessian is
\begin{equation}
  \nabla^2 f(-1, 1) = \begin{bmatrix} 2 + 1200\times 1 - 400\times 1 & 400 \\ 400 & 200 \end{bmatrix}
  = \begin{bmatrix} 802 & 400 \\ 400 & 200 \end{bmatrix}.
\end{equation}
The full Newton step $\alpha = 1$ would overshoot significantly because this local quadratic model is a poor approximation of the true function (which has $\kappa \approx 2009$ globally and the Hessian changes rapidly along the valley). The damping factor $\alpha = 0.22$ restricts the step to a region where the quadratic approximation remains reasonable, providing \emph{global} convergence guarantees at the cost of slower convergence than the undamped Newton step would provide near the solution.
\end{remark}

\subsection{Newton Decrement as a Stopping Criterion}

A key advantage of Newton's method over gradient methods is its natural stopping criterion: the \emph{Newton decrement}
\begin{equation}
  \lambda(x)^2 = \nabla f(x)^{\top}[\nabla^2 f(x)]^{-1}\nabla f(x) = p_k^{\top}\nabla^2 f(x_k) p_k,
  \label{eq:newton_decrement}
\end{equation}
where $p_k = [\nabla^2 f(x_k)]^{-1}\nabla f(x_k)$ is the Newton direction. The quantity $\lambda(x)^2/2$ is an estimate of $f(x) - f^{\star}$ and is invariant under affine coordinate transformations. The algorithm terminates when $\lambda(x_k)^2/2 \leq \varepsilon$ for a user-specified tolerance $\varepsilon$, providing a certificate of near-optimality. In contrast, gradient methods require comparing $\|\nabla f(x_k)\|$ to a tolerance, which is scale-dependent and may be too conservative or too aggressive depending on the problem geometry.

\subsection{Limitations of Newton's Method}

While Newton's method is superior in iteration count, it has important practical limitations:

\begin{enumerate}[nosep]
\item \textbf{Computational cost:} Forming $\nabla^2 f(x_k)$ costs $O(d^2)$ storage and $O(d^2)$ computation; solving the linear system costs $O(d^3)$. For $d \gg 2$ (modern deep learning has $d \sim 10^7$--$10^{11}$), this is completely intractable.

\item \textbf{Hessian non-convexity:} Newton's method requires $\nabla^2 f(x_k) \succ 0$. At saddle points or near indefinite Hessian regions (common in deep learning), the Newton direction may be an ascent direction. Hessian regularisation ($\lambda I$) or trust-region methods are required.

\item \textbf{Global vs.\ local convergence:} Theorem~\ref{thm:newton} is a \emph{local} result. Far from the optimum, Newton steps can be highly unstable (see the $\alpha = 0.22$ requirement on Rosenbrock). Quasi-Newton methods (e.g., L-BFGS) approximate $H^{-1}$ using gradient history, reducing cost to $O(d \cdot m)$ per step where $m$ is the memory window, balancing cost and convergence speed.

\item \textbf{Single iteration on quadratics:} This remarkable property (Proposition~\ref{prop:newton_quad}) holds only for exact quadratics. Even a slight non-quadratic perturbation (like the $\sin(x_1)$ term in Benchmark B) means Newton requires multiple iterations, though far fewer than GD.

\item \textbf{Quasi-Newton approximation:} The \emph{Broyden--Fletcher--Goldfarb--Shanno} (BFGS) and \emph{limited-memory BFGS} (L-BFGS) methods approximate $[\nabla^2 f(x_k)]^{-1}$ using $m$ recent gradient-iterate pairs $\{(s_j, y_j)\}_{j=k-m}^{k-1}$ where $s_j = x_{j+1} - x_j$ and $y_j = \nabla f(x_{j+1}) - \nabla f(x_j)$ \cite{nocedal2006numerical}. The approximation satisfies the secant condition $B_{k+1} s_k = y_k$ and preserves positive definiteness via the Sherman--Morrison--Woodbury formula. L-BFGS requires $O(m d)$ storage and $O(m d)$ per iteration (vs.\ Newton's $O(d^2)$/$O(d^3)$), making it the method of choice for large $d$ when second-order information is beneficial.
\end{enumerate}

\noindent\textbf{Practical recommendation (Q3):} Newton's method should be used when: (a) $d \leq 1000$ (Hessian storage feasible), (b) the objective is near-quadratic (few Newton steps needed), and (c) high accuracy is required. For larger $d$, L-BFGS with memory $m = 10$--30 provides a good approximation at $O(md)$ cost. Always use damping ($\alpha < 1$) or a trust-region radius for non-convex problems; never apply undamped Newton to an objective with non-positive-definite Hessian regions.

\subsection{Part A: Finite Differences and Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

When analytical gradients are unavailable, derivatives can be approximated numerically. The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad e_i \in \mathbb{R}^d \text{ is the $i$-th unit vector.}
  \label{eq:fd}
\end{equation}
By Taylor expansion, the truncation error is:
\begin{equation}
  \left|\frac{f(x + \delta e_i) - f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| \leq \frac{\delta}{2}\left|\frac{\partial^2 f}{\partial x_i^2}(x)\right| + O(\delta^2).
\end{equation}
Thus the approximation error is $O(\delta)$: larger $\delta$ gives larger truncation error. Combined with floating-point round-off error $O(\varepsilon_{\mathrm{mach}}/\delta)$, the optimal step size is $\delta^{\star} = \sqrt{\varepsilon_{\mathrm{mach}}} \approx 10^{-8}$. In these experiments, $\delta = 0.05$ is near-optimal while $\delta = 0.8$ introduces large truncation error because the function curvature changes significantly over a step of that size.

\begin{remark}[Central vs.\ Forward Finite Differences]
The \emph{central} finite-difference approximation,
\begin{equation}
  \frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x - \delta e_i)}{2\delta},
  \label{eq:central_fd}
\end{equation}
achieves $O(\delta^2)$ truncation error (versus $O(\delta)$ for forward differences~\eqref{eq:fd}), since the first-order error terms cancel by symmetry. This requires $2d$ function evaluations (vs.\ $d+1$) but allows using a step roughly $10^4$ times larger for the same accuracy. The optimal step for central differences is $\delta_{\mathrm{central}}^{\star} = \varepsilon_{\mathrm{mach}}^{1/3} \approx 10^{-5}$, which is less susceptible to floating-point cancellation. For the two-dimensional benchmarks in this report ($d=2$), central differences cost only one extra evaluation per iteration, making them preferable whenever accuracy matters.
\end{remark}

\subsubsection{Nesterov Random Search (Zeroth-Order Method)}

\begin{algorithm}[H]
\caption{Nesterov Gaussian Smoothing (Random Search) \cite{nesterov2017random}}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, smoothing parameter $\delta > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent This zeroth-order method estimates the gradient via a single random directional finite difference, requiring only \emph{two} function evaluations per iteration regardless of dimension $d$. The estimator $\hat{g}_k$ is an unbiased estimator of $\nabla f_\delta(x_k)$ where $f_\delta$ is a Gaussian-smoothed version of $f$ \cite{nesterov2017random}. The variance of $\hat{g}_k$ is $O(d)$, meaning convergence slows in high dimensions.

\begin{remark}[Connection to SGD]
The Nesterov random search method is structurally analogous to SGD (Section~\ref{sec:q2}): both replace the full gradient with a stochastic estimate, and both suffer from a variance term that limits convergence to a neighbourhood of the optimum for constant step sizes. The key difference is the source of randomness --- in SGD it is data sub-sampling (with $O(1/b)$ variance), while in random search it is direction sampling (with $O(d)$ variance). Dimension $d$ thus plays the role of inverse batch size in the convergence speed comparison.
\end{remark}

\subsection{Part B: Derivative-Free Methods on Rosenbrock (Benchmark C)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm \cite{nelder1965simplex} maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ and iteratively transforms it through four geometric operations:

\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): Move the worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2$): If reflection improved the best point, try a further step.
\item \textbf{Contraction} ($\rho = 0.5$): If reflection worsened things, contract toward the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): If contraction fails, shrink all vertices toward the current best.
\end{enumerate}

These operations require no gradient information; only function evaluations at the vertices. In the implementation, the initial simplex step is 0.35 and the algorithm runs for 160 iterations.

\begin{remark}[Why Nelder--Mead Succeeds on Rosenbrock]
Nelder--Mead's effectiveness on Rosenbrock ($f \approx 0.001$ at termination) despite lacking gradient information is explained by the \emph{adaptive simplex geometry}. Unlike fixed-stencil finite differences, the simplex can deform to match the curved valley: reflection rotates the simplex to align with the valley direction, expansion elongates it along the descent axis, and contraction resizes it when approaching the narrow optimum. This geometric adaptivity is equivalent to an implicit coordinate transformation --- the elongated simplex near the Rosenbrock valley effectively performs an anisotropic search aligned with the local curvature. On separable or axis-aligned functions, this offers no advantage over coordinate descent, but on curved manifolds like the Rosenbrock banana, it dramatically outperforms isotropic search strategies (such as random search with spherical perturbations, which requires $O(d\kappa)$ more iterations).
\end{remark}

\subsection{Coordinate Descent as a Structured Derivative-Free Alternative}

For problems with separable or block-separable structure, \emph{coordinate descent} minimises over one coordinate (or block) at a time:
\begin{equation}
  x_k^{(i)} = \arg\min_{t} f(x_k^{(1)}, \ldots, x_k^{(i-1)},\, t,\, x_k^{(i+1)}, \ldots, x_k^{(d)}),
\end{equation}
requiring only $O(1)$ function evaluations per step (for exact 1D minimisation) or one partial derivative (for gradient-based updates). On Benchmark B, coordinate descent would alternate between minimising over $x_1$ (solving $2(x_1-1) + \cos(x_1) = 0$ numerically) and $x_2$ (exact: $x_2 = 2$), converging in approximately two steps due to the near-separable structure. On Rosenbrock, coordinate descent performs poorly because the coupling term $100(x_2 - x_1^2)^2$ creates strong inter-coordinate dependence; progress along $x_1$ and $x_2$ are tightly coupled, making one-at-a-time minimisation inefficient. This illustrates when structured derivative-free methods outperform general-purpose ones: separability is a prerequisite.

\subsection{Nelder--Mead vs.\ Grid Search: A Direct Comparison}

On Benchmark C, Nelder--Mead with 160 iterations uses $\leq 160 \times 3 = 480$ function evaluations and achieves $f \approx 0.001$, far superior to grid search's best value ($f \approx 0.003$ from 3025 evaluations at resolution $\Delta x \approx 0.074$). This comparison highlights key trade-offs:

\begin{itemize}[nosep]
\item \textbf{Nelder--Mead}: adaptive, exploits problem structure, achieves good accuracy with few evaluations, but has no guarantee of finding the global optimum and may converge to local minima. On the unimodal Rosenbrock, it reliably finds near-global solutions.
\item \textbf{Grid search}: systematic, guarantees global coverage to within grid resolution, but requires $O(N^d)$ evaluations. Refining the grid from $55\times55$ to $550\times550$ (for $10\times$ better resolution) would require $100\times$ more evaluations (302,500 points), which is infeasible for even modest-dimensional problems.
\end{itemize}

In practice, for low-dimensional black-box optimisation without gradient information, Nelder--Mead or Bayesian optimisation (which models the objective as a Gaussian process) are vastly more efficient than grid search.

\subsection{Problem Formulation}

Minimise Benchmark B subject to a half-space constraint:
\begin{equation}
  \min_{x \in \mathbb{R}^2}\ f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)
  \quad \text{s.t.}\ x_1 \geq 0.5.
  \label{eq:q5_problem}
\end{equation}
Starting point $x_0 = (0.2, 4.0)$ is \emph{infeasible} ($x_{0,1} = 0.2 < 0.5$). The unconstrained minimiser $(0.582, 2.0)$ satisfies $x_1^{\star} = 0.582 > 0.5$, so the \textbf{constrained minimum coincides with the unconstrained minimum} and the constraint is inactive at the solution. This makes the problem a useful test case: it checks whether methods maintain feasibility without unnecessarily penalising the true optimum.

\subsubsection{KKT Conditions}

The Karush--Kuhn--Tucker (KKT) conditions for \eqref{eq:q5_problem} state that $x^{\star}$ is a local minimum if and only if there exists a multiplier $\mu^{\star} \geq 0$ such that:
\begin{align}
  \nabla f(x^{\star}) - \mu^{\star} e_1 &= 0, \label{eq:kkt_stationarity} \\
  \mu^{\star}(x^{\star}_1 - 0.5) &= 0 \quad (\text{complementary slackness}), \\
  x^{\star}_1 &\geq 0.5 \quad (\text{feasibility}).
\end{align}
Since the constraint $x_1 \geq 0.5$ is \emph{inactive} at $x^{\star} = (0.582, 2.0)$, complementary slackness gives $\mu^{\star} = 0$, and the stationarity condition reduces to $\nabla f(x^{\star}) = 0$, the unconstrained condition.

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$ (or infeasible), step size $\alpha > 0$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = \bigl(\max(0.5,\, z_{1}),\, z_{2}\bigr)$ \Comment{Project onto feasible set}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent For the half-space $\mathcal{X} = \{x : x_1 \geq 0.5\}$, the projection is simply $\Pi_{\mathcal{X}}(z) = (\max(0.5, z_1), z_2)$, a $O(1)$ operation. Projected GD maintains \emph{exact feasibility} at every iteration by construction.

\begin{proposition}[Projected GD Convergence \cite{boyd2004convex}]\label{prop:pgd_conv}
For a convex $L$-smooth function $f$ and closed convex set $\mathcal{X}$, projected gradient descent with step $\alpha = 1/L$ achieves:
\begin{equation}
  f\!\left(\bar{x}_K\right) - f^{\star} \leq \frac{L\|x_0 - x^{\star}\|^2}{2K} = O\!\left(\frac{1}{K}\right),
  \quad \bar{x}_K = \frac{1}{K}\sum_{k=0}^{K-1} x_k.
\end{equation}
If $f$ is additionally $\mu$-strongly convex on $\mathcal{X}$, the last iterate satisfies $\|x_k - x^{\star}\|^2 \leq (1-\mu/L)^k\|x_0-x^{\star}\|^2$, i.e., geometric convergence. In our experiment, $\mathcal{X}$ is a half-space and $f$ is strongly convex near the minimum, so projected GD converges geometrically once feasibility is established.
\end{proposition}

\begin{algorithm}[H]
\caption{Quadratic Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha > 0$, penalty weight $\lambda > 0$, iterations $N$
\State Define: $F_{\lambda}(x) \gets f(x) + \frac{\lambda}{2} \max(0,\, 0.5 - x_1)^2$ \Comment{Penalty function}
\For{$k = 0, 1, \ldots, N-1$}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{remark}
The implementation uses a linear penalty $\lambda \max(0, 0.5 - x_1)$ rather than quadratic, giving a gradient contribution $-\lambda \mathbf{1}_{x_1 < 0.5} \cdot e_1$ when violated. As $\lambda \to \infty$, the penalised minimiser converges to the constrained minimiser, but very large $\lambda$ introduces stiffness requiring smaller step sizes.
\end{remark}

\begin{proposition}[Penalty Method Convergence]\label{prop:penalty_conv}
Let $x^{\star}_\lambda$ denote the unconstrained minimiser of $F_\lambda(x) = f(x) + \lambda\max(0,\, c(x))$ where $c(x) = 0.5 - x_1$. Under mild regularity (LICQ, second-order sufficiency), as $\lambda \to \infty$:
\begin{equation}
  x^{\star}_\lambda \to x^{\star}, \qquad \lambda\,\max(0,\, c(x^{\star}_\lambda)) \to \mu^{\star},
\end{equation}
where $x^{\star}$ is the constrained minimiser and $\mu^{\star}$ is the optimal KKT multiplier. Here $\mu^{\star} = 0$ (constraint inactive), so $x^{\star}_\lambda \to (0.582, 2.0)$ and the residual constraint violation $\max(0, c(x^{\star}_\lambda)) \to 0$ for any finite $\lambda > 0$ large enough to steer the iterate past the boundary.
\end{proposition}

\subsection{Lagrangian Duality Perspective}

The penalty method can be understood through Lagrangian duality. The Lagrangian for \eqref{eq:q5_problem} is:
\begin{equation}
  L(x, \mu) = f(x) - \mu(x_1 - 0.5), \quad \mu \geq 0.
\end{equation}
The dual function is $d(\mu) = \min_x L(x, \mu)$, and strong duality holds (Slater's condition is satisfied since the constraint set is convex and the problem has a strictly feasible point). The penalty objective $F_\lambda(x) = f(x) + \lambda \max(0, 0.5 - x_1)$ can be interpreted as augmenting the Lagrangian: as $\lambda \to \infty$, the unconstrained minimum of $F_\lambda$ approaches the constrained minimum, recovering the primal solution. The penalty weight $\lambda$ approximates the optimal dual multiplier $\mu^{\star} = 0$ (since the constraint is inactive); the fact that $\lambda$ must be nonzero to steer from the infeasible initial point reflects the need to navigate through the infeasible region before the constraint becomes inactive.

\textbf{Projected GD} enforces feasibility exactly at every iteration via the projection oracle. This is the fundamental advantage: feasibility is a hard constraint, never traded off against objective improvement. The cost is requiring a projection oracle $\Pi_{\mathcal{X}}$, which is trivial ($O(1)$) for half-spaces but can be expensive ($O(d^3)$ or worse) for general convex sets.

\textbf{Penalty methods} convert the constrained problem to an unconstrained one, requiring only gradient evaluations of the augmented objective. The penalty weight $\lambda$ controls the feasibility-accuracy trade-off: small $\lambda$ under-penalises, large $\lambda$ introduces stiffness. In practice, $\lambda$ must be increased progressively to achieve both feasibility and optimality without numerical difficulties. The need to reduce $\alpha$ for large $\lambda$ (from 0.05 to 0.03) reflects this stiffness.

\begin{remark}[Stiffness and Step Size Reduction for Large $\lambda$]
The penalty-augmented objective $F_\lambda(x) = f(x) + \lambda\max(0, c(x))$ has effective smoothness constant $L_\lambda \approx L + \lambda$ when the constraint gradient $\nabla c$ is bounded. GD stability requires $\alpha \leq 1/L_\lambda \approx 1/(L + \lambda)$; tripling $\lambda$ from 1.5 to 4.5 therefore requires approximately tripling the denominator, hence the reduction in $\alpha$ from 0.05 to 0.03 seen in Table~\ref{tab:q5_params}. This stiffness issue is analogous to stiff ODEs, where $\lambda$ plays the role of the large eigenvalue forcing small time steps. Implicit methods (which solve the penalised step implicitly) avoid this restriction but require solving a nonlinear system per iteration --- exactly the motivation for augmented Lagrangian methods, which use a fixed moderate $\lambda$ and update the dual variable instead.
\end{remark}

\subsubsection{Augmented Lagrangian Method}
The \emph{augmented Lagrangian} (AL) method \cite{nocedal2006numerical} overcomes the stiffness limitation by combining a Lagrange multiplier estimate $\hat{\mu}$ with a quadratic penalty:
\begin{equation}
  L_\rho(x, \hat\mu) = f(x) - \hat\mu\,(x_1 - 0.5) + \frac{\rho}{2}\max\bigl(0,\, 0.5 - x_1 + \hat\mu/\rho\bigr)^2.
  \label{eq:aug_lag}
\end{equation}
Alternating minimisation over $x$ (with fixed $\hat\mu$) and dual ascent $\hat\mu \gets \hat\mu + \rho\,c(x_k)$ converges to the KKT point at a \emph{fixed} penalty $\rho$, without needing $\rho \to \infty$. This resolves the conditioning issue of pure penalty methods. For our problem with $\mu^{\star} = 0$, AL reduces to the pure penalty method with a well-chosen fixed $\rho$, but for problems with active constraints ($\mu^{\star} \neq 0$), AL is strictly superior. The dual update extracts the correct multiplier estimate, allowing accurate constraint satisfaction without ill-conditioning.

\subsection{Theoretical Background}

\begin{definition}[Linear Programme]
A linear programme (LP) minimises a linear objective over a convex polytope:
\begin{equation}
  \min_{x \in \mathcal{X}}\ a^{\top} x, \quad \mathcal{X} = \{x : Ax \leq b,\, l \leq x \leq u\}.
\end{equation}
\end{definition}

\begin{theorem}[LP Optimum at a Vertex]
The minimum of a linear objective over a bounded polytope $\mathcal{X}$ is always attained at a vertex (extreme point) of $\mathcal{X}$.
\end{theorem}

\begin{definition}[Frank--Wolfe Gap]
At iterate $x_k$, the Frank--Wolfe gap is $g_k = \nabla f(x_k)^{\top}(x_k - z_k)$ where $z_k = \arg\min_{z \in \mathcal{X}} \nabla f(x_k)^{\top} z$. This quantity upper-bounds the suboptimality: $f(x_k) - f^{\star} \leq g_k / \beta$, and $g_k = 0$ is a necessary condition for optimality. Note the connection to KKT stationarity conditions (cf.\ Section~\ref{sec:q5}): $g_k = 0$ means $\nabla f(x_k)^{\top}(x_k - z) \leq 0$ for all $z \in \mathcal{X}$, which is precisely the first-order optimality condition for constrained optimisation.
\end{definition}

\subsection{Feasible Set and Linear Programme}

Feasible set: $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ {-5} \leq x_2 \leq 10\}$.

\textbf{Linear Programme:} For $f(x) = a^{\top} x$ with $a = [1, 2]^{\top}$, the minimum over $\mathcal{X}$ is obtained by minimising each coordinate independently (since the feasible set is a box):
\begin{equation}
  x^{\star} = (0.5,\, {-5}), \qquad f^{\star} = 0.5 + 2({-5}) = {-9.5}.
\end{equation}
Since both components of $a$ are positive, the objective decreases with decreasing $x_i$, so the minimum is at the lower bounds of the box. This illustrates the vertex theorem: $x^{\star} = (0.5, -5)$ is a vertex of $\mathcal{X}$.

\begin{remark}[Frank--Wolfe Solves the LP in One Step]
Applied to the LP objective $f(x) = a^{\top}x$ with $a = [1,2]^{\top}$, Frank--Wolfe immediately computes $z_0 = \arg\min_{z \in \mathcal{X}} a^{\top} z = (0.5, -5)$ (the LP optimum), since $\nabla f = a$ is constant. The update $x_1 = \beta x_0 + (1-\beta) z_0$ places $x_1$ on the line segment from $x_0$ to $x^{\star}$; after $k$ steps, $x_k = \beta^k x_0 + (1 - \beta^k) x^{\star}$. Thus Frank--Wolfe converges at the geometric rate $O(\beta^k)$ for linear objectives, far better than its $O(1/k)$ guarantee for general smooth objectives. The LP is solved in one step of the linear minimisation oracle itself, with subsequent FW iterations simply interpolating toward $z_0$.
\end{remark}

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient) \cite{frank1956algorithm}}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^{\top} z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (step toward LMO solution)}
\EndFor
\end{algorithmic}
\end{algorithm}

\noindent For box constraints, the LP subproblem decouples per coordinate:
\begin{equation}
  (z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring $O(d)$ operations --- far cheaper than a general LP. The step $(1-\beta)$ toward $z_k$ plays the role of the learning rate; a smaller $(1-\beta)$ (larger $\beta$) means a more conservative step.

\begin{remark}
The standard Frank--Wolfe algorithm uses the diminishing step size $\gamma_k = 2/(k+2)$, which achieves the optimal $O(L/k)$ convergence rate for smooth convex objectives. The fixed-$\beta$ variant used here is a simplified implementation that converges but may oscillate near the optimum.
\end{remark}

\begin{theorem}[Frank--Wolfe Convergence Rate \cite{frank1956algorithm,jaggi2013revisiting}]
For a convex $L$-smooth function $f$ and compact convex set $\mathcal{X}$ with diameter $D = \max_{x,y \in \mathcal{X}}\|x-y\|$, the Frank--Wolfe algorithm with step size $\gamma_k = 2/(k+2)$ achieves:
\begin{equation}
  f(x_k) - f^{\star} \leq \frac{2L D^2}{k+2} = O\!\left(\frac{1}{k}\right).
\end{equation}
This $O(1/k)$ rate is slower than projected gradient descent's $O(1/k^2)$ but requires only a linear oracle rather than a projection.
\end{theorem}

\subsection{Interior vs.\ Boundary Behaviour and Comparison with Projected GD}

\textbf{Interior optimum:} Frank--Wolfe converges but exhibits characteristic zig-zag oscillation. The LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$ as the gradient near the optimum becomes small and its direction changes, and the convex combination averages these oscillations at rate $O(1/k)$.

\textbf{Boundary optimum:} Frank--Wolfe is well-suited to boundary optima because the LP oracle naturally selects the vertex closest to the constrained optimum. The iterate approaches the boundary without needing explicit projection. However, the fixed $\beta$ schedule prevents $x_k$ from reaching $f^{\star}$ exactly. To see why: with fixed $\beta = 0.93$, $x_k = \beta^k x_0 + (1-\beta^k)z^{\star}$ only if $z_k = z^{\star}$ at every step. In practice, near the boundary the gradient changes direction and $z_k$ oscillates, preventing convergence to $x^{\star} = (0.5, 0)$ exactly. The standard Frank--Wolfe schedule $\gamma_k = 2/(k+2)$ guarantees $O(LD^2/k)$ suboptimality with no such oscillation (the weights $\gamma_k \to 0$ suppress the vertex oscillation effect). For the boundary case, even after 140 iterations with $\beta = 0.93$, $f(x_{140}) = 0.364 > f^{\star} = 0.25$ (gap of 0.114), indicating the fixed-$\beta$ schedule requires approximately $140 \times (0.364/0.25 - 1) / (1 - 0.93) \approx 930$ more iterations to close this gap.

\textbf{Frank--Wolfe vs.\ Projected GD:} Projected GD achieves $O(1/k)$ ergodic convergence (or $O(1/k^2)$ with Nesterov acceleration) but requires a projection oracle. Frank--Wolfe achieves $O(1/k)$ using only a linear oracle, which is generally cheaper. The trade-off is particularly favourable when $\mathcal{X}$ is described by a complex constraint set (e.g., the nuclear norm ball, the simplex, flow polytopes) where projection is expensive but linear minimisation is efficient.

\begin{remark}[Frank--Wolfe Sparsity Property]
A distinctive feature of Frank--Wolfe is that each iterate $x_{k+1} = \beta x_k + (1-\beta) z_k$ is a convex combination of at most $k+1$ vertices of $\mathcal{X}$. For sparse-structure problems (e.g., optimisation over the $\ell_1$ ball), this means the solution is automatically sparse --- Frank--Wolfe implicitly performs sparse regularisation. This is exploited in LASSO, low-rank matrix completion, and deep learning sparsification.
\end{remark}

\begin{remark}[Away-Steps and Pairwise Frank--Wolfe]
The basic Frank--Wolfe method suffers from the ``zig-zag'' phenomenon for interior optima. The \emph{away-step} variant adds a step that moves \emph{away} from the worst active vertex, reducing the active set and allowing linear convergence on strongly convex problems. For the box-constrained problems in this report, this would involve removing vertices from the convex combination rather than always adding new ones, significantly improving convergence speed near interior optima.
\end{remark}

\noindent\textbf{Practical recommendation (Q6):} Use Frank--Wolfe when: (a) the feasible set has a cheap linear oracle (simplex, flow polytopes, nuclear norm balls), (b) sparse solutions are desired (the sparsity property), or (c) the problem is large-scale and projection is expensive. For moderate-scale problems where projections are tractable, Projected GD with Nesterov acceleration achieves $O(1/k^2)$ (better than FW's $O(1/k)$). The away-step Frank--Wolfe variant achieves linear convergence on strongly convex objectives and is recommended over the standard variant for problems where interior optima are expected.

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has systematically evaluated twelve optimisation algorithms across three benchmarks of increasing difficulty, yielding the following key conclusions:

\begin{enumerate}[leftmargin=1.5em]

\item \textbf{Adaptive step-size methods (Q1):} The Polyak step achieves geometric convergence when $f^{\star}$ is known exactly but diverges catastrophically under misspecification, limiting its practical utility. Adagrad and RMSprop are robust choices for anisotropic problems; RMSprop's exponential forgetting avoids Adagrad's pathological step-size decay. Heavy Ball provides effective acceleration when hyperparameters are well-tuned.

\item \textbf{Momentum and stochastic methods (Q2):} Nesterov acceleration achieves the optimal $O(1/k^2)$ convergence rate and is the strongest first-order method on Rosenbrock (achieving $7\times$ improvement over GD). Adam combines momentum and adaptive scaling effectively for well-conditioned and moderately ill-conditioned problems but is outperformed by pure Nesterov acceleration on highly ill-conditioned deterministic problems. Mini-batch SGD converges faster per epoch with smaller batches but introduces more gradient noise, a fundamental variance-speed trade-off.

\item \textbf{Newton's method (Q3):} Newton's method converges in a single iteration on quadratic objectives and achieves locally quadratic convergence near any non-degenerate minimum. It requires $O(d^3)$ per iteration for Hessian inversion but dramatically reduces iteration count, making it superior for low-to-medium-dimensional problems with expensive objective evaluations. Damping ($\alpha < 1$) is essential for globalisation on non-convex problems like Rosenbrock.

\item \textbf{Derivative-free methods (Q4):} Finite-difference gradients with appropriately chosen $\delta$ approximate exact gradients with $O(\delta)$ error; the choice $\delta \approx 10^{-4}$ achieves good accuracy. Random search scales independently of dimension (2 evaluations per step) but converges slowly. Nelder--Mead is highly effective in low dimensions, particularly on ill-conditioned problems, at the cost of lacking convergence guarantees. Grid search provides global search capability at exponential computational cost.

\item \textbf{Constrained optimisation (Q5):} Projected gradient descent enforces exact feasibility at every iteration and converges to the constrained minimum; it is the method of choice when a cheap projection oracle exists. Penalty methods offer algorithmic simplicity but require careful tuning of the penalty weight and may require smaller step sizes for large $\lambda$. The linear penalty approximation used here effectively enforces feasibility for $\lambda \geq 1.8$.

\item \textbf{Frank--Wolfe and linear programming (Q6):} The LP fundamental theorem guarantees optima at vertices of the feasible polytope. Frank--Wolfe solves a linear oracle per iteration (cheap for box constraints) and naturally handles boundary optima without projection, at the cost of $O(1/k)$ convergence and characteristic oscillation near interior optima. The fixed-$\beta$ step schedule should be replaced by a diminishing schedule $\gamma_k = 2/(k+2)$ for provably convergent behaviour.

\end{enumerate}

\begin{lstlisting}[caption={Gradient Descent}]
def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Heavy Ball (Polyak Momentum)}]
def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z     # look-ahead point
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)   # bias correction
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)   # Newton direction
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                      for i in range(len(g))])   # linear oracle
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # convex combination
        x_hist.append(x.copy()); f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
