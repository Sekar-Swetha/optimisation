\begin{abstract}
This report presents a systematic comparative study of six families of optimisation algorithms applied to three benchmark problems: a linear regression quadratic loss (Benchmark~A), a toy neural network loss with sinusoidal perturbation (Benchmark~B), and the Rosenbrock function (Benchmark~C). The methods investigated are: \textbf{(1)}~adaptive step-size methods including the Polyak rule, Adagrad, RMSprop, and Heavy Ball; \textbf{(2)}~momentum-based and stochastic methods including Nesterov acceleration, Adam, and mini-batch SGD; \textbf{(3)}~Newton's method exploiting second-order curvature; \textbf{(4)}~derivative-free methods (finite differences, Nesterov random search, Nelder--Mead, grid search); \textbf{(5)}~constrained optimisation via projected gradient descent and penalty methods; and \textbf{(6)}~the Frank--Wolfe conditional gradient algorithm for linear-constrained problems. Key findings include: Newton's method attains the quadratic loss minimum in a single iteration; the Polyak step size diverges whenever the specified optimum $f^\star$ underestimates the true minimum; Nesterov acceleration yields an $O(1/k^2)$ rate on convex problems versus the $O(1/k)$ rate of vanilla gradient descent; and for the constrained problem considered here the constraint is inactive at the unconstrained minimum, so projected gradient descent, penalty methods, and unconstrained gradient descent all converge to the same feasible optimum. Results are supported by convergence plots, trajectory visualisations, and quantitative comparisons.
\end{abstract}

\section{Benchmark Problems and Experimental Setup}
\label{sec:benchmarks}
% ============================================================

All experiments use a fixed random seed (\texttt{np.random.seed(42)}) for reproducibility. The three benchmark functions span qualitatively different optimisation landscapes: a well-conditioned convex quadratic (A), a mildly anisotropic near-quadratic with a sinusoidal perturbation (B), and the notoriously ill-conditioned nonconvex Rosenbrock function (C).

\subsection{Benchmark A: Linear Regression Quadratic Loss}

A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \overset{\text{i.i.d.}}{\sim} \mathcal{N}(0, 1).
\end{equation}
The mean squared error loss is:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \left(\theta^T x^{(i)} - y^{(i)}\right)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
The gradient and Hessian are:
\begin{equation}
\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y), \qquad \nabla^2 J = H = \frac{1}{m}X^T X.
\end{equation}
The Hessian $H \in \mathbb{R}^{2\times 2}$ is constant (independent of $\theta$), symmetric positive semidefinite, and numerically positive definite. The OLS minimiser $\theta_{\text{OLS}} = H^{-1}(X^T y / m)$ achieves $J(\theta_{\text{OLS}}) = J^\star \approx 0.4826$ (not zero due to irreducible noise). The condition number of $H$ is $\kappa(H) \approx 1.5$ in this instance, making Benchmark~A well-conditioned. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}

\begin{equation}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient and Hessian are:
\begin{equation}
\nabla f(x) = \begin{bmatrix} 2(x_1 - 1) + \cos(x_1) \\ 10(x_2 - 2) \end{bmatrix}, \qquad
\nabla^2 f(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The Hessian is diagonal with eigenvalues $\lambda_1(x_1) = 2 - \sin(x_1) \in [1, 3]$ and $\lambda_2 = 10$. The condition number varies in $[10/3, 10]$, creating mild anisotropy. Setting $\nabla f = 0$ yields $x_2^\star = 2$ and $x_1^\star \approx 0.5824$ (from $2(x_1 - 1) + \cos(x_1) = 0$), giving $f^\star \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}

\begin{equation}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is $(x^\star_1, x^\star_2) = (1, 1)$ with $f^\star = 0$. The gradient is:
\begin{equation}
\nabla f(x) = \begin{bmatrix} -2(1 - x_1) - 400 x_1(x_2 - x_1^2) \\ 200(x_2 - x_1^2) \end{bmatrix},
\end{equation}
and the Hessian is:
\begin{equation}
\nabla^2 f(x) = \begin{bmatrix} 2 + 1200 x_1^2 - 400 x_2 & -400 x_1 \\ -400 x_1 & 200 \end{bmatrix}.
\end{equation}
At the optimum, $\nabla^2 f(1,1) = \bigl[\begin{smallmatrix}802 & -400 \\ -400 & 200\end{smallmatrix}\bigr]$ with eigenvalues approximately $2$ and $1000$, giving condition number $\kappa \approx 500$. Combined with a narrow curved valley along $x_2 \approx x_1^2$, Rosenbrock is one of the standard hard test functions for optimisation.

The difficulty arises from two compounding effects: \textbf{(1)}~the large condition number forces gradient descent to use a tiny step size ($\alpha < 2/\lambda_{\max} \approx 0.001$), leading to slow progress; and \textbf{(2)}~the valley curvature changes strongly along the path, meaning the Hessian at the starting point is very different from the Hessian at the optimum, invalidating global quadratic models. Gradient methods following the $x_2$ direction rapidly (steep curvature $200$) stall in the $x_1$ direction (shallow curvature near $2$), producing the characteristic ``banana'' trajectory. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Benchmark D: Constrained Feasible Set}

For constrained optimisation (Questions~5--6):
\begin{equation}
\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}.
\end{equation}
This is a compact convex box, for which projection and linear minimisation are both $O(d)$ operations.

\section{Question 1: Adaptive Step-Size Methods \hfill [30 marks]}
\label{sec:q1}
% ============================================================

\subsection{Theoretical Background}

For gradient descent $x_{k+1} = x_k - \alpha_k \nabla f(x_k)$, the choice of step size $\alpha_k$ is critical. A fixed $\alpha < 2/L$ (where $L$ is the Lipschitz constant of $\nabla f$) guarantees convergence on $L$-smooth functions, but a single global constant is often either too conservative (slow convergence) or too aggressive (divergence). Adaptive methods address this by selecting $\alpha_k$ based on local information.

\begin{definition}
A function $f$ is \emph{$L$-smooth} if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x - y\|$ for all $x,y$. It is \emph{$\mu$-strongly convex} if $f(y) \geq f(x) + \nabla f(x)^T(y-x) + \frac{\mu}{2}\|y-x\|^2$ for all $x,y$.
\end{definition}

For Benchmark~A, $L = \lambda_{\max}(H)$ and $\mu = \lambda_{\min}(H)$, so gradient descent converges at rate $(1 - \mu/L)^k$, which is fast given the small condition number. For Benchmark~C, the effective $L$ near the starting point is large, forcing a very small fixed step size.

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, stabiliser $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$ \Comment{Optimal step for quadratic oracle}
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step minimises $f$ along the gradient direction under a subgradient model, setting $\alpha_k$ so that a perfect gradient step would reach $f^\star$ in one step. It achieves superlinear convergence for strongly convex functions \emph{when $f^\star$ is exact}, but is very sensitive to misspecification: if the supplied $f^\star < f_{\text{true}}^\star$, then as $x_k \to x^\star$ we have $f(x_k) \to f^\star_{\text{true}} > f^\star$, so $\alpha_k = (f(x_k)-f^\star)/\|g_k\|^2 \to \infty$ as $\|g_k\| \to 0$, causing divergence.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, stabiliser $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Coordinate-wise squared gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad adapts the learning rate per coordinate using the sum of all past squared gradients. Coordinates with historically large gradients receive smaller effective steps. The effective step size for coordinate $j$ at iteration $k$ is $\alpha_0/\sqrt{\sum_{t=0}^k g_{t,j}^2}$, which decays monotonically. This guarantees convergence on convex problems but causes the effective rate to vanish on non-stationary tasks.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay factor $\beta \in (0,1)$, stabiliser $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{EMA of squared gradients}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop replaces the full accumulation in Adagrad with an exponential moving average (decay $\beta = 0.9$), so that the effective rate responds to recent gradients and does not decay to zero. The parameter $\beta$ controls the memory horizon: $1/(1-\beta) = 10$ effective steps are remembered.

\begin{algorithm}[H]
\caption{Heavy Ball (Polyak Momentum)}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum coefficient $\beta$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$ \Comment{Momentum accumulation}
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

Heavy Ball introduces an exponential moving average of past gradients as a ``momentum'' term. For quadratic functions, the optimal heavy-ball parameters are $\alpha = 4/(\sqrt{L}+\sqrt{\mu})^2$ and $\beta = ((\sqrt{L}-\sqrt{\mu})/(\sqrt{L}+\sqrt{\mu}))^2$, achieving linear convergence with rate $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1)$, which is substantially faster than GD's rate $(\kappa-1)/(\kappa+1)$.

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^\star$ is known exactly, Polyak converges fastest (Rosenbrock: $f = 0.024$ at 120 iterations). Otherwise, Heavy Ball and Adagrad both match GD on Benchmarks~A and~B, while Heavy Ball outperforms GD on Benchmark~C ($f = 1.40$ vs $f = 3.51$) through momentum accumulation.
\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to the exponential moving average ($\beta = 0.9$, memory horizon $\approx 10$ steps). Heavy Ball shows characteristic momentum overshoot (visible in trajectory oscillation on Benchmark~B) followed by damped oscillation.
\item \textbf{Stability:} GD baseline is most stable but slowest. Adagrad is uniformly stable: since $G_k = \sum_{t=0}^k g_t^2$ is monotonically non-decreasing, the effective step $\alpha_k = \alpha_0/\sqrt{G_k}$ is monotonically non-increasing, guaranteeing no blow-up. Polyak is \emph{unstable} when $f^\star$ is misspecified.
\item \textbf{Adagrad vs RMSprop:} For Benchmark~A (stationary distribution, large $m$), Adagrad performs well as the accumulated gradients represent the true problem structure. RMSprop's advantage emerges in non-stationary settings (e.g., different phases of training) where old gradient information should be discarded.
\item \textbf{Practical implication:} In practice, exact $f^\star$ is rarely known, limiting Polyak's applicability. RMSprop and Adagrad are more robust general choices. Adam (Question~2) addresses both issues by combining momentum and per-coordinate adaptation with bias correction.
\end{itemize}

\section{Question 2: Momentum and Stochastic Methods \hfill [30 marks]}
\label{sec:q2}
% ============================================================

\subsection{Theoretical Background}

Standard gradient descent on $\mu$-strongly convex, $L$-smooth functions achieves a convergence rate of $(1 - \mu/L)^k$ per iteration. Nesterov~(1983) showed that this rate is \emph{not optimal} and proposed an accelerated scheme achieving the information-theoretic optimal rate of $O(1/k^2)$ for convex functions and $(1 - \sqrt{\mu/L})^k$ for strongly convex functions, doubling the exponent compared to GD.

\begin{theorem}[Nesterov Acceleration, informal]
For a convex $L$-smooth function $f$, the Nesterov accelerated gradient method with step size $\alpha = 1/L$ satisfies $f(x_k) - f^\star \leq \frac{2L\|x_0 - x^\star\|^2}{(k+1)^2}$, achieving $O(1/k^2)$ convergence versus $O(1/k)$ for GD.
\end{theorem}

Adam combines first-order momentum (Nesterov / Heavy Ball) with per-coordinate second-moment normalisation (RMSprop), providing both acceleration and adaptive scaling.

\begin{remark}[Heavy Ball vs Nesterov on Nonconvex Functions]
While Heavy Ball achieves the same $O(1/k^2)$ rate as Nesterov for strongly convex functions (under optimal parameter choice), it does \emph{not} guarantee acceleration on general convex or nonconvex functions. Heavy Ball computes the gradient at the current point $x_k$, while Nesterov computes it at the lookahead $\tilde{x}_k = x_k + \beta_k z_k$. The lookahead point is further ``downhill'' than $x_k$, so the gradient at $\tilde{x}_k$ provides a more ``corrective'' signal that prevents overshooting. This explains why on Rosenbrock (strongly nonconvex), Nesterov significantly outperforms Heavy Ball (Table~\ref{tab:q2_final}).
\end{remark}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Momentum / Acceleration}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, maximum momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Momentum schedule: $\beta_k \to 1$ at rate $O(1/k)$}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead point}
    \State $g_k \gets \nabla f(\tilde{x}_k)$ \Comment{Gradient evaluated at lookahead}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, g_k$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at the predicted position $\tilde{x}_k = x_k + \beta_k z_k$ rather than the current $x_k$. This ``look before you leap'' strategy allows Nesterov's method to correct its trajectory before overshooting, and is what enables the improved $O(1/k^2)$ convergence rate. The adaptive schedule $\beta_k = (k-1)/(k+2)$ matches the schedule from the $O(1/k^2)$ analysis; $\beta_{\max}$ prevents instability in the early nonlinear regime.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, exponential decay rates $\beta_1, \beta_2 \in (0,1)$, stabiliser $\varepsilon$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (momentum)}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (adaptive scaling)}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction for initialisation at zero}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam combines momentum ($m_t$) with per-coordinate adaptive scaling ($v_t$), inheriting advantages of both RMSprop and heavy-ball-style momentum methods. The bias correction terms $(1 - \beta_1^t)^{-1}$ and $(1 - \beta_2^t)^{-1}$ prevent the early iterates from being biased towards zero (since both moments are initialised at zero). Default parameters $\beta_1 = 0.9$, $\beta_2 = 0.999$, $\varepsilon = 10^{-8}$ are standard; the slightly lower $\beta_1 \in \{0.80, 0.82\}$ used here reduces momentum to improve stability on the nonconvex Rosenbrock.

\subsection{Mini-Batch Stochastic Gradient Descent}
\label{subsec:sgd}

Mini-batch SGD on Benchmark~A uses $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at every epoch. The gradient estimator for mini-batch $\mathcal{B}$ of size $b$ is:
\begin{equation}
\hat{g}(\theta) = \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}}).
\end{equation}
This is an unbiased estimator of the full gradient $\nabla J(\theta)$, with variance $\text{Var}[\hat{g}] \propto \sigma^2_{\text{noise}} / b$. Larger batches reduce variance but perform fewer updates per epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y)$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute data indices $\{1,\ldots,m\}$
    \For{each mini-batch $\mathcal{B}$ of size $b$}
        \State $g \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$
        \State $\theta \gets \theta - \alpha\, g$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

\subsection{Local Approximations of $g(x) = x^4$}

Newton's method is motivated by second-order Taylor approximations. Given $g: \mathbb{R} \to \mathbb{R}$ with $g(x) = x^4$, the Taylor expansion around $x_0 = 0.25$ is:

\textbf{First-order approximation} (linear, used by gradient descent):
\begin{equation}
g^{(1)}(x) = g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4\cdot 0.25^3(x - 0.25) = 0.003906 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order approximation} (quadratic, used by Newton's method):
\begin{equation}
g^{(2)}(x) = g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2 = 0.003906 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton update $x_{k+1} = x_k - g'(x_k)/g''(x_k)$ minimises $g^{(2)}$, yielding the stationary point of the local quadratic model. For $g(x) = x^4$: $x_{k+1} = x_k - 4x_k^3/(12x_k^2) = x_k - x_k/3 = \frac{2}{3}x_k$. Starting from $x_0 = 0.25$, Newton converges at rate $|\frac{2}{3}|$ (linear here, since $g'(0) = 0$ is a degenerate root). For functions with non-degenerate minima ($g''(x^\star) > 0$), Newton achieves quadratic convergence.

\subsection{Newton's Method for Multivariate Optimisation}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, regularisation $\lambda_\text{damp}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda_\text{damp} I$ \Comment{Regularised Hessian for numerical stability}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction: solves $H_k p_k = g_k$}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The damping $\lambda_\text{damp} = 10^{-8}$ ensures $H_k$ is numerically positive definite, preventing the solve from failing when the Hessian has small eigenvalues. The damped Newton step ($\alpha < 1$) is used on Benchmarks B and C where the Hessian changes rapidly, and a full Newton step ($\alpha = 1$) would overshoot.

\begin{theorem}[Newton's Quadratic Convergence, informal]
Near a non-degenerate local minimum $x^\star$ (where $\nabla^2 f(x^\star) \succ 0$), Newton's method with $\alpha = 1$ converges quadratically: $\|x_{k+1} - x^\star\| \leq C \|x_k - x^\star\|^2$ for some constant $C > 0$. This means the number of correct digits doubles at each iteration.
\end{theorem}

\subsection{Why Newton Outperforms Gradient Descent: Theoretical Insight}

The efficiency gain of Newton's method over gradient descent can be understood geometrically. Gradient descent moves in the direction of steepest descent in the Euclidean metric, which is optimal for the ball $\{y : \|y-x\|_2 \leq r\}$ but suboptimal for the ``ellipsoid'' of the Hessian. Newton's method moves in the direction of steepest descent in the \emph{Hessian metric} $\|y-x\|_H = \|y-x\|_{\nabla^2 f(x)}$, which is the natural geometry of the problem near $x$.

Concretely, for a quadratic $f(x) = \frac{1}{2}x^T H x - b^T x$, the gradient descent iteration $x_{k+1} = x_k - \alpha H x_k + \alpha b$ has convergence factor $\rho = 1 - \alpha \lambda_{\min}(H)$ if $\alpha < 2/\lambda_{\max}(H)$, leading to a worst-case factor $\rho_\text{GD} = (\kappa-1)/(\kappa+1)$. Newton's iteration with $\alpha = 1$ gives $x_1 = H^{-1}b$ exactly in one step, regardless of condition number.

This \emph{affine invariance} property --- that Newton's method is unaffected by linear transformations of the variable space --- is its most powerful attribute. It explains why Newton succeeds so dramatically on Benchmark~A (quadratic, exact in 1 step) and still provides significant benefits on Benchmarks~B and~C despite the non-quadratic structure.

\subsection{Computational Cost Analysis}

Newton's method requires computing and solving a $d \times d$ linear system at each iteration, with cost $O(d^2)$ for the Hessian and $O(d^3)$ for the solve ($O(d^{2.37})$ asymptotically). For $d = 2$, this is negligible, but for large neural networks ($d \sim 10^8$), storing the Hessian alone requires $O(d^2)$ memory --- far beyond feasibility. Quasi-Newton methods (L-BFGS) approximate $H^{-1}$ using only gradient information at $O(md)$ memory for $m$ stored vectors, providing a practical middle ground between first and second-order methods. In the context of this assignment ($d = 2$), Newton's overhead is trivial and the quadratic convergence rate is the dominant factor.

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark~B)}

\subsubsection{Finite Difference Gradient Approximation}

When analytical gradients are unavailable (black-box functions, simulation outputs, non-differentiable objectives), the gradient can be approximated using function evaluations. The forward finite-difference approximation replaces each partial derivative with:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}, \quad i = 1,\ldots,d.
\end{equation}
This requires $d + 1$ function evaluations per gradient estimate. The approximation error has two competing components:
\begin{equation}
\left|\frac{f(x+\delta e_i)-f(x)}{\delta} - \frac{\partial f}{\partial x_i}(x)\right| \leq \frac{L_2 \delta}{2} + \frac{\varepsilon_\text{machine}}{\delta},
\end{equation}
where $L_2$ bounds the second derivative and $\varepsilon_\text{machine} \approx 10^{-16}$ is floating-point machine precision. The optimal $\delta^\star = \sqrt{2\varepsilon_\text{machine}/L_2} \approx 10^{-8}$. The experiments test $\delta = 0.05$ (near-optimal for this problem) and $\delta = 0.8$ (too large, dominant truncation error).

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov Random Search (Zeroth-Order)}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, perturbation $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$, normalise: $u_k \gets u_k / \|u_k\|$ (uniform on sphere)
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta u_k) - f(x_k)}{\delta}\, u_k$ \Comment{Smoothed directional derivative}
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

This zeroth-order method estimates the directional derivative along a random direction $u_k$, requiring only \emph{two function evaluations} per iteration regardless of dimension $d$ --- cost is dimension-independent. The key property is that $\mathbb{E}[\hat{g}_k \mid x_k]$ provides an unbiased estimate of the \emph{smoothed gradient} $\nabla f_\delta(x_k) = \mathbb{E}_{u \sim \text{Unif}(\mathbb{S}^{d-1})}[f(x_k+\delta u) \cdot u] / \delta$, which converges to $\nabla f(x_k)$ as $\delta \to 0$ for differentiable $f$. The tradeoff is statistical: in $d = 2$ dimensions, the signal-to-noise ratio of $\hat{g}_k$ relative to $\nabla f$ is $O(1/\sqrt{d})$, so Nesterov random search is more competitive in low dimensions.

\subsection{Part B: Derivative-Free Methods on Benchmark~C}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices (for $d=2$) and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): Move the worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2$): If reflection improved significantly, try a further step beyond the centroid.
\item \textbf{Contraction} ($\rho = 0.5$): If reflection did not improve, contract the worst vertex towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): If contraction also failed, contract all vertices towards the best.
\end{enumerate}
No gradient information is required; convergence is achieved purely through the adaptive geometry of the simplex. The initial simplex step size is $0.35$, and 160 iterations are performed.

The fundamental trade-off in derivative-free optimisation is between \emph{computational cost per step} and \emph{quality of the descent direction}. Methods with accurate gradient estimates converge faster per step but require more function evaluations. Nesterov random search is the extreme case: minimal evaluations per step but slow convergence per step. Grid search is the opposite extreme: guaranteed global coverage but combinatorially expensive.

\textbf{Dimension dependence:} This is the critical consideration for large-scale problems. FD gradient requires $d+1$ evaluations (scales linearly with dimension). Nesterov random search requires only 2 evaluations per step regardless of $d$, but pays a statistical penalty: the directional derivative along $u_k$ is a noisy estimate of $\nabla f \cdot u_k$, and the variance grows with $d$ because the component $\|u_k\| = 1$ means each coordinate gets $O(1/\sqrt{d})$ of the update. The effective convergence rate for Nesterov random search in dimension $d$ is $O(d/k)$ for convex functions, versus $O(1/k)$ for full gradient descent --- a $d$-fold penalty.

\textbf{Nelder--Mead limitations:} While effective for $d \leq 10$, Nelder--Mead has no global convergence guarantee even for smooth functions, and empirically fails to converge on many problems in $d > 10$. The standard MATLAB \texttt{fminsearch} is based on Nelder--Mead for exactly this reason: it handles low-dimensional problems robustly without requiring gradient information.

\subsection{Problem Setup and Analysis}

The problem is to minimise Benchmark~B subject to $x_1 \geq 0.5$, starting from the infeasible initial point $x_0 = (0.2, 4.0)$:
\begin{equation}
\min_{x \in \mathbb{R}^2}\; f(x_1, x_2) = (x_1-1)^2 + 5(x_2-2)^2 + \sin(x_1) \quad \text{subject to}\quad x_1 \geq 0.5.
\end{equation}

\textbf{Key observation (KKT analysis):} The unconstrained minimum of $f$ satisfies $\nabla f(x^\star) = 0$, giving $x^\star \approx (0.5824, 2.0)$. Since $x^\star_1 \approx 0.5824 > 0.5$, the inequality constraint is \emph{inactive} at the optimum. By the Karush--Kuhn--Tucker (KKT) conditions for the problem $\min_{x_1 \geq 0.5} f(x)$:
\begin{equation}
\nabla f(x^\star) - \mu^\star \mathbf{e}_1 = 0, \quad \mu^\star \geq 0, \quad \mu^\star (x_1^\star - 0.5) = 0 \quad \text{(complementary slackness)}.
\end{equation}
Since $x_1^\star > 0.5$ (strict feasibility), complementary slackness forces $\mu^\star = 0$, so $\nabla f(x^\star) = 0$ --- exactly the unconstrained stationarity condition. Therefore, $x^\star_\text{constrained} = x^\star_\text{unconstrained} \approx (0.5824, 2.0)$, with $f^\star \approx 0.7244$.

This means all constrained methods (if they converge) will find the same solution as unconstrained GD. The differences among the methods are entirely in the convergence speed and path, particularly in how quickly they escape the infeasible initial region.

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1})$ \Comment{Projection: $x_1 \leftarrow \max(0.5,\, x_1)$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, 0.5 - x_1)$ \Comment{Exact penalty on constraint violation}
    \State $x_{k+1} \gets x_k - \alpha\, \nabla_x F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

where $\nabla_x F_\lambda(x) = \nabla f(x) + \lambda \nabla_x \max(0, 0.5-x_1) = \nabla f(x) - \lambda \mathbf{e}_1 \cdot \mathbf{1}[x_1 < 0.5]$.

Settings: Projected GD ($\alpha = 0.08$), Penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$), unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\subsection{Feasible Set and Linear Programme}

The feasible set is $\mathcal{X} = \{x \in \mathbb{R}^2 : 0.5 \leq x_1 \leq 5,\ -5 \leq x_2 \leq 10\}$, a compact convex box with four vertices: $(0.5,-5)$, $(5,-5)$, $(0.5,10)$, $(5,10)$.

\textbf{Linear Programme (Part I):} Minimise $f(x) = a^T x = x_1 + 2x_2$ over $\mathcal{X}$.

\begin{proposition}
For a linear objective $f(x) = a^T x$ over a bounded polytope $\mathcal{X}$, the minimum is attained at a vertex. Since each coefficient $a_i > 0$, the minimum over the box occurs at the lower bound in each coordinate:
\begin{equation}
x^\star = (0.5,\; -5), \qquad f^\star = 0.5 + 2(-5) = -9.5.
\end{equation}
\end{proposition}

This illustrates the fundamental theorem of linear programming: the optimum of a linear objective over a bounded polytope lies at a vertex (extreme point).

The parallel contour lines sweep across the box, and the minimum is attained at the lower-left corner $(0.5, -5)$ where $f = -9.5$. \subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{z \in \mathcal{X}}\, \nabla f(x_k)^T z$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination: $\gamma_k = 1-\beta$ is step towards $z_k$}
\EndFor
\end{algorithmic}
\end{algorithm}

The LMO for a box constraint decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
This requires only $O(d)$ operations --- no projection, no linear system solve. The iterate $x_{k+1}$ is always a convex combination of feasible points, so $x_{k+1} \in \mathcal{X}$ automatically.

\begin{theorem}[Frank--Wolfe Convergence Rate]
For a convex $L$-smooth function $f$ on compact convex $\mathcal{X}$ with diameter $D$, the Frank--Wolfe algorithm with $\gamma_k = 2/(k+2)$ achieves $f(x_k) - f^\star \leq \frac{2LD^2}{k+2} = O(1/k)$.
\end{theorem}

Here we use a fixed $\gamma_k = 1-\beta$ (constant step), which also converges at $O(1/k)$ but with a different constant. The fixed-step version does not require knowing $L$ or the diameter $D$.

\subsection{Interior Optimum: $f(x) = (x_1 - 1)^2 + (x_2 - 5)^2$}

The unconstrained minimum $(1, 5)$ lies strictly \emph{inside} $\mathcal{X}$ (since $0.5 < 1 < 5$ and $-5 < 5 < 10$). Two mixing parameters are tested: $\beta = 0.90$ (larger step $\gamma = 0.10$) and $\beta = 0.985$ (smaller step $\gamma = 0.015$).

\subsection{Interior vs Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits a characteristic ``zig-zag'' pattern because the LP vertex $z_k$ jumps between extreme points of $\mathcal{X}$, and the convex combination averages these over time. Near the interior optimum, the gradient is small, but the LMO still selects a vertex far from $x^\star$, causing oscillation. The $O(1/k)$ rate is fundamental: it cannot be improved for the general Frank--Wolfe method on convex problems (tight lower bound).

\textbf{Boundary optimum:} Frank--Wolfe is \emph{particularly} well-suited to boundary optima. The LMO naturally selects the vertex closest to the constrained optimum ($z_k = (0.5, -5)$ then $(0.5, 10)$ alternating, but $x_1 \to 0.5$ monotonically through the convex combination). This is the key advantage over projected gradient descent for complex constraint sets: the LMO is a linear minimisation problem (easy for box, simplex, spectahedron), while the projection can be expensive (e.g., projecting onto the spectahedron requires a full eigendecomposition). The fixed-$\beta$ schedule does not allow the step size to decay at the $O(1/k)$ rate; using the standard $\gamma_k = 2/(k+2)$ schedule would improve convergence. Additionally, Frank--Wolfe iterates are \emph{sparse}: since $z_k$ is always a vertex, $x_{k+1}$ is a convex combination of vertices, producing iterates that are sparse in the simplex sense --- a desirable property for sparsity-promoting constraints like the $\ell_1$ ball or the nuclear-norm ball.

\textbf{Frank--Wolfe gap:} A useful primal-dual gap measure is the Frank--Wolfe gap: $g_k = \nabla f(x_k)^T(x_k - z_k) \geq 0$. When $g_k \leq \varepsilon$, the current point is an $\varepsilon$-approximate stationary point. The standard convergence result guarantees $\min_{k \leq T} g_k \leq O(LD^2/T)$, providing a convergence certificate without needing to know $f^\star$.

\subsection{Connecting to Optimisation Theory}

The experimental results confirm several key theoretical predictions from the CS7DS2 course:

\begin{enumerate}[nosep]
\item \textbf{Newton's quadratic convergence} (Section~\ref{sec:q3}): The one-step exact convergence on Benchmark~A is a direct consequence of the affine invariance of Newton's method on quadratics: $x_1 = x_0 - H^{-1}\nabla J(x_0) = \theta_\text{OLS}$ exactly. On non-quadratic problems, Newton's quadratic rate near the minimum enables dramatic speedups even with damping.

\item \textbf{Nesterov's $O(1/k^2)$ rate} (Section~\ref{sec:q2}): The $7.1\times$ improvement of Nesterov over GD on Rosenbrock after 150 iterations ($f = 0.479$ vs $f = 3.388$) is consistent with the improved convergence rate. The lookahead mechanism (gradient at $\tilde{x}_k$ rather than $x_k$) is what enables this rate improvement.

\item \textbf{Polyak divergence} (Section~\ref{sec:q1}): The catastrophic failure of Polyak on Benchmarks~A and~B when $f^\star = 0$ is misspecified ($f^\star_\text{true} > 0$) directly illustrates the theoretical analysis: $\alpha_k = (f(x_k) - f^\star)/\|g_k\|^2 \to \infty$ as $\|g_k\| \to 0$ while $f(x_k) \not\to f^\star$.

\item \textbf{Adagrad monotone decay} (Section~\ref{sec:q1}): The step size plots (Figure~\ref{fig:q1_stepsize_A}) directly confirm the $O(1/\sqrt{k})$ decay rate of the Adagrad effective step, matching the theoretical analysis.

\item \textbf{Frank--Wolfe $O(1/k)$ rate} (Section~\ref{sec:q6}): The boundary case converges to $f = 0.364$ versus $f^\star = 0.25$ after 140 iterations, consistent with the sub-linear $O(1/k)$ guarantee that requires many more iterations than gradient-based methods near the optimum.

\item \textbf{KKT inactivity} (Section~\ref{sec:q5}): When the constraint is inactive at the optimum ($x^\star_1 \approx 0.582 > 0.5$), all constrained methods converge to the unconstrained solution, as predicted by the KKT conditions. The Lagrange multiplier $\mu^\star = 0$ at the optimum, consistent with complementary slackness.

\item \textbf{Condition number and convergence speed} (Sections~\ref{sec:q1}--\ref{sec:q3}): The most dramatic differences between methods appear on Benchmark~C (Rosenbrock, $\kappa \approx 500$), while Benchmark~A (condition number $\approx 1.5$) shows little discrimination between methods. This confirms the theoretical prediction that method differences are most pronounced when $\kappa \gg 1$.
\end{enumerate}

\subsection{Final Remarks}

This study demonstrates that no single optimisation method dominates across all problem types. The choice of method should be guided by:
\begin{itemize}[nosep]
\item \textbf{Problem structure:} Is the function convex? Is the Hessian available? Is the problem smooth?
\item \textbf{Computational budget:} How many function/gradient evaluations are affordable?
\item \textbf{Available information:} Is $f^\star$ known? Are gradients analytically available?
\item \textbf{Constraints:} Are there projectable constraints (favoring projected GD) or complex constraints (favoring Frank--Wolfe)?
\end{itemize}
For the data analysis setting (large datasets, moderate dimensions, smooth losses), Adam and Nesterov acceleration represent the state-of-the-art, offering the best balance of convergence speed, robustness, and computational cost per iteration.

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

\begin{lstlisting}[caption={Polyak Step Size}]
def polyak_step(grad_fn, loss_fn, x0, f_star, eps, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        g = grad_fn(x)
        z = beta * z + alpha * g
        x = x - z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Nesterov Momentum}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex}]
def nelder_mead(loss_fn, x0, step, n_iters,
                alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1])
        f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1])
            f_e = loss_fn(x_e)
            simplex[-1] = (x_e if f_e < f_r else x_r)
            f_vals[-1] = (f_e if f_e < f_r else f_r)
        else:
            x_c = centroid + rho * ((x_r if f_r < f_vals[-1] else simplex[-1]) - centroid)
            f_c = loss_fn(x_c)
            if f_c <= min(f_r, f_vals[-1]):
                simplex[-1] = x_c; f_vals[-1] = f_c
            else:
                for i in range(1, n + 1):
                    simplex[i] = simplex[0] + sigma * (simplex[i] - simplex[0])
                    f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = fw_lp_box(g, bounds)  # Linear minimisation oracle
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z   # Convex combination
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)

def fw_lp_box(grad, bounds):
    z = np.zeros(len(grad))
    for i in range(len(grad)):
        z[i] = bounds[i][0] if grad[i] > 0 else bounds[i][1]
    return z
\end{lstlisting}
