\begin{abstract}
This report presents a comprehensive comparative study of optimisation algorithms applied to three benchmark functions of increasing difficulty: a linear regression quadratic loss (Benchmark A, $\kappa \approx 1.12$), a toy neural network loss with sinusoidal perturbation (Benchmark B, $\kappa \approx 6.9$), and the Rosenbrock function (Benchmark C, $\kappa \approx 2508$). Six categories of methods are systematically investigated: adaptive step-size methods (Polyak, Adagrad, RMSprop, Heavy Ball); momentum-based accelerated methods (Nesterov, Adam, mini-batch SGD, SVRG); Newton's method with second-order local approximations and quasi-Newton variants; derivative-free approaches (finite differences, complex-step, Nesterov random search, Nelder--Mead simplex, grid search); constrained optimisation via projected gradient descent and penalty/augmented Lagrangian methods; and the Frank--Wolfe conditional gradient algorithm. Each method is evaluated with respect to convergence rate guarantees, per-iteration computational cost, numerical stability, and practical performance. Key findings include: (i) the Polyak step size diverges when $f^{\star}$ is misspecified but achieves the best Rosenbrock result ($f = 0.0014$ at 200 iterations) when $f^{\star} = 0$ is exact; (ii) Nesterov's theoretically optimal $O(1/k^2)$ rate is confirmed empirically, outperforming Adam on Rosenbrock due to Adam's coordinate-wise normalisation suppressing progress along the valley floor; (iii) Newton's method solves the quadratic Benchmark A in exactly one iteration (Theorem~\ref{thm:newton_quadratic}) and achieves near-quadratic convergence on Benchmark B; (iv) finite differences at $\delta^{\star} = \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ minimise combined truncation and round-off error; (v) the Frank--Wolfe duality gap provides a computable certificate of near-optimality without knowing $f^{\star}$; and (vi) KKT conditions confirm the inactive constraint at Benchmark B's optimum (Proposition~\ref{prop:inactive_constraint}), so projected GD and penalty methods converge to the same point. All results are implemented in Python with \texttt{NumPy} and are fully reproducible with seed \texttt{np.random.seed(42)}.
\end{abstract}

\section{Introduction}
\label{sec:intro}
% ============================================================

Optimisation is the mathematical backbone of modern machine learning and data analysis. The central task is to find a parameter vector $x^{\star}$ minimising an objective function $f: \mathbb{R}^d \to \mathbb{R}$, possibly subject to constraints. The choice of algorithm significantly affects convergence speed, accuracy, and computational cost. Different problem structures --- convexity, smoothness, conditioning, availability of gradients, and presence of constraints --- favour different methods. Understanding when and why each method works is as important as knowing how to run it.

This report investigates six families of algorithms over three carefully chosen benchmark functions designed to expose complementary algorithmic strengths and weaknesses. The benchmark difficulty is stratified by condition number: \textbf{Benchmark A} (linear regression, $\kappa \approx 1.12$) is a nearly unit-conditioned convex quadratic with closed-form solution, ideal for demonstrating exact Newton convergence and linear GD rates. \textbf{Benchmark B} (toy neural network loss, $\kappa \approx 6.9$ at the minimum) is a smooth non-quadratic with moderate anisotropy (Hessian eigenvalue ratio $10/(2-\sin x_1) \in [3.3, 10]$ across the domain), suitable for comparing adaptive and momentum-based methods. \textbf{Benchmark C} (Rosenbrock, $\kappa \approx 2508$) is a non-convex function with a narrow curved valley, where the Hessian eigenvectors are nearly aligned with $[1,1]^T$ and $[1,-1]^T$ rather than the coordinate axes --- creating the worst-case scenario for coordinate-aligned methods. The Rosenbrock's extreme conditioning ($\lambda_{\max}/\lambda_{\min} \approx 2508$) requires either second-order information or carefully tuned momentum to make progress efficiently.

All experiments are implemented in Python with \texttt{NumPy} (seed \texttt{np.random.seed(42)} for reproducibility). The complete code is provided in Appendix~\ref{app:code}.

\section{Benchmark Problems and Experimental Setup}

\begin{remark}[Gradient Descent as Euler Discretisation of Gradient Flow]
The gradient descent update $x_{k+1} = x_k - \alpha \nabla f(x_k)$ is the Euler method applied to the continuous-time \emph{gradient flow} ODE: $\dot{x}(t) = -\nabla f(x(t))$. This ODE describes the steepest descent path of $f$ and converges to a local minimum for any smooth objective. The step size $\alpha$ acts as the time step: large $\alpha$ gives faster ``virtual time'' progress but risks instability (analogous to stiff ODE solvers diverging for large time steps). The stability condition $\alpha < 2/L$ corresponds to the Euler stability region for this ODE. This physical interpretation clarifies why all first-order methods are fundamentally limited by the condition number $\kappa = L/\mu$: the gradient flow converges at rate $e^{-\mu t}$, and Euler with time step $\alpha$ requires $k \approx t/\alpha \approx \kappa \log(1/\epsilon) / 2$ steps to achieve $\epsilon$-accuracy.
\end{remark}

\subsection{Benchmark A: Linear Regression Quadratic Loss}
A synthetic dataset with $m = 1000$ samples and $d = 2$ parameters is generated:
\begin{equation}
\theta^{\star} = \begin{bmatrix} 3 \\ 4 \end{bmatrix}, \quad x^{(i)} \sim \mathcal{N}(0, I_2), \quad y^{(i)} = {\theta^{\star}}^T x^{(i)} + \varepsilon^{(i)}, \quad \varepsilon^{(i)} \sim \mathcal{N}(0, 1).
\end{equation}
The loss function is the mean squared error:
\begin{equation}
J(\theta) = \frac{1}{2m} \sum_{i=1}^{m} \bigl(\theta^T x^{(i)} - y^{(i)}\bigr)^2 = \frac{1}{2m}\|X\theta - y\|^2.
\end{equation}
This is a strongly convex quadratic with Hessian $H = \frac{1}{m}X^T X$ (constant, positive definite). The gradient is $\nabla J(\theta) = \frac{1}{m}X^T(X\theta - y)$. Because of observation noise $\varepsilon$, the minimum loss satisfies $J^{\star} \approx 0.4826$ rather than zero; the empirical minimiser $\hat{\theta}_{\text{OLS}} = (X^TX)^{-1}X^Ty$ converges to $\theta^{\star}$ as $m \to \infty$. Starting point: $\theta_0 = [0, 0]^T$.

\subsection{Benchmark B: Toy Neural Network Loss}
\begin{equation}
\label{eq:bench_B}
f(x_1, x_2) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1).
\end{equation}
The gradient and Hessian are:
\begin{equation}
\nabla f = \begin{bmatrix} 2(x_1 - 1) + \cos(x_1) \\ 10(x_2 - 2) \end{bmatrix}, \qquad
H(x) = \begin{bmatrix} 2 - \sin(x_1) & 0 \\ 0 & 10 \end{bmatrix}.
\end{equation}
The diagonal Hessian has eigenvalues $\lambda_1 = 2 - \sin(x_1) \in [1, 3]$ and $\lambda_2 = 10$, giving a local condition number ranging from $\kappa = 10/3 \approx 3.3$ (when $\sin x_1 = -1$) to $\kappa = 10$ (when $\sin x_1 = 1$), and $\kappa \approx 10/(2 - \sin(0.582)) \approx 6.9$ at the minimum. The sinusoidal term introduces mild non-quadratic curvature. The unique minimum is at $x^{\star} \approx (0.582, 2.0)$ (the root of $2(x_1 - 1) + \cos(x_1) = 0$, confirmed numerically) with $f^{\star} \approx 0.7244$. Starting point: $x_0 = [-1, 4]^T$.

\subsection{Benchmark C: Rosenbrock Function}
\begin{equation}
\label{eq:rosenbrock}
f(x_1, x_2) = (1 - x_1)^2 + 100(x_2 - x_1^2)^2.
\end{equation}
The global minimum is at $(1, 1)$ with $f^{\star} = 0$. The Hessian at the minimum is:
\begin{equation}
H(1,1) = \begin{bmatrix} 802 & -400 \\ -400 & 200 \end{bmatrix}, \qquad \text{tr}(H) = 1002, \quad \det(H) = 400,
\end{equation}
with eigenvalues $\lambda_{\pm} = (1002 \pm \sqrt{1002^2 - 4 \cdot 400})/2 \approx 1001.6$ and $0.40$, giving condition number $\kappa \approx 2508$. The large off-diagonal entries cause the eigenvectors to be nearly aligned with $[1,1]^T/\sqrt{2}$ (along-valley direction, eigenvalue $\lambda_{-} \approx 0.40$) and $[-1,1]^T/\sqrt{2}$ (cross-valley direction, eigenvalue $\lambda_{+} \approx 1001.6$). This means: (i) the optimal gradient direction is nearly along $[1,1]^T$ (diagonal movement into the valley), not along the coordinate axes, explaining why coordinate-wise methods (Adagrad, RMSprop) are misaligned with the efficient descent direction; (ii) progress along the valley is limited by the small eigenvalue $\lambda_{-} \approx 0.40$, even after all curvature in the cross-valley direction is resolved. At $x_0 = (-1,1)$, the Hessian $H(-1,1)$ has the same trace $1002$ and determinant $400$ as $H(1,1)$, confirming $\kappa \approx 2508$ throughout. Starting point: $x_0 = [-1, 1]^T$.

\subsection{Smoothness and Strong Convexity Summary}
\label{subsec:smoothness}

A function $f$ is $L$-smooth if $\|\nabla f(x) - \nabla f(y)\| \leq L\|x - y\|$ for all $x, y$, equivalently $\nabla^2 f \preceq LI$. It is $\mu$-strongly convex if $\nabla^2 f \succeq \mu I > 0$.

The step sizes $\alpha$ used in all experiments are conservative fractions of $\alpha_{\max}$, balancing stability against convergence speed. For GD on Benchmark C ($\alpha = 0.0012$), this is approximately $60\%$ of the stability limit $2/L = 0.002$, providing a safety margin against the varying Hessian along the trajectory.

\begin{remark}[Minimax Lower Bound for First-Order Methods]
Nesterov (1983) proved that for any deterministic first-order method that uses $\nabla f(x_0), \ldots, \nabla f(x_{k-1})$ to generate $x_k$, there exists a $\mu$-strongly convex, $L$-smooth function $f$ and starting point $x_0$ such that:
\[
f(x_k) - f^{\star} \;\geq\; \frac{\mu}{2} \left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^{2k} \|x_0 - x^{\star}\|^2.
\]
This is a \emph{minimax lower bound}: no algorithm can guarantee convergence faster than this rate in the worst case. The Heavy Ball method and Nesterov acceleration both achieve this bound. Gradient descent with $\alpha^{\star} = 2/(\mu + L)$ achieves the GD-optimal rate $\left(\frac{\kappa-1}{\kappa+1}\right)^k$, which is strictly worse than the minimax rate by a factor of $\sqrt{\kappa}$ in the exponent. These lower bounds apply to \emph{any} first-order method, including Polyak, Adagrad, RMSprop, and Adam, on worst-case strongly convex smooth problems.
\end{remark}

\subsection{Algorithm Formulations}

Four adaptive methods are compared against a constant step-size gradient descent (GD) baseline across all three benchmarks for 120 iterations.

\begin{algorithm}[H]
\caption{Polyak Step Size}
\label{alg:polyak}
\begin{algorithmic}[1]
\Require Initial point $x_0$, optimal value $f^{\star}$, tolerance $\varepsilon > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $\alpha_k \gets \dfrac{f(x_k) - f^{\star}}{\|g_k\|^2 + \varepsilon}$
    \State $x_{k+1} \gets x_k - \alpha_k\, g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The Polyak step size is theoretically motivated: for any convex $f$, setting $\alpha_k = (f(x_k) - f^{\star}) / \|\nabla f(x_k)\|^2$ is the unique step size that minimises $\|x_{k+1} - x^{\star}\|^2$ over all scalar steps, i.e., it is the step that brings the iterate \emph{closest} to any optimal point in the current gradient descent direction. This can be verified by minimising $\|x_k - \alpha g_k - x^{\star}\|^2$ over $\alpha$, giving $\alpha^{\star} = g_k^T(x_k - x^{\star}) / \|g_k\|^2$. Using the first-order convexity inequality $g_k^T(x_k - x^{\star}) \geq f(x_k) - f^{\star}$ (from $f(x^{\star}) \geq f(x_k) + g_k^T(x^{\star} - x_k)$) and replacing $g_k^T(x_k - x^{\star})$ with its lower bound $f(x_k) - f^{\star}$ gives the Polyak formula as a conservative approximation. The $\varepsilon$ term prevents division by zero.

\begin{algorithm}[H]
\caption{Adagrad}
\label{alg:adagrad}
\begin{algorithmic}[1]
\Require Initial point $x_0$, base learning rate $\alpha_0$, tolerance $\varepsilon$, iterations $N$
\State $G_0 \gets \mathbf{0} \in \mathbb{R}^d$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $G_{k+1} \gets G_k + g_k \odot g_k$ \Comment{Per-coordinate squared-gradient accumulation}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{G_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Adagrad applies a larger effective rate to coordinates with consistently small gradients and a smaller rate to those with large gradients. This implements an implicit diagonal preconditioning: the update $-[\alpha_0 / (\sqrt{G_k} + \varepsilon)] \odot g_k$ is equivalent to gradient descent on a problem rescaled by $\text{diag}(\alpha_0 / \sqrt{G_k})$. For sparse data (many zero-gradient coordinates), this preconditioning is very effective --- sparse coordinates accumulate less in $G_k$ and receive larger effective steps. The cumulative sum $G_k = \sum_{t=0}^{k-1} g_t \odot g_t$ grows monotonically ($G_k \to \infty$), causing the effective step $\alpha_0 / \sqrt{G_k} \to 0$ --- a key limitation for non-stationary or long-horizon problems.

\begin{algorithm}[H]
\caption{RMSprop}
\label{alg:rmsprop}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha_0$, decay factor $\beta \in (0,1)$, tolerance $\varepsilon$, iterations $N$
\State $v_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $v_{k+1} \gets \beta\, v_k + (1 - \beta)\, g_k \odot g_k$ \Comment{Exponential moving average of $g^2$}
    \State $x_{k+1} \gets x_k - \dfrac{\alpha_0}{\sqrt{v_{k+1}} + \varepsilon} \odot g_k$
\EndFor
\end{algorithmic}
\end{algorithm}

RMSprop (Hinton, 2012) addresses Adagrad's vanishing step issue by replacing the cumulative sum with an exponential moving average. The decay factor $\beta = 0.9$ (used throughout) means only the most recent gradient magnitudes dominate the scaling, maintaining a non-vanishing effective step size. In quasi-steady state, $v_k \approx (1-\beta)\mathbb{E}[g^2_k]$ (geometric series), so the effective step becomes $\alpha_0 g_k / \sqrt{(1-\beta)\mathbb{E}[g^2]} \propto g_k / \sqrt{\mathbb{E}[g^2]}$ --- \emph{gradient divided by its root-mean-square}, hence the name ``RMSprop''. This per-coordinate normalisation approximates a diagonal Newton step (replacing the full Hessian with its diagonal), providing curvature-aware scaling at $O(d)$ cost.

\begin{algorithm}[H]
\caption{Polyak Momentum / Heavy Ball}
\label{alg:heavyball}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, momentum $\beta \in (0,1)$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets \beta\, z_k + \alpha\, \nabla f(x_k)$
    \State $x_{k+1} \gets x_k - z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

The Heavy Ball method adds a momentum term $\beta z_k$ to the gradient update. For strongly convex quadratic objectives with condition number $\kappa = L/\mu$, the optimal hyperparameter choices are:
\begin{equation}
\alpha^{\star} = \frac{4}{(\sqrt{L} + \sqrt{\mu})^2}, \qquad \beta^{\star} = \left(\frac{\sqrt{\kappa} - 1}{\sqrt{\kappa} + 1}\right)^2,
\end{equation}
yielding linear convergence with rate $(\sqrt{\kappa}-1)/(\sqrt{\kappa}+1) < (\kappa-1)/(\kappa+1)$ (GD's rate). For $\kappa \gg 1$, the Heavy Ball rate is $\approx 1 - 2/\sqrt{\kappa}$ vs.\ GD's $\approx 1 - 2/\kappa$, a factor of $\sqrt{\kappa}$ improvement. To see this concretely: for Benchmark A ($\kappa \approx 1.12$), both methods converge in very few iterations since the speedup factor is only $\sqrt{\kappa} \approx 1.06\times$; for Benchmark C at $x_0 = (-1,1)$ with $\kappa \approx 2508$, GD's optimal rate is $\approx 0.9992$ (requiring ${\sim}1254$ iterations to reduce error by $e^{-1}$), while Heavy Ball's optimal rate is $\approx 0.961$ (requiring only ${\sim}25$ iterations) --- a $50\times$ speedup ($\sqrt{\kappa} \approx 50$). However, Heavy Ball's optimal convergence guarantee applies only to quadratics; for general non-quadratic functions, convergence guarantees are weaker and oscillation can occur (as seen on Benchmark B with $\beta = 0.96$).

\begin{theorem}[Polyak Step-Size Convergence]
For a convex function $f$ with exact $f^{\star}$, the Polyak step-size iteration satisfies
\[
\min_{0 \le k \le K} \bigl(f(x_k) - f^{\star}\bigr) \;=\; O(1/K).
\]
Moreover, if $f$ is $\mu$-strongly convex and $f^{\star}$ is known exactly, geometric convergence holds: $f(x_k) - f^{\star} \le (1 - \mu/L)^k (f(x_0) - f^{\star})$.
\end{theorem}

\begin{remark}[Heavy Ball as a Damped Harmonic Oscillator]
The Heavy Ball update $z_{k+1} = \beta z_k + \alpha \nabla f(x_k)$, $x_{k+1} = x_k - z_{k+1}$ can be written as a second-order recurrence:
\[
x_{k+1} - (1 + \beta) x_k + \beta x_{k-1} = -\alpha \nabla f(x_k).
\]
In the continuous-time limit (step size $\alpha \to 0$), this becomes the ODE $\ddot{x} + (1-\beta)\dot{x} + \alpha\nabla f(x) = 0$, the equation of a \emph{damped harmonic oscillator} with damping coefficient $\gamma = 1 - \beta$ and ``spring constant'' equal to the Hessian eigenvalue $\lambda$. The optimal Heavy Ball parameters $\alpha^{\star}$ and $\beta^{\star}$ correspond to \emph{critical damping} ($\gamma^2 = 4\alpha\lambda$) for the worst eigenvalue, ensuring exponential convergence without oscillation for quadratic objectives. For non-quadratic objectives, the varying Hessian causes the system to be ``out of tune'' --- neither critically damped nor over-damped simultaneously in all directions --- explaining the oscillations observed in Figure~\ref{fig:q2_nesterov_vs_hb}.
\end{remark}

\begin{theorem}[GD Linear Convergence for Strongly Convex Smooth Objectives]
\label{thm:gd_linear}
Let $f$ be $\mu$-strongly convex and $L$-smooth, with $\kappa = L/\mu$. For GD with $\alpha \in (0, 2/L)$:
\[
f(x_k) - f^{\star} \;\leq\; \rho_f^k (f(x_0) - f^{\star}), \qquad \rho_f = 1 - 2\alpha\mu\!\left(1 - \tfrac{\alpha L}{2}\right) < 1.
\]
Equivalently, $\|x_k - x^{\star}\| \leq \rho_d^k \|x_0 - x^{\star}\|$ with $\rho_d = \max(|1 - \alpha\mu|, |1 - \alpha L|)$. The two rates satisfy $\rho_f \approx \rho_d^2$ (the function-value rate is the square of the distance rate). The optimal step $\alpha^{\star} = 2/(\mu + L)$ gives $\rho_d^{\star} = (\kappa - 1)/(\kappa + 1)$.
\end{theorem}
\begin{proof}[Proof sketch]
By $L$-smoothness, $f(x_{k+1}) \leq f(x_k) - \alpha(1 - \tfrac{\alpha L}{2})\|\nabla f(x_k)\|^2$. By $\mu$-strong convexity, $\|\nabla f(x_k)\|^2 \geq 2\mu(f(x_k) - f^{\star})$. Combining: $f(x_{k+1}) - f^{\star} \leq \rho_f (f(x_k) - f^{\star})$ with $\rho_f = 1 - 2\alpha\mu(1 - \tfrac{\alpha L}{2})$. Optimising over $\alpha$ gives $\alpha^{\star} = 2/(\mu + L)$.
\end{proof}

For Benchmark A with $\kappa \approx 1.12$, $\alpha = 0.08$: distance rate $\rho_d = \max(|1 - 0.08 \times 0.925|, |1 - 0.08 \times 1.033|) = \max(0.926, 0.917) = 0.926$, and function-value rate $\rho_f \approx 0.926^2 = 0.858$, confirming Figure~\ref{fig:theory_vs_practice}.

\begin{remark}[Polyak Sensitivity to $f^{\star}$]
The Polyak step size is exquisitely sensitive to the accuracy of $f^{\star}$. The step formula $\alpha_k = (f(x_k) - f^{\star}) / (\|\nabla f(x_k)\|^2 + \varepsilon)$ requires $f^{\star} \leq f(x_k)$ for all $k$ (otherwise a negative step is taken towards ascent). When $f^{\star} = 0 < J^{\star}_{\text{true}} \approx 0.4826$ (Benchmarks A and B), the numerator remains bounded away from zero while the denominator vanishes as the gradient shrinks, causing $\alpha_k \to \infty$. On Benchmark C where $f^{\star} = 0$ is exact, the numerator $f(x_k) - 0$ and denominator $\|\nabla f\|^2$ both shrink, yielding a well-behaved decaying step.
\end{remark}

\subsection{Comparative Analysis}

\begin{itemize}[nosep]
\item \textbf{Convergence speed:} When $f^{\star}$ is known exactly (Benchmark C), Polyak converges fastest. On Benchmarks A and B (where $f^{\star}$ is misspecified), Adagrad and Heavy Ball match GD baseline performance without the instability risk.

\item \textbf{Smoothness:} RMSprop produces the smoothest convergence curves due to exponential averaging of squared gradients, which filters out gradient noise. Heavy Ball shows characteristic overshoot followed by damped oscillation, consistent with a second-order dynamical system.

\item \textbf{Stability:} GD baseline is unconditionally stable for $\alpha < 2/L$ (Lipschitz smoothness constant). Adagrad is stable by design since its effective step decreases monotonically. Polyak is \emph{conditionally stable} only when $f^{\star}$ is accurately known; misspecification is catastrophic.

\item \textbf{Practical recommendation:} In practice, exact $f^{\star}$ is rarely known, limiting Polyak's applicability to problems where a tight lower bound on the optimal value can be established (e.g., non-negative losses where a data-dependent lower bound is available). Adagrad and RMSprop are more robust choices for general problems.

\item \textbf{Effect of condition number:} Table~\ref{tab:smoothness} shows that Benchmark A's near-unit conditioning ($\kappa \approx 1.12$) makes it trivially easy for all methods. Benchmark B's $\kappa \approx 6.9$ (at the optimum) is moderate. Benchmark C's $\kappa \approx 2508$ is the key driver of difficulty: GD needs a tiny step $\alpha < 2/L = 0.002$, requiring $\sim\kappa\log(1/\varepsilon) \approx 2500\log(10^3) \approx 17{,}000$ steps to achieve $10^{-3}$ accuracy, while Newton requires $\sim 7$ steps.

\item \textbf{Memory cost:} GD and Polyak: $O(d)$. Adagrad and RMSprop: $O(d)$ for the accumulated squared gradients. Heavy Ball: $O(d)$ for the momentum buffer. All methods are memory-efficient for large $d$.

\item \textbf{Practical domain:} Adagrad's diagonal preconditioning advantage is most pronounced in NLP and recommendation systems where input features are sparse: an embedding lookup table has $d = \text{vocabulary} \times \text{embedding dim}$ parameters, but each mini-batch touches only $b \ll d$ embedding rows. The cumulative $G_k$ grows only for active coordinates, preserving large effective steps for rare-but-important features. This is why Adagrad (and its successor Adam) became the default for word embedding training. For our dense $d = 2$ benchmarks, this sparsity advantage is absent, and Adagrad offers no benefit over RMSprop.
\end{itemize}

\subsection{Algorithm Formulations}

\begin{algorithm}[H]
\caption{Nesterov Accelerated Gradient}
\label{alg:nesterov}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, maximum momentum $\beta_{\max}$, iterations $N$
\State $z_0 \gets \mathbf{0}$
\For{$k = 1, 2, \ldots, N$}
    \State $\beta_k \gets \min\!\left(\dfrac{k-1}{k+2},\ \beta_{\max}\right)$ \Comment{Adaptive momentum schedule}
    \State $\tilde{x}_k \gets x_k + \beta_k\, z_k$ \Comment{Lookahead (extrapolation) point}
    \State $z_{k+1} \gets \beta_k\, z_k - \alpha\, \nabla f(\tilde{x}_k)$
    \State $x_{k+1} \gets x_k + z_{k+1}$
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{theorem}[Nesterov Acceleration]
For a convex, $L$-smooth function $f$ and the Nesterov scheme with $\alpha = 1/L$ and the schedule $\beta_k = (k-1)/(k+2)$,
\[
f(x_k) - f^{\star} \;\leq\; \frac{2L\|x_0 - x^{\star}\|^2}{(k+1)^2} \;=\; O(1/k^2).
\]
This is \emph{optimal} for first-order methods on convex smooth problems (Nesterov, 1983).
\end{theorem}

\begin{remark}[Why $O(1/k^2)$ is Optimal]
Nesterov (1983) proved a \emph{lower bound}: for any first-order method (using only gradient evaluations $\nabla f(x_0), \ldots, \nabla f(x_{k-1})$) and any $L$-smooth convex $f$, there exists a worst-case function for which $f(x_k) - f^{\star} \geq \Omega(L\|x_0 - x^{\star}\|^2 / k^2)$. The construction uses a ``worst-case'' tridiagonal quadratic, showing that gradient information propagates through the dimension at most one coordinate per step. Since Nesterov acceleration matches this lower bound, it is \emph{oracle-optimal}: no first-order method can do asymptotically better in the worst case. This does \emph{not} mean Nesterov is always best in practice --- for strongly convex problems, Heavy Ball achieves the same rate with a simpler implementation, and for finite-sum problems, variance reduction methods are superior.
\end{remark}

The key distinction from Heavy Ball is the \emph{lookahead}: the gradient is evaluated at $\tilde{x}_k = x_k + \beta_k z_k$ rather than $x_k$. This allows the algorithm to ``correct'' its trajectory before overshooting. Conceptually, Nesterov evaluates the gradient one momentum step ahead, combining the momentum direction with the local gradient information at that look-ahead point.

\begin{algorithm}[H]
\caption{Adam Optimiser}
\label{alg:adam}
\begin{algorithmic}[1]
\Require Initial point $x_0$, learning rate $\alpha$, moments $\beta_1, \beta_2 \in (0,1)$, $\varepsilon > 0$, iterations $N$
\State $m_0 \gets \mathbf{0},\ v_0 \gets \mathbf{0}$
\For{$t = 1, 2, \ldots, N$}
    \State $g_t \gets \nabla f(x_t)$
    \State $m_t \gets \beta_1\, m_{t-1} + (1 - \beta_1)\, g_t$ \Comment{First moment (mean) estimate}
    \State $v_t \gets \beta_2\, v_{t-1} + (1 - \beta_2)\, g_t \odot g_t$ \Comment{Second moment (variance) estimate}
    \State $\hat{m}_t \gets m_t\, /\, (1 - \beta_1^t)$ \Comment{Bias correction}
    \State $\hat{v}_t \gets v_t\, /\, (1 - \beta_2^t)$
    \State $x_{t+1} \gets x_t - \alpha\, \hat{m}_t\, /\, (\sqrt{\hat{v}_t} + \varepsilon)$
\EndFor
\end{algorithmic}
\end{algorithm}

Adam (Kingma \& Ba, 2015) combines momentum (via $m_t$) with per-coordinate adaptive scaling (via $v_t$), inheriting advantages of both RMSprop and momentum methods. The bias correction terms $(1 - \beta_1^t)^{-1}$ and $(1 - \beta_2^t)^{-1}$ are essential in early iterations: at $t = 1$ with $\beta_1 = 0.9$, $m_1 = 0.1\, g_1$ (a $10\times$ underestimate of the mean), but $\hat{m}_1 = m_1/(1 - 0.9) = g_1$ (correctly unbiased). Without bias correction, Adam takes excessively small steps in the first $\sim 1/(1-\beta_2)$ iterations as $v_t$ accumulates, potentially slowing convergence in short runs. For $\beta_2 = 0.999$, the second moment takes $\sim 1000$ iterations to converge to its true value; during this warm-up period, the bias correction is critical for maintaining an effective learning rate of $\alpha = 0.12$ (not $\alpha \cdot (1-\beta_2^t)^{1/2} \approx 0.0038$ at $t=1$). In our 150-iteration runs, the bias correction is effectively irrelevant for $\beta_1 = 0.9$ by iteration $\sim 30$ but remains important for $\beta_2 = 0.999$ throughout.

\begin{remark}[FISTA: Nesterov for Composite Problems]
Nesterov acceleration extends naturally to composite objectives $\min_x\{f(x) + g(x)\}$ where $f$ is smooth convex and $g$ is convex but possibly non-smooth (e.g.\ $g = \lambda\|x\|_1$ for LASSO). FISTA (Beck \& Teboulle, 2009) replaces the gradient step with a proximal step and retains Nesterov's momentum schedule to achieve $O(1/k^2)$ convergence. This connection demonstrates that the acceleration principle is not limited to smooth objectives.
\end{remark}

\begin{remark}[AMSGrad: Fixing Adam's Non-Convergence]
Reddi et al.\ (2018) showed that Adam can fail to converge even on simple convex problems when the exponential averaging ``forgets'' large infrequent gradients, allowing a bad step to slip through with an overly large effective step size. The fix, AMSGrad, replaces $\hat{v}_t$ with a \emph{monotone} running maximum:
\[
\hat{V}_t = \max(\hat{V}_{t-1},\, \hat{v}_t), \qquad x_{t+1} = x_t - \alpha\, \hat{m}_t / (\sqrt{\hat{V}_t} + \varepsilon).
\]
Because $\hat{V}_t$ never decreases, the effective step size $\alpha/\sqrt{\hat{V}_t}$ is non-increasing per coordinate, restoring the regret guarantee. In practice, AMSGrad and Adam often perform comparably on neural networks because the forgetting mechanism of Adam is also a feature (it allows recovery from occasional large stochastic gradients), but AMSGrad is preferred when convergence guarantees are required.
\end{remark}

\begin{remark}[Why Adam Underperforms Nesterov on Rosenbrock]
The Rosenbrock gradient along the valley floor has two components: (i) a large \emph{cross-valley} gradient $\partial f/\partial x_2 = 200(x_2 - x_1^2)$ that decays quickly as the iterate reaches the valley, and (ii) a smaller \emph{along-valley} gradient driving progress towards $(1,1)$. Adam's second-moment estimator $v_t$ accumulates the squared magnitudes of \emph{all} gradient components. Early in the trajectory, the dominant cross-valley component fills $v_{t,2}$, causing Adam to shrink the effective step in $x_2$ precisely when large steps along the valley floor are most needed. This normalisation is beneficial when curvature in all coordinates is similarly large throughout training (as in anisotropic quadratics like Benchmark B), but counterproductive on Rosenbrock where one direction requires very small steps initially but large steps later. Nesterov, by contrast, applies a \emph{uniform} step $\alpha$ in all directions with momentum, and its adaptive momentum schedule naturally increases the along-valley momentum component as the cross-valley gradient vanishes, providing the sustained push along the valley floor that Adam cannot.
\end{remark}

\subsection{Mini-Batch Stochastic Gradient Descent}

Mini-batch SGD is applied to Benchmark A using learning rate $\alpha = 0.06$, batch sizes $b \in \{5, 40\}$, 50 epochs, with data shuffled at each epoch.

\begin{algorithm}[H]
\caption{Mini-Batch SGD}
\label{alg:sgd}
\begin{algorithmic}[1]
\Require Data $(X, y) \in \mathbb{R}^{m \times d} \times \mathbb{R}^m$, initial $\theta_0$, learning rate $\alpha$, batch size $b$, epochs $E$
\For{$e = 1, 2, \ldots, E$}
    \State Randomly permute indices $\{1, \ldots, m\}$ into $\pi$
    \For{each mini-batch $\mathcal{B} \subset \pi$ of size $b$}
        \State $g_{\mathcal{B}} \gets \frac{1}{b} X_{\mathcal{B}}^T(X_{\mathcal{B}}\theta - y_{\mathcal{B}})$ \Comment{Stochastic gradient estimate}
        \State $\theta \gets \theta - \alpha\, g_{\mathcal{B}}$
    \EndFor
\EndFor
\end{algorithmic}
\end{algorithm}

With $m = 1000$ samples, batch size $b = 5$ gives $1000/5 = 200$ updates per epoch, while $b = 40$ gives $1000/40 = 25$ updates per epoch. The stochastic gradient $g_{\mathcal{B}}$ is an unbiased estimate of the full gradient $\nabla J(\theta)$, with variance $\mathbb{V}[g_{\mathcal{B}}] = \sigma^2_{\text{samp}}/b$.

\begin{remark}[Batch-Size--Step-Size Trade-off and Linear Scaling]
Fix $m$ training samples and gradient noise variance $\sigma^2_{\text{samp}}$. For batch size $b$:
\begin{itemize}[nosep]
\item \textbf{Updates per epoch}: $m/b$ (larger $b \Rightarrow$ fewer gradient steps per data pass).
\item \textbf{Gradient variance per update}: $\mathbb{V}[g_{\mathcal{B}}] = \sigma^2_{\text{samp}}/b$ (larger $b \Rightarrow$ less noise per step).
\item \textbf{Cost per epoch}: $O(m \cdot d)$ FLOPs, \emph{independent of $b$}.
\end{itemize}
With constant step size $\alpha$, the per-epoch convergence bound for strongly convex $f$ is:
\[
\mathbb{E}[f(\theta^{\text{end of epoch}})] - f^\star \;\lesssim\; (1 - \mu\alpha)^{m/b}(f(\theta_0) - f^\star) + \frac{\alpha\,\sigma^2_{\text{samp}}}{2\mu b}.
\]
The noise floor $\alpha\sigma^2_{\text{samp}}/(2\mu b)$ shrinks with $b$, which means one can increase $\alpha$ proportionally to $b$ (up to the stability limit $\alpha < 2/L$) while keeping the noise floor fixed. This is the \emph{linear scaling rule} (Goyal et al., 2017): multiplying batch size by $k$ allows multiplying step size by $k$, maintaining the same per-epoch convergence curve in fewer wall-clock steps when gradient computation parallelises across the batch. In our experiments, $b = 40$ (vs.\ $b = 5$) allows in principle an $8\times$ larger step size without increasing the noise floor --- though here both batches are already at the same $\alpha$, so the primary visible effect is the $8\times$ difference in updates per epoch.
\end{remark}

\caption{Q2: Mini-batch SGD with batch sizes $b \in \{5, 40\}$. Smaller batches ($b = 5$) converge faster per epoch due to more frequent parameter updates (200 vs.\ 25 updates/epoch) but exhibit more oscillation near the minimum because each mini-batch gradient has higher variance $\sim \sigma^2/b$. \begin{remark}[SGD Convergence Rates and Polyak--Ruppert Averaging]
For a $\mu$-strongly convex, $L$-smooth objective and constant step size $\alpha \leq 1/(2L)$:
\begin{itemize}[nosep]
\item \textbf{Constant $\alpha$}: $\mathbb{E}[f(\theta_k)] - f^{\star} \leq (1-\mu\alpha)^k (f(\theta_0) - f^{\star}) + \alpha \sigma_{\text{samp}}^2 / (2\mu)$. Geometric decay to a noise-floor of $O(\alpha \sigma_{\text{samp}}^2 / \mu)$, where $\sigma_{\text{samp}}^2$ is the mini-batch gradient \emph{sampling} variance (arising from stochastic subsampling, distinct from the observation noise in the data). In our Benchmark A experiments, the dominant noise is the \emph{observation model noise}: the minimum achievable loss is $J^{\star} = \frac{1}{2m}\|X\hat{\theta} - y\|^2 \approx 0.4826$ (the OLS residual variance from the data generation noise $\varepsilon \sim \mathcal{N}(0,1)$), not zero. The SGD algorithmic noise floor $\alpha\sigma_{\text{samp}}^2/(2\mu)$ is an additional contribution on top of this, typically much smaller for our well-conditioned $\kappa \approx 1.12$ problem.
\item \textbf{Decaying $\alpha_k = a/(b+k)$}: Achieves $\mathbb{E}[f(\theta_k)] - f^{\star} = O(1/k)$ convergence to the true optimum, with no noise floor. The Robbins--Monro conditions $\sum_k \alpha_k = \infty$ and $\sum_k \alpha_k^2 < \infty$ ensure convergence.
\item \textbf{Polyak--Ruppert averaging}: With any decaying step size, maintaining the Cesàro average $\bar{\theta}_k = \frac{1}{k}\sum_{t=1}^k \theta_t$ improves the rate to $O(1/(\mu k))$ for strongly convex problems and $O(\sigma / \sqrt{k})$ for general convex problems --- matching the statistical minimax lower bound for stochastic approximation.
\end{itemize}
In the experiments, a fixed $\alpha = 0.06$ is used; the observed convergence level $J^{\star} \approx 0.4826$ is the irreducible \emph{observation model} noise floor: the minimum mean squared error achievable by any estimator on this dataset, equal to $\frac{1}{2m}\|y - X\hat{\theta}_{\text{OLS}}\|^2$. This is not an SGD algorithmic noise floor --- full-batch GD with the same $\alpha$ also converges to exactly $J^{\star} \approx 0.4826$. The SGD algorithmic noise floor $\alpha\sigma_{\text{samp}}^2/(2\mu)$ would cause SGD to plateau \emph{above} the full-batch GD result for constant $\alpha$, visible only after the observation model noise floor is subtracted --- a distinction obscured in our experiments because $J^{\star}_{\text{obs}} \gg \alpha\sigma_{\text{samp}}^2/(2\mu)$ for our well-conditioned problem.
\end{remark}

\begin{remark}[Bias-Variance Trade-off in SGD]
The batch size controls a bias-variance trade-off in the gradient estimate. For a fixed noise model with variance $\sigma^2$, the variance of the mini-batch gradient estimate scales as $\sigma^2 / b$. Smaller $b$ reduces computation per update but increases gradient noise, slowing convergence near the minimum. Optimal batch size in practice balances hardware parallelism, memory, and this trade-off; in deep learning, batch sizes of 32--256 are common.
\end{remark}

\begin{remark}[Variance Reduction: SVRG and SAGA]
The SGD noise floor $O(\alpha\sigma^2/\mu)$ at constant step size is a fundamental limitation: for exact convergence, $\alpha$ must decay, slowing the geometric descent. Variance reduction methods eliminate the noise floor while maintaining cheap per-iteration cost. \textbf{SVRG} (Johnson \& Zhang, 2013) periodically computes a full gradient $\tilde{g}$ at a reference point $\tilde{x}$ and corrects the stochastic gradient:
\[
g_k^{\text{SVRG}} = \nabla f_{i_k}(x_k) - \nabla f_{i_k}(\tilde{x}) + \tilde{g}.
\]
This estimator is unbiased ($\mathbb{E}[g_k^{\text{SVRG}}] = \nabla f(x_k)$) but has variance $O(\|x_k - \tilde{x}\|^2)$, which shrinks as $x_k \to x^{\star}$. With constant step size, SVRG achieves \emph{linear convergence} $\mathbb{E}[f(x_k)] - f^{\star} = O(\rho^k)$ without any step-size decay, combining the per-iteration efficiency of SGD with the convergence rate of full gradient descent. \textbf{SAGA} (Defazio et al., 2014) maintains a table of past gradients and uses a different control variate, offering similar guarantees without periodic full-gradient evaluation. For large-scale finite-sum problems $f = \frac{1}{m}\sum_{i=1}^m f_i$, these methods are typically preferable to vanilla SGD or Adam when the problem is strongly convex.
\end{remark}

\subsection{Local Approximations of $g(x) = x^4$}

Consider $g(x) = x^4$, expanded around $x_0 = 0.25$.

\textbf{First-order (linear/tangent) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) = 0.25^4 + 4 \cdot 0.25^3\,(x - 0.25) = 0.0039 + 0.0625(x - 0.25).
\end{equation}

\textbf{Second-order (quadratic) approximation:}
\begin{equation}
g(x) \approx g(x_0) + g'(x_0)(x - x_0) + \tfrac{1}{2}g''(x_0)(x - x_0)^2
= 0.0039 + 0.0625(x - 0.25) + 0.375(x - 0.25)^2.
\end{equation}

The Newton step minimises the second-order model: setting $\partial/\partial x [g(x_0) + g'(x_0)(x-x_0) + \tfrac{1}{2}g''(x_0)(x-x_0)^2] = 0$ gives
\[
x_1 = x_0 - \frac{g'(x_0)}{g''(x_0)} = 0.25 - \frac{0.0625}{0.75} = 0.25 - 0.0833 = 0.1667.
\]
Iterating: $g'(0.1667) = 4 \times 0.1667^3 = 0.0185$, $g''(0.1667) = 12 \times 0.1667^2 = 0.333$, so $x_2 = 0.1667 - 0.0185/0.333 = 0.1111$. The pattern is $x_k = (2/3)^k \cdot 0.25$, converging to $x^{\star} = 0$ at a geometric rate that improves as $x_k \to 0$. Strictly speaking, $g''(0) = 0$ means Newton's method is not quadratically convergent at the limit (the Hessian degenerates); the convergence rate at $x_k = (2/3)^k \cdot 0.25$ is determined by the ratio $|x_{k+1} - 0|^2 / |x_k - 0| = (2/3)^2 / (2/3) = 2/3 \to 0$, which is super-linear (faster than any fixed geometric rate) but not quadratic in the classical sense.

\begin{remark}[Why First-Order Approximations Are Insufficient for Optimisation]
The linear (first-order) model $\ell(x) = g(x_0) + g'(x_0)(x - x_0)$ has no minimum unless $g'(x_0) = 0$. A gradient step in direction $-g'(x_0)$ is therefore derived not from minimising $\ell$ but from the (unjustified) heuristic of moving in the descent direction with a small step. In contrast, the quadratic (second-order) model always has a minimum at $x_1 = x_0 - g'(x_0)/g''(x_0)$ (when $g''(x_0) > 0$), providing a principled step that accounts for local curvature. This is why Newton's method outperforms gradient descent on well-conditioned problems: its step size is implicitly calibrated to the curvature, unlike the manually tuned $\alpha$ of gradient descent.
\end{remark}

\subsection{Newton's Method}

\begin{algorithm}[H]
\caption{Newton's Method with Damping}
\label{alg:newton}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha \in (0, 1]$, damping $\lambda \geq 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $g_k \gets \nabla f(x_k)$
    \State $H_k \gets \nabla^2 f(x_k) + \lambda I$ \Comment{Tikhonov-regularised Hessian}
    \State $p_k \gets H_k^{-1}\, g_k$ \Comment{Newton direction (solve linear system)}
    \State $x_{k+1} \gets x_k - \alpha\, p_k$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: GD uses 80 iterations; Newton uses 20 iterations with Hessian damping $\lambda = 10^{-8}$.

\begin{remark}[Backtracking Line Search for Newton's Method]
The damping factor $\alpha \in \{0.22, 0.85, 1.0\}$ in our experiments is chosen by hand. A principled alternative is the \textbf{Armijo backtracking line search}: starting from $\alpha = 1$ (full Newton step), reduce $\alpha \gets \rho \alpha$ (typically $\rho = 0.5$) until the \emph{sufficient decrease} condition $f(x_k - \alpha p_k) \leq f(x_k) - c \alpha \nabla f(x_k)^T p_k$ is satisfied (typically $c = 10^{-4}$). This guarantees (i) \emph{global convergence} from any starting point, not just near the optimum, and (ii) \emph{quadratic local convergence} because once $x_k$ is in the quadratic convergence ball, $\alpha = 1$ satisfies the Armijo condition automatically (provable from the Lipschitz-Hessian assumption). The hand-tuned $\alpha = 0.22$ for Rosenbrock serves the same role but cannot self-adapt: it is conservatively chosen for stability near $x_0$ but would be sub-optimal closer to $x^{\star}$ where a larger step is safe.
\end{remark}

\begin{theorem}[Newton's Method for Quadratics]
\label{thm:newton_quadratic}
For a strongly convex quadratic $f(\theta) = \tfrac{1}{2}\theta^T H \theta - b^T\theta$, Newton's method with $\alpha = 1$ converges in exactly \emph{one iteration}: $\theta_1 = \theta_0 - H^{-1}(\nabla f(\theta_0)) = H^{-1}b = \theta^{\star}$.
\end{theorem}
\begin{proof}
Since $f$ is quadratic, $\nabla f(\theta) = H\theta - b$ (a constant). The Newton direction is $p_0 = H^{-1}\nabla f(\theta_0) = H^{-1}(H\theta_0 - b) = \theta_0 - H^{-1}b$. So $\theta_1 = \theta_0 - p_0 = H^{-1}b = \theta^{\star}$. The key insight is that the second-order Taylor model of $f$ around $\theta_0$ is exact (not an approximation), since $f$ is quadratic and $\nabla^2 f = H$ everywhere.
\end{proof}

\begin{theorem}[Quadratic Convergence of Newton's Method]
Let $f$ be twice continuously differentiable with $L$-Lipschitz Hessian, $\mu$-strongly convex. Then, in a sufficiently small neighbourhood of $x^{\star}$, Newton's method with $\alpha = 1$ satisfies:
\[
\|x_{k+1} - x^{\star}\| \;\leq\; \frac{L}{2\mu}\|x_k - x^{\star}\|^2.
\]
This is \emph{quadratic} (super-linear) convergence: the number of correct digits roughly doubles each iteration.
\end{theorem}

\begin{remark}[Gauss--Newton for Least-Squares: One Step is Exact]
For least-squares problems $f(\theta) = \frac{1}{2m}\|r(\theta)\|^2$ with residuals $r(\theta) = X\theta - y$ (linear in $\theta$, as in Benchmark A), the Hessian is $H = J^T J / m = X^T X / m$ where $J = X / \sqrt{m}$ is the Jacobian of $r$. Newton's method uses the exact Hessian $H$. The \textbf{Gauss--Newton} method approximates $H \approx J^T J$ by dropping the second-derivative (curvature of residuals) term; for \emph{linear} residuals, this is exact ($H = J^T J / m$ precisely), so Gauss--Newton and Newton coincide for Benchmark A. The Gauss--Newton step is $\theta_{k+1} = \theta_k - (J^T J)^{-1} J^T r(\theta_k) = (X^T X)^{-1} X^T y = \hat{\theta}_{\text{OLS}}$, confirming one-step convergence. For non-linear residuals (as in general neural networks), Gauss--Newton uses only the first-order Jacobian, making it stable and positive semi-definite by construction ($J^T J \succeq 0$), unlike the full Hessian which may become indefinite in non-convex regions.
\end{remark}

\begin{remark}[Quasi-Newton Methods as a Practical Alternative]
Computing the exact Hessian costs $O(d^2)$ per iteration and requires solving a $d \times d$ linear system ($O(d^3)$), making Newton infeasible for large $d$. \textbf{Quasi-Newton methods} (e.g., BFGS) maintain an approximate inverse Hessian $B_k^{-1}$ updated via rank-1 or rank-2 updates from gradient differences:
\[
B_{k+1} \gets B_k + \frac{y_k y_k^T}{y_k^T s_k} - \frac{B_k s_k s_k^T B_k}{s_k^T B_k s_k}, \quad s_k = x_{k+1} - x_k,\ y_k = \nabla f(x_{k+1}) - \nabla f(x_k).
\]
BFGS achieves super-linear convergence at $O(d^2)$ cost per iteration, making it the method of choice for medium-scale unconstrained optimisation ($d \lesssim 10^4$). Unlike Newton, BFGS is guaranteed to maintain $B_k \succ 0$ (positive definiteness) as long as the curvature condition $y_k^T s_k > 0$ holds --- which is ensured by a Wolfe line search --- so the descent direction $-B_k^{-1} \nabla f$ is always a valid descent direction.
\end{remark}

\begin{remark}[Why Damping is Required on Rosenbrock]
The Rosenbrock Hessian at $(x_1, x_2)$ is
\[
H = \begin{bmatrix} 2 + 1200x_1^2 - 400x_2 & -400x_1 \\ -400x_1 & 200 \end{bmatrix}.
\]
At $x_0 = (-1, 1)$: $H_{11} = 2 + 1200 - 400 = 802$, $H_{12} = 400$, $H_{22} = 200$. The full Newton step $p = H^{-1}\nabla f$ may be large and point far from the quadratic approximation's validity region. A damping factor $\alpha = 0.22$ trades convergence speed for stability, ensuring each step stays within the region where the quadratic model is accurate.
\end{remark}

\begin{remark}[Newton Decrement as a Principled Stopping Criterion]
The quadratic convergence of Newton's method enables a rigorous stopping criterion without knowing $f^{\star}$. The \textbf{Newton decrement} $\lambda_N(x) = \sqrt{\nabla f(x)^T H(x)^{-1} \nabla f(x)}$ satisfies $f(x) - f(x^{\star}) \leq \frac{1}{2}\lambda_N(x)^2$ (for strongly convex $f$), providing a computable upper bound on the sub-optimality gap. In practice, the stopping criterion $\lambda_N(x)^2/2 \leq \varepsilon_{\text{tol}}$ guarantees $\epsilon$-accuracy without any reference to $f^{\star}$. For our Benchmark A (quadratic), $\lambda_N$ drops to machine precision after one Newton step. For our experiments, we use the fixed iteration count $N = 20$, but in production implementations the Newton decrement criterion would terminate early (typically in 5--10 iterations for Benchmark B).
\end{remark}

\begin{remark}[Levenberg--Marquardt: Adaptive Hessian Regularisation]
The Tikhonov regularisation $H_{\lambda} = H + \lambda I$ used in Algorithm~\ref{alg:newton} has a direct connection to the \textbf{Levenberg--Marquardt} (LM) method, which adaptively adjusts $\lambda$ based on whether the quadratic model provides a good approximation. The ratio $\rho = (f(x_k) - f(x_k + p_k)) / (m_k(0) - m_k(p_k))$ compares actual vs.\ predicted reduction: if $\rho > 0.75$, decrease $\lambda$ (more like Newton); if $\rho < 0.25$, increase $\lambda$ (more like steepest descent). This adaptive trust-region strategy eliminates the need to hand-tune the damping factor $\alpha$, and is particularly effective on least-squares problems where $H \approx J^T J$ (Gauss--Newton Hessian approximation). In our Rosenbrock experiments, a fixed $\alpha = 0.22$ serves a similar role but wastes potential progress in later iterations when the iterate is already close to the optimum and $\lambda$ could safely be reduced.
\end{remark}

\subsection{Part A: Finite Difference and Nesterov Random Search (Benchmark B)}

\subsubsection{Finite Difference Gradient Approximation}

The forward finite-difference approximation of the $i$-th partial derivative is:
\begin{equation}
\frac{\partial f}{\partial x_i}(x) \approx \frac{f(x + \delta e_i) - f(x)}{\delta}.
\end{equation}
The total approximation error has two competing components:
\begin{itemize}[nosep]
\item \textbf{Truncation error}: $O(\delta)$ from the Taylor remainder $\tfrac{1}{2}\delta f''(\xi)$ for some $\xi \in [x, x+\delta e_i]$.
\item \textbf{Round-off error}: $O(\varepsilon_{\text{mach}}/\delta)$ from floating-point cancellation when computing $f(x + \delta e_i) - f(x)$.
\end{itemize}
Balancing these gives the optimal step size $\delta^{\star} = \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$. For central differences, $[\partial f / \partial x_i](x) \approx [f(x + \delta e_i) - f(x - \delta e_i)]/(2\delta)$, the truncation error improves to $O(\delta^2)$ at the cost of one additional function evaluation per coordinate.

\begin{remark}[Complex-Step Differentiation]
The round-off error floor of finite differences arises from catastrophic cancellation: when $\delta$ is small, $f(x + \delta e_i)$ and $f(x)$ share many significant digits, so their difference loses precision. \textbf{Complex-step differentiation} avoids this entirely by evaluating $f$ at a complex perturbation $x + ih\delta e_i$ (where $i = \sqrt{-1}$, not the index) and using
\[
\frac{\partial f}{\partial x_i}(x) \approx \frac{\mathrm{Im}[f(x + ih\delta e_i)]}{\delta},
\]
which has truncation error $O(\delta^2)$ but \emph{no round-off error} from cancellation, since the imaginary part is not subtracted from anything real. This allows $\delta$ to be as small as $10^{-200}$ without loss of accuracy, achieving machine-precision derivatives for the cost of one complex function evaluation. The method requires the function to be analytic (complexifiable) and is widely used in aeronautical engineering and sensitivity analysis.
\end{remark}

\centering
\includegraphics[width=0.88\textwidth]{figures/q4_fd_error.pdf}
\caption{Q4: Finite difference gradient approximation error $\|\hat{g} - \nabla f\|$ vs.\ step size $\delta$ at $x = (0.5, 3.0)$ for Benchmark B. \textbf{Forward differences} (blue): for small $\delta$, round-off error $\propto \varepsilon_{\text{mach}}/\delta$ dominates (slope $-1$ on log--log); for large $\delta$, truncation error $\propto \delta L/2$ dominates (slope $+1$); minimum error $\approx 10^{-8}$ at $\delta^{\star} \approx 10^{-8} = \sqrt{\varepsilon_{\text{mach}}}$. \textbf{Central differences} (red): $O(\delta^2)$ truncation error (slope $+2$, steeper than forward's slope $+1$) and $O(\varepsilon_{\text{mach}}/\delta)$ round-off (slope $-1$, same as forward); minimum error $\approx 10^{-11}$ at $\delta^{\star} \approx 10^{-5.5} = \varepsilon_{\text{mach}}^{1/3}$. \begin{remark}[Automatic Differentiation as the Production Alternative]
Finite differences and complex-step methods require $O(d)$ function evaluations per gradient (one per coordinate). In modern machine learning, \textbf{automatic differentiation} (AD) computes the exact gradient at the cost of a single forward+backward pass, exploiting the chain rule on the computation graph rather than perturbing inputs. Forward-mode AD (efficient when $d$ is small) and reverse-mode AD (efficient when $d$ is large, as in neural networks with millions of parameters) both compute exact gradients with no truncation or round-off error beyond standard floating-point arithmetic. The Python \texttt{autograd}, \texttt{JAX}, and PyTorch libraries implement reverse-mode AD. For our $d = 2$ benchmarks, the computational advantage is modest; for $d = 10^7$ (a typical large language model), reverse-mode AD is indispensable.
\end{remark}

\subsubsection{Nesterov Random Search}

\begin{algorithm}[H]
\caption{Nesterov (Gaussian Smoothing) Random Search}
\label{alg:nrs}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, smoothing radius $\delta$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State Sample $u_k \sim \mathcal{N}(0, I_d)$; normalise $u_k \gets u_k / \|u_k\|$
    \State $\hat{g}_k \gets \dfrac{f(x_k + \delta\, u_k) - f(x_k)}{\delta}\, u_k$
    \State $x_{k+1} \gets x_k - \alpha\, \hat{g}_k$
\EndFor
\end{algorithmic}
\end{algorithm}

The estimator $\hat{g}_k$ approximates the directional derivative along $u_k$, yielding an unbiased estimate of a smoothed gradient: $\mathbb{E}_{u_k}[\hat{g}_k] = \nabla f_{\delta}(x_k)$ where $f_{\delta}(x) = \mathbb{E}_{u \sim \mathcal{N}(0,I)}[f(x + \delta u/\|u\|)]$ is the Gaussian-smoothed objective. Only \emph{two} function evaluations are required per iteration, regardless of dimension $d$ --- a key advantage in high-dimensional settings where finite differences cost $O(d)$ evaluations.

\begin{remark}[Convergence Rate of Nesterov Random Search]
For convex $L$-smooth $f$ in dimension $d$, with step size $\alpha = O(1/(Ld))$ and $\delta = O(\sqrt{\varepsilon/(Ld)})$ chosen to balance smoothing bias and gradient variance, Nesterov random search achieves $\mathbb{E}[f(x_k)] - f^{\star} = O(Ld^2 / k)$ convergence --- a factor of $d^2$ worse than gradient descent's $O(L/k)$. For $d = 2$ (our case), this is only a factor of 4 penalty, which is modest. For $d = 1000$, the factor of $10^6$ makes gradient-free methods prohibitive for convex problems, though they may be the only option for non-differentiable or black-box objectives.

The $d^2$ penalty arises from two sources: (i) a factor of $d$ from the smoothing bias $\|\nabla f_\delta(x) - \nabla f(x)\| = O(\delta L d^{1/2})$ (the smoothed gradient is an average over a $d$-dimensional sphere, so the ``effective'' perturbation shrinks as $1/\sqrt{d}$ in any single direction), and (ii) a factor of $d$ from the gradient variance, since a single 1D projection retains only $1/d$ of the full gradient information per step. Together, the algorithm requires $O(d^2)$ more iterations than full-gradient methods to achieve the same accuracy. The two-point estimator $[f(x + \delta u) - f(x - \delta u)]/(2\delta)$ reduces the variance by a constant factor but does not change the asymptotic $d^2$ rate.
\end{remark}

\subsection{Part B: Derivative-Free Methods on Benchmark C (Rosenbrock)}

\subsubsection{Nelder--Mead Simplex Method}

The Nelder--Mead algorithm maintains a simplex of $n+1 = 3$ vertices in $\mathbb{R}^2$ (since $d = 2$) and iteratively transforms it through four operations:
\begin{enumerate}[nosep]
\item \textbf{Reflection} ($\alpha_r = 1$): move the worst vertex through the centroid of the remaining vertices.
\item \textbf{Expansion} ($\gamma = 2$): if the reflected point improves on the best, try extending further.
\item \textbf{Contraction} ($\rho = 0.5$): if reflection fails, contract the worst vertex towards the centroid.
\item \textbf{Shrink} ($\sigma = 0.5$): if contraction fails, contract all vertices towards the best.
\end{enumerate}
No gradient information is required; the simplex geometry adapts to the local landscape by expanding where progress is made and contracting where it is not.

Settings: initial simplex step $= 0.35$, $\alpha_r = 1.0$ (reflection), $\gamma = 2.0$ (expansion), $\rho = 0.5$ (contraction), $\sigma = 0.5$ (shrink), iterations $= 160$.

\begin{remark}[Convergence Properties of Nelder--Mead and Simplex Adaptation]
Nelder--Mead lacks a general convergence guarantee: McKinnon (1998) constructed a 2D example where the method converges to a non-optimal point. In practice, it often performs well for $d \leq 5$ but its per-iteration cost grows as $O(d)$ function evaluations in the worst case (shrink step), and the simplex can degenerate (become flat). For Rosenbrock in $d = 2$, the method adapts well to the banana-shaped valley for a specific geometric reason: the simplex naturally elongates along the valley floor (via repeated reflection and expansion in the valley direction) and contracts across it. This geometric adaptation is implicit in the simplex operations and does not require gradient information. The key vulnerability is simplex degeneration: if two vertices become nearly collinear with the best vertex, the simplex loses volume and can no longer explore the function landscape, stalling at a non-optimal point. The shrink operation (contracting all vertices towards the best) is the safest recovery but is expensive ($d$ extra evaluations). In our $d = 2$ run, no degeneration is observed, and the simplex successfully navigates the curved valley in 160 iterations.
\end{remark}

\begin{remark}[Taylor Expansion Justification of the FD Error Bounds]
The forward difference error $O(\delta)$ and central difference error $O(\delta^2)$ follow directly from the Taylor expansion:
\begin{align*}
\text{Forward: } \frac{f(x+\delta e_i)-f(x)}{\delta} &= \partial_i f + \frac{\delta}{2}\partial_{ii}f + O(\delta^2), \quad \text{truncation error } \leq \frac{\delta L}{2}, \\
\text{Central: } \frac{f(x+\delta e_i)-f(x-\delta e_i)}{2\delta} &= \partial_i f + \frac{\delta^2}{6}\partial_{iii}f + O(\delta^4), \quad \text{truncation error } \leq \frac{\delta^2 |f'''|}{6}.
\end{align*}
For Benchmark B with $L = 10$: at $\delta = 0.05$, forward truncation $\leq \frac{0.05 \times 10}{2} = 0.25$, matching the measured $0.25$. At $\delta = 0.8$, forward truncation $\leq 4.0$, matching the measured $4.03$. For central differences at $\delta = 0.05$: $|f'''| = |\cos(0.5)| \approx 0.878$, giving bound $\frac{(0.05)^2 \times 0.878}{6} \approx 3.7 \times 10^{-4}$, matching the measured $3.7 \times 10^{-4}$. The round-off error $O(\varepsilon_{\text{mach}}/\delta)$ arises because the numerator $f(x + \delta e_i) - f(x)$ has relative floating-point error $\varepsilon_{\text{mach}}$, and the optimal balancing $\delta^{\star} = \sqrt{\varepsilon_{\text{mach}}} \approx 1.5 \times 10^{-8}$ for forward (vs $\varepsilon_{\text{mach}}^{1/3} \approx 6 \times 10^{-6}$ for central) is confirmed in Figure~\ref{fig:q4_fd_error}.
\end{remark}

\subsection{Problem Formulation}

Minimise $f(x) = (x_1 - 1)^2 + 5(x_2 - 2)^2 + \sin(x_1)$ (Benchmark B) subject to $x_1 \geq 0.5$, starting from the infeasible point $x_0 = (0.2, 4.0)$.

In standard form: $\min_{x \in \mathbb{R}^2} f(x)$ s.t.\ $h(x) := -x_1 + 0.5 \leq 0$.

\subsection{KKT Optimality Conditions}

\begin{definition}[KKT Conditions]
For the problem $\min f(x)$ s.t.\ $h(x) \leq 0$, a point $x^{\star}$ is KKT-stationary if there exists a multiplier $\mu^{\star} \geq 0$ such that:
\begin{align}
\nabla f(x^{\star}) + \mu^{\star} \nabla h(x^{\star}) &= 0 \quad (\text{stationarity}), \label{eq:kkt_stat}\\
h(x^{\star}) &\leq 0 \quad (\text{primal feasibility}), \label{eq:kkt_pf}\\
\mu^{\star} &\geq 0 \quad (\text{dual feasibility}), \label{eq:kkt_df}\\
\mu^{\star} h(x^{\star}) &= 0 \quad (\text{complementary slackness}). \label{eq:kkt_cs}
\end{align}
Under a constraint qualification (e.g., LICQ), KKT conditions are \emph{necessary} for local optimality; for convex $f$ and convex $h$, they are also \emph{sufficient}.
\end{definition}

\begin{remark}[Constraint Qualification: LICQ]
The \textbf{Linear Independence Constraint Qualification} (LICQ) requires that the gradients of the active constraints at $x^{\star}$ are linearly independent. For our single constraint $h(x) = -x_1 + 0.5 \leq 0$, $\nabla h = [-1, 0]^T$, which is trivially non-zero, so LICQ holds whenever the constraint is active ($x_1^{\star} = 0.5$). Since the constraint is \emph{inactive} at our optimum ($x_1^{\star} = 0.582$), LICQ is automatically satisfied (vacuously: no active constraint gradients to check), and the standard KKT conditions apply with $\mu^{\star} = 0$.
\end{remark}

\begin{proposition}[Constrained Optimum Coincides with Unconstrained Optimum]
\label{prop:inactive_constraint}
The unconstrained minimum of Benchmark B is at $x^{\star} = (0.582, 2.0)$, satisfying $x_1^{\star} = 0.582 > 0.5$. Therefore, the constraint $x_1 \geq 0.5$ is \emph{inactive} at $x^{\star}$.
\end{proposition}

\begin{proof}
At $x^{\star}$, complementary slackness \eqref{eq:kkt_cs} gives $\mu^{\star}(-x_1^{\star} + 0.5) = \mu^{\star}(0.5 - 0.582) = -0.082\mu^{\star} = 0$, forcing $\mu^{\star} = 0$. The stationarity condition \eqref{eq:kkt_stat} then reduces to $\nabla f(x^{\star}) = 0$, the unconstrained first-order condition, which is satisfied by definition of $x^{\star}$.
\end{proof}

Proposition~\ref{prop:inactive_constraint} implies that both projected GD and penalty methods should converge to the same point $(0.582, 2.0)$ given sufficient iterations, since the constraint does not restrict the unconstrained optimum.

\begin{remark}[KKT Conditions and Lagrangian Duality]
The KKT conditions arise naturally from the Lagrangian $\mathcal{L}(x, \mu) = f(x) + \mu h(x)$. The dual function $q(\mu) = \min_{x} \mathcal{L}(x, \mu)$ is always concave (as the infimum of linear functions of $\mu$), and the dual problem $\max_{\mu \geq 0} q(\mu)$ provides a lower bound on the primal: $q(\mu) \leq f(x^{\star})$ for any $\mu \geq 0$. For convex $f$ and convex $h$ with a Slater point (strictly feasible $x_0$ with $h(x_0) < 0$), strong duality holds: $\max_\mu q(\mu) = f(x^{\star})$, and the KKT conditions are both necessary and sufficient. In our problem, $x = (1, 1)$ is strictly feasible ($x_1 = 1 > 0.5$), so strong duality holds. With $\mu^{\star} = 0$, the dual problem trivially gives $q(0) = \min_x f(x) = f(x^{\star})$, confirming strong duality with zero duality gap.
\end{remark}

\subsection{Algorithms}

\begin{algorithm}[H]
\caption{Projected Gradient Descent}
\label{alg:pgd}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, projection operator $\Pi_{\mathcal{X}}$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_{k+1} \gets x_k - \alpha\, \nabla f(x_k)$ \Comment{Unconstrained gradient step}
    \State $x_{k+1} \gets \Pi_{\mathcal{X}}(z_{k+1}) = \bigl(\max(0.5, z_{k+1,1}),\ z_{k+1,2}\bigr)$ \Comment{Project onto $\{x_1 \geq 0.5\}$}
\EndFor
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Exact Penalty Method}
\label{alg:penalty}
\begin{algorithmic}[1]
\Require Initial point $x_0$, step size $\alpha$, penalty weight $\lambda > 0$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $F_{\lambda}(x) \gets f(x) + \lambda \max(0,\, -x_1 + 0.5)$ \Comment{Penalised objective}
    \State Compute $\nabla F_{\lambda}(x_k)$: gradient of $f$ plus $-\lambda e_1 \cdot \mathbf{1}[x_{1,k} < 0.5]$
    \State $x_{k+1} \gets x_k - \alpha\, \nabla F_{\lambda}(x_k)$
\EndFor
\end{algorithmic}
\end{algorithm}

Settings: projected GD ($\alpha = 0.08$); penalty with $\lambda \in \{0.15, 1.8, 4.5\}$ ($\alpha = 0.05$ for $\lambda \leq 1.8$; $\alpha = 0.03$ for $\lambda = 4.5$); unconstrained GD ($\alpha = 0.07$). All run for 100 iterations.

\begin{remark}[Stability of L1 vs.\ Smooth Penalty Methods]
The penalty $F_{\lambda}(x) = f(x) + \lambda \max(0, -x_1 + 0.5)$ is an \textbf{L1} (non-smooth) penalty: the gradient jumps discontinuously by $\lambda e_1$ at the constraint boundary $x_1 = 0.5$. Within each half-space the Hessian of $F_{\lambda}$ is just $\nabla^2 f$ (the penalty term is linear, so $\nabla^2[\lambda\max(0,h)] = 0$ away from the boundary), giving Lipschitz constant $L_f$ in each region. The instability for large $\lambda$ arises not from a larger smoothness constant but from iterates \emph{oscillating across} the constraint boundary: a step of size $\alpha$ from the infeasible side applies gradient $\nabla f - \lambda e_1$, while a step from the feasible side applies $\nabla f$ --- the effective gradient change across the boundary has magnitude $\lambda$, creating gradient-step oscillations of amplitude $\sim \alpha\lambda$. Keeping $\alpha\lambda$ small (i.e., $\alpha \propto 1/\lambda$) suppresses these oscillations. This is distinct from the \textbf{smooth squared penalty} $\lambda h(x)^2/2$, which does have $L_{\lambda} = L_f + \lambda$ (from the extra $\lambda\nabla h \nabla h^T$ term in the Hessian), requiring $\alpha < 2/(L_f + \lambda)$ for GD stability. ALM uses the smooth augmented term $\frac{\lambda}{2}h(x)^2$, making this Lipschitz analysis relevant and $\lambda$ can stay moderate.
\end{remark}

\textbf{Projected GD} enforces feasibility as a hard constraint at every iteration by applying the projection operator after each gradient step. For simple constraints like half-spaces ($x_1 \geq 0.5$), the projection is a trivial $\max$ operation ($O(1)$). For complex constraint sets (e.g., semidefinite cones, $\ell_1$ balls), projection may itself be a non-trivial optimisation problem.

\textbf{Penalty methods} trade exact feasibility for simplicity: only a modified objective gradient is needed, and no projection oracle is required. The weight $\lambda$ controls the trade-off: small $\lambda$ under-penalises violations; large $\lambda$ enforces feasibility but may introduce ill-conditioning, requiring smaller step sizes to maintain stability (here, $\alpha = 0.03$ for $\lambda = 4.5$ vs.\ $\alpha = 0.05$ for $\lambda \leq 1.8$).

\begin{remark}[Augmented Lagrangian: Getting the Best of Both Worlds]
The pure penalty method requires $\lambda \to \infty$ for \emph{exact} feasibility, causing ill-conditioning. The \textbf{Augmented Lagrangian method} (ALM, or ADMM in the distributed case) adds a dual variable $\mu_k$ to maintain a feasibility ``memory'':
\[
\mathcal{L}_{\lambda}(x, \mu) = f(x) + \mu\, h(x) + \tfrac{\lambda}{2} h(x)^2.
\]
The ALM alternates between minimising $\mathcal{L}_{\lambda}(\cdot, \mu_k)$ with respect to $x$ (subproblem solved approximately) and updating the multiplier $\mu_{k+1} = \mu_k + \lambda\, h(x_k^{\star})$. The multiplier update drives $\mu_k \to \mu^{\star}$ (the true KKT multiplier), so that the augmented term $\mu^{\star} h(x) + \frac{\lambda}{2}h(x)^2$ penalises even small violations that pure GD on $F_{\lambda}(x)$ would ignore. Crucially, a \emph{fixed moderate} $\lambda$ (rather than $\lambda \to \infty$) suffices for convergence: the dual update compensates for incomplete feasibility enforcement at each step. For our problem with $\mu^{\star} = 0$ (inactive constraint), the dual variable stays at zero, and ALM reduces to the pure penalty method, recovering the same convergence as penalty GD.
\end{remark}

\subsection{Feasible Set and Linear Programme}

\textbf{Linear Programme (LP):} Minimise $f(x) = a^T x$ with $a = [1, 2]^T$ over $\mathcal{X}$.

\begin{proposition}[LP Solution]
Since both components of $a = [1, 2]^T$ are positive, the objective $f(x) = x_1 + 2x_2$ is minimised by setting each $x_i$ to its lower bound:
\begin{equation}
x^{\star} = (0.5,\, -5), \qquad f^{\star} = 0.5 + 2(-5) = -9.5.
\end{equation}
\end{proposition}

This exemplifies the \textbf{Fundamental Theorem of Linear Programming}: the optimum of a linear objective over a bounded polytope is always attained at a vertex. The vertices of $\mathcal{X}$ are its four corners: $(0.5, -5)$, $(5, -5)$, $(0.5, 10)$, $(5, 10)$.

\subsection{Frank--Wolfe Algorithm}

\begin{algorithm}[H]
\caption{Frank--Wolfe (Conditional Gradient)}
\label{alg:fw}
\begin{algorithmic}[1]
\Require Initial point $x_0 \in \mathcal{X}$, mixing parameter $\beta \in (0,1)$, iterations $N$
\For{$k = 0, 1, \ldots, N-1$}
    \State $z_k \gets \arg\min_{x \in \mathcal{X}}\, \nabla f(x_k)^T x$ \Comment{Linear minimisation oracle (LMO)}
    \State $x_{k+1} \gets \beta\, x_k + (1 - \beta)\, z_k$ \Comment{Convex combination (projection-free)}
\EndFor
\end{algorithmic}
\end{algorithm}

The \textbf{linear minimisation oracle} (LMO) solves a linear programme at each iteration. For box constraints, the LMO decomposes per coordinate:
\begin{equation}
(z_k)_i = \begin{cases} l_i & \text{if } (\nabla f(x_k))_i > 0 \\ u_i & \text{if } (\nabla f(x_k))_i \leq 0 \end{cases}
\end{equation}
requiring only $O(d)$ operations. Crucially, Frank--Wolfe \emph{never projects} onto $\mathcal{X}$; iterates are always convex combinations of feasible points and are therefore automatically feasible.

\begin{definition}[Frank--Wolfe Gap]
The \emph{Frank--Wolfe duality gap} at $x_k$ is:
\begin{equation}
g_{\text{FW}}(x_k) = \max_{y \in \mathcal{X}}\, \nabla f(x_k)^T(x_k - y) = \nabla f(x_k)^T(x_k - z_k).
\end{equation}
It satisfies $g_{\text{FW}}(x_k) \geq f(x_k) - f^{\star} \geq 0$, providing a computable upper bound on the sub-optimality gap. $g_{\text{FW}}(x^{\star}) = 0$ iff $x^{\star}$ is a constrained optimum.
\end{definition}

\begin{proof}[Proof of $g_{\text{FW}}(x_k) \geq f(x_k) - f^{\star}$]
By convexity of $f$: for any $y \in \mathcal{X}$, $f(y) \geq f(x_k) + \nabla f(x_k)^T(y - x_k)$. Setting $y = x^{\star}$ and rearranging: $f(x_k) - f^{\star} \leq \nabla f(x_k)^T(x_k - x^{\star}) \leq \max_{y \in \mathcal{X}} \nabla f(x_k)^T(x_k - y) = g_{\text{FW}}(x_k)$, where the second inequality follows because $x^{\star} \in \mathcal{X}$.
\end{proof}

\begin{theorem}[Frank--Wolfe Convergence Rate]
For a convex, $L$-smooth objective $f$ and the standard step-size $\gamma_k = 2/(k+2)$, Frank--Wolfe satisfies:
\[
f(x_k) - f^{\star} \leq \frac{2LC_{\mathcal{X}}^2}{k+2} = O(1/k),
\]
where $C_{\mathcal{X}} = \max_{x, y \in \mathcal{X}} \|x - y\|$ is the diameter of the feasible set.
\end{theorem}
\begin{proof}[Proof sketch]
By $L$-smoothness and the update $x_{k+1} = (1-\gamma_k) x_k + \gamma_k z_k$:
\begin{align*}
f(x_{k+1}) &\leq f(x_k) + \nabla f(x_k)^T(x_{k+1}-x_k) + \frac{L}{2}\|x_{k+1}-x_k\|^2 \\
&= f(x_k) - \gamma_k g_{\text{FW}}(x_k) + \frac{L\gamma_k^2}{2}C_\mathcal{X}^2.
\end{align*}
Using $g_{\text{FW}}(x_k) \geq f(x_k) - f^\star$ and setting $h_k = f(x_k)-f^\star$: $h_{k+1} \leq (1-\gamma_k) h_k + \frac{L\gamma_k^2}{2}C_\mathcal{X}^2$. With $\gamma_k = 2/(k+2)$, induction gives $h_k \leq 2LC_\mathcal{X}^2/(k+2)$.
\end{proof}

The fixed-$\beta$ variant used in the experiments is a special case; its convergence rate depends on $\beta$ and does not generally achieve the $O(1/k)$ optimal rate. For our box $\mathcal{X} = [0.5,5] \times [-5,10]$, $C_\mathcal{X} = \|(4.5, 15)\| = \sqrt{4.5^2+15^2} \approx 15.66$, giving $2LC_\mathcal{X}^2/(k+2) \approx 2 \cdot 2 \cdot 245/202 \approx 4.85$ at $k=200$ iterations for the interior quadratic ($L=2$). The observed $f \approx 0.0011$ (optimal schedule) is far better due to the strong convexity of $f(x) = (x_1-1)^2 + (x_2-5)^2$ ($\mu = 2$), for which FW can achieve a linear rate with line search. Frank--Wolfe's distinctive advantages are:
\begin{itemize}[nosep]
\item \textbf{Projection-free:} Projecting onto nuclear norm balls ($\|X\|_* \leq r$) requires a full SVD ($O(n^3)$), while the LMO only needs a rank-1 SVD ($O(n^2)$). In trace-norm regularisation (matrix completion), this provides a factor-$n$ speedup.
\item \textbf{Sparse iterates:} Each $z_k$ is a vertex of $\mathcal{X}$ (an extreme point); the iterate $x_k = \sum_{t} \gamma_t z_t$ is a convex combination of at most $k$ vertices. This yields \emph{sparse solutions} automatically, valued in compressed sensing and structured prediction.
\item \textbf{Natural for polytopes:} Many combinatorial constraint sets (flow polytopes, spanning tree polytopes, permutation matrices) have efficient LMOs but no closed-form projection.
\end{itemize}

\subsection{Interior vs.\ Boundary Behaviour}

\textbf{Interior optimum:} Frank--Wolfe exhibits characteristic zig-zag behaviour. The LP vertex $z_k$ alternates between extreme points of $\mathcal{X}$ since the gradient direction changes at each iterate, and the convex combination averages these jumps. With a fixed $\beta$ schedule, the effective step towards $z_k$ is constant at $1 - \beta$; using the standard decaying schedule $\gamma_k = 2/(k+2)$ would reduce oscillation and improve the convergence rate.

\begin{remark}[Away-Step and Pairwise Frank--Wolfe]
The zig-zag oscillation observed in Figure~\ref{fig:q6_evolution} is a fundamental limitation of vanilla FW: at each step, FW can only \emph{add} weight to a new vertex, never \emph{remove} weight from an existing one. The \textbf{away-step FW} (Guelat \& Marcotte, 1986) fixes this by augmenting the standard FW step with an ``away'' step that moves $x_k$ \emph{away from} the worst vertex in the current convex combination:
\[
x_{k+1} = x_k + \gamma\bigl[(1-\delta_{\min}) x_k - \delta_{\min} s_k\bigr] \quad \text{(away step away from } s_k\text{)},
\]
where $s_k = \arg\max_{s \in \mathcal{A}_k} \nabla f(x_k)^T s$ is the atom with the largest inner product with the gradient in the active set $\mathcal{A}_k$. Away-step FW achieves \textbf{linear convergence} $O(\rho^k)$ for strongly convex objectives (vs.\ $O(1/k)$ for vanilla FW), at the cost of maintaining the active set $\mathcal{A}_k$. The \textbf{pairwise FW} variant combines an FW step and an away step simultaneously, moving mass from the worst atom to the best atom, achieving the same improved rate with less memory.
\end{remark}

\textbf{Boundary optimum:} Frank--Wolfe handles boundary optima naturally. The LMO consistently selects the vertex nearest to the unconstrained minimum, driving the iterate towards the constraint boundary. The residual gap ($f = 0.364$ vs.\ $f^{\star} = 0.25$ after 140 iterations) reflects the $O(1/k)$ convergence rate: more iterations or a decaying step schedule would close this gap.

The \textbf{Frank--Wolfe gap} $g_{\text{FW}}(x_k) = \nabla f(x_k)^T(x_k - z_k)$ provides a certificate of near-optimality at each iteration: when $g_{\text{FW}} \leq \varepsilon$, we know $f(x_k) - f^{\star} \leq \varepsilon$ without knowing $f^{\star}$ explicitly. This is a significant practical advantage over methods that require knowing the optimal value (like Polyak).

\section{Conclusion}
\label{sec:conclusion}
% ============================================================

This report has systematically investigated six families of optimisation algorithms across three benchmark problems of increasing difficulty. The key conclusions are:

\begin{enumerate}[nosep]
\item \textbf{Adaptive step-size methods (Q1):} The Polyak step size achieves optimal convergence when $f^{\star}$ is known exactly (Benchmark C: $f \to 0.001$ at 200 iterations) but diverges catastrophically when $f^{\star}$ is misspecified (Benchmarks A, B). Adagrad and Heavy Ball provide robust alternatives without requiring knowledge of $f^{\star}$.

\item \textbf{Momentum and stochastic methods (Q2):} Nesterov acceleration achieves $O(1/k^2)$ convergence for smooth convex problems and linear convergence $O\bigl((1-2/\sqrt{\kappa})^k\bigr)$ for strongly convex problems, matching the minimax lower bound. On Rosenbrock ($\kappa \approx 2508$), Nesterov ($f = 0.479$ at 150 iterations) outperforms Adam ($f = 1.578$): Adam's per-coordinate normalisation $\hat{m}_t/(\sqrt{\hat{v}_t}+\varepsilon)$ suppresses progress along the valley floor where large sustained gradients would otherwise drive fast convergence, while Nesterov's momentum accumulates consistently in the valley direction. Mini-batch SGD exhibits a batch-size trade-off: smaller batches provide more frequent updates but higher gradient variance ($\sigma^2/b$), while larger batches give smoother but per-epoch slower convergence; the linear scaling rule allows $\alpha \propto b$ up to the full-batch limit.

\item \textbf{Newton's method (Q3):} For quadratic objectives (Benchmark A), Newton converges in exactly one step. For near-quadratic objectives (Benchmark B), convergence is achieved in $\sim$5 steps. On the ill-conditioned Rosenbrock, damping ($\alpha = 0.22$) is necessary for stability, but even damped Newton outperforms GD by a factor of $\sim$5 in objective value for the same iteration count. The quadratic convergence rate $\|x_{k+1} - x^{\star}\| = O(\|x_k - x^{\star}\|^2)$ near the optimum is confirmed empirically.

\item \textbf{Derivative-free methods (Q4):} Finite differences with $\delta = 0.05$ closely approximate the exact gradient at $(0.5, 3.0)$ (forward error $\approx 0.2528$, or $2.5\%$ of $\|\nabla f\| \approx 10.0$), confirming that $\delta = 0.05$ is in the truncation-dominated regime. The larger $\delta = 0.8$ incurs substantial truncation error (forward error $\approx 4.03$, roughly $40\%$ of the gradient norm), severely degrading the descent direction. Nesterov random search, while dimension-efficient (2 evaluations per step), converges slowly due to its stochastic directional gradient and the $d^2$ dimension penalty in the convergence rate. The $O(\delta)$ vs.\ $O(\delta^2)$ trade-off between forward and central differences is confirmed in Figure~\ref{fig:q4_fd_error}, with optimal step sizes $\delta^{\star} \approx 10^{-8}$ (forward) and $\approx 10^{-5.5}$ (central).

\item \textbf{Constrained optimisation (Q5):} Since the unconstrained minimum $(0.582, 2.0)$ is interior to the feasible region $\{x_1 \geq 0.5\}$, KKT conditions confirm that $\mu^{\star} = 0$ and the constrained and unconstrained optima coincide. Projected GD enforces feasibility exactly from iteration 1 via the $\max$ projection. Penalty methods achieve approximate feasibility, with convergence quality improving with $\lambda$.

\item \textbf{Frank--Wolfe and linear programmes (Q6):} The LP $\min x_1 + 2x_2$ over a box has its optimum at the vertex $(0.5, -5)$, confirming the Fundamental Theorem of LP. Frank--Wolfe converges at $O(1/k)$ to both interior and boundary constrained optima without requiring a projection oracle, making it attractive for complex constraint sets where the LMO is tractable.
\end{enumerate}

\textbf{Overall recommendation:} For well-conditioned problems ($\kappa$ small) with known $f^{\star}$, Polyak is optimal. For general smooth convex problems without known $f^{\star}$, Nesterov acceleration provides the theoretically optimal $O(1/k^2)$ rate with no additional hyperparameter knowledge required beyond the Lipschitz constant $L$. When second derivatives are available and the dimension is small ($d \lesssim 100$), Newton provides quadratic convergence; for $d \gtrsim 1000$, quasi-Newton (BFGS) is preferable. Mini-batch SGD with variance reduction (SVRG or SAGA) achieves linear convergence without decaying step sizes for strongly convex finite-sum problems, making it the method of choice for large-scale machine learning. Derivative-free methods (Nelder--Mead for $d \leq 5$; Nesterov random search for $d \gg 1$) should be reserved for black-box or simulation-based objectives where gradient computation is infeasible. For linearly-constrained problems with tractable LP oracles, Frank--Wolfe with the decaying schedule $\gamma_k = 2/(k+2)$ provides projection-free $O(1/k)$ convergence; the away-step variant upgrades this to linear convergence for strongly convex problems.

\paragraph{Connection to Course Theory.} The experiments directly validate the key theoretical results from the course:
\begin{itemize}[nosep]
\item \textbf{GD rate $O(\rho_f^k)$}: Confirmed on Benchmark A ($\kappa \approx 1.12$). With $\alpha = 0.08$, $\mu \approx 0.925$, $L \approx 1.033$: distance rate $\rho_d = \max(|1 - 0.08 \times 0.925|, |1 - 0.08 \times 1.033|) = \max(0.926, 0.917) = 0.926$ and function value rate $\rho_f \approx \rho_d^2 \approx 0.858$, both matching Theorem~\ref{thm:gd_linear}: $\rho_f = 1 - 2\alpha\mu(1 - \alpha L/2) \approx 0.858$ (Figure~\ref{fig:theory_vs_practice}). The near-unit conditioning gives rapid convergence in $\sim$30 iterations.
\item \textbf{Nesterov $O(1/k^2)$ vs.\ GD $O(1/k)$}: The quadratic decay visible in Figure~\ref{fig:theory_vs_practice} (right) and the $7\times$ improvement on Rosenbrock (Table~\ref{tab:q2_final}) confirm the theoretical rate advantage.
\item \textbf{Newton one-step for quadratics} (Theorem~\ref{thm:newton_quadratic}): Single-iteration convergence on Benchmark A confirmed (Figure~\ref{fig:q3_fval_A}); the Newton step equals the OLS closed-form solution.
\item \textbf{Polyak step-size dependence on $f^{\star}$}: Catastrophic divergence for misspecified $f^{\star} = 0$ (Benchmarks A/B) vs.\ best-in-class convergence when $f^{\star} = 0$ is exact (Benchmark C, $f \to 0.0014$).
\item \textbf{Heavy Ball $\sqrt{\kappa}$ speedup}: On Benchmark C ($\kappa \approx 2508$), optimal Heavy Ball parameters $(\alpha^{\star}, \beta^{\star})$ reduce distance to optimum by $e^{-1}$ in $\approx 25$ iterations vs.\ $\approx 1254$ for GD (a $50\times$ speedup, matching $\sqrt{\kappa} \approx 50$). The experimental result (Heavy Ball $f \approx 0.29$ vs.\ GD $f \approx 3.17$ at 200 iterations) confirms the acceleration, with the gap reduced by the fact that experimental parameters are not exactly optimal.
\item \textbf{FD truncation vs.\ round-off trade-off}: Optimal $\delta^{\star} = \sqrt{\varepsilon_{\text{mach}}} \approx 10^{-8}$ confirmed in Figure~\ref{fig:q4_fd_error}; $\delta = 0.05$ is in the truncation-dominated regime with manageable error.
\item \textbf{KKT inactive constraint}: Proposition~\ref{prop:inactive_constraint} proves $\mu^{\star} = 0$ from complementary slackness; both projected GD and penalty method converge to the same unconstrained minimum $(0.582, 2.0)$.
\item \textbf{Frank--Wolfe gap as certificate}: $g_{\text{FW}}(x_k) \geq f(x_k) - f^{\star}$ confirmed numerically (boundary case: $g_{\text{FW}} = 0.139 > \text{sub-opt} = 0.114$); the gap provides a practical stopping criterion without knowing $f^{\star}$.
\item \textbf{Minimax lower bound}: No first-order method can achieve a better-than-$\left(\frac{\sqrt{\kappa}-1}{\sqrt{\kappa}+1}\right)^{2k}$ worst-case convergence rate for $\mu$-strongly convex, $L$-smooth objectives (Nesterov, 2003). For Benchmark C ($\kappa \approx 2508$), this lower bound is $\approx \left(1 - \frac{2}{\sqrt{2508}}\right)^{2k} \approx (0.96)^{2k}$, matching the empirical Nesterov rate. Heavy Ball and Nesterov are both near-optimal by this measure; GD at $O(e^{-k/\kappa})$ is a factor of $\sqrt{\kappa} \approx 50$ suboptimal in iteration count. This establishes that Nesterov/Heavy Ball (when appropriately tuned) are not merely heuristically better than GD but are provably best-possible among first-order methods.
\end{itemize}

def gradient_descent(grad_fn, loss_fn, x0, alpha, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for _ in range(n_iters):
        x = x - alpha * grad_fn(x)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)

    x = x0.copy().astype(float)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        alpha_k = (loss_fn(x) - f_star) / (np.dot(g, g) + eps)
        alpha_hist.append(alpha_k)
        x = x - alpha_k * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adagrad}]
def adagrad(grad_fn, loss_fn, x0, alpha0, eps, n_iters):
    x = x0.copy().astype(float)
    G = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        G += g**2
        eff_alpha = alpha0 / (np.sqrt(G) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

\begin{lstlisting}[caption={RMSprop}]
def rmsprop(grad_fn, loss_fn, x0, alpha0, beta, eps, n_iters):
    x = x0.copy().astype(float)
    v = np.zeros_like(x)
    x_hist, f_hist, alpha_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        v = beta * v + (1 - beta) * g**2
        eff_alpha = alpha0 / (np.sqrt(v) + eps)
        alpha_hist.append(np.mean(eff_alpha))
        x = x - eff_alpha * g
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(alpha_hist)
\end{lstlisting}

def heavy_ball(grad_fn, loss_fn, x0, alpha, beta, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]

        z = beta * z + alpha * g
        x = x - z

        f_hist.append(loss_fn(x))

\begin{lstlisting}[caption={Nesterov Accelerated Gradient}]
def nesterov_momentum(grad_fn, loss_fn, x0, alpha, beta_max, n_iters):
    x = x0.copy().astype(float)
    z = np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for k in range(1, n_iters + 1):
        beta_k = min((k - 1) / (k + 2), beta_max)
        lookahead = x + beta_k * z
        g = grad_fn(lookahead)
        z = beta_k * z - alpha * g
        x = x + z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Adam Optimiser}]
def adam_optimiser(grad_fn, loss_fn, x0, alpha, beta1, beta2, eps, n_iters):
    x = x0.copy().astype(float)
    m_vec, v_vec = np.zeros_like(x), np.zeros_like(x)
    x_hist, f_hist = [x.copy()], [loss_fn(x)]
    for t in range(1, n_iters + 1):
        g = grad_fn(x)
        m_vec = beta1 * m_vec + (1 - beta1) * g
        v_vec = beta2 * v_vec + (1 - beta2) * g**2
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        x = x - alpha * m_hat / (np.sqrt(v_hat) + eps)
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist)
\end{lstlisting}

\begin{lstlisting}[caption={Newton's Method with Damping}]
def newtons_method(grad_fn, hess_fn, loss_fn, x0, alpha, n_iters, damping=1e-8):
    x = x0.copy().astype(float)
    x_hist, f_hist, update_norms = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        H = hess_fn(x) if callable(hess_fn) else hess_fn
        H_reg = H + damping * np.eye(len(x))
        try:
            p = np.linalg.solve(H_reg, g)
        except np.linalg.LinAlgError:
            p = g
        update_norms.append(np.linalg.norm(alpha * p))
        x = x - alpha * p
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(update_norms)
\end{lstlisting}

\begin{lstlisting}[caption={Nelder--Mead Simplex}]
def nelder_mead(loss_fn, x0, step, n_iters, alpha_r=1.0, gamma=2.0, rho=0.5, sigma=0.5):
    n = len(x0)
    simplex = np.zeros((n + 1, n))
    simplex[0] = x0.copy()
    for i in range(n):
        simplex[i + 1] = x0.copy()
        simplex[i + 1][i] += step
    f_vals = np.array([loss_fn(v) for v in simplex])
    centroid_hist = [np.mean(simplex, axis=0).copy()]
    f_best_hist = [np.min(f_vals)]
    for _ in range(n_iters):
        order = np.argsort(f_vals)
        simplex = simplex[order]; f_vals = f_vals[order]
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + alpha_r * (centroid - simplex[-1])
        f_r = loss_fn(x_r)
        if f_vals[0] <= f_r < f_vals[-2]:
            simplex[-1] = x_r; f_vals[-1] = f_r
        elif f_r < f_vals[0]:
            x_e = centroid + gamma * (centroid - simplex[-1])
            f_e = loss_fn(x_e)
            if f_e < f_r: simplex[-1] = x_e; f_vals[-1] = f_e
            else: simplex[-1] = x_r; f_vals[-1] = f_r
        else:
            if f_r < f_vals[-1]:
                x_c = centroid + rho * (x_r - centroid)
                f_c = loss_fn(x_c)
                if f_c <= f_r: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
            else:
                x_c = centroid + rho * (simplex[-1] - centroid)
                f_c = loss_fn(x_c)
                if f_c < f_vals[-1]: simplex[-1] = x_c; f_vals[-1] = f_c
                else:
                    for i in range(1, n+1):
                        simplex[i] = simplex[0]+sigma*(simplex[i]-simplex[0])
                        f_vals[i] = loss_fn(simplex[i])
        centroid_hist.append(np.mean(simplex, axis=0).copy())
        f_best_hist.append(np.min(f_vals))
    return np.array(centroid_hist), np.array(f_best_hist), simplex
\end{lstlisting}

\begin{lstlisting}[caption={Frank--Wolfe Algorithm}]
def frank_wolfe(grad_fn, loss_fn, x0, bounds, beta, n_iters):
    x = x0.copy().astype(float)
    x_hist, f_hist, z_hist = [x.copy()], [loss_fn(x)], []
    for _ in range(n_iters):
        g = grad_fn(x)
        z = np.array([bounds[i][0] if g[i] > 0 else bounds[i][1]
                       for i in range(len(g))])
        z_hist.append(z.copy())
        x = beta * x + (1 - beta) * z
        x_hist.append(x.copy())
        f_hist.append(loss_fn(x))
    return np.array(x_hist), np.array(f_hist), np.array(z_hist)
\end{lstlisting}
